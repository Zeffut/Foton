1//! Serverbound packet reporting a recipe book screen's settings.
2
3use foton_macros::{ReadFrom, ServerPacket};
4
5use super::RecipeBookType;
6
7/// Sent when the player opens or closes a recipe book, or toggles its
8/// craftable-only filter.
9///
10/// Vanilla parity: `ServerboundRecipeBookChangeSettingsPacket`: the book type
11/// as an enum ordinal, then the two booleans.
12#[derive(ReadFrom, ServerPacket, Clone, Copy, Debug, PartialEq, Eq)]
13pub struct SRecipeBookChangeSettings {
14    pub book_type: RecipeBookType,
15    pub is_open: bool,
16    pub is_filtering: bool,
17}
18
19#[cfg(test)]
20mod tests {
21    use std::io::Cursor;
22
23    use foton_utils::serial::ReadFrom as _;
24
25    use super::*;
26
27    #[test]
28    fn reads_the_book_before_its_two_flags() {
29        let buffer = [2u8, 1, 0];
30        let packet = SRecipeBookChangeSettings::read(&mut Cursor::new(buffer.as_slice()))
31            .expect("packet should parse");
32
33        assert_eq!(
34            packet,
35            SRecipeBookChangeSettings {
36                book_type: RecipeBookType::BlastFurnace,
37                is_open: true,
38                is_filtering: false,
39            }
40        );
41    }
42
43    /// Vanilla's `readEnum` throws on an ordinal past the end, which drops the
44    /// connection; a fifth book must not be read as some other one.
45    #[test]
46    fn an_unknown_book_is_rejected() {
47        let buffer = [4u8, 1, 1];
48        assert!(SRecipeBookChangeSettings::read(&mut Cursor::new(buffer.as_slice())).is_err());
49    }
50}
