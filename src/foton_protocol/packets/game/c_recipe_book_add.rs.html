1//! Clientbound packet adding recipes to the recipe book.
2
3use std::io::{Result, Write};
4
5use foton_macros::{ClientPacket, WriteTo};
6use foton_registry::packets::play::C_RECIPE_BOOK_ADD;
7use foton_registry::recipe::RecipeDisplayEntry;
8use foton_utils::serial::WriteTo;

10/// One recipe display handed to the recipe book.
11///
12/// Vanilla parity: `ClientboundRecipeBookAddPacket.Entry`, whose two booleans
13/// travel as one flag byte.
14#[derive(Debug, Clone)]
15pub struct RecipeBookAddEntry {
16    pub contents: RecipeDisplayEntry,
17    /// Whether the client pops an unlock toast for it.
18    pub notification: bool,
19    /// Whether the book marks it new until the player looks at it.
20    pub highlight: bool,
21}
22
23impl RecipeBookAddEntry {
24    const FLAG_NOTIFICATION: u8 = 1;
25    const FLAG_HIGHLIGHT: u8 = 2;
26}
27
28impl WriteTo for RecipeBookAddEntry {
29    fn write(&self, writer: &mut impl Write) -> Result<()> {
30        self.contents.write(writer)?;
31        let mut flags = 0;
32        if self.notification {
33            flags |= Self::FLAG_NOTIFICATION;
34        }
35        if self.highlight {
36            flags |= Self::FLAG_HIGHLIGHT;
37        }
38        flags.write(writer)
39    }
40}

42/// Adds recipe displays to the client's recipe book.
43///
44/// Vanilla parity: `ClientboundRecipeBookAddPacket`. With `replace` the client
45/// first forgets every recipe it knew, which is how the whole book is sent on
46/// joining; without it the entries are unlocks, and the ones flagged for it
47/// pop a toast.
48#[derive(ClientPacket, WriteTo, Clone, Debug)]
49#[packet_id(Play = C_RECIPE_BOOK_ADD)]
50pub struct CRecipeBookAdd {
51    pub entries: Vec<RecipeBookAddEntry>,
52    pub replace: bool,
53}

55#[cfg(test)]
56mod tests {
57    use std::fmt::Write as _;
58
59    use foton_registry::recipe::RecipeDisplayEntry;
60    use foton_registry::{REGISTRY, init_vanilla_registry};
61    use foton_utils::Identifier;
62
63    use super::*;

65    /// `ClientboundRecipeBookAddPacket.STREAM_CODEC` run by vanilla 26.2 over
66    /// `minecraft:cake` (display 0, ungrouped, toast and highlight) and
67    /// `minecraft:iron_ingot_from_smelting_raw_iron` (display 1, group 0,
68    /// highlight only), each built from its datapack JSON with vanilla's own
69    /// recipe classes.
70    const VANILLA: &str = concat!(
71        "0200010303090a01090496080590080100000a01090496080590080100000a01090496080590080100000a01",
72        "04d908060e6d696e6563726166743a656767730a0104d9080a0104d4070a0104d4070a0104d40705da080100",
73        "0004e8020003010902960802960802960802d908000e6d696e6563726166743a6567677302d90802d40702d4",
74        "0702d4070301020a0104a3070105a40701000004ea02c8013f3333330106010102a3070200",
75    );
76
77    fn entry(key: &'static str, id: i32, group: Option<i32>) -> RecipeDisplayEntry {
78        let recipe = REGISTRY
79            .recipes
80            .book_recipe(&Identifier::vanilla_static(key))
81            .unwrap_or_else(|| panic!("{key} is a vanilla recipe"));
82        let [display] = <[RecipeDisplayEntry; 1]>::try_from(recipe.displays)
83            .unwrap_or_else(|_| panic!("{key} has exactly one display"));
84        RecipeDisplayEntry {
85            id,
86            group,
87            ..display
88        }
89    }
90
91    /// The whole chain in one: the recipe's own data, vanilla's rules for
92    /// turning it into a display (the milk bucket's remainder, the tag kept as
93    /// a tag, the furnace's fuel and category), and the wire shape of every
94    /// field down to the optional group being a shifted `VarInt`.
95    #[test]
96    fn encodes_vanilla_recipes_byte_for_byte_like_vanilla() {
97        init_vanilla_registry();
98        let packet = CRecipeBookAdd {
99            entries: vec![
100                RecipeBookAddEntry {
101                    contents: entry("cake", 0, None),
102                    notification: true,
103                    highlight: true,
104                },
105                RecipeBookAddEntry {
106                    contents: entry("iron_ingot_from_smelting_raw_iron", 1, Some(0)),
107                    notification: false,
108                    highlight: true,
109                },
110            ],
111            replace: false,
112        };
113
114        let mut bytes = Vec::new();
115        packet.write(&mut bytes).expect("packet encodes");
116
117        let hex = bytes.iter().fold(String::new(), |mut hex, byte| {
118            let _ = write!(hex, "{byte:02x}");
119            hex
120        });
121        assert_eq!(hex, VANILLA);
122    }
