9/// A command block minecart's editor being saved.
10///
11/// Vanilla parity: `ServerboundSetCommandMinecartPacket`. A minecart has no
12/// mode and no conditional flag -- it runs when an activator rail says so -- so
13/// the packet is only the entity, the command and the output checkbox.
14#[derive(ServerPacket, Clone, Debug)]
15pub struct SSetCommandMinecart {
16    /// Runtime id of the minecart being edited.
17    pub entity: i32,
18    /// The command to store.
19    pub command: String,
20    /// Whether the minecart keeps its last output.
21    pub track_output: bool,
22}
23
24impl ReadFrom for SSetCommandMinecart {
25    fn read(data: &mut Cursor<&[u8]>) -> Result<Self> {
26        let entity = VarInt::read(data)?.0;
27        let command = String::read_prefixed_bound::<VarInt>(data, MAX_COMMAND_LENGTH)?;
28        let track_output = bool::read(data)?;
29
30        Ok(Self {
31            entity,
32            command,
33            track_output,
34        })
35    }
36}

38#[cfg(test)]
39mod tests {
40    use super::*;
41
42    /// The entity id is a var int and the command a var-int-prefixed string;
43    /// reading the id as a plain `i32` would desynchronise everything after it.
44    #[test]
45    fn the_entity_id_is_a_var_int_before_the_command() {
46        let bytes = [0x80, 0x02, 2, b'h', b'i', 1];
47        let packet =
48            SSetCommandMinecart::read(&mut Cursor::new(bytes.as_slice())).expect("packet parses");
49
50        assert_eq!(packet.entity, 256);
51        assert_eq!(packet.command, "hi");
52        assert!(packet.track_output);
53    }
