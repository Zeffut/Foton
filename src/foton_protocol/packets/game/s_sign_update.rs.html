1use std::io::Cursor;
2
3use foton_macros::ServerPacket;
4use foton_utils::BlockPos;
5use foton_utils::serial::ReadFrom;
6
7/// Maximum characters per sign line.
8pub const MAX_SIGN_LINE_LENGTH: usize = 384;
9
10/// Serverbound packet sent when a player finishes editing a sign.
11#[derive(ServerPacket, Clone, Debug)]
12pub struct SSignUpdate {
13    /// The position of the sign block.
14    pub pos: BlockPos,
15    /// Whether updating the front text (true) or back text (false).
16    pub is_front_text: bool,
17    /// The four lines of text. Each line is max 384 characters.
18    pub lines: [String; 4],
19}
20
21impl ReadFrom for SSignUpdate {
22    fn read(data: &mut Cursor<&[u8]>) -> std::io::Result<Self> {
23        use foton_utils::serial::prefixed_read::read_utf;
24
25        let pos = BlockPos::read(data)?;
26        let is_front_text = bool::read(data)?;
27        // Vanilla parity: `input.readUtf(384)`. `read_prefixed_bound` counts
28        // bytes and decodes UTF-8 strictly, where vanilla counts UTF-16 units
29        // and replaces bad sequences. A line of 384 section signs is 768 bytes
30        // and perfectly legal to vanilla, but failed to decode here -- and a
31        // decode failure drops the player. `s_rename_item` already reads its
32        // text this way.
33        let lines = [
34            read_utf(data, MAX_SIGN_LINE_LENGTH)?,
35            read_utf(data, MAX_SIGN_LINE_LENGTH)?,
36            read_utf(data, MAX_SIGN_LINE_LENGTH)?,
37            read_utf(data, MAX_SIGN_LINE_LENGTH)?,
38        ];
39
40        Ok(Self {
41            pos,
42            is_front_text,
43            lines,
44        })
45    }
46}

48#[cfg(test)]
49mod tests {
50    use std::io::Cursor;
51
52    use foton_utils::codec::VarInt;
53    use foton_utils::serial::{ReadFrom, WriteTo};
54
55    use super::{MAX_SIGN_LINE_LENGTH, SSignUpdate};
56
57    /// Builds the wire form of a sign update carrying `lines`.
58    fn encode(lines: [&str; 4]) -> Vec<u8> {
59        let mut bytes = Vec::new();
60        // BlockPos, then the front/back flag.
61        0i64.write(&mut bytes).expect("position should encode");
62        true.write(&mut bytes).expect("flag should encode");
63        for line in lines {
64            VarInt(i32::try_from(line.len()).expect("test line fits"))
65                .write(&mut bytes)
66                .expect("length should encode");
67            bytes.extend_from_slice(line.as_bytes());
68        }
69        bytes
70    }
71
72    /// A full line of multi-byte characters is legal and must decode.
73    ///
74    /// Vanilla reads these with `readUtf(384)`, which counts UTF-16 units.
75    /// Foton bounded bytes instead, so 384 section signs -- 768 bytes, and
76    /// exactly what a player writing a decorated sign produces -- were refused,
77    /// and a decode failure drops the player.
78    #[test]
79    fn a_line_of_multi_byte_characters_still_fits() {
80        let line = "\u{00a7}".repeat(MAX_SIGN_LINE_LENGTH);
81        assert_eq!(
82            line.len(),
83            MAX_SIGN_LINE_LENGTH * 2,
84            "the test line is two bytes per character"
85        );
86
87        let bytes = encode([&line, "", "", ""]);
88        let packet = SSignUpdate::read(&mut Cursor::new(bytes.as_slice()))
89            .expect("a 384-character line is within vanilla's limit");
90        assert_eq!(packet.lines[0], line);
91    }
92
93    /// The limit is still a limit.
94    #[test]
95    fn a_line_past_the_limit_is_still_refused() {
96        let line = "a".repeat(MAX_SIGN_LINE_LENGTH * 3 + 1);
97        let bytes = encode([&line, "", "", ""]);
98        assert!(
99            SSignUpdate::read(&mut Cursor::new(bytes.as_slice())).is_err(),
100            "past three bytes per UTF-16 unit there is nothing legitimate left"
101        );
102    }
