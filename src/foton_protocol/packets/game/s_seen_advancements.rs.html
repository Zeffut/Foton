1//! Serverbound seen advancements packet -- which tab the client is looking at.
2
3use std::io::{Cursor, Result};
4
5use foton_macros::{ReadFrom, ServerPacket};
6use foton_utils::Identifier;
7use foton_utils::serial::ReadFrom;
8
9/// What the client just did with the advancement screen.
10///
11/// Vanilla parity: `ServerboundSeenAdvancementsPacket.Action`. The wire form is
12/// the ordinal, so the declaration order is protocol-observable.
13#[derive(ReadFrom, Clone, Copy, Debug, PartialEq, Eq)]
14#[read(as = VarInt)]
15pub enum SeenAdvancementsAction {
16    /// The client switched to a tab, naming its root advancement.
17    OpenedTab = 0,
18    /// The client closed the advancement screen.
19    ClosedScreen = 1,
20}
21
22/// Tells the server which advancement tab the client has open, so the server
23/// knows where to send a selection back to.
24///
25/// Vanilla parity: `ServerboundSeenAdvancementsPacket`. The tab is *not* a
26/// nullable field: vanilla writes the identifier bare and only when the action
27/// is `OpenedTab`, so there is no boolean between the action and the
28/// identifier.
29#[derive(ServerPacket, Clone, Debug, PartialEq, Eq)]
30pub struct SSeenAdvancements {
31    /// What the client did.
32    pub action: SeenAdvancementsAction,
33    /// The root advancement of the tab that was opened. Present exactly when
34    /// `action` is `OpenedTab`.
35    pub tab: Option<Identifier>,
36}
37
38impl ReadFrom for SSeenAdvancements {
39    fn read(data: &mut Cursor<&[u8]>) -> Result<Self> {
40        let action = SeenAdvancementsAction::read(data)?;
41        // Vanilla reads the identifier straight after the action, with no
42        // nullable prefix. Decoding it as an `Option<Identifier>` would take
43        // the identifier's own length byte for the flag.
44        let tab = match action {
45            SeenAdvancementsAction::OpenedTab => Some(Identifier::read(data)?),
46            SeenAdvancementsAction::ClosedScreen => None,
47        };
48        Ok(Self { action, tab })
49    }
50}

52#[cfg(test)]
53mod tests {
54    use super::*;
55
56    #[test]
57    fn a_closed_screen_is_one_byte_and_carries_no_tab() {
58        let buffer = [1u8];
59        let mut data = Cursor::new(buffer.as_slice());
60        let packet = SSeenAdvancements::read(&mut data).expect("packet should parse");
61
62        assert_eq!(packet.action, SeenAdvancementsAction::ClosedScreen);
63        assert_eq!(packet.tab, None);
64        assert_eq!(data.position(), 1);
65    }
66
67    #[test]
68    fn an_opened_tab_reads_its_identifier_straight_after_the_action() {
69        let buffer = b"\x00\x14minecraft:story/root";
70        let mut data = Cursor::new(buffer.as_slice());
71        let packet = SSeenAdvancements::read(&mut data).expect("packet should parse");
72
73        assert_eq!(packet.action, SeenAdvancementsAction::OpenedTab);
74        assert_eq!(packet.tab, Some(Identifier::vanilla_static("story/root")));
75        // Every byte and no more: `0x14` is the identifier's length, not a flag.
76        assert_eq!(data.position(), buffer.len() as u64);
77    }
78
79    /// The trap. This buffer decodes cleanly both ways and the two readings
80    /// disagree, so turning `tab` into an `Option<Identifier>` -- which reads
81    /// like the obvious tidy-up -- fails here instead of silently handing the
82    /// server a tab the client never opened.
83    #[test]
84    fn a_nullable_prefix_would_decode_a_different_identifier() {
85        // 45 characters, which is also the byte value of the `-` in front of
86        // it: an `Option` reader eats the length byte as its `true` flag, then
87        // finds a perfectly good 45-byte length prefix waiting behind it.
88        const NULLABLE_READING: &str = "minecraft:a_tab_the_client_never_asked_to_see";
89        const ACTUAL_READING: &str = "-minecraft:a_tab_the_client_never_asked_to_see";
90
91        assert_eq!(NULLABLE_READING.len(), 45);
92        assert_eq!(ACTUAL_READING.len(), 46);
93        assert_eq!(ACTUAL_READING.as_bytes()[0], 45);
94
95        let mut buffer = vec![0, 46];
96        buffer.extend_from_slice(ACTUAL_READING.as_bytes());
97        let mut data = Cursor::new(buffer.as_slice());
98        let packet = SSeenAdvancements::read(&mut data).expect("packet should parse");
99
100        assert_eq!(
101            packet.tab,
102            Some(Identifier::new(
103                "-minecraft",
104                "a_tab_the_client_never_asked_to_see"
105            ))
106        );
107        assert_ne!(
108            packet.tab.map(|tab| tab.to_string()),
109            Some(NULLABLE_READING.to_string())
110        );
111    }
112
113    #[test]
114    fn an_unknown_action_is_rejected() {
115        let buffer = [2u8];
116        let mut data = Cursor::new(buffer.as_slice());
117
118        assert!(SSeenAdvancements::read(&mut data).is_err());
119    }
