1//! Clientbound packet carrying a merchant's trades to the open trading screen.
2
3use foton_macros::{ClientPacket, WriteTo};
4use foton_registry::{packets::play::C_MERCHANT_OFFERS, trading::MerchantOffers};
5
6/// Sent right after a trading menu opens, and again whenever the offers change.
7///
8/// Vanilla parity: `ClientboundMerchantOffersPacket`. The client will not draw a
9/// trading screen at all until this arrives, which is why vanilla sends it from
10/// `Merchant.openTradingScreen` immediately after the menu packet.
11#[derive(ClientPacket, WriteTo, Clone, Debug)]
12#[packet_id(Play = C_MERCHANT_OFFERS)]
13pub struct CMerchantOffers {
14    #[write(as = VarInt)]
15    pub container_id: i32,
16    pub offers: MerchantOffers,
17    /// The merchant's level, 1..=5, which picks the badge the screen draws.
18    #[write(as = VarInt)]
19    pub villager_level: i32,
20    /// Experience toward the next level.
21    #[write(as = VarInt)]
22    pub villager_xp: i32,
23    /// Whether to draw the level badge and the experience bar at all.
24    ///
25    /// Vanilla parity: `showProgress`, which a wandering trader sends as false.
26    pub show_progress: bool,
27    /// Whether the screen may tell the player that a sold-out trade will return.
28    pub can_restock: bool,
29}
