133/// The formatting fields of a text component, without its content.
134///
135/// Vanilla parity: `Style`, whose codec shares its field names with the
136/// component codec, so the components crate's own types describe it.
137#[derive(Clone, PartialEq, Eq, Default, Serialize, Deserialize)]
138#[serde(default)]
139pub struct TextStyle {
140    #[serde(flatten)]
141    pub format: Format,
142    #[serde(flatten)]
143    pub interactions: Interactivity,
144}
145
146impl std::fmt::Debug for TextStyle {
147    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
148        formatter.write_str("TextStyle { .. }")
149    }
150}
151
152impl TextStyle {
153    /// The style a component carries.
154    #[must_use]
155    pub fn of(component: &TextComponent) -> Self {
156        Self {
157            format: component.format.clone(),
158            interactions: component.interactions.clone(),
159        }
160    }
161
162    /// Applies this style to a piece of text.
163    #[must_use]
164    pub fn apply(&self, mut text: TextComponent) -> TextComponent {
165        text.format = self.format.clone();
166        text.interactions = self.interactions.clone();
167        text
168    }
169}
170
171impl WriteTo for TextStyle {
172    fn write(&self, writer: &mut impl Write) -> Result<()> {
173        write_style(
174            &self.format,
175            &self.interactions,
176            &mut LimitedWriter::new(writer, usize::MAX),
177        )
178    }
179}

181/// How a score is drawn beside its holder.
182///
183/// Vanilla parity: `NumberFormat` and its three implementations.
184#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
185#[serde(rename_all = "snake_case")]
186pub enum NumberFormat {
187    /// `BlankFormat`: the number is not drawn.
188    Blank,
189    /// `FixedFormat`: this text is drawn in place of the number.
190    Fixed(TextComponent),
191    /// `StyledFormat`: the number, in this style.
192    Styled(TextStyle),
193}
194
195impl NumberFormat {
196    /// `NumberFormat.format`, the text drawn for a score value.
197    #[must_use]
198    pub fn format(&self, value: i32) -> TextComponent {
199        match self {
200            Self::Blank => TextComponent::plain(""),
201            Self::Fixed(text) => text.clone(),
202            Self::Styled(style) => style.apply(TextComponent::plain(value.to_string())),
203        }
204    }
205}
206
207impl WriteTo for NumberFormat {
208    /// `NumberFormatTypes.STREAM_CODEC`: the registry id of the format type
209    /// (`blank`, `styled`, `fixed`, in registration order), then its payload.
210    fn write(&self, writer: &mut impl Write) -> Result<()> {
211        match self {
212            Self::Blank => VarInt(0).write(writer),
213            Self::Styled(style) => {
214                VarInt(1).write(writer)?;
215                style.write(writer)
216            }
217            Self::Fixed(text) => {
218                VarInt(2).write(writer)?;
219                text.write(writer)
220            }
221        }
222    }
223}

225#[cfg(test)]
226mod tests {
227    use super::*;
228    use crate::packets::game::{
229        CResetScore, CSetDisplayObjective, CSetObjective, CSetScore, ObjectiveLook, ObjectiveMethod,
230    };
231    use text_components::TextComponent;
232
233    fn encoded(packet: &impl WriteTo) -> Vec<u8> {
234        let mut bytes = Vec::new();
235        packet
236            .write(&mut bytes)
237            .expect("writing to a Vec cannot fail");
238        bytes
239    }
240
241    /// The slot is its id as a `VarInt`, then the objective name; `team_black` is 3.
242    #[test]
243    fn a_display_slot_is_its_id_then_the_name() {
244        let packet = CSetDisplayObjective {
245            slot: DisplaySlot::TeamBlack,
246            objective_name: "kills".to_owned(),
247        };
248        assert_eq!(encoded(&packet), [&[3, 5][..], b"kills"].concat());
249    }
250
251    /// A removal is the name and the method byte alone; an add carries the
252    /// look, ending in the absent-number-format flag.
253    #[test]
254    fn objective_methods_carry_the_look_only_for_add_and_change() {
255        let remove = CSetObjective {
256            name: "a".to_owned(),
257            method: ObjectiveMethod::Remove,
258        };
259        assert_eq!(encoded(&remove), [1, b'a', 1]);
260
261        let add = CSetObjective {
262            name: "a".to_owned(),
263            method: ObjectiveMethod::Add(ObjectiveLook {
264                display_name: TextComponent::plain("A"),
265                render_type: ObjectiveRenderType::Hearts,
266                number_format: None,
267            }),
268        };
269        let bytes = encoded(&add);
270        assert_eq!(bytes[..3], [1, b'a', 0]);
271        assert_eq!(bytes[bytes.len() - 2..], [1, 0]);
272    }
273
274    /// Number formats are the registry id of their type: blank 0, styled 1,
275    /// fixed 2 -- the order `NumberFormatTypes.bootstrap` registers them in.
276    #[test]
277    fn number_format_types_use_their_registry_ids() {
278        let score = |format| CSetScore {
279            owner: "o".to_owned(),
280            objective_name: "b".to_owned(),
281            score: 300,
282            display: None,
283            number_format: Some(format),
284        };
285        // owner, objective, score 300 as VarInt (0xAC 0x02), no display, present format.
286        let blank = encoded(&score(NumberFormat::Blank));
287        assert_eq!(blank, [1, b'o', 1, b'b', 0xAC, 0x02, 0, 1, 0]);
288        let fixed = encoded(&score(NumberFormat::Fixed(TextComponent::plain("x"))));
289        assert_eq!(fixed[..8], [1, b'o', 1, b'b', 0xAC, 0x02, 0, 1]);
290        assert_eq!(fixed[8], 2);
291    }
292
293    #[test]
294    fn a_reset_names_an_objective_only_when_it_has_one() {
295        let all = CResetScore {
296            owner: "o".to_owned(),
297            objective_name: None,
298        };
299        assert_eq!(encoded(&all), [1, b'o', 0]);
300        let one = CResetScore {
301            owner: "o".to_owned(),
302            objective_name: Some("b".to_owned()),
303        };
304        assert_eq!(encoded(&one), [1, b'o', 1, 1, b'b']);
305    }
