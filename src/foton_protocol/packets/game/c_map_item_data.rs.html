12/// One marker drawn on top of a map.
13///
14/// Vanilla parity: `net.minecraft.world.level.saveddata.maps.MapDecoration`.
15/// `x`, `y` and `rot` are already in the map's own coordinate space: `x` and
16/// `y` span the image at two units per pixel, and `rot` is a sixteenth of a
17/// turn.
18#[derive(Clone, Debug, PartialEq)]
19pub struct MapDecorationData {
20    pub decoration_type: MapDecorationTypeRef,
21    pub x: i8,
22    pub y: i8,
23    pub rot: i8,
24    pub name: Option<TextComponent>,
25}
26
27impl WriteTo for MapDecorationData {
28    fn write(&self, writer: &mut impl Write) -> Result<()> {
29        // Registry holder reference: id + 1, because 0 means an inline value.
30        VarInt(self.decoration_type.id() as i32 + 1).write(writer)?;
31        self.x.write(writer)?;
32        self.y.write(writer)?;
33        self.rot.write(writer)?;
34        self.name.write(writer)
35    }
36}

38/// A rectangle of the map's color array that changed since the last packet.
39///
40/// Vanilla parity: `MapItemSavedData.MapPatch`.
41#[derive(Clone, Debug, PartialEq, Eq)]
42pub struct MapPatch {
43    pub start_x: u8,
44    pub start_y: u8,
45    pub width: u8,
46    pub height: u8,
47    /// `width * height` packed `MapColor` bytes, row-major.
48    pub map_colors: Vec<u8>,
49}
50
51impl WriteTo for MapPatch {
52    fn write(&self, writer: &mut impl Write) -> Result<()> {
53        // Vanilla writes width first and uses a zero width as the "absent"
54        // marker, so the field order here is not the field order of the record.
55        self.width.write(writer)?;
56        self.height.write(writer)?;
57        self.start_x.write(writer)?;
58        self.start_y.write(writer)?;
59        self.map_colors.write_prefixed::<VarInt>(writer)
60    }
61}

63/// Sends one map's scale, lock state, markers and changed colors to a client.
64///
65/// Vanilla: `ClientboundMapItemDataPacket`.
66#[derive(ClientPacket, Clone, Debug)]
67#[packet_id(Play = C_MAP_ITEM_DATA)]
68pub struct CMapItemData {
69    /// The `minecraft:map_id` component value the client keys its copy by.
70    pub map_id: i32,
71    pub scale: u8,
72    pub locked: bool,
73    /// `None` leaves the client's existing markers alone; `Some` replaces them
74    /// wholesale.
75    pub decorations: Option<Vec<MapDecorationData>>,
76    /// `None` leaves the client's existing pixels alone.
77    pub color_patch: Option<MapPatch>,
78}
79
80impl WriteTo for CMapItemData {
81    fn write(&self, writer: &mut impl Write) -> Result<()> {
82        VarInt(self.map_id).write(writer)?;
83        self.scale.write(writer)?;
84        self.locked.write(writer)?;
85        self.decorations.write(writer)?;
86        match &self.color_patch {
87            Some(patch) => patch.write(writer),
88            // Vanilla signals an absent patch with a zero width byte rather
89            // than the usual optional boolean.
90            None => 0u8.write(writer),
91        }
92    }
93}

95#[cfg(test)]
96mod tests {
97    use foton_utils::serial::WriteTo as _;
98
99    use super::{CMapItemData, MapPatch};
100
101    /// The patch header is written width-first and an absent patch is a single
102    /// zero byte, neither of which matches the shape of any other optional in
103    /// the protocol -- so getting it wrong would desync every map silently.
104    #[test]
105    fn an_absent_patch_is_one_zero_byte_and_a_present_one_leads_with_its_width() {
106        let mut absent = Vec::new();
107        CMapItemData {
108            map_id: 7,
109            scale: 0,
110            locked: false,
111            decorations: None,
112            color_patch: None,
113        }
114        .write(&mut absent)
115        .expect("map packet should encode");
116        assert_eq!(absent, vec![7, 0, 0, 0, 0]);
117
118        let mut present = Vec::new();
119        CMapItemData {
120            map_id: 1,
121            scale: 2,
122            locked: true,
123            decorations: None,
124            color_patch: Some(MapPatch {
125                start_x: 3,
126                start_y: 4,
127                width: 2,
128                height: 1,
129                map_colors: vec![0x0c, 0x0d],
130            }),
131        }
132        .write(&mut present)
133        .expect("map packet should encode");
134        assert_eq!(present, vec![1, 2, 1, 0, 2, 1, 3, 4, 2, 0x0c, 0x0d]);
135    }
