1//! Clientbound update advancements packet -- what the advancement screen
2//! draws and how far the player has got through it.
3//!
4//! Vanilla parity: `ClientboundUpdateAdvancementsPacket`, together with the
5//! `AdvancementHolder`, `Advancement` and `DisplayInfo` stream codecs it
6//! delegates to. Vanilla keeps the removals in a set and the progress in a
7//! map; Foton keeps ordered vectors, so the same packet always encodes to the
8//! same bytes and can be tested against them.
9

18use text_components::TextComponent;

21/// Set when the advancement has a tab background, which only a root does.
22const DISPLAY_HAS_BACKGROUND: i32 = 1;
23/// Set when earning the advancement pops a toast.
24const DISPLAY_SHOW_TOAST: i32 = 2;
25/// Set when the advancement stays hidden until it is earned.
26const DISPLAY_HIDDEN: i32 = 4;
27
28/// One entry of the `added` list: an advancement plus the two text components
29/// resolved for the player this packet is addressed to.
30///
31/// The resolved text is kept next to the definition rather than replacing it,
32/// so the id, the parent, the icon, the layout and the requirements are still
33/// read straight off the registry entry. Building one is the only way to pair
34/// the two, which keeps the resolved text from drifting away from the
35/// advancement it belongs to.
36#[derive(Debug, Clone)]
37pub struct AddedAdvancement {
38    advancement: AdvancementRef,
39    /// The display's `title` and `description`, resolved. Present exactly when
40    /// the advancement has a display.
41    resolved_text: Option<(TextComponent, TextComponent)>,
42}
43
44impl AddedAdvancement {
45    /// Pairs `advancement` with its title and description resolved for
46    /// `player`.
47    #[must_use]
48    pub fn new<T: TextResolutor>(advancement: AdvancementRef, player: &T) -> Self {
49        let resolved_text = advancement.display.as_ref().map(|display| {
50            (
51                display.title.resolve(player),
52                display.description.resolve(player),
53            )
54        });
55        Self {
56            advancement,
57            resolved_text,
58        }
59    }
60
61    /// The definition this entry describes.
62    #[must_use]
63    pub const fn advancement(&self) -> AdvancementRef {
64        self.advancement
65    }
66}

68/// Writes one `DisplayInfo` with its text already resolved.
69///
70/// Vanilla parity: `DisplayInfo.serializeToNetwork`. `announce_chat` is
71/// deliberately not on the wire -- the server is the one that broadcasts that
72/// message, and the client's decoder hard-codes `false` for it.
73fn write_display(
74    display: &DisplayInfo,
75    title: &TextComponent,
76    description: &TextComponent,
77    writer: &mut impl Write,
78) -> Result<()> {
79    title.write(writer)?;
80    description.write(writer)?;
81    display.icon.template().write(writer)?;
82    VarInt(display.advancement_type as i32).write(writer)?;
83
84    let mut flags = 0;
85    if display.background.is_some() {
86        flags |= DISPLAY_HAS_BACKGROUND;
87    }
88    if display.show_toast {
89        flags |= DISPLAY_SHOW_TOAST;
90    }
91    if display.hidden {
92        flags |= DISPLAY_HIDDEN;
93    }
94    // Vanilla uses `writeInt` here, not `writeVarInt`: the flags are four
95    // big-endian bytes even though they never exceed 7.
96    flags.write(writer)?;
97
98    if let Some(background) = display.background.as_ref() {
99        background.write(writer)?;
100    }
101    display.x.write(writer)?;
102    display.y.write(writer)
103}
104
105impl WriteTo for AddedAdvancement {
106    fn write(&self, writer: &mut impl Write) -> Result<()> {
107        self.advancement.key.write(writer)?;
108        self.advancement.parent.write(writer)?;
109
110        match (&self.advancement.display, &self.resolved_text) {
111            (Some(display), Some((title, description))) => {
112                true.write(writer)?;
113                write_display(display, title, description, writer)?;
114            }
115            _ => false.write(writer)?,
116        }
117
118        // Vanilla: `AdvancementRequirements.write`, a list of lists of strings.
119        let groups = self.advancement.requirements.groups;
120        VarInt(i32::try_from(groups.len()).unwrap_or(i32::MAX)).write(writer)?;
121        for group in groups {
122            VarInt(i32::try_from(group.len()).unwrap_or(i32::MAX)).write(writer)?;
123            for name in *group {
124                name.write_prefixed::<VarInt>(writer)?;
125            }
126        }
127
128        self.advancement.sends_telemetry_event.write(writer)
129    }
130}

132/// Adds, removes and re-scores the advancements one client knows about.
133///
134/// Vanilla parity: `ClientboundUpdateAdvancementsPacket`. Neither the criteria
135/// nor the rewards travel: the client only needs the tree, the artwork and the
136/// progress, and it rebuilds each advancement's requirements from the entry it
137/// was sent.
138#[derive(ClientPacket, Clone, Debug)]
139#[packet_id(Play = C_UPDATE_ADVANCEMENTS)]
140pub struct CUpdateAdvancements {
141    /// Whether the client throws away everything it already had first.
142    pub reset: bool,
143    /// The advancements the client is told about, in wire order.
144    pub added: Vec<AddedAdvancement>,
145    /// The advancements the client forgets.
146    pub removed: Vec<Identifier>,
147    /// The progress to apply, keyed by advancement.
148    pub progress: Vec<(Identifier, AdvancementProgress)>,
149    /// Whether the advancement screen can be opened at all. New in 26.2.
150    pub show_advancements: bool,
151}
152
153impl CUpdateAdvancements {
154    /// Builds the packet, resolving every added advancement's title and
155    /// description for `player`.
156    #[must_use]
157    pub fn new<T: TextResolutor>(
158        reset: bool,
159        added: impl IntoIterator<Item = AdvancementRef>,
160        removed: Vec<Identifier>,
161        progress: Vec<(Identifier, AdvancementProgress)>,
162        show_advancements: bool,
163        player: &T,
164    ) -> Self {
165        Self {
166            reset,
167            added: added
168                .into_iter()
169                .map(|advancement| AddedAdvancement::new(advancement, player))
170                .collect(),
171            removed,
172            progress,
173            show_advancements,
174        }
175    }
176}
177
178impl WriteTo for CUpdateAdvancements {
179    fn write(&self, writer: &mut impl Write) -> Result<()> {
180        self.reset.write(writer)?;
181        self.added.write(writer)?;
182        self.removed.write(writer)?;
183        self.progress.write(writer)?;
184        // Easy to forget: 26.2 added this after the progress map.
185        self.show_advancements.write(writer)
186    }
187}

198    /// When `story/root`'s only criterion was met, in epoch milliseconds.
199    const OBTAINED: i64 = 1_234;
200
201    fn encoded(value: &impl WriteTo) -> Vec<u8> {
202        let mut bytes = Vec::new();
203        value
204            .write(&mut bytes)
205            .expect("writing to a Vec cannot fail");
206        bytes
207    }
208
209    /// `story/root` with its one criterion met.
210    fn story_root_progress() -> AdvancementProgress {
211        let mut progress = AdvancementProgress::new();
212        progress.update(STORY_ROOT.requirements);
213        assert!(
214            progress.grant("crafting_table", OBTAINED),
215            "story/root's criterion is called crafting_table"
216        );
217        progress
218    }

220    /// One added advancement and one progress entry, as a player would get on
221    /// login.
222    fn story_root_packet(show_advancements: bool) -> CUpdateAdvancements {
223        CUpdateAdvancements::new(
224            true,
225            [&STORY_ROOT],
226            Vec::new(),
227            vec![(STORY_ROOT.key.clone(), story_root_progress())],
228            show_advancements,
229            &NoResolutor,
230        )
231    }

233    /// The whole packet, byte for byte. The three opaque runs -- the two text
234    /// components' NBT and the icon's item stack template -- are encoded on
235    /// their own rather than spelled out, so the parts this packet is
236    /// responsible for stay readable.
237    #[test]
238    fn an_added_advancement_encodes_exactly_as_vanilla_reads_it() {
239        init_vanilla_registry();
240
241        let display = STORY_ROOT.display.as_ref().expect("story/root is drawn");
242        let title = encoded(&display.title.resolve(&NoResolutor));
243        let description = encoded(&display.description.resolve(&NoResolutor));
244        let icon = encoded(display.icon.template());
245        let key = encoded(&STORY_ROOT.key);
246        let background = encoded(
247            display
248                .background
249                .as_ref()
250                .expect("story/root has a tab background"),
251        );
252
253        let mut expected = Vec::new();
254        expected.push(1); // reset
255        expected.push(1); // one added advancement
256        expected.extend_from_slice(&key);
257        expected.push(0); // no parent: story/root is a root
258        expected.push(1); // it has a display
259        expected.extend_from_slice(&title);
260        expected.extend_from_slice(&description);
261        expected.extend_from_slice(&icon);
262        expected.push(0); // frame: task, written as a var-int
263        expected.extend_from_slice(&[0, 0, 0, 1]); // flags: a plain int, background only
264        expected.extend_from_slice(&background);
265        expected.extend_from_slice(&0.0f32.to_be_bytes()); // x
266        expected.extend_from_slice(&1.75f32.to_be_bytes()); // y
267        expected.push(1); // one requirement group
268        expected.push(1); // holding one criterion
269        expected.push(14);
270        expected.extend_from_slice(b"crafting_table");
271        expected.push(1); // sends_telemetry_event
272        expected.push(0); // nothing removed
273        expected.push(1); // one progress entry
274        expected.extend_from_slice(&key);
275        expected.push(1); // covering one criterion
276        expected.push(14);
277        expected.extend_from_slice(b"crafting_table");
278        expected.push(1); // which was obtained
279        expected.extend_from_slice(&OBTAINED.to_be_bytes());
280        expected.push(1); // show_advancements
281
282        assert_eq!(encoded(&story_root_packet(true)), expected);
283    }

285    /// The display flags are `writeInt`, not `writeVarInt`. A var-int would
286    /// encode `story/root`'s flags as the single byte `1` and shift every
287    /// following field three bytes earlier, which desynchronizes the whole
288    /// packet rather than failing loudly.
289    #[test]
290    fn the_display_flags_occupy_four_bytes() {
291        init_vanilla_registry();
292
293        let display = STORY_ROOT.display.as_ref().expect("story/root is drawn");
294        let title = encoded(&display.title.resolve(&NoResolutor));
295        let description = encoded(&display.description.resolve(&NoResolutor));
296        let icon = encoded(display.icon.template());
297        let key = encoded(&STORY_ROOT.key);
298        let background = encoded(
299            display
300                .background
301                .as_ref()
302                .expect("story/root has a tab background"),
303        );
304
305        // reset, added count, id, absent parent, present display, the three
306        // opaque runs, and the one-byte frame var-int.
307        let flags_at = 1 + 1 + key.len() + 1 + 1 + title.len() + description.len() + icon.len() + 1;
308        let bytes = encoded(&story_root_packet(true));
309
310        assert_eq!(&bytes[flags_at..flags_at + 4], &[0, 0, 0, 1]);
311        // The background identifier follows the flags, and only because the
312        // background bit was set.
313        assert_eq!(
314            &bytes[flags_at + 4..flags_at + 4 + background.len()],
315            &background[..]
316        );
317    }

319    /// 26.2 appended `show_advancements` after the progress map. Dropping it
