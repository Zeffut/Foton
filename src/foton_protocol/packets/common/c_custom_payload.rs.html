1use foton_macros::{ClientPacket, WriteTo};
2use foton_registry::packets::config::C_CUSTOM_PAYLOAD;
3use foton_registry::packets::play::C_CUSTOM_PAYLOAD as PLAY_C_CUSTOM_PAYLOAD;
4use foton_utils::Identifier;
5
6#[derive(WriteTo, ClientPacket, Clone, Debug)]
7#[packet_id(Config = C_CUSTOM_PAYLOAD, Play = PLAY_C_CUSTOM_PAYLOAD)]
8pub struct CCustomPayload {
9    pub identifier: Identifier,
10    // Vanilla's custom payload codec writes the bytes remaining in the packet.
11    #[write(as = NoPrefixVec)]
12    pub payload: Box<[u8]>,
13}
14
15impl CCustomPayload {
16    #[must_use]
17    pub const fn new(identifier: Identifier, payload: Box<[u8]>) -> Self {
18        Self {
19            identifier,
20            payload,
21        }
22    }
23}

25#[cfg(test)]
26mod tests {
27    use foton_utils::{Identifier, serial::WriteTo};
28
29    use super::CCustomPayload;
30
31    #[test]
32    fn custom_payload_bytes_follow_identifier_without_a_length_prefix() {
33        let identifier: Identifier = "fixture:voice".parse().expect("valid fixture identifier");
34        for body in [Vec::new(), vec![0x55, 0x66]] {
35            let packet = CCustomPayload::new(identifier.clone(), body.clone().into_boxed_slice());
36            let mut encoded = Vec::new();
37            packet.write(&mut encoded).expect("encode custom payload");
38
39            let mut expected = vec![13];
40            expected.extend_from_slice(b"fixture:voice");
41            expected.extend_from_slice(&body);
42            assert_eq!(encoded, expected);
43        }
44    }
