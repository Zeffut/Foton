23// raw -> compress -> encrypt
24/// A writer that can encrypt data.
25pub enum EncryptionWriter<W: AsyncWrite + Unpin> {
26    /// A writer that encrypts data.
27    Encrypt(Box<StreamEncryptor<W>>),
28    /// A writer that does not encrypt data.
29    None(W),
30}
31
32impl<W: AsyncWrite + Unpin> EncryptionWriter<W> {
33    /// Upgrades the writer to encrypt data.
34    ///
35    /// # Panics
36    /// - If the writer is already encrypting data.
37    #[must_use]
38    #[cfg_attr(
39        not(test),
40        expect(
41            clippy::panic,
42            reason = "the only caller checks the same condition first and refuses there; this arm is the type making that check enforceable"
43        )
44    )]
45    pub fn upgrade(self, cipher: Aes128Cfb8Enc) -> Self {
46        match self {
47            Self::None(stream) => Self::Encrypt(Box::new(StreamEncryptor::new(cipher, stream))),
48            Self::Encrypt(_) => panic!("Cannot upgrade a stream that already has a cipher!"),
49        }
50    }
51}
52
53impl<W: AsyncWrite + Unpin> AsyncWrite for EncryptionWriter<W> {
54    fn poll_write(
55        self: Pin<&mut Self>,
56        cx: &mut Context<'_>,
57        buf: &[u8],
58    ) -> Poll<io::Result<usize>> {
59        match self.get_mut() {
60            Self::Encrypt(writer) => {
61                let writer = Pin::new(writer);
62                writer.poll_write(cx, buf)
63            }
64            Self::None(writer) => {
65                let writer = Pin::new(writer);
66                writer.poll_write(cx, buf)
67            }
68        }
69    }
70
71    fn poll_flush(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
72        match self.get_mut() {
73            Self::Encrypt(writer) => {
74                let writer = Pin::new(writer);
75                writer.poll_flush(cx)
76            }
77            Self::None(writer) => {
78                let writer = Pin::new(writer);
79                writer.poll_flush(cx)
80            }
81        }
82    }
83
84    fn poll_shutdown(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
85        match self.get_mut() {
86            Self::Encrypt(writer) => {
87                let writer = Pin::new(writer);
88                writer.poll_shutdown(cx)
89            }
90            Self::None(writer) => {
91                let writer = Pin::new(writer);
92                writer.poll_shutdown(cx)
93            }
94        }
95    }
96}

98/// Encoder: Server -> Client
99/// Supports `ZLib` endecoding/compression
100/// Supports Aes128 Encryption
101pub struct TCPNetworkEncoder<W: AsyncWrite + Unpin> {
102    writer: EncryptionWriter<W>,
103}
104
105impl<W: AsyncWrite + Unpin> TCPNetworkEncoder<W> {
106    /// Creates a new `TCPNetworkEncoder`.
107    pub const fn new(writer: W) -> Self {
108        Self {
109            writer: EncryptionWriter::None(writer),
110        }
111    }
112
113    /// NOTE: Encryption can only be set; a minecraft stream cannot go back to being unencrypted
114    ///
115    /// # Panics
116    /// - If the stream is already encrypted.
117    #[cfg_attr(
118        not(test),
119        expect(
120            clippy::panic,
121            reason = "refusing to re-key a live stream is a programming error, and the Panics note above says so"
122        )
123    )]
124    pub fn set_encryption(&mut self, key: &[u8; 16]) {
125        if matches!(self.writer, EncryptionWriter::Encrypt(_)) {
126            panic!("Cannot upgrade a stream that already has a cipher!");
127        }
128        // `new_from_slices` is the fallible constructor because it takes
129        // slices of unknown length. AES-128-CFB8 wants sixteen bytes of key
130        // and sixteen of IV, and `key` is a `[u8; 16]`, so the sizes are
131        // already settled by the type: `new` takes the sized arrays and
132        // cannot fail, which retires the `expect` rather than excusing it.
133        let cipher = Aes128Cfb8Enc::new(key.into(), key.into());
134        replace_with::replace_with_or_abort(&mut self.writer, |encoder| encoder.upgrade(cipher));
135    }
136
137    /// Writes a packet to the stream.
138    ///
139    /// # Errors
140    /// - If the packet fails to write.
141    /// - If the stream fails to flush.
142    pub async fn write_packet(&mut self, packet: &EncodedPacket) -> Result<(), PacketError> {
143        self.writer
144            .write_all(&packet.encoded_data)
145            .await
146            .map_err(|e| PacketError::EncryptionFailed(e.to_string()))?;
147
148        self.writer
149            .flush()
150            .await
151            .map_err(|e| PacketError::EncryptionFailed(e.to_string()))
152    }
153}
