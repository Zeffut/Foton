1//! Shared vanilla `TamableAnimal` state and hooks.
2//!
3//! Vanilla parity: `TamableAnimal` and `OwnableEntity`. A tameable animal is an
4//! [`Animal`] that remembers one player, obeys a sit order, and teleports to its
5//! owner when it falls too far behind. Wolf, cat and parrot all sit on this, and
6//! none of them behaves right without it: the sit order, the collar owner check
7//! and the "who may I attack" rule are all decided here rather than per mob.
8

34/// Bit of the synced flags byte that marks a sitting pose.
35///
36/// Vanilla parity: the `& 1` of `TamableAnimal.isInSittingPose`.
37const FLAG_IN_SITTING_POSE: i8 = 1;
38
39/// Bit of the synced flags byte that marks a tamed animal.
40///
41/// Vanilla parity: the `& 4` of `TamableAnimal.isTame`.
42const FLAG_TAME: i8 = 4;
43
44/// Squared distance past which a pet teleports rather than walks to its owner.
45///
46/// Vanilla parity: `TamableAnimal.TELEPORT_WHEN_DISTANCE_IS_SQ`.
47pub const TELEPORT_WHEN_DISTANCE_IS_SQ: f64 = 144.0;
48
49/// How many landing spots a teleporting pet tries before giving up.
50///
51/// Vanilla parity: the ten attempts of `TamableAnimal.teleportToAroundBlockPos`.
52const TELEPORT_ATTEMPTS: i32 = 10;
53
54/// Smallest horizontal offset a teleport may land at.
55///
56/// Vanilla parity: `MIN_HORIZONTAL_DISTANCE_FROM_TARGET_AFTER_TELEPORTING`, which
57/// is why a pet never lands on top of the player it was following.
58const MIN_HORIZONTAL_TELEPORT_DISTANCE: i32 = 2;
59
60/// Largest horizontal offset a teleport may land at.
61///
62/// Vanilla parity: `MAX_HORIZONTAL_DISTANCE_FROM_TARGET_AFTER_TELEPORTING`.
63const MAX_HORIZONTAL_TELEPORT_DISTANCE: i32 = 3;
64
65/// Largest vertical offset a teleport may land at.
66///
67/// Vanilla parity: `MAX_VERTICAL_DISTANCE_FROM_TARGET_AFTER_TELEPORTING`.
68const MAX_VERTICAL_TELEPORT_DISTANCE: i32 = 1;
69
70/// How deep the owner chain is walked before it is treated as a cycle.
71///
72/// Vanilla walks it with a set of everything it has seen; Foton keeps the set
73/// and adds this cap so a corrupted save cannot spin the server.
74const MAX_OWNER_CHAIN_DEPTH: usize = 16;
75
76/// Runtime fields shared by vanilla tameable animals.
77///
78/// Vanilla parity: the single `orderedToSit` field of `TamableAnimal`. The tame
79/// flag, the sitting pose and the owner all live in synchronized entity data
80/// instead, so the entity owns them and this only holds what is server-side.
81#[derive(Debug)]
82pub struct TamableAnimalBase {
83    ordered_to_sit: SyncMutex<bool>,
84}
85
86impl TamableAnimalBase {
87    /// Creates default tameable runtime state.
88    #[must_use]
89    pub const fn new() -> Self {
90        Self {
91            ordered_to_sit: SyncMutex::new(false),
92        }
93    }
94
95    /// Returns vanilla `TamableAnimal.orderedToSit`.
96    #[must_use]
97    pub fn is_ordered_to_sit(&self) -> bool {
98        *self.ordered_to_sit.lock()
99    }
100
101    /// Sets vanilla `TamableAnimal.orderedToSit`.
102    pub fn set_ordered_to_sit(&self, ordered_to_sit: bool) {
103        *self.ordered_to_sit.lock() = ordered_to_sit;
104    }
105}
106
107impl Default for TamableAnimalBase {
108    fn default() -> Self {
109        Self::new()
110    }
111}

113/// Vanilla-shaped behavior shared by entities that extend `TamableAnimal`.
114pub trait TamableAnimal: Animal {
115    /// Returns shared tameable runtime state.
116    fn tamable_base(&self) -> &TamableAnimalBase;
117
118    /// Returns the synchronized `TamableAnimal.DATA_FLAGS_ID` byte.
119    fn tamable_flags(&self) -> i8;
120
121    /// Sets the synchronized `TamableAnimal.DATA_FLAGS_ID` byte.
122    fn set_tamable_flags(&self, flags: i8);
123
124    /// Returns the synchronized owner reference.
125    ///
126    /// Vanilla parity: `TamableAnimal.DATA_OWNERUUID_ID`. Vanilla stores an
127    /// `EntityReference`; Foton stores the UUID it resolves from.
128    fn owner_uuid(&self) -> Option<Uuid>;
129
130    /// Sets the synchronized owner reference.
131    fn set_owner_uuid(&self, owner: Option<Uuid>);
132
133    /// Returns vanilla `TamableAnimal.isTame`.
134    fn is_tame(&self) -> bool {
135        self.tamable_flags() & FLAG_TAME != 0
136    }
137
138    /// Applies vanilla `TamableAnimal.setTame`.
139    fn set_tame(&self, is_tame: bool, include_side_effects: bool) {
140        let current = self.tamable_flags();
141        let updated = if is_tame {
142            current | FLAG_TAME
143        } else {
144            current & !FLAG_TAME
145        };
146        self.set_tamable_flags(updated);
147
148        if include_side_effects {
149            self.apply_taming_side_effects();
150        }
151    }
152
153    /// Hook for vanilla `TamableAnimal.applyTamingSideEffects`.
154    fn apply_taming_side_effects(&self) {}
155
156    /// Returns vanilla `TamableAnimal.isInSittingPose`.
157    fn is_in_sitting_pose(&self) -> bool {
158        self.tamable_flags() & FLAG_IN_SITTING_POSE != 0
159    }
160
161    /// Applies vanilla `TamableAnimal.setInSittingPose`.
162    fn set_in_sitting_pose(&self, value: bool) {
163        let current = self.tamable_flags();
164        let updated = if value {
165            current | FLAG_IN_SITTING_POSE
166        } else {
167            current & !FLAG_IN_SITTING_POSE
168        };
169        self.set_tamable_flags(updated);
170    }
171
172    /// Returns vanilla `TamableAnimal.isOrderedToSit`.
173    fn is_ordered_to_sit(&self) -> bool {
174        self.tamable_base().is_ordered_to_sit()
175    }
176
177    /// Sets vanilla `TamableAnimal.setOrderedToSit`.
178    fn set_ordered_to_sit(&self, ordered_to_sit: bool) {
179        self.tamable_base().set_ordered_to_sit(ordered_to_sit);
180    }
181
182    /// Returns vanilla `OwnableEntity.getOwner`.
183    fn owner(&self) -> Option<SharedEntity> {
184        let uuid = self.owner_uuid()?;
185        let world = self.level()?;
186        let owner = world.get_entity_by_uuid(&uuid)?;
187        owner.as_living_entity()?;
188        Some(owner)
189    }
190
191    /// Returns vanilla `OwnableEntity.getRootOwner`.
192    ///
193    /// A pet owned by a pet answers to whoever is at the top of the chain. A
194    /// cycle resolves to nothing, exactly as vanilla's `seen` set makes it.
195    fn root_owner(&self) -> Option<SharedEntity> {
196        let mut seen = FxHashSet::default();
197        seen.insert(self.uuid());
198
199        let mut owner = self.owner()?;
200        for _ in 0..MAX_OWNER_CHAIN_DEPTH {
201            if !seen.insert(owner.uuid()) {
202                return None;
203            }
204            let Some(tamable) = owner.as_tamable_animal() else {
205                return Some(owner);
206            };
207            let Some(next) = tamable.owner() else {
208                return Some(owner);
209            };
210            owner = next;
211        }
212
213        None
214    }
215
216    /// Returns vanilla `TamableAnimal.isOwnedBy`.
217    fn is_owned_by(&self, entity: &dyn Entity) -> bool {
218        self.owner_uuid() == Some(entity.uuid())
219    }
220
221    /// Applies vanilla `TamableAnimal.tame`.
222    fn tame(&self, player: &Player) {
223        self.set_tame(true, true);
224        self.set_owner_uuid(Some(player.gameprofile.id));
225        triggers::entity::tame_animal(player, self.as_entity_event_source());
226    }
227
228    /// Returns vanilla `TamableAnimal.canAttack`.
229    fn can_attack_tamable(&self, target: &dyn LivingEntity) -> bool {
230        if self.is_owned_by(target.as_entity_event_source()) {
231            return false;
232        }
233        Mob::can_attack(self, target)
234    }
235
236    /// Returns vanilla `TamableAnimal.wantsToAttack`.
237    fn wants_to_attack(&self, _target: &dyn LivingEntity, _owner: &dyn Entity) -> bool {
238        true
239    }
240
241    /// Returns vanilla `TamableAnimal.considersEntityAsAlly`.
242    fn considers_entity_as_ally_tamable(&self, other: &dyn Entity) -> bool {
243        if !self.is_tame() {
244            return false;
245        }
246        let Some(owner) = self.root_owner() else {
247            return false;
248        };
249        if owner.uuid() == other.uuid() {
250            return true;
251        }
252        owner.is_allied_to(other)
253    }
254
255    /// Applies vanilla `TamableAnimal.feed`.
256    fn feed(
257        &self,
258        player: &Player,
259        hand: InteractionHand,
260        item_stack: &ItemStack,
261        healing_factor: f32,
262        default_heal: f32,
263    ) {
264        let nutrition = item_stack.get(FOOD).map(FoodProperties::nutrition);
265        Mob::use_player_item(self, player, hand);
266        self.heal(nutrition.map_or(default_heal, |nutrition| healing_factor * nutrition as f32));
267        self.play_eating_sound();
268    }
269
270    /// Shows the vanilla taming outcome.
271    ///
272    /// Vanilla parity: `TamableAnimal.spawnTamingParticles` runs on the client
273    /// off entity events 6 and 7, so the server's whole job is broadcasting one
274    /// of them.
275    fn spawn_taming_particles(&self, success: bool) {
276        self.broadcast_entity_event(if success {
277            EntityStatus::TamingSucceeded
278        } else {
279            EntityStatus::TamingFailed
280        });
281    }
282
283    /// Returns vanilla `TamableAnimal.unableToMoveToOwner`.
284    fn unable_to_move_to_owner(&self) -> bool {
285        if self.is_ordered_to_sit() || self.is_passenger() || self.may_be_leashed() {
286            return true;
287        }
288
289        self.owner()
290            .is_some_and(|owner| owner.as_player().is_some_and(Entity::is_spectator))
291    }
292
293    /// Returns vanilla `TamableAnimal.canFlyToOwner`.
294    fn can_fly_to_owner(&self) -> bool {
295        false
296    }
297
298    /// Returns vanilla `TamableAnimal.shouldTryTeleportToOwner`.
299    fn should_try_teleport_to_owner(&self) -> bool {
300        self.owner().is_some_and(|owner| {
301            self.position().distance_squared(owner.position()) >= TELEPORT_WHEN_DISTANCE_IS_SQ
302        })
303    }
304
305    /// Applies vanilla `TamableAnimal.tryToTeleportToOwner`.
306    fn try_to_teleport_to_owner(&self) {
307        let Some(owner) = self.owner() else {
308            return;
309        };
310        self.teleport_to_around_block_pos(owner.block_position());
311    }
312
313    /// Applies vanilla `TamableAnimal.teleportToAroundBlockPos`.
314    fn teleport_to_around_block_pos(&self, target_pos: BlockPos) {
315        for _ in 0..TELEPORT_ATTEMPTS {
316            let xd = rand::random_range(
317                -MAX_HORIZONTAL_TELEPORT_DISTANCE..=MAX_HORIZONTAL_TELEPORT_DISTANCE,
318            );
319            let zd = rand::random_range(
320                -MAX_HORIZONTAL_TELEPORT_DISTANCE..=MAX_HORIZONTAL_TELEPORT_DISTANCE,
321            );
322            if xd.abs() < MIN_HORIZONTAL_TELEPORT_DISTANCE
323                && zd.abs() < MIN_HORIZONTAL_TELEPORT_DISTANCE
324            {
325                continue;
326            }
327
328            let yd = rand::random_range(
329                -MAX_VERTICAL_TELEPORT_DISTANCE..=MAX_VERTICAL_TELEPORT_DISTANCE,
330            );
331            if self.maybe_teleport_to(BlockPos::new(
332                target_pos.x() + xd,
333                target_pos.y() + yd,
334                target_pos.z() + zd,
335            )) {
336                return;
337            }
338        }
339    }
340
341    /// Applies vanilla `TamableAnimal.maybeTeleportTo`.
342    fn maybe_teleport_to(&self, pos: BlockPos) -> bool {
343        if !self.can_teleport_to(pos) {
344            return false;
345        }
346
347        let (x, y, z) = pos.get_bottom_center();
348        if self.try_set_position(DVec3::new(x, y, z)).is_err() {
349            return false;
350        }
351        self.mob_base().navigation().lock().stop();
352        true
353    }
354
355    /// Applies vanilla `TamableAnimal.canTeleportTo`.
356    fn can_teleport_to(&self, pos: BlockPos) -> bool {
357        let Some(world) = self.level() else {
358            return false;
359        };
360
361        let mut context = PathfindingContext::new(world.as_ref(), self.block_position());
362        if WalkPathEvaluator::path_type_static(&mut context, pos) != PathType::Walkable {
363            return false;
364        }
365
366        if !self.can_fly_to_owner()
367            && world
368                .get_block_state(pos.below())
369                .get_block()
370                .has_tag(&BlockTag::LEAVES)
371        {
372            return false;
373        }
374
375        let current = self.block_position();
376        let delta = DVec3::new(
377            f64::from(pos.x() - current.x()),
378            f64::from(pos.y() - current.y()),
379            f64::from(pos.z() - current.z()),
380        );
381        let moved_box = self.bounding_box().translate(delta);
382        !collision::has_collision(
383            &WorldCollisionProvider::for_entity(&world, self.as_entity_event_source()),
384            moved_box,
385        )
386    }

388    /// Sends the vanilla death message a pet's owner gets.
389    ///
390    /// Vanilla parity: the `TamableAnimal.die` override, which sends the owner
391    /// `getCombatTracker().getDeathMessage()`. Foton has no combat tracker yet,
392    /// so the message is built the same way [`Player`] already builds its own:
393    /// the `death.attack.<message_id>` key with the victim's display name.
394    fn notify_owner_of_death(&self, source: &DamageSource) {
395        let Some(world) = self.level() else {
396            return;
397        };
398        if !world.get_game_rule(&SHOW_DEATH_MESSAGES) {
399            return;
400        }
401        let Some(owner) = self.owner() else {
402            return;
403        };
404        let Some(player) = owner.as_player() else {
405            return;
406        };
407
408        let content = TranslatedMessage {
409            key: format!("death.attack.{}", source.damage_type.message_id).into(),
410            fallback: None,
411            args: Some(Box::new([self.display_name()])),
412        }
413        .component();
414        player.send_packet(CSystemChat {
415            content,
416            overlay: false,
417        });
418    }
419
420    /// Saves vanilla tameable animal fields.
421    fn save_tamable_animal(&self, nbt: &mut NbtCompound) {
422        if let Some(owner) = self.owner_uuid() {
423            nbt.insert("Owner", NbtTag::IntArray(owner.to_int_array().to_vec()));
424        }
425        nbt.insert("Sitting", i8::from(self.is_ordered_to_sit()));
426    }
427
428    /// Loads vanilla tameable animal fields.
429    fn load_tamable_animal(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
430        let owner = nbt
431            .int_array("Owner")
432            .and_then(|values| Uuid::from_int_array(&values));
433        if let Some(owner) = owner {
434            self.set_owner_uuid(Some(owner));
435            self.set_tame(true, false);
436        } else {
437            self.set_owner_uuid(None);
438            self.set_tame(false, true);
439        }
440
441        let ordered_to_sit = nbt.byte("Sitting").is_some_and(|value| value != 0);
442        self.set_ordered_to_sit(ordered_to_sit);
443        self.set_in_sitting_pose(ordered_to_sit);
444    }
445}

447/// Returns whether `entity` is a tamed animal.
448///
449/// Vanilla parity: the `entity instanceof TamableAnimal animal && animal.isTame()`
450/// test that `Wolf.wantsToAttack` and `Fox.FoxAlertableEntitiesSelector` share.
451#[must_use]
452pub fn is_tamed(entity: &dyn Entity) -> bool {
453    entity
454        .as_tamable_animal()
455        .is_some_and(TamableAnimal::is_tame)
456}
457
458/// Updates the tame flag when the entity supports vanilla taming.
459pub fn set_tamed(entity: &dyn Entity, tamed: bool) -> bool {
460    let Some(animal) = entity.as_tamable_animal() else {
461        return false;
462    };
463    animal.set_tame(tamed, true);
464    true
465}
466
467/// Returns the vanilla owner UUID, when the entity supports taming.
468pub fn owner_uuid(entity: &dyn Entity) -> Option<Uuid> {
469    entity
470        .as_tamable_animal()
471        .and_then(TamableAnimal::owner_uuid)
472}
473
474/// Updates the vanilla owner UUID, returning whether the entity is tamable.
475pub fn set_owner_uuid(entity: &dyn Entity, owner: Option<Uuid>) -> bool {
476    let Some(animal) = entity.as_tamable_animal() else {
477        return false;
478    };
479    animal.set_owner_uuid(owner);
480    true
481}

487    #[test]
488    fn the_tame_and_sitting_flags_occupy_different_bits() {
489        // The two live in one synced byte; overlapping them would make a sitting
490        // pet read as tamed, which is what the client draws the collar from.
491        assert_eq!(FLAG_TAME & FLAG_IN_SITTING_POSE, 0);
492    }

494    #[test]
495    fn a_fresh_tamable_animal_is_not_ordered_to_sit() {
496        let base = TamableAnimalBase::new();
497        assert!(!base.is_ordered_to_sit());
498
499        base.set_ordered_to_sit(true);
500        assert!(base.is_ordered_to_sit());
501    }
