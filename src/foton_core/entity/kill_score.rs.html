1//! Crediting a kill.
2//!
3//! Vanilla parity: `Entity.awardKillScore` and the `ServerPlayer` override,
4//! which is one decision with three outcomes -- the two kill triggers and the
5//! kill statistics -- and so is kept whole here rather than split across them.
6
7use foton_registry::stat::Stat;
8use foton_registry::{vanilla_custom_stats, vanilla_stat_types};
9
10use crate::advancement::triggers;
11use crate::entity::Entity;
12use crate::entity::damage::DamageSource;
13
14/// Credits `killer` with killing `victim`.
15///
16/// The `ServerPlayer` override wraps the base in `victim != this`, so a player
17/// who kills themselves is credited with nothing -- which is why the guard is
18/// here rather than in either half.
19pub fn award_kill_score(killer: &dyn Entity, victim: &dyn Entity, killing_blow: &DamageSource) {
20    if killer.as_player().is_some() && killer.id() == victim.id() {
21        return;
22    }
23
24    // The base `Entity.awardKillScore`, which only speaks when a player died.
25    if let Some(victim_player) = victim.as_player() {
26        triggers::entity::entity_killed_player(victim_player, killer, killing_blow);
27    }
28
29    let Some(killer_player) = killer.as_player() else {
30        return;
31    };
32    // TODO: drive the `killCount` scoreboard objectives here too, once Foton
33    // has statistic-keyed objectives.
34    if victim.as_player().is_some() {
35        killer_player.award_custom_stat(&vanilla_custom_stats::PLAYER_KILLS);
36    } else {
37        killer_player.award_custom_stat(&vanilla_custom_stats::MOB_KILLS);
38    }
39    killer_player.award_stat(Stat::new(&vanilla_stat_types::KILLED, victim.entity_type()));
40    triggers::entity::player_killed_entity(killer_player, victim, killing_blow);
41}
