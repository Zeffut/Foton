1//! Mobs that fit in a bucket.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.Bucketable`. The interface is
4//! two halves that have to agree: the mob writes itself into a bucket when a
5//! player scoops it up, and `MobBucketItem` reads it back out when the bucket
6//! is emptied. Foton had only the second half, and it spawned a fresh mob
7//! rather than the one that went in; this is the first half plus the state the
8//! second half was missing.
9

27/// A mob a water bucket can pick up.
28///
29/// Vanilla parity: the `Bucketable` interface. Vanilla's two `saveDefaultData`
30/// / `loadDefaultData` statics are [`save_default_data_to_bucket_tag`] and
31/// [`load_default_data_from_bucket_tag`] here, because Rust has no static
32/// interface methods.
33pub trait Bucketable: LivingEntity {
34    /// Vanilla parity: `Bucketable.fromBucket`.
35    #[expect(
36        clippy::wrong_self_convention,
37        reason = "this is vanilla's `fromBucket()` getter, not a conversion"
38    )]
39    fn from_bucket(&self) -> bool;
40
41    /// Vanilla parity: `Bucketable.setFromBucket`.
42    fn set_from_bucket(&self, from_bucket: bool);
43
44    /// Vanilla parity: `Bucketable.saveToBucketTag`.
45    fn save_to_bucket_tag(&self, bucket: &mut ItemStack);
46
47    /// Vanilla parity: `Bucketable.loadFromBucketTag`.
48    fn load_from_bucket_tag(&self, tag: BorrowedNbtCompoundView<'_, '_>);
49
50    /// Vanilla parity: `Bucketable.getBucketItemStack`.
51    fn bucket_item_stack(&self) -> ItemStack;
52
53    /// Vanilla parity: `Bucketable.getPickupSound`.
54    fn pickup_sound(&self) -> SoundEventRef;
55
56    /// Vanilla parity: `Bucketable.canBePickedUpWithBucket`.
57    fn can_be_picked_up_with_bucket(&self, item_stack: &ItemStack) -> bool {
58        item_stack.is(&vanilla_items::WATER_BUCKET)
59    }
60}

62/// Writes the flags every bucketed mob carries into the bucket.
63///
64/// Vanilla parity: `Bucketable.saveDefaultDataToBucketTag`. Each flag is only
65/// written when it is set, so an ordinary mob leaves an all-but-empty tag
66/// behind -- which is why the bucket of a plain fish stacks with another.
67pub fn save_default_data_to_bucket_tag(mob: &dyn Mob, bucket: &mut ItemStack) {
68    if let Some(custom_name) = mob.custom_name() {
69        bucket.set(vanilla_components::CUSTOM_NAME, custom_name);
70    }
71
72    let mut tag = NbtCompound::new();
73    if mob.is_no_ai() {
74        tag.insert("NoAI", true);
75    }
76    if mob.is_silent() {
77        tag.insert("Silent", true);
78    }
79    if mob.is_no_gravity() {
80        tag.insert("NoGravity", true);
81    }
82    if mob.has_glowing_tag() {
83        tag.insert("Glowing", true);
84    }
85    if mob.is_invulnerable() {
86        tag.insert("Invulnerable", true);
87    }
88    if mob.is_persistence_required() {
89        tag.insert("PersistenceRequired", true);
90    }
91    tag.insert("Health", mob.get_health());
92
93    set_bucket_entity_data(bucket, tag);
94}

96/// Reads those flags back out of the bucket.
97///
98/// Vanilla parity: `Bucketable.loadDefaultDataFromBucketTag`. Vanilla's
99/// `ifPresent` shape matters: a key that is absent leaves the mob's own default
100/// alone rather than clearing it.
101pub fn load_default_data_from_bucket_tag(mob: &dyn Mob, tag: BorrowedNbtCompoundView<'_, '_>) {
102    if let Some(no_ai) = tag.byte("NoAI") {
103        mob.set_no_ai(no_ai != 0);
104    }
105    if let Some(silent) = tag.byte("Silent") {
106        mob.set_silent(silent != 0);
107    }
108    if let Some(no_gravity) = tag.byte("NoGravity") {
109        mob.set_no_gravity(no_gravity != 0);
110    }
111    if let Some(glowing) = tag.byte("Glowing") {
112        mob.set_glowing_tag(glowing != 0);
113    }
114    if let Some(invulnerable) = tag.byte("Invulnerable") {
115        mob.set_invulnerable(invulnerable != 0);
116    }
117    if tag
118        .byte("PersistenceRequired")
119        .is_some_and(|flag| flag != 0)
120    {
121        mob.set_persistence_required();
122    }
123    if let Some(health) = tag.float("Health") {
124        mob.set_health(health);
125    }
126}

128/// Merges `tag` into a bucket's `minecraft:bucket_entity_data`.
129///
130/// Vanilla parity: `CustomData.update(DataComponents.BUCKET_ENTITY_DATA, ...)`,
131/// which reads the component, hands the tag to the mutator and writes it back.
132pub fn set_bucket_entity_data(bucket: &mut ItemStack, tag: NbtCompound) {
133    let Some(custom_data) = CustomData::try_from_compound(tag) else {
134        return;
135    };
136    bucket.set(vanilla_components::BUCKET_ENTITY_DATA, custom_data);
137}

139/// Hands a bucket's `minecraft:bucket_entity_data` to `read` as a borrowed view.
140///
141/// Vanilla parity: the
142/// `itemStack.getOrDefault(BUCKET_ENTITY_DATA, CustomData.EMPTY).copyTag()` of
143/// `MobBucketItem.spawn`. The component stores owned NBT and every loader in
144/// Foton reads the borrowed shape, so the tag makes one round trip through
145/// bytes here rather than in each caller.
146pub fn read_bucket_entity_data(bucket: &ItemStack, read: impl FnOnce(BorrowedNbtCompoundView)) {
147    let Some(saved) = bucket.get(vanilla_components::BUCKET_ENTITY_DATA) else {
148        return;
149    };
150    let mut bytes = Vec::new();
151    saved.copy_tag().write(&mut bytes);
152    let Ok(borrowed) = read_borrowed_compound(&mut Cursor::new(bytes.as_slice())) else {
153        return;
154    };
155    read(BorrowedNbtCompoundView::from(&borrowed));
156}

158/// Picks a mob up with the water bucket in `hand`.
159///
160/// Vanilla parity: `Bucketable.bucketMobPickup`. Returns `None` -- vanilla's
161/// `Optional.empty()` -- when the held item is not a bucket this mob answers
162/// to, so the caller falls through to its own interaction.
163#[must_use]
164pub fn bucket_mob_pickup<T: Bucketable + Mob + ?Sized>(
165    player: &Player,
166    hand: InteractionHand,
167    pickup_entity: &T,
168) -> Option<InteractionResult> {
169    let held = {
170        let inventory = player.inventory.lock();
171        let held = inventory.get_item_in_hand(hand);
172        held.copy_with_count(held.count())
173    };
174    if !pickup_entity.can_be_picked_up_with_bucket(&held) || !Entity::is_alive(pickup_entity) {
175        return None;
176    }
177
178    pickup_entity.play_sound(pickup_entity.pickup_sound(), 1.0, 1.0);
179    let mut bucket = pickup_entity.bucket_item_stack();
180    pickup_entity.save_to_bucket_tag(&mut bucket);
181
182    // Vanilla parity: `ItemUtils.createFilledResult(itemStack, player, bucket,
183    // false)` -- the `false` is `limitCreativeStackSize`, so a creative player
184    // still gets the filled bucket rather than nothing.
185    // The filled bucket is moved into the inventory below, and the trigger
186    // wants the same stack vanilla hands it.
187    let filled = bucket.clone();
188    let overflow = {
189        let mut inventory = player.inventory.lock();
190        inventory.apply_filled_result(hand, bucket, player.has_infinite_materials(), false)
191    };
192    if !overflow.is_empty() {
193        let _ = player.drop_item(overflow, false, false);
194    }
195
196    triggers::item::filled_bucket(player, &filled);
197    pickup_entity.drop_leash();
198    pickup_entity.set_removed(RemovalReason::Discarded);
199    Some(InteractionResult::Success)
200}
