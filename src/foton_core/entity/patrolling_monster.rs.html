1//! Monsters that march a long line across the world rather than stand around.
2//!
3//! Vanilla parity: `PatrollingMonster`. An illager patrol is the only reason a
4//! pillager appears hundreds of blocks from anything: the group picks a point
5//! up to five hundred blocks away, walks to it, and picks another. The leader
6//! carries the ominous banner and drags the rest along, which is also the seam
7//! raids hang off -- killing a banner-carrying captain is what gives a player
8//! Bad Omen.
9
10use foton_utils::BlockPos;
11use foton_utils::locks::SyncMutex;
12use simdnbt::borrow::NbtCompound as BorrowedNbtCompoundView;
13use simdnbt::owned::{NbtCompound, NbtTag};
14
15use crate::entity::Mob;

17/// NBT key vanilla stores the patrol destination under.
18pub const TAG_PATROL_TARGET: &str = "patrol_target";
19/// NBT key vanilla stores the leader flag under.
20pub const TAG_PATROL_LEADER: &str = "PatrolLeader";
21/// NBT key vanilla stores the patrolling flag under.
22pub const TAG_PATROLLING: &str = "Patrolling";
23
24/// Chance a patroller that spawned on its own turns out to be the captain.
25///
26/// Vanilla parity: the `nextFloat() < 0.06F` of
27/// `PatrollingMonster.finalizeSpawn`.
28pub const PATROL_LEADER_SPAWN_CHANCE: f32 = 0.06;
29
30/// How far a patroller throws its next waypoint, in each direction.
31///
32/// Vanilla parity: the `-500 + random.nextInt(1000)` of `findPatrolTarget`.
33const PATROL_TARGET_SPREAD: i32 = 500;
34
35/// Squared distance past which even a patrol is allowed to despawn.
36///
37/// Vanilla parity: the `distSqr > 16384.0` of `removeWhenFarAway`.
38const PATROL_PERSISTENCE_DISTANCE_SQR: f64 = 16_384.0;
39
40/// Where a patroller is going and whether it leads.
41///
42/// Vanilla keeps these three on the mob; Foton groups them so an entity holds
43/// one field, the way it holds a [`crate::entity::MobBase`].
44#[derive(Debug)]
45pub struct PatrolState {
46    /// The point this mob is walking towards, if it has one.
47    target: SyncMutex<Option<BlockPos>>,
48    /// Whether this mob is the banner-carrying captain.
49    leader: SyncMutex<bool>,
50    /// Whether this mob is on patrol at all.
51    patrolling: SyncMutex<bool>,
52}
53
54impl PatrolState {
55    /// Creates the state of a mob that is not patrolling.
56    #[must_use]
57    pub const fn new() -> Self {
58        Self {
59            target: SyncMutex::new(None),
60            leader: SyncMutex::new(false),
61            patrolling: SyncMutex::new(false),
62        }
63    }
64}
65
66impl Default for PatrolState {
67    fn default() -> Self {
68        Self::new()
69    }
70}

72/// A monster that walks a patrol.
73///
74/// Vanilla parity: the `PatrollingMonster` class.
75pub trait PatrollingMonster: Mob {
76    /// Returns this mob's patrol state.
77    fn patrol_state(&self) -> &PatrolState;
78
79    /// Returns whether this mob may carry the ominous banner.
80    ///
81    /// Vanilla parity: `canBeLeader`, which only the ravager answers no to.
82    fn can_be_leader(&self) -> bool {
83        true
84    }
85
86    /// Returns whether this mob may be swept into a passing patrol.
87    ///
88    /// Vanilla parity: `canJoinPatrol`.
89    fn can_join_patrol(&self) -> bool {
90        true
91    }
92
93    /// Returns the point this mob is walking towards.
94    fn patrol_target(&self) -> Option<BlockPos> {
95        *self.patrol_state().target.lock()
96    }
97
98    /// Returns whether this mob has somewhere to walk to.
99    fn has_patrol_target(&self) -> bool {
100        self.patrol_target().is_some()
101    }
102
103    /// Sends this mob to `target`, putting it on patrol.
104    ///
105    /// Vanilla parity: `setPatrolTarget`, which also flips `patrolling`.
106    fn set_patrol_target(&self, target: BlockPos) {
107        *self.patrol_state().target.lock() = Some(target);
108        *self.patrol_state().patrolling.lock() = true;
109    }
110
111    /// Picks a fresh waypoint up to five hundred blocks away.
112    ///
113    /// Vanilla parity: `findPatrolTarget`.
114    fn find_patrol_target(&self) {
115        let position = self.block_position();
116        let target = position.offset(
117            rand::random_range(-PATROL_TARGET_SPREAD..PATROL_TARGET_SPREAD),
118            0,
119            rand::random_range(-PATROL_TARGET_SPREAD..PATROL_TARGET_SPREAD),
120        );
121        self.set_patrol_target(target);
122    }
123
124    /// Returns whether this mob carries the ominous banner.
125    fn is_patrol_leader(&self) -> bool {
126        *self.patrol_state().leader.lock()
127    }
128
129    /// Makes this mob the captain, putting it on patrol.
130    ///
131    /// Vanilla parity: `setPatrolLeader`.
132    fn set_patrol_leader(&self, leader: bool) {
133        *self.patrol_state().leader.lock() = leader;
134        *self.patrol_state().patrolling.lock() = true;
135    }
136
137    /// Returns whether this mob is on patrol.
138    fn is_patrolling(&self) -> bool {
139        *self.patrol_state().patrolling.lock()
140    }
141
142    /// Sets whether this mob is on patrol.
143    fn set_patrolling(&self, patrolling: bool) {
144        *self.patrol_state().patrolling.lock() = patrolling;
145    }
146
147    /// Returns vanilla `PatrollingMonster.removeWhenFarAway`.
148    ///
149    /// A patrol is kept loaded so it can actually arrive somewhere, but not
150    /// past the distance at which the chunks are gone anyway.
151    fn remove_when_far_away_patrolling(&self, dist_sqr: f64) -> bool {
152        !self.is_patrolling() || dist_sqr > PATROL_PERSISTENCE_DISTANCE_SQR
153    }
154}

156/// Writes the patrol state the way vanilla does.
157///
158/// Vanilla parity: `PatrollingMonster.addAdditionalSaveData`.
159pub fn write_patrol_state(mob: &dyn PatrollingMonster, nbt: &mut NbtCompound) {
160    if let Some(target) = mob.patrol_target() {
161        nbt.insert(
162            TAG_PATROL_TARGET,
163            NbtTag::IntArray(vec![target.x(), target.y(), target.z()]),
164        );
165    }
166    nbt.insert(TAG_PATROL_LEADER, i8::from(mob.is_patrol_leader()));
167    nbt.insert(TAG_PATROLLING, i8::from(mob.is_patrolling()));
168}
169
170/// Reads the patrol state the way vanilla does.
171///
172/// Vanilla parity: `PatrollingMonster.readAdditionalSaveData`. The two flags go
173/// straight into the state rather than through the setters, because vanilla's
174/// setters also force `patrolling` on and a saved idle captain would come back
175/// walking.
176pub fn read_patrol_state(mob: &dyn PatrollingMonster, nbt: BorrowedNbtCompoundView<'_, '_>) {
177    let target = nbt
178        .int_array(TAG_PATROL_TARGET)
179        .filter(|position| position.len() == 3)
180        .map(|position| BlockPos::new(position[0], position[1], position[2]));
181    *mob.patrol_state().target.lock() = target;
182    *mob.patrol_state().leader.lock() = nbt.byte(TAG_PATROL_LEADER).is_some_and(|value| value != 0);
183    *mob.patrol_state().patrolling.lock() =
184        nbt.byte(TAG_PATROLLING).is_some_and(|value| value != 0);
185}
