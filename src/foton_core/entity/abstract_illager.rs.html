1//! The four illagers, and what they have in common.
2//!
3//! Vanilla parity: `AbstractIllager`. Very little of the class is behavior --
4//! it adds one arm pose, one alliance rule and one door goal on top of
5//! [`crate::entity::Raider`] -- but the alliance rule is what keeps a
6//! pillager's bolt from killing the vindicator standing in front of it, and the
7//! arm pose is the whole of what a player sees an illager doing.
8
9use foton_registry::{REGISTRY, TaggedRegistryExt as _, vanilla_entity_type_tags::EntityTypeTag};
10
11use crate::entity::Entity;
12use crate::entity::raider::Raider;
13
14/// What an illager is doing with its arms.
15///
16/// Vanilla parity: `AbstractIllager.IllagerArmPose`. Vanilla never sends this:
17/// the client derives it from the synced flags every tick and picks a model
18/// pose. Foton keeps the derivation server-side anyway, because it is the one
19/// place the meaning of those flags is written down, and because a mob whose
20/// pose says `CROSSBOW_CHARGE` while nothing is charging is a bug worth
21/// catching in a test rather than in a screenshot.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum IllagerArmPose {
24    /// Arms folded: the idle pose.
25    Crossed,
26    /// Mid-swing.
27    Attacking,
28    /// Hands raised, casting.
29    Spellcasting,
30    /// Drawing a bow.
31    BowAndArrow,
32    /// Holding a loaded crossbow.
33    CrossbowHold,
34    /// Winding a crossbow.
35    CrossbowCharge,
36    /// Jumping over a fallen village.
37    Celebrating,
38    /// Arms down: the pose a pillager without a crossbow falls back to.
39    Neutral,
40}
41
42/// An illager.
43///
44/// Vanilla parity: the `AbstractIllager` class.
45pub trait AbstractIllager: Raider {
46    /// Returns what this illager is doing with its arms.
47    ///
48    /// Vanilla parity: `getArmPose`, which defaults to folded arms.
49    fn arm_pose(&self) -> IllagerArmPose {
50        IllagerArmPose::Crossed
51    }
52
53    /// Returns vanilla `AbstractIllager.considersEntityAsAlly`.
54    ///
55    /// The tag holds the four illagers, so a pillager, a vindicator, an evoker
56    /// and an illusioner never target one another. Vanilla also requires both
57    /// entities to be teamless, which is free here: Foton puts no entity on a
58    /// scoreboard team.
59    ///
60    /// Vanilla's `canAttack` override -- refusing to hit a baby villager -- is
61    /// absent with the villagers it names.
62    fn considers_entity_as_ally_illager(&self, other: &dyn Entity) -> bool {
63        REGISTRY
64            .entity_types
65            .is_in_tag(other.entity_type(), &EntityTypeTag::ILLAGER_FRIENDS)
66    }
67}
