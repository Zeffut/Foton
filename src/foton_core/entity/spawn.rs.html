6/// Vanilla `EntitySpawnReason`.
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8pub enum EntitySpawnReason {
9    /// The periodic mob spawning that fills loaded chunks.
10    Natural,
11    /// Placed while the chunk was being generated, before any player saw it.
12    ChunkGeneration,
13    /// A monster spawner block.
14    Spawner,
15    /// Placed as part of a structure, such as a pillager outpost's guards.
16    Structure,
17    /// Born from two parents.
18    Breeding,
19    /// Summoned by another mob -- a witch's cat, an evoker's vexes.
20    MobSummoned,
21    /// Spawned riding another mob, such as a chicken jockey.
22    Jockey,
23    /// Part of a scripted event, such as a raid wave.
24    Event,
25    /// The result of one mob turning into another. See
26    /// [`crate::entity::conversion`].
27    Conversion,
28    /// Called in by a zombie that took damage.
29    Reinforcement,
30    /// Released by a trigger, such as a trial spawner's projectile.
31    Triggered,
32    /// Poured out of a bucket.
33    Bucket,
34    /// Placed by a spawn egg.
35    SpawnItemUse,
36    /// Created by `/summon` or another command.
37    Command,
38    /// Fired out of a dispenser.
39    Dispenser,
40    /// A pillager patrol.
41    Patrol,
42    /// A trial spawner.
43    TrialSpawner,
44    /// Read back from disk rather than newly created.
45    Load,
46    /// Re-created on the far side of a portal.
47    DimensionTravel,
48}
49
50impl EntitySpawnReason {
51    /// Whether a spawner block produced this mob.
52    ///
53    /// Vanilla parity: `EntitySpawnReason.isSpawner`. Both spawner kinds
54    /// count, which is what exempts them from the spawn-position checks a
55    /// natural spawn has to pass.
56    #[must_use]
57    pub const fn is_spawner(self) -> bool {
58        matches!(self, Self::Spawner | Self::TrialSpawner)
59    }
60
61    /// Whether this reason lets a mob appear regardless of light level.
62    ///
63    /// Vanilla parity: `EntitySpawnReason.ignoresLightRequirements` -- only a
64    /// trial spawner, which is why an ordinary spawner still respects light.
65    #[must_use]
66    pub const fn ignores_light_requirements(self) -> bool {
67        matches!(self, Self::TrialSpawner)
68    }
69}

71#[derive(Debug, Clone, Copy, PartialEq)]
72pub enum SpawnGroupData {
73    AgeableMob(AgeableMobGroupData),
74    /// Vanilla `Rabbit.RabbitGroupData`, which is an `AgeableMobGroupData` that
75    /// also carries the variant every rabbit of the group is born with.
76    Rabbit(RabbitGroupData),
77    /// Vanilla `TropicalFish.TropicalFishGroupData`, which carries the variant
78    /// a shoal shares. Vanilla derives it from
79    /// `AbstractSchoolingFish.SchoolSpawnGroupData` and so also carries the
80    /// school leader; Foton has no schooling fish, so that half is absent.
81    TropicalFish(TropicalFishGroupData),
82    /// Vanilla `Horse.HorseGroupData`, which carries the coat every horse of a
83    /// herd is born with.
84    Horse(HorseGroupData),
85    /// Vanilla `Llama.LlamaGroupData`, the same idea for a llama herd.
86    Llama(LlamaGroupData),
87    /// Vanilla `Axolotl.AxolotlGroupData`, which carries the two colors a
88    /// cluster draws from rather than one shared color.
89    Axolotl(AxolotlGroupData),
90}
91
92impl SpawnGroupData {
93    /// Returns the ageable layer, for the kinds that extend one.
94    #[must_use]
95    pub const fn ageable(&self) -> Option<&AgeableMobGroupData> {
96        match self {
97            Self::AgeableMob(group_data) => Some(group_data),
98            Self::Rabbit(group_data) => Some(&group_data.ageable),
99            Self::Horse(group_data) => Some(&group_data.ageable),
100            Self::Llama(group_data) => Some(&group_data.ageable),
101            Self::Axolotl(group_data) => Some(&group_data.ageable),
102            Self::TropicalFish(_) => None,
103        }
104    }
105
106    /// Returns the ageable layer for mutation.
107    #[must_use]
108    pub const fn ageable_mut(&mut self) -> Option<&mut AgeableMobGroupData> {
109        match self {
110            Self::AgeableMob(group_data) => Some(group_data),
111            Self::Rabbit(group_data) => Some(&mut group_data.ageable),
112            Self::Horse(group_data) => Some(&mut group_data.ageable),
113            Self::Llama(group_data) => Some(&mut group_data.ageable),
114            Self::Axolotl(group_data) => Some(&mut group_data.ageable),
115            Self::TropicalFish(_) => None,
116        }
117    }
118}

171/// Vanilla `Axolotl.AxolotlGroupData`.
172#[derive(Debug, Clone, Copy, PartialEq)]
173pub struct AxolotlGroupData {
174    ageable: AgeableMobGroupData,
175    types: [AxolotlVariant; 2],
176}
177
178impl AxolotlGroupData {
179    /// Creates group data for a cluster of axolotls.
180    ///
181    /// Vanilla parity: `AxolotlGroupData(Variant...)` calls `super(false)`, so
182    /// the cluster never rolls the shared baby chance -- the axolotl decides
183    /// that for itself from the group size instead.
184    #[must_use]
185    pub const fn new(types: [AxolotlVariant; 2]) -> Self {
186        Self {
187            ageable: AgeableMobGroupData::with_should_spawn_baby(false),
188            types,
189        }
190    }
191
192    /// How many axolotls of the cluster have spawned already.
193    ///
194    /// Vanilla parity: the inherited `AgeableMobGroupData.getGroupSize`, which
195    /// the axolotl reads itself rather than leaving to the shared baby roll.
196    #[must_use]
197    pub const fn group_size(self) -> i32 {
198        self.ageable.group_size()
199    }
200
201    /// Picks one of the two colors the cluster draws from.
202    ///
203    /// Vanilla parity: `AxolotlGroupData.getVariant`.
204    #[must_use]
205    pub fn variant(self, pick: impl FnOnce(usize) -> usize) -> AxolotlVariant {
206        self.types[pick(self.types.len()) % self.types.len()]
207    }
208}

257#[derive(Debug, Clone, Copy, PartialEq)]
258pub struct AgeableMobGroupData {
259    group_size: i32,
260    should_spawn_baby: bool,
261    baby_spawn_chance: f32,
262}
263
264impl AgeableMobGroupData {
265    pub const DEFAULT_BABY_SPAWN_CHANCE: f32 = 0.05;
266
267    #[must_use]
268    pub const fn new(should_spawn_baby: bool, baby_spawn_chance: f32) -> Self {
269        Self {
270            group_size: 0,
271            should_spawn_baby,
272            baby_spawn_chance,
273        }
274    }
275
276    #[must_use]
277    pub const fn with_should_spawn_baby(should_spawn_baby: bool) -> Self {
278        Self::new(should_spawn_baby, Self::DEFAULT_BABY_SPAWN_CHANCE)
279    }
280
281    #[must_use]
282    pub const fn with_baby_spawn_chance(baby_spawn_chance: f32) -> Self {
283        Self::new(true, baby_spawn_chance)
284    }
285
286    #[must_use]
287    pub const fn group_size(self) -> i32 {
288        self.group_size
289    }
290
291    #[must_use]
292    pub const fn should_spawn_baby(self) -> bool {
293        self.should_spawn_baby
294    }
295
296    #[must_use]
297    pub const fn baby_spawn_chance(self) -> f32 {
298        self.baby_spawn_chance
299    }
300
301    pub const fn increase_group_size_by_one(&mut self) {
302        self.group_size += 1;
303    }
304
305    #[must_use]
306    pub const fn needs_baby_spawn_roll(self) -> bool {
307        self.should_spawn_baby && self.group_size > 0
308    }
309
310    pub fn finalize_ageable_spawn(&mut self, baby_roll: impl FnOnce() -> f32) -> bool {
311        let spawn_baby = self.needs_baby_spawn_roll() && baby_roll() <= self.baby_spawn_chance;
312        self.increase_group_size_by_one();
313        spawn_baby
314    }
315}

317#[cfg(test)]
318mod tests {
319    use super::AgeableMobGroupData;
320
321    #[test]
322    fn ageable_group_data_increments_before_later_baby_rolls_can_apply() {
323        let mut group_data = AgeableMobGroupData::with_should_spawn_baby(true);
324
325        assert!(!group_data.finalize_ageable_spawn(|| {
326            panic!("first group member should not roll for baby spawn")
327        }));
328        assert_eq!(group_data.group_size(), 1);
329
330        assert!(group_data.finalize_ageable_spawn(|| 0.05));
331        assert_eq!(group_data.group_size(), 2);
332    }
333
334    #[test]
335    fn ageable_group_data_can_disable_baby_spawns() {
336        let mut group_data = AgeableMobGroupData::with_should_spawn_baby(false);
337
338        assert!(
339            !group_data
340                .finalize_ageable_spawn(|| { panic!("disabled baby spawning should not roll") })
341        );
342        assert!(
343            !group_data
344                .finalize_ageable_spawn(|| { panic!("disabled baby spawning should not roll") })
345        );
346        assert_eq!(group_data.group_size(), 2);
347    }
