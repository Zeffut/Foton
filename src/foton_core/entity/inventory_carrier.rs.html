1//! Mobs that carry a container of their own.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.npc.InventoryCarrier`. The
4//! piglin and the villager already carried one each with their own copy of the
5//! save code; this is the shared interface behind them, and it is what lets a
6//! brain behavior reach an inventory whose shape it does not know.
7
8use std::sync::Arc;

10use foton_protocol::packets::game::CTakeItemEntity;
11use foton_registry::item_stack::ItemStack;
12use foton_utils::ChunkPos;
13use foton_utils::locks::SyncMutex;
14use simdnbt::borrow::NbtCompound as BorrowedNbtCompoundView;
15use simdnbt::owned::{NbtCompound, NbtList, NbtTag};
16
17use crate::entity::entities::ItemEntity;
18use crate::entity::{Mob, RemovalReason, SharedEntity};
19use crate::inventory::container::{Container as _, SimpleContainer};
20use crate::world::World;

22/// The NBT key a carried inventory is stored under.
23///
24/// Vanilla parity: `InventoryCarrier.TAG_INVENTORY`.
25pub const TAG_INVENTORY: &str = "Inventory";
26
27/// A mob with a container of its own.
28///
29/// Vanilla parity: the `InventoryCarrier` interface.
30pub trait InventoryCarrier: Mob {
31    /// Vanilla parity: `InventoryCarrier.getInventory`.
32    fn carried_inventory(&self) -> &SyncMutex<SimpleContainer>;
33}
34
35/// Writes the carried inventory.
36///
37/// Vanilla parity: `InventoryCarrier.writeInventoryToTag`.
38pub fn save_inventory(inventory: &SimpleContainer, nbt: &mut NbtCompound) {
39    let items: Vec<NbtCompound> = inventory
40        .items()
41        .iter()
42        .filter(|item| !item.is_empty())
43        .filter_map(|item| match item.to_nbt_tag_ref() {
44            NbtTag::Compound(compound) => Some(compound),
45            _ => None,
46        })
47        .collect();
48    nbt.insert(TAG_INVENTORY, NbtList::Compound(items));
49}
50
51/// Reads the carried inventory back.
52///
53/// Vanilla parity: `InventoryCarrier.readInventoryFromTag`.
54pub fn load_inventory(inventory: &mut SimpleContainer, nbt: BorrowedNbtCompoundView<'_, '_>) {
55    let Some(list) = nbt.list(TAG_INVENTORY).and_then(|list| list.compounds()) else {
56        return;
57    };
58    for compound in list {
59        let Some(mut item) = ItemStack::from_borrowed_compound(&compound) else {
60            continue;
61        };
62        inventory.add(&mut item);
63    }
64}

66/// Whether `stack` would fit in this container at all.
67///
68/// Vanilla parity: `SimpleContainer.canAddItem`, which asks only whether some
69/// slot could take part of it -- an item that half fits still counts.
70#[must_use]
71pub fn can_add_item(inventory: &SimpleContainer, stack: &ItemStack) -> bool {
72    inventory.items().iter().any(|slot| {
73        slot.is_empty()
74            || (ItemStack::is_same_item_same_components(slot, stack)
75                && slot.count() < slot.max_stack_size())
76    })
77}

79/// Takes a dropped item into the carrier's own container.
80///
81/// Vanilla parity: the static `InventoryCarrier.pickUpItem`. The count is read
82/// before the add so the pickup animation shows how many actually went in, and
83/// an item that only half fits is left on the ground with the remainder.
84pub fn pick_up_item(
85    world: &Arc<World>,
86    carrier: &dyn InventoryCarrier,
87    item_entity: &SharedEntity,
88) {
89    use foton_utils::Downcast as _;
90
91    let Some(item) = item_entity.downcast_ref::<ItemEntity>() else {
92        return;
93    };
94    let stack = item.get_item();
95    if !carrier.wants_to_pick_up(world, &stack) {
96        return;
97    }
98
99    let mut remainder = stack.copy_with_count(stack.count());
100    let count = remainder.count();
101    {
102        let mut inventory = carrier.carried_inventory().lock();
103        if !can_add_item(&inventory, &remainder) {
104            return;
105        }
106        inventory.add(&mut remainder);
107    }
108
109    // Vanilla parity: `mob.take(itemEntity, count - remainder.getCount())`,
110    // which is the pickup animation every client draws.
111    let taken = count - remainder.count();
112    if taken > 0 {
113        world.broadcast_to_nearby(
114            ChunkPos::from_entity_pos(item_entity.position()),
115            CTakeItemEntity::new(item_entity.id(), carrier.id(), taken),
116            None,
117        );
118    }
119
120    if remainder.is_empty() {
121        item_entity.set_removed(RemovalReason::Discarded);
122    } else {
123        item.set_item(remainder);
124    }
125}
