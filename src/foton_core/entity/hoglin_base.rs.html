45/// Hits `target` for a randomized share of the attacker's damage, then launches it.
46///
47/// Vanilla parity: `HoglinBase.hurtAndThrowTarget`. The damage roll is half the
48/// attribute plus up to the whole of it, so an adult hoglin's six hits for
49/// between three and nine; a baby's is not rolled at all, and a baby never
50/// throws.
51pub fn hurt_and_throw_target(
52    world: &World,
53    body: &dyn LivingEntity,
54    target: &SharedEntity,
55) -> bool {
56    let attack_damage = body
57        .attributes()
58        .lock()
59        .required_value(vanilla_attributes::ATTACK_DAMAGE) as f32;
60
61    #[expect(
62        clippy::cast_possible_truncation,
63        reason = "vanilla rolls `nextInt((int)attackDamage)`, truncating the same way"
64    )]
65    let whole_damage = attack_damage as i32;
66    let actual_damage = if !body.is_baby() && whole_damage > 0 {
67        #[expect(
68            clippy::cast_precision_loss,
69            reason = "an attack damage attribute is far below the f32 integer limit"
70        )]
71        let roll = rand::random_range(0..whole_damage) as f32;
72        attack_damage / 2.0 + roll
73    } else {
74        attack_damage
75    };
76
77    let damage_source = DamageSource::environment(&vanilla_damage_types::MOB_ATTACK)
78        .with_causing_entity(body.id())
79        .with_direct_entity(body.id());
80    let was_hurt = target.hurt(world, &damage_source, actual_damage);
81    if was_hurt && !body.is_baby() {
82        // Vanilla also runs `EnchantmentHelper.doPostAttackEffects` here. A
83        // hoglin has no weapon and no enchantments, so there is nothing for it
84        // to run; a zoglin is the same.
85        throw_target(body, target);
86    }
87    was_hurt
88}

90/// Launches `target` away from `body`.
91///
92/// Vanilla parity: `HoglinBase.throwTarget`. Vanilla feeds `nextInt(21) - 10`
93/// straight into `Vec3.yRot`, which takes **radians**, so the push is swung by
94/// up to ten radians rather than ten degrees -- effectively a random direction.
95/// It is ported as written, because that scatter is what the hit looks like in
96/// game.
97pub fn throw_target(body: &dyn LivingEntity, target: &SharedEntity) {
98    let knockback_power = body
99        .attributes()
100        .lock()
101        .get_value(vanilla_attributes::ATTACK_KNOCKBACK)
102        .unwrap_or(0.0);
103    let knockback_resistance = target
104        .as_living_entity()
105        .and_then(|living| {
106            living
107                .attributes()
108                .lock()
109                .get_value(vanilla_attributes::KNOCKBACK_RESISTANCE)
110        })
111        .unwrap_or(0.0);
112    let effective_power = knockback_power - knockback_resistance;
113    if effective_power <= 0.0 {
114        return;
115    }
116
117    let body_position = body.position();
118    let target_position = target.position();
119    let horizontal = DVec3::new(
120        target_position.x - body_position.x,
121        0.0,
122        target_position.z - body_position.z,
123    );
124    // Vanilla parity: `Vec3.normalize` returns `ZERO` rather than refusing when
125    // the two are stacked, and `throwTarget` carries on -- so a hoglin standing
126    // exactly on its target still launches it straight up.
127    let direction = horizontal.normalize_or_zero();
128
129    let push_angle = f64::from(rand::random_range(0..21) - 10);
130    let horizontal_scale = effective_power * f64::from(0.5f32.mul_add(rand::random::<f32>(), 0.2));
131    let scaled = direction * horizontal_scale;
132    let (sin, cos) = push_angle.sin_cos();
133    let push = DVec3::new(
134        scaled.x * cos + scaled.z * sin,
135        0.0,
136        scaled.z * cos - scaled.x * sin,
137    );
138    let vertical_scale = effective_power * f64::from(rand::random::<f32>()) * 0.5;
139
140    target.push_impulse(DVec3::new(push.x, vertical_scale, push.z));
141    target.mark_hurt();
142}
