1//! Shared vanilla `AbstractHorse` state and hooks.
2//!
3//! Vanilla parity: `AbstractHorse`. Everything a horse-shaped mob does that is
4//! not its coat lives here: the saddle and armor slots, the temper that turns
5//! a bucking wild horse into a tame one, the rearing, the steering seam a rider
6//! drives, and the NBT round trip that keeps all of it across a save.
7//!
8//! Vanilla makes this class an `OwnableEntity` rather than a `TamableAnimal`:
9//! a horse has an owner and a tame flag, but no sit order and no owner-following,
10//! so it deliberately does not sit on Foton's [`TamableAnimal`] layer.
11//!
12//! [`TamableAnimal`]: crate::entity::TamableAnimal

221/// Crosses two parent attribute values into a foal's.
222///
223/// Vanilla parity: `AbstractHorse.createOffspringAttribute`. The average of the
224/// parents, spread over their gap plus a margin, is why two good horses breed a
225/// better one more often than not without ever guaranteeing it.
226pub(crate) fn create_offspring_attribute(
227    own_value: f64,
228    partner_value: f64,
229    range_min: f64,
230    range_max: f64,
231    quality_roll: &mut dyn FnMut() -> f64,
232) -> f64 {
233    debug_assert!(range_max > range_min, "incorrect range for an attribute");
234
235    let own_value = own_value.clamp(range_min, range_max);
236    let partner_value = partner_value.clamp(range_min, range_max);
237    let margin = BREEDING_CROSS_FACTOR * (range_max - range_min);
238    let range = (own_value - partner_value).abs() + margin * 2.0;
239    let average = f64::midpoint(own_value, partner_value);
240    let baby_quality = (quality_roll() + quality_roll() + quality_roll()) / 3.0 - 0.5;
241    let new_value = range.mul_add(baby_quality, average);
242
243    if new_value > range_max {
244        return range_max - (new_value - range_max);
245    }
246    if new_value < range_min {
247        return range_min + (range_min - new_value);
248    }
249    new_value
250}

252/// Vanilla `SoundType` values that make a horse clop on wood.
253///
254/// Vanilla parity: `AbstractHorse.isWoodSoundType`. Vanilla compares the sound
255/// type by reference; `SoundType` is a plain value in Foton's generated registry,
256/// so the five of them are compared field by field instead.
257const WOOD_SOUND_TYPES: [SoundType; 5] = [
258    sound_types::WOOD,
259    sound_types::NETHER_WOOD,
260    sound_types::STEM,
261    sound_types::CHERRY_WOOD,
262    sound_types::BAMBOO_WOOD,
263];
264
265fn is_same_sound_type(left: SoundType, right: SoundType) -> bool {
266    ptr::eq(left.break_sound, right.break_sound)
267        && ptr::eq(left.step_sound, right.step_sound)
268        && ptr::eq(left.place_sound, right.place_sound)
269        && ptr::eq(left.hit_sound, right.hit_sound)
270        && ptr::eq(left.fall_sound, right.fall_sound)
271}
272
273fn is_wood_sound_type(sound_type: SoundType) -> bool {
274    WOOD_SOUND_TYPES
275        .into_iter()
276        .any(|wood| is_same_sound_type(sound_type, wood))
277}

315/// Runtime fields shared by every vanilla horse-shaped mob.
316pub struct AbstractHorseBase {
317    state: SyncMutex<AbstractHorseState>,
318    /// Vanilla parity: `AbstractHorse.owner`, an `EntityReference<LivingEntity>`.
319    /// Foton stores the UUID it resolves from, the way [`crate::entity::TamableAnimal`] does.
320    owner: SyncMutex<Option<Uuid>>,
321    /// Vanilla parity: `AbstractHorse.inventory`. Held behind its own handle so
322    /// an open horse screen keeps working across an unrelated state change, and
323    /// so `hasInventoryChanged` can compare identity after a resize.
324    inventory: SyncMutex<Shared<SimpleContainer>>,
325}

327impl fmt::Debug for AbstractHorseBase {
328    /// `SimpleContainer` is not `Debug`, so the inventory is summarized by size.
329    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
330        f.debug_struct("AbstractHorseBase")
331            .field("state", &*self.state.lock())
332            .field("owner", &*self.owner.lock())
333            .field(
334                "inventory_size",
335                &self.inventory.lock().lock().items().len(),
336            )
337            .finish()
338    }
339}

360    /// Replaces the inventory with one of `size` slots, carrying items across.
361    ///
362    /// Vanilla parity: `AbstractHorse.createInventory`.
363    pub fn create_inventory(&self, size: usize) {
364        let mut slot = self.inventory.lock();
365        let carried: Vec<ItemStack> = {
366            let old = slot.lock();
367            old.items()
368                .iter()
369                .take(size)
370                .map(|item| item.copy_with_count(item.count()))
371                .collect()
372        };
373
374        let mut items = vec![ItemStack::empty(); size];
375        for (index, item) in carried.into_iter().enumerate() {
376            items[index] = item;
377        }
378        *slot = SimpleContainer::from_items(items).into_shared();
379    }

657    /// Applies vanilla `AbstractHorse.handleEating`.
658    fn handle_eating(&self, player: &Player, item_stack: &ItemStack) -> bool {
659        let mut item_used = false;
660        let (heal, age_up, temper) = if item_stack.is(&vanilla_items::WHEAT) {
661            (2.0, 20, 3)
662        } else if item_stack.is(&vanilla_items::SUGAR) {
663            (1.0, 30, 3)
664        } else if item_stack.is(&vanilla_items::HAY_BLOCK) {
665            (20.0, 180, 0)
666        } else if item_stack.is(&vanilla_items::APPLE) {
667            (3.0, 60, 3)
668        } else if item_stack.is(&vanilla_items::RED_MUSHROOM) {
669            (3.0, 0, 3)
670        } else if item_stack.is(&vanilla_items::CARROT) {
671            (3.0, 60, 3)
672        } else if item_stack.is(&vanilla_items::GOLDEN_CARROT) {
673            if self.is_tamed() && self.get_age() == 0 && !self.is_in_love() {
674                item_used = true;
675                self.set_in_love(Some(player));
676            }
677            (4.0, 60, 5)
678        } else if item_stack.is(&vanilla_items::GOLDEN_APPLE)
679            || item_stack.is(&vanilla_items::ENCHANTED_GOLDEN_APPLE)
680        {
681            if self.is_tamed() && self.get_age() == 0 && !self.is_in_love() {
682                item_used = true;
683                self.set_in_love(Some(player));
684            }
685            (10.0, 240, 10)
686        } else {
687            (0.0, 0, 0)
688        };
689
690        let item_used = self.apply_eating_effects(item_used, heal, age_up, temper);
691        if item_used {
692            self.horse_eating_effect();
693            self.game_event(&vanilla_game_events::EAT);
694        }
695        item_used
696    }

698    /// Applies the heal, age-up and temper part shared by every `handleEating`.
699    ///
700    /// Vanilla repeats these three blocks verbatim in `AbstractHorse.handleEating`
701    /// and `Llama.handleEating`; only the table above them differs. `item_used`
702    /// carries in whatever the table already decided, because the temper block
703    /// reads it.
704    fn apply_eating_effects(
705        &self,
706        mut item_used: bool,
707        heal: f32,
708        age_up: i32,
709        temper: i32,
710    ) -> bool {
711        if self.get_health() < self.get_max_health() && heal > 0.0 {
712            self.heal(heal);
713            item_used = true;
714        }
715
716        if AgeableMob::is_baby(self) && age_up > 0 && !self.is_age_locked() {
717            // VANILLA CLIENT-LOCAL: `handleEating` adds the happy-villager particles.
718            self.age_up(age_up, false);
719            item_used = true;
720        }
721
722        if temper > 0 && (item_used || !self.is_tamed()) && self.temper() < self.max_temper() {
723            self.modify_temper(temper);
724            item_used = true;
725        }
726
727        item_used
728    }

784        };

850    fn abstract_horse_cause_fall_damage(
851        &self,
852        fall_distance: f64,
853        damage_modifier: f32,
854        source: &DamageSource,
855    ) -> bool {
856        if fall_distance > 1.0 {

873        self.propagate_fall_to_passengers(fall_distance, damage_modifier, source);
874        self.play_block_fall_sound();

886    /// Applies vanilla `AbstractHorse.tick`'s counters.
887    ///
888    /// The eat and mouth animations are client-side interpolation; the rearing
889    /// one is not, because `getPassengerAttachmentPoint` shifts a rider by it.
890    fn tick_abstract_horse(&self) {
891        {
892            let mut state = self.abstract_horse_base().state.lock();
893            if state.mouth_counter > 0 {
894                state.mouth_counter += 1;
895            }
896            if state.tail_counter > 0 {
897                state.tail_counter += 1;
898                if state.tail_counter > TAIL_TICKS {
899                    state.tail_counter = 0;
900                }
901            }
902            if state.sprint_counter > 0 {
903                state.sprint_counter += 1;
904                if state.sprint_counter > SPRINT_TICKS {
905                    state.sprint_counter = 0;
906                }
907            }
908        }
909
910        let close_mouth = {
911            let mut state = self.abstract_horse_base().state.lock();
912            let close = state.mouth_counter > MOUTH_OPEN_TICKS;
913            if close {
914                state.mouth_counter = 0;
915            }
916            close
917        };
918        if close_mouth {
919            self.set_horse_flag(FLAG_OPEN_MOUTH, false);
920        }
921
922        let stop_standing = {
923            let mut state = self.abstract_horse_base().state.lock();
924            state.stand_counter > 0 && {
925                state.stand_counter -= 1;
926                state.stand_counter <= 0
927            }
928        };
929        if stop_standing {
930            self.clear_standing();
931        }
932
933        let standing = self.is_standing();
934        let mut state = self.abstract_horse_base().state.lock();
935        state.stand_anim_o = state.stand_anim;
936        if standing {
937            state.stand_anim += (1.0 - state.stand_anim).mul_add(0.4, 0.05);
938            state.stand_anim = state.stand_anim.min(1.0);
939        } else {
940            state.allow_stand_sliding = false;
941            let anim = state.stand_anim;
942            state.stand_anim += (0.8 * anim * anim).mul_add(anim, -anim).mul_add(0.6, -0.05);
943            state.stand_anim = state.stand_anim.max(0.0);
944        }
945    }

1003    /// The body of [`Self::follow_mommy`], callable from an override.
1004    ///
1005    /// Rust has no `super`, and the llama only wants to add a guard in front.
1006    fn follow_mommy_default(&self, world: &Arc<World>) {
1007        if !self.is_bred() || !AgeableMob::is_baby(self) || self.is_eating() {
1008            return;
1009        }
1010
1011        let mommy_targeting = TargetingConditions::for_non_combat()
1012            .range(MOMMY_SEARCH_RANGE)
1013            .ignore_line_of_sight();
1014        let search_box = self.bounding_box().inflate(MOMMY_SEARCH_RANGE);
1015        let Some(pathfinder) = self.as_pathfinder_mob() else {
1016            return;
1017        };
1018        let mommy = world.nearest_entity_in_aabb_matching(&search_box, self.position(), |entity| {
1019            let Some(candidate) = entity.as_living_entity() else {
1020                return false;
1021            };
1022            entity
1023                .as_abstract_horse()
1024                .is_some_and(AbstractHorse::is_bred)
1025                && mommy_targeting.test(world.as_ref(), Some(pathfinder), candidate)
1026        });
1027
1028        let Some(mommy) = mommy else {
1029            return;
1030        };
1031        if self.position().distance_squared(mommy.position()) > MOMMY_FOLLOW_DISTANCE_SQR {
1032            pathfinder.move_to_pos_with_reach(mommy.position(), 0, 1.0);
1033        }
1034    }

1246            .state

1323    /// Drops the horse's inventory contents.
1324    ///
1325    /// Vanilla parity: the inventory half of `AbstractHorse.dropEquipment`. The
1326    /// saddle and body armor are equipment, which Foton already drops through
1327    /// `Mob.dropCustomDeathLoot`.
1328    fn drop_abstract_horse_inventory(&self) {
1329        let inventory = self.abstract_horse_base().inventory();
1330        let dropped: Vec<ItemStack> = {
1331            let mut container = inventory.lock();
1332            container
1333                .items_mut()
1334                .iter_mut()
1335                .filter(|item| {
1336                    !item.is_empty()
1337                        && !item.has_enchantment_effect(
1338                            EnchantmentEffectComponent::PreventEquipmentDrop,
1339                        )
1340                })
1341                .map(mem::take)
1342                .collect()
1343        };
1344        for item_stack in dropped {
1345            self.spawn_at_location(item_stack, 0.0);
1346        }
1347    }

1349    /// Returns the rider offset vanilla adds while the horse rears.
1350    ///
1351    /// Vanilla parity: `AbstractHorse.getPassengerAttachmentPoint`.
1352    fn abstract_horse_rearing_rider_offset(&self, scale: f32) -> DVec3 {
1353        let stand_anim_o = f64::from(self.abstract_horse_base().stand_anim_o());
1354        let scale = f64::from(scale);
1355        let offset = DVec3::new(
1356            0.0,
1357            0.15 * stand_anim_o * scale,
1358            -0.7 * stand_anim_o * scale,
1359        );
1360        let yaw = -f64::from(self.rotation().0).to_radians();
1361        let (sin, cos) = yaw.sin_cos();
1362        DVec3::new(
1363            offset.x.mul_add(cos, offset.z * sin),
1364            offset.y,
1365            offset.z.mul_add(cos, -(offset.x * sin)),
1366        )
1367    }
