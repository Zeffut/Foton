1//! Shared vanilla `AbstractNautilus` state and hooks.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.nautilus.AbstractNautilus`.
4//! A nautilus is the game's only tameable mount that lives underwater: it is
5//! tamed with food, saddled, ridden with the mouse rather than with a walk
6//! input, dashes when the rider charges a jump, and keeps its rider breathing
7//! while they sit on it. Both the nautilus and the zombie nautilus are this
8//! class plus a sound table, so everything here is shared.

86/// How long a nautilus waits between dashes.
87///
88/// Vanilla parity: `AbstractNautilus.DASH_COOLDOWN_TICKS`.

95/// The shortest a dash can look, whatever else happens.

97/// Vanilla parity: `AbstractNautilus.DASH_MINIMUM_DURATION_TICKS`.
98const DASH_MINIMUM_DURATION_TICKS: i32 = 5;
99/// How hard a dash pushes while the nautilus is swimming.
100///
101/// Vanilla parity: `AbstractNautilus.DASH_MOMENTUM_IN_WATER`.
102const DASH_MOMENTUM_IN_WATER: f32 = 1.2;
103/// How hard a dash pushes while the nautilus is beached.
104///
105/// Vanilla parity: `AbstractNautilus.DASH_MOMENTUM_ON_LAND`.
106const DASH_MOMENTUM_ON_LAND: f32 = 0.5;
107
108/// How much a rider's pitch turns the nautilus.
109///
110/// Vanilla parity: the `controller.getXRot() * 0.5F` of
111/// `AbstractNautilus.getRiddenRotation`.
112const RIDDEN_PITCH_FACTOR: f32 = 0.5;
113/// How fast the nautilus swings round to the rider's yaw.
114///
115/// Vanilla parity: the `turnSpeed` of `AbstractNautilus.tickRidden`.
116const RIDDEN_TURN_SPEED: f32 = 0.5;
117/// What backing up costs a rider's look direction.
118///
119/// Vanilla parity: the `*= -0.5F` of `AbstractNautilus.getRiddenInput`.
120const RIDDEN_BACKWARDS_FACTOR: f32 = -0.5;

122/// The chance in three that a feeding tames a nautilus.
123///
124/// Vanilla parity: the `random.nextInt(3) == 0` of `AbstractNautilus.tryToTame`.
125const TAME_CHANCE_IN: i32 = 3;
126/// How much of a food's nutrition a nautilus heals for.
127///
128/// Vanilla parity: the `2.0F` of the `feed` call in `AbstractNautilus.mobInteract`.

131///
132/// Vanilla parity: the `1.0F` of the same call.

181impl AbstractNautilusBase {
182    /// Creates nautilus runtime state with an empty inventory.
183    ///
184    /// The real size arrives from [`AbstractNautilus::create_nautilus_inventory`],
185    /// which the constructor calls the way vanilla's does -- the column count is
186    /// the mob's to answer, and the mob does not exist yet here.
187    #[must_use]
188    pub fn new() -> Self {
189        Self {
190            state: SyncMutex::new(AbstractNautilusState::new()),
191            inventory: SyncMutex::new(SimpleContainer::new(0).into_shared()),
192        }
193    }
194
195    /// Returns the current inventory handle.
196    #[must_use]
197    pub fn inventory(&self) -> Shared<SimpleContainer> {
198        Arc::clone(&self.inventory.lock())
199    }
200
201    /// Replaces the inventory with one of `size` slots, carrying items across.
202    ///
203    /// Vanilla parity: `AbstractNautilus.createInventory`.
204    pub fn create_inventory(&self, size: usize) {
205        let mut slot = self.inventory.lock();
206        let carried: Vec<ItemStack> = {
207            let old = slot.lock();
208            old.items()
209                .iter()
210                .take(size)
211                .map(|item| item.copy_with_count(item.count()))
212                .collect()
213        };
214
215        let mut items = vec![ItemStack::empty(); size];
216        for (index, item) in carried.into_iter().enumerate() {
217            items[index] = item;
218        }
219        *slot = SimpleContainer::from_items(items).into_shared();
220    }
221
222    /// Returns vanilla `AbstractNautilus.dashCooldown`.
223    #[must_use]
224    pub fn dash_cooldown(&self) -> i32 {
225        self.state.lock().dash_cooldown
226    }
227
228    fn set_dash_cooldown(&self, dash_cooldown: i32) {
229        self.state.lock().dash_cooldown = dash_cooldown;
230    }
231
232    fn take_player_jump_pending_scale(&self) -> f32 {
233        let mut state = self.state.lock();
234        let pending_scale = state.player_jump_pending_scale;
235        state.player_jump_pending_scale = 0.0;
236        pending_scale
237    }
238
239    fn set_player_jump_pending_scale(&self, scale: f32) {
240        self.state.lock().player_jump_pending_scale = scale;
241    }
242}

252    /// Writes the synchronized `AbstractNautilus.DASH` flag.
253    ///
254    /// Implementations only touch the synced value; the cooldown side effect is

268    /// Applies vanilla `AbstractNautilus.setDashing`.
269    ///
270    /// Vanilla's `onSyncedDataUpdated` arms the cooldown the first time `DASH`
271    /// changes with the clock at zero, which is the only thing that starts the
272    /// server's cooldown when a rider's `handleStartJump` raises the flag.
273    /// Foton has no synced-data change hook, so the same rule lives here: every
274    /// vanilla path that writes `DASH` goes through `setDashing`, so the two are
275    /// equivalent.
276    fn set_dashing(&self, is_dashing: bool) {
277        self.set_dash_flag(is_dashing);
278        let base = self.abstract_nautilus_base();
279        if base.dash_cooldown() == 0 {
280            base.set_dash_cooldown(DASH_COOLDOWN_TICKS);
281        }
282    }

284    /// Returns vanilla `AbstractNautilus.isFood`.
285    ///
286    /// An untamed adult only takes its taming items; everything else it eats is
287    /// the food tag.
288    fn is_nautilus_food(&self, item_stack: &ItemStack) -> bool {
289        let tag = if !self.is_tame() && !AgeableMob::is_baby(self) {
290            &ItemTag::NAUTILUS_TAMING_ITEMS
291        } else {
292            &ItemTag::NAUTILUS_FOOD
293        };
294        REGISTRY.items.is_in_tag(item_stack.item(), tag)
295    }
296
297    /// Applies vanilla `AbstractNautilus.usePlayerItem`, which hands back the
298    /// water bucket a bucket of fish leaves behind.
299    ///
300    /// Rust has no `super`, and this is what `Mob::use_player_item` is
301    /// overridden with, so the `else` branch spells `Mob.usePlayerItem` out
302    /// rather than calling it -- calling it would come straight back here.
303    fn use_nautilus_player_item(&self, player: &Player, hand: InteractionHand) {
304        let is_bucket_food = {
305            let inventory = player.inventory.lock();
306            REGISTRY.items.is_in_tag(
307                inventory.get_item_in_hand(hand).item(),
308                &ItemTag::NAUTILUS_BUCKET_FOOD,
309            )
310        };
311        if !is_bucket_food {
312            player.inventory.lock().shrink_item_in_hand(hand, 1);
313            return;
314        }
315
316        let overflow = {
317            let mut inventory = player.inventory.lock();
318            inventory.apply_filled_result(
319                hand,
320                ItemStack::new(&vanilla_items::WATER_BUCKET),
321                player.has_infinite_materials(),
322                true,
323            )
324        };
325        if !overflow.is_empty() {
326            let _ = player.drop_item(overflow, false, false);
327        }
328    }

340    /// Returns vanilla `AbstractNautilus.getRiddenInput`.
341    ///
342    /// Unlike every land mount, forward is the rider's *look* direction, so a
343    /// nautilus climbs and dives with the mouse.
344    fn nautilus_ridden_input(&self, controller: &Player) -> DVec3 {
345        let input = controller.travel_input();
346        let strafe = input.sideways();
347        if input.forward() == 0.0 {
348            return DVec3::new(f64::from(strafe), 0.0, 0.0);
349        }
350
351        let pitch_radians = controller.rotation().1.to_radians();
352        let mut forward_look = pitch_radians.cos();
353        let mut up_look = -pitch_radians.sin();
354        if input.forward() < 0.0 {
355            forward_look *= RIDDEN_BACKWARDS_FACTOR;
356            up_look *= RIDDEN_BACKWARDS_FACTOR;
357        }
358
359        DVec3::new(
360            f64::from(strafe),
361            f64::from(up_look),
362            f64::from(forward_look),
363        )
364    }

366    /// Applies vanilla `AbstractNautilus.tickRidden`.
367    fn nautilus_tick_ridden(&self, controller: &Player) {
368        let (controller_yaw, controller_pitch) = controller.rotation();
369        let mut yaw = self.rotation().0;
370        let difference = wrap_degrees(controller_yaw - yaw);
371        yaw += difference * RIDDEN_TURN_SPEED;
372        self.set_rotation((yaw, controller_pitch * RIDDEN_PITCH_FACTOR));
373        self.base().set_old_yaw_to_current();
374        self.set_y_body_rot(yaw);
375        self.set_y_head_rot(yaw);
376
377        if !self.is_server_driven_movement() {
378            return;
379        }
380
381        let pending_scale = self
382            .abstract_nautilus_base()
383            .take_player_jump_pending_scale();
384        if pending_scale > 0.0 && !self.is_jumping() {
385            self.execute_riders_jump(pending_scale, controller);
386        }
387    }

389    /// Returns vanilla `AbstractNautilus.getRiddenSpeed`.
390    fn nautilus_ridden_speed(&self) -> f32 {
391        #[expect(
392            clippy::cast_possible_truncation,
393            reason = "a movement-speed attribute, immediately used as a speed"
394        )]
395        let movement_speed = self
396            .attributes()
397            .lock()
398            .required_value(vanilla_attributes::MOVEMENT_SPEED) as f32;
399        if self.is_in_water() {
400            RIDDEN_SPEED_MODIFIER_IN_WATER * movement_speed
401        } else {
402            RIDDEN_SPEED_MODIFIER_ON_LAND * movement_speed
403        }
404    }

406    /// Applies vanilla `AbstractNautilus.travelInWater`, which swims rather
407    /// than sinking and bleeds a tenth of its speed a tick.
408    fn nautilus_travel_in_water(&self, input: DVec3) -> Option<MoveResult> {
409        self.move_relative(self.get_speed(), input);
410        let result = self.move_entity(MoverType::SelfMovement, self.velocity());
411        self.set_velocity(self.velocity() * NAUTILUS_WATER_RESISTANCE);
412        result

429    /// Returns vanilla `AbstractNautilus.getNautilusRestrictionRadius`.
430    fn nautilus_restriction_radius(&self) -> i32 {
431        if !AgeableMob::is_baby(self) && !self.has_item_in_slot(EquipmentSlot::Saddle) {
432            LARGE_RESTRICTION_RADIUS
433        } else {

438    /// Applies vanilla `AbstractNautilus.checkRestriction`, which is what keeps
439    /// a tame nautilus near where its owner left it.
440    fn check_nautilus_restriction(&self) {
441        if self.is_leashed() || self.is_vehicle() || !self.is_tame() {
442            return;
443        }
444
445        let radius = self.nautilus_restriction_radius();
446        let position = self.block_position();
447        let keeps_home = self.has_home()
448            && block_closer_than(
449                self.home_position(),
450                position,
451                f64::from(radius + RESTRICTION_RADIUS_BUFFER),
452            )
453            && radius == self.home_radius();
454        if !keeps_home {
455            self.set_home_to(position, radius);
456        }
457    }

459    /// Applies vanilla `AbstractNautilus.applyEffects`, the rider's air supply.
460    fn apply_nautilus_effects(&self, world: &Arc<World>) {
461        let Some(passenger) = self.first_passenger() else {
462            return;
463        };
464        let Some(player) = passenger.as_player() else {
465            return;
466        };
467
468        let has_effect = player.has_mob_effect(vanilla_mob_effects::BREATH_OF_THE_NAUTILUS);
469        let should_refresh = world.game_time() % EFFECT_REFRESH_RATE == 0;
470        if has_effect && !should_refresh {
471            return;
472        }
473
474        player.add_mob_effect(
475            MobEffectInstance::with_duration(
476                vanilla_mob_effects::BREATH_OF_THE_NAUTILUS,
477                EFFECT_DURATION,
478                0,
479            )
480            .with_ambient(true),
481        );
482    }

484    /// Applies vanilla `AbstractNautilus.tick`.
485    ///
486    /// Vanilla also spawns the bubble trail here; `Level.addParticle` is
487    /// client-local, so the server has nothing to send.
488    fn tick_nautilus(&self) {
489        if let Some(world) = self.level() {
490            self.apply_nautilus_effects(&world);
491        }
492
493        let base = self.abstract_nautilus_base();
494        if self.is_dashing() && base.dash_cooldown() < DASH_FLAG_CLEAR_AT {
495            self.set_dashing(false);
496        }
497
498        let cooldown = base.dash_cooldown();
499        if cooldown > 0 {
500            base.set_dash_cooldown(cooldown - 1);
501            if cooldown - 1 == 0 {
502                self.make_sound(self.dash_ready_sound());
503            }
504        }
505    }

512    /// Applies vanilla `AbstractNautilus.onPlayerJump`.
513    ///
514    /// Vanilla only reaches this from `LocalPlayer`, so it never fires on a
515    /// dedicated server; it is here because a locally simulated nautilus still
516    /// needs the pending scale [`Self::execute_riders_jump`] consumes.
517    fn nautilus_on_player_jump(&self, jump_amount: i32) {
518        if !Mob::is_saddled(self) || self.abstract_nautilus_base().dash_cooldown() > 0 {
519            return;
520        }
521        self.abstract_nautilus_base()
522            .set_player_jump_pending_scale(Entity::player_jump_pending_scale(self, jump_amount));
523    }

525    /// Applies vanilla `AbstractNautilus.executeRidersJump`.
526    fn execute_riders_jump(&self, amount: f32, controller: &Player) {
527        let momentum = if self.is_in_water() {
528            DASH_MOMENTUM_IN_WATER
529        } else {
530            DASH_MOMENTUM_ON_LAND
531        };
532        let scale = f64::from(momentum * amount)
533            * self
534                .attributes()
535                .lock()
536                .required_value(vanilla_attributes::MOVEMENT_SPEED)
537            * f64::from(self.block_speed_factor());
538        self.set_velocity(self.velocity() + controller.look_angle() * scale);
539        self.mark_velocity_sync();
540        self.abstract_nautilus_base()
541            .set_dash_cooldown(DASH_COOLDOWN_TICKS);
542        self.set_dashing(true);
543    }

545    /// Applies vanilla `AbstractNautilus.handleStartJump`.
546    fn nautilus_handle_start_jump(&self) {
547        self.make_sound(self.dash_sound());
548        if let Some(world) = self.level() {
549            world.game_event(
550                &vanilla_game_events::ENTITY_ACTION,
551                self.block_position(),
552                &GameEventContext::new(Some(self.as_entity_event_source()), None),
553            );
554        }

556    }

575    /// Applies vanilla `AbstractNautilus.mobInteract`.
576    fn nautilus_mob_interact(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
577        if AgeableMob::is_baby(self) {
578            return Animal::mob_interact_animal(self, player, hand);
579        }
580
581        if self.is_tame() && player.is_secondary_use_active() {
582            self.open_custom_inventory_screen(player);
583            return InteractionResult::Success;
584        }
585
586        let item_stack = {
587            let inventory = player.inventory.lock();
588            let item_stack = inventory.get_item_in_hand(hand);
589            item_stack.copy_with_count(item_stack.count())
590        };
591        let is_food = self.is_nautilus_food(&item_stack);
592        if !item_stack.is_empty() {
593            if !self.is_tame() && is_food {
594                self.use_nautilus_player_item(player, hand);
595                self.try_to_tame_nautilus(player);
596                return InteractionResult::SuccessServer;
597            }
598
599            if is_food && self.get_health() < self.get_max_health() {
600                self.feed(
601                    player,
602                    hand,
603                    &item_stack,
604                    FEED_HEALING_FACTOR,
605                    FEED_DEFAULT_HEAL,
606                );
607                return InteractionResult::Success;
608            }
609
610            let interaction_result =
611                LivingEntity::interact_living_entity_with_equippable(self, player, hand);
612            if interaction_result.consumes_action() {
613                return interaction_result;
614            }
615        }
616
617        if self.is_tame() && !player.is_secondary_use_active() && !is_food {
618            self.nautilus_do_player_ride(player);
619            return InteractionResult::Success;
620        }
621
622        Animal::mob_interact_animal(self, player, hand)
623    }

625    /// Applies vanilla `AbstractNautilus.tryToTame`, a one-in-three roll.
626    fn try_to_tame_nautilus(&self, player: &Player) {
627        if rand::random_range(0..TAME_CHANCE_IN) == 0 {
628            self.tame(player);
629            self.mob_base().navigation().lock().stop();
630            self.broadcast_entity_event(EntityStatus::TamingSucceeded);
631        } else {
632            self.broadcast_entity_event(EntityStatus::TamingFailed);
633        }
634
635        self.play_eating_sound();
636    }

696    /// Applies vanilla `AbstractNautilus.hurtServer`, whose only addition is
697    /// remembering who did it.
698    fn nautilus_hurt_server(&self, world: &World, source: &DamageSource, amount: f32) -> bool {
699        let was_hurt = self.living_hurt_server(world, source, amount);
700        if !was_hurt {
701            return false;
702        }
703
704        let Some(attacker) = source
705            .causing_entity_id
706            .and_then(|id| world.get_entity_by_id(id))
707        else {
708            return true;
709        };
710        if attacker.as_living_entity().is_some() {
711            set_anger_target(world, self, &attacker);
712        }
713        true
714    }

719    }

722/// Applies vanilla `NautilusAi.setAngerTarget`.
723///
724/// Lives beside the mob rather than in its brain module because both nautilus
725/// brains share it and `AbstractNautilus.hurtServer` is its only caller.
726pub fn set_anger_target<B: AbstractNautilus + ?Sized>(
727    world: &World,
728    body: &B,
729    target: &SharedEntity,
730) {
731    let Some(target_living) = target.as_living_entity() else {
732        return;
733    };
734    let Some(brain) = Mob::brain(body) else {
735        return;
736    };
737    let Some(body_living) = body.as_entity_event_source().as_living_entity() else {
738        return;
739    };
740    if !is_entity_attackable_ignoring_line_of_sight(world, body_living, target_living) {
741        return;
742    }
743
744    brain.erase_memory(memory_module_types::CANT_REACH_WALK_TARGET_SINCE.id());
745    brain.set_memory_with_expiry(
746        memory_module_types::ANGRY_AT,
747        target.uuid(),
748        ANGER_DURATION_TICKS,
749    );
750}

757/// Returns vanilla `AbstractNautilus.checkNautilusSpawnRules`.
758///
759/// A nautilus spawns in the twenty blocks of water just under the surface, with
760/// water below it and water above it.
761#[must_use]
762pub fn check_nautilus_spawn_rules(world: &Arc<World>, pos: BlockPos) -> bool {
763    let sea_level = world.sea_level;
764    let min_spawn_level = sea_level - SPAWN_DEPTH_BELOW_SEA_LEVEL;
765    pos.y() >= min_spawn_level
766        && pos.y() <= sea_level - SPAWN_MIN_DEPTH
767        && world
768            .get_block_state(pos.below())
769            .get_fluid_state()
770            .fluid_id
771            .has_tag(&vanilla_fluid_tags::FluidTag::WATER)
772        && world.get_block_state(pos.above()).get_block() == &vanilla_blocks::WATER
773}

779/// The shallowest a nautilus spawns below sea level.
780///
781/// Vanilla parity: the `seaLevel - 5` of the same check.
782const SPAWN_MIN_DEPTH: i32 = 5;

804/// How long a nautilus waits between fights nobody started.
805///
806/// Vanilla parity: `NautilusAi.TIME_BETWEEN_NON_PLAYER_ATTACKS`.
807const TIME_BETWEEN_NON_PLAYER_ATTACKS: UniformIntProvider = UniformIntProvider {
808    min_inclusive: 2400,
809    max_inclusive: 3600,
810};

812/// Rolls the pause a nautilus takes before it looks for a fight of its own.
813#[must_use]
814pub fn sample_time_between_non_player_attacks() -> i32 {
815    rand::random_range(
816        TIME_BETWEEN_NON_PLAYER_ATTACKS.min_inclusive
817            ..=TIME_BETWEEN_NON_PLAYER_ATTACKS.max_inclusive,
818    )
819}
