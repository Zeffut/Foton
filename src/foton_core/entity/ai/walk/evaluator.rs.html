1//! The contract the A* search needs from whatever decides where a mob may go.
2
3use crate::entity::ai::node::{Node, NodeStore};
4use crate::entity::ai::path::PathfindingContext;
5
6use super::collision::WalkNodeCollision;
7use super::node_evaluator::Neighbors;
8
9/// Decides which positions a mob can occupy and which it can step to.
10///
11/// Vanilla parity: `NodeEvaluator`. The A* search itself is the same whether a
12/// mob walks, swims or flies; only this changes, which is why vanilla has one
13/// `PathFinder` and several evaluators.
14pub trait NodeEvaluator {
15    /// Returns the node the search starts from.
16    ///
17    /// Vanilla parity: `NodeEvaluator.getStart`.
18    fn get_start(&mut self, context: &mut PathfindingContext<'_>) -> i32;
19
20    /// Returns the nodes reachable in one step from `pos_hash`.
21    ///
22    /// Vanilla parity: `NodeEvaluator.getNeighbors`.
23    fn get_neighbors(
24        &mut self,
25        context: &mut PathfindingContext<'_>,
26        collision: &mut dyn WalkNodeCollision,
27        pos_hash: i32,
28    ) -> Neighbors;
29
30    /// Returns the node behind `hash`, if the evaluator has seen it.
31    fn node(&self, hash: i32) -> Option<&Node>;
32
33    /// Returns the node behind `hash` for the search to score.
34    fn node_mut(&mut self, hash: i32) -> Option<&mut Node>;
35
36    /// Returns the node store the open set orders.
37    fn nodes_mut(&mut self) -> &mut NodeStore;
38
39    /// Clears the per-search scoring left on every node.
40    fn reset_search_state(&mut self);
41
42    /// Drops every node, so the next search starts from nothing.
43    ///
44    /// Vanilla parity: `NodeEvaluator.done`.
45    fn clear_nodes(&mut self);
46}
