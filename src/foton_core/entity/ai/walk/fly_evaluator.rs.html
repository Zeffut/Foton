1//! Node evaluation for mobs that fly.
2//!
3//! Vanilla parity: `FlyNodeEvaluator`. It extends the walking evaluator rather
4//! than replacing it: the mob's bounding box is classified the same way, but
5//! open air counts as somewhere to be rather than somewhere to fall through,
6//! and a step may go up or down as freely as sideways. That is the whole
7//! difference between a parrot and a chicken.
8

23/// Extra cost of a node a flier could land on.
24///
25/// Vanilla parity: the `best.costMalus++` of `FlyNodeEvaluator.findAcceptedNode`
26/// when the node is walkable, which nudges a bird toward open air rather than
27/// scraping along the ground.
28const WALKABLE_NODE_MALUS: f32 = 1.0;
29
30/// How many random start positions a small mob tries.
31///
32/// Vanilla parity: `FlyNodeEvaluator.MAX_START_NODE_CANDIDATES`.
33const MAX_START_NODE_CANDIDATES: i32 = 10;
34
35/// Size below which a mob counts as small for the start search.
36///
37/// Vanilla parity: `FlyNodeEvaluator.SMALL_MOB_SIZE`.
38const SMALL_MOB_SIZE: f64 = 1.0;
39
40/// Box a small mob's start candidates are drawn from.
41///
42/// Vanilla parity: `FlyNodeEvaluator.SMALL_MOB_INFLATED_START_NODE_BOUNDING_BOX`.
43const SMALL_MOB_INFLATED_START_BOX: f64 = 1.1;
44
45/// Classifies one block the way a flying mob sees it.
46///
47/// Vanilla parity: `FlyNodeEvaluator.getPathType`. Open air one block above
48/// anything that is neither air, water nor walkable becomes `WALKABLE`, which
49/// is what makes a bird treat a rooftop as a destination.
50#[must_use]
51pub(super) fn fly_path_type(
52    context: &mut PathfindingContext<'_>,
53    x: i32,
54    y: i32,
55    z: i32,
56) -> PathType {
57    let mut block_path_type = context.get_path_type_from_state(x, y, z);
58    if block_path_type == PathType::Open && y > context.level().min_y() {
59        let below_pos = BlockPos::new(x, y - 1, z);
60        block_path_type = match context.get_path_type_from_state(x, y - 1, z) {
61            PathType::Fire | PathType::Lava => PathType::Fire,
62            PathType::Damaging => PathType::Damaging,
63            PathType::Cocoa => PathType::Cocoa,
64            PathType::Fence => {
65                if below_pos == context.mob_position() {
66                    PathType::Open
67                } else {
68                    PathType::Fence
69                }
70            }
71            PathType::Walkable | PathType::Open | PathType::Water => PathType::Open,
72            _ => PathType::Walkable,
73        };
74    }
75
76    if matches!(block_path_type, PathType::Walkable | PathType::Open) {
77        block_path_type =
78            WalkPathEvaluator::check_neighbour_blocks(context, x, y, z, block_path_type);
79    }
80
81    block_path_type
82}

84/// Decides where a flying mob may go.
85///
86/// Vanilla parity: `FlyNodeEvaluator`.
87#[derive(Debug, Clone)]
88pub struct FlyNodeEvaluator {
89    /// The walking evaluator this one extends.
90    ///
91    /// Vanilla parity: the superclass. It owns the node store, the mob settings
92    /// and the bounding-box classification; only the neighbor search, the start
93    /// node and the per-block path type differ, and the last of those is a mode
94    /// on the walker rather than a second copy of the scan.
95    walk: WalkNodeEvaluator,
96    /// Path types already computed this search.
97    ///
98    /// Vanilla parity: `FlyNodeEvaluator.pathTypeByPosCache`. The neighbor
99    /// search asks about the same twenty-six positions from several directions.
100    path_type_cache: FxHashMap<i64, PathType>,
101}
102
103impl FlyNodeEvaluator {
104    /// Creates the evaluator for one flying mob.
105    #[must_use]
106    pub fn new(settings: MobPathSettings) -> Self {
107        Self {
108            walk: WalkNodeEvaluator::new(settings).with_path_type_mode(PathTypeMode::Fly),
109            path_type_cache: FxHashMap::default(),
110        }
111    }
112
113    /// Vanilla parity: `FlyNodeEvaluator.getCachedPathType`.
114    fn cached_path_type(
115        &mut self,
116        context: &mut PathfindingContext<'_>,
117        x: i32,
118        y: i32,
119        z: i32,
120    ) -> PathType {
121        let key = PackedBlockPos::from(BlockPos::new(x, y, z)).as_raw();
122        if let Some(path_type) = self.path_type_cache.get(&key) {
123            return *path_type;
124        }
125        let path_type = self.walk.get_path_type_of_mob(context, x, y, z);
126        self.path_type_cache.insert(key, path_type);
127        path_type
128    }
129
130    /// Vanilla parity: `FlyNodeEvaluator.findAcceptedNode`.
131    fn find_accepted_node(
132        &mut self,
133        context: &mut PathfindingContext<'_>,
134        x: i32,
135        y: i32,
136        z: i32,
137    ) -> Option<i32> {
138        let path_type = self.cached_path_type(context, x, y, z);
139        let path_cost = self.walk.settings().pathfinding_malus(path_type);
140        if path_cost < 0.0 {
141            return None;
142        }
143
144        let node = self.walk.nodes_mut().get_node(x, y, z);
145        node.path_type = path_type;
146        node.cost_malus = node.cost_malus.max(path_cost);
147        if path_type == PathType::Walkable {
148            node.cost_malus += WALKABLE_NODE_MALUS;
149        }
150        Some(node.hash())
151    }
152
153    /// Vanilla parity: `FlyNodeEvaluator.isOpen`.
154    fn is_open(&self, hash: Option<i32>) -> bool {
155        hash.and_then(|hash| self.walk.node(hash))
156            .is_some_and(|node| !node.closed)
157    }
158
159    /// Vanilla parity: `FlyNodeEvaluator.hasMalus`.
160    fn has_malus(&self, hash: Option<i32>) -> bool {
161        hash.and_then(|hash| self.walk.node(hash))
162            .is_some_and(|node| node.cost_malus >= 0.0)
163    }
164
165    /// Vanilla parity: `FlyNodeEvaluator.canStartAt`.
166    ///
167    /// Unlike the walking check this accepts an open node, because open air is
168    /// exactly where a flier starts.
169    fn can_start_at(&mut self, context: &mut PathfindingContext<'_>, pos: BlockPos) -> bool {
170        let path_type = self.cached_path_type(context, pos.x(), pos.y(), pos.z());
171        self.walk.settings().pathfinding_malus(path_type) >= 0.0
172    }
173
174    /// Vanilla parity: `FlyNodeEvaluator.iteratePathfindingStartNodeCandidatePositions`.
175    fn start_node_candidates(&self) -> Vec<BlockPos> {
176        let settings = self.walk.settings();
177        let bounding_box = settings.bounding_box();
178        let mob_y = settings.mob_position().y();
179
180        if bounding_box.size() >= SMALL_MOB_SIZE {
181            return vec![
182                BlockPos::containing(bounding_box.min_x(), f64::from(mob_y), bounding_box.min_z()),
183                BlockPos::containing(bounding_box.min_x(), f64::from(mob_y), bounding_box.max_z()),
184                BlockPos::containing(bounding_box.max_x(), f64::from(mob_y), bounding_box.min_z()),
185                BlockPos::containing(bounding_box.max_x(), f64::from(mob_y), bounding_box.max_z()),
186            ];
187        }
188
189        let x_padding = (SMALL_MOB_INFLATED_START_BOX - bounding_box.width()).max(0.0);
190        let y_padding = (SMALL_MOB_INFLATED_START_BOX - bounding_box.height()).max(0.0);
191        let z_padding = (SMALL_MOB_INFLATED_START_BOX - bounding_box.depth()).max(0.0);
192        let inflated = bounding_box.inflate_xyz(x_padding, y_padding, z_padding);
193
194        let min = BlockPos::containing(inflated.min_x(), inflated.min_y(), inflated.min_z());
195        let max = BlockPos::containing(inflated.max_x(), inflated.max_y(), inflated.max_z());
196        (0..MAX_START_NODE_CANDIDATES)
197            .map(|_| {
198                BlockPos::new(
199                    rand::random_range(min.x()..=max.x()),
200                    rand::random_range(min.y()..=max.y()),
201                    rand::random_range(min.z()..=max.z()),
202                )
203            })
204            .collect()
205    }
206}

208impl NodeEvaluator for FlyNodeEvaluator {
209    /// Vanilla parity: `FlyNodeEvaluator.getStart`.
210    fn get_start(&mut self, context: &mut PathfindingContext<'_>) -> i32 {
211        let settings = self.walk.settings();
212        let position = settings.mob_position_vec();
213        let mut start_y = if settings.can_float() && settings.in_water() {
214            let mut surface = settings.mob_position().y();
215            while context
216                .get_block_state(BlockPos::containing(
217                    position.x,
218                    f64::from(surface),
219                    position.z,
220                ))
221                .get_fluid_state()
222                .is_water()
223            {
224                surface += 1;
225            }
226            surface
227        } else {
228            foton_math::fast_floor(position.y + 0.5)
229        };
230        if start_y < context.level().min_y() {
231            start_y = context.level().min_y();
232        }
233
234        let start_pos = BlockPos::containing(position.x, f64::from(start_y), position.z);
235        if !self.can_start_at(context, start_pos) {
236            for candidate in self.start_node_candidates() {
237                if self.can_start_at(context, candidate) {
238                    return self.walk.get_start_node(context, candidate);
239                }
240            }
241        }
242
243        self.walk.get_start_node(context, start_pos)
244    }
245
246    /// Vanilla parity: `FlyNodeEvaluator.getNeighbors`: the six faces, then the
247    /// twelve edge diagonals, then the eight corners, each gated on every
248    /// straight step it is built from.
249    #[expect(
250        clippy::too_many_lines,
251        reason = "vanilla lists all twenty-six steps and their gates by hand; \
252                  compressing them hides which gate belongs to which corner"
253    )]
254    fn get_neighbors(
255        &mut self,
256        context: &mut PathfindingContext<'_>,
257        _collision: &mut dyn WalkNodeCollision,
258        pos_hash: i32,
259    ) -> Neighbors {
260        let mut neighbors = Neighbors::new();
261        let Some(node) = self.walk.node(pos_hash) else {
262            return neighbors;
263        };
264        let (x, y, z) = (node.x, node.y, node.z);
265
266        let south = self.find_accepted_node(context, x, y, z + 1);
267        let west = self.find_accepted_node(context, x - 1, y, z);
268        let east = self.find_accepted_node(context, x + 1, y, z);
269        let north = self.find_accepted_node(context, x, y, z - 1);
270        let up = self.find_accepted_node(context, x, y + 1, z);
271        let down = self.find_accepted_node(context, x, y - 1, z);
272        for face in [south, west, east, north, up, down] {
273            if self.is_open(face) {
274                neighbors.push(face.unwrap_or_default());
275            }
276        }
277
278        let south_up = self.find_accepted_node(context, x, y + 1, z + 1);
279        let west_up = self.find_accepted_node(context, x - 1, y + 1, z);
280        let east_up = self.find_accepted_node(context, x + 1, y + 1, z);
281        let north_up = self.find_accepted_node(context, x, y + 1, z - 1);
282        let south_down = self.find_accepted_node(context, x, y - 1, z + 1);
283        let west_down = self.find_accepted_node(context, x - 1, y - 1, z);
284        let east_down = self.find_accepted_node(context, x + 1, y - 1, z);
285        let north_down = self.find_accepted_node(context, x, y - 1, z - 1);
286        let north_east = self.find_accepted_node(context, x + 1, y, z - 1);
287        let south_east = self.find_accepted_node(context, x + 1, y, z + 1);
288        let north_west = self.find_accepted_node(context, x - 1, y, z - 1);
289        let south_west = self.find_accepted_node(context, x - 1, y, z + 1);
290
291        for (edge, gates) in [
292            (south_up, [south, up]),
293            (west_up, [west, up]),
294            (east_up, [east, up]),
295            (north_up, [north, up]),
296            (south_down, [south, down]),
297            (west_down, [west, down]),
298            (east_down, [east, down]),
299            (north_down, [north, down]),
300            (north_east, [north, east]),
301            (south_east, [south, east]),
302            (north_west, [north, west]),
303            (south_west, [south, west]),
304        ] {
305            if self.is_open(edge) && gates.iter().all(|gate| self.has_malus(*gate)) {
306                neighbors.push(edge.unwrap_or_default());
307            }
308        }
309
310        for (corner_x, corner_y, corner_z, gates) in [
311            (
312                x + 1,
313                y + 1,
314                z - 1,
315                [north_east, north, east, up, north_up, east_up],
316            ),
317            (
318                x + 1,
319                y + 1,
320                z + 1,
321                [south_east, south, east, up, south_up, east_up],
322            ),
323            (
324                x - 1,
325                y + 1,
326                z - 1,
327                [north_west, north, west, up, north_up, west_up],
328            ),
329            (
330                x - 1,
331                y + 1,
332                z + 1,
333                [south_west, south, west, up, south_up, west_up],
334            ),
335            (
336                x + 1,
337                y - 1,
338                z - 1,
339                [north_east, north, east, down, north_down, east_down],
340            ),
341            (
342                x + 1,
343                y - 1,
344                z + 1,
345                [south_east, south, east, down, south_down, east_down],
346            ),
347            (
348                x - 1,
349                y - 1,
350                z - 1,
351                [north_west, north, west, down, north_down, west_down],
352            ),
353            (
354                x - 1,
355                y - 1,
356                z + 1,
357                [south_west, south, west, down, south_down, west_down],
358            ),
359        ] {
360            let corner = self.find_accepted_node(context, corner_x, corner_y, corner_z);
361            if self.is_open(corner) && gates.iter().all(|gate| self.has_malus(*gate)) {
362                neighbors.push(corner.unwrap_or_default());
363            }
364        }
365
366        neighbors
367    }

371    }

383    }

388        self.walk.clear_nodes();
389        self.path_type_cache.clear();
390    }
