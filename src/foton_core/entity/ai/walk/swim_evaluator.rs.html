1//! Node evaluation for mobs that swim.
2//!
3//! Vanilla parity: `SwimNodeEvaluator`. Where the walking evaluator asks what a
4//! mob can stand on, this one asks what it can be inside: every node is a block
5//! of water, and the search is free in all three axes rather than bound to a
6//! floor.
7

22/// Extra cost of a node the mob would have to leave the water for.
23///
24/// Vanilla parity: the `8.0F` `SwimNodeEvaluator.findAcceptedNode` adds when the
25/// block holds no fluid, which is what keeps a fish in the pond when it could
26/// technically flop onto the bank.
27const DRY_NODE_MALUS: f32 = 8.0;
28
29/// The four horizontal directions, in vanilla's order.
30///
31/// Vanilla parity: `Direction.Plane.HORIZONTAL`, walked to build the diagonals.
32const HORIZONTAL: [Direction; 4] = [
33    Direction::North,
34    Direction::East,
35    Direction::South,
36    Direction::West,
37];
38
39/// Decides where a swimming mob may go.
40///
41/// Vanilla parity: `SwimNodeEvaluator`.
42///
43/// Nothing constructs one yet. `WaterBoundPathNavigation` is the next step, and
44/// it is what will hand this to the pathfinder.
45#[derive(Debug, Clone)]
46pub struct SwimNodeEvaluator {
47    settings: MobPathSettings,
48    nodes: NodeStore,
49    /// Whether the mob may break the surface.
50    ///
51    /// Vanilla parity: the `allowBreaching` constructor flag, true for dolphins
52    /// and false for fish, which is why a dolphin jumps and a cod does not.
53    allow_breaching: bool,
54    /// Path types already computed this search.
55    ///
56    /// Vanilla parity: `pathTypesByPosCache`. The swim evaluator asks for the
57    /// same position from several directions, and each answer walks the mob's
58    /// whole bounding box.
59    path_type_cache: FxHashMap<i64, PathType>,
60}
61
62impl SwimNodeEvaluator {
63    /// Creates the evaluator for one swimming mob.
64    #[must_use]
65    pub fn new(settings: MobPathSettings, allow_breaching: bool) -> Self {
66        Self {
67            settings,
68            nodes: NodeStore::new(),
69            allow_breaching,
70            path_type_cache: FxHashMap::default(),
71        }
72    }
73
74    /// Returns the settings the evaluator was built from.
75    #[must_use]
76    pub const fn settings(&self) -> &MobPathSettings {
77        &self.settings
78    }
79
80    /// Returns the node at `hash`, if it has been reached.
81    #[must_use]
82    pub fn node(&self, hash: i32) -> Option<&Node> {
83        self.nodes.get(hash)
84    }
85
86    /// Returns whether a node can still be stepped to.
87    ///
88    /// Vanilla parity: `SwimNodeEvaluator.isNodeValid`.
89    fn is_node_valid(&self, hash: Option<i32>) -> bool {
90        hash.and_then(|hash| self.nodes.get(hash))
91            .is_some_and(|node| !node.closed)
92    }
93
94    /// Returns whether a node is reachable at all.
95    ///
96    /// Vanilla parity: `SwimNodeEvaluator.hasMalus`, the gate on building a
97    /// diagonal out of two straight steps.
98    fn has_malus(&self, hash: Option<i32>) -> bool {
99        hash.and_then(|hash| self.nodes.get(hash))
100            .is_some_and(|node| node.cost_malus >= 0.0)
101    }
102
103    /// Returns the node at these coordinates if a swimming mob may occupy it.
104    ///
105    /// Vanilla parity: `SwimNodeEvaluator.findAcceptedNode`.
106    fn find_accepted_node(
107        &mut self,
108        context: &mut PathfindingContext<'_>,
109        x: i32,
110        y: i32,
111        z: i32,
112    ) -> Option<i32> {
113        let path_type = self.cached_path_type(context, x, y, z);
114        let accepted =
115            path_type == PathType::Water || (self.allow_breaching && path_type == PathType::Breach);
116        if !accepted {
117            return None;
118        }
119
120        let path_cost = self.settings.pathfinding_malus(path_type);
121        if path_cost < 0.0 {
122            return None;
123        }
124
125        // Read the fluid before borrowing the node store, which borrows `self`.
126        let dry = context
127            .get_block_state(BlockPos::new(x, y, z))
128            .get_fluid_state()
129            .is_empty();
130
131        let node = self.nodes.get_node(x, y, z);
132        node.path_type = path_type;
133        node.cost_malus = node.cost_malus.max(path_cost);
134        if dry {
135            node.cost_malus += DRY_NODE_MALUS;
136        }
137        Some(node.hash())
138    }
139
140    /// Returns the path type at these coordinates, computing it once per search.
141    ///
142    /// Vanilla parity: `SwimNodeEvaluator.getCachedBlockType`.
143    fn cached_path_type(
144        &mut self,
145        context: &mut PathfindingContext<'_>,
146        x: i32,
147        y: i32,
148        z: i32,
149    ) -> PathType {
150        let key = PackedBlockPos::from(BlockPos::new(x, y, z)).as_raw();
151        if let Some(path_type) = self.path_type_cache.get(&key) {
152            return *path_type;
153        }
154        let path_type = self.path_type_of_mob(context, x, y, z);
155        self.path_type_cache.insert(key, path_type);
156        path_type
157    }
158
159    /// Classifies the block a swimming mob would occupy.
160    ///
161    /// Vanilla parity: `SwimNodeEvaluator.getPathTypeOfMob`, which walks the
162    /// mob's whole bounding box and refuses the position as soon as one cell is
163    /// not water.
164    fn path_type_of_mob(
165        &self,
166        context: &mut PathfindingContext<'_>,
167        x: i32,
168        y: i32,
169        z: i32,
170    ) -> PathType {
171        for offset_x in x..x + self.settings.entity_width() {
172            for offset_y in y..y + self.settings.entity_height() {
173                for offset_z in z..z + self.settings.entity_depth() {
174                    let pos = BlockPos::new(offset_x, offset_y, offset_z);
175                    let state = context.get_block_state(pos);
176                    let fluid = state.get_fluid_state();
177                    let below = context.get_block_state(pos.below());
178
179                    if fluid.is_empty()
180                        && below.is_pathfindable(PathComputationType::Water)
181                        && state.is_air()
182                    {
183                        return PathType::Breach;
184                    }
185
186                    if !is_water_fluid(fluid.fluid_id) {
187                        return PathType::Blocked;
188                    }
189                }
190            }
191        }
192
193        let state = context.get_block_state(BlockPos::new(x, y, z));
194        if state.is_pathfindable(PathComputationType::Water) {
195            PathType::Water
196        } else {
197            PathType::Blocked
198        }
199    }
200}

202impl NodeEvaluator for SwimNodeEvaluator {
203    /// Vanilla parity: `SwimNodeEvaluator.getStart`, which starts from the
204    /// bottom corner of the mob's box rather than the block it stands on.
205    fn get_start(&mut self, _context: &mut PathfindingContext<'_>) -> i32 {
206        let bounding_box = self.settings.bounding_box();
207        let node = self.nodes.get_node(
208            bounding_box.min_x().floor() as i32,
209            (bounding_box.min_y() + 0.5).floor() as i32,
210            bounding_box.min_z().floor() as i32,
211        );
212        node.hash()
213    }
214
215    /// Vanilla parity: `SwimNodeEvaluator.getNeighbors`: all six faces, then a
216    /// diagonal for each pair of horizontal neighbors that both worked out.
217    fn get_neighbors(
218        &mut self,
219        context: &mut PathfindingContext<'_>,
220        _collision: &mut dyn WalkNodeCollision,
221        pos_hash: i32,
222    ) -> Neighbors {
223        let mut neighbors = Neighbors::new();
224        let Some(node) = self.nodes.get(pos_hash) else {
225            return neighbors;
226        };
227        let (x, y, z) = (node.x, node.y, node.z);
228
229        let mut by_direction: [Option<i32>; Direction::ALL.len()] = [None; Direction::ALL.len()];
230        for (index, direction) in Direction::ALL.into_iter().enumerate() {
231            let (step_x, step_y, step_z) = direction.offset();
232            let found = self.find_accepted_node(context, x + step_x, y + step_y, z + step_z);
233            by_direction[index] = found;
234            if self.is_node_valid(found) {
235                neighbors.push(found.unwrap_or_default());
236            }
237        }
238
239        for direction in HORIZONTAL {
240            let clockwise = direction.rotate_y_clockwise();
241            let straight = by_direction[direction_index(direction)];
242            let turned = by_direction[direction_index(clockwise)];
243            if !self.has_malus(straight) || !self.has_malus(turned) {
244                continue;
245            }
246
247            let (step_x, _, step_z) = direction.offset();
248            let (turn_x, _, turn_z) = clockwise.offset();
249            let diagonal =
250                self.find_accepted_node(context, x + step_x + turn_x, y, z + step_z + turn_z);
251            if self.is_node_valid(diagonal) {
252                neighbors.push(diagonal.unwrap_or_default());
253            }
254        }
255
256        neighbors
257    }
258
259    fn node(&self, hash: i32) -> Option<&Node> {
260        self.nodes.get(hash)
261    }
262
263    fn node_mut(&mut self, hash: i32) -> Option<&mut Node> {
264        self.nodes.get_mut(hash)
265    }
266
267    fn nodes_mut(&mut self) -> &mut NodeStore {
268        &mut self.nodes
269    }
270
271    fn reset_search_state(&mut self) {
272        self.nodes.reset_search_state();
273    }
274
275    /// Vanilla parity: `SwimNodeEvaluator.done`, which drops the path-type cache
276    /// along with the nodes.
277    fn clear_nodes(&mut self) {
278        self.nodes.clear();
279        self.path_type_cache.clear();
280    }
281}
282
283/// Returns a direction's slot in [`Direction::ALL`].
284fn direction_index(direction: Direction) -> usize {
285    Direction::ALL
286        .into_iter()
287        .position(|candidate| candidate == direction)
288        .unwrap_or_default()
289}

301    /// A world that is water everywhere except where told otherwise.
302    struct Pool {
303        overrides: Vec<(BlockPos, BlockStateId)>,
304    }
305
306    impl Pool {
307        fn new() -> Self {
308            // Building the pool already reads block states, so the registry has
309            // to be up before the first one is asked for.
310            init_vanilla_registry();
311            init_behaviors();
312            Self {
313                overrides: Vec::new(),
314            }
315        }
316
317        fn with(mut self, pos: BlockPos, state: BlockStateId) -> Self {
318            self.overrides.push((pos, state));
319            self
320        }
321    }

323    impl LevelReader for Pool {
324        fn get_block_state(&self, pos: BlockPos) -> BlockStateId {
325            self.overrides
326                .iter()
327                .find_map(|(at, state)| (*at == pos).then_some(*state))
328                .unwrap_or_else(|| vanilla_blocks::WATER.default_state())
329        }

331        fn raw_brightness(&self, _pos: BlockPos, _sky_darkening: u8) -> u8 {
332            0
333        }

336            -64
