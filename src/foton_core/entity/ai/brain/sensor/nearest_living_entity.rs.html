3use rustc_hash::FxHashSet;
4
5use super::{Sensor, follow_range, is_entity_targetable};
6use crate::entity::LivingEntity;
7use crate::entity::SharedEntity;
8use crate::entity::ai::brain::context::BrainContext;
9use crate::entity::ai::brain::memory::{
10    EntityMemory, MemoryModuleId, NearestVisibleLivingEntities, memory_module_types,
11};
12
13/// Remembers every living entity within follow range, nearest first.
14///
15/// Vanilla parity: `net.minecraft.world.entity.ai.sensing.NearestLivingEntitySensor`.
16pub struct NearestLivingEntitySensor;
17
18impl Sensor for NearestLivingEntitySensor {
19    fn required_memories(&self) -> Vec<MemoryModuleId> {
20        vec![
21            memory_module_types::NEAREST_LIVING_ENTITIES.id(),
22            memory_module_types::NEAREST_VISIBLE_LIVING_ENTITIES.id(),
23        ]
24    }
25
26    fn do_tick(&mut self, ctx: &BrainContext<'_>) {
27        let body = ctx.mob();
28        let range = follow_range(body);
29        let search_area = body.bounding_box().inflate_xyz(range, range, range);
30        let body_id = body.id();
31
32        let mut nearby: Vec<SharedEntity> =
33            ctx.world()
34                .get_entities_in_aabb_matching(&search_area, |entity| {
35                    entity.id() != body_id
36                        && entity
37                            .as_living_entity()
38                            .is_some_and(LivingEntity::is_alive)
39                });
40        let body_position = body.position();
41        nearby.sort_by(|left, right| {
42            let left = left.position().distance_squared(body_position);
43            let right = right.position().distance_squared(body_position);
44            left.total_cmp(&right)
45        });
46
47        let mut visible = FxHashSet::default();
48        for entity in &nearby {
49            let Some(living) = entity.as_living_entity() else {
50                continue;
51            };
52            if is_entity_targetable(ctx.world(), body, living) {
53                visible.insert(entity.id());
54            }
55        }
56
57        let remembered: Vec<EntityMemory> = nearby.iter().map(EntityMemory::new).collect();
58        ctx.brain().set_memory(
59            memory_module_types::NEAREST_LIVING_ENTITIES,
60            remembered.clone(),
61        );
62        ctx.brain().set_memory(
63            memory_module_types::NEAREST_VISIBLE_LIVING_ENTITIES,
64            NearestVisibleLivingEntities::new(remembered, visible),
65        );
66    }
67}
