1//! Vanilla `VillagerBabiesSensor`.
2
3use foton_registry::vanilla_entities;
4
5use super::Sensor;
6
7use crate::entity::ai::brain::behavior::utils;
8use crate::entity::ai::brain::context::BrainContext;
9use crate::entity::ai::brain::memory::{EntityMemory, MemoryModuleId, memory_module_types};
10
11/// Remembers the village's other children.
12///
13/// Vanilla parity: `net.minecraft.world.entity.ai.sensing.VillagerBabiesSensor`.
14/// It scans nothing of its own: it filters the visible-mobs memory
15/// `NearestLivingEntitySensor` already fills, so its range is whatever that one
16/// last wrote and its cost is a walk over a short list.
17pub struct VillagerBabiesSensor;
18
19impl Sensor for VillagerBabiesSensor {
20    fn required_memories(&self) -> Vec<MemoryModuleId> {
21        vec![memory_module_types::VISIBLE_VILLAGER_BABIES.id()]
22    }
23
24    fn do_tick(&mut self, ctx: &BrainContext<'_>) {
25        let babies = ctx
26            .brain()
27            .get_memory(memory_module_types::NEAREST_VISIBLE_LIVING_ENTITIES)
28            .map(|visible| {
29                visible.find_all(|candidate| {
30                    utils::is_of_type(candidate, &vanilla_entities::VILLAGER) && candidate.is_baby()
31                })
32            })
33            .unwrap_or_default();
34        // Vanilla writes the list whether or not it is empty; the memory map
35        // erases an empty collection either way, which is what leaves
36        // `VISIBLE_VILLAGER_BABIES` absent for the play package to gate on.
37        ctx.brain().set_memory(
38            memory_module_types::VISIBLE_VILLAGER_BABIES,
39            babies.iter().map(EntityMemory::new).collect::<Vec<_>>(),
40        );
41    }
42}
