3use foton_registry::vanilla_entity_type_tags::EntityTypeTag;
4use foton_registry::{REGISTRY, TaggedRegistryExt as _};
5
6use crate::entity::SharedEntity;
7use crate::entity::ai::brain::context::BrainContext;
8use crate::entity::ai::brain::memory::{EntityMemory, MemoryModuleId, memory_module_types};
9
10use super::{Sensor, follow_range, is_entity_attackable};
11
12/// Vanilla parity: `AxolotlAttackablesSensor.TARGET_DETECTION_DISTANCE`, which
13/// the sensor squares before comparing.
14const TARGET_DETECTION_DISTANCE_SQR: f64 = 64.0;
15
16/// Remembers the nearest thing an axolotl would pick a fight with.
17///
18/// Vanilla parity: `net.minecraft.world.entity.ai.sensing.AxolotlAttackablesSensor`.
19/// Two lists feed it: the drowned and the guardians it always hates, and the
20/// fish it hunts -- but only while it has no hunting cooldown, which is what
21/// stops one axolotl clearing a whole reef.
22pub struct AxolotlAttackablesSensor;
23
24impl Sensor for AxolotlAttackablesSensor {
25    fn required_memories(&self) -> Vec<MemoryModuleId> {
26        vec![
27            memory_module_types::NEAREST_ATTACKABLE.id(),
28            memory_module_types::HAS_HUNTING_COOLDOWN.id(),
29        ]
30    }
31
32    fn do_tick(&mut self, ctx: &BrainContext<'_>) {
33        let body = ctx.mob();
34        let range = follow_range(body);
35        let search_area = body.bounding_box().inflate_xyz(range, range, range);
36        let body_id = body.id();
37        let body_position = body.position();
38        // Vanilla reads the cooldown once per candidate; the answer cannot
39        // change inside one scan, so it is read once here instead.
40        let hunting = !ctx
41            .brain()
42            .has_memory_value(memory_module_types::HAS_HUNTING_COOLDOWN.id());
43
44        let mut candidates: Vec<SharedEntity> =
45            ctx.world()
46                .get_entities_in_aabb_matching(&search_area, |entity| {
47                    if entity.id() == body_id {
48                        return false;
49                    }
50                    let Some(living) = entity.as_living_entity() else {
51                        return false;
52                    };
53                    if entity.position().distance_squared(body_position)
54                        > TARGET_DETECTION_DISTANCE_SQR
55                    {
56                        return false;
57                    }
58                    if !living.is_in_water() {
59                        return false;
60                    }
61                    let always_hostile = REGISTRY.entity_types.is_in_tag(
62                        entity.entity_type(),
63                        &EntityTypeTag::AXOLOTL_ALWAYS_HOSTILES,
64                    );
65                    let hunt_target = hunting
66                        && REGISTRY
67                            .entity_types
68                            .is_in_tag(entity.entity_type(), &EntityTypeTag::AXOLOTL_HUNT_TARGETS);
69                    if !always_hostile && !hunt_target {
70                        return false;
71                    }
72                    is_entity_attackable(ctx.world(), body, living)
73                });
74
75        candidates.sort_by(|left, right| {
76            let left = left.position().distance_squared(body_position);
77            let right = right.position().distance_squared(body_position);
78            left.total_cmp(&right)
79        });
80
81        ctx.brain().set_memory_or_erase(
82            memory_module_types::NEAREST_ATTACKABLE,
83            candidates.first().map(EntityMemory::new),
84        );
85    }
86}
