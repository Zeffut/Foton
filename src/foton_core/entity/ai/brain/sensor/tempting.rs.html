9use crate::entity::ai::targeting::TargetingConditions;

14/// What counts as a temptation for this mob.
15type Temptations = Box<dyn Fn(&dyn PathfinderMob, &ItemStack) -> bool + Send>;
16
17/// Remembers the nearest player holding something this mob wants.
18///
19/// Vanilla parity: `net.minecraft.world.entity.ai.sensing.TemptingSensor`.
20pub struct TemptingSensor {
21    temptations: Temptations,
22}
23
24impl TemptingSensor {
25    /// Tempts a mob with whatever its own `isFood` accepts.
26    ///
27    /// Vanilla parity: `TemptingSensor.forAnimal`.
28    #[must_use]
29    pub fn for_animal() -> Self {
30        Self::new(|mob, item_stack| {
31            mob.as_animal()
32                .is_some_and(|animal| animal.is_food(item_stack))
33        })
34    }
35
36    /// Tempts a mob with a fixed set of items.
37    ///
38    /// Vanilla parity: `new TemptingSensor(Predicate<ItemStack>)`.
39    #[must_use]
40    pub fn new(
41        temptations: impl Fn(&dyn PathfinderMob, &ItemStack) -> bool + Send + 'static,
42    ) -> Self {
43        Self {
44            temptations: Box::new(temptations),
45        }
46    }
47
48    fn is_holding_temptation(&self, mob: &dyn PathfinderMob, player: &Player) -> bool {
49        [EquipmentSlot::MainHand, EquipmentSlot::OffHand]
50            .into_iter()
51            .any(|slot| {
52                let mut tempting = false;
53                player.with_equipment_slot(slot, &mut |item_stack| {
54                    tempting = (self.temptations)(mob, item_stack);
55                });
56                tempting
57            })
58    }
59}
60
61impl Sensor for TemptingSensor {
62    fn required_memories(&self) -> Vec<MemoryModuleId> {
63        vec![memory_module_types::TEMPTING_PLAYER.id()]
64    }
65
66    fn do_tick(&mut self, ctx: &BrainContext<'_>) {
67        let body = ctx.mob();
68        let range = body
69            .attributes()
70            .lock()
71            .required_value(vanilla_attributes::TEMPT_RANGE);
72        // Vanilla parity: `TEMPT_TARGETING`, a non-combat condition that
73        // ignores line of sight so a mob follows food held behind a fence.
74        let conditions = TargetingConditions::for_non_combat()
75            .ignore_line_of_sight()
76            .range(range);
77
78        let tempter = ctx
79            .world()
80            .nearest_player(body.position(), range, |player| {
81                !player.is_spectator()
82                    && conditions.test(ctx.world(), Some(body), player)
83                    && self.is_holding_temptation(body, player)
84            });
85
86        ctx.brain().set_memory_or_erase(
87            memory_module_types::TEMPTING_PLAYER,
88            tempter.map(|player| EntityMemory::new(&(player as SharedEntity))),
89        );
90    }
91}
