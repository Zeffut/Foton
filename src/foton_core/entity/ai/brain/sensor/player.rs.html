1//! Vanilla `PlayerSensor`.
2
3use std::sync::Arc;
4
5use super::{Sensor, follow_range, is_entity_attackable, is_entity_targetable};
6use crate::entity::ai::brain::context::BrainContext;
7use crate::entity::ai::brain::memory::{EntityMemory, MemoryModuleId, memory_module_types};
8use crate::entity::{Entity as _, SharedEntity};
9use crate::player::Player;
10
11/// Remembers the nearby players, and which of them are visible or attackable.
12///
13/// Vanilla parity: `net.minecraft.world.entity.ai.sensing.PlayerSensor`.
14pub struct PlayerSensor;
15
16impl Sensor for PlayerSensor {
17    fn required_memories(&self) -> Vec<MemoryModuleId> {
18        vec![
19            memory_module_types::NEAREST_PLAYERS.id(),
20            memory_module_types::NEAREST_VISIBLE_PLAYER.id(),
21            memory_module_types::NEAREST_VISIBLE_ATTACKABLE_PLAYER.id(),
22            memory_module_types::NEAREST_VISIBLE_ATTACKABLE_PLAYERS.id(),
23        ]
24    }
25
26    fn do_tick(&mut self, ctx: &BrainContext<'_>) {
27        let body = ctx.mob();
28        let body_position = body.position();
29        let range = follow_range(body);
30        let range_sqr = range * range;
31
32        let mut players: Vec<Arc<Player>> = Vec::new();
33        ctx.world().players.iter_players(|_, player| {
34            if !player.is_spectator()
35                && player.position().distance_squared(body_position) <= range_sqr
36            {
37                players.push(Arc::clone(player));
38            }
39            true
40        });
41        players.sort_by(|left, right| {
42            let left = left.position().distance_squared(body_position);
43            let right = right.position().distance_squared(body_position);
44            left.total_cmp(&right)
45        });
46
47        let brain = ctx.brain();
48        brain.set_memory(
49            memory_module_types::NEAREST_PLAYERS,
50            players
51                .iter()
52                .map(|player| EntityMemory::new(&(Arc::clone(player) as SharedEntity)))
53                .collect::<Vec<_>>(),
54        );
55
56        let visible: Vec<&Arc<Player>> = players
57            .iter()
58            .filter(|player| is_entity_targetable(ctx.world(), body, player.as_ref()))
59            .collect();
60        brain.set_memory_or_erase(
61            memory_module_types::NEAREST_VISIBLE_PLAYER,
62            visible
63                .first()
64                .map(|player| EntityMemory::new(&(Arc::clone(player) as SharedEntity))),
65        );
66
67        let attackable: Vec<&Arc<Player>> = visible
68            .into_iter()
69            .filter(|player| is_entity_attackable(ctx.world(), body, player.as_ref()))
70            .collect();
71        brain.set_memory(
72            memory_module_types::NEAREST_VISIBLE_ATTACKABLE_PLAYERS,
73            attackable
74                .iter()
75                .map(|player| EntityMemory::new(&(Arc::clone(player) as SharedEntity)))
76                .collect::<Vec<_>>(),
77        );
78        brain.set_memory_or_erase(
79            memory_module_types::NEAREST_VISIBLE_ATTACKABLE_PLAYER,
80            attackable
81                .first()
82                .map(|player| EntityMemory::new(&(Arc::clone(player) as SharedEntity))),
83        );
84    }
85}
