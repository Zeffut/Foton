16/// The smallest speed factor a charge's knockback is scaled by.
17///
18/// Vanilla parity: the low bound of the `Mth.clamp(..., 0.2F, 2.0F)` in
19/// `ChargeAttack.dealKnockBack`.
20const MIN_KNOCKBACK_SPEED_FACTOR: f32 = 0.2;
21/// The largest speed factor a charge's knockback is scaled by.
22const MAX_KNOCKBACK_SPEED_FACTOR: f32 = 2.0;
23/// What one level of speed or slowness is worth to a charge's knockback.
24///
25/// Vanilla parity: the `0.25F` of `ChargeAttack.dealKnockBack`.
26const KNOCKBACK_PER_EFFECT_LEVEL: f32 = 0.25;
27/// The rotation limit that lets a charging body snap straight at its target.
28///
29/// Vanilla parity: the `360.0F` pair of `ChargeAttack.tick`'s `lookAt`.
30const FULL_TURN: f32 = 360.0;
31
32/// Rushes `ATTACK_TARGET` in a straight line and rams whatever it reaches.
33///
34/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.ChargeAttack`. The
35/// charge is not pathfinding: [`Self::start`] freezes one velocity vector at
36/// the target and the body carries it until it hits something, runs out of
37/// charge distance, loses sight of the target, or is tamed mid-run.
38pub struct ChargeAttack {
39    time_between_attacks: i32,
40    charge_targeting: TargetingConditions,
41    speed: f32,
42    knockback_force: f32,
43    max_charge_distance: f64,
44    max_target_detection_distance: f64,
45    charge_sound: SoundEventRef,
46    charge_velocity_vector: DVec3,
47    start_position: DVec3,
48    entry_condition: [(MemoryModuleId, MemoryStatus); 2],
49}

51impl ChargeAttack {
52    /// Vanilla parity: the `ChargeAttack` constructor.
53    #[must_use]
54    pub const fn new(
55        time_between_attacks: i32,
56        charge_targeting: TargetingConditions,
57        speed: f32,
58        knockback_force: f32,
59        max_charge_distance: f64,
60        max_target_detection_distance: f64,
61        charge_sound: SoundEventRef,
62    ) -> Self {
63        Self {
64            time_between_attacks,
65            charge_targeting,
66            speed,
67            knockback_force,
68            max_charge_distance,
69            max_target_detection_distance,
70            charge_sound,
71            charge_velocity_vector: DVec3::ZERO,
72            start_position: DVec3::ZERO,
73            entry_condition: [
74                (
75                    memory_module_types::CHARGE_COOLDOWN_TICKS.id(),
76                    MemoryStatus::ValueAbsent,
77                ),
78                (
79                    memory_module_types::ATTACK_TARGET.id(),
80                    MemoryStatus::ValuePresent,
81                ),
82            ],
83        }
84    }

86    /// Vanilla parity: `ChargeAttack.dealDamageToTarget`.
87    fn deal_damage_to_target(ctx: &BrainContext<'_>, target: &SharedEntity) {
88        let body = ctx.mob();
89        let damage_source = Self::damage_source(ctx);
90        #[expect(
91            clippy::cast_possible_truncation,
92            reason = "an attack-damage attribute, immediately used as damage"
93        )]
94        let damage = body
95            .attributes()
96            .lock()
97            .required_value(vanilla_attributes::ATTACK_DAMAGE) as f32;
98        if !target.hurt(ctx.world(), &damage_source, damage) {
99            return;
100        }
101
102        // Vanilla parity: `EnchantmentHelper.doPostAttackEffects`, whose item
103        // source is the attacker's weapon -- a nautilus never holds one, but
104        // the victim half of the call still runs off the target's own armor.
105        let weapon_item = {
106            let mut main_hand = ItemStack::empty();
107            body.with_equipment_slot(EquipmentSlot::MainHand, &mut |item_stack| {
108                main_hand = item_stack.copy_with_count(item_stack.count());
109            });
110            main_hand
111        };
112        let context = EnchantmentPostAttackContext::new(
113            target.as_ref(),
114            Some(body.as_entity_event_source()),
115            Some(body.as_entity_event_source()),
116            &damage_source,
117        );
118        enchantment_helper::do_post_attack_effects_with_item_source(
119            ctx.world(),
120            target.as_ref(),
121            &weapon_item,
122            &context,
123        );
124    }
125
126    /// Vanilla parity: `ChargeAttack.dealKnockBack`.
127    fn deal_knockback(ctx: &BrainContext<'_>, target: &SharedEntity, speed: f32, force: f32) {
128        let body = ctx.mob();
129        let speed_level = body
130            .mob_effect(vanilla_mob_effects::SPEED)
131            .map_or(0, |effect| effect.amplifier() + 1);
132        let slowness_level = body
133            .mob_effect(vanilla_mob_effects::SLOWNESS)
134            .map_or(0, |effect| effect.amplifier() + 1);
135        #[expect(
136            clippy::cast_precision_loss,
137            reason = "an effect amplifier, which vanilla widens the same way"
138        )]
139        let speed_boost_power = KNOCKBACK_PER_EFFECT_LEVEL * (speed_level - slowness_level) as f32;
140        #[expect(
141            clippy::cast_possible_truncation,
142            reason = "a movement-speed attribute, immediately used as a speed"
143        )]
144        let movement_speed = body
145            .attributes()
146            .lock()
147            .required_value(vanilla_attributes::MOVEMENT_SPEED) as f32;
148        let speed_factor = (speed * movement_speed)
149            .clamp(MIN_KNOCKBACK_SPEED_FACTOR, MAX_KNOCKBACK_SPEED_FACTOR)
150            + speed_boost_power;
151
152        body.cause_extra_knockback(
153            target.as_ref(),
154            f64::from(speed_factor * force),
155            body.velocity(),
156        );
157    }
158
159    /// The body of `ChargeAttack.stop`, callable while the charge is mid-tick.
160    fn end_charge(ctx: &BrainContext<'_>, time_between_attacks: i32) {
161        ctx.brain().set_memory(
162            memory_module_types::CHARGE_COOLDOWN_TICKS,
163            time_between_attacks,
164        );
165        ctx.brain()
166            .erase_memory(memory_module_types::ATTACK_TARGET.id());
167    }
168
169    /// Vanilla parity: `level.damageSources().mobAttack(body)`.
170    fn damage_source(ctx: &BrainContext<'_>) -> DamageSource {
171        let body = ctx.mob();
172        DamageSource::environment(&vanilla_damage_types::MOB_ATTACK)
173            .with_causing_entity(body.id())
174            .with_direct_entity(body.id())
175            .with_source_position(body.position())
176    }
177}

179impl TimedBehavior for ChargeAttack {
180    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
181        &self.entry_condition
182    }
183
184    fn check_extra_start_conditions(&mut self, ctx: &BrainContext<'_>) -> bool {
185        ctx.brain()
186            .has_memory_value(memory_module_types::ATTACK_TARGET.id())
187    }
188
189    /// Vanilla parity: `ChargeAttack.canStillUse`.
190    fn can_still_use(&mut self, ctx: &BrainContext<'_>) -> bool {
191        let brain = ctx.brain();
192        let Some(target) = brain
193            .get_memory(memory_module_types::ATTACK_TARGET)
194            .and_then(|memory| memory.get())
195        else {
196            return false;
197        };
198        let body = ctx.mob();
199        // Vanilla parity: the `body instanceof TamableAnimal tamed && tamed.isTame()`
200        // bail-out -- a nautilus tamed mid-charge breaks it off.
201        if is_tamed(body.as_entity_event_source()) {
202            return false;
203        }
204        if body.position().distance_squared(self.start_position)
205            >= self.max_charge_distance * self.max_charge_distance
206        {
207            return false;
208        }
209        if target.position().distance_squared(body.position())
210            >= self.max_target_detection_distance * self.max_target_detection_distance
211        {
212            return false;
213        }
214        if !body.has_line_of_sight(target.as_ref()) {
215            return false;
216        }
217        !brain.has_memory_value(memory_module_types::CHARGE_COOLDOWN_TICKS.id())
218    }
219
220    /// Vanilla parity: `ChargeAttack.start`.
221    fn start(&mut self, ctx: &BrainContext<'_>) {
222        let body = ctx.mob();
223        self.start_position = body.position();
224        let Some(target) = ctx
225            .brain()
226            .get_memory(memory_module_types::ATTACK_TARGET)
227            .and_then(|memory| memory.get())
228        else {
229            return;
230        };
231        let direction = (target.position() - body.position()).normalize_or_zero();
232        self.charge_velocity_vector = direction * f64::from(self.speed);
233        if self.can_still_use(ctx) {
234            body.play_sound(self.charge_sound, 1.0, 1.0);
235        }
236    }
237
238    /// Vanilla parity: `ChargeAttack.tick`.
239    fn tick(&mut self, ctx: &BrainContext<'_>) {
240        let Some(target) = ctx
241            .brain()
242            .get_memory(memory_module_types::ATTACK_TARGET)
243            .and_then(|memory| memory.get())
244        else {
245            return;
246        };
247        let body = ctx.mob();
248        Mob::look_at(body, target.as_ref(), FULL_TURN, FULL_TURN);
249        body.set_velocity(self.charge_velocity_vector);
250
251        let Some(body_living) = body.as_entity_event_source().as_living_entity() else {
252            return;
253        };
254        let charge_targeting = &self.charge_targeting;
255        let rammed = ctx
256            .world()
257            .get_entities_in_aabb_matching(&body.bounding_box(), |candidate| {
258                candidate.as_living_entity().is_some_and(|living| {
259                    charge_targeting.test(ctx.world(), Some(body_living), living)
260                })
261            })
262            .into_iter()
263            .next();
264        let Some(rammed) = rammed else {
265            return;
266        };
267        // Vanilla parity: a rider is not rammed by the mob carrying them.
268        if body.has_passenger(rammed.as_ref()) {
269            return;
270        }
271
272        Self::deal_damage_to_target(ctx, &rammed);
273        Self::deal_knockback(ctx, &rammed, self.speed, self.knockback_force);
274        Self::end_charge(ctx, self.time_between_attacks);
275    }
276
277    /// Vanilla parity: `ChargeAttack.stop`, which is also what [`Self::tick`]
278    /// calls the moment the charge connects.
279    fn stop(&mut self, ctx: &BrainContext<'_>) {
280        Self::end_charge(ctx, self.time_between_attacks);
281    }
282
283    fn debug_name(&self) -> &'static str {
284        "ChargeAttack"
285    }
