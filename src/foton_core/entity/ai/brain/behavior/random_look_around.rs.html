3use foton_utils::value_providers::UniformIntProvider;
4use glam::DVec3;
5
6use super::{BrainContext, MemoryModuleId, MemoryStatus, TimedBehavior};
7use crate::entity::ai::brain::memory::memory_module_types;
8use crate::entity::ai::brain::position_tracker::PositionTracker;
9
10/// Vanilla parity: the `Mth.clamp(..., -90.0F, 90.0F)` of `start`.
11const MIN_PITCH: f32 = -90.0;
12const MAX_PITCH: f32 = 90.0;
13
14/// Looks somewhere at random, and then not again for a while.
15///
16/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.RandomLookAround`.
17/// The gaze cooldown it sets is what keeps a mob's head still between glances
18/// rather than swiveling every tick.
19pub struct RandomLookAround {
20    interval: UniformIntProvider,
21    max_yaw: f32,
22    min_pitch: f32,
23    pitch_range: f32,
24}
25
26/// Vanilla parity: the `entryCondition` map built in the constructor.
27const ENTRY_CONDITION: &[(MemoryModuleId, MemoryStatus)] = &[
28    (
29        memory_module_types::LOOK_TARGET.id(),
30        MemoryStatus::ValueAbsent,
31    ),
32    (
33        memory_module_types::GAZE_COOLDOWN_TICKS.id(),
34        MemoryStatus::ValueAbsent,
35    ),
36];
37
38impl RandomLookAround {
39    /// Vanilla parity: `new RandomLookAround(IntProvider, float, float, float)`.
40    ///
41    /// # Panics
42    ///
43    /// If `min_pitch` is above `max_pitch`, which is the same
44    /// `IllegalArgumentException` vanilla's constructor throws.
45    #[must_use]
46    pub fn new(interval: UniformIntProvider, max_yaw: f32, min_pitch: f32, max_pitch: f32) -> Self {
47        assert!(
48            min_pitch <= max_pitch,
49            "Minimum pitch is larger than maximum pitch! {min_pitch} > {max_pitch}"
50        );
51        Self {
52            interval,
53            max_yaw,
54            min_pitch,
55            pitch_range: max_pitch - min_pitch,
56        }
57    }
58}
59
60impl TimedBehavior for RandomLookAround {
61    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
62        ENTRY_CONDITION
63    }
64
65    fn start(&mut self, ctx: &BrainContext<'_>) {
66        let body = ctx.mob();
67        let pitch =
68            (rand::random::<f32>() * self.pitch_range + self.min_pitch).clamp(MIN_PITCH, MAX_PITCH);
69        let yaw = wrap_degrees(
70            body.rotation().0 + 2.0 * rand::random::<f32>() * self.max_yaw - self.max_yaw,
71        );
72        let look = body.calculate_view_vector(pitch, yaw);
73
74        let position = body.position();
75        let eye = DVec3::new(position.x, body.get_eye_y(), position.z);
76        ctx.brain().set_memory(
77            memory_module_types::LOOK_TARGET,
78            PositionTracker::of_position(eye + look),
79        );
80        ctx.brain().set_memory(
81            memory_module_types::GAZE_COOLDOWN_TICKS,
82            rand::random_range(self.interval.min_inclusive..=self.interval.max_inclusive),
83        );
84    }
85
86    fn debug_name(&self) -> &'static str {
87        "RandomLookAround"
88    }
89}
90
91/// Vanilla parity: `Mth.wrapDegrees`.
92fn wrap_degrees(mut degrees: f32) -> f32 {
93    degrees %= 360.0;
94    if degrees >= 180.0 {
95        degrees -= 360.0;
96    }
97    if degrees < -180.0 {
98        degrees += 360.0;
99    }
100    degrees
101}
