14/// How far a child runs from whoever is chasing it.
15///
16/// Vanilla parity: `PlayTagWithOtherKids.MAX_FLEE_XZ_DIST` and `MAX_FLEE_Y_DIST`.
17const MAX_FLEE_XZ_DIST: i32 = 20;
18const MAX_FLEE_Y_DIST: i32 = 8;
19
20/// How fast, either way. Vanilla gives the chase and the escape the same pace.
21///
22/// Vanilla parity: `PlayTagWithOtherKids.FLEE_SPEED_MODIFIER` and
23/// `CHASE_SPEED_MODIFIER`.
24const PLAY_SPEED_MODIFIER: f64 = 0.6;
25
26/// How many children may pile onto one of their own before the rest look for
27/// somebody else to chase.
28///
29/// Vanilla parity: `PlayTagWithOtherKids.MAX_CHASERS_PER_TARGET`.
30const MAX_CHASERS_PER_TARGET: usize = 5;
31
32/// One tick in this many starts or changes a game.
33///
34/// Vanilla parity: `PlayTagWithOtherKids.AVERAGE_WAIT_TIME_BETWEEN_RUNS`, the
35/// `nextInt(10) != 0` the whole behavior opens with.
36const AVERAGE_WAIT_TIME_BETWEEN_RUNS: i32 = 10;
37
38/// How many spots a fleeing child tries before giving up on this tick.
39///
40/// Vanilla parity: the `for (int j = 0; j < 10; j++)` of the flee branch.
41const FLEE_ATTEMPTS: i32 = 10;
42
43/// The game the village's children play with each other.
44///
45/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.PlayTagWithOtherKids`.
46///
47/// It is one behavior for both halves of tag, and which half a child is playing
48/// is read off everybody else's `INTERACTION_TARGET`: a child somebody is
49/// already chasing runs, and a child nobody is chasing joins the chase --
50/// preferring one that is already being chased, so the village plays one game
51/// rather than a dozen. Running is bounded to the village, which is what keeps
52/// a game of tag from emptying it.
53pub struct PlayTagWithOtherKids;
54
55impl Trigger for PlayTagWithOtherKids {
56    fn required_memories(&self) -> Vec<MemoryModuleId> {
57        vec![
58            memory_module_types::VISIBLE_VILLAGER_BABIES.id(),
59            memory_module_types::WALK_TARGET.id(),
60            memory_module_types::LOOK_TARGET.id(),
61            memory_module_types::INTERACTION_TARGET.id(),
62        ]
63    }
64
65    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
66        let brain = ctx.brain();
67        // Vanilla parity: `i.present(VISIBLE_VILLAGER_BABIES)` and
68        // `i.absent(WALK_TARGET)` -- a child already going somewhere is left to
69        // get there.
70        let Some(friends) = brain.get_memory(memory_module_types::VISIBLE_VILLAGER_BABIES) else {
71            return false;
72        };
73        if brain.has_memory_value(memory_module_types::WALK_TARGET.id()) {
74            return false;
75        }
76        if rand::random_range(0..AVERAGE_WAIT_TIME_BETWEEN_RUNS) != 0 {
77            return false;
78        }
79
80        let friends: Vec<SharedEntity> = friends.iter().filter_map(EntityMemory::get).collect();
81        let me = ctx.mob().id();
82        if friends.iter().any(|friend| chasing(friend) == Some(me)) {
83            run_away(ctx);
84            return true;
85        }
86
87        // Vanilla parity: the preference for somebody already being chased,
88        // then `findAny` over the rest -- which is what turns a crowd of
89        // children into one game with one `it` rather than a dozen pairs.
90        let target = being_chased(&friends).or_else(|| friends.first().cloned());
91        if let Some(kid) = target {
92            chase_kid(ctx, &kid);
93        }
94        true
95    }
96
97    fn debug_name(&self) -> &'static str {
98        "PlayTagWithOtherKids"
99    }
100}

102/// Vanilla parity: the flee branch, which only accepts a spot still inside the
103/// village.
104fn run_away(ctx: &BrainContext<'_>) {
105    let mob = ctx.mob();
106    for _ in 0..FLEE_ATTEMPTS {
107        let Some(pos) = land_random_pos(mob, MAX_FLEE_XZ_DIST, MAX_FLEE_Y_DIST) else {
108            continue;
109        };
110        if !ctx
111            .world()
112            .is_village(BlockPos::containing(pos.x, pos.y, pos.z))
113        {
114            continue;
115        }
116        ctx.brain().set_memory(
117            memory_module_types::WALK_TARGET,
118            WalkTarget::of_position(pos, PLAY_SPEED_MODIFIER, 0),
119        );
120        break;
121    }
122}

124/// Vanilla parity: the private `PlayTagWithOtherKids.chaseKid`.
125fn chase_kid(ctx: &BrainContext<'_>, kid: &SharedEntity) {
126    let brain = ctx.brain();
127    brain.set_memory(
128        memory_module_types::INTERACTION_TARGET,
129        utils::remember(kid),
130    );
131    brain.set_memory(
132        memory_module_types::LOOK_TARGET,
133        PositionTracker::of_entity(kid, true),
134    );
135    brain.set_memory(
136        memory_module_types::WALK_TARGET,
137        WalkTarget::new(
138            PositionTracker::of_entity(kid, false),
139            PLAY_SPEED_MODIFIER,
140            1,
141        ),
142    );
143}

145/// Vanilla parity: the private `PlayTagWithOtherKids.findSomeoneBeingChased`,
146/// which takes the least-chased child that somebody is already after and that
147/// has not already drawn a crowd.
148fn being_chased(friends: &[SharedEntity]) -> Option<SharedEntity> {
149    let mut chasers: FxHashMap<i32, usize> = FxHashMap::default();
150    for friend in friends {
151        if let Some(chased) = chasing(friend) {
152            *chasers.entry(chased).or_default() += 1;
153        }
154    }
155
156    let mut candidates: Vec<(i32, usize)> = chasers
157        .into_iter()
158        .filter(|&(_, count)| count > 0 && count <= MAX_CHASERS_PER_TARGET)
159        .collect();
160    // Vanilla sorts by chaser count and takes the first, so the child with the
161    // shortest tail behind it is the one joined.
162    candidates.sort_unstable_by_key(|&(id, count)| (count, id));
163    let least_chased = candidates.first()?.0;
164    friends
165        .iter()
166        .find(|friend| friend.id() == least_chased)
167        .cloned()
168}

170/// Who this child is chasing, if anybody.
171///
172/// Vanilla parity: the private `whoAreYouChasing` and `isChasingSomeone`, both
173/// of which read the other child's own `INTERACTION_TARGET`.
174fn chasing(friend: &SharedEntity) -> Option<i32> {
175    let brain = friend.as_mob().and_then(Mob::brain)?;
176    brain
177        .get_memory(memory_module_types::INTERACTION_TARGET)
178        .map(|target| target.id())
179}
