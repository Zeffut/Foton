20/// How many times `getRandomSwimmablePos` re-rolls before giving up.
21///
22/// Vanilla parity: the `attempts++ < 10` of `BehaviorUtils.getRandomSwimmablePos`.
23const RANDOM_SWIMMABLE_POS_ATTEMPTS: usize = 10;
24
25/// Vanilla parity: `RandomStroll.MAX_XZ_DIST`.
26const MAX_XZ_DIST: i32 = 10;
27/// Vanilla parity: `RandomStroll.MAX_Y_DIST`.
28const MAX_Y_DIST: i32 = 7;
29/// Vanilla parity: `RandomStroll.SWIM_XY_DISTANCE_TIERS`.
30///
31/// A swimming stroll walks outward through these tiers, each one aimed along
32/// the direction the last found, so the mob drifts in a line rather than
33/// picking one point at random and turning for it.
34const SWIM_XY_DISTANCE_TIERS: [(i32, i32); 6] = [(1, 1), (3, 3), (5, 5), (6, 5), (7, 7), (10, 7)];
35
36/// Vanilla parity: the `-2` flying height of `RandomStroll.getTargetFlyPos`,
37/// which is what keeps a wandering flier drifting gently downward.
38const FLY_TARGET_HEIGHT: i32 = -2;
39/// Vanilla parity: the `(float) (Math.PI / 2)` of `getTargetFlyPos`.
40const FLY_MAX_XZ_RADIANS_FROM_DIR: f64 = FRAC_PI_2;

42/// Where a stroll aims for.
43type TargetPicker = Box<dyn Fn(&dyn PathfinderMob) -> Option<DVec3> + Send>;
44/// Whether a stroll may run at all.
45type StrollGuard = Box<dyn Fn(&dyn PathfinderMob) -> bool + Send>;

47/// Sets `WALK_TARGET` to somewhere nearby, once.
48///
49/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.RandomStroll`.
50pub struct RandomStroll {
51    speed_modifier: f64,
52    fetch_target_pos: TargetPicker,
53    can_run: StrollGuard,
54}
55
56impl RandomStroll {
57    /// Strolls up to ten blocks out.
58    ///
59    /// Vanilla parity: `RandomStroll.stroll(float)`.
60    #[must_use]
61    pub fn stroll(speed_modifier: f64) -> Self {
62        Self::stroll_within(speed_modifier, MAX_XZ_DIST, MAX_Y_DIST)
63    }
64
65    /// Strolls no further than the given distances.
66    ///
67    /// Vanilla parity: `RandomStroll.stroll(float, int, int)`.
68    #[must_use]
69    pub fn stroll_within(
70        speed_modifier: f64,
71        max_horizontal_distance: i32,
72        max_vertical_distance: i32,
73    ) -> Self {
74        Self {
75            speed_modifier,
76            fetch_target_pos: Box::new(move |mob| {
77                land_random_pos(mob, max_horizontal_distance, max_vertical_distance)
78            }),
79            can_run: Box::new(|_| true),
80        }
81    }
82
83    /// Refuses to stroll out of water.
84    ///
85    /// Vanilla parity: the `mayStrollFromWater = false` overload.
86    #[must_use]
87    pub fn not_from_water(mut self) -> Self {
88        self.can_run = Box::new(|mob| !mob.is_in_water());
89        self
90    }
91
92    /// Drifts to a spot in the air ahead of where the mob is looking.
93    ///
94    /// Vanilla parity: `RandomStroll.fly(float)`, whose `getTargetFlyPos` aims
95    /// the search along the view vector inside a quarter turn either way, so a
96    /// flier wanders forward rather than doubling back on itself.
97    #[must_use]
98    pub fn fly(speed_modifier: f64) -> Self {
99        Self {
100            speed_modifier,
101            fetch_target_pos: Box::new(|mob| {
102                // Vanilla parity: `body.getViewVector(0.0F)`, whose zero
103                // partial tick reads the *previous* tick's rotation.
104                let (old_yaw, old_pitch) = mob.base().old_rotation();
105                let view = mob.calculate_view_vector(old_pitch, old_yaw);
106                air_and_water_random_pos(
107                    mob,
108                    MAX_XZ_DIST,
109                    MAX_Y_DIST,
110                    FLY_TARGET_HEIGHT,
111                    view.x,
112                    view.z,
113                    FLY_MAX_XZ_RADIANS_FROM_DIR,
114                )
115            }),
116            can_run: Box::new(|_| true),
117        }
118    }
119
120    /// Strolls along the water, and only while in it.
121    ///
122    /// Vanilla parity: `RandomStroll.swim(float)`.
123    #[must_use]
124    pub fn swim(speed_modifier: f64) -> Self {
125        Self {
126            speed_modifier,
127            fetch_target_pos: Box::new(target_swim_pos),
128            can_run: Box::new(<dyn PathfinderMob>::is_in_water),
129        }
130    }
131}

133/// Vanilla parity: `RandomStroll.getTargetSwimPos`.
134///
135/// Each tier after the first extends the previous answer outward rather than
136/// rolling again, and the walk stops at the first tier that leaves the water or
137/// the mob's home radius -- so the target is the furthest point along a line
138/// that is still swimmable.
139fn target_swim_pos(mob: &dyn PathfinderMob) -> Option<DVec3> {
140    let world = mob.level()?;
141    let mut fallback: Option<DVec3> = None;
142    let mut target_pos: Option<DVec3> = None;
143
144    for (horizontal, vertical) in SWIM_XY_DISTANCE_TIERS {
145        target_pos = match fallback {
146            None => random_swimmable_pos(mob, horizontal, vertical),
147            Some(previous) => {
148                let step = (previous - mob.position()).normalize_or_zero()
149                    * DVec3::new(
150                        f64::from(horizontal),
151                        f64::from(vertical),
152                        f64::from(horizontal),
153                    );
154                Some(mob.position() + step)
155            }
156        };
157
158        let leaves_water = target_pos.is_none_or(|pos| {
159            !world
160                .get_block_state(BlockPos::containing(pos.x, pos.y, pos.z))
161                .get_fluid_state()
162                .is_water()
163        });
164        let outside_home = target_pos.is_some_and(|pos| {
165            mob.has_home()
166                && block_center_distance_sqr(mob.home_position(), mob.position())
167                    < (f64::from(mob.home_radius()) + f64::from(horizontal) + 1.0).powi(2)
168                && !mob.is_within_home_vec(pos)
169        });
170        if leaves_water || outside_home {
171            return fallback;
172        }
173
174        fallback = target_pos;
175    }
176
177    target_pos
178}

180/// Vanilla parity: `BehaviorUtils.getRandomSwimmablePos`, which re-rolls up to
181/// ten times for a point a swimmer can actually path into.
182fn random_swimmable_pos(
183    mob: &dyn PathfinderMob,
184    horizontal_dist: i32,
185    vertical_dist: i32,
186) -> Option<DVec3> {
187    let world = mob.level()?;
188    for _ in 0..=RANDOM_SWIMMABLE_POS_ATTEMPTS {
189        let pos = default_random_pos(mob, horizontal_dist, vertical_dist)?;
190        let state = world.get_block_state(BlockPos::containing(pos.x, pos.y, pos.z));
191        if BLOCK_BEHAVIORS
192            .get_behavior(state.get_block())
193            .is_pathfindable(state, PathComputationType::Water)
194        {
195            return Some(pos);
196        }
197    }
198    None
199}
200
201fn block_center_distance_sqr(pos: BlockPos, target: DVec3) -> f64 {
202    let (x, y, z) = pos.get_center();
203    DVec3::new(x, y, z).distance_squared(target)
204}

206impl Trigger for RandomStroll {
207    fn required_memories(&self) -> Vec<MemoryModuleId> {
208        vec![memory_module_types::WALK_TARGET.id()]
209    }
210
211    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
212        // Vanilla parity: the `i.absent(WALK_TARGET)` of the builder group.
213        if ctx
214            .brain()
215            .has_memory_value(memory_module_types::WALK_TARGET.id())
216        {
217            return false;
218        }
219        if !(self.can_run)(ctx.mob()) {
220            return false;
221        }
222
223        let target = (self.fetch_target_pos)(ctx.mob());
224        ctx.brain().set_memory_or_erase(
225            memory_module_types::WALK_TARGET,
226            target.map(|pos| WalkTarget::of_position(pos, self.speed_modifier, 0)),
227        );
228        true
229    }
230
231    fn debug_name(&self) -> &'static str {
232        "RandomStroll"
233    }
