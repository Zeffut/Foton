13/// Keeps a baby trailing the nearest adult of its kind.
14///
15/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.BabyFollowAdult`.
16pub struct BabyFollowAdult {
17    follow_range: UniformIntProvider,
18    speed_modifier: SpeedModifier,
19    nearest_visible_adult: MemoryModuleType<EntityMemory>,
20    target_eye: bool,
21}
22
23/// How fast the baby closes the gap, which the body may decide per tick.
24type SpeedModifier = Box<dyn Fn(&dyn PathfinderMob) -> f64 + Send>;
25
26impl BabyFollowAdult {
27    /// Vanilla parity: `BabyFollowAdult.create(UniformInt, float)`.
28    #[must_use]
29    pub fn new(follow_range: UniformIntProvider, speed_modifier: f64) -> Self {
30        Self::variable(follow_range, move |_| speed_modifier)
31    }
32
33    /// Follows at a speed the baby picks.
34    ///
35    /// Vanilla parity: the `Function<LivingEntity, Float> speedModifier`
36    /// overload -- an axolotl calf paddles after its parent faster in water
37    /// than it waddles after it on land.
38    #[must_use]
39    pub fn variable(
40        follow_range: UniformIntProvider,
41        speed_modifier: impl Fn(&dyn PathfinderMob) -> f64 + Send + 'static,
42    ) -> Self {
43        Self {
44            follow_range,
45            speed_modifier: Box::new(speed_modifier),
46            nearest_visible_adult: memory_module_types::NEAREST_VISIBLE_ADULT,
47            target_eye: false,
48        }
49    }
50
51    /// Follows the adult a different memory points at.
52    ///
53    /// Vanilla parity: the `nearestVisibleType` argument of the four-argument
54    /// `BabyFollowAdult.create`.
55    #[must_use]
56    pub const fn following(mut self, memory: MemoryModuleType<EntityMemory>) -> Self {
57        self.nearest_visible_adult = memory;
58        self
59    }
60
61    /// Follows the adult's eyes rather than its feet.
62    ///
63    /// Vanilla parity: the `targetEye` argument of the four-argument
64    /// `BabyFollowAdult.create`. A ghastling hovers, and the feet of the
65    /// four-block-tall adult it is following are four blocks below where it
66    /// wants to be.
67    #[must_use]
68    pub const fn targeting_eye(mut self) -> Self {
69        self.target_eye = true;
70        self
71    }
72}
73
74impl Trigger for BabyFollowAdult {
75    fn required_memories(&self) -> Vec<MemoryModuleId> {
76        vec![
77            self.nearest_visible_adult.id(),
78            memory_module_types::LOOK_TARGET.id(),
79            memory_module_types::WALK_TARGET.id(),
80        ]
81    }
82
83    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
84        let brain = ctx.brain();
85        if brain.has_memory_value(memory_module_types::WALK_TARGET.id()) {
86            return false;
87        }
88        if !ctx.mob().is_baby() {
89            return false;
90        }
91        let Some(adult) = brain
92            .get_memory(self.nearest_visible_adult)
93            .and_then(|memory| memory.get())
94        else {
95            return false;
96        };
97
98        let distance_sqr = adult.position().distance_squared(ctx.mob().position());
99        let outer = f64::from(self.follow_range.max_inclusive + 1);
100        let inner = f64::from(self.follow_range.min_inclusive);
101        if distance_sqr >= outer * outer || distance_sqr < inner * inner {
102            return false;
103        }
104
105        brain.set_memory(
106            memory_module_types::LOOK_TARGET,
107            PositionTracker::of_entity_targeting(&adult, true, self.target_eye),
108        );
109        brain.set_memory(
110            memory_module_types::WALK_TARGET,
111            WalkTarget::new(
112                PositionTracker::of_entity_targeting(&adult, self.target_eye, self.target_eye),
113                (self.speed_modifier)(ctx.mob()),
114                self.follow_range.min_inclusive - 1,
115            ),
116        );
117        true
118    }
119
120    fn debug_name(&self) -> &'static str {
121        "BabyFollowAdult"
122    }
123}
