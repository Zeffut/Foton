1//! The long jump: `LongJumpUtil`, `LongJumpToRandomPos`,
2//! `LongJumpToPreferredBlock` and `LongJumpMidJump`.
3//!
4//! Vanilla splits these across four files, but they are one mechanism -- a mob
5//! picks a landing spot it cannot walk to, solves a ballistic arc that clears
6//! everything between, crouches for two seconds and throws itself -- and the
7//! three behaviors only make sense read together.

29/// Vanilla parity: `LongJumpToRandomPos.FIND_JUMP_TRIES`.
30const FIND_JUMP_TRIES: i32 = 20;
31/// Vanilla parity: `LongJumpToRandomPos.PREPARE_JUMP_DURATION`.
32const PREPARE_JUMP_DURATION: i64 = 40;
33/// Vanilla parity: `LongJumpToRandomPos.MIN_PATHFIND_DISTANCE_TO_VALID_JUMP`.
34const MIN_PATHFIND_DISTANCE_TO_VALID_JUMP: i32 = 8;
35/// Vanilla parity: `LongJumpToRandomPos.TIME_OUT_DURATION`.
36const LONG_JUMP_TIME_OUT: i32 = 200;
37/// Vanilla parity: `LongJumpMidJump.TIME_OUT_DURATION`.
38const MID_JUMP_TIME_OUT: i32 = 100;
39/// Vanilla parity: `LongJumpToRandomPos.ALLOWED_ANGLES`, in degrees.
40const ALLOWED_ANGLES: [i32; 4] = [65, 70, 75, 80];
41/// Vanilla parity: the `scale(0.95F)` that ends `calculateJumpVectorForAngle`.
42const JUMP_VELOCITY_SCALE: f64 = 0.95;
43/// Vanilla parity: the `0.5` the target is pulled back by so the mob lands on
44/// the block rather than into its far face.
45const TARGET_PULLBACK: f64 = 0.5;
46/// Vanilla parity: the `minDimension * 0.9F` step of `isClearTransition`.
47const TRANSITION_STEP_SCALE: f64 = 0.9;
48/// Vanilla parity: the `multiply(0.1F, 1.0, 0.1F)` a landing mob is slowed by.
49const LANDING_HORIZONTAL_DAMPING: f64 = 0.1;
50/// Vanilla parity: the `2.0F` volume of the landing sound.
51const LANDING_SOUND_VOLUME: f32 = 2.0;

53/// A candidate landing block and how heavily it is weighted.
54///
55/// Vanilla parity: `LongJumpToRandomPos.PossibleJump`, whose weight is the
56/// squared distance -- so a mob prefers the longest jump it can make.
57#[derive(Debug, Clone, Copy)]
58struct PossibleJump {
59    target_pos: BlockPos,
60    weight: i32,
61}

63/// Whether a mob will accept landing on a given block.

65/// Vanilla parity: the `BiPredicate<E, BlockPos> acceptableLandingSpot`.

68/// Vanilla parity: `LongJumpToRandomPos.defaultAcceptableLandingSpot`.
69#[must_use]
70pub fn default_acceptable_landing_spot(body: &dyn PathfinderMob, target_pos: BlockPos) -> bool {
71    let Some(world) = body.level() else {
72        return false;
73    };
74    if !world.get_block_state(target_pos.below()).is_solid_render() {
75        return false;
76    }
77
78    let mut context = PathfindingContext::new(world.as_ref(), body.block_position());
79    let path_type = WalkPathEvaluator::path_type_static(&mut context, target_pos);
80    body.get_pathfinding_malus(path_type) == 0.0
81}

155    /// Replaces the test for a block worth landing on.

157    /// Vanilla parity: the six-argument constructor's `acceptableLandingSpot`.

167    /// Aims most jumps at one kind of block.
168    ///
169    /// Vanilla parity: `LongJumpToPreferredBlock`, which is how a frog ends up
170    /// on the lily pads and big dripleaves rather than anywhere it can reach.
171    #[must_use]
172    pub fn preferring(mut self, preferred_block_tag: Identifier, chance: f32) -> Self {

174        self.preferred_blocks_chance = chance;
175        self

178    /// Puts the cooldown on at half length, the way vanilla does whenever the
179    /// jump cannot start or cannot continue.
180    fn set_half_cooldown(&self, ctx: &BrainContext<'_>) {
181        ctx.brain().set_memory(
182            memory_module_types::LONG_JUMP_COOLDOWN_TICKS,
183            sample(self.time_between_long_jumps) / 2,
184        );
185    }
186
187    /// Vanilla parity: `LongJumpToRandomPos.getJumpCandidate` plus the
188    /// `LongJumpToPreferredBlock` override.
189    fn take_jump_candidate(&mut self, ctx: &BrainContext<'_>) -> Option<PossibleJump> {
190        if !self.currently_wanting_preferred_ones {
191            return take_weighted(&mut self.jump_candidates);
192        }
193
194        let Some(tag) = self.preferred_block_tag.clone() else {
195            return take_weighted(&mut self.jump_candidates);
196        };
197        let world = ctx.world();
198
199        while let Some(candidate) = take_weighted(&mut self.jump_candidates) {
200            if world
201                .get_block_state(candidate.target_pos.below())
202                .get_block()
203                .has_tag(&tag)
204            {
205                return Some(candidate);
206            }
207            self.not_preferred_jump_candidates.push(candidate);
208        }
209
210        if self.not_preferred_jump_candidates.is_empty() {
211            None
212        } else {
213            Some(self.not_preferred_jump_candidates.remove(0))
214        }
215    }
216
217    /// Vanilla parity: `LongJumpToRandomPos.pickCandidate`.
218    fn pick_candidate(&mut self, ctx: &BrainContext<'_>, timestamp: i64) {
219        while !self.jump_candidates.is_empty() || !self.not_preferred_jump_candidates.is_empty() {
220            let Some(candidate) = self.take_jump_candidate(ctx) else {
221                return;
222            };
223            let target_pos = candidate.target_pos;
224            if !self.is_acceptable_landing_position(ctx, target_pos) {
225                continue;
226            }
227
228            let (cx, cy, cz) = target_pos.get_center();
229            let Some(jump_vector) = self.calculate_optimal_jump_vector(ctx, DVec3::new(cx, cy, cz))
230            else {
231                continue;
232            };
233
234            ctx.brain().set_memory(
235                memory_module_types::LOOK_TARGET,
236                PositionTracker::of_block(target_pos),
237            );
238            // Vanilla only jumps to somewhere it could not simply walk to.
239            let walkable = ctx.mob().create_path_to(target_pos, 0).is_some_and(|path| {
240                path.can_reach()
241                    && path.node_count() <= MIN_PATHFIND_DISTANCE_TO_VALID_JUMP as usize
242            });
243            if !walkable {
244                self.chosen_jump = Some(jump_vector);
245                self.prepare_jump_start = timestamp;
246                return;
247            }
248        }
249    }
250
251    /// Vanilla parity: `LongJumpToRandomPos.isAcceptableLandingPosition`, which
252    /// refuses a block directly above or below the mob.
253    fn is_acceptable_landing_position(&self, ctx: &BrainContext<'_>, target_pos: BlockPos) -> bool {
254        let body_pos = ctx.mob().block_position();
255        if body_pos.x() == target_pos.x() && body_pos.z() == target_pos.z() {
256            return false;
257        }
258        (self.acceptable_landing_spot)(ctx.mob(), target_pos)
259    }
260
261    /// Vanilla parity: `LongJumpToRandomPos.calculateOptimalJumpVector`, which
262    /// tries the four launch angles in a random order and takes the first that
263    /// both reaches and clears.
264    fn calculate_optimal_jump_vector(
265        &self,
266        ctx: &BrainContext<'_>,
267        target_pos: DVec3,
268    ) -> Option<DVec3> {
269        let body = ctx.mob();
270        let mut angles = ALLOWED_ANGLES;
271        for index in (1..angles.len()).rev() {
272            angles.swap(index, rand::random_range(0..=index));
273        }
274
275        let jump_strength = body
276            .attributes()
277            .lock()
278            .required_value(vanilla_attributes::JUMP_STRENGTH);
279        let max_jump_velocity = jump_strength * f64::from(self.max_jump_velocity_multiplier);
280
281        angles.into_iter().find_map(|angle| {
282            calculate_jump_vector_for_angle(body, target_pos, max_jump_velocity, angle, true)
283        })
284    }
285}

287impl TimedBehavior for LongJumpToRandomPos {
288    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
289        LONG_JUMP_ENTRY_CONDITION
290    }
291
292    fn duration(&self) -> (i32, i32) {
293        (LONG_JUMP_TIME_OUT, LONG_JUMP_TIME_OUT)
294    }
295
296    fn check_extra_start_conditions(&mut self, ctx: &BrainContext<'_>) -> bool {
297        let body = ctx.mob();
298        let on_honey = ctx
299            .world()
300            .get_block_state(body.block_position())
301            .get_block()
302            == &vanilla_blocks::HONEY_BLOCK;
303        let can_start = body.on_ground() && !body.is_in_water() && !body.is_in_lava() && !on_honey;
304        if !can_start {
305            self.set_half_cooldown(ctx);
306        }
307        can_start
308    }
309
310    fn can_still_use(&mut self, ctx: &BrainContext<'_>) -> bool {
311        let body = ctx.mob();
312        let is_valid = self.initial_position == Some(body.position())
313            && self.find_jump_tries > 0
314            && !body.is_in_water()
315            && (self.chosen_jump.is_some() || !self.jump_candidates.is_empty());
316
317        if !is_valid
318            && !ctx
319                .brain()
320                .has_memory_value(memory_module_types::LONG_JUMP_MID_JUMP.id())
321        {
322            self.set_half_cooldown(ctx);
323            ctx.brain()
324                .erase_memory(memory_module_types::LOOK_TARGET.id());
325        }
326
327        is_valid
328    }
329
330    fn start(&mut self, ctx: &BrainContext<'_>) {
331        let body = ctx.mob();
332        self.chosen_jump = None;
333        self.find_jump_tries = FIND_JUMP_TRIES;
334        self.initial_position = Some(body.position());
335        self.not_preferred_jump_candidates.clear();
336        self.currently_wanting_preferred_ones =
337            rand::random::<f32>() < self.preferred_blocks_chance;
338
339        let mob_pos = body.block_position();
340        self.jump_candidates.clear();
341        for x in (mob_pos.x() - self.max_long_jump_width)..=(mob_pos.x() + self.max_long_jump_width)
342        {
343            for y in (mob_pos.y() - self.max_long_jump_height)
344                ..=(mob_pos.y() + self.max_long_jump_height)
345            {
346                for z in (mob_pos.z() - self.max_long_jump_width)
347                    ..=(mob_pos.z() + self.max_long_jump_width)
348                {
349                    let pos = BlockPos::new(x, y, z);
350                    if pos == mob_pos {
351                        continue;
352                    }
353                    let weight = distance_squared_ceil(mob_pos, pos);
354                    self.jump_candidates.push(PossibleJump {
355                        target_pos: pos,
356                        weight,
357                    });
358                }
359            }
360        }
361    }
362
363    fn tick(&mut self, ctx: &BrainContext<'_>) {
364        let timestamp = ctx.game_time();
365        let Some(chosen_jump) = self.chosen_jump else {
366            self.find_jump_tries -= 1;
367            self.pick_candidate(ctx, timestamp);
368            return;
369        };
370
371        if timestamp - self.prepare_jump_start < PREPARE_JUMP_DURATION {
372            return;
373        }
374
375        let body = ctx.mob();
376        let (_, pitch) = body.rotation();
377        body.set_rotation((body.y_body_rot(), pitch));
378        body.set_discard_friction(true);
379
380        // Vanilla parity: the jump boost lengthens the arc without changing its
381        // direction.
382        let length = chosen_jump.length();
383        let boosted = length + f64::from(body.get_jump_boost_power());
384        body.set_velocity(chosen_jump * (boosted / length));
385
386        ctx.brain()
387            .set_memory(memory_module_types::LONG_JUMP_MID_JUMP, true);
388        body.play_sound(self.jump_sound, 1.0, 1.0);
389    }
390
391    fn debug_name(&self) -> &'static str {
392        "LongJumpToRandomPos"
393    }
394}

476/// Vanilla parity: `LongJumpUtil.calculateJumpVectorForAngle`.
477///
478/// Solves the projectile equation for the launch speed that lands on
479/// `target_pos` at `angle`, then walks the resulting arc to check it clears.
480/// `None` means the target is out of reach at this angle, or something is in
481/// the way.
482#[must_use]
483pub fn calculate_jump_vector_for_angle(
484    body: &dyn PathfinderMob,
485    target_pos: DVec3,
486    max_jump_velocity: f64,
487    angle: i32,
488    check_collision: bool,
489) -> Option<DVec3> {
490    let mob_pos = body.position();
491    let direction_plane = DVec3::new(target_pos.x - mob_pos.x, 0.0, target_pos.z - mob_pos.z)
492        .normalize_or_zero()
493        * TARGET_PULLBACK;
494    let direction = (target_pos - direction_plane) - mob_pos;
495
496    let angrad = f64::from(angle) * PI / 180.0;
497    let xz_ang = direction.z.atan2(direction.x);
498    let horizontal = DVec3::new(direction.x, 0.0, direction.z);
499    let r2 = horizontal.length_squared();
500    let r = r2.sqrt();
501    let y = direction.y;
502    let g = body.get_gravity();
503
504    let sin2ang = (2.0 * angrad).sin();
505    let cosangsqr = angrad.cos().powi(2);
506    let sinangrad = angrad.sin();
507    let cosangrad = angrad.cos();
508    let sinxz_ang = xz_ang.sin();
509    let cosxz_ang = xz_ang.cos();
510
511    let v0sqr = r2 * g / (r * sin2ang - 2.0 * y * cosangsqr);
512    if v0sqr < 0.0 {
513        return None;
514    }
515    let v0 = v0sqr.sqrt();
516    if v0 > max_jump_velocity {
517        return None;
518    }
519
520    let v0r = v0 * cosangrad;
521    let v0y = v0 * sinangrad;
522
523    if check_collision {
524        let samples = (r / v0r).ceil() as i32 * 2;
525        let mut ri = 0.0;
526        let mut previous_pos: Option<DVec3> = None;
527        let dimensions = body.dimensions_for_pose(EntityPose::LongJumping);
528
529        for _ in 0..(samples - 1) {
530            ri += r / f64::from(samples);
531            let yi =
532                sinangrad / cosangrad * ri - ri.powi(2) * g / (2.0 * v0sqr * cosangrad.powi(2));
533            let sample_pos = DVec3::new(
534                mob_pos.x + ri * cosxz_ang,
535                mob_pos.y + yi,
536                mob_pos.z + ri * sinxz_ang,
537            );
538            if let Some(previous) = previous_pos
539                && !is_clear_transition(body, dimensions, previous, sample_pos)
540            {
541                return None;
542            }
543            previous_pos = Some(sample_pos);
544        }
545    }
546
547    Some(DVec3::new(v0r * cosxz_ang, v0y, v0r * sinxz_ang) * JUMP_VELOCITY_SCALE)
548}

550/// Vanilla parity: `LongJumpUtil.isClearTransition`, which steps the mob's own
551/// hitbox along one segment of the arc.
552fn is_clear_transition(
553    body: &dyn PathfinderMob,
554    dimensions: EntityDimensions,
555    position1: DVec3,
556    position2: DVec3,
557) -> bool {
558    let Some(world) = body.level() else {
559        return false;
560    };
561    let direction = position2 - position1;
562    let min_dimension = f64::from(dimensions.width.min(dimensions.height));
563    if min_dimension <= 0.0 {
564        return true;
565    }
566    let checks = (direction.length() / min_dimension).ceil() as i32;
567    let normalized = direction.normalize_or_zero();
568    let mut next = position1;
569
570    for index in 0..checks {
571        next = if index == checks - 1 {
572            position2
573        } else {
574            next + normalized * (min_dimension * TRANSITION_STEP_SCALE)
575        };
576        if has_collision(
577            &WorldCollisionProvider::new(&world),
578            body.make_bounding_box_at(next),
579        ) {
580            return false;
581        }
582    }
583
584    true
585}

587/// Vanilla parity: `WeightedRandom.getRandomItem` followed by the `remove`.
588fn take_weighted(candidates: &mut Vec<PossibleJump>) -> Option<PossibleJump> {
589    let total: i64 = candidates.iter().map(|jump| i64::from(jump.weight)).sum();
590    if total <= 0 {
591        return (!candidates.is_empty()).then(|| candidates.remove(0));
592    }
593
594    let mut roll = rand::random_range(0..total);
595    for (index, candidate) in candidates.iter().enumerate() {
596        roll -= i64::from(candidate.weight);
597        if roll < 0 {
598            return Some(candidates.remove(index));
599        }
600    }
601    None
602}

604/// Draws one value from an inclusive range.
605///
606/// Vanilla parity: `UniformInt.sample(RandomSource)`. Foton's provider takes an
607/// explicit generator; the jump cooldown is incidental live randomness, so it
608/// uses the runtime one rather than a seeded stream.
609fn sample(provider: UniformIntProvider) -> i32 {
610    rand::random_range(provider.min_inclusive..=provider.max_inclusive)
611}

613/// Vanilla parity: the `Mth.ceil(mobPos.distSqr(pos))` weight.
614fn distance_squared_ceil(from: BlockPos, to: BlockPos) -> i32 {
615    let dx = f64::from(to.x() - from.x());
616    let dy = f64::from(to.y() - from.y());
617    let dz = f64::from(to.z() - from.z());
618    dx.mul_add(dx, dy.mul_add(dy, dz * dz)).ceil() as i32
619}
