1//! Vanilla `SetWalkTargetFromLookTarget`.
2
3use super::{BrainContext, Trigger};
4use crate::entity::PathfinderMob;
5use crate::entity::ai::brain::memory::{MemoryModuleId, WalkTarget, memory_module_types};
6
7/// Whether the body may walk to what it is looking at, and how fast.
8type WalkGuard = Box<dyn Fn(&dyn PathfinderMob) -> Option<f64> + Send>;
9
10/// Walks toward whatever is in `LOOK_TARGET`.
11///
12/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.SetWalkTargetFromLookTarget`.
13pub struct SetWalkTargetFromLookTarget {
14    speed_modifier: WalkGuard,
15    close_enough_distance: i32,
16}
17
18impl SetWalkTargetFromLookTarget {
19    /// Always walks, at a fixed speed.
20    ///
21    /// Vanilla parity: `SetWalkTargetFromLookTarget.create(float, int)`.
22    #[must_use]
23    pub fn new(speed_modifier: f64, close_enough_distance: i32) -> Self {
24        Self::conditional(move |_| Some(speed_modifier), close_enough_distance)
25    }
26
27    /// Walks only when `speed_modifier` answers with a speed.
28    ///
29    /// Vanilla parity: the `Predicate` plus `Function<LivingEntity, Float>`
30    /// overload, collapsed into one closure because a mob that must not walk
31    /// and a mob with no speed are the same case.
32    #[must_use]
33    pub fn conditional(
34        speed_modifier: impl Fn(&dyn PathfinderMob) -> Option<f64> + Send + 'static,
35        close_enough_distance: i32,
36    ) -> Self {
37        Self {
38            speed_modifier: Box::new(speed_modifier),
39            close_enough_distance,
40        }
41    }
42}
43
44impl Trigger for SetWalkTargetFromLookTarget {
45    fn required_memories(&self) -> Vec<MemoryModuleId> {
46        vec![
47            memory_module_types::WALK_TARGET.id(),
48            memory_module_types::LOOK_TARGET.id(),
49        ]
50    }
51
52    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
53        let brain = ctx.brain();
54        if brain.has_memory_value(memory_module_types::WALK_TARGET.id()) {
55            return false;
56        }
57        let Some(look_target) = brain.get_memory(memory_module_types::LOOK_TARGET) else {
58            return false;
59        };
60        let Some(speed_modifier) = (self.speed_modifier)(ctx.mob()) else {
61            return false;
62        };
63
64        brain.set_memory(
65            memory_module_types::WALK_TARGET,
66            WalkTarget::new(look_target, speed_modifier, self.close_enough_distance),
67        );
68        true
69    }
70
71    fn debug_name(&self) -> &'static str {
72        "SetWalkTargetFromLookTarget"
73    }
74}
