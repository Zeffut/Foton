11/// Vanilla parity: `SleepInBed.COOLDOWN_AFTER_BEING_WOKEN`.
12const COOLDOWN_AFTER_BEING_WOKEN: i64 = 100;
13/// Vanilla parity: the `timestamp + 40L` of `SleepInBed.stop`.
14const COOLDOWN_AFTER_GETTING_UP: i64 = 40;
15/// How near the bed a villager has to stand before it climbs in.
16///
17/// Vanilla parity: the `closerToCenterThan(body.position(), 2.0)` of
18/// `SleepInBed.checkExtraStartConditions`.
19const CLOSE_ENOUGH_TO_CLIMB_IN: f64 = 2.0;
20/// Vanilla parity: the `closerToCenterThan(body.position(), 1.14)` of
21/// `SleepInBed.canStillUse`.
22const CLOSE_ENOUGH_TO_STAY_IN: f64 = 1.14;
23/// Vanilla parity: the `body.getY() > bedPos.getY() + 0.4` of `canStillUse`,
24/// which is how a villager notices the bed under it has been mined.
25const MIN_HEIGHT_ABOVE_BED: f64 = 0.4;
26
27/// Puts the body in the bed it remembers as home.
28///
29/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.SleepInBed`. It runs
30/// in the REST package, so a villager only ever tries this once the
31/// `villager_schedule` timeline has put it in [`Activity::Rest`], and
32/// [`WakeUp`] in the core package gets it out again the moment the schedule
33/// moves on.
34///
35/// MISSING FOUNDATION: vanilla's `start` also closes the doors it opened on the
36/// way home, through `InteractWithDoor.closeDoorsThatIHaveOpenedOrPassedThrough`.
37/// Getting into bed is also when a villager shuts the doors it left open on
38/// the way home -- the same `DOORS_TO_CLOSE` pass [`InteractWithDoor`] runs as
39/// it walks, which is why that one is public.
40///
41/// [`InteractWithDoor`]: super::InteractWithDoor
42pub struct SleepInBed {
43    next_ok_start_time: i64,
44}
45

53    }
54}

62/// Vanilla parity: the `ImmutableMap` handed to `SleepInBed`'s `super(...)`.
63const SLEEP_ENTRY_CONDITION: &[(MemoryModuleId, MemoryStatus)] = &[
64    (memory_module_types::HOME.id(), MemoryStatus::ValuePresent),
65    (
66        memory_module_types::LAST_WOKEN.id(),
67        MemoryStatus::Registered,
68    ),
69    (
70        memory_module_types::LAST_SLEPT.id(),
71        MemoryStatus::Registered,
72    ),
73    (
74        memory_module_types::WALK_TARGET.id(),
75        MemoryStatus::Registered,
76    ),
77    (
78        memory_module_types::CANT_REACH_WALK_TARGET_SINCE.id(),
79        MemoryStatus::Registered,
80    ),
81];
82
83impl TimedBehavior for SleepInBed {
84    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
85        SLEEP_ENTRY_CONDITION
86    }
87
88    /// Vanilla parity: `SleepInBed.timedOut`, which returns `false` -- a
89    /// villager stays asleep until the schedule or the bed says otherwise.
90    fn times_out(&self) -> bool {
91        false
92    }
93
94    fn check_extra_start_conditions(&mut self, ctx: &BrainContext<'_>) -> bool {
95        let mob = ctx.mob();
96        if mob.is_passenger() {
97            return false;
98        }
99        let brain = ctx.brain();
100        let Some(home) = brain.get_memory(memory_module_types::HOME) else {
101            return false;
102        };
103        let world = ctx.world();
104        if home.dimension != world.key {
105            return false;
106        }
107
108        // Vanilla parity: the hundred-tick pause after being shaken awake, so a
109        // woken villager does not climb straight back in.
110        if let Some(last_woken) = brain.get_memory(memory_module_types::LAST_WOKEN) {
111            let since = world.game_time() - last_woken;
112            if since > 0 && since < COOLDOWN_AFTER_BEING_WOKEN {
113                return false;
114            }
115        }
116
117        if !utils::block_closer_to_center_than(home.pos, mob.position(), CLOSE_ENOUGH_TO_CLIMB_IN) {
118            return false;
119        }
120        let state = world.get_block_state(home.pos);
121        state.is_bed() && !state.get_value(&BlockStateProperties::OCCUPIED)
122    }
123
124    fn can_still_use(&mut self, ctx: &BrainContext<'_>) -> bool {
125        let brain = ctx.brain();
126        let Some(home) = brain.get_memory(memory_module_types::HOME) else {
127            return false;
128        };
129        let mob = ctx.mob();
130        brain.is_active(Activity::Rest)
131            && mob.position().y > f64::from(home.pos.y()) + MIN_HEIGHT_ABOVE_BED
132            && utils::block_closer_to_center_than(home.pos, mob.position(), CLOSE_ENOUGH_TO_STAY_IN)
133    }
134
135    fn start(&mut self, ctx: &BrainContext<'_>) {
136        if ctx.game_time() <= self.next_ok_start_time {
137            return;
138        }
139        let brain = ctx.brain();
140        let Some(home) = brain.get_memory(memory_module_types::HOME) else {
141            return;
142        };
143        // Vanilla parity: the `closeDoorsThatIHaveOpenedOrPassedThrough(level,
144        // body, null, null, ..)` of `SleepInBed.start` -- with no path nodes to
145        // excuse, so every door still tracked is a candidate.
146        super::close_doors_behind(ctx.world(), ctx.mob(), brain, None, None);
147        if let Err(error) = ctx.mob().start_sleeping(home.pos) {
148            log::debug!(
149                "villager {} could not lie down in its bed: {error}",
150                ctx.mob().id()
151            );
152            return;
153        }
154        brain.set_memory(memory_module_types::LAST_SLEPT, ctx.game_time());
155        brain.erase_memory(memory_module_types::WALK_TARGET.id());
156        brain.erase_memory(memory_module_types::CANT_REACH_WALK_TARGET_SINCE.id());
157    }
158
159    fn stop(&mut self, ctx: &BrainContext<'_>) {
160        if !ctx.mob().is_sleeping() {
161            return;
162        }
163        ctx.mob().stop_sleeping();
164        self.next_ok_start_time = ctx.game_time() + COOLDOWN_AFTER_GETTING_UP;
165    }
166
167    fn debug_name(&self) -> &'static str {
168        "SleepInBed"
169    }
170}

172/// Gets the body out of bed as soon as its schedule stops saying REST.
173///
174/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.WakeUp`. It sits in
175/// the core package, so it runs whatever activity the villager is in, and it is
176/// what writes `LAST_WOKEN` -- through `stopSleeping` on the villager side --
177/// that keeps `SleepInBed` from putting it straight back.
178pub struct WakeUp;
179
180impl Trigger for WakeUp {
181    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
182        let mob = ctx.mob();
183        if ctx.brain().is_active(Activity::Rest) || !mob.is_sleeping() {
184            return false;
185        }
186        mob.stop_sleeping();
187        true
188    }
189
190    fn debug_name(&self) -> &'static str {
191        "WakeUp"
192    }
193}
