12/// Vanilla parity: `VillageBoundRandomStroll.MAX_XZ_DIST`.
13const MAX_XZ_DIST: i32 = 10;
14/// Vanilla parity: `VillageBoundRandomStroll.MAX_Y_DIST`.
15const MAX_Y_DIST: i32 = 7;

17const VILLAGE_SEARCH_SECTION_RADIUS: i32 = 2;

19/// Wanders, but drifts back toward the village when it has strayed out.
20///
21/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.VillageBoundRandomStroll`.
22/// Inside a village this is a plain land stroll; outside one it aims at the
23/// nearest section that is closer to a village center, which is what keeps a
24/// panicking or idling villager from wandering off into the dark forever.
25pub struct VillageBoundRandomStroll {
26    speed_modifier: f64,
27    max_xz_dist: i32,
28    max_y_dist: i32,
29}
30
31impl VillageBoundRandomStroll {
32    /// Vanilla parity: the one-argument `VillageBoundRandomStroll.create`.
33    #[must_use]
34    pub const fn new(speed_modifier: f64) -> Self {
35        Self::with_range(speed_modifier, MAX_XZ_DIST, MAX_Y_DIST)
36    }
37
38    /// Vanilla parity: the three-argument `VillageBoundRandomStroll.create`.
39    #[must_use]
40    pub const fn with_range(speed_modifier: f64, max_xz_dist: i32, max_y_dist: i32) -> Self {
41        Self {
42            speed_modifier,
43            max_xz_dist,
44            max_y_dist,
45        }
46    }
47}

49impl Trigger for VillageBoundRandomStroll {
50    fn required_memories(&self) -> Vec<MemoryModuleId> {
51        vec![memory_module_types::WALK_TARGET.id()]
52    }
53
54    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
55        let brain = ctx.brain();
56        if brain.has_memory_value(memory_module_types::WALK_TARGET.id()) {
57            return false;
58        }
59
60        let mob = ctx.mob();
61        let world = ctx.world();
62        let body_pos = mob.block_position();
63        let stroll_to = if world.is_village(body_pos) {
64            land_random_pos(mob, self.max_xz_dist, self.max_y_dist)
65        } else {
66            let section = SectionPos::from_block_pos(body_pos);
67            let toward = find_section_closest_to_village(ctx, section);
68            if toward == section {
69                land_random_pos(mob, self.max_xz_dist, self.max_y_dist)
70            } else {
71                default_random_pos_towards(
72                    mob,
73                    self.max_xz_dist,
74                    self.max_y_dist,
75                    section_bottom_center(toward),
76                    FRAC_PI_2,
77                )
78            }
79        };
80
81        brain.set_memory_or_erase(
82            memory_module_types::WALK_TARGET,
83            stroll_to.map(|pos| WalkTarget::of_position(pos, self.speed_modifier, 0)),
84        );
85        true
86    }
87
88    fn debug_name(&self) -> &'static str {
89        "VillageBoundRandomStroll"
90    }
91}

93/// Vanilla parity: `BehaviorUtils.findSectionClosestToVillage`, which walks the
94/// cube of sections around `center` and keeps the one nearest a village.
95fn find_section_closest_to_village(ctx: &BrainContext<'_>, center: SectionPos) -> SectionPos {
96    let world = ctx.world();
97    let mut best = center;
98    let mut best_distance = world.sections_to_village(center);
99    for x in -VILLAGE_SEARCH_SECTION_RADIUS..=VILLAGE_SEARCH_SECTION_RADIUS {
100        for y in -VILLAGE_SEARCH_SECTION_RADIUS..=VILLAGE_SEARCH_SECTION_RADIUS {
101            for z in -VILLAGE_SEARCH_SECTION_RADIUS..=VILLAGE_SEARCH_SECTION_RADIUS {
102                let candidate = SectionPos::new(center.x() + x, center.y() + y, center.z() + z);
103                let distance = world.sections_to_village(candidate);
104                if distance < best_distance {
105                    best_distance = distance;
106                    best = candidate;
107                }
108            }
109        }
110    }
111    best
112}

114/// Vanilla parity: `Vec3.atBottomCenterOf(sectionPos.center())`.
115fn section_bottom_center(section: SectionPos) -> DVec3 {
116    DVec3::new(
117        f64::from((section.x() << 4) + 8) + 0.5,
118        f64::from((section.y() << 4) + 8),
119        f64::from((section.z() << 4) + 8) + 0.5,
120    )
121}
