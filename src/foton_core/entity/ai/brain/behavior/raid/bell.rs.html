11/// The chance a villager standing at the bell leaves it alone this tick.
12///
13/// Vanilla parity: `RingBell.BELL_RING_CHANCE`, whose `nextFloat() <= 0.95F`
14/// means one tick in twenty actually rings.
15const BELL_RING_CHANCE: f32 = 0.95;
16
17/// How close a villager has to be to reach the rope.
18///
19/// Vanilla parity: `RingBell.RING_BELL_FROM_DISTANCE`.
20const RING_BELL_FROM_DISTANCE: f64 = 3.0;
21
22/// Sends a villager that heard a bell into hiding.
23///
24/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.ReactToBell`.
25///
26/// The `HEARD_BELL_TIME` this reads is written by [`BellBlockEntity`] on
27/// everything within thirty-two blocks of a rung bell. The raid check is what
28/// keeps the two halves of the village apart: a bell rung during a raid is the
29/// alarm the `PRE_RAID` and RAID packages are already answering, and this would
30/// only pull the villager out of them, so it stands down and lets
31/// [`SetRaidStatus`] keep the villager where it is.
32///
33/// [`BellBlockEntity`]: crate::block_entity::entities::BellBlockEntity
34/// [`SetRaidStatus`]: super::SetRaidStatus
35pub struct ReactToBell;
36
37impl Trigger for ReactToBell {
38    fn required_memories(&self) -> Vec<MemoryModuleId> {
39        vec![memory_module_types::HEARD_BELL_TIME.id()]
40    }
41
42    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
43        if !ctx
44            .brain()
45            .has_memory_value(memory_module_types::HEARD_BELL_TIME.id())
46        {
47            return false;
48        }
49        if ctx
50            .world()
51            .get_raid_at(ctx.mob().block_position())
52            .is_none()
53        {
54            ctx.brain().set_active_activity_if_possible(Activity::Hide);
55        }
56        true
57    }
58
59    fn debug_name(&self) -> &'static str {
60        "ReactToBell"
61    }
62}

64/// Rings the village bell a villager is standing at.
65///
66/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.RingBell`. It runs
67/// only in the `PRE_RAID` package, which is also what walks the villager to the
68/// meeting point in the first place -- so the raid alarm is a villager reaching
69/// the bell and pulling it, rather than anything the raid does itself.
70pub struct RingBell;
71
72impl Trigger for RingBell {
73    fn required_memories(&self) -> Vec<MemoryModuleId> {
74        vec![memory_module_types::MEETING_POINT.id()]
75    }
76
77    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
78        let Some(meeting_point) = ctx.brain().get_memory(memory_module_types::MEETING_POINT) else {
79            return false;
80        };
81        if rand::random::<f32>() <= BELL_RING_CHANCE {
82            return false;
83        }
84
85        // Vanilla returns `true` once the roll passes, whether or not the bell
86        // was in reach or still standing.
87        let mob = ctx.mob();
88        let pos = meeting_point.pos;
89        if !utils::block_closer_than(pos, mob.block_position(), RING_BELL_FROM_DISTANCE) {
90            return true;
91        }
92        let world = ctx.world();
93        if world.get_block_state(pos).get_block() == &vanilla_blocks::BELL {
94            BellBlock::attempt_to_ring(world, pos, None, Some(mob));
95        }
96        true
97    }
98
99    fn debug_name(&self) -> &'static str {
100        "RingBell"
101    }
102}
