3use foton_registry::{RegistryEntry as _, vanilla_poi_types};

6use crate::entity::ai::brain::behavior::{BrainContext, Trigger, utils};

10/// How long a villager keeps trying to reach a hiding place after the bell.
11///
12/// Vanilla parity: `SetHiddenState.HIDE_TIMEOUT`, measured from the
13/// `HEARD_BELL_TIME` rather than from when the hiding started -- so a villager
14/// that cannot get indoors gives up fifteen seconds after the alarm instead of
15/// standing in a doorway forever.
16const HIDE_TIMEOUT: i64 = 300;
17
18/// Ticks per second, for the `seconds` `SetHiddenState` is built with.
19const TICKS_PER_SECOND: i32 = 20;
20
21/// Vanilla parity: the `p -> p.is(PoiTypes.HOME)` both searches use.
22fn is_home_poi(poi_type_id: usize) -> bool {
23    poi_type_id == vanilla_poi_types::HOME.id()
24}

26/// Sends a villager to the nearest bed it can find, and remembers where.
27///
28/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.LocateHidingPlace`.
29///
30/// The search is three fallbacks deep: a bed already within reach, then any bed
31/// in the radius at random, then the villager's own home. Occupancy is ignored
32/// throughout -- a raid is no time to be fussy about whose bed it is -- and the
33/// four memories it erases are the ones that would otherwise keep the villager
34/// walking somewhere else: its path, what it was looking at, and whoever it was
35/// in the middle of trading with or courting.
36pub struct LocateHidingPlace {
37    radius: i32,
38    speed_modifier: f64,
39    close_enough_dist: i32,
40}
41
42impl LocateHidingPlace {
43    /// Vanilla parity: `LocateHidingPlace.create`.
44    #[must_use]
45    pub const fn new(radius: i32, speed_modifier: f64, close_enough_dist: i32) -> Self {
46        Self {
47            radius,
48            speed_modifier,
49            close_enough_dist,
50        }
51    }
52}
53
54impl Trigger for LocateHidingPlace {
55    fn required_memories(&self) -> Vec<MemoryModuleId> {
56        vec![
57            memory_module_types::WALK_TARGET.id(),
58            memory_module_types::HOME.id(),
59            memory_module_types::HIDING_PLACE.id(),
60            memory_module_types::PATH.id(),
61            memory_module_types::LOOK_TARGET.id(),
62            memory_module_types::BREED_TARGET.id(),
63            memory_module_types::INTERACTION_TARGET.id(),
64        ]
65    }
66
67    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
68        let brain = ctx.brain();
69        // Vanilla parity: the `i.absent(WALK_TARGET)` of the group -- a
70        // villager already walking somewhere is left to get there.
71        if brain.has_memory_value(memory_module_types::WALK_TARGET.id()) {
72            return false;
73        }
74
75        let mob = ctx.mob();
76        let body_block = mob.block_position();
77        let body_position = mob.position();
78        let close_enough = f64::from(self.close_enough_dist);
79
80        let found = {
81            let storage = ctx.world().poi_storage.lock();
82            storage
83                .find(
84                    &is_home_poi,
85                    &|_| true,
86                    body_block,
87                    self.close_enough_dist + 1,
88                    OccupationStatus::Any,
89                )
90                .filter(|&pos| utils::block_closer_to_center_than(pos, body_position, close_enough))
91                .or_else(|| {
92                    storage.get_random(
93                        &is_home_poi,
94                        &|_| true,
95                        OccupationStatus::Any,
96                        body_block,
97                        self.radius,
98                        &mut rand::rng(),
99                    )
100                })
101        };
102        // Vanilla parity: the last `.or(...)`, which falls back to the bed this
103        // villager already owns however far away it is.
104        let Some(pos) = found.or_else(|| {
105            brain
106                .get_memory(memory_module_types::HOME)
107                .map(|home| home.pos)
108        }) else {
109            return true;
110        };
111
112        brain.erase_memory(memory_module_types::PATH.id());
113        brain.erase_memory(memory_module_types::LOOK_TARGET.id());
114        brain.erase_memory(memory_module_types::BREED_TARGET.id());
115        brain.erase_memory(memory_module_types::INTERACTION_TARGET.id());
116        brain.set_memory(
117            memory_module_types::HIDING_PLACE,
118            GlobalPos::new(ctx.world().key.clone(), pos),
119        );
120        if !utils::block_closer_to_center_than(pos, body_position, close_enough) {
121            brain.set_memory(
122                memory_module_types::WALK_TARGET,
123                WalkTarget::of_block(pos, self.speed_modifier, self.close_enough_dist),
124            );
125        }
126        true
127    }
128
129    fn debug_name(&self) -> &'static str {
130        "LocateHidingPlace"
131    }
132}

134/// Counts a villager's time indoors and lets it out again when that is up.
135///
136/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.SetHiddenState`.
137///
138/// Two clocks end the hiding, and both have to be watched: the villager has
139/// spent its fifteen seconds actually at the hiding place, or five seconds have
140/// passed since the bell without it getting there. Whichever runs out first
141/// drops both memories and hands the villager back to its schedule -- which is
142/// the only way out of the HIDE activity, since that package has no
143/// `UpdateActivityFromSchedule` of its own.
144pub struct SetHiddenState {
145    stay_hidden_ticks: i32,
146    close_enough_dist: f64,
147    /// Vanilla parity: the `MutableInt ticksHidden` the builder closes over.
148    ticks_hidden: i32,
149}
150
151impl SetHiddenState {
152    /// Vanilla parity: `SetHiddenState.create(seconds, closeEnoughDist)`.
153    #[must_use]
154    pub const fn new(seconds: i32, close_enough_dist: i32) -> Self {
155        Self {
156            stay_hidden_ticks: seconds * TICKS_PER_SECOND,
157            close_enough_dist: close_enough_dist as f64,
158            ticks_hidden: 0,
159        }
160    }
161}
162
163impl Trigger for SetHiddenState {
164    fn required_memories(&self) -> Vec<MemoryModuleId> {
165        vec![
166            memory_module_types::HIDING_PLACE.id(),
167            memory_module_types::HEARD_BELL_TIME.id(),
168        ]
169    }
170
171    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
172        let brain = ctx.brain();
173        let Some(hiding_place) = brain.get_memory(memory_module_types::HIDING_PLACE) else {
174            return false;
175        };
176        let Some(heard_bell_time) = brain.get_memory(memory_module_types::HEARD_BELL_TIME) else {
177            return false;
178        };
179
180        let timed_out_trying_to_hide = heard_bell_time + HIDE_TIMEOUT <= ctx.game_time();
181        if self.ticks_hidden <= self.stay_hidden_ticks && !timed_out_trying_to_hide {
182            if utils::block_closer_than(
183                hiding_place.pos,
184                ctx.mob().block_position(),
185                self.close_enough_dist,
186            ) {
187                self.ticks_hidden += 1;
188            }
189            return true;
190        }
191
192        brain.erase_memory(memory_module_types::HEARD_BELL_TIME.id());
193        brain.erase_memory(memory_module_types::HIDING_PLACE.id());
194        brain.update_activity_from_schedule(ctx.world(), ctx.game_time());
195        self.ticks_hidden = 0;
196        true
197    }
198
199    fn debug_name(&self) -> &'static str {
200        "SetHiddenState"
201    }
202}
