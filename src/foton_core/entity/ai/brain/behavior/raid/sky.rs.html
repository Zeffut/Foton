12/// How many positions are tried before the villager gives up on the sky.
13///
14/// Vanilla parity: the `for (int i = 0; i < 10; i++)` of `getOutdoorPosition`.
15const OUTDOOR_ATTEMPTS: i32 = 10;
16
17/// The horizontal spread of one attempt.
18///
19/// Vanilla parity: the `random.nextInt(20) - 10` of `getOutdoorPosition`.
20const OUTDOOR_HORIZONTAL_SPREAD: i32 = 20;
21
22/// The vertical spread of one attempt.
23///
24/// Vanilla parity: the `random.nextInt(6) - 3` of `getOutdoorPosition`.
25const OUTDOOR_VERTICAL_SPREAD: i32 = 6;
26
27/// Whether `target` is open to the sky and not above the body.
28///
29/// Vanilla parity: `MoveToSkySeeingSpot.hasNoBlocksAbove`. The height
30/// comparison is what stops a villager celebrating on a roof it cannot reach:
31/// a column is only a candidate if its motion-blocking top is at or below where
32/// the villager already stands.
33#[must_use]
34pub fn has_no_blocks_above(world: &World, body: &dyn PathfinderMob, target: BlockPos) -> bool {
35    world.can_see_sky(target)
36        && f64::from(
37            world
38                .heightmap_pos(HeightmapType::MotionBlocking, target)
39                .y(),
40        ) <= body.position().y
41}
42
43/// Vanilla parity: the private `MoveToSkySeeingSpot.getOutdoorPosition`.
44fn outdoor_position(world: &World, body: &dyn PathfinderMob) -> Option<DVec3> {
45    let pos = body.block_position();
46    for _ in 0..OUTDOOR_ATTEMPTS {
47        let candidate = pos.offset(
48            rand::random_range(0..OUTDOOR_HORIZONTAL_SPREAD) - OUTDOOR_HORIZONTAL_SPREAD / 2,
49            rand::random_range(0..OUTDOOR_VERTICAL_SPREAD) - OUTDOOR_VERTICAL_SPREAD / 2,
50            rand::random_range(0..OUTDOOR_HORIZONTAL_SPREAD) - OUTDOOR_HORIZONTAL_SPREAD / 2,
51        );
52        if has_no_blocks_above(world, body, candidate) {
53            // Vanilla parity: `Vec3.atBottomCenterOf`, which centers the column
54            // but keeps the block's own floor.
55            let (x, _, z) = candidate.get_center();
56            return Some(DVec3::new(x, f64::from(candidate.y()), z));
57        }
58    }
59    None
60}

62/// Walks a villager out from under a roof.
63///
64/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.MoveToSkySeeingSpot`.
65/// It is the first thing the RAID package offers once the village has won,
66/// because the fireworks
67/// [`CelebrateVillagersSurvivedRaid`](crate::entity::entities::mobs::npc::villager_ai::CelebrateVillagersSurvivedRaid)
68/// throws are only thrown from a block that can see the sky.
69pub struct MoveToSkySeeingSpot {
70    speed_modifier: f64,
71}
72
73impl MoveToSkySeeingSpot {
74    /// Vanilla parity: `MoveToSkySeeingSpot.create`.
75    #[must_use]
76    pub const fn new(speed_modifier: f64) -> Self {
77        Self { speed_modifier }
78    }
79}
80
81impl Trigger for MoveToSkySeeingSpot {
82    fn required_memories(&self) -> Vec<MemoryModuleId> {
83        vec![memory_module_types::WALK_TARGET.id()]
84    }
85
86    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
87        let brain = ctx.brain();
88        if brain.has_memory_value(memory_module_types::WALK_TARGET.id()) {
89            return false;
90        }
91
92        let world = ctx.world();
93        let mob = ctx.mob();
94        // Vanilla returns `false` from under open sky, so a villager already
95        // outdoors leaves the roll to whatever else the gate is offering.
96        if world.can_see_sky(mob.block_position()) {
97            return false;
98        }
99
100        if let Some(target) = outdoor_position(world, mob) {
101            brain.set_memory(
102                memory_module_types::WALK_TARGET,
103                WalkTarget::of_position(target, self.speed_modifier, 0),
104            );
105        }
106        true
107    }
108
109    fn debug_name(&self) -> &'static str {
110        "MoveToSkySeeingSpot"
111    }
112}
