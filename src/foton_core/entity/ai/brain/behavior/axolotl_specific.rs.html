1//! The two behaviors that make an axolotl play dead.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.axolotl.PlayDead` and
4//! `ValidatePlayDead`. They are a pair: the first one holds the axolotl still
5//! and heals it, and the second one, running in the core activity, counts the
6//! clock down and hands control back when it runs out.
7
8use foton_registry::vanilla_mob_effects;
9
10use super::{BrainContext, MemoryModuleId, MemoryStatus, TimedBehavior, Trigger};
11use crate::entity::MobEffectInstance;
12use crate::entity::ai::brain::memory::memory_module_types;
13
14/// Vanilla parity: `Axolotl.TOTAL_PLAYDEAD_TIME`, which is both the memory's
15/// starting value and the behavior's duration.
16pub const TOTAL_PLAYDEAD_TIME: i32 = 200;
17
18/// Holds an axolotl still and heals it while it plays dead.
19///
20/// Vanilla parity: `net.minecraft.world.entity.animal.axolotl.PlayDead`.
21pub struct PlayDead;
22
23impl TimedBehavior for PlayDead {
24    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
25        const CONDITIONS: &[(MemoryModuleId, MemoryStatus)] = &[
26            (
27                memory_module_types::PLAY_DEAD_TICKS.id(),
28                MemoryStatus::ValuePresent,
29            ),
30            (
31                memory_module_types::HURT_BY_ENTITY.id(),
32                MemoryStatus::ValuePresent,
33            ),
34        ];
35        CONDITIONS
36    }
37
38    fn duration(&self) -> (i32, i32) {
39        (TOTAL_PLAYDEAD_TIME, TOTAL_PLAYDEAD_TIME)
40    }
41
42    fn check_extra_start_conditions(&mut self, ctx: &BrainContext<'_>) -> bool {
43        ctx.mob().is_in_water()
44    }
45
46    fn can_still_use(&mut self, ctx: &BrainContext<'_>) -> bool {
47        ctx.mob().is_in_water()
48            && ctx
49                .brain()
50                .has_memory_value(memory_module_types::PLAY_DEAD_TICKS.id())
51    }
52
53    fn start(&mut self, ctx: &BrainContext<'_>) {
54        let brain = ctx.brain();
55        brain.erase_memory(memory_module_types::WALK_TARGET.id());
56        brain.erase_memory(memory_module_types::LOOK_TARGET.id());
57        ctx.mob().add_mob_effect(MobEffectInstance::with_duration(
58            vanilla_mob_effects::REGENERATION,
59            TOTAL_PLAYDEAD_TIME,
60            0,
61        ));
62    }
63
64    fn debug_name(&self) -> &'static str {
65        "PlayDead"
66    }
67}
68
69/// Counts the play-dead clock down and ends the act when it hits zero.
70///
71/// Vanilla parity: `net.minecraft.world.entity.animal.axolotl.ValidatePlayDead`.
72/// It lives in the core activity, so it keeps running while `PLAY_DEAD` holds
73/// every other activity out.
74pub struct ValidatePlayDead;
75
76impl Trigger for ValidatePlayDead {
77    fn required_memories(&self) -> Vec<MemoryModuleId> {
78        vec![
79            memory_module_types::PLAY_DEAD_TICKS.id(),
80            memory_module_types::HURT_BY_ENTITY.id(),
81        ]
82    }
83
84    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
85        let brain = ctx.brain();
86        let Some(ticks) = brain.get_memory(memory_module_types::PLAY_DEAD_TICKS) else {
87            return false;
88        };
89
90        if ticks <= 0 {
91            brain.erase_memory(memory_module_types::PLAY_DEAD_TICKS.id());
92            brain.erase_memory(memory_module_types::HURT_BY_ENTITY.id());
93            brain.use_default_activity();
94        } else {
95            brain.set_memory(memory_module_types::PLAY_DEAD_TICKS, ticks - 1);
96        }
97
98        true
99    }
100
101    fn debug_name(&self) -> &'static str {
102        "ValidatePlayDead"
103    }
104}
