3/// One named slice of a brain's behavior schedule.
4///
5/// Vanilla parity: `net.minecraft.world.entity.schedule.Activity`.
6///
7/// Vanilla registers these into `BuiltInRegistries.ACTIVITY`, but the registry
8/// is a hardcoded Java list -- no datapack can add to it, no packet carries an
9/// activity id and nothing writes one to disk. `FotonExtractor` emits no
10/// `activity` asset, so mirroring the Java constants as an enum keeps the data
11/// coming from the vanilla source without inventing a registry that would only
12/// ever hold these twenty-six entries.
13#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
14pub enum Activity {
15    /// Behaviors that run whatever else the mob is doing.
16    Core,
17    /// The fallback activity every brain starts in.
18    Idle,
19    Work,
20    Play,
21    Rest,
22    Meet,
23    Panic,
24    Raid,
25    PreRaid,
26    Hide,
27    Fight,
28    Celebrate,
29    AdmireItem,
30    Avoid,
31    Ride,
32    PlayDead,
33    LongJump,
34    Ram,
35    Tongue,
36    Swim,
37    LaySpawn,
38    Sniff,
39    Investigate,
40    Roar,
41    Emerge,
42    Dig,
43}

114    /// Looks an activity up by the key a timeline keyframe stores.
115    ///
116    /// Vanilla parity: `BuiltInRegistries.ACTIVITY.byNameCodec()`, which the
117    /// `AttributeTypes.ACTIVITY` value codec is. A bare path resolves in the
118    /// `minecraft` namespace, as every `ResourceLocation` does.
119    #[must_use]
120    pub fn by_key(key: &str) -> Option<Self> {
121        let path = key.strip_prefix("minecraft:").unwrap_or(key);
122        if path.contains(':') {
123            return None;
124        }
125        Self::ALL.iter().copied().find(|entry| entry.name() == path)
126    }
127}

129/// The environment attribute a brain reads its scheduled activity from.
130///
131/// Vanilla parity: the `EnvironmentAttribute<Activity>` held in `Brain.schedule`
132/// and passed to `Brain.setSchedule`. 26.2 replaced the old `Schedule` /
133/// `ScheduleBuilder` data type with a timeline track, so a "schedule" is now
134/// the name of an activity-valued environment attribute; vanilla declares
135/// exactly the two below in `EnvironmentAttributes` and nothing can add a
136/// third, so this is a closed set rather than a registry.
137#[derive(Debug, Clone, Copy, PartialEq, Eq)]
138pub enum ScheduleAttribute {
139    /// Vanilla `EnvironmentAttributes.VILLAGER_ACTIVITY`.
140    VillagerActivity,
141    /// Vanilla `EnvironmentAttributes.BABY_VILLAGER_ACTIVITY`.
142    BabyVillagerActivity,
143}
144
145impl ScheduleAttribute {
146    /// Returns the name the `villager_schedule` timeline tracks this under.
147    #[must_use]
148    pub const fn attribute_name(self) -> &'static str {
149        match self {
150            Self::VillagerActivity => "minecraft:gameplay/villager_activity",
151            Self::BabyVillagerActivity => "minecraft:gameplay/baby_villager_activity",
152        }
153    }
154
155    /// The value the attribute falls back to where no timeline covers it.
156    ///
157    /// Vanilla parity: the `defaultValue(Activity.IDLE)` both attributes are
158    /// registered with.
159    pub const DEFAULT_ACTIVITY: Activity = Activity::Idle;
