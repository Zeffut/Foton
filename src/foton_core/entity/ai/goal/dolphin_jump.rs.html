11/// How far ahead the dolphin checks before it breaches.
12///
13/// Vanilla parity: `DolphinJumpGoal.STEPS_TO_CHECK`, which skips two and three
14/// so the arc is only sampled where it matters.
15const STEPS_TO_CHECK: [i32; 6] = [0, 1, 4, 5, 6, 7];
16
17/// Forward push a breach starts with.
18const JUMP_FORWARD: f64 = 0.6;
19/// Upward push a breach starts with.
20const JUMP_UP: f64 = 0.7;
21/// Below this squared vertical speed the arc is treated as flat.
22const FLAT_ARC_SPEED_SQR: f64 = 0.03;
23/// Degrees of pitch below which a flat arc counts as level.
24const LEVEL_PITCH: f32 = 10.0;
25/// How fast the pitch falls back to level at the top of the arc.
26const PITCH_LEVEL_LERP: f32 = 0.2;
27
28/// Leaps clear of the water and arcs back in.
29///
30/// Vanilla parity: `DolphinJumpGoal`. The dolphin only starts if the water and
31/// the air ahead of it are both clear the whole way, which is why one never
32/// jumps into a cliff.
33pub(crate) struct DolphinJumpGoal {
34    interval: i32,
35    breached: bool,
36}
37
38impl DolphinJumpGoal {
39    #[must_use]
40    pub(crate) const fn new(interval: i32) -> Self {
41        Self {
42            interval: reduced_tick_delay(interval),
43            breached: false,
44        }
45    }
46
47    fn water_is_clear(mob: &dyn PathfinderMob, pos: BlockPos) -> bool {
48        let Some(world) = mob.level() else {
49            return false;
50        };
51        let state = world.get_block_state(pos);
52        state.get_fluid_state().is_water() && !state.blocks_motion()
53    }
54
55    fn surface_is_clear(mob: &dyn PathfinderMob, pos: BlockPos) -> bool {
56        let Some(world) = mob.level() else {
57            return false;
58        };
59        world.get_block_state(pos.above()).is_air()
60            && world.get_block_state(pos.above().above()).is_air()
61    }
62}
63
64impl Goal for DolphinJumpGoal {
65    fn controls(&self) -> GoalControls {
66        // Vanilla parity: `JumpGoal` sets MOVE and JUMP.
67        GoalControls::MOVE | GoalControls::JUMP
68    }
69
70    fn is_interruptable(&self) -> bool {
71        false
72    }
73
74    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
75        if rand::random_range(0..self.interval.max(1)) != 0 {
76            return false;
77        }
78
79        let motion = mob.direction_yaw();
80        let (step_x, step_z) = motion.offset_xz();
81        let mob_pos = mob.block_position();
82
83        STEPS_TO_CHECK.into_iter().all(|step| {
84            let pos = mob_pos.offset(step_x * step, 0, step_z * step);
85            Self::water_is_clear(mob, pos) && Self::surface_is_clear(mob, pos)
86        })
87    }
88
89    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
90        let vertical_speed = mob.velocity().y;
91        let pitch = mob.rotation().1;
92        let arc_is_over = vertical_speed * vertical_speed < FLAT_ARC_SPEED_SQR
93            && pitch != 0.0
94            && pitch.abs() < LEVEL_PITCH
95            && mob.is_in_water();
96
97        !arc_is_over && !mob.on_ground()
98    }
99
100    fn start(&mut self, mob: &dyn PathfinderMob) {
101        let direction = mob.direction_yaw();
102        let (step_x, step_z) = direction.offset_xz();
103        mob.set_velocity(
104            mob.velocity()
105                + DVec3::new(
106                    f64::from(step_x) * JUMP_FORWARD,
107                    JUMP_UP,
108                    f64::from(step_z) * JUMP_FORWARD,
109                ),
110        );
111        mob.mob_base().navigation().lock().stop();
112    }
113
114    fn stop(&mut self, mob: &dyn PathfinderMob) {
115        let (yaw, _) = mob.rotation();
116        mob.set_rotation((yaw, 0.0));
117    }
118
119    fn tick(&mut self, mob: &dyn PathfinderMob) {
120        // Vanilla never clears `breached`, so the jump sound plays once for the
121        // life of the goal rather than once per leap.
122        let already_breached = self.breached;
123        if !already_breached && let Some(world) = mob.level() {
124            self.breached = world
125                .get_block_state(mob.block_position())
126                .get_fluid_state()
127                .is_water();
128        }
129
130        if self.breached && !already_breached {
131            mob.play_sound(&sound_events::ENTITY_DOLPHIN_JUMP, 1.0, 1.0);
132        }
133
134        let movement = mob.velocity();
135        let (yaw, pitch) = mob.rotation();
136        if movement.y * movement.y < FLAT_ARC_SPEED_SQR && pitch != 0.0 {
137            mob.set_rotation((yaw, rot_lerp(PITCH_LEVEL_LERP, pitch, 0.0)));
138        } else if movement.length() > 1.0e-5 {
139            let horizontal = movement.x.hypot(movement.z);
140            let rotation = (-movement.y).atan2(horizontal).to_degrees();
141            mob.set_rotation((yaw, rotation as f32));
142        }
143    }
144}
145
146/// Vanilla parity: `Mth.rotLerp`.
147fn rot_lerp(delta: f32, from: f32, to: f32) -> f32 {
148    delta.mul_add(wrap_degrees(to - from), from)
149}
150
151fn wrap_degrees(mut degrees: f32) -> f32 {
152    degrees %= 360.0;
153    if degrees >= 180.0 {
154        degrees -= 360.0;
155    }
156    if degrees < -180.0 {
157        degrees += 360.0;
158    }
159    degrees
160}
