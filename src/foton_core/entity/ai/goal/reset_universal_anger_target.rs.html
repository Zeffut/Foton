1//! Letting go of a grudge when the universal-anger rule is on.
2
3use foton_registry::vanilla_game_rules::UNIVERSAL_ANGER;
4use foton_utils::WorldAabb;
5
6use super::selector::{Goal, GoalControls};
7use super::target_goal::follow_distance;
8use crate::entity::PathfinderMob;
9
10/// How far up and down the alert reaches.
11///
12/// Vanilla parity: `ResetUniversalAngerTargetGoal.ALERT_RANGE_Y`.
13const ALERT_RANGE_Y: f64 = 10.0;
14
15/// Turns a mob's grudge against one player into a grudge against all of them.
16///
17/// Vanilla parity: `ResetUniversalAngerTargetGoal`. It only ever runs under the
18/// `universalAnger` game rule, which is what makes a whole pack round on the
19/// next player it sees rather than on the one who threw the punch.
20pub(crate) struct ResetUniversalAngerTargetGoal {
21    alert_others_of_same_type: bool,
22    last_hurt_by_player_timestamp: i32,
23}
24
25impl ResetUniversalAngerTargetGoal {
26    #[must_use]
27    pub(crate) const fn new(alert_others_of_same_type: bool) -> Self {
28        Self {
29            alert_others_of_same_type,
30            last_hurt_by_player_timestamp: 0,
31        }
32    }
33
34    /// Vanilla parity: `ResetUniversalAngerTargetGoal.wasHurtByPlayer`.
35    fn was_hurt_by_player(&self, mob: &dyn PathfinderMob) -> bool {
36        mob.last_hurt_by_mob()
37            .is_some_and(|attacker| attacker.as_player().is_some())
38            && mob.last_hurt_by_mob_timestamp() > self.last_hurt_by_player_timestamp
39    }
40
41    /// Vanilla parity: `ResetUniversalAngerTargetGoal.getNearbyMobsOfSameType`.
42    fn refresh_others_of_same_type(mob: &dyn PathfinderMob) {
43        let Some(world) = mob.level() else {
44            return;
45        };
46
47        let within = follow_distance(mob);
48        let position = mob.position();
49        let search_box = WorldAabb::new(
50            position.x,
51            position.y,
52            position.z,
53            position.x + 1.0,
54            position.y + 1.0,
55            position.z + 1.0,
56        )
57        .inflate_xyz(within, ALERT_RANGE_Y, within);
58        let mob_type_key = mob.downcast_type_key();
59        let mob_uuid = mob.uuid();
60
61        for entity in world.get_entities_in_aabb_matching(&search_box, |entity| {
62            entity.uuid() != mob_uuid
63                && entity.downcast_type_key() == mob_type_key
64                && !entity.is_spectator()
65        }) {
66            if let Some(other) = entity.as_neutral_mob() {
67                other.forget_current_target_and_refresh_universal_anger();
68            }
69        }
70    }
71}
72
73impl Goal for ResetUniversalAngerTargetGoal {
74    /// Vanilla parity: `ResetUniversalAngerTargetGoal` never calls `setFlags`,
75    /// so it holds no control at all. Claiming `TARGET` would let the
76    /// `HurtByTargetGoal` that provoked the mob block the reset outright.
77    fn controls(&self) -> GoalControls {
78        GoalControls::EMPTY
79    }
80
81    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
82        mob.level()
83            .is_some_and(|world| world.get_game_rule(&UNIVERSAL_ANGER))
84            && self.was_hurt_by_player(mob)
85    }
86
87    fn start(&mut self, mob: &dyn PathfinderMob) {
88        self.last_hurt_by_player_timestamp = mob.last_hurt_by_mob_timestamp();
89        let Some(neutral) = mob.as_neutral_mob() else {
90            return;
91        };
92        neutral.forget_current_target_and_refresh_universal_anger();
93
94        if self.alert_others_of_same_type {
95            Self::refresh_others_of_same_type(mob);
96        }
97    }
98}
