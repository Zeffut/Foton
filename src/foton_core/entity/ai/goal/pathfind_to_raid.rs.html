1//! Walk to the raid, sweeping up any raider passed on the way.
2//!
3//! Vanilla parity: `PathfindToRaidGoal`. This is what stops a wave that spawned
4//! ninety blocks out from standing in a field: it aims the mob at the village
5//! center in short hops, re-aiming each time the path runs out. The second half
6//! is the recruitment sweep -- once a second, every unattached raider within
7//! sixteen blocks is pulled into the raid, which is how a passing patrol ends
8//! up fighting somebody else's fight.
9
10use core::f64::consts::FRAC_PI_2;
11
12use glam::DVec3;
13
14use super::default_random_pos_towards;
15use super::selector::{Goal, GoalControls};
16use crate::entity::PathfinderMob;
17
18/// Ticks between two recruitment sweeps.
19///
20/// Vanilla parity: `PathfindToRaidGoal.RECRUITMENT_SEARCH_TICK_DELAY`.
21const RECRUITMENT_SEARCH_TICK_DELAY: i32 = 20;
22
23/// Speed a raider walks towards its raid at.
24///
25/// Vanilla parity: `PathfindToRaidGoal.SPEED_MODIFIER`.
26const SPEED_MODIFIER: f64 = 1.0;
27
28/// How far each hop towards the village center reaches.
29///
30/// Vanilla parity: the `getPosTowards(mob, 15, 4, ..)` of `tick`.
31const HOP_HORIZONTAL_DISTANCE: i32 = 15;
32const HOP_VERTICAL_DISTANCE: i32 = 4;
33
34/// How far a raider will look for others to recruit.
35///
36/// Vanilla parity: the `inflate(16.0)` of `recruitNearby`.
37const RECRUITMENT_RANGE: f64 = 16.0;
38
39/// Walks a raider to its raid.
40///
41/// Vanilla parity: `PathfindToRaidGoal`.
42pub(crate) struct PathfindToRaidGoal {
43    /// Tick count past which the next recruitment sweep runs.
44    recruitment_tick: i32,
45}
46
47impl PathfindToRaidGoal {
48    /// Creates the goal.
49    #[must_use]
50    pub(crate) const fn new() -> Self {
51        Self {
52            recruitment_tick: 0,
53        }
54    }
55
56    /// Returns whether the mob is in a raid it still has to reach.
57    ///
58    /// Vanilla parity: the shared body of `canUse` and `canContinueToUse`.
59    fn is_heading_to_raid(mob: &dyn PathfinderMob) -> bool {
60        let Some(raider) = mob.as_raider() else {
61            return false;
62        };
63        let Some(status) = raider.current_raid_status() else {
64            return false;
65        };
66        if !status.active || status.over {
67            return false;
68        }
69        mob.level()
70            .is_some_and(|world| !world.is_village(mob.block_position()))
71    }
72}
73
74impl Default for PathfindToRaidGoal {
75    fn default() -> Self {
76        Self::new()
77    }
78}
79
80impl Goal for PathfindToRaidGoal {
81    fn controls(&self) -> GoalControls {
82        GoalControls::MOVE
83    }
84
85    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
86        mob.target().is_none()
87            && mob.controlling_passenger_mob().is_none()
88            && Self::is_heading_to_raid(mob)
89    }
90
91    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
92        Self::is_heading_to_raid(mob)
93    }
94
95    fn tick(&mut self, mob: &dyn PathfinderMob) {
96        let Some(raider) = mob.as_raider() else {
97            return;
98        };
99        let Some(raid) = raider.current_raid() else {
100            return;
101        };
102        if !raid.is_active() {
103            return;
104        }
105        let Some(world) = mob.level() else {
106            return;
107        };
108
109        if mob.tick_count() > self.recruitment_tick {
110            self.recruitment_tick = mob.tick_count() + RECRUITMENT_SEARCH_TICK_DELAY;
111            let search = mob.bounding_box().inflate(RECRUITMENT_RANGE);
112            let recruits = world.get_entities_in_aabb_matching(&search, |entity| {
113                entity
114                    .as_raider()
115                    .is_some_and(|other| !other.has_active_raid() && other.is_recruitable())
116            });
117            for recruit in recruits {
118                let Some(recruit_raider) = recruit.as_raider() else {
119                    continue;
120                };
121                raid.join_raid(&world, raid.groups_spawned(), recruit_raider, None, true);
122            }
123        }
124
125        if mob.is_path_finding() {
126            return;
127        }
128        let (center_x, center_y, center_z) = raid.center().get_bottom_center();
129        let Some(next) = default_random_pos_towards(
130            mob,
131            HOP_HORIZONTAL_DISTANCE,
132            HOP_VERTICAL_DISTANCE,
133            DVec3::new(center_x, center_y, center_z),
134            FRAC_PI_2,
135        ) else {
136            return;
137        };
138        mob.move_to_pos(next, SPEED_MODIFIER);
139    }
140}
