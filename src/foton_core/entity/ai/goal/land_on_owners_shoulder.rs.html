1use foton_utils::Downcast as _;
2
3use super::selector::{Goal, GoalControls};
4use crate::entity::entities::ParrotEntity;
5use crate::entity::{Entity, Mob, PathfinderMob, TamableAnimal};

7/// Puts a tamed parrot on its owner's shoulder.
8///
9/// Vanilla parity: `LandOnOwnersShoulderGoal`. The goal is uninterruptible only
10/// once the bird has actually landed, which is what stops another goal pulling
11/// it off mid-hop.
12pub struct LandOnOwnersShoulderGoal {
13    is_sitting_on_shoulder: bool,
14}
15
16impl LandOnOwnersShoulderGoal {
17    #[must_use]
18    pub(crate) const fn new() -> Self {
19        Self {
20            is_sitting_on_shoulder: false,
21        }
22    }
23}
24
25impl Goal for LandOnOwnersShoulderGoal {
26    fn controls(&self) -> GoalControls {
27        // Vanilla parity: `LandOnOwnersShoulderGoal` never calls `setFlags`.
28        GoalControls::EMPTY
29    }
30
31    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
32        let Some(parrot) = mob.downcast_ref::<ParrotEntity>() else {
33            return false;
34        };
35        let Some(owner) = parrot.owner() else {
36            return false;
37        };
38        let Some(player) = owner.as_player() else {
39            return false;
40        };
41
42        let owner_can_be_sat_on = !player.is_spectator()
43            && !player.abilities.lock().flying
44            && !player.is_in_water()
45            && !player.is_in_powder_snow();
46
47        !parrot.is_ordered_to_sit() && owner_can_be_sat_on && parrot.can_sit_on_shoulder()
48    }
49
50    fn is_interruptable(&self) -> bool {
51        !self.is_sitting_on_shoulder
52    }
53
54    fn start(&mut self, _mob: &dyn PathfinderMob) {
55        self.is_sitting_on_shoulder = false;
56    }
57
58    fn requires_update_every_tick(&self) -> bool {
59        true
60    }
61
62    fn tick(&mut self, mob: &dyn PathfinderMob) {
63        if self.is_sitting_on_shoulder {
64            return;
65        }
66        let Some(parrot) = mob.downcast_ref::<ParrotEntity>() else {
67            return;
68        };
69        if parrot.is_in_sitting_pose() || parrot.is_leashed() {
70            return;
71        }
72        let Some(owner) = parrot.owner() else {
73            return;
74        };
75        let Some(player) = owner.as_player() else {
76            return;
77        };
78        if !parrot.bounding_box().intersects(player.bounding_box()) {
79            return;
80        }
81
82        self.is_sitting_on_shoulder = parrot.set_entity_on_shoulder(player);
83    }
84}
