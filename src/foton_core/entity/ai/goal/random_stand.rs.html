1//! A horse rearing up for no reason at all.
2//!
3//! Vanilla parity: `RandomStandGoal`. The counter runs on every tick and only
4//! clears once it beats a thousand-sided roll, which is why a paddock of horses
5//! rears every so often rather than on a fixed beat.
6
7use super::selector::{Goal, GoalControls};
8use crate::entity::{AbstractHorse, LivingEntity, PathfinderMob};
9
10/// The die the stand counter is rolled against.
11///
12/// Vanilla parity: the `random.nextInt(1000)` of `RandomStandGoal.canUse`.
13const STAND_ROLL_SIDES: i32 = 1000;
14
15/// The second roll a horse still has to win once its counter is up.
16///
17/// Vanilla parity: the `random.nextInt(10) == 0` of the same method.
18const STAND_CONFIRM_SIDES: i32 = 10;
19
20/// Rears up at random.
21///
22/// Vanilla parity: `RandomStandGoal`.
23pub struct RandomStandGoal {
24    next_stand: i32,
25    /// Vanilla reads `getAmbientStandInterval()` off the horse in the goal's
26    /// constructor; Foton builds goals before the mob exists, so the interval is
27    /// read from the mob on the first reset instead.
28    initialized: bool,
29}
30
31impl RandomStandGoal {
32    #[must_use]
33    pub(crate) const fn new() -> Self {
34        Self {
35            next_stand: 0,
36            initialized: false,
37        }
38    }
39
40    fn reset_stand_interval(&mut self, horse: &dyn AbstractHorse) {
41        self.next_stand = -horse.ambient_stand_interval();
42        self.initialized = true;
43    }
44}
45
46impl Goal for RandomStandGoal {
47    fn controls(&self) -> GoalControls {
48        GoalControls::EMPTY
49    }
50
51    fn requires_update_every_tick(&self) -> bool {
52        true
53    }
54
55    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
56        let Some(horse) = mob.as_abstract_horse() else {
57            return false;
58        };
59        if !self.initialized {
60            self.reset_stand_interval(horse);
61        }
62
63        self.next_stand += 1;
64        if self.next_stand <= 0 || rand::random_range(0..STAND_ROLL_SIDES) >= self.next_stand {
65            return false;
66        }
67
68        self.reset_stand_interval(horse);
69        !LivingEntity::is_immobile(horse) && rand::random_range(0..STAND_CONFIRM_SIDES) == 0
70    }
71
72    fn can_continue_to_use(&mut self, _mob: &dyn PathfinderMob) -> bool {
73        false
74    }
75
76    fn start(&mut self, mob: &dyn PathfinderMob) {
77        let Some(horse) = mob.as_abstract_horse() else {
78            return;
79        };
80        horse.stand_if_possible();
81        if let Some(sound) = horse.ambient_stand_sound() {
82            // Vanilla calls `Entity.playSound(SoundEvent)`, which is a flat 1/1.
83            horse.play_sound(sound, 1.0, 1.0);
84        }
85    }
86}
