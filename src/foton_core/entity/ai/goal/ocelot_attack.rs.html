1use glam::DVec3;
2
3use super::selector::{Goal, GoalControls};
4use crate::entity::{PathfinderMob, SharedEntity};
5
6/// Ticks between two bites.
7///
8/// Vanilla parity: the `this.attackTime = 20` of `OcelotAttackGoal.tick`.
9const ATTACK_INTERVAL_TICKS: i32 = 20;
10
11/// Distance past which the cat gives up the chase.
12///
13/// Vanilla parity: the `225.0` of `OcelotAttackGoal`, fifteen blocks.
14const GIVE_UP_DISTANCE_SQR: f64 = 225.0;
15
16/// Distance inside which the cat sprints rather than stalks.
17///
18/// Vanilla parity: the `16.0` of `OcelotAttackGoal.tick`.
19const SPRINT_DISTANCE_SQR: f64 = 16.0;
20
21/// Speed while walking toward prey that is still far off.
22const WALK_SPEED_MODIFIER: f64 = 0.8;
23
24/// Speed while closing the last few blocks.
25const SPRINT_SPEED_MODIFIER: f64 = 1.33;
26
27/// Speed while creeping up on prey.
28const CROUCH_SPEED_MODIFIER: f64 = 0.6;
29
30/// The stalk-and-pounce attack a cat or an ocelot uses on chickens.
31///
32/// Vanilla parity: `OcelotAttackGoal`. The three speeds are what the cat's
33/// `customServerAiStep` reads back to decide between crouching, walking and
34/// sprinting, so they are the goal's whole visible output.
35pub struct OcelotAttackGoal {
36    target: Option<SharedEntity>,
37    attack_time: i32,
38}
39
40impl OcelotAttackGoal {
41    #[must_use]
42    pub(crate) const fn new() -> Self {
43        Self {
44            target: None,
45            attack_time: 0,
46        }
47    }
48}
49
50impl Goal for OcelotAttackGoal {
51    fn controls(&self) -> GoalControls {
52        GoalControls::MOVE | GoalControls::LOOK
53    }
54
55    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
56        self.target = mob.target();
57        self.target.is_some()
58    }
59
60    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
61        let Some(target) = self.target.clone() else {
62            return false;
63        };
64        if !target.is_alive() {
65            return false;
66        }
67        if mob.position().distance_squared(target.position()) > GIVE_UP_DISTANCE_SQR {
68            return false;
69        }
70
71        !mob.mob_base().navigation().lock().is_done() || self.can_use(mob)
72    }
73
74    fn stop(&mut self, mob: &dyn PathfinderMob) {
75        self.target = None;
76        mob.mob_base().navigation().lock().stop();
77    }
78
79    fn requires_update_every_tick(&self) -> bool {
80        true
81    }
82
83    fn tick(&mut self, mob: &dyn PathfinderMob) {
84        let Some(target) = self.target.clone() else {
85            return;
86        };
87
88        let target_position = target.position();
89        mob.mob_base().controls().lock().look_control.set_look_at(
90            DVec3::new(target_position.x, target.get_eye_y(), target_position.z),
91            30.0,
92            30.0,
93        );
94
95        let melee_radius = mob.bounding_box().width() * 2.0;
96        let melee_radius_sqr = melee_radius * melee_radius;
97        let distance_sqr = mob.position().distance_squared(target_position);
98        let speed_modifier =
99            if distance_sqr > melee_radius_sqr && distance_sqr < SPRINT_DISTANCE_SQR {
100                SPRINT_SPEED_MODIFIER
101            } else if distance_sqr < GIVE_UP_DISTANCE_SQR {
102                CROUCH_SPEED_MODIFIER
103            } else {
104                WALK_SPEED_MODIFIER
105            };
106
107        mob.move_to_pos(target_position, speed_modifier);
108        self.attack_time = (self.attack_time - 1).max(0);
109        if distance_sqr > melee_radius_sqr || self.attack_time > 0 {
110            return;
111        }
112
113        self.attack_time = ATTACK_INTERVAL_TICKS;
114        if let Some(world) = mob.level() {
115            let _ = mob.do_hurt_target(&world, &target);
116        }
117    }
118}
