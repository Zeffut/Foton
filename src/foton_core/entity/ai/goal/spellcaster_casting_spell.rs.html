1//! Spellcaster casting-spell goal.
2//!
3//! Vanilla parity: `SpellcasterIllager.SpellcasterCastingSpellGoal`. This is
4//! the goal that holds an evoker still with its arms up: while a spell is
5//! running it takes the movement and look controls away from everything else,
6//! so the caster cannot walk out of its own animation. Clearing the spell on
7//! stop is what puts the arms down.
8
9use super::selector::{Goal, GoalControls};
10use crate::entity::{IllagerSpell, Mob, PathfinderMob, SharedEntity};
11
12/// A subclass override of where the caster looks while casting.
13///
14/// Vanilla parity: overriding `SpellcasterCastingSpellGoal.tick`, which the
15/// evoker does so it keeps looking at the sheep it is recoloring.
16type LookTarget = fn(&dyn PathfinderMob) -> Option<SharedEntity>;
17
18/// Stands still for the length of a spell.
19///
20/// Vanilla parity: `SpellcasterIllager.SpellcasterCastingSpellGoal`.
21pub(crate) struct SpellcasterCastingSpellGoal {
22    /// What the caster watches while casting.
23    look_target: LookTarget,
24}
25
26impl SpellcasterCastingSpellGoal {
27    /// Creates the goal, watching whatever the caster is attacking.
28    #[must_use]
29    pub(crate) const fn new() -> Self {
30        Self {
31            look_target: Mob::target,
32        }
33    }
34
35    /// Replaces what the caster watches while casting.
36    ///
37    /// Vanilla parity: `Evoker.EvokerCastingSpellGoal`, which falls back to the
38    /// wololo target when there is nothing to attack.
39    #[must_use]
40    pub(crate) const fn with_look_target(mut self, look_target: LookTarget) -> Self {
41        self.look_target = look_target;
42        self
43    }
44}
45
46impl Default for SpellcasterCastingSpellGoal {
47    fn default() -> Self {
48        Self::new()
49    }
50}
51
52impl Goal for SpellcasterCastingSpellGoal {
53    fn controls(&self) -> GoalControls {
54        GoalControls::MOVE | GoalControls::LOOK
55    }
56
57    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
58        mob.as_spellcaster_illager()
59            .is_some_and(|caster| caster.spell_casting_time() > 0)
60    }
61
62    fn start(&mut self, mob: &dyn PathfinderMob) {
63        mob.mob_base().navigation().lock().stop();
64    }
65
66    fn stop(&mut self, mob: &dyn PathfinderMob) {
67        if let Some(caster) = mob.as_spellcaster_illager() {
68            caster.set_is_casting_spell(IllagerSpell::None);
69        }
70    }
71
72    fn tick(&mut self, mob: &dyn PathfinderMob) {
73        let Some(watched) = (self.look_target)(mob) else {
74            return;
75        };
76        Mob::look_at(
77            mob,
78            watched.as_ref(),
79            mob.max_head_y_rot(),
80            mob.max_head_x_rot(),
81        );
82    }
83}
