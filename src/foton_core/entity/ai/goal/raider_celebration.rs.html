1//! Raider celebration goal.
2//!
3//! Vanilla parity: `Raider.RaiderCelebration`. When a raid wins, the survivors
4//! stop attacking, jump on the spot and shout. It is the only place the
5//! celebrating flag is ever set, and the flag is what puts a vindicator's arms
6//! down and a spellcaster's hands together.
7//!
8//! The goal is written whole, but it can never start: its gate is a raid the
9//! village lost, and Foton has no raid to lose. See
10//! [`crate::entity::raider`] for why.
11
12use super::reduced_tick_delay;
13use super::selector::{Goal, GoalControls};
14use crate::entity::{LivingEntity, PathfinderMob};
15
16/// Average ticks between two celebratory shouts.
17///
18/// Vanilla parity: the `nextInt(adjustedTickDelay(100))` of `tick`.
19const SHOUT_INTERVAL: i32 = 100;
20
21/// Average ticks between two hops.
22///
23/// Vanilla parity: the `nextInt(adjustedTickDelay(50))` of `tick`.
24const JUMP_INTERVAL: i32 = 50;
25
26/// Jumps and shouts over a fallen village.
27///
28/// Vanilla parity: `Raider.RaiderCelebration`.
29pub(crate) struct RaiderCelebrationGoal;
30
31impl RaiderCelebrationGoal {
32    /// Creates the goal.
33    #[must_use]
34    pub(crate) const fn new() -> Self {
35        Self
36    }
37}
38
39impl Default for RaiderCelebrationGoal {
40    fn default() -> Self {
41        Self::new()
42    }
43}
44
45impl Goal for RaiderCelebrationGoal {
46    fn controls(&self) -> GoalControls {
47        GoalControls::MOVE
48    }
49
50    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
51        let Some(raider) = mob.as_raider() else {
52            return false;
53        };
54        LivingEntity::is_alive(mob)
55            && mob.target().is_none()
56            && raider
57                .current_raid_status()
58                .is_some_and(|status| status.loss)
59    }
60
61    fn start(&mut self, mob: &dyn PathfinderMob) {
62        if let Some(raider) = mob.as_raider() {
63            raider.set_celebrating(true);
64        }
65    }
66
67    fn stop(&mut self, mob: &dyn PathfinderMob) {
68        if let Some(raider) = mob.as_raider() {
69            raider.set_celebrating(false);
70        }
71    }
72
73    fn tick(&mut self, mob: &dyn PathfinderMob) {
74        let Some(raider) = mob.as_raider() else {
75            return;
76        };
77
78        // The two bounds are halved because this goal does not ask to be
79        // ticked every tick and a mob's goals only tick on even ticks
80        // otherwise; that is vanilla's `Goal.adjustedTickDelay`.
81        if !mob.is_silent() && rand::random_range(0..reduced_tick_delay(SHOUT_INTERVAL)) == 0 {
82            mob.play_sound(raider.celebrate_sound(), 1.0, 1.0);
83        }
84        if !mob.is_passenger() && rand::random_range(0..reduced_tick_delay(JUMP_INTERVAL)) == 0 {
85            mob.jump_control_jump();
86        }
87    }
88}
