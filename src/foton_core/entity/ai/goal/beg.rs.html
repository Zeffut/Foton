14/// Shortest time a wolf keeps begging for.
15///
16/// Vanilla parity: the `40 + random.nextInt(40)` of `BegGoal.start`.
17const MIN_LOOK_TICKS: i32 = 40;
18
19/// Extra random time on top of [`MIN_LOOK_TICKS`].
20const EXTRA_LOOK_TICKS: i32 = 40;
21
22/// Makes a wolf sit up and stare at food a player is holding.
23///
24/// Vanilla parity: `BegGoal`. The only server-visible effect is the synced
25/// "interested" flag, which is what tilts the wolf's head on the client.
26pub struct BegGoal {
27    player: Option<Arc<Player>>,
28    look_distance: f64,
29    look_time: i32,
30    interesting: Box<dyn Fn(&ItemStack) -> bool + Send>,
31}
32
33impl BegGoal {
34    /// Creates the goal with the predicate for "food worth begging for".
35    ///
36    /// Vanilla hardcodes `itemStack.is(Items.BONE) || wolf.isFood(itemStack)`;
37    /// the predicate is a parameter here so the wolf keeps owning what its own
38    /// food is.
39    #[must_use]
40    pub(crate) fn new(
41        look_distance: f64,
42        interesting: impl Fn(&ItemStack) -> bool + Send + 'static,
43    ) -> Self {
44        Self {
45            player: None,
46            look_distance,
47            look_time: 0,
48            interesting: Box::new(interesting),
49        }
50    }
51
52    fn player_holding_interesting(&self, player: &Player) -> bool {
53        player.is_holding(&mut |item_stack| (self.interesting)(item_stack))
54    }
55
56    fn find_player(&mut self, mob: &dyn PathfinderMob) {
57        let Some(world) = mob.level() else {
58            self.player = None;
59            return;
60        };
61
62        let beg_targeting = TargetingConditions::for_non_combat().range(self.look_distance);
63        self.player = world.nearest_player(mob.position(), self.look_distance, |player| {
64            beg_targeting.test(world.as_ref(), Some(mob), player)
65        });
66    }
67}
68
69impl Goal for BegGoal {
70    fn controls(&self) -> GoalControls {
71        GoalControls::LOOK
72    }
73
74    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
75        self.find_player(mob);
76        let Some(player) = self.player.clone() else {
77            return false;
78        };
79        self.player_holding_interesting(&player)
80    }
81
82    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
83        let Some(player) = self.player.clone() else {
84            return false;
85        };
86        if !Entity::is_alive(player.as_ref()) {
87            return false;
88        }
89        if mob.position().distance_squared(player.position())
90            > self.look_distance * self.look_distance
91        {
92            return false;
93        }
94
95        self.look_time > 0 && self.player_holding_interesting(&player)
96    }
97
98    fn start(&mut self, mob: &dyn PathfinderMob) {
99        set_interested(mob, true);
100        self.look_time =
101            reduced_tick_delay(MIN_LOOK_TICKS + rand::random_range(0..EXTRA_LOOK_TICKS));
102    }
103
104    fn stop(&mut self, mob: &dyn PathfinderMob) {
105        set_interested(mob, false);
106        self.player = None;
107    }
108
109    fn tick(&mut self, mob: &dyn PathfinderMob) {
110        let Some(player) = &self.player else {
111            return;
112        };
113
114        let position = player.position();
115        mob.mob_base().controls().lock().look_control.set_look_at(
116            DVec3::new(position.x, player.get_eye_y(), position.z),
117            10.0,
118            mob.max_head_x_rot(),
119        );
120        self.look_time -= 1;
121    }
122}

124/// Sets vanilla `Wolf.setIsInterested`.
125///
126/// Vanilla's `BegGoal` is typed on `Wolf` and calls the setter directly; only
127/// the wolf has the synced flag, so the goal reaches for the wolf here too.
128fn set_interested(mob: &dyn PathfinderMob, interested: bool) {
129    if let Some(wolf) = mob.downcast_ref::<WolfEntity>() {
130        wolf.set_is_interested(interested);
131    }
