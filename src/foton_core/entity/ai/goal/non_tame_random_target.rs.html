1use super::nearest_attackable_target::NearestAttackableTargetGoal;
2use super::selector::{Goal, GoalControls};
3use crate::entity::{LivingEntity, PathfinderMob, TamableAnimal};
4use crate::world::World;

6/// How often an untamed pet looks for prey.
7///
8/// Vanilla parity: the `10` `NonTameRandomTargetGoal` passes to its super.
9const RANDOM_INTERVAL: i32 = 10;
10
11/// Lets an untamed pet hunt, and stops the moment it is tamed.
12///
13/// Vanilla parity: `NonTameRandomTargetGoal`. This is why a wild wolf chases
14/// sheep and a tamed one walks past them.
15pub struct NonTameRandomTargetGoal {
16    nearest: NearestAttackableTargetGoal,
17}
18
19impl NonTameRandomTargetGoal {
20    #[must_use]
21    pub(crate) fn new(
22        must_see: bool,
23        selector: impl Fn(Option<&dyn LivingEntity>, &dyn LivingEntity, &World) -> bool
24        + Send
25        + Sync
26        + 'static,
27    ) -> Self {
28        Self {
29            nearest: NearestAttackableTargetGoal::new_with_interval(
30                RANDOM_INTERVAL,
31                must_see,
32                false,
33                selector,
34            ),
35        }
36    }
37}
38
39impl Goal for NonTameRandomTargetGoal {
40    fn controls(&self) -> GoalControls {
41        GoalControls::TARGET
42    }
43
44    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
45        let is_tame = mob.as_tamable_animal().is_some_and(TamableAnimal::is_tame);
46        !is_tame && self.nearest.can_use(mob)
47    }
48
49    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
50        self.nearest.current_target_passes_conditions(mob)
51    }
52
53    fn start(&mut self, mob: &dyn PathfinderMob) {
54        self.nearest.start(mob);
55    }
56
57    fn stop(&mut self, mob: &dyn PathfinderMob) {
58        self.nearest.stop(mob);
59    }
60}
