1//! Drifting somewhere else for no reason at all.
2//!
3//! Vanilla parity: `Ghast.RandomFloatAroundGoal`, a public goal shared by the
4//! ghast and the happy ghast. It is the whole of a ghast's idle movement: pick
5//! a point within sixteen blocks, hand it to the move control, and let the move
6//! control coast there.
7

17/// How many spots the goal tries before it settles for whatever it last drew.
18///
19/// Vanilla parity: `RandomFloatAroundGoal.MAX_ATTEMPTS`.
20const MAX_ATTEMPTS: i32 = 64;
21
22/// Half the span of one axis of the search box, in blocks.
23///
24/// Vanilla parity: the `16.0F` of `chooseRandomPosition`.
25const SEARCH_SPAN: f64 = 16.0;
26
27/// Squared distance below which the current destination is too close to bother
28/// flying to.
29///
30/// Vanilla parity: the `dd < 1.0` of `RandomFloatAroundGoal.canUse`.
31const TOO_CLOSE_SQR: f64 = 1.0;
32
33/// Squared distance beyond which the current destination is written off.
34///
35/// Vanilla parity: the `dd > 3600.0` of the same method.
36const TOO_FAR_SQR: f64 = 3600.0;
37
38/// Speed multiplier of an idle drift.
39const FLOAT_SPEED_MODIFIER: f64 = 1.0;
40
41/// Picks somewhere for a floating mob to drift to.
42///
43/// Vanilla parity: `Ghast.RandomFloatAroundGoal.getSuitableFlyToPosition`.
44/// `distance_to_blocks` is the happy ghast's "stay near something solid"
45/// preference; a ghast passes `0`, which accepts any point.
46pub(crate) fn suitable_fly_to_position(mob: &dyn Mob, distance_to_blocks: i32) -> DVec3 {
47    let Some(world) = mob.level() else {
48        return mob.position();
49    };
50    let center = mob.position();
51    let mut result = None;
52
53    for _ in 0..MAX_ATTEMPTS {
54        let candidate = choose_random_position_with_restriction(mob, center);
55        if candidate.is_some() {
56            result = candidate;
57        }
58        if let Some(candidate) = candidate
59            && is_good_target(&world, candidate, distance_to_blocks)
60        {
61            return candidate;
62        }
63    }
64
65    let mut result = result.unwrap_or_else(|| choose_random_position(center));
66
67    // Vanilla parity: a target under the terrain is mirrored back to the same
68    // distance below the mob instead, so a ghast that rolls a point inside a
69    // hill dives rather than swimming through it.
70    let pos = BlockPos::containing(result.x, result.y, result.z);
71    let height_y = world.heightmap_at(HeightmapType::MotionBlocking, pos.x(), pos.z());
72    if height_y < pos.y() && height_y > world.min_y() {
73        result = DVec3::new(result.x, center.y - (center.y - result.y).abs(), result.z);
74    }
75
76    result
77}

79/// Vanilla parity: `RandomFloatAroundGoal.isGoodTarget`.
80fn is_good_target(world: &World, target: DVec3, distance_to_blocks: i32) -> bool {
81    if distance_to_blocks <= 0 {
82        return true;
83    }
84
85    let pos = BlockPos::containing(target.x, target.y, target.z);
86    if !world.get_block_state(pos).is_air() {
87        return false;
88    }
89
90    Direction::ALL.into_iter().any(|direction| {
91        (1..distance_to_blocks).any(|step| {
92            !world
93                .get_block_state(pos.relative_n(direction, step))
94                .is_air()
95        })
96    })
97}

99/// Vanilla parity: `RandomFloatAroundGoal.chooseRandomPosition`.
100fn choose_random_position(center: DVec3) -> DVec3 {
101    fn offset() -> f64 {
102        f64::from(rand::random::<f32>().mul_add(2.0, -1.0)) * SEARCH_SPAN
103    }
104
105    DVec3::new(
106        center.x + offset(),
107        center.y + offset(),
108        center.z + offset(),
109    )
110}
111
112/// Vanilla parity: `RandomFloatAroundGoal.chooseRandomPositionWithRestriction`.
113fn choose_random_position_with_restriction(mob: &dyn Mob, center: DVec3) -> Option<DVec3> {
114    let target = choose_random_position(center);
115    if mob.has_home() && !mob.is_within_home_vec(target) {
116        return None;
117    }
118
119    Some(target)
120}

122/// Drifts to a random point whenever the move control has nowhere useful to be.
123///
124/// Vanilla parity: `Ghast.RandomFloatAroundGoal`.
125pub(crate) struct RandomFloatAroundGoal {
126    distance_to_blocks: i32,
127}
128
129impl RandomFloatAroundGoal {
130    /// Creates the ghast's own configuration, which accepts any point.
131    #[must_use]
132    pub(crate) const fn new() -> Self {
133        Self::with_distance_to_blocks(0)
134    }
135
136    /// Creates a drift that prefers points near solid blocks.
137    ///
138    /// Vanilla parity: the two-argument `RandomFloatAroundGoal(mob, distance)`.
139    #[must_use]
140    pub(crate) const fn with_distance_to_blocks(distance_to_blocks: i32) -> Self {
141        Self { distance_to_blocks }
142    }
143}
144
145impl Goal for RandomFloatAroundGoal {
146    fn controls(&self) -> GoalControls {
147        GoalControls::MOVE
148    }
149
150    /// Vanilla parity: `RandomFloatAroundGoal.canUse`. The mob re-rolls when it
151    /// has nowhere to be, when it has already arrived, and when whatever it was
152    /// aiming at has ended up sixty blocks away.
153    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
154        let (has_wanted, wanted_position) = {
155            let controls = mob.mob_base().controls().lock();
156            (
157                controls.move_control.has_wanted(),
158                controls.move_control.wanted_position(),
159            )
160        };
161        if !has_wanted {
162            return true;
163        }
164
165        let distance_sqr = wanted_position.distance_squared(mob.position());
166        distance_sqr < TOO_CLOSE_SQR || distance_sqr > TOO_FAR_SQR
167    }
168
169    /// Vanilla parity: `RandomFloatAroundGoal.canContinueToUse` returns false,
170    /// so the goal runs for exactly the one tick it takes to pick a point.
171    fn can_continue_to_use(&mut self, _mob: &dyn PathfinderMob) -> bool {
172        false
173    }
174
175    fn start(&mut self, mob: &dyn PathfinderMob) {
176        let target = suitable_fly_to_position(mob, self.distance_to_blocks);
177        mob.mob_base()
178            .controls()
179            .lock()
180            .move_control
181            .set_wanted_position(target, FLOAT_SPEED_MODIFIER);
182    }
183}
