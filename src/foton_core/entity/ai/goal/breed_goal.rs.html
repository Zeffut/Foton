11const PARTNER_SEARCH_RANGE: f64 = 8.0;
12const BREED_DISTANCE_SQR: f64 = 9.0;
13const BREED_TIME: i32 = 60;
14
15/// What a pair of animals does once they have finished courting.
16///
17/// Vanilla parity: the overridable `BreedGoal.breed`. A turtle replaces it to
18/// carry an egg away instead of producing a hatchling on the spot.
19type BreedAction = Box<dyn Fn(&Arc<World>, &dyn Animal, &dyn Animal) + Send + Sync>;
20
21pub struct BreedGoal {
22    partner: Option<SharedEntity>,
23    love_time: i32,
24    speed_modifier: f64,
25    breed: Option<BreedAction>,
26}
27
28impl BreedGoal {
29    #[must_use]
30    pub(crate) const fn new(speed_modifier: f64) -> Self {
31        Self {
32            partner: None,
33            love_time: 0,
34            speed_modifier,
35            breed: None,
36        }
37    }
38
39    /// Creates a breed goal that ends in something other than a newborn.
40    ///
41    /// Vanilla parity: overriding `BreedGoal.breed`.
42    #[must_use]
43    pub(crate) fn with_breed_action(
44        speed_modifier: f64,
45        breed: impl Fn(&Arc<World>, &dyn Animal, &dyn Animal) + Send + Sync + 'static,
46    ) -> Self {
47        Self {
48            partner: None,
49            love_time: 0,
50            speed_modifier,
51            breed: Some(Box::new(breed)),
52        }
53    }
54
55    fn get_free_partner(mob: &dyn PathfinderMob, animal: &dyn Animal) -> Option<SharedEntity> {
56        let world = mob.level()?;
57        let search_box = mob.bounding_box().inflate(PARTNER_SEARCH_RANGE);
58        let partner_targeting = TargetingConditions::for_non_combat()
59            .range(PARTNER_SEARCH_RANGE)
60            .ignore_line_of_sight();
61
62        world.nearest_entity_in_aabb_matching(&search_box, mob.position(), |entity| {
63            let Some(candidate) = entity.as_animal() else {
64                return false;
65            };
66            if !partner_targeting.test(world.as_ref(), Some(mob), candidate) {
67                return false;
68            }
69            if !animal.can_mate(candidate) {
70                return false;
71            }
72
73            !entity
74                .as_pathfinder_mob()
75                .is_some_and(PathfinderMob::is_panicking)
76        })
77    }
78}
79
80impl Goal for BreedGoal {
81    fn controls(&self) -> GoalControls {
82        GoalControls::MOVE | GoalControls::LOOK
83    }
84
85    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
86        let Some(animal) = mob.as_animal() else {
87            return false;
88        };
89        if !animal.is_in_love() {
90            return false;
91        }
92
93        self.partner = Self::get_free_partner(mob, animal);
94        self.partner.is_some()
95    }
96
97    fn can_continue_to_use(&mut self, _mob: &dyn PathfinderMob) -> bool {
98        let Some(partner) = &self.partner else {
99            return false;
100        };
101        if !partner.is_alive() {
102            return false;
103        }
104        if self.love_time >= BREED_TIME {
105            return false;
106        }
107        if partner
108            .as_pathfinder_mob()
109            .is_some_and(PathfinderMob::is_panicking)
110        {
111            return false;
112        }
113
114        partner.as_animal().is_some_and(Animal::is_in_love)
115    }
116
117    fn stop(&mut self, _mob: &dyn PathfinderMob) {
118        self.partner = None;
119        self.love_time = 0;
120    }
121
122    fn tick(&mut self, mob: &dyn PathfinderMob) {
123        let Some(partner) = &self.partner else {
124            return;
125        };
126        let Some(animal) = mob.as_animal() else {
127            return;
128        };
129        let Some(partner_animal) = partner.as_animal() else {
130            return;
131        };
132
133        let partner_position = partner.position();
134        mob.mob_base().controls().lock().look_control.set_look_at(
135            DVec3::new(partner_position.x, partner.get_eye_y(), partner_position.z),
136            10.0,
137            mob.max_head_x_rot(),
138        );
139        mob.move_to_pos(partner_position, self.speed_modifier);
140
141        self.love_time += 1;
142        if self.love_time < reduced_tick_delay(BREED_TIME)
143            || mob.position().distance_squared(partner_position) >= BREED_DISTANCE_SQR
144        {
145            return;
146        }
147
148        let Some(world) = mob.level() else {
149            return;
150        };
151        match &self.breed {
152            Some(breed) => breed(&world, animal, partner_animal),
153            None => animal.spawn_child_from_breeding(&world, partner_animal),
154        }
155    }
156}
