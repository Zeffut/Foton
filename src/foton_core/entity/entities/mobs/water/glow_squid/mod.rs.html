1//! Glow squid entity.
2//!
3//! Vanilla parity: `GlowSquid`, which extends `Squid` and changes four sounds,
4//! one particle, and the dark ticks it counts after being hurt. Everything else
5//! comes from [`super::squid_common`], which is what makes this file short.
6//!
7//! The dark ticks are the whole point of the mob: hurt one and it stops
8//! glowing for five seconds, so the light a cave gets from a shoal of them is
9//! something a player can put out.
10

54/// A glow squid.

71impl GlowSquidEntity {
72    /// Creates a glow squid at runtime.
73    #[must_use]
74    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
75        Self::new_with_base(
76            EntityBase::new(id, position, entity_type.dimensions, world),
77            entity_type,
78        )
79    }
80
81    /// Creates a glow squid from saved base data.
82    #[must_use]
83    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
84        Self::new_with_base(
85            EntityBase::from_load(load, entity_type.dimensions),
86            entity_type,
87        )
88    }
89
90    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
91        let living_base = LivingEntityBase::new(entity_type);
92        let mob_base = MobBase::new();
93        let ageable_base = AgeableMobBase::new();
94        let mut entity_data = GlowSquidEntityData::new();
95        living_base.initialize_synced_data(&mut entity_data);
96
97        {
98            let mut goals = mob_base.goal_selector().lock();
99            let hooks = squid_common::hooks_for::<Self>();
100            goals.add_goal(0, SquidRandomMovementGoal::new(hooks));
101            goals.add_goal(1, SquidFleeGoal::new(hooks));
102        }
103
104        Self {
105            base,
106            entity_type,
107            living_base,
108            mob_base,
109            ageable_base,
110            entity_data: SyncMutex::new(entity_data),
111            state: SyncMutex::new(SquidState::new()),
112        }
113    }
114
115    /// Returns how much longer this squid stays dark.
116    #[must_use]
117    pub fn dark_ticks_remaining(&self) -> i32 {
118        *self.entity_data.lock().dark_ticks_remaining.get()
119    }
120
121    fn set_dark_ticks(&self, ticks: i32) {
122        self.entity_data.lock().dark_ticks_remaining.set(ticks);
123    }
124}

139    /// Vanilla parity: `AgeableWaterCreature.baseTick`, which reads the air left
140    /// before the shared tick spends it.
141    fn base_tick(&self) {
142        let air_before_tick = self.air_supply();
143        Mob::base_tick_mob(self);
144        if let Some(world) = self.level() {
145            squid_common::handle_air_supply(self, &world, air_before_tick);
146        }
147    }
148
149    fn get_default_gravity(&self) -> f64 {
150        GRAVITY
151    }
152
153    fn dimensions_for_pose(&self, _pose: EntityPose) -> EntityDimensions {
154        let scale = LivingEntity::get_scale(self);
155        if AgeableMob::is_baby(self) {
156            BABY_DIMENSIONS.scale(scale)
157        } else {
158            self.entity_type.dimensions.scale(scale)
159        }
160    }
161
162    fn sound_source(&self) -> SoundSource {
163        SoundSource::Neutral
164    }
165
166    fn movement_emission(&self) -> EntityMovementEmission {
167        EntityMovementEmission::Events
168    }
169
170    fn play_step_sound(&self, _pos: BlockPos, _block_state: BlockStateId) {}
171
172    fn save_additional(&self, nbt: &mut NbtCompound) {
173        self.save_mob(nbt);
174        self.save_ageable_mob(nbt);
175        nbt.insert("DarkTicksRemaining", self.dark_ticks_remaining());
176    }
177
178    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
179        self.load_mob(nbt);
180        self.load_ageable_mob(nbt);
181        self.set_dark_ticks(nbt.int("DarkTicksRemaining").unwrap_or(0));
182    }
183}

225    /// Inks, and goes dark for five seconds.
226    ///
227    /// Vanilla parity: `GlowSquid.hurtServer` on top of `Squid.hurtServer`.
228    /// Going dark is what makes a shoal of these a light source a player can
229    /// switch off.
230    fn before_actually_hurt(&self, source: &DamageSource, _amount: f32) {
231        self.set_dark_ticks(DARK_TICKS_ON_HURT);
232
233        if source.causing_entity_id.is_none() {
234            return;
235        }
236        if let Some(world) = self.level() {
237            squid_common::spawn_ink(
238                self,
239                &world,
240                &sound_events::ENTITY_GLOW_SQUID_SQUIRT,
241                &vanilla_particle_types::GLOW_SQUID_INK,
242            );
243        }
244    }
245
246    fn ai_step(&self) -> Option<MoveResult> {
247        let result = self.default_ai_step();
248        squid_common::tick_tentacles(self, &self.state);
249        if self.is_in_water() {
250            squid_common::face_travel_direction(self);
251        }
252
253        // Vanilla parity: `GlowSquid.aiStep` counts the darkness down.
254        let dark = self.dark_ticks_remaining();
255        if dark > 0 {
256            self.set_dark_ticks(dark - 1);
257        }
258
259        AgeableMob::tick_ageable_mob(self);
260        result
261    }
262
263    /// Vanilla parity: `Squid.travel`, which throws away the travel input.
264    fn travel(&self, _input: DVec3) -> Option<MoveResult> {
265        self.move_entity(MoverType::SelfMovement, self.velocity())
266    }
267}

330    /// Returns whether this mob accepts where the spawner put it.
331    ///
332    /// Vanilla parity: `GlowSquid.checkGlowSquidSpawnRules` adds a darkness
333    /// test to the surface water rule, which is why they gather in flooded
334    /// caves rather than in open sea.
335    fn check_spawn_rules(
336        &self,
337        world: &Arc<World>,
338        _spawn_reason: EntitySpawnReason,
339        pos: BlockPos,
340    ) -> bool {
341        use crate::world::LevelReader as _;
342        check_surface_water_animal_spawn_rules(world, pos)
343            && world.max_local_raw_brightness(pos, world.sky_darkening()) == 0
344    }
345
346    fn finalize_spawn(
347        &self,
348        world: &Arc<World>,
349        spawn_reason: EntitySpawnReason,
350        group_data: Option<SpawnGroupData>,
351    ) -> Option<SpawnGroupData> {
352        let group_data = group_data.unwrap_or(SpawnGroupData::AgeableMob(
353            AgeableMobGroupData::with_baby_spawn_chance(BABY_SPAWN_CHANCE),
354        ));
355        self.finalize_spawn_ageable_mob(world, spawn_reason, Some(group_data))
356    }

360    }
361

363        self.entity_data.lock().mob_mut().mob_flags.set(flags);
364    }
