1//! Pufferfish entity.
2//!
3//! Vanilla parity: `Pufferfish`, `AbstractFish` and `WaterAnimal`. A pufferfish
4//! is a cod with a threat display: anything scary within two blocks inflates
5//! it over two stages, and while it is inflated it stings whatever touches it
6//! for damage and poison that both scale with how puffed up it is. Deflating
7//! is on a timer of its own, which is why it stays a ball for a while after
8//! the swimmer has gone.
9

46/// Vanilla `Pufferfish.STATE_SMALL`.
47const STATE_SMALL: i32 = 0;
48/// Vanilla `Pufferfish.STATE_MID`.
49const STATE_MID: i32 = 1;
50/// Vanilla `Pufferfish.STATE_FULL`.
51const STATE_FULL: i32 = 2;
52
53/// Ticks of inflating before the second stage.
54///
55/// Vanilla parity: the `inflateCounter > 40` of `Pufferfish.tick`.
56const MID_TO_FULL_TICKS: i32 = 40;
57/// Ticks of calm before the fish drops back to half puffed.
58///
59/// Vanilla parity: the `deflateTimer > 60` of the same method.
60const FULL_TO_MID_TICKS: i32 = 60;
61/// Ticks of calm before it is flat again.
62const MID_TO_SMALL_TICKS: i32 = 100;
63
64/// How far a pufferfish notices something worth puffing up at.
65///
66/// Vanilla parity: the `inflate(2.0)` of `PufferfishPuffGoal.canUse`.
67const PUFF_SEARCH_RANGE: f64 = 2.0;
68/// How far past its own box a puffed fish stings.
69///
70/// Vanilla parity: the `inflate(0.3)` of `Pufferfish.aiStep`.
71const STING_REACH: f64 = 0.3;
72/// Poison ticks a sting inflicts per puff stage.
73///
74/// Vanilla parity: the `60 * puffState` of `Pufferfish.touch`.
75const POISON_TICKS_PER_STATE: i32 = 60;

113    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
114        let living_base = LivingEntityBase::new(entity_type);
115        let mob_base = MobBase::new();
116        // Vanilla parity: `WaterAnimal` clears the water malus.
117        mob_base
118            .pathfinding_malus()
119            .lock()
120            .set(PathType::Water, 0.0);
121        let mut entity_data = PufferfishEntityData::new();
122        living_base.initialize_synced_data(&mut entity_data);
123
124        {
125            // Vanilla parity: `AbstractFish.registerGoals` plus the puff goal.
126            let mut goals = mob_base.goal_selector().lock();
127            goals.add_goal(0, PanicGoal::new(fish::PANIC_SPEED_MODIFIER));
128            goals.add_goal(1, PufferfishPuffGoal::new());
129            goals.add_goal(
130                2,
131                AvoidEntityGoal::with_selector(
132                    fish::AVOID_PLAYER_RANGE,
133                    fish::AVOID_WALK_SPEED,
134                    fish::AVOID_SPRINT_SPEED,
135                    |_, target, _| fish::is_player_to_flee(target),
136                ),
137            );
138            goals.add_goal(
139                4,
140                RandomSwimmingGoal::new(fish::SWIM_SPEED_MODIFIER, fish::SWIM_INTERVAL_TICKS),
141            );
142        }
143
144        let pufferfish = Self {
145            base,
146            entity_type,
147            living_base,
148            mob_base,
149            inflate_counter: SyncMutex::new(0),
150            deflate_timer: SyncMutex::new(0),
151            entity_data: SyncMutex::new(entity_data),
152        };
153        // Vanilla parity: the `refreshDimensions()` of the `Pufferfish`
154        // constructor. Without it the fish carries the unscaled registry hitbox
155        // until its first puff.
156        pufferfish.refresh_dimensions();
157        pufferfish
158    }

197    /// Vanilla parity: `Pufferfish.getScale`.
198    #[must_use]
199    const fn puff_scale(puff_state: i32) -> f32 {
200        match puff_state {
201            STATE_SMALL => 0.5,
202            STATE_MID => 0.7,
203            _ => 1.0,
204        }

207    /// Returns whether this entity is one a pufferfish inflates at.
208    ///
209    /// Vanilla parity: the `SCARY_MOB` selector, which spares creative players
210    /// and everything in `not_scary_for_pufferfish` -- the other pufferfish,
211    /// the guardians and the axolotl among them.
212    fn is_scary(target: &dyn LivingEntity) -> bool {
213        if target
214            .as_player()
215            .is_some_and(|player| player.game_mode() == GameType::Creative)
216        {
217            return false;
218        }
219
220        !REGISTRY.entity_types.is_in_tag(
221            target.entity_type(),
222            &EntityTypeTag::NOT_SCARY_FOR_PUFFERFISH,
223        )
224    }
225
226    /// Vanilla parity: the `TARGETING_CONDITIONS` of `Pufferfish`.
227    fn targeting_conditions() -> TargetingConditions {
228        TargetingConditions::for_non_combat()
229            .ignore_invisibility_testing()
230            .ignore_line_of_sight()
231            .selector(|_, target, _| Self::is_scary(target))
232    }

234    /// Vanilla parity: `Pufferfish.tick`, the inflate and deflate clocks.
235    fn tick_puff_state(&self) {
236        if !Entity::is_alive(self) || !self.is_effective_ai() {
237            return;
238        }
239
240        let inflate_counter = *self.inflate_counter.lock();
241        if inflate_counter > 0 {
242            let puff_state = self.puff_state();
243            if puff_state == STATE_SMALL {
244                self.make_sound(Some(&sound_events::ENTITY_PUFFER_FISH_BLOW_UP));
245                self.set_puff_state(STATE_MID);
246            } else if inflate_counter > MID_TO_FULL_TICKS && puff_state == STATE_MID {
247                self.make_sound(Some(&sound_events::ENTITY_PUFFER_FISH_BLOW_UP));
248                self.set_puff_state(STATE_FULL);
249            }
250
251            *self.inflate_counter.lock() += 1;
252            return;
253        }
254
255        if self.puff_state() == STATE_SMALL {
256            return;
257        }
258
259        let deflate_timer = *self.deflate_timer.lock();
260        if deflate_timer > FULL_TO_MID_TICKS && self.puff_state() == STATE_FULL {
261            self.make_sound(Some(&sound_events::ENTITY_PUFFER_FISH_BLOW_OUT));
262            self.set_puff_state(STATE_MID);
263        } else if deflate_timer > MID_TO_SMALL_TICKS && self.puff_state() == STATE_MID {
264            self.make_sound(Some(&sound_events::ENTITY_PUFFER_FISH_BLOW_OUT));
265            self.set_puff_state(STATE_SMALL);
266        }
267
268        *self.deflate_timer.lock() += 1;
269    }
270
271    /// Vanilla parity: the sting sweep of `Pufferfish.aiStep`.
272    fn sting_nearby_mobs(&self) {
273        let puff_state = self.puff_state();
274        if puff_state <= STATE_SMALL || !Entity::is_alive(self) {
275            return;
276        }
277        let Some(world) = self.level() else {
278            return;
279        };
280
281        let targeting = Self::targeting_conditions();
282        let search_box = self.bounding_box().inflate(STING_REACH);
283        for entity in world.get_entities_in_aabb_matching(&search_box, |entity| {
284            entity.is_mob()
285                && entity
286                    .as_living_entity()
287                    .is_some_and(|target| targeting.test(world.as_ref(), Some(self), target))
288        }) {
289            if !entity.is_alive() {
290                continue;
291            }
292            let Some(target) = entity.as_living_entity() else {
293                continue;
294            };
295            self.touch(&world, target, puff_state);
296        }
297    }
298
299    /// Vanilla parity: `Pufferfish.touch`.
300    fn touch(&self, world: &Arc<World>, target: &dyn LivingEntity, puff_state: i32) {
301        let damage_source = DamageSource::environment(&vanilla_damage_types::MOB_ATTACK)
302            .with_causing_entity(self.id())
303            .with_direct_entity(self.id())
304            .with_source_position(self.position());
305        if !target.hurt_server(world, &damage_source, (1 + puff_state) as f32) {
306            return;
307        }
308
309        target.add_mob_effect(MobEffectInstance::with_duration(
310            vanilla_mob_effects::POISON,
311            POISON_TICKS_PER_STATE * puff_state,
312            0,
313        ));
314        self.play_sound(&sound_events::ENTITY_PUFFER_FISH_STING, 1.0, 1.0);
315    }
316}

339        let air_before_tick = self.air_supply();

346    /// Vanilla parity: `Pufferfish.getDefaultDimensions`, which scales with how
347    /// puffed up the fish is.
348    fn dimensions_for_pose(&self, _pose: EntityPose) -> EntityDimensions {
349        let scale = LivingEntity::get_scale(self) * Self::puff_scale(self.puff_state());
350        if self.entity_type.fixed {
351            self.entity_type.dimensions
352        } else {
353            self.entity_type.dimensions.scale(scale)
354        }
355    }

358    /// a current rather than being swept along by it.

360        false
361    }

363    fn sound_source(&self) -> SoundSource {
364        SoundSource::Neutral
365    }
366
367    /// Vanilla parity: `AbstractFish.playStepSound` is empty.
368    fn play_step_sound(&self, _pos: BlockPos, _block_state: BlockStateId) {}
369
370    /// Vanilla parity: `Pufferfish.playerTouch`, which is the sting a swimmer
371    /// gets for brushing past, complete with the client-side screen event.
372    fn player_touch(self: Arc<Self>, player: &Arc<Player>) {
373        let puff_state = self.puff_state();
374        if puff_state <= STATE_SMALL {
375            return;
376        }
377        let Some(world) = self.level() else {
378            return;
379        };
380
381        let damage_source = DamageSource::environment(&vanilla_damage_types::MOB_ATTACK)
382            .with_causing_entity(self.id())
383            .with_direct_entity(self.id())
384            .with_source_position(self.position());
385        if !player.hurt_server(&world, &damage_source, (1 + puff_state) as f32) {
386            return;
387        }
388
389        if !self.is_silent() {
390            player.send_packet(CGameEvent {
391                event: GameEventType::PufferFishSting,
392                data: 0.0,
393            });
394        }
395
396        player.add_mob_effect(MobEffectInstance::with_duration(
397            vanilla_mob_effects::POISON,
398            POISON_TICKS_PER_STATE * puff_state,
399            0,
400        ));
401    }
402
403    fn save_additional(&self, nbt: &mut NbtCompound) {
404        self.save_mob(nbt);
405        nbt.insert("FromBucket", self.from_bucket());
406        nbt.insert("PuffState", self.puff_state());
407    }
408
409    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
410        self.load_mob(nbt);
411        self.set_from_bucket(nbt.byte("FromBucket").is_some_and(|flag| flag != 0));
412        self.set_puff_state(nbt.int("PuffState").unwrap_or(STATE_SMALL).min(STATE_FULL));
413    }

537/// Inflates the fish while something frightening is beside it.
538///
539/// Vanilla parity: `Pufferfish.PufferfishPuffGoal`. The goal only starts and
540/// stops the inflate counter; the two-stage clock itself is in `tick`.
541struct PufferfishPuffGoal;
542
543impl PufferfishPuffGoal {
544    const fn new() -> Self {
545        Self
546    }
547}
548
549impl Goal for PufferfishPuffGoal {
550    fn controls(&self) -> GoalControls {
551        GoalControls::EMPTY
552    }
553
554    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
555        let Some(world) = mob.level() else {
556            return false;
557        };
558
559        let targeting = PufferfishEntity::targeting_conditions();
560        let search_box = mob.bounding_box().inflate(PUFF_SEARCH_RANGE);
561        !world
562            .get_entities_in_aabb_matching(&search_box, |entity| {
563                entity
564                    .as_living_entity()
565                    .is_some_and(|target| targeting.test(world.as_ref(), Some(mob), target))
566            })
567            .is_empty()
568    }
569
570    fn start(&mut self, mob: &dyn PathfinderMob) {
571        let Some(pufferfish) = mob.downcast_ref::<PufferfishEntity>() else {
572            return;
573        };
574        *pufferfish.inflate_counter.lock() = 1;
575        *pufferfish.deflate_timer.lock() = 0;
576    }
577
578    fn stop(&mut self, mob: &dyn PathfinderMob) {
579        if let Some(pufferfish) = mob.downcast_ref::<PufferfishEntity>() {
580            *pufferfish.inflate_counter.lock() = 0;
581        }
582    }
583}
