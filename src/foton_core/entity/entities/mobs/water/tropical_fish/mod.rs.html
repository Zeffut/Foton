44/// How big a tropical fish's body is.
45///
46/// Vanilla parity: `TropicalFish.Base`, the low byte of a pattern's packed id.
47#[derive(Debug, Clone, Copy, PartialEq, Eq)]
48pub enum TropicalFishBase {
49    /// Vanilla `SMALL`.
50    Small,
51    /// Vanilla `LARGE`.
52    Large,
53}
54
55impl TropicalFishBase {
56    const fn id(self) -> i32 {
57        match self {
58            Self::Small => 0,
59            Self::Large => 1,
60        }
61    }
62}

64/// One of the twelve tropical fish shapes.
65///
66/// Vanilla parity: `TropicalFish.Pattern`. The packed id is
67/// `base.id | index << 8`, so the six small shapes are `0x0000..=0x0500` and
68/// the six large ones are `0x0001..=0x0501`.
69#[derive(Debug, Clone, Copy, PartialEq, Eq)]
70pub enum TropicalFishPattern {
71    /// Vanilla `KOB`.
72    Kob,
73    /// Vanilla `SUNSTREAK`.
74    Sunstreak,
75    /// Vanilla `SNOOPER`.
76    Snooper,
77    /// Vanilla `DASHER`.
78    Dasher,
79    /// Vanilla `BRINELY`.
80    Brinely,
81    /// Vanilla `SPOTTY`.
82    Spotty,
83    /// Vanilla `FLOPPER`.
84    Flopper,
85    /// Vanilla `STRIPEY`.
86    Stripey,
87    /// Vanilla `GLITTER`.
88    Glitter,
89    /// Vanilla `BLOCKFISH`.
90    Blockfish,
91    /// Vanilla `BETTY`.
92    Betty,
93    /// Vanilla `CLAYFISH`.
94    Clayfish,
95}
96
97impl TropicalFishPattern {
98    /// Vanilla parity: `TropicalFish.Pattern.values()`, in declaration order.
99    pub const VALUES: [Self; 12] = [
100        Self::Kob,
101        Self::Sunstreak,
102        Self::Snooper,
103        Self::Dasher,
104        Self::Brinely,
105        Self::Spotty,
106        Self::Flopper,
107        Self::Stripey,
108        Self::Glitter,
109        Self::Blockfish,
110        Self::Betty,
111        Self::Clayfish,
112    ];
113
114    /// Returns the body size this pattern is drawn on.
115    #[must_use]
116    pub const fn base(self) -> TropicalFishBase {
117        match self {
118            Self::Kob
119            | Self::Sunstreak
120            | Self::Snooper
121            | Self::Dasher
122            | Self::Brinely
123            | Self::Spotty => TropicalFishBase::Small,
124            Self::Flopper
125            | Self::Stripey
126            | Self::Glitter
127            | Self::Blockfish
128            | Self::Betty
129            | Self::Clayfish => TropicalFishBase::Large,
130        }
131    }
132
133    /// Returns this pattern's index within its body size.
134    const fn index(self) -> i32 {
135        match self {
136            Self::Kob | Self::Flopper => 0,
137            Self::Sunstreak | Self::Stripey => 1,
138            Self::Snooper | Self::Glitter => 2,
139            Self::Dasher | Self::Blockfish => 3,
140            Self::Brinely | Self::Betty => 4,
141            Self::Spotty | Self::Clayfish => 5,
142        }
143    }
144
145    /// Vanilla parity: `TropicalFish.Pattern.getPackedId`.
146    #[must_use]
147    pub const fn packed_id(self) -> i32 {
148        self.base().id() | (self.index() << 8)
149    }
150
151    /// Vanilla parity: `TropicalFish.Pattern.byId`, which falls back to `KOB`.
152    #[must_use]
153    pub fn by_id(packed_id: i32) -> Self {
154        Self::VALUES
155            .into_iter()
156            .find(|pattern| pattern.packed_id() == packed_id)
157            .unwrap_or(Self::Kob)
158    }
159
160    /// Returns the name this pattern is saved and translated under.
161    #[must_use]
162    pub const fn serialized_name(self) -> &'static str {
163        match self {
164            Self::Kob => "kob",
165            Self::Sunstreak => "sunstreak",
166            Self::Snooper => "snooper",
167            Self::Dasher => "dasher",
168            Self::Brinely => "brinely",
169            Self::Spotty => "spotty",
170            Self::Flopper => "flopper",
171            Self::Stripey => "stripey",
172            Self::Glitter => "glitter",
173            Self::Blockfish => "blockfish",
174            Self::Betty => "betty",
175            Self::Clayfish => "clayfish",
176        }
177    }
178}

180/// A whole tropical fish appearance.
181///
182/// Vanilla parity: the `TropicalFish.Variant` record, which is stored and sent
183/// as the single packed `int` [`TropicalFishVariant::packed_id`] returns.
184#[derive(Debug, Clone, Copy, PartialEq, Eq)]
185pub struct TropicalFishVariant {
186    pattern: TropicalFishPattern,
187    base_color: DyeColor,
188    pattern_color: DyeColor,
189}
190
191impl TropicalFishVariant {
192    /// Vanilla `TropicalFish.DEFAULT_VARIANT`.
193    pub const DEFAULT: Self = Self::new(TropicalFishPattern::Kob, DyeColor::White, DyeColor::White);
194
195    /// Creates a variant from its three parts.
196    #[must_use]
197    pub const fn new(
198        pattern: TropicalFishPattern,
199        base_color: DyeColor,
200        pattern_color: DyeColor,
201    ) -> Self {
202        Self {
203            pattern,
204            base_color,
205            pattern_color,
206        }
207    }
208
209    /// Vanilla parity: `TropicalFish.packVariant`.
210    ///
211    /// The pattern keeps the low sixteen bits, the base color the third byte
212    /// and the pattern color the fourth.
213    #[must_use]
214    pub const fn packed_id(self) -> i32 {
215        (self.pattern.packed_id() & 0xFFFF)
216            | ((self.base_color.id() & 0xFF) << 16)
217            | ((self.pattern_color.id() & 0xFF) << 24)
218    }
219
220    /// Vanilla parity: the `Variant(int)` constructor.
221    #[must_use]
222    pub fn from_packed_id(packed_id: i32) -> Self {
223        Self {
224            pattern: TropicalFishPattern::by_id(packed_id & 0xFFFF),
225            base_color: DyeColor::by_id((packed_id >> 16) & 0xFF),
226            pattern_color: DyeColor::by_id((packed_id >> 24) & 0xFF),
227        }
228    }
229
230    /// Returns the shape.
231    #[must_use]
232    pub const fn pattern(self) -> TropicalFishPattern {
233        self.pattern
234    }
235
236    /// Returns the body color.
237    #[must_use]
238    pub const fn base_color(self) -> DyeColor {
239        self.base_color
240    }
241
242    /// Returns the color the pattern is drawn in.
243    #[must_use]
244    pub const fn pattern_color(self) -> DyeColor {
245        self.pattern_color
246    }
247}

382    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
383        let living_base = LivingEntityBase::new(entity_type);
384        let mob_base = MobBase::new();
385        // Vanilla parity: `WaterAnimal` clears the water malus.
386        mob_base
387            .pathfinding_malus()
388            .lock()
389            .set(PathType::Water, 0.0);
390        let mut entity_data = TropicalFishEntityData::new();
391        living_base.initialize_synced_data(&mut entity_data);
392
393        {
394            // Vanilla parity: `AbstractFish.registerGoals`.
395            let mut goals = mob_base.goal_selector().lock();
396            goals.add_goal(0, PanicGoal::new(fish::PANIC_SPEED_MODIFIER));
397            goals.add_goal(
398                2,
399                AvoidEntityGoal::with_selector(
400                    fish::AVOID_PLAYER_RANGE,
401                    fish::AVOID_WALK_SPEED,
402                    fish::AVOID_SPRINT_SPEED,
403                    |_, target, _| fish::is_player_to_flee(target),
404                ),
405            );
406            goals.add_goal(
407                4,
408                RandomSwimmingGoal::new(fish::SWIM_SPEED_MODIFIER, fish::SWIM_INTERVAL_TICKS),
409            );
410        }
411
412        Self {
413            base,
414            entity_type,
415            living_base,
416            mob_base,
417            entity_data: SyncMutex::new(entity_data),
418            is_school: AtomicBool::new(true),
419        }
420    }

485    /// Vanilla parity: `TropicalFish.checkTropicalFishSpawnRules`. Warm oceans
486    /// let a fish spawn at any depth; anywhere else it has to be near the top.
487    #[must_use]
488    pub fn check_tropical_fish_spawn_rules(world: &Arc<World>, pos: BlockPos) -> bool {
489        if !world
490            .get_block_state(pos.below())
491            .get_fluid_state()
492            .is_water()
493            || world.get_block_state(pos.above()).get_block() != &vanilla_blocks::WATER
494        {
495            return false;
496        }
497
498        world.biome_at(pos).is_some_and(|biome| {
499            biome.has_tag(&BiomeTag::ALLOWS_TROPICAL_FISH_SPAWNS_AT_ANY_HEIGHT)
500        }) || check_surface_water_animal_spawn_rules(world, pos)
501    }

645    /// Vanilla parity: `TropicalFish.finalizeSpawn`. Nine spawns in ten join a
646    /// shoal that shares one of the named combinations; the tenth is a
647    /// one-off, rolled freely and excluded from schooling.
648    fn finalize_spawn(
649        &self,
650        world: &Arc<World>,
651        spawn_reason: EntitySpawnReason,
652        group_data: Option<SpawnGroupData>,
653    ) -> Option<SpawnGroupData> {
654        let (variant, group_data) = match group_data {
655            Some(SpawnGroupData::TropicalFish(shoal)) => {
656                (shoal.variant(), SpawnGroupData::TropicalFish(shoal))
657            }
658            _ if rand::random::<f32>() < COMMON_VARIANT_CHANCE => {
659                let variant = COMMON_VARIANTS[rand::random_range(0..COMMON_VARIANTS.len())];
660                (
661                    variant,
662                    SpawnGroupData::TropicalFish(TropicalFishGroupData::new(variant)),
663                )
664            }
665            _ => {
666                self.is_school.store(false, Ordering::Relaxed);
667                let variant = TropicalFishVariant::new(
668                    TropicalFishPattern::VALUES
669                        [rand::random_range(0..TropicalFishPattern::VALUES.len())],
670                    DyeColor::VALUES[rand::random_range(0..DyeColor::VALUES.len())],
671                    DyeColor::VALUES[rand::random_range(0..DyeColor::VALUES.len())],
672                );
673                (
674                    variant,
