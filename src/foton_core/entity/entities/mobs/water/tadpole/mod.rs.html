1//! Tadpole entity.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.frog.Tadpole`. A tadpole
4//! is a fish with a clock: it swims, follows a player holding a slime ball, and
5//! after twenty minutes turns into a frog. It is what a `FrogspawnBlock` hatches
6//! into, which is the loop this closes.

45/// How long a tadpole takes to become a frog.
46///
47/// Vanilla parity: `Tadpole.ticksToBeFrog`, which is `Math.abs(-24000)`.
48pub const TICKS_TO_BE_FROG: i32 = 24_000;
49
50/// How far a fed tadpole's age jumps per second of feeding.
51///
52/// Vanilla parity: the `* 20` of `Tadpole.ageUp`.
53const TICKS_PER_FED_SECOND: i32 = 20;
54
55/// Vanilla parity: the `0.15F` volume of `Tadpole.ageUp`'s grow-up sound.
56const GROW_UP_SOUND_VOLUME: f32 = 0.15;
57
58/// Vanilla parity: `FrogspawnBlock.MIN_TADPOLES_SPAWN`.
59pub const MIN_TADPOLES_SPAWN: i32 = 2;
60/// Vanilla parity: `FrogspawnBlock.MAX_TADPOLES_SPAWN`, exclusive in the
61/// `random.nextInt(2, 6)` the block rolls.
62pub const MAX_TADPOLES_SPAWN_EXCLUSIVE: i32 = 6;

64/// A tadpole.
65#[entity_behavior(class = "Tadpole")]
66pub struct TadpoleEntity {
67    base: EntityBase,
68    entity_type: EntityTypeRef,
69    living_base: LivingEntityBase,
70    mob_base: MobBase,
71    brain: Brain,
72    /// Vanilla `Tadpole.age`, a plain field rather than the `AgeableMob` clock:
73    /// a tadpole is a fish and has no baby form.
74    age: SyncMutex<i32>,
75    entity_data: SyncMutex<TadpoleEntityData>,
76}

83impl TadpoleEntity {
84    /// Creates a tadpole at runtime.
85    #[must_use]
86    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
87        Self::new_with_base(
88            EntityBase::new(id, position, entity_type.dimensions, world),
89            entity_type,
90        )
91    }
92
93    /// Creates a tadpole from saved base data.
94    #[must_use]
95    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
96        Self::new_with_base(
97            EntityBase::from_load(load, entity_type.dimensions),
98            entity_type,
99        )
100    }
101
102    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
103        let living_base = LivingEntityBase::new(entity_type);
104        let mob_base = MobBase::new();
105        // Vanilla parity: `WaterAnimal` clears the water malus, so open water is
106        // free to path through rather than merely allowed.
107        mob_base
108            .pathfinding_malus()
109            .lock()
110            .set(PathType::Water, 0.0);
111        let mut entity_data = TadpoleEntityData::new();
112        living_base.initialize_synced_data(&mut entity_data);
113
114        Self {
115            base,
116            entity_type,
117            living_base,
118            mob_base,
119            brain: tadpole_ai::make_brain(),
120            age: SyncMutex::new(0),
121            entity_data: SyncMutex::new(entity_data),
122        }
123    }
124
125    /// Returns vanilla `Tadpole.getAge`.
126    #[must_use]
127    pub fn age(&self) -> i32 {
128        *self.age.lock()
129    }
130
131    /// Returns vanilla `Tadpole.isAgeLocked`.
132    #[must_use]
133    pub fn is_age_locked(&self) -> bool {
134        *self.entity_data.lock().age_locked.get()
135    }
136
137    /// Sets vanilla `Tadpole.setAgeLocked`.
138    pub fn set_age_locked(&self, locked: bool) {
139        self.entity_data.lock().age_locked.set(locked);
140    }
141
142    /// Sets vanilla `Tadpole.setAge`, which is also where a tadpole notices it
143    /// has run out of time and turns into a frog.
144    pub fn set_age(&self, age: i32) {
145        *self.age.lock() = age;
146        if age >= TICKS_TO_BE_FROG {
147            self.grow_up();
148        }
149    }
150
151    /// Returns vanilla `Tadpole.getTicksLeftUntilAdult`.
152    #[must_use]
153    pub fn ticks_left_until_adult(&self) -> i32 {
154        (TICKS_TO_BE_FROG - self.age()).max(0)
155    }
156
157    /// Vanilla parity: `Tadpole.ageUp`, the conversion into a frog.
158    ///
159    /// This is the far end of the frogspawn loop: the block hatches tadpoles and
160    /// twenty minutes later each of them is a frog that can lay spawn of its own.
161    fn grow_up(&self) {
162        let Some(world) = self.level() else {
163            return;
164        };
165
166        let converted = convert_to(
167            self,
168            ConversionParams::single(false, false),
169            |id, position, level| FrogEntity::new(&vanilla_entities::FROG, id, position, level),
170            |frog| {
171                frog.finalize_spawn(&world, EntitySpawnReason::Conversion, None);
172                frog.set_persistence_required();
173            },
174        );
175
176        if converted.is_some() {
177            self.play_sound(
178                &sound_events::ENTITY_TADPOLE_GROW_UP,
179                GROW_UP_SOUND_VOLUME,
180                1.0,
181            );
182        }
183    }
184
185    /// Returns whether the stack is vanilla frog food.
186    ///
187    /// Vanilla parity: `Tadpole.isFood`, which reads `#minecraft:frog_food`
188    /// rather than an `Animal.isFood` a fish does not have.
189    #[must_use]
190    pub fn is_food(item_stack: &ItemStack) -> bool {
191        REGISTRY
192            .items
193            .is_in_tag(item_stack.item(), &ItemTag::FROG_FOOD)
194    }
195
196    /// Vanilla parity: `Tadpole.feed`.
197    ///
198    /// The happy-villager puff vanilla adds here is a client-local
199    /// `Level.addParticle`, so there is no server-side work for it.
200    fn feed(&self, player: &Player, hand: InteractionHand) {
201        Mob::use_player_item(self, player, hand);
202        let speed_up =
203            AgeableMobBase::get_speed_up_seconds_when_feeding(self.ticks_left_until_adult());
204        self.set_age(self.age() + speed_up * TICKS_PER_FED_SECOND);
205    }
206}

208impl Entity for TadpoleEntity {
209    fn base(&self) -> &EntityBase {
210        &self.base
211    }
212
213    fn entity_type(&self) -> EntityTypeRef {
214        self.entity_type
215    }
216
217    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
218        Some(&self.entity_data)
219    }
220
221    /// Vanilla parity: `WaterAnimal.baseTick`, which reads the air left before
222    /// the shared tick spends it.
223    fn base_tick(&self) {
224        let air_before_tick = self.air_supply();
225        Mob::base_tick_mob(self);
226        if let Some(world) = self.level() {
227            fish::handle_air_supply(self, &world, air_before_tick);
228        }
229    }
230
231    fn sound_source(&self) -> SoundSource {
232        SoundSource::Neutral
233    }
234
235    /// Vanilla parity: `AbstractFish.playStepSound` is empty; a fish has no feet.
236    fn play_step_sound(&self, _pos: BlockPos, _block_state: BlockStateId) {}
237
238    fn save_additional(&self, nbt: &mut NbtCompound) {
239        self.save_mob(nbt);
240        nbt.insert("Age", self.age());
241        nbt.insert("AgeLocked", self.is_age_locked());
242        self.brain.save(nbt);
243    }
244
245    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
246        self.load_mob(nbt);
247        // The age is assigned directly rather than through `set_age`: a tadpole
248        // loaded at or past its deadline must not convert while the chunk is
249        // still being read in. Its next `ai_step` grows it up instead.
250        *self.age.lock() = nbt.int("Age").unwrap_or(0);
251        self.set_age_locked(nbt.byte("AgeLocked").is_some_and(|flag| flag != 0));
252        self.brain.load(nbt);
253    }
254}

389    /// Vanilla parity: `Tadpole.mobInteract`. Feeding hurries it toward being a
390    /// frog.
391    ///
392    /// MISSING FOUNDATION: vanilla then falls through to
393    /// `Bucketable.bucketMobPickup` and to the golden-dandelion age lock. Foton's
394    /// `MobBucketItem` does not carry a tadpole yet, so a water bucket used on
395    /// one does nothing rather than picking it up.
396    fn mob_interact(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
397        let item_stack = {
398            let inventory = player.inventory.lock();
399            let item_stack = inventory.get_item_in_hand(hand);
400            item_stack.copy_with_count(item_stack.count())
401        };
402
403        if Self::is_food(&item_stack) && !self.is_age_locked() {
404            self.feed(player, hand);
405            return InteractionResult::Success;
406        }
407
408        InteractionResult::Pass
409    }

422/// Spawns the tadpoles a hatching frogspawn block releases.
423///
424/// Vanilla parity: `FrogspawnBlock.spawnTadpoles`, which lives here rather than
425/// in the block so the entity owns its own construction.
426pub fn spawn_tadpoles_from_frogspawn(world: &Arc<World>, pos: BlockPos) {
427    let count = rand::random_range(MIN_TADPOLES_SPAWN..MAX_TADPOLES_SPAWN_EXCLUSIVE);
428
429    for _ in 0..count {
430        let x = f64::from(pos.x()) + random_tadpole_position_offset();
431        let z = f64::from(pos.z()) + random_tadpole_position_offset();
432        let y = f64::from(pos.y()) - 0.5;
433
434        let Some(entity) = ENTITIES.create(
435            &vanilla_entities::TADPOLE,
436            next_entity_id(),
437            DVec3::new(x, y, z),
438            Arc::downgrade(world),
439        ) else {
440            return;
441        };
442
443        // Vanilla parity: the `random.nextInt(1, 361)` degrees of yaw.
444        entity.set_rotation((rand::random_range(1..361) as f32, 0.0));
445        entity.set_old_position_to_current();
446        if let Some(mob) = entity.as_mob() {
447            mob.set_persistence_required();
448        }
449        let _added = world.try_add_entity(entity);
450    }
451}

453/// Vanilla parity: `FrogspawnBlock.getRandomTadpolePositionOffset`, which keeps
454/// a hatchling's hitbox inside the block it came out of.
455fn random_tadpole_position_offset() -> f64 {
456    rand::random::<f64>().clamp(0.2, 0.799_999_997_019_767_8)
457}
