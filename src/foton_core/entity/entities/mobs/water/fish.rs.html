111/// Drains air out of water and refills it in.
112///
113/// Vanilla parity: `WaterAnimal.handleAirSupply`, the mirror image of a land
114/// mob drowning.
115pub(super) fn handle_air_supply<M: LivingEntity + ?Sized>(
116    fish: &M,
117    world: &World,
118    air_before_tick: i32,
119) {
120    if Entity::is_alive(fish) && !fish.is_in_water() {
121        fish.set_air_supply(air_before_tick - 1);
122        if fish.should_take_drowning_damage() {
123            fish.set_air_supply(0);
124            fish.hurt_server(
125                world,
126                &DamageSource::environment(&vanilla_damage_types::DROWN),
127                SUFFOCATION_DAMAGE,
128            );
129        }
130    } else {
131        fish.set_air_supply(FISH_AIR_SUPPLY);
132    }
133}

135/// Throws the fish about when it is stranded on land.
136///
137/// Vanilla parity: the flop branch of `AbstractFish.aiStep`.
138pub(super) fn flop<M: LivingEntity + ?Sized>(fish: &M, flop_sound: SoundEventRef) {
139    if fish.is_in_water() || !fish.on_ground() {
140        return;
141    }
142
143    let scatter = || (rand::random::<f64>() * 2.0 - 1.0) * FLOP_SCATTER;
144    fish.set_velocity(fish.velocity() + DVec3::new(scatter(), FLOP_LIFT, scatter()));
145    fish.set_on_ground(false);
146    fish.play_sound(flop_sound, 1.0, 1.0);
147}

149/// Swims instead of wading.
150///
151/// Vanilla parity: `AbstractFish.travelInWater`, which replaces the shared
152/// water physics outright: a fish pushes itself along, keeps nine tenths of its
153/// speed, and drifts down when it has nothing to chase.
154pub(super) fn travel_in_water<M: Mob + ?Sized>(fish: &M, input: DVec3) -> Option<MoveResult> {
155    fish.move_relative(SWIM_ACCELERATION, input);
156    let result = fish.move_entity(MoverType::SelfMovement, fish.velocity());
157    fish.set_velocity(fish.velocity() * SWIM_DRAG);
158    if fish.target().is_none() {
159        fish.set_velocity(fish.velocity() + DVec3::new(0.0, IDLE_SINK, 0.0));
160    }
161    result
162}

164/// Steers in three dimensions rather than along the ground.
165///
166/// Vanilla parity: `AbstractFish.FishMoveControl.tick`.
167pub(super) fn tick_move_control<M: Mob + ?Sized>(fish: &M) {
168    if fish.is_eye_in_water() {
169        fish.set_velocity(fish.velocity() + DVec3::new(0.0, SUBMERGED_LIFT, 0.0));
170    }
171
172    let (operation, wanted_position, speed_modifier) = {
173        let controls = fish.mob_base().controls().lock();
174        let move_control = controls.move_control;
175        (
176            move_control.operation(),
177            move_control.wanted_position(),
178            move_control.speed_modifier(),
179        )
180    };
181
182    let navigating = matches!(operation, MoveControlOperation::MoveTo)
183        && !fish.mob_base().navigation().lock().is_done();
184    if !navigating {
185        fish.set_mob_speed(0.0);
186        return;
187    }
188
189    let movement_speed = fish
190        .attributes()
191        .lock()
192        .required_value(vanilla_attributes::MOVEMENT_SPEED);
193    let target_speed = (speed_modifier * movement_speed) as f32;
194    let current_speed = fish.get_speed();
195    let speed = SPEED_LERP.mul_add(target_speed - current_speed, current_speed);
196    fish.set_mob_speed(speed);
197
198    let delta = wanted_position - fish.position();
199    if delta.y != 0.0 {
200        let distance = delta.length();
201        if distance > 0.0 {
202            let lift = f64::from(speed) * (delta.y / distance) * VERTICAL_STEER;
203            fish.set_velocity(fish.velocity() + DVec3::new(0.0, lift, 0.0));
204        }
205    }
206
207    if delta.x != 0.0 || delta.z != 0.0 {
208        let wanted_yaw = (delta.z.atan2(delta.x).to_degrees() as f32) - 90.0;
209        let (yaw, pitch) = fish.rotation();
210        let turned = rotlerp(yaw, wanted_yaw, TURN_RATE);
211        fish.set_rotation((turned, pitch));
212        fish.set_y_body_rot(turned);
213    }
214}

216/// Returns whether a fish should flee this entity.
217///
218/// Vanilla parity: the `Player.class` filter of the fish `AvoidEntityGoal`,
219/// which is why a fish darts from a swimmer but ignores a squid.
220pub(super) fn is_player_to_flee(target: &dyn LivingEntity) -> bool {
221    target
222        .as_player()
223        .is_some_and(|player| !target.is_spectator() && !player.has_infinite_materials())
