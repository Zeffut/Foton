1//! The villager's brain: the activity packages that make up its day.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.ai.behavior.VillagerGoalPackages`
4//! plus the `Brain.Provider` in `Villager`.
5//!
6//! # What drives it
7//!
8//! The `villager_schedule` timeline carries two activity-valued tracks, and
9//! [`Brain::update_activity_from_schedule`] reads whichever one the villager's
10//! [`ScheduleAttribute`] names. An adult steps IDLE -> WORK -> MEET -> IDLE ->
11//! REST across the day; a baby steps IDLE -> PLAY -> IDLE -> PLAY -> REST.
12//! Every package but PANIC ends with `UpdateActivityFromSchedule` at priority
13//! 99, so the lowest-priority thing a villager does is notice the hour.
14//!
15//! # Two deliberate differences from vanilla
16//!
17//! **No `refreshBrain`.** Vanilla rebuilds a villager's whole brain whenever its
18//! profession changes or it grows up, because the job-site predicates, the
19//! composter special case and the baby/adult package split are all baked in at
20//! construction. Foton instead: hands the tick's context to the job-site
21//! predicates (see [`AcquirePoi`]); lets one [`WorkAtPoi`] look at the block it
22//! is standing at; and registers both WORK and PLAY on every villager, which is
23//! safe because the schedule attribute never names PLAY for an adult and WORK
24//! needs a `JOB_SITE` a baby cannot hold. Growing up therefore only has to swap
25//! the schedule attribute, which `age_boundary_changed` does.
26//!
27//! **The farming weights are the farmer's.** Vanilla weights `HarvestFarmland`
28//! and `UseBonemeal` differently for a farmer and for everybody else, because
29//! the weight is fixed when the brain is built. Both refuse to start for a
30//! non-farmer -- `HarvestFarmland` tests the profession, and only the farmer
31//! asks for bone meal -- and a `RunOne` entry that never starts does not change
32//! the relative order of the ones that do, so the farmer's weights are used
33//! throughout.
34//!
35//! [`Brain::update_activity_from_schedule`]: crate::entity::ai::brain::Brain::update_activity_from_schedule
36//! [`ScheduleAttribute`]: crate::entity::ai::brain::ScheduleAttribute

180/// Reaches for the villager behind a brain context.
181///
182/// Every behavior in this module is registered on a villager's own brain, so
183/// the downcast only fails if one is put on some other mob -- in which case
184/// doing nothing is the right answer.
185fn villager<'a>(ctx: &'a BrainContext<'a>) -> Option<&'a VillagerEntity> {
186    ctx.mob().downcast_ref::<VillagerEntity>()
187}

189/// Whether this profession holds no workstation at all.
190///
191/// Vanilla parity: the `PoiType.NONE` predicate `VillagerProfession.NONE` and
192/// `NITWIT` are registered with as their `heldJobSite`.
193fn is_jobless(profession_path: &str) -> bool {
194    profession_path == "none" || profession_path == "nitwit"
195}

197/// Vanilla parity: `VillagerProfession.heldJobSite`, which every named
198/// profession registers as `poiType -> poiType.is(jobSite)` -- and the POI and
199/// the profession share a registry key.
200fn held_job_site(ctx: &BrainContext<'_>, poi_type: PoiTypeRef) -> bool {
201    let Some(villager) = villager(ctx) else {
202        return false;
203    };
204    let profession = villager.profession();
205    !is_jobless(&profession.key.path) && poi_type.key == profession.key
206}
207
208/// Vanilla parity: `VillagerProfession.acquirableJobSite`, which is the held one
209/// for a named profession, `ALL_ACQUIRABLE_JOBS` for `NONE`, and nothing for
210/// `NITWIT`.
211fn acquirable_job_site(ctx: &BrainContext<'_>, poi_type: PoiTypeRef) -> bool {
212    let Some(villager) = villager(ctx) else {
213        return false;
214    };
215    let profession = villager.profession();
216    match profession.key.path.as_ref() {
217        "nitwit" => false,
218        "none" => REGISTRY
219            .poi_types
220            .is_in_tag(poi_type, &PoiTag::ACQUIRABLE_JOB_SITE),
221        _ => poi_type.key == profession.key,
222    }
223}

241/// Builds a villager's brain.
242///
243/// Vanilla parity: `Villager.BRAIN_PROVIDER` plus `registerBrainGoals`. The
244/// schedule attribute is set by the caller, which knows whether this villager
245/// is a baby.
246#[must_use]
247pub fn make_brain() -> Brain {
248    Brain::new_with_memories(
249        SENSORS,
250        // Nothing writes these until the villager itself does, so they have to
251        // be registered up front or the reads would miss.
252        &[
253            memory_module_types::LAST_SLEPT.id(),
254            memory_module_types::LAST_WOKEN.id(),
255            memory_module_types::LAST_WORKED_AT_POI.id(),
256            memory_module_types::HOME.id(),
257            memory_module_types::JOB_SITE.id(),
258            memory_module_types::POTENTIAL_JOB_SITE.id(),
259            memory_module_types::MEETING_POINT.id(),
260        ],
261        vec![
262            core_package(),
263            work_package(),
264            rest_package(),
265            meet_package(),
266            idle_package(),
267            play_package(),
268            panic_package(),
269            pre_raid_package(),
270            raid_package(),
271            hide_package(),
272        ],
273    )
274}

881/// Whether the raid over this villager has been won.
882///
883/// Vanilla parity: `VillagerGoalPackages.raidExistsAndNotVictory`, whose name
884/// says the opposite of what it does -- the body is `currentRaid != null &&

889        .get_raid_at(mob.block_position())
890        .is_some_and(|raid| raid.is_victory())
891}
