1//! What a frightened villager does.
2//!
3//! Vanilla parity: `VillagerPanicTrigger` and `VillagerCalmDown`, the pair that
4//! switch a villager into and back out of [`Activity::Panic`].
5
6use super::villager;
7use crate::entity::ai::brain::Activity;
8use crate::entity::ai::brain::behavior::{
9    BrainContext, MemoryModuleId, MemoryStatus, TimedBehavior, Trigger,
10};
11use crate::entity::ai::brain::memory::memory_module_types;
12
13/// Vanilla parity: `VillagerCalmDown.SAFE_DISTANCE_FROM_DANGER`, squared.
14const SAFE_DISTANCE_FROM_DANGER_SQR: f64 = 36.0;
15
16/// Vanilla parity: the `timestamp % 100L` of `VillagerPanicTrigger.tick`.
17const GOLEM_CHECK_INTERVAL: i64 = 100;
18
19/// Vanilla parity: the `3` of that same call -- a frightened village settles
20/// for less agreement than a chatting one.
21const VILLAGERS_NEEDED_TO_AGREE_WHEN_PANICKING: i32 = 3;
22
23/// Switches a villager to PANIC while something is hurting or hunting it.
24///
25/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.VillagerPanicTrigger`,
26/// which sits at priority zero of the core package so it overrides whatever the
27/// schedule had the villager doing.
28///
29/// Its tick is also the urgent half of raising an iron golem: a village under
30/// attack needs only three villagers to agree rather than the five that
31/// gossiping asks for, and it asks every five seconds for as long as the fright
32/// lasts.
33pub struct VillagerPanicTrigger;
34
35impl VillagerPanicTrigger {
36    /// Vanilla parity: `VillagerPanicTrigger.hasHostile`.
37    #[must_use]
38    pub fn has_hostile(ctx: &BrainContext<'_>) -> bool {
39        ctx.brain()
40            .has_memory_value(memory_module_types::NEAREST_HOSTILE.id())
41    }
42
43    /// Vanilla parity: `VillagerPanicTrigger.isHurt`.
44    #[must_use]
45    pub fn is_hurt(ctx: &BrainContext<'_>) -> bool {
46        ctx.brain()
47            .has_memory_value(memory_module_types::HURT_BY.id())
48    }
49}
50
51impl TimedBehavior for VillagerPanicTrigger {
52    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
53        &[]
54    }
55
56    fn can_still_use(&mut self, ctx: &BrainContext<'_>) -> bool {
57        Self::is_hurt(ctx) || Self::has_hostile(ctx)
58    }
59
60    /// Vanilla parity: the `timestamp % 100L == 0L` of
61    /// `VillagerPanicTrigger.tick`.
62    fn tick(&mut self, ctx: &BrainContext<'_>) {
63        if ctx.game_time() % GOLEM_CHECK_INTERVAL != 0 {
64            return;
65        }
66        if let Some(villager) = villager(ctx) {
67            villager.spawn_golem_if_needed(
68                ctx.world(),
69                ctx.game_time(),
70                VILLAGERS_NEEDED_TO_AGREE_WHEN_PANICKING,
71            );
72        }
73    }
74
75    fn start(&mut self, ctx: &BrainContext<'_>) {
76        if !Self::is_hurt(ctx) && !Self::has_hostile(ctx) {
77            return;
78        }
79        let brain = ctx.brain();
80        if !brain.is_active(Activity::Panic) {
81            // Vanilla drops everything it was walking toward or talking to, so
82            // the panic does not inherit a stale destination.
83            brain.erase_memory(memory_module_types::PATH.id());
84            brain.erase_memory(memory_module_types::WALK_TARGET.id());
85            brain.erase_memory(memory_module_types::LOOK_TARGET.id());
86            brain.erase_memory(memory_module_types::BREED_TARGET.id());
87            brain.erase_memory(memory_module_types::INTERACTION_TARGET.id());
88        }
89        brain.set_active_activity_if_possible(Activity::Panic);
90    }
91
92    fn debug_name(&self) -> &'static str {
93        "VillagerPanicTrigger"
94    }
95}
96
97/// Ends the panic once nothing dangerous is left in sight.
98///
99/// Vanilla parity: `net.minecraft.world.entity.ai.behavior.VillagerCalmDown`.
100/// It is what puts a villager back on its schedule, which is why it re-reads it
101/// on the spot rather than waiting for the PANIC package's own update.
102pub struct VillagerCalmDown;
103
104impl Trigger for VillagerCalmDown {
105    fn required_memories(&self) -> Vec<MemoryModuleId> {
106        vec![
107            memory_module_types::HURT_BY.id(),
108            memory_module_types::HURT_BY_ENTITY.id(),
109            memory_module_types::NEAREST_HOSTILE.id(),
110        ]
111    }
112
113    fn trigger(&mut self, ctx: &BrainContext<'_>) -> bool {
114        let brain = ctx.brain();
115        let hurt_by_entity_close = brain
116            .get_memory(memory_module_types::HURT_BY_ENTITY)
117            .and_then(|memory| memory.get())
118            .is_some_and(|entity| {
119                entity.position().distance_squared(ctx.mob().position())
120                    <= SAFE_DISTANCE_FROM_DANGER_SQR
121            });
122        let feel_scared = brain.has_memory_value(memory_module_types::HURT_BY.id())
123            || brain.has_memory_value(memory_module_types::NEAREST_HOSTILE.id())
124            || hurt_by_entity_close;
125        if !feel_scared {
126            brain.erase_memory(memory_module_types::HURT_BY.id());
127            brain.erase_memory(memory_module_types::HURT_BY_ENTITY.id());
128            brain.update_activity_from_schedule(ctx.world(), ctx.game_time());
129        }
130        true
131    }
132
133    fn debug_name(&self) -> &'static str {
134        "VillagerCalmDown"
135    }
136}
