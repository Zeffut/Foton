1//! The villager.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.npc.villager.Villager` and the
4//! half of `AbstractVillager` that is not the `Merchant` seam (which lives in
5//! [`super::merchant_state`]).
6//!
7//! What is here is the trading loop end to end -- a profession and a level, the
8//! trades that follow from them, the screen a player buys through, restocking,
9//! leveling up, the reputation that discounts a price -- and the seam onto the
10//! brain that runs the villager's day. The day itself is in
11//! [`super::villager_ai`].

242    fn new_with_base(
243        base: EntityBase,
244        entity_type: EntityTypeRef,
245        id: i32,
246        world: Weak<World>,
247    ) -> Self {
248        let living_base = LivingEntityBase::new(entity_type);
249        let mob_base = MobBase::new();
250        let ageable_base = AgeableMobBase::new();
251        let mut entity_data = VillagerEntityData::new();
252        living_base.initialize_synced_data(&mut entity_data);
253
254        {
255            // Vanilla parity: the three navigation lines of the `Villager`
256            // constructor. `setCanOpenDoors` is the one with teeth: without it
257            // a closed door is not pathfindable, so a villager would never plan
258            // a route through one and `InteractWithDoor` would have nothing to
259            // open.
260            let mut navigation = mob_base.navigation().lock();
261            navigation.set_can_open_doors(true);
262            navigation.set_can_float(true);
263            navigation
264                .set_required_path_length(REQUIRED_PATH_LENGTH, f64::from(REQUIRED_PATH_LENGTH));
265        }
266
267        let villager = Self {
268            base,
269            entity_type,
270            living_base,
271            mob_base,
272            ageable_base,
273            entity_data: SyncMutex::new(entity_data),
274            merchant: Arc::new(MerchantState::villager(id, world)),
275            inventory: SyncMutex::new(SimpleContainer::new(Self::INVENTORY_SIZE)),
276            gossips: SyncMutex::new(GossipContainer::new()),
277            brain: villager_ai::make_brain(),
278            food_level: SyncMutex::new(0),
279            last_gossip_time: SyncMutex::new(0),
280            last_gossip_decay_time: SyncMutex::new(0),
281            last_restock_game_time: SyncMutex::new(0),
282            number_of_restocks_today: SyncMutex::new(0),
283        };
284        villager.update_schedule();
285        // Vanilla parity: the `setCanPickUpLoot(true)` of the `Villager`
286        // constructor, which is what makes `Mob.aiStep` offer the ground to
287        // `wantsToPickUp` at all.
288        Mob::set_can_pick_up_loot(&villager, true);
289        villager
290    }

433    /// Rolls the trades this profession offers at this level and adds them.
434    ///
435    /// Vanilla parity: `Villager.updateTrades`, which looks up the trade set
436    /// for the current level and appends what it draws to the existing list --
437    /// so leveling up adds trades rather than replacing them.
438    pub fn update_trades(&self) {
439        let profession = self.profession();
440        let Some(trade_set) = TradeSet::for_profession(&profession.key, self.villager_level())
441        else {
442            return;
443        };
444
445        let mut rng = rand::rng();
446        let villager_variant = self.villager_type().key.clone();
447        let position = self.position();
448        let mut context = LootContext::new(&mut rng)
449            .with_origin(position.x, position.y, position.z)
450            .allowing_additional_cost_component()
451            .with_this_entity(EntityRef {
452                entity_type: Some(&self.entity_type.key),
453                villager_variant: Some(&villager_variant),
454                ..EntityRef::default()
455            });
456
457        let mut offers = self.merchant.offers().lock();
458        trade_set.add_offers(&mut context, &mut offers);
459    }

486    /// Records something that happened to this villager.
487    ///
488    /// Vanilla parity: `Villager.onReputationEventFrom`.
489    pub fn on_reputation_event_from(&self, event: ReputationEventType, source: Uuid) {
490        let mut gossips = self.gossips.lock();
491        match event {
492            ReputationEventType::ZombieVillagerCured => {
493                gossips.add(
494                    source,
495                    GossipType::MajorPositive,
496                    GossipType::REPUTATION_CHANGE_PER_EVERLASTING_MEMORY,
497                );
498                gossips.add(
499                    source,
500                    GossipType::MinorPositive,
501                    GossipType::REPUTATION_CHANGE_PER_EVENT,
502                );
503            }
504            ReputationEventType::Trade => {
505                gossips.add(
506                    source,
507                    GossipType::Trading,
508                    GossipType::REPUTATION_CHANGE_PER_TRADE,
509                );
510            }
511            ReputationEventType::VillagerHurt => {
512                gossips.add(
513                    source,
514                    GossipType::MinorNegative,
515                    GossipType::REPUTATION_CHANGE_PER_EVENT,
516                );
517            }
518            ReputationEventType::VillagerKilled => {
519                gossips.add(
520                    source,
521                    GossipType::MajorNegative,
522                    GossipType::REPUTATION_CHANGE_PER_EVENT,
523                );
524            }
525            // Vanilla's `Villager.onReputationEventFrom` has no arm for a golem
526            // being killed; only the iron golem's own handler reacts to that.
527            ReputationEventType::GolemKilled => {}
528        }
529    }

531    /// Swaps gossip with another villager standing nearby.
532    ///
533    /// Vanilla parity: `Villager.gossip`, gated on both villagers being off
534    /// their thousand-two-hundred-tick cooldown. Two villagers finding time to
535    /// talk is also what vanilla reads as the village being calm and populated
536    /// enough to want an iron golem, so the golem check hangs off the end of it.
537    pub fn gossip_with(&self, world: &Arc<World>, other: &Self, timestamp: i64) {
538        let mine = *self.last_gossip_time.lock();
539        let theirs = *other.last_gossip_time.lock();
540        let ready = |last: i64| timestamp < last || timestamp >= last + GOSSIP_COOLDOWN;
541        if !ready(mine) || !ready(theirs) {
542            return;
543        }
544
545        let source = other.gossips.lock().copy();
546        let mut rng = rand::rng();
547        self.gossips
548            .lock()
549            .transfer_from(&source, &mut rng, MAX_GOSSIP_TOPICS);
550        *self.last_gossip_time.lock() = timestamp;
551        *other.last_gossip_time.lock() = timestamp;
552        self.spawn_golem_if_needed(world, timestamp, VILLAGERS_NEEDED_TO_AGREE_WHEN_GOSSIPING);
553    }

555    /// Raises an iron golem if enough of the village agrees it wants one.
556    ///
557    /// Vanilla parity: `Villager.spawnGolemIfNeeded`. Nothing counts the golems
558    /// a village already has; what stops a second one is that every villager
559    /// who agreed to the first is told about it -- see
560    /// [`GolemSensor::golem_detected`] -- and spends the next thirty seconds
561    /// not wanting another. Every villager in range is told, not only the ones
562    /// that agreed, so a villager who was asleep through the vote does not
563    /// immediately call for a second golem.
564    ///
565    /// [`GolemSensor::golem_detected`]: crate::entity::ai::brain::sensor::GolemSensor::golem_detected
566    pub fn spawn_golem_if_needed(
567        &self,
568        world: &Arc<World>,
569        timestamp: i64,
570        villagers_needed_to_agree: i32,
571    ) {
572        if !self.wants_to_spawn_golem(timestamp) {
573            return;
574        }
575
576        let search_box = self.bounding_box().inflate(GOLEM_AGREEMENT_RANGE);
577        let nearby: Vec<SharedEntity> = world
578            .get_entities_in_aabb_matching(&search_box, |entity| {
579                entity.downcast_ref::<Self>().is_some()
580            });
581        let agreeing = nearby
582            .iter()
583            .filter_map(|entity| entity.downcast_ref::<Self>())
584            .filter(|villager| villager.wants_to_spawn_golem(timestamp))
585            .take(MAX_VILLAGERS_THAT_AGREE)
586            .count();
587        if i32::try_from(agreeing).unwrap_or(i32::MAX) < villagers_needed_to_agree {
588            return;
589        }
590
591        let spawned = try_spawn_mob(
592            &vanilla_entities::IRON_GOLEM,
593            EntitySpawnReason::MobSummoned,
594            world,
595            self.block_position(),
596            GOLEM_SPAWN_ATTEMPTS,
597            GOLEM_SPAWN_RANGE_XZ,
598            GOLEM_SPAWN_RANGE_Y,
599            SpawnStrategy::LegacyIronGolem,
600            false,
601        );
602        if spawned.is_none() {
603            return;
604        }
605        for entity in &nearby {
606            if let Some(brain) = entity.as_mob().and_then(Mob::brain) {
607                GolemSensor::golem_detected(brain);
608            }
609        }
610    }

645    /// Whether this villager may restock right now.
646    ///
647    /// Vanilla parity: `Villager.shouldRestock` and the `allowedToRestock` it
648    /// guards. Note that the day rollover is what resets the counter, and that
649    /// resetting it also catches the demand up on the restocks that were missed.
650    #[must_use]
651    pub fn should_restock(&self, game_time: i64) -> bool {
652        let is_new_day = {
653            let last = *self.last_restock_game_time.lock();
654            game_time > last + HALF_DAY_TICKS
655        };
656        if is_new_day {
657            *self.last_restock_game_time.lock() = game_time;
658            self.reset_number_of_restocks();
659        }
660        self.allowed_to_restock(game_time) && self.merchant.needs_to_restock()
661    }

663    fn allowed_to_restock(&self, game_time: i64) -> bool {
664        let restocks = *self.number_of_restocks_today.lock();
665        restocks == 0
666            || (restocks < MAX_RESTOCKS_PER_DAY
667                && game_time > *self.last_restock_game_time.lock() + RESTOCK_COOLDOWN_TICKS)
668    }

675        self.merchant.reset_uses();

680    /// Vanilla parity: `Villager.resetNumberOfRestocks`, which catches demand
681    /// up on whatever restocks the day did not use before clearing the count.
682    fn reset_number_of_restocks(&self) {
683        let missed = MAX_RESTOCKS_PER_DAY - *self.number_of_restocks_today.lock();
684        if missed > 0 {

686        }

691    }

765    /// Vanilla parity: `Villager.eatUntilFull`.
766    fn eat_until_full(&self) {
767        if *self.food_level.lock() >= BREEDING_FOOD_THRESHOLD {
768            return;
769        }
770        let mut inventory = self.inventory.lock();
771        for slot in 0..inventory.get_container_size() {
772            let value = food_points(inventory.get_item(slot));
773            if value == 0 {
774                continue;
775            }
776            while inventory.get_item(slot).count() > 0 {
777                *self.food_level.lock() += value;
778                inventory.remove_item(slot, 1);
779                if *self.food_level.lock() >= BREEDING_FOOD_THRESHOLD {
780                    return;
781                }
782            }
783        }
784    }

798    /// Bakes bread out of the wheat this villager is carrying.
799    ///
800    /// Vanilla parity: `WorkAtComposter.makeBread`, which is the only thing that
801    /// turns a farmer's harvest into food a village can eat and breed on. Bread
802    /// it cannot carry is dropped rather than lost.
803    pub fn make_bread(&self) {
804        let leftover = {
805            let mut inventory = self.inventory.lock();
806            if inventory.count_item(&vanilla_items::BREAD) > MAX_BREAD_BEFORE_BAKING {
807                return;
808            }
809            let wheat = inventory.count_item(&vanilla_items::WHEAT);
810            let loaves = MAX_BREAD_PER_BAKE.min(wheat / WHEAT_PER_BREAD);
811            if loaves == 0 {
812                return;
813            }
814            inventory.remove_item_type(&vanilla_items::WHEAT, loaves * WHEAT_PER_BREAD);
815            let mut bread = ItemStack::with_count(&vanilla_items::BREAD, loaves);
816            inventory.add(&mut bread);
817            bread
818        };
819        if !leftover.is_empty() {
820            self.spawn_at_location(leftover, BREAD_DROP_OFFSET);
821        }
822    }

893    /// Gives back the POI ticket a memory holds.
894    ///
895    /// Vanilla parity: `Villager.releasePoi`, which only releases when the block
896    /// is still a POI this villager's memory is allowed to hold -- so a job site
897    /// that has since become something else is left alone.
898    pub fn release_poi(&self, memory: MemoryModuleType<GlobalPos>) {
899        let Some(world) = Entity::level(self) else {
900            return;
901        };
902        let Some(held) = self.brain.get_memory(memory) else {
903            return;
904        };
905        if held.dimension != world.key {
906            return;
907        }
908        let mut storage = world.poi_storage.lock();
909        let Some(poi_type) = storage
910            .get_type(held.pos)
911            .and_then(|id| REGISTRY.poi_types.by_id(id))
912        else {
913            return;
914        };
915        if self.poi_memory_accepts(memory, poi_type) {
916            let _released = storage.release_ticket(held.pos);
917        }
918    }

920    /// Vanilla parity: the `POI_MEMORIES` map, which pairs each POI memory with
921    /// the predicate saying whether this villager may hold that kind of POI in
922    /// it.
923    fn poi_memory_accepts(
924        &self,
925        memory: MemoryModuleType<GlobalPos>,
926        poi_type: PoiTypeRef,
927    ) -> bool {
928        let profession = self.profession();
929        let jobless = profession.key.path == "none" || profession.key.path == "nitwit";
930        match memory.id().key() {
931            "minecraft:home" => poi_type.key == vanilla_poi_types::HOME.key,
932            "minecraft:meeting_point" => poi_type.key == vanilla_poi_types::MEETING.key,
933            "minecraft:job_site" => !jobless && poi_type.key == profession.key,
934            "minecraft:potential_job_site" => REGISTRY
935                .poi_types
936                .is_in_tag(poi_type, &PoiTag::ACQUIRABLE_JOB_SITE),
937            _ => false,
938        }
939    }

941    /// Runs the once-a-tick trading bookkeeping.
942    ///
943    /// Vanilla parity: the `updateMerchantTimer` and `lastTradedPlayer` halves
944    /// of `Villager.customServerAiStep`.
945    fn tick_merchant(&self, world: &Arc<World>) {
946        if !self.is_trading()
947            && let Some(level_up) = self.merchant.tick_level_up_timer()
948        {
949            if level_up {
950                self.increase_merchant_career();
951            }
952            self.living_base
953                .add_mob_effect(MobEffectInstance::with_duration(
954                    vanilla_mob_effects::REGENERATION,
955                    LEVEL_UP_REGENERATION_TICKS,
956                    0,
957                ));
958        }
959
960        if let Some(player) = self.merchant.take_last_traded_player() {
961            self.on_reputation_event_from(ReputationEventType::Trade, player);
962            self.broadcast_entity_event(EntityStatus::VillagerHappy);
963        }
964
965        // Vanilla parity: a villager whose profession was taken away mid-trade
966        // has its screen closed.
967        if self.profession().key.path == "none" && self.is_trading() {
968            self.stop_trading();
969        }
970
971        let game_time = world.game_time();
972        if self.should_restock(game_time) {
973            self.restock(game_time);
974        }
975    }

1334    /// Vanilla parity: `Villager.mobInteract`.
1335    fn mob_interact(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
1336        let stack = player.inventory.lock().get_item_in_hand(hand).clone();
1337        if stack.item().key.path == "villager_spawn_egg"
1338            || !LivingEntity::is_alive(self)
1339            || self.is_trading()
1340            || LivingEntity::is_sleeping(self)
1341        {
1342            return InteractionResult::Pass;
1343        }
1344
1345        if AgeableMob::is_baby(self) {
1346            self.set_unhappy();
1347            return InteractionResult::Success;
1348        }
1349
1350        let offers = self.offers();
1351        let no_offers = offers.is_empty();
1352        if hand == InteractionHand::MainHand && no_offers {
1353            self.set_unhappy();
1354        }
1355        if no_offers {
1356            return InteractionResult::Consume;
1357        }
1358
1359        self.start_trading(player);
1360        InteractionResult::Success
1361    }
