1//! Illusioner entity.
2//!
3//! Vanilla parity: `Illusioner`. The illager nothing spawns: it exists only
4//! behind a summon command, and it is the only illager that carries a bow. Both
5//! of its spells attack a player's information rather than their health -- one
6//! blinds, the other splits the illusioner into four decoys the client draws
7//! and the server knows nothing about.

56/// Experience an illusioner drops.
57///
58/// Vanilla parity: the `xpReward = 5` of the constructor.
59const XP_REWARD: i32 = 5;
60
61/// Ticks between two arrows.
62///
63/// Vanilla parity: the `20` interval of `RangedBowAttackGoal`.
64const BOW_ATTACK_INTERVAL: i32 = 20;
65
66/// Range within which the illusioner shoots rather than approaches.
67const BOW_ATTACK_RADIUS: f64 = 15.0;
68
69/// Speed the illusioner closes the distance at.
70///
71/// Vanilla parity: the `0.5` of `RangedBowAttackGoal`, half a skeleton's.
72const BOW_APPROACH_SPEED: f64 = 0.5;
73
74/// Speed the arrow leaves the bow at.
75///
76/// Vanilla parity: the `1.6F` of `Projectile.spawnProjectileUsingShoot`.
77const ARROW_POWER: f32 = 1.6;
78
79/// Spread of a mob-fired arrow before difficulty is applied.
80///
81/// Vanilla parity: the `14 - difficulty * 4` of `performRangedAttack`.
82const ARROW_UNCERTAINTY_BASE: f32 = 14.0;
83
84/// How much each difficulty step tightens the spread.
85const ARROW_UNCERTAINTY_PER_DIFFICULTY: f32 = 4.0;
86
87/// Fraction of a target's height an arrow is aimed at.
88///
89/// Vanilla parity: the `getY(0.3333333333333333)` of `performRangedAttack`.
90const AIM_HEIGHT_FRACTION: f64 = 1.0 / 3.0;
91
92/// Speed an illusioner wanders at.
93const STROLL_SPEED_MODIFIER: f64 = 0.6;
94
95/// Distance at which an illusioner watches a player.
96const LOOK_AT_PLAYER_RANGE: f64 = 3.0;
97
98/// How often an illusioner bothers to watch a player.
99const LOOK_AT_PLAYER_PROBABILITY: f32 = 1.0;
100
101/// Distance at which an illusioner watches another mob.
102const LOOK_AT_MOB_RANGE: f64 = 8.0;
103
104/// How often an illusioner bothers to watch another mob.
105const LOOK_AT_MOB_PROBABILITY: f32 = 0.02;
106
107/// How long a caster keeps chasing a target it has lost sight of.
108const UNSEEN_MEMORY_TICKS: i32 = 300;
109
110/// Speed a follower patrols at.
111const PATROL_SPEED: f64 = 0.7;
112
113/// Speed the captain patrols at.
114const PATROL_LEADER_SPEED: f64 = 0.595;
115/// Speed a raider walks the streets of the village it is raiding at.
116///
117/// Vanilla parity: the `1.05F` of `new RaiderMoveThroughVillageGoal(this, 1.05F, 1)`.
118const VILLAGE_WALK_SPEED_MODIFIER: f64 = 1.05;
119
120/// How close to a house counts as having reached it.
121///
122/// Vanilla parity: the `1` of the same goal.
123const VILLAGE_POI_ARRIVAL_DISTANCE: f64 = 1.0;

143impl IllusionerEntity {
144    /// Creates an illusioner at runtime.
145    #[must_use]
146    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
147        Self::new_with_base(
148            EntityBase::new(id, position, entity_type.dimensions, world),
149            entity_type,
150        )
151    }
152
153    /// Creates an illusioner from saved base data.
154    #[must_use]
155    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
156        Self::new_with_base(
157            EntityBase::from_load(load, entity_type.dimensions),
158            entity_type,
159        )
160    }
161
162    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
163        let living_base = LivingEntityBase::new(entity_type);
164        let mob_base = MobBase::new();
165        let mut entity_data = IllusionerEntityData::new();
166        living_base.initialize_synced_data(&mut entity_data);
167
168        {
169            // Vanilla parity: the goal order of `Illusioner.registerGoals`,
170            // over the ones `PatrollingMonster` and `Raider` add.
171            let mut goals = mob_base.goal_selector().lock();
172            goals.add_goal(0, FloatGoal::new(&mob_base));
173            goals.add_goal(1, SpellcasterCastingSpellGoal::new());
174            goals.add_goal(4, IllusionerMirrorSpellGoal::new());
175            goals.add_goal(
176                4,
177                LongDistancePatrolGoal::new(PATROL_SPEED, PATROL_LEADER_SPEED),
178            );
179            goals.add_goal(5, IllusionerBlindnessSpellGoal::new());
180            goals.add_goal(1, ObtainRaidLeaderBannerGoal::new());
181            goals.add_goal(3, PathfindToRaidGoal::new());
182            goals.add_goal(
183                4,
184                RaiderMoveThroughVillageGoal::new(
185                    VILLAGE_WALK_SPEED_MODIFIER,
186                    VILLAGE_POI_ARRIVAL_DISTANCE,
187                ),
188            );
189            goals.add_goal(5, RaiderCelebrationGoal::new());
190            goals.add_goal(
191                6,
192                RangedBowAttackGoal::new(
193                    BOW_ATTACK_INTERVAL,
194                    BOW_ATTACK_RADIUS,
195                    BOW_APPROACH_SPEED,
196                    fire_arrow,
197                ),
198            );
199            goals.add_goal(8, RandomStrollGoal::new(STROLL_SPEED_MODIFIER));
200            goals.add_goal(
201                9,
202                LookAtPlayerGoal::new_with_probability(
203                    LOOK_AT_PLAYER_RANGE,
204                    LOOK_AT_PLAYER_PROBABILITY,
205                ),
206            );
207            goals.add_goal(
208                10,
209                LookAtPlayerGoal::new_for_living_entities(
210                    LOOK_AT_MOB_RANGE,
211                    LOOK_AT_MOB_PROBABILITY,
212                    |_, target, _| target.as_mob().is_some(),
213                ),
214            );
215            // Vanilla also flees a creaking at priority 3, and adds three
216            // raid-only goals. Foton has no creaking and no raid.
217        }
218
219        {
220            let mut targets = mob_base.target_selector().lock();
221            targets.add_goal(
222                1,
223                HurtByTargetGoal::new()
224                    .with_ignored_damage_filter(|entity| entity.as_raider().is_some())
225                    .set_alert_others([]),
226            );
227            targets.add_goal(
228                2,
229                NearestAttackableTargetGoal::new_for_players(true, |_, _, _| true)
230                    .with_unseen_memory_ticks(UNSEEN_MEMORY_TICKS),
231            );
232            targets.add_goal(
233                3,
234                NearestAttackableTargetGoal::new(false, |_, target, _| {
235                    target.entity_type() == &vanilla_entities::IRON_GOLEM
236                })
237                .with_unseen_memory_ticks(UNSEEN_MEMORY_TICKS),
238            );
239            // Vanilla also hunts villagers at priority 3; Foton has none.
240        }
241
242        let illusioner = Self {
243            base,
244            entity_type,
245            living_base,
246            mob_base,
247            entity_data: SyncMutex::new(entity_data),
248            patrol_state: PatrolState::new(),
249            raider_state: RaiderState::new(),
250            spellcaster_state: SpellcasterState::new(),
251        };
252        illusioner.set_xp_reward(XP_REWARD);
253        illusioner
254    }
255}

257/// Looses an arrow at `target`.
258///
259/// Vanilla parity: `Illusioner.performRangedAttack`. Foton's arrow already
260/// applies the same `distance * 0.2` lob, so the aim point is the only part
261/// worth spelling out. Vanilla draws the projectile from the illusioner's
262/// quiver; Foton's mobs have none, so the arrow is a plain one.
263fn fire_arrow(mob: &dyn PathfinderMob, target: DVec3, power: f32) {
264    let Some(illusioner) = mob.downcast_ref::<IllusionerEntity>() else {
265        return;
266    };
267    let Some(world) = mob.level() else {
268        return;
269    };
270
271    // The goal hands over the target's feet; vanilla aims a third of the way
272    // up its body instead, which is the difference between hitting a player
273    // and hitting the block they are standing on.
274    let target = mob.target().map_or(target, |entity| {
275        let height = f64::from(entity.base().dimensions().height);
276        target.with_y(height.mul_add(AIM_HEIGHT_FRACTION, entity.position().y))
277    });
278
279    let difficulty = u8::from(world.difficulty());
280    let uncertainty =
281        ARROW_UNCERTAINTY_PER_DIFFICULTY.mul_add(-f32::from(difficulty), ARROW_UNCERTAINTY_BASE);
282    let arrow = ArrowEntity::shoot_at(&world, illusioner, target, ARROW_POWER, uncertainty);
283    // Vanilla parity: the `ProjectileUtil.getMobArrow(.., power, ..)`
284    // of `performRangedAttack` -- a shallower draw hits softer.
285    arrow.set_base_damage_from_mob(power);
286    drop(arrow);
287
288    world.play_sound_at(
289        &sound_events::ENTITY_SKELETON_SHOOT,
290        illusioner.sound_source(),
291        illusioner.position(),
292        1.0,
293        0.4f32.mul_add(rand::random::<f32>(), 0.8).recip(),
294        None,
295    );
296}

386    /// Returns whether this mob accepts where the spawner put it.
387    ///
388    /// Vanilla parity: the `Monster::checkMonsterSpawnRules` `SpawnPlacements`

394        world: &Arc<World>,
395        spawn_reason: EntitySpawnReason,
396        pos: BlockPos,

398        check_monster_spawn_rules(world, spawn_reason, pos)

401    fn finalize_spawn(
402        &self,
403        world: &Arc<World>,
404        spawn_reason: EntitySpawnReason,
405        group_data: Option<SpawnGroupData>,
406    ) -> Option<SpawnGroupData> {
407        // Vanilla parity: `Illusioner.finalizeSpawn` arms it before the shared
408        // raider spawn work.
409        self.living_base()
410            .equipment()
411            .lock()
412            .set(EquipmentSlot::MainHand, ItemStack::new(&vanilla_items::BOW));
413        finalize_spawn_raider(self, spawn_reason);
414        self.finalize_spawn_mob_base(world, spawn_reason, group_data)
415    }

496impl AbstractIllager for IllusionerEntity {

507        IllagerArmPose::Crossed
