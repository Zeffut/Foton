1//! Vindicator entity.
2//!
3//! Vanilla parity: `Vindicator`. The illager with the axe. Outside a raid it is
4//! a plain melee hostile with a patrol; inside one it breaks doors. The one
5//! oddity is Johnny: a vindicator renamed with a name tag stops caring what
6//! it is attacking and goes after every living thing in range, which is the
7//! single most dangerous mob in the game to let loose in a village.

53/// NBT key vanilla stores the Johnny flag under.
54///
55/// Vanilla parity: `Vindicator.TAG_JOHNNY`.
56const TAG_JOHNNY: &str = "Johnny";

58/// The name that turns a vindicator on everything alive.
59const JOHNNY_NAME: &str = "Johnny";

61/// Distance at which a patrol stops shouting and charges.
62const HOLD_GROUND_RADIUS: f32 = 10.0;

64/// Speed a vindicator closes on its target at.
65///
66/// Vanilla parity: the `1.0` of `MeleeAttackGoal`.
67const MELEE_SPEED_MODIFIER: f64 = 1.0;
68
69/// Speed a vindicator wanders at.
70const STROLL_SPEED_MODIFIER: f64 = 0.6;
71
72/// Distance at which a vindicator watches a player.
73///
74/// Vanilla parity: the `3.0F` of the player `LookAtPlayerGoal`.
75const LOOK_AT_PLAYER_RANGE: f64 = 3.0;
76
77/// How often a vindicator bothers to watch a player.
78const LOOK_AT_PLAYER_PROBABILITY: f32 = 1.0;
79
80/// Distance at which a vindicator watches another mob.
81const LOOK_AT_MOB_RANGE: f64 = 8.0;
82
83/// How often a vindicator bothers to watch another mob.
84const LOOK_AT_MOB_PROBABILITY: f32 = 0.02;
85
86/// Speed a follower patrols at.
87const PATROL_SPEED: f64 = 0.7;
88
89/// Speed the captain patrols at.
90const PATROL_LEADER_SPEED: f64 = 0.595;
91/// Speed a raider walks the streets of the village it is raiding at.
92///
93/// Vanilla parity: the `1.05F` of `new RaiderMoveThroughVillageGoal(this, 1.05F, 1)`.
94const VILLAGE_WALK_SPEED_MODIFIER: f64 = 1.05;

96/// How close to a house counts as having reached it.
97///
98/// Vanilla parity: the `1` of the same goal.

101/// A vindicator.
102#[entity_behavior(class = "Vindicator")]
103pub struct VindicatorEntity {
104    base: EntityBase,
105    entity_type: EntityTypeRef,
106    living_base: LivingEntityBase,
107    mob_base: MobBase,
108    entity_data: SyncMutex<VindicatorEntityData>,

120impl VindicatorEntity {
121    /// Creates a vindicator at runtime.
122    #[must_use]
123    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
124        Self::new_with_base(
125            EntityBase::new(id, position, entity_type.dimensions, world),
126            entity_type,
127        )
128    }
129
130    /// Creates a vindicator from saved base data.
131    #[must_use]
132    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
133        Self::new_with_base(
134            EntityBase::from_load(load, entity_type.dimensions),
135            entity_type,
136        )
137    }
138
139    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
140        let living_base = LivingEntityBase::new(entity_type);
141        let mob_base = MobBase::new();
142        let mut entity_data = VindicatorEntityData::new();
143        living_base.initialize_synced_data(&mut entity_data);
144
145        {
146            // Vanilla parity: the goal order of `Vindicator.registerGoals`,
147            // over the ones `PatrollingMonster` and `Raider` add.
148            let mut goals = mob_base.goal_selector().lock();
149            goals.add_goal(0, FloatGoal::new(&mob_base));
150            goals.add_goal(2, VindicatorBreakDoorGoal::new());
151            goals.add_goal(3, RaiderOpenDoorGoal::new());
152            goals.add_goal(4, HoldGroundAttackGoal::new(HOLD_GROUND_RADIUS));
153            goals.add_goal(
154                4,
155                LongDistancePatrolGoal::new(PATROL_SPEED, PATROL_LEADER_SPEED),
156            );
157            goals.add_goal(5, MeleeAttackGoal::new(MELEE_SPEED_MODIFIER, false));
158            goals.add_goal(1, ObtainRaidLeaderBannerGoal::new());
159            goals.add_goal(3, PathfindToRaidGoal::new());
160            goals.add_goal(
161                4,
162                RaiderMoveThroughVillageGoal::new(
163                    VILLAGE_WALK_SPEED_MODIFIER,
164                    VILLAGE_POI_ARRIVAL_DISTANCE,
165                ),
166            );
167            goals.add_goal(5, RaiderCelebrationGoal::new());
168            goals.add_goal(8, RandomStrollGoal::new(STROLL_SPEED_MODIFIER));
169            goals.add_goal(
170                9,
171                LookAtPlayerGoal::new_with_probability(
172                    LOOK_AT_PLAYER_RANGE,
173                    LOOK_AT_PLAYER_PROBABILITY,
174                ),
175            );
176            goals.add_goal(
177                10,
178                LookAtPlayerGoal::new_for_living_entities(
179                    LOOK_AT_MOB_RANGE,
180                    LOOK_AT_MOB_PROBABILITY,
181                    |_, target, _| target.as_mob().is_some(),
182                ),
183            );
184            // Vanilla also flees a creaking at priority 1, and adds three
185            // raid-only goals. Foton has no creaking and no raid.
186        }
187
188        {
189            let mut targets = mob_base.target_selector().lock();
190            targets.add_goal(
191                1,
192                HurtByTargetGoal::new()
193                    .with_ignored_damage_filter(|entity| entity.as_raider().is_some())
194                    .set_alert_others([]),
195            );
196            targets.add_goal(
197                2,
198                NearestAttackableTargetGoal::new_for_players(true, |_, _, _| true),
199            );
200            targets.add_goal(
201                3,
202                NearestAttackableTargetGoal::new(true, |_, target, _| {
203                    target.entity_type() == &vanilla_entities::IRON_GOLEM
204                }),
205            );
206            targets.add_goal(4, VindicatorJohnnyAttackGoal::new());
207            // Vanilla also hunts villagers at priority 3; Foton has none.
208        }
209
210        Self {
211            base,
212            entity_type,
213            living_base,
214            mob_base,
215            entity_data: SyncMutex::new(entity_data),
216            patrol_state: PatrolState::new(),
217            raider_state: RaiderState::new(),
218            is_johnny: SyncMutex::new(false),
219        }
220    }
221
222    /// Returns whether this vindicator answers to Johnny.
223    #[must_use]
224    pub fn is_johnny(&self) -> bool {
225        *self.is_johnny.lock()
226    }
227
228    /// Gives this vindicator an iron axe unless a raid is about to arm it.
229    ///
230    /// Vanilla parity: `Vindicator.populateDefaultEquipmentSlots`.
231    fn populate_default_equipment(&self) {
232        if self.current_raid_status().is_some() {
233            return;
234        }
235        self.living_base().equipment().lock().set(
236            EquipmentSlot::MainHand,
237            ItemStack::new(&vanilla_items::IRON_AXE),
238        );
239    }
240}

263    fn is_allied_to(&self, other: &dyn Entity) -> bool {

265    }
266
267    /// Latches Johnny the first time the name is applied.
268    ///
269    /// Vanilla parity: `Vindicator.setCustomName`. The flag is one-way:
270    /// renaming a Johnny to anything else does not calm it down.
271    fn set_custom_name(&self, custom_name: Option<TextComponent>) {
272        let becomes_johnny = !self.is_johnny()
273            && custom_name
274                .as_ref()
275                .is_some_and(|name| name.to_plain(&DisplayResolutor) == JOHNNY_NAME);
276        self.entity_set_custom_name(custom_name);
277        if becomes_johnny {
278            *self.is_johnny.lock() = true;
279        }
280    }
281
282    fn save_additional(&self, nbt: &mut NbtCompound) {
283        self.save_mob(nbt);
284        write_patrol_state(self, nbt);
285        write_raider_state(self, nbt);
286        if self.is_johnny() {
287            nbt.insert(TAG_JOHNNY, i8::from(true));
288        }
289    }
290
291    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
292        self.load_mob(nbt);
293        read_patrol_state(self, nbt);
294        read_raider_state(self, nbt);
295        *self.is_johnny.lock() = nbt.byte(TAG_JOHNNY).is_some_and(|value| value != 0);
296    }
297}

337impl Mob for VindicatorEntity {
338    /// Vanilla parity: `Vindicator` derives from `Monster`.
339    fn is_monster(&self) -> bool {
340        true
341    }
342
343    fn mob_base(&self) -> &MobBase {
344        &self.mob_base
345    }
346
347    /// Returns whether this mob accepts where the spawner put it.
348    ///
349    /// Vanilla parity: the `Monster::checkMonsterSpawnRules` `SpawnPlacements`
350    /// registers for the vindicator -- unlike the pillager, which gets the
351    /// patrol rule, a vindicator needs the dark.
352    fn check_spawn_rules(
353        &self,
354        world: &Arc<World>,
355        spawn_reason: EntitySpawnReason,
356        pos: BlockPos,
357    ) -> bool {
358        check_monster_spawn_rules(world, spawn_reason, pos)
359    }
360
361    fn finalize_spawn(
362        &self,
363        world: &Arc<World>,
364        spawn_reason: EntitySpawnReason,
365        group_data: Option<SpawnGroupData>,
366    ) -> Option<SpawnGroupData> {
367        finalize_spawn_raider(self, spawn_reason);
368        // Vanilla parity: `Vindicator.finalizeSpawn` turns door opening on for
369        // every vindicator, raid or not.
370        self.mob_base().navigation().lock().set_can_open_doors(true);
371        self.populate_default_equipment();
372        self.finalize_spawn_mob_base(world, spawn_reason, group_data)
373    }
374
375    /// Vanilla parity: `Vindicator.customServerAiStep`, which re-checks door
376    /// opening against the raid every tick. Foton has no raid, so a vindicator
377    /// that had doors enabled at spawn keeps them; one that did not, does not.
378    fn custom_server_ai_step(&self) {
379        if self.is_no_ai() {
380            return;
381        }
382        let raided = self.has_active_raid();
383        self.mob_base()
384            .navigation()
385            .lock()
386            .set_can_open_doors(raided);
387    }
388
389    fn remove_when_far_away(&self, dist_sqr: f64) -> bool {
390        self.remove_when_far_away_raider(dist_sqr)
391    }
392
393    fn requires_custom_persistence(&self) -> bool {
394        self.requires_custom_persistence_raider() || self.is_passenger() || self.is_leashed()
395    }
396
397    /// Vanilla parity: `Raider.updateNoActionTime`.
398    fn update_no_action_time(&self) {
399        self.increment_no_action_time();
400        self.increment_no_action_time();
401    }

405    }
