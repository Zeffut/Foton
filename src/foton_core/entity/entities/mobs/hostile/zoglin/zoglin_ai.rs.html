1//! The zoglin's brain.
2//!
3//! Vanilla parity: the static `getActivities`, `initCoreActivity`,
4//! `initIdleActivity` and `initFightActivity` of
5//! `net.minecraft.world.entity.monster.Zoglin`. Vanilla keeps them on the mob
6//! class rather than in a separate `Ai` helper; Foton splits them out so the
7//! entity file stays about the entity.
8

10use foton_utils::value_providers::UniformIntProvider;

12use crate::entity::ai::brain::behavior::{

14    RunOne, SetEntityLookTargetSometimes, SetWalkTargetFromAttackTargetIfTargetOutOfReach,

22/// Vanilla parity: `Zoglin.ATTACK_INTERVAL`.
23const ATTACK_INTERVAL: i64 = 40;
24/// Vanilla parity: `Zoglin.BABY_ATTACK_INTERVAL`.
25const BABY_ATTACK_INTERVAL: i64 = 15;
26/// Vanilla parity: `Zoglin.MOVEMENT_SPEED_WHEN_FIGHTING`, the `1.0F` of the
27/// fight activity's walk behavior.
28const SPEED_MULTIPLIER_WHEN_CHASING: f64 = 1.0;
29/// Vanilla parity: `Zoglin.SPEED_MULTIPLIER_WHEN_IDLING`.
30const SPEED_MULTIPLIER_WHEN_IDLING: f64 = 0.4;
31/// Vanilla parity: the `LookAtTargetSink(45, 90)` of the core activity.
32const LOOK_AT_TARGET_MIN_DURATION: i32 = 45;
33const LOOK_AT_TARGET_MAX_DURATION: i32 = 90;
34/// Vanilla parity: the `SetEntityLookTargetSometimes.create(8.0F, UniformInt.of(30, 60))`
35/// of the idle activity.
36const GAZE_RANGE: f64 = 8.0;
37const GAZE_INTERVAL: UniformIntProvider = UniformIntProvider {
38    min_inclusive: 30,
39    max_inclusive: 60,
40};
41/// Vanilla parity: the `DoNothing(30, 60)` of the idle gate.
42const IDLE_DO_NOTHING_MIN: i32 = 30;
43const IDLE_DO_NOTHING_MAX: i32 = 60;
44/// Vanilla parity: the `SetWalkTargetFromLookTarget.create(0.4F, 3)` of the same gate.
45const IDLE_LOOK_WALK_CLOSE_ENOUGH: i32 = 3;
46
47/// The sensors a zoglin runs.
48///
49/// Vanilla parity: the sensor list of `Zoglin.BRAIN_PROVIDER`, which is the
50/// shortest of any brain mob: it needs to know what is nearby, and nothing else.
51pub const SENSORS: &[SensorType] = &[
52    SensorType::NearestLivingEntities,
53    SensorType::NearestPlayers,
54];
55
56/// Builds a zoglin's brain.
57///
58/// Vanilla parity: `Zoglin.BRAIN_PROVIDER` feeding `Zoglin.getActivities`.
59#[must_use]
60pub fn make_brain() -> Brain {
61    Brain::new(
62        SENSORS,
63        vec![core_activity(), idle_activity(), fight_activity()],
64    )
65}

67/// Vanilla parity: `Zoglin.initCoreActivity`.
68fn core_activity() -> ActivityData {
69    ActivityData::create(
70        Activity::Core,
71        0,
72        vec![
73            Behavior::boxed(LookAtTargetSink::new(
74                LOOK_AT_TARGET_MIN_DURATION,
75                LOOK_AT_TARGET_MAX_DURATION,
76            )),
77            Behavior::boxed(MoveToTargetSink::new()),
78        ],
79    )
80}
81
82/// Vanilla parity: `Zoglin.initIdleActivity`.
83fn idle_activity() -> ActivityData {
84    ActivityData::create(
85        Activity::Idle,
86        10,
87        vec![
88            OneShot::boxed(StartAttacking::new(find_nearest_valid_attack_target)),
89            OneShot::boxed(SetEntityLookTargetSometimes::any_within(
90                GAZE_RANGE,
91                GAZE_INTERVAL,
92            )),
93            Box::new(RunOne::unconditional(vec![
94                (
95                    OneShot::boxed(RandomStroll::stroll(SPEED_MULTIPLIER_WHEN_IDLING)),
96                    2,
97                ),
98                (
99                    OneShot::boxed(SetWalkTargetFromLookTarget::new(
100                        SPEED_MULTIPLIER_WHEN_IDLING,
101                        IDLE_LOOK_WALK_CLOSE_ENOUGH,
102                    )),
103                    2,
104                ),
105                (
106                    Box::new(DoNothing::new(IDLE_DO_NOTHING_MIN, IDLE_DO_NOTHING_MAX)),
107                    1,
108                ),
109            ])),
110        ],
111    )
112}
113
114/// Vanilla parity: `Zoglin.initFightActivity`.
115fn fight_activity() -> ActivityData {
116    ActivityData::create(
117        Activity::Fight,
118        10,
119        vec![
120            OneShot::boxed(SetWalkTargetFromAttackTargetIfTargetOutOfReach::new(
121                SPEED_MULTIPLIER_WHEN_CHASING,
122            )),
123            OneShot::boxed(MeleeAttack::conditional(
124                |mob| !mob.is_baby(),
125                ATTACK_INTERVAL,
126            )),
127            OneShot::boxed(MeleeAttack::conditional(
128                LivingEntity::is_baby,
129                BABY_ATTACK_INTERVAL,
130            )),
131            OneShot::boxed(StopAttackingIfTargetInvalid::new()),
132        ],
133    )
134    .gated_by(memory_module_types::ATTACK_TARGET.id())
135}

137/// Vanilla parity: the private `Zoglin.findNearestValidAttackTarget`, which
138/// attacks anything visible that is neither another zoglin nor a creeper.
139fn find_nearest_valid_attack_target(ctx: &BrainContext<'_>) -> Option<SharedEntity> {
140    ctx.brain()
141        .get_memory(memory_module_types::NEAREST_VISIBLE_LIVING_ENTITIES)?
142        .find_closest(|candidate| {
143            let raw = candidate.as_entity_event_source();
144            !utils::is_of_type(raw, &vanilla_entities::ZOGLIN)
145                && !utils::is_of_type(raw, &vanilla_entities::CREEPER)
146                && is_entity_attackable(ctx.world(), ctx.mob(), candidate)
147        })
148}
