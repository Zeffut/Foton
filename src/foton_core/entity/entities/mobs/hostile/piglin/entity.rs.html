1//! Piglin entity.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.monster.piglin.Piglin`. The mob
4//! the Nether's economy runs on: it wears gold, hunts hoglins in packs, is
5//! calmed by gold armor, and trades a gold ingot for whatever
6//! `gameplay/piglin_bartering` rolls.
7//!
8//! Everything it does is brain-driven -- see [`super::piglin_ai`]. This file is
9//! the body: the inventory it carries, the crossbow it fires, the equipment it
10//! spawns with, and the clock that turns it into a
11//! [`crate::entity::entities::ZombifiedPiglinEntity`] in the overworld.

277    /// What this piglin is doing with its hands, for the client.
278    ///
279    /// Vanilla parity: `Piglin.getArmPose`.
280    #[must_use]
281    pub fn arm_pose(&self) -> PiglinArmPose {
282        if self.is_dancing() {
283            return PiglinArmPose::Dancing;
284        }
285        if piglin_ai::is_loved_item(&self.get_item_in_hand(InteractionHand::OffHand)) {
286            return PiglinArmPose::AdmiringItem;
287        }
288        if Mob::is_aggressive(self) && abstract_piglin::is_holding_melee_weapon(self) {
289            return PiglinArmPose::AttackingWithMeleeWeapon;
290        }
291        if self.is_charging_crossbow() {
292            return PiglinArmPose::CrossbowCharge;
293        }
294        let main_hand = self.get_item_in_hand(InteractionHand::MainHand);
295        if main_hand.is(&vanilla_items::CROSSBOW) && crossbow_is_charged(&main_hand) {
296            return PiglinArmPose::CrossbowHold;
297        }
298        PiglinArmPose::Default
299    }

311    /// Whether the carried inventory has room for `item_stack`.
312    ///
313    /// Vanilla parity: `Piglin.canAddToInventory`, which delegates to
314    /// `SimpleContainer.canAddItem`.
315    #[must_use]
316    pub fn can_add_to_inventory(&self, item_stack: &ItemStack) -> bool {
317        let inventory = self.inventory.lock();
318        for slot in 0..inventory.get_container_size() {
319            let existing = inventory.get_item(slot);
320            if existing.is_empty() {
321                return true;
322            }
323            if ItemStack::is_same_item_same_components(existing, item_stack)
324                && existing.count() + item_stack.count() <= existing.max_stack_size()
325            {
326                return true;
327            }
328        }
329        false
330    }

332    /// Empties the carried inventory.
333    ///
334    /// Vanilla parity: `SimpleContainer.removeAllItems`.
335    pub fn remove_all_inventory_items(&self) -> Vec<ItemStack> {
336        let mut inventory = self.inventory.lock();
337        let mut taken = Vec::new();
338        for slot in 0..inventory.get_container_size() {
339            let item = inventory.remove_item_no_update(slot);
340            if !item.is_empty() {
341                taken.push(item);
342            }
343        }
344        taken

364    /// Vanilla parity: the private `Piglin.createSpawnWeapon`.
365    fn create_spawn_weapon() -> ItemStack {
366        if rand::random::<f32>() < PROBABILITY_OF_SPAWNING_WITH_CROSSBOW {
367            return ItemStack::new(&vanilla_items::CROSSBOW);
368        }
369        if rand::random_range(0..ONE_IN_N_BLADES_IS_A_SPEAR) == 0 {
370            return ItemStack::new(&vanilla_items::GOLDEN_SPEAR);
371        }
372        ItemStack::new(&vanilla_items::GOLDEN_SWORD)
373    }

405    /// Turns this piglin into a zombified piglin.
406    ///
407    /// Vanilla parity: `Piglin.finishConversion`, which cancels the admiring,
408    /// spills the carried inventory, and then runs `AbstractPiglin`'s own
409    /// conversion with `ConversionParams.single(this, true, true)`.
410    pub fn finish_conversion(&self) -> Option<Arc<ZombifiedPiglinEntity>> {
411        piglin_ai::cancel_admiring(self);
412        for item in self.remove_all_inventory_items() {
413            let _ = self.spawn_at_location(item, 0.0);
414        }
415
416        let equipment: Vec<(EquipmentSlot, ItemStack)> = EquipmentSlot::ALL
417            .into_iter()
418            .map(|slot| (slot, self.get_item_by_slot(slot)))
419            .filter(|(_, item)| !item.is_empty())
420            .collect();
421
422        convert_to(
423            self,
424            ConversionParams::single(true, true).with_reason(PiglinZombification),
425            |id, position, world| {
426                ZombifiedPiglinEntity::new(&vanilla_entities::ZOMBIFIED_PIGLIN, id, position, world)
427            },
428            |zombified| {
429                // The baby flag comes across in `copy_common_state`.
430                // Vanilla's `keepEquipment` moves the slots inside
431                // `ConversionType.SINGLE.convert`; Foton's conversion leaves
432                // that to the caller, so a piglin's gold armor comes across
433                // here rather than being lost.
434                for (slot, item) in equipment {
435                    zombified.set_item_slot(slot, item);
436                }
437                zombified
438                    .living_base()
439                    .add_mob_effect(MobEffectInstance::with_duration(
440                        vanilla_mob_effects::NAUSEA,
441                        CONVERSION_NAUSEA_TICKS,
442                        0,
443                    ));
444            },
445        )
446    }

448    /// The hooks the brain's [`crate::entity::ai::brain::behavior::CrossbowAttack`]
449    /// drives this piglin through.
450    #[must_use]
451    pub fn crossbow_hooks() -> CrossbowAttackHooks {
452        CrossbowAttackHooks {
453            set_charging_crossbow: |ctx: &BrainContext<'_>, charging: bool| {
454                if let Some(piglin) = ctx.mob().downcast_ref::<Self>() {
455                    piglin.set_charging_crossbow(charging);
456                }
457            },
458            perform_ranged_attack: |ctx: &BrainContext<'_>, target: &SharedEntity, _power: f32| {
459                let Some(piglin) = ctx.mob().downcast_ref::<Self>() else {
460                    return;
461                };
462                // Vanilla parity: `Piglin.performRangedAttack` ignores the
463                // power the behavior passes and fires at `MOB_ARROW_POWER`.
464                perform_crossbow_attack(ctx.world(), piglin, target, MOB_ARROW_POWER);
465                // Vanilla parity: `Piglin.onCrossbowAttackPerformed`.
466                piglin.set_no_action_time(0);
467            },
468        }
469    }

559    fn save_additional(&self, nbt: &mut NbtCompound) {

573        Mob::set_can_pick_up_loot(self, nbt.byte("CanPickUpLoot").is_none_or(|flag| flag != 0));
574        self.set_immune_to_zombification(nbt.byte("IsImmuneToZombification").unwrap_or(0) != 0);
575        self.set_baby(nbt.byte("IsBaby").unwrap_or(0) != 0);
576        {
577            let mut state = self.state.lock();
578            state.time_in_overworld = nbt.int("TimeInOverworld").unwrap_or(0);
579            state.cannot_hunt = nbt.byte("CannotHunt").unwrap_or(0) != 0;
580        }
581        abstract_piglin::load_inventory(&mut self.inventory.lock(), nbt);
582        self.brain.load(nbt);
583    }

623        if !was_hurt {
624            return false;
625        }
626        let Some(world) = self.level() else {
627            return true;
628        };
629        let Some(attacker) = source
630            .causing_entity_id
631            .and_then(|id| world.get_entity_by_id(id))
632        else {
633            return true;
634        };
635        if let Some(living) = attacker.as_living_entity() {
636            piglin_ai::was_hurt_by(&world, self, &attacker, living);
637        }
638        true
639    }

722    /// item ahead of the armor and damage comparisons the base makes.
723    fn can_replace_current_item(
724        &self,
725        new_item_stack: &ItemStack,
726        current_item_stack: &ItemStack,
727        slot: EquipmentSlot,
728    ) -> bool {
729        use foton_registry::enchantment_effect::EnchantmentEffectComponent;
730
731        if current_item_stack.has_enchantment_effect(EnchantmentEffectComponent::PreventArmorChange)
732        {
733            return false;
734        }
735
736        let preferred = self.preferred_weapon_type();
737        let wanted = |stack: &ItemStack| {
738            piglin_ai::is_loved_item(stack)
739                || preferred.is_some_and(|tag| REGISTRY.items.is_in_tag(stack.item(), tag))
740        };
741        let new_wanted = wanted(new_item_stack);
742        let current_wanted = wanted(current_item_stack);
743        if new_wanted && !current_wanted {
744            return true;
745        }
746        if !new_wanted && current_wanted {
747            return false;
748        }
749        self.mob_can_replace_current_item(new_item_stack, current_item_stack, slot)
750    }
