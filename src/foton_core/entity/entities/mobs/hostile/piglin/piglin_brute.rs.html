76impl PiglinBruteEntity {
77    /// Creates a piglin brute at runtime.
78    #[must_use]
79    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
80        Self::new_with_base(
81            EntityBase::new(id, position, entity_type.dimensions, world),
82            entity_type,
83        )
84    }
85
86    /// Creates a piglin brute from saved base data.
87    #[must_use]
88    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
89        Self::new_with_base(
90            EntityBase::from_load(load, entity_type.dimensions),
91            entity_type,
92        )
93    }
94
95    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
96        let living_base = LivingEntityBase::new(entity_type);
97        let mob_base = MobBase::new();
98        mob_base.set_xp_reward(XP_REWARD);
99        let mut entity_data = PiglinBruteEntityData::new();
100        living_base.initialize_synced_data(&mut entity_data);
101
102        let brute = Self {
103            base,
104            entity_type,
105            living_base,
106            mob_base,
107            entity_data: SyncMutex::new(entity_data),
108            state: SyncMutex::new(PiglinBruteState {
109                time_in_overworld: 0,
110            }),
111            brain: piglin_brute_ai::make_brain(),
112        };
113        abstract_piglin::apply_constructor(&brute);
114        brute
115    }
116

120        &self.brain
121    }

172impl ConvertiblePiglin for PiglinBruteEntity {
173    /// Vanilla parity: `AbstractPiglin.isConverting`.
174    fn is_converting(&self) -> bool {
175        let Some(world) = self.level() else {
176            return false;
177        };
178        !self.is_immune_to_zombification()
179            && !self.is_no_ai()
180            && world.dimension_type.piglins_zombify
181    }
182
183    fn bump_time_in_overworld(&self, converting: bool) -> i32 {
184        let mut state = self.state.lock();
185        if converting {
186            state.time_in_overworld += 1;
187        } else {
188            state.time_in_overworld = 0;
189        }
190        state.time_in_overworld
191    }
192
193    /// Vanilla parity: `PiglinBrute.playConvertedSound`.
194    fn play_converted_sound(&self) {
195        self.make_sound(Some(
196            &sound_events::ENTITY_PIGLIN_BRUTE_CONVERTED_TO_ZOMBIFIED,
197        ));
198    }
199
200    /// Vanilla parity: `AbstractPiglin.finishConversion`, which a brute does
201    /// not override -- so unlike a piglin it spills nothing on the way.
202    fn convert_to_zombified(&self) {
203        let equipment: Vec<(EquipmentSlot, ItemStack)> = EquipmentSlot::ALL
204            .into_iter()
205            .map(|slot| (slot, self.get_item_by_slot(slot)))
206            .filter(|(_, item)| !item.is_empty())
207            .collect();
208
209        convert_to(
210            self,
211            ConversionParams::single(true, true).with_reason(PiglinZombification),
212            |id, position, world| {
213                ZombifiedPiglinEntity::new(&vanilla_entities::ZOMBIFIED_PIGLIN, id, position, world)
214            },
215            |zombified| {
216                for (slot, item) in equipment {
217                    zombified.set_item_slot(slot, item);
218                }
219                zombified
220                    .living_base()
221                    .add_mob_effect(MobEffectInstance::with_duration(
222                        vanilla_mob_effects::NAUSEA,
223                        CONVERSION_NAUSEA_TICKS,
224                        0,
225                    ));
226            },
227        );
228    }
229}

293    fn set_health(&self, health: f32) {
294        let max_health = self.get_max_health();
295        let clamped = health.clamp(0.0, max_health);
296        self.entity_data
297            .lock()
298            .living_entity_mut()
299            .health
300            .set(clamped);
301    }
302
303    /// Vanilla parity: `PiglinBrute.hurtServer`.
304    fn hurt_server(&self, world: &World, source: &DamageSource, amount: f32) -> bool {
305        let was_hurt = self.living_hurt_server(world, source, amount);
306        if !was_hurt {
307            return false;
308        }
309        let Some(world) = self.level() else {
310            return true;
311        };
312        let Some(attacker) = source
313            .causing_entity_id
314            .and_then(|id| world.get_entity_by_id(id))
315        else {
316            return true;
317        };
318        if let Some(living) = attacker.as_living_entity() {
319            piglin_brute_ai::was_hurt_by(&world, &self.brain, self, &attacker, living);
320        }
321        true
322    }
323

326    }

357        false

359
360    /// Vanilla parity: `PiglinBrute.wantsToPickUp`, which is only ever true for
361    /// the golden axe it fights with.
362    fn wants_to_pick_up(&self, world: &World, item_stack: &ItemStack) -> bool {
363        item_stack.is(&vanilla_items::GOLDEN_AXE) && self.mob_wants_to_pick_up(world, item_stack)
364    }
365
366    /// Vanilla parity: `PiglinBrute.customServerAiStep`.
367    fn custom_server_ai_step(&self) {
368        let Some(world) = self.level() else {
369            return;
370        };
371        self.brain.tick(&world, self);
372        let activity_changed = piglin_brute_ai::update_activity(&self.brain);
373        Mob::set_aggressive(
374            self,
375            self.brain
376                .has_memory_value(memory_module_types::ATTACK_TARGET.id()),
377        );
378        // Vanilla parity: `playActivitySound` on a change, then the random
379        // `maybePlayActivitySound`; both only speak while fighting.
380        if activity_changed && self.brain.active_non_core_activity() == Some(Activity::Fight) {
381            self.play_angry_sound();
382        }
383        if piglin_brute_ai::should_play_activity_sound(&self.brain) {
384            self.play_angry_sound();
385        }
386        abstract_piglin::tick_conversion(self);
387    }
388
389    /// Vanilla parity: `PiglinBrute.getAmbientSound`.
390    fn ambient_sound(&self) -> Option<SoundEventRef> {
391        Some(&sound_events::ENTITY_PIGLIN_BRUTE_AMBIENT)
392    }
393
394    /// Vanilla parity: `PiglinBrute.finalizeSpawn`, which pins the brute's home
395    /// to where it spawned and hands it its axe.
396    fn finalize_spawn(
397        &self,
398        world: &Arc<World>,
399        spawn_reason: EntitySpawnReason,
400        group_data: Option<SpawnGroupData>,
401    ) -> Option<SpawnGroupData> {
402        piglin_brute_ai::init_memories(
403            &self.brain,
404            GlobalPos::new(world.key.clone(), self.block_position()),
405        );
406        self.set_item_slot(
407            EquipmentSlot::MainHand,
408            ItemStack::new(&vanilla_items::GOLDEN_AXE),
409        );
410        self.finalize_spawn_mob_base(world, spawn_reason, group_data)
411    }
412
413    fn mob_flags(&self) -> i8 {
414        *self.entity_data.lock().mob().mob_flags.get()
415    }
416
417    fn set_mob_flags(&self, flags: i8) {
418        self.entity_data.lock().mob_mut().mob_flags.set(flags);
419    }
420}
