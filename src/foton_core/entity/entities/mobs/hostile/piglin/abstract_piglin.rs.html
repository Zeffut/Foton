2//!
3//! Vanilla parity: `net.minecraft.world.entity.monster.piglin.AbstractPiglin`.
4//! Rust has no abstract base class, so the shared body is these free functions
5//! and the two mobs call them from their own overrides.

17/// How long a piglin has to stand in the overworld before it turns.
18///
19/// Vanilla parity: `AbstractPiglin.CONVERSION_TIME`.
20pub const CONVERSION_TIME: i32 = 300;
21
22/// How badly a piglin wants to keep clear of fire.
23///
24/// Vanilla parity: the two `setPathfindingMalus` calls of the
25/// `AbstractPiglin` constructor.
26const FIRE_IN_NEIGHBOR_MALUS: f32 = 16.0;
27const FIRE_MALUS: f32 = -1.0;
28
29/// What a piglin holds in its hands, on the client.
30///
31/// Vanilla parity: `net.minecraft.world.entity.monster.piglin.PiglinArmPose`.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub enum PiglinArmPose {
34    AttackingWithMeleeWeapon,
35    CrossbowHold,
36    CrossbowCharge,
37    AdmiringItem,
38    Dancing,
39    Default,
40}

42/// Applies the shared part of the `AbstractPiglin` constructor.
43///
44/// Vanilla parity: `setCanPickUpLoot(true)`, `applyOpenDoorsAbility()` and the
45/// two fire maluses -- the second of which is `-1.0`, meaning a piglin will not
46/// path into fire at any cost.
47pub fn apply_constructor<P: Mob + ?Sized>(piglin: &P) {
48    Mob::set_can_pick_up_loot(piglin, true);
49    piglin
50        .mob_base()
51        .navigation()
52        .lock()
53        .set_can_open_doors(true);
54    piglin.set_pathfinding_malus(PathType::FireInNeighbor, FIRE_IN_NEIGHBOR_MALUS);
55    piglin.set_pathfinding_malus(PathType::Fire, FIRE_MALUS);
56}

58/// Whether the main hand holds something that counts as a weapon.
59///
60/// Vanilla parity: `AbstractPiglin.isHoldingMeleeWeapon`, which asks for the
61/// `tool` component rather than a tag -- so a golden shovel counts.
62#[must_use]
63pub fn is_holding_melee_weapon(body: &dyn LivingEntity) -> bool {
64    body.get_item_in_hand(InteractionHand::MainHand)
65        .get(TOOL)
66        .is_some()
67}

69/// A piglin that can be converted and its clock.
70///
71/// The two mobs keep the same three pieces of state and run the same clock, so
72/// the clock takes them through this trait rather than being written twice.
73pub trait ConvertiblePiglin: Mob {
74    /// Vanilla parity: `AbstractPiglin.isConverting`.
75    fn is_converting(&self) -> bool;
76
77    /// Returns and advances `timeInOverworld`, or resets it to zero.
78    fn bump_time_in_overworld(&self, converting: bool) -> i32;
79
80    /// Vanilla parity: `AbstractPiglin.playConvertedSound`.
81    fn play_converted_sound(&self);
82
83    /// Vanilla parity: `AbstractPiglin.finishConversion`.
84    fn convert_to_zombified(&self);
85}
86
87/// Runs the overworld conversion clock.
88///
89/// Vanilla parity: `AbstractPiglin.customServerAiStep`. The converted sound is
90/// skipped on peaceful, which is the one difficulty check in the whole method.
91pub fn tick_conversion<P: ConvertiblePiglin + ?Sized>(piglin: &P) {
92    let converting = piglin.is_converting();
93    let time_in_overworld = piglin.bump_time_in_overworld(converting);
94    if time_in_overworld <= CONVERSION_TIME {
95        return;
96    }
97
98    if piglin
99        .level()
100        .is_some_and(|world| world.difficulty() != Difficulty::Peaceful)
101    {
102        piglin.play_converted_sound();
103    }
104    piglin.convert_to_zombified();
105}

107/// The NBT key a piglin's carried inventory is stored under.
108///
109/// Vanilla parity: `InventoryCarrier.TAG_INVENTORY`.
110const TAG_INVENTORY: &str = "Inventory";
111
112/// Writes the carried inventory.
113///
114/// Vanilla parity: `InventoryCarrier.writeInventoryToTag`.
115pub fn save_inventory(inventory: &SimpleContainer, nbt: &mut NbtCompound) {
116    let items: Vec<NbtCompound> = inventory
117        .items()
118        .iter()
119        .filter(|item| !item.is_empty())
120        .filter_map(|item| match item.to_nbt_tag_ref() {
121            NbtTag::Compound(compound) => Some(compound),
122            _ => None,
123        })
124        .collect();
125    nbt.insert(TAG_INVENTORY, NbtList::Compound(items));
126}
127
128/// Reads the carried inventory back.
129///
130/// Vanilla parity: `InventoryCarrier.readInventoryFromTag`.
131pub fn load_inventory(inventory: &mut SimpleContainer, nbt: BorrowedNbtCompoundView<'_, '_>) {
132    let Some(list) = nbt.list(TAG_INVENTORY).and_then(|list| list.compounds()) else {
133        return;
134    };
135    for compound in list {
136        let Some(mut item) = ItemStack::from_borrowed_compound(&compound) else {
137            continue;
138        };
139        inventory.add(&mut item);
140    }
