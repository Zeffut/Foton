32/// How far the evoker counts its own vexes.
33///
34/// Vanilla parity: the `range(16.0)` and `inflate(16.0)` of the summon goal.
35const VEX_COUNT_RANGE: f64 = 16.0;
36
37/// Upper bound on the roll the vex count is compared against.
38///
39/// Vanilla parity: the `random.nextInt(8) + 1 > vexes` of `canUse`, which is
40/// why an evoker with a full escort only rarely summons more.
41const VEX_ROLL_BOUND: i32 = 8;
42
43/// How many vexes one summon brings.
44///
45/// Vanilla parity: the three-iteration loop of `performSpellCasting`.
46const VEX_SUMMON_COUNT: i32 = 3;
47
48/// Half-width of the square a summoned vex appears in.
49///
50/// Vanilla parity: the two offsets of `performSpellCasting`, each a
51/// `-2 + nextInt(5)`.
52const VEX_SPAWN_SPREAD: i32 = 2;
53
54/// How far above the evoker a summoned vex appears.
55const VEX_SPAWN_HEIGHT: i32 = 1;
56
57/// Shortest a summoned vex lives, in seconds.
58///
59/// Vanilla parity: the `20 * (30 + nextInt(90))` limited life, which is what
60/// makes an evoker's escort run out rather than accumulate.
61const VEX_LIFE_MIN_SECONDS: i32 = 30;
62
63/// Width of the random part of a summoned vex's life, in seconds.
64const VEX_LIFE_SPREAD_SECONDS: i32 = 90;
65
66/// Ticks in a second.
67const TICKS_PER_SECOND: i32 = 20;
68
69/// Ticks the evoker keeps its hands up while summoning.
70const SUMMON_CASTING_TIME: i32 = 100;
71
72/// Ticks between two summons.
73const SUMMON_CASTING_INTERVAL: i32 = 340;
74
75/// Ticks the evoker keeps its hands up while calling fangs.
76const FANGS_CASTING_TIME: i32 = 40;
77
78/// Ticks between two fang lines.
79const FANGS_CASTING_INTERVAL: i32 = 100;
80
81/// Squared distance inside which the fangs come up in two rings instead of a
82/// line.
83///
84/// Vanilla parity: the `distanceToSqr(target) < 9.0` of `performSpellCasting`.
85const FANGS_RING_DISTANCE_SQR: f64 = 9.0;
86
87/// Fangs in the inner ring.
88const FANGS_INNER_RING_COUNT: i32 = 5;
89
90/// Radius of the inner ring.
91const FANGS_INNER_RING_RADIUS: f64 = 1.5;
92
93/// Fangs in the outer ring.
94const FANGS_OUTER_RING_COUNT: i32 = 8;
95
96/// Radius of the outer ring.
97const FANGS_OUTER_RING_RADIUS: f64 = 2.5;
98
99/// Warmup the outer ring rises with, so it follows the inner one.
100const FANGS_OUTER_RING_DELAY: i32 = 3;
101
102/// Fangs in a line.
103const FANGS_LINE_COUNT: i32 = 16;
104
105/// Spacing between two fangs in a line.
106const FANGS_LINE_SPACING: f64 = 1.25;
107
108/// Warmup the evoker takes before recoloring a sheep.
109const WOLOLO_CAST_WARMUP_TIME: i32 = 40;
110
111/// Ticks the evoker keeps its hands up while recoloring.
112const WOLOLO_CASTING_TIME: i32 = 60;
113
114/// Ticks between two recolorings.
115const WOLOLO_CASTING_INTERVAL: i32 = 140;
116
117/// How far the evoker looks for a blue sheep, horizontally.
118const WOLOLO_SEARCH_RANGE: f64 = 16.0;
119
120/// How far the evoker looks for a blue sheep, vertically.
121const WOLOLO_SEARCH_HEIGHT: f64 = 4.0;

123/// Summons vexes.
124///
125/// Vanilla parity: `Evoker.EvokerSummonSpellGoal`.
126///
127pub(super) struct EvokerSummonSpellGoal {
128    base: SpellcasterUseSpellBase,
129    /// Who counts as one of this evoker's vexes.
130    vex_count_targeting: TargetingConditions,
131}
132
133impl EvokerSummonSpellGoal {
134    /// Creates the goal.
135    #[must_use]
136    pub(super) fn new() -> Self {
137        Self {
138            base: SpellcasterUseSpellBase::new(
139                IllagerSpell::SummonVex,
140                DEFAULT_CAST_WARMUP_TIME,
141                SUMMON_CASTING_TIME,
142                SUMMON_CASTING_INTERVAL,
143                Some(&sound_events::ENTITY_EVOKER_PREPARE_SUMMON),
144            ),
145            vex_count_targeting: TargetingConditions::for_non_combat()
146                .range(VEX_COUNT_RANGE)
147                .ignore_line_of_sight()
148                .ignore_invisibility_testing(),
149        }
150    }
151
152    /// Counts the vexes already around the evoker.
153    ///
154    /// Vanilla parity: the `getNearbyEntities(Vex.class, ..)` of `canUse`.
155    fn nearby_vexes(&self, mob: &dyn PathfinderMob) -> usize {
156        let Some(world) = mob.level() else {
157            return 0;
158        };
159        let search_box = mob.bounding_box().inflate(VEX_COUNT_RANGE);
160        let level = world.as_ref();
161        world
162            .get_entities_in_aabb_matching(&search_box, |entity| {
163                entity.entity_type() == &vanilla_entities::VEX
164                    && entity.as_living_entity().is_some_and(|living| {
165                        self.vex_count_targeting.test(level, Some(mob), living)
166                    })
167            })
168            .len()
169    }
170}

191    }

195    }

198/// Puts three vexes in the air around the evoker.
199///
200/// Vanilla parity: `EvokerSummonSpellGoal.performSpellCasting`. Each vex is
201/// bound to the spot it appeared at, so the escort hangs over the evoker
202/// rather than scattering, and each is given thirty seconds to two minutes to
203/// live. Vanilla also copies the evoker's scoreboard team onto the vex; Foton
204/// puts no entity on a team.
205fn summon_vexes(mob: &dyn PathfinderMob) {
206    let Some(world) = mob.level() else {
207        return;
208    };
209    let origin = mob.block_position();
210
211    for _ in 0..VEX_SUMMON_COUNT {
212        let pos = origin.offset(
213            rand::random_range(-VEX_SPAWN_SPREAD..=VEX_SPAWN_SPREAD),
214            VEX_SPAWN_HEIGHT,
215            rand::random_range(-VEX_SPAWN_SPREAD..=VEX_SPAWN_SPREAD),
216        );
217        let (x, y, z) = pos.get_bottom_center();
218        let spawn = DVec3::new(x, y, z);
219
220        let vex = Arc::new(VexEntity::new(
221            &vanilla_entities::VEX,
222            next_entity_id(),
223            spawn,
224            Arc::downgrade(&world),
225        ));
226        vex.finalize_spawn(&world, EntitySpawnReason::MobSummoned, None);
227        vex.set_owner(mob.as_entity_event_source());
228        vex.set_bound_origin(Some(pos));
229        vex.set_limited_life(
230            TICKS_PER_SECOND
231                * (VEX_LIFE_MIN_SECONDS + rand::random_range(0..VEX_LIFE_SPREAD_SECONDS)),
232        );
233
234        let entity: SharedEntity = vex;
235        if let Err(error) = world.try_add_entity(entity) {
236            log::debug!("evoker failed to summon a vex: {error}");
237            continue;
238        }
239        world.game_event_at(
240            &vanilla_game_events::ENTITY_PLACE,
241            spawn,
242            &GameEventContext::new(Some(mob.as_entity_event_source()), None),
243        );
244    }
245}

292/// Lays out the fangs.
293///
294/// Vanilla parity: `EvokerAttackSpellGoal.performSpellCasting`. Close in, the
295/// fangs come up in two rings around the evoker; further out, in a line walking
296/// towards the target, each one three ticks behind the last.
297fn cast_fangs(mob: &dyn PathfinderMob) {
298    let Some(target) = mob.target() else {
299        return;
300    };
301    let Some(world) = mob.level() else {
302        return;
303    };
304
305    let position = mob.position();
306    let target_position = target.position();
307    let min_y = target_position.y.min(position.y);
308    let max_y = target_position.y.max(position.y) + 1.0;
309    let angle_to_target =
310        (target_position.z - position.z).atan2(target_position.x - position.x) as f32;
311
312    if position.distance_squared(target_position) < FANGS_RING_DISTANCE_SQR {
313        for i in 0..FANGS_INNER_RING_COUNT {
314            let angle = f64::from(i).mul_add(PI * 0.4, f64::from(angle_to_target)) as f32;
315            place_fangs(
316                mob,
317                &world,
318                ring_point(position, angle, FANGS_INNER_RING_RADIUS),
319                min_y,
320                max_y,
321                angle,
322                0,
323            );
324        }
325        for i in 0..FANGS_OUTER_RING_COUNT {
326            let angle =
327                f64::from(i).mul_add(TAU / 8.0, f64::from(angle_to_target) + TAU / 5.0) as f32;
328            place_fangs(
329                mob,
330                &world,
331                ring_point(position, angle, FANGS_OUTER_RING_RADIUS),
332                min_y,
333                max_y,
334                angle,
335                FANGS_OUTER_RING_DELAY,
336            );
337        }
338        return;
339    }
340
341    for i in 0..FANGS_LINE_COUNT {
342        let reach = FANGS_LINE_SPACING * f64::from(i + 1);
343        place_fangs(
344            mob,
345            &world,
346            ring_point(position, angle_to_target, reach),
347            min_y,
348            max_y,
349            angle_to_target,
350            i,
351        );
352    }
353}

355/// Returns the point `radius` away from `origin` at `angle`.
356fn ring_point(origin: DVec3, angle: f32, radius: f64) -> (f64, f64) {
357    let cos = f64::from(trig::cos(f64::from(angle)));
358    let sin = f64::from(trig::sin(f64::from(angle)));
359    (cos.mul_add(radius, origin.x), sin.mul_add(radius, origin.z))
360}

362/// Drops one pair of fangs onto the first solid surface below `max_y`.
363///
364/// Vanilla parity: `EvokerAttackSpellGoal.createSpellEntity`. Fangs only rise
365/// out of a face that could be stood on, which is why a fang line stops at the
366/// edge of a hole rather than continuing into it.
367fn place_fangs(
368    mob: &dyn PathfinderMob,
369    world: &Arc<World>,
370    (x, z): (f64, f64),
371    min_y: f64,
372    max_y: f64,
373    angle: f32,
374    delay_ticks: i32,
375) {
376    let mut pos = BlockPos::containing(x, max_y, z);
377    let mut top_offset = 0.0;
378    let mut found = false;
379    let floor = min_y.floor() as i32 - 1;
380
381    while pos.y() >= floor {
382        let below = pos.below();
383        let below_state = world.get_block_state(below);
384        if world.is_face_sturdy(below_state, below, Direction::Up) {
385            let state = world.get_block_state(pos);
386            if !state.is_air() {
387                let shape = state.get_collision_shape_at(pos);
388                if !shape.is_empty() {
389                    top_offset = shape.max(Axis::Y);
390                }
391            }
392            found = true;
393            break;
394        }
395        pos = pos.below();
396    }
397
398    if !found {
399        return;
400    }
401
402    let spawn = DVec3::new(x, f64::from(pos.y()) + top_offset, z);
403    let fangs = Arc::new(EvokerFangsEntity::new(
404        &vanilla_entities::EVOKER_FANGS,
405        next_entity_id(),
406        spawn,
407        Arc::downgrade(world),
408    ));
409    fangs.place(spawn, angle, delay_ticks);
410    fangs.set_owner_uuid(Some(mob.uuid()));
411
412    let entity: SharedEntity = fangs;
413    if let Err(error) = world.try_add_entity(entity) {
414        log::debug!("evoker failed to raise its fangs: {error}");
415        return;
416    }
417
418    world.game_event_at(
419        &vanilla_game_events::ENTITY_PLACE,
420        spawn,
421        &GameEventContext::new(Some(mob.as_entity_event_source()), None),
422    );
423}

425/// Turns a blue sheep red.
426///
427/// Vanilla parity: `Evoker.EvokerWololoSpellGoal`. The only spell an evoker
428/// casts with nothing to fight, and the only one whose target is not the thing
429/// it is attacking, which is why it overrides the whole gate.
430pub(super) struct EvokerWololoSpellGoal {
431    base: SpellcasterUseSpellBase,
432    /// Which sheep is worth recoloring.
433    wololo_targeting: TargetingConditions,
434}
435
436impl EvokerWololoSpellGoal {
437    /// Creates the goal.
438    #[must_use]
439    pub(super) fn new() -> Self {
440        Self {
441            base: SpellcasterUseSpellBase::new(
442                IllagerSpell::Wololo,
443                WOLOLO_CAST_WARMUP_TIME,
444                WOLOLO_CASTING_TIME,
445                WOLOLO_CASTING_INTERVAL,
446                Some(&sound_events::ENTITY_EVOKER_PREPARE_WOLOLO),
447            ),
448            wololo_targeting: TargetingConditions::for_non_combat()
449                .range(WOLOLO_SEARCH_RANGE)
450                .selector(|_, target, _| {
451                    target
452                        .downcast_ref::<SheepEntity>()
453                        .is_some_and(|sheep| sheep.color() == DyeColor::Blue)
454                }),
455        }
456    }
457}
458
459impl Goal for EvokerWololoSpellGoal {
460    fn controls(&self) -> GoalControls {
461        GoalControls::EMPTY
462    }
463
464    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
465        let Some(evoker) = mob.downcast_ref::<EvokerEntity>() else {
466            return false;
467        };
468        if mob.target().is_some() || evoker.is_casting_spell() || !self.base.is_off_cooldown(mob) {
469            return false;
470        }
471        let Some(world) = mob.level() else {
472            return false;
473        };
474        if !world.get_game_rule(&MOB_GRIEFING) {
475            return false;
476        }
477
478        let search_box = mob.bounding_box().inflate_xyz(
479            WOLOLO_SEARCH_RANGE,
480            WOLOLO_SEARCH_HEIGHT,
481            WOLOLO_SEARCH_RANGE,
482        );
483        let level = world.as_ref();
484        let sheep = world.get_entities_in_aabb_matching(&search_box, |entity| {
485            entity
486                .as_living_entity()
487                .is_some_and(|living| self.wololo_targeting.test(level, Some(mob), living))
488        });
489        if sheep.is_empty() {
490            return false;
491        }
492
493        let picked = &sheep[rand::random_range(0..sheep.len())];
494        evoker.set_wololo_target(Some(picked));
495        true
496    }
497
498    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
499        mob.downcast_ref::<EvokerEntity>()
500            .is_some_and(|evoker| evoker.wololo_target().is_some())
501            && self.base.attack_warmup_delay() > 0
502    }
503
504    fn start(&mut self, mob: &dyn PathfinderMob) {
505        self.base.start(mob);
506    }
507
508    fn stop(&mut self, mob: &dyn PathfinderMob) {
509        if let Some(evoker) = mob.downcast_ref::<EvokerEntity>() {
510            evoker.set_wololo_target(None);
511        }
512    }
513
514    fn tick(&mut self, mob: &dyn PathfinderMob) {
515        self.base.tick(mob, |mob| {
516            let Some(evoker) = mob.downcast_ref::<EvokerEntity>() else {
517                return;
518            };
519            let Some(target) = evoker.wololo_target() else {
520                return;
521            };
522            let Some(sheep) = target.downcast_ref::<SheepEntity>() else {
523                return;
524            };
525            if LivingEntity::is_alive(sheep) {
526                sheep.set_color(DyeColor::Red);
527            }
528        });
529    }
