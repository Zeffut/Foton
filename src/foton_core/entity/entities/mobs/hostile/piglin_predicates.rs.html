1//! The two piglin predicates the sensor and the brain both ask.
2//!
3//! Vanilla parity: `PiglinAi.isWearingSafeArmor` and
4//! `PiglinAi.isPlayerHoldingLovedItem`. They live one level above the piglin so
5//! that [`crate::entity::ai::brain::sensor`] can reach them without depending on
6//! the piglin's private module; vanilla has the same two callers.
7
8use foton_registry::equipment::EquipmentSlot;
9use foton_registry::vanilla_item_tags::ItemTag;
10use foton_registry::{REGISTRY, TaggedRegistryExt as _, vanilla_entities};
11
12use crate::entity::LivingEntity;
13use crate::entity::ai::brain::behavior::utils;
14
15/// Returns whether `entity` wears any piece of gold armor.
16///
17/// Vanilla parity: `PiglinAi.isWearingSafeArmor`. One piece is enough, which is
18/// why a single golden boot buys the whole truce.
19#[must_use]
20pub fn is_wearing_safe_armor(entity: &dyn LivingEntity) -> bool {
21    EquipmentSlot::ARMOR_SLOTS.into_iter().any(|slot| {
22        REGISTRY.items.is_in_tag(
23            entity.get_item_by_slot(slot).item(),
24            &ItemTag::PIGLIN_SAFE_ARMOR,
25        )
26    })
27}
28
29/// Returns whether `entity` is a player holding gold.
30///
31/// Vanilla parity: `PiglinAi.isPlayerHoldingLovedItem`, which is what makes a
32/// piglin stare at you when you take out an ingot.
33#[must_use]
34pub fn is_player_holding_loved_item(entity: &dyn LivingEntity) -> bool {
35    utils::is_of_type(entity.as_entity_event_source(), &vanilla_entities::PLAYER)
36        && entity.is_holding(&mut |item| {
37            REGISTRY
38                .items
39                .is_in_tag(item.item(), &ItemTag::PIGLIN_LOVED)
40        })
41}
