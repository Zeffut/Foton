1//! Phantom entity.
2//!
3//! Vanilla parity: `Phantom` and its four goals. A phantom never paths: it
4//! circles an anchor point high above whatever it is hunting, and every eight
5//! to twelve seconds it drops the circle and swoops straight through its
6//! target. Everything it does is written into one `moveTargetPoint` that
7//! `PhantomMoveControl` then flies at.

251    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
252        let living_base = LivingEntityBase::new(entity_type);
253        let mob_base = MobBase::new();
254        let mut entity_data = PhantomEntityData::new();
255        living_base.initialize_synced_data(&mut entity_data);
256        mob_base.set_xp_reward(XP_REWARD);
257
258        {
259            // Keep vanilla Phantom goal priorities in the same order.
260            let mut goals = mob_base.goal_selector().lock();
261            goals.add_goal(1, PhantomAttackStrategyGoal::new());

263            goals.add_goal(3, PhantomCircleAroundAnchorGoal::new());
264        }
265
266        {
267            let mut targets = mob_base.target_selector().lock();
268            targets.add_goal(1, PhantomAttackPlayerTargetGoal::new());
269        }
270
271        Self {
272            base,
273            entity_type,
274            living_base,
275            mob_base,
276            entity_data: SyncMutex::new(entity_data),
277            move_target_point: SyncMutex::new(DVec3::ZERO),
278            anchor_point: SyncMutex::new(None),
279            attack_phase: SyncMutex::new(AttackPhase::Circle),
280            flight_speed: SyncMutex::new(PHANTOM_INITIAL_SPEED),
281        }
282    }

327/// Hunts the highest player in range.
328///
329/// Vanilla parity: `Phantom.PhantomAttackPlayerTargetGoal`. Sorting by height
330/// is what makes a phantom pick the player on the roof rather than the one in
331/// the cellar.
332struct PhantomAttackPlayerTargetGoal {
333    targeting: TargetingConditions,
334    next_scan_tick: i32,
335}
336
337impl PhantomAttackPlayerTargetGoal {
338    const fn new() -> Self {
339        Self {
340            targeting: TargetingConditions::for_combat().range(TARGET_RANGE),
341            next_scan_tick: reduced_tick_delay(FIRST_SCAN_DELAY_TICKS),
342        }
343    }
344}
345
346impl Goal for PhantomAttackPlayerTargetGoal {
347    fn controls(&self) -> GoalControls {
348        GoalControls::TARGET
349    }
350
351    /// Vanilla parity: `PhantomAttackPlayerTargetGoal.canUse`.
352    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
353        if self.next_scan_tick > 0 {
354            self.next_scan_tick -= 1;
355            return false;
356        }
357        self.next_scan_tick = reduced_tick_delay(SCAN_INTERVAL_TICKS);
358
359        let Some(world) = mob.level() else {
360            return false;
361        };
362        let search_box = mob.bounding_box().inflate_xyz(
363            TARGET_SCAN_HORIZONTAL,
364            TARGET_SCAN_VERTICAL,
365            TARGET_SCAN_HORIZONTAL,
366        );
367        let level = world.as_ref();
368        let mut players: Vec<SharedEntity> =
369            world.get_entities_in_aabb_matching(&search_box, |entity| {
370                entity.as_player().is_some()
371                    && entity
372                        .as_living_entity()
373                        .is_some_and(|player| self.targeting.test(level, Some(mob), player))
374            });
375        if players.is_empty() {
376            return false;
377        }
378
379        // Vanilla sorts by descending Y and takes the first player that passes
380        // the default targeting conditions.
381        players.sort_by(|left, right| {
382            right
383                .position()
384                .y
385                .partial_cmp(&left.position().y)
386                .unwrap_or(Ordering::Equal)
387        });
388
389        let default_conditions = TargetingConditions::default();
390        for player in players {
391            let passes = player
392                .as_living_entity()
393                .is_some_and(|living| default_conditions.test(level, Some(mob), living));
394            if passes {
395                mob.set_target(Some(&player));
396                return true;
397            }
398        }
399
400        false
401    }
402
403    /// Vanilla parity: `PhantomAttackPlayerTargetGoal.canContinueToUse`.
404    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
405        let (Some(world), Some(target)) = (mob.level(), mob.target()) else {
406            return false;
407        };
408        let conditions = TargetingConditions::default();
409        target
410            .as_living_entity()
411            .is_some_and(|living| conditions.test(world.as_ref(), Some(mob), living))
412    }
413}

415/// Alternates between circling and swooping.
416///
417/// Vanilla parity: `Phantom.PhantomAttackStrategyGoal`. It holds no control at
418/// all -- its whole job is to flip the attack phase and move the anchor.
419struct PhantomAttackStrategyGoal {
420    next_sweep_tick: i32,
421}
422
423impl PhantomAttackStrategyGoal {
424    const fn new() -> Self {
425        Self { next_sweep_tick: 0 }
426    }
427
428    /// Moves the anchor high above whatever the phantom is hunting.
429    ///
430    /// Vanilla parity: `PhantomAttackStrategyGoal.setAnchorAboveTarget`, which
431    /// only moves an anchor that already exists.
432    fn set_anchor_above_target(mob: &dyn PathfinderMob) {
433        let (Some(phantom), Some(target), Some(world)) =
434            (phantom_of(mob), mob.target(), mob.level())
435        else {
436            return;
437        };
438        let mut anchor = phantom.anchor_point.lock();
439        if anchor.is_none() {
440            return;
441        }
442
443        let above = target
444            .block_position()
445            .above_n(ANCHOR_ABOVE_TARGET_MIN + rand::random_range(0..ANCHOR_ABOVE_TARGET_SPAN));
446        let sea_level = world.sea_level;
447        *anchor = Some(if above.y() < sea_level {
448            BlockPos::new(above.x(), sea_level + 1, above.z())
449        } else {
450            above
451        });
452    }
453}
454
455impl Goal for PhantomAttackStrategyGoal {
456    fn controls(&self) -> GoalControls {
457        GoalControls::EMPTY
458    }
459
460    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
461        let (Some(world), Some(target)) = (mob.level(), mob.target()) else {
462            return false;
463        };
464        let conditions = TargetingConditions::default();
465        target
466            .as_living_entity()
467            .is_some_and(|living| conditions.test(world.as_ref(), Some(mob), living))
468    }
469
470    fn start(&mut self, mob: &dyn PathfinderMob) {
471        self.next_sweep_tick = FIRST_SWEEP_DELAY_TICKS;
472        if let Some(phantom) = phantom_of(mob) {
473            *phantom.attack_phase.lock() = AttackPhase::Circle;
474        }
475        Self::set_anchor_above_target(mob);
476    }
477
478    /// Vanilla parity: `PhantomAttackStrategyGoal.stop`, which parks the anchor
479    /// well above the terrain so the phantom keeps circling after it gives up.
480    fn stop(&mut self, mob: &dyn PathfinderMob) {
481        let (Some(phantom), Some(world)) = (phantom_of(mob), mob.level()) else {
482            return;
483        };
484        let mut anchor = phantom.anchor_point.lock();
485        let Some(current) = *anchor else {
486            return;
487        };
488
489        let ground_y = world.heightmap_at(HeightmapType::MotionBlocking, current.x(), current.z());
490        *anchor = Some(
491            BlockPos::new(current.x(), ground_y, current.z())
492                .above_n(ANCHOR_ABOVE_GROUND_MIN + rand::random_range(0..ANCHOR_ABOVE_GROUND_SPAN)),
493        );
494    }
495
496    /// Vanilla parity: `PhantomAttackStrategyGoal.tick`.
497    fn tick(&mut self, mob: &dyn PathfinderMob) {
498        let Some(phantom) = phantom_of(mob) else {
499            return;
500        };
501        if *phantom.attack_phase.lock() != AttackPhase::Circle {
502            return;
503        }
504
505        self.next_sweep_tick -= 1;
506        if self.next_sweep_tick > 0 {
507            return;
508        }
509
510        *phantom.attack_phase.lock() = AttackPhase::Swoop;
511        Self::set_anchor_above_target(mob);
512        self.next_sweep_tick = (SWEEP_INTERVAL_MIN_SECONDS
513            + rand::random_range(0..SWEEP_INTERVAL_SPAN_SECONDS))
514            * TICKS_PER_SECOND;
515        mob.play_sound(
516            &sound_events::ENTITY_PHANTOM_SWOOP,
517            10.0,
518            0.1f32.mul_add(rand::random::<f32>(), 0.95),
519        );
520    }
521}

523/// Wheels around the anchor point.
524///
525/// Vanilla parity: `Phantom.PhantomCircleAroundAnchorGoal`.
526struct PhantomCircleAroundAnchorGoal {
527    angle: f32,
528    distance: f32,
529    height: f32,
530    clockwise: f32,
531}
532
533impl PhantomCircleAroundAnchorGoal {
534    const fn new() -> Self {
535        Self {
536            angle: 0.0,
537            distance: 0.0,
538            height: 0.0,
539            clockwise: 1.0,
540        }
541    }
542
543    /// Advances a step around the circle and writes the new point.
544    ///
545    /// Vanilla parity: `PhantomCircleAroundAnchorGoal.selectNext`.
546    fn select_next(&mut self, mob: &dyn PathfinderMob) {
547        let Some(phantom) = phantom_of(mob) else {
548            return;
549        };
550
551        let anchor = {
552            let mut anchor = phantom.anchor_point.lock();
553            *anchor.get_or_insert_with(|| mob.block_position())
554        };
555
556        self.angle += self.clockwise * CIRCLE_STEP_DEGREES.to_radians();
557        *phantom.move_target_point.lock() = DVec3::new(
558            f64::from(anchor.x()) + f64::from(self.distance * self.angle.cos()),
559            f64::from(anchor.y()) + f64::from(CIRCLE_HEIGHT_MIN + self.height),
560            f64::from(anchor.z()) + f64::from(self.distance * self.angle.sin()),
561        );
562    }
563}
564
565impl Goal for PhantomCircleAroundAnchorGoal {
566    /// Vanilla parity: `PhantomMoveTargetGoal` sets `Goal.Flag.MOVE`.
567    fn controls(&self) -> GoalControls {
568        GoalControls::MOVE
569    }
570
571    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
572        mob.target().is_none()
573            || phantom_of(mob)
574                .is_some_and(|phantom| *phantom.attack_phase.lock() == AttackPhase::Circle)
575    }
576
577    fn start(&mut self, mob: &dyn PathfinderMob) {
578        self.distance = rand::random::<f32>().mul_add(CIRCLE_RADIUS_SPAN, CIRCLE_RADIUS_MIN);
579        self.height = rand::random::<f32>().mul_add(CIRCLE_HEIGHT_SPAN, CIRCLE_HEIGHT_MIN);
580        self.clockwise = if rand::random::<bool>() { 1.0 } else { -1.0 };
581        self.select_next(mob);
582    }
583
584    /// Vanilla parity: `PhantomCircleAroundAnchorGoal.tick`.
585    fn tick(&mut self, mob: &dyn PathfinderMob) {
586        if rand::random_range(0..CIRCLE_HEIGHT_REROLL_TICKS) == 0 {
587            self.height = rand::random::<f32>().mul_add(CIRCLE_HEIGHT_SPAN, CIRCLE_HEIGHT_MIN);
588        }
589
590        if rand::random_range(0..CIRCLE_RADIUS_REROLL_TICKS) == 0 {
591            self.distance += 1.0;
592            if self.distance > CIRCLE_RADIUS_MAX {
593                self.distance = CIRCLE_RADIUS_MIN;
594                self.clockwise = -self.clockwise;
595            }
596        }
597
598        if rand::random_range(0..CIRCLE_ANGLE_REROLL_TICKS) == 0 {
599            self.angle = rand::random::<f32>() * TAU;
600            self.select_next(mob);
601        }
602
603        let Some(phantom) = phantom_of(mob) else {
604            return;
605        };
606        let Some(world) = mob.level() else {
607            return;
608        };
609
610        let position = mob.position();
611        let target_point = phantom.move_target_point();
612        if target_point.distance_squared(position) < TOUCHING_TARGET_SQR {
613            self.select_next(mob);
614        }
615
616        // Vanilla parity: the phantom refuses to aim into the ground or the
617        // ceiling, and flips the height offset the moment it would.
618        let block_position = mob.block_position();
619        let target_point = phantom.move_target_point();
620        if target_point.y < position.y && !world.get_block_state(block_position.below()).is_air() {
621            self.height = self.height.max(1.0);
622            self.select_next(mob);
623        }
624
625        let target_point = phantom.move_target_point();
626        if target_point.y > position.y && !world.get_block_state(block_position.above()).is_air() {
627            self.height = self.height.min(-1.0);
628            self.select_next(mob);
629        }
630    }
631}

633/// Dives straight through the target.
634///
635/// Vanilla parity: `Phantom.PhantomSweepAttackGoal`.
636struct PhantomSweepAttackGoal {
637    /// Whether a cat has scared the phantom off this swoop.
638    is_scared_of_cat: bool,
639    /// The tick the next cat scan is due on.
640    cat_search_tick: i32,
641}
642
643impl PhantomSweepAttackGoal {
644    const fn new() -> Self {
645        Self {
646            is_scared_of_cat: false,
647            cat_search_tick: 0,
648        }
649    }
650
651    /// Makes every living cat within sixteen blocks hiss, and reports whether
652    /// there was one.
653    ///
654    /// Vanilla parity: the cat scan of `PhantomSweepAttackGoal.canContinueToUse`.
655    fn hiss_at_nearby_cats(mob: &dyn PathfinderMob) -> bool {
656        let Some(world) = mob.level() else {
657            return false;
658        };
659
660        let search_box = mob.bounding_box().inflate(CAT_SEARCH_RANGE);
661        let cats = world.get_entities_in_aabb_matching(&search_box, |entity| {
662            entity.is_alive() && entity.downcast_ref::<CatEntity>().is_some()
663        });
664        for entity in &cats {
665            if let Some(cat) = entity.downcast_ref::<CatEntity>() {
666                cat.hiss();
667            }
668        }
669
670        !cats.is_empty()
671    }
672}
673
674impl Goal for PhantomSweepAttackGoal {
675    fn controls(&self) -> GoalControls {
676        GoalControls::MOVE
677    }
678
679    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
680        mob.target().is_some()
681            && phantom_of(mob)
682                .is_some_and(|phantom| *phantom.attack_phase.lock() == AttackPhase::Swoop)
683    }
684
685    /// Vanilla parity: `PhantomSweepAttackGoal.canContinueToUse`. A cat within
686    /// sixteen blocks calls the swoop off, which is the whole reason a cat by
687    /// the bed keeps phantoms away.
688    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
689        let Some(target) = mob.target() else {
690            return false;
691        };
692        let alive = target
693            .as_living_entity()
694            .is_some_and(LivingEntity::is_alive);
695        if !alive {
696            return false;
697        }
698        if let Some(player) = target.as_player()
699            && (player.is_spectator() || player.has_infinite_materials())
700        {
701            return false;
702        }
703        if !self.can_use(mob) {
704            return false;
705        }
706
707        if mob.tick_count() > self.cat_search_tick {
708            self.cat_search_tick = mob.tick_count() + CAT_SEARCH_TICK_DELAY;
709            self.is_scared_of_cat = Self::hiss_at_nearby_cats(mob);
710        }
711
712        !self.is_scared_of_cat
713    }
714
715    fn stop(&mut self, mob: &dyn PathfinderMob) {
716        mob.set_target(None);
717        if let Some(phantom) = phantom_of(mob) {
718            *phantom.attack_phase.lock() = AttackPhase::Circle;
719        }
720    }
721
722    /// Vanilla parity: `PhantomSweepAttackGoal.tick`.
723    fn tick(&mut self, mob: &dyn PathfinderMob) {
724        let (Some(phantom), Some(target)) = (phantom_of(mob), mob.target()) else {
725            return;
726        };
727
728        let target_position = target.position();
729        *phantom.move_target_point.lock() = DVec3::new(
730            target_position.x,
731            target_position.y + target.bounding_box().height() * 0.5,
732            target_position.z,
733        );
734
735        if mob
736            .bounding_box()
737            .inflate(SWEEP_HIT_INFLATE)
738            .intersects(target.bounding_box())
739        {
740            if let Some(world) = mob.level() {
741                let _ = mob.do_hurt_target(world.as_ref(), &target);
742                *phantom.attack_phase.lock() = AttackPhase::Circle;
743                if !mob.is_silent() {
744                    world.level_event(
745                        level_events::SOUND_PHANTOM_BITE,
746                        mob.block_position(),
747                        0,
748                        None,
749                    );
750                }
751            }
752            return;
753        }
754
755        // Vanilla also breaks the swoop off on `hurtTime > 0`, the red-flash
756        // timer. Foton models `invulnerableTime` but not `hurtTime`, so only
757        // the wall half of the check is here.
758        if mob.horizontal_collision() {
759            *phantom.attack_phase.lock() = AttackPhase::Circle;
760        }
761    }
762}
