46/// Experience this mob drops.
47///
48/// Vanilla parity: the `this.xpReward = 5` of the `Monster` constructor, which
49/// every monster inherits and this one does not override.
50const XP_REWARD: i32 = 5;

52/// Ticks between shots on Hard.

54/// Vanilla parity: `Bogged.getHardAttackInterval`.
55const HARD_ATTACK_INTERVAL_TICKS: i32 = 50;
56
57/// Ticks between shots on every other difficulty.
58///
59/// Vanilla parity: `Bogged.getAttackInterval`.
60const ATTACK_INTERVAL_TICKS: i32 = 70;

62/// Range within which a bogged will loose an arrow.
63///
64/// Vanilla parity: the `attackRadius` of `AbstractSkeleton`'s bow goal.
65const ATTACK_RADIUS: f64 = 15.0;
66
67/// Speed of the arrows a bogged fires.
68///
69/// Vanilla parity: the `1.6F` velocity of `performRangedAttack`.

73///
74/// Vanilla parity: `14 - difficulty * 4`, with difficulty 2.

78///
79/// Vanilla parity: the `MobEffectInstance(MobEffects.POISON, 100)` of

84const STROLL_SPEED_MODIFIER: f64 = 1.0;

93const LOOK_AT_PLAYER_RANGE: f64 = 8.0;

115impl BoggedEntity {
116    /// Creates a bogged at runtime.
117    #[must_use]
118    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
119        Self::new_with_base(
120            EntityBase::new(id, position, entity_type.dimensions, world),
121            entity_type,
122        )
123    }
124
125    /// Creates a bogged from saved base data.
126    #[must_use]
127    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
128        Self::new_with_base(
129            EntityBase::from_load(load, entity_type.dimensions),
130            entity_type,
131        )
132    }
133
134    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
135        let living_base = LivingEntityBase::new(entity_type);
136        let mob_base = MobBase::new();
137        mob_base.set_xp_reward(XP_REWARD);
138        let mut entity_data = BoggedEntityData::new();
139        living_base.initialize_synced_data(&mut entity_data);
140
141        {
142            // A bogged keeps the AbstractSkeleton goal set unchanged.
143            let mut goals = mob_base.goal_selector().lock();
144            goals.add_goal(2, RestrictSunGoal::new());
145            goals.add_goal(3, FleeSunGoal::new(1.0));
146            goals.add_goal(
147                4,
148                RangedBowAttackGoal::by_difficulty(
149                    HARD_ATTACK_INTERVAL_TICKS,
150                    ATTACK_INTERVAL_TICKS,
151                    ATTACK_RADIUS,
152                    BOW_APPROACH_SPEED,
153                    fire_arrow,
154                ),
155            );
156            goals.add_goal(5, WaterAvoidingRandomStrollGoal::new(STROLL_SPEED_MODIFIER));
157            goals.add_goal(6, LookAtPlayerGoal::new(LOOK_AT_PLAYER_RANGE));
158            goals.add_goal(6, RandomLookAroundGoal::new());
159            // TODO: vanilla also flees wolves at priority 3.
160        }
161
162        {
163            let mut targets = mob_base.target_selector().lock();
164            targets.add_goal(1, HurtByTargetGoal::new());
165            targets.add_goal(
166                2,
167                NearestAttackableTargetGoal::new_for_players(true, |_, _, _| true),
168            );
169        }
170
171        Self {
172            base,
173            entity_type,
174            living_base,
175            mob_base,
176            entity_data: SyncMutex::new(entity_data),
177        }
178    }
179
180    /// Returns whether the mushrooms have already been taken.
181    ///
182    /// Vanilla parity: `Bogged.isSheared`.
183    #[must_use]
184    pub fn is_sheared(&self) -> bool {
185        *self.entity_data.lock().bogged().sheared.get()
186    }
187
188    /// Sets the shear flag.
189    ///
190    /// Vanilla parity: `Bogged.setSheared`.
191    pub fn set_sheared(&self, sheared: bool) {
192        self.entity_data.lock().bogged_mut().sheared.set(sheared);
193    }
194
195    /// Returns whether shears would take anything off this bogged.
196    ///
197    /// Vanilla parity: `Bogged.readyForShearing`. A bogged only grows its
198    /// mushrooms once, so a sheared one is worth nothing to a farm.
199    #[must_use]
200    pub fn ready_for_shearing(&self) -> bool {
201        !self.is_sheared()
202    }
203
204    /// Takes the mushrooms off and drops them.
205    ///
206    /// Vanilla parity: `Bogged.shear` and `Bogged.spawnShearedMushrooms`.
207    pub fn shear(&self, world: &Arc<World>, sound_source: SoundSource, tool: &ItemStack) {
208        world.play_sound_at(
209            &sound_events::ENTITY_BOGGED_SHEAR,
210            sound_source,
211            self.position(),
212            1.0,
213            1.0,
214            None,
215        );
216
217        let mut rng = rand::rng();
218        // Vanilla drops from the mob's full height rather than its eye height.
219        let drop_height = f64::from(self.base.dimensions().height);
220        for dropped in shearing_loot_items_with_rng(
221            self,
222            &vanilla_loot_tables::SHEARING_BOGGED,
223            tool,
224            &mut rng,
225        ) {
226            let _ = self.spawn_at_location(dropped, drop_height);
227        }
228
229        self.set_sheared(true);
230    }
231}

233/// Looses a poisoned arrow at `target`.
234///
235/// Vanilla parity: `AbstractSkeleton.performRangedAttack` with the
236/// `Bogged.getArrow` override.
237fn fire_arrow(mob: &dyn PathfinderMob, target: DVec3, power: f32) {
238    let Some(archer) = mob.downcast_ref::<BoggedEntity>() else {
239        return;
240    };
241    let Some(world) = archer.level() else {
242        return;
243    };
244    // Vanilla parity: `AbstractSkeleton.performRangedAttack` reads the bow out
245    // of whichever hand holds one and hands it to `ProjectileUtil.getMobArrow`,
246    // so the arrow can read Power and Flame off it when it lands. Nothing
247    // leaves a quiver: `Monster.getProjectile` conjures the arrow when the mob
248    // carries none, which is why a skeleton never runs out.
249    let bow = archer.get_item_in_hand(weapon_holding_hand(archer, &vanilla_items::BOW));
250    let arrow = ArrowEntity::shoot_at(&world, archer, target, ARROW_POWER, ARROW_UNCERTAINTY);
251    // Vanilla parity: the `ProjectileUtil.getMobArrow(.., power, ..)`
252    // of `performRangedAttack` -- a shallower draw hits softer.
253    arrow.set_base_damage_from_mob(power);
254    if bow.is(&vanilla_items::BOW) {
255        arrow.set_fired_from_weapon(Some(bow));
256    }
257    // Vanilla parity: `Bogged.getArrow` poisons every arrow it fires.
258    arrow.add_effect(MobEffectInstance::with_duration(
259        vanilla_mob_effects::POISON,
260        POISON_TICKS,
261        0,
262    ));
263
264    world.play_sound_at(
265        &sound_events::ENTITY_SKELETON_SHOOT,
266        SoundSource::Hostile,
267        archer.position(),
268        1.0,
269        0.4f32.mul_add(rand::random::<f32>(), 0.8).recip(),
270        None,
271    );
272}

347impl Mob for BoggedEntity {
348    /// Arms the skeleton with the bow it shoots with.
349    ///
350    /// Vanilla parity: `AbstractSkeleton.finalizeSpawn`, which runs the shared
351    /// `Mob.finalizeSpawn` and then `populateDefaultEquipmentSlots` -- and that
352    /// is the only place a skeleton's bow comes from.
353    fn finalize_spawn(
354        &self,
355        world: &Arc<World>,
356        spawn_reason: EntitySpawnReason,
357        group_data: Option<SpawnGroupData>,
358    ) -> Option<SpawnGroupData> {
359        let group_data = self.finalize_spawn_mob_base(world, spawn_reason, group_data);
360        // Vanilla parity: the `setCanPickUpLoot` of
361        // `AbstractSkeleton.finalizeSpawn`, which is what lets a skeleton pick
362        // your bow up off the ground.
363        self.roll_spawn_can_pick_up_loot(world);
364        self.set_item_in_hand(
365            InteractionHand::MainHand,
366            ItemStack::new(&vanilla_items::BOW),
367        );
368        group_data
369    }
370    /// Vanilla parity: `Bogged` derives from `AbstractSkeleton`, and so from
371    /// `Monster`.
372    fn is_monster(&self) -> bool {
373        true
374    }

376    /// Returns whether this mob accepts where the spawner put it.
377    ///
378    /// Vanilla parity: the `Monster::checkMonsterSpawnRules` a bogged is

381        &self,
382        world: &Arc<World>,

391    }

394        PathfinderMob::tick_pathfinder_goal_selectors(self);
395    }

397    fn tick_path_navigation(&self) {
398        PathfinderMob::tick_pathfinder_path_navigation(self);
399    }
400
401    fn ambient_sound(&self) -> Option<SoundEventRef> {
402        Some(&sound_events::ENTITY_BOGGED_AMBIENT)
403    }
404
405    /// Shears the mushrooms off.
406    ///
407    /// Vanilla parity: `Bogged.mobInteract`.
408    fn mob_interact(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
409        let item_stack = {
410            let inventory = player.inventory.lock();
411            let item_stack = inventory.get_item_in_hand(hand);
412            item_stack.copy_with_count(item_stack.count())
413        };
414        if !item_stack.is(&vanilla_items::SHEARS) || !self.ready_for_shearing() {
415            return InteractionResult::Pass;
416        }
417
418        let Some(world) = self.level() else {
419            return InteractionResult::Pass;
420        };
421        // Vanilla passes `SoundSource.PLAYERS` here rather than the mob's own
422        // source, because the shears are the player's.
423        self.shear(&world, SoundSource::Players, &item_stack);
424        world.game_event_at(
425            &vanilla_game_events::SHEAR,
426            self.position(),
427            &GameEventContext::new(Some(player as &dyn Entity), None),
428        );
429        player.hurt_item_in_hand(hand, 1);
430
431        InteractionResult::Success
432    }
433
434    fn mob_flags(&self) -> i8 {
435        *self.entity_data.lock().mob().mob_flags.get()
436    }
437
438    fn set_mob_flags(&self, flags: i8) {
439        self.entity_data.lock().mob_mut().mob_flags.set(flags);
440    }
441}

448mod tests {

451    use foton_registry::{init_vanilla_registry, vanilla_entities};
452    use simdnbt::borrow::read_compound as read_borrowed_compound;

457    fn bogged() -> BoggedEntity {
458        init_vanilla_registry();
459        BoggedEntity::new(
460            &vanilla_entities::BOGGED,
461            next_entity_id(),
462            DVec3::new(8.5, 64.0, 8.5),
463            Weak::new(),
464        )

467    fn reload(nbt: &NbtCompound) -> BoggedEntity {
468        let mut bytes = Vec::new();
469        nbt.write(&mut bytes);
470        let borrowed = read_borrowed_compound(&mut Cursor::new(&bytes))
471            .unwrap_or_else(|error| panic!("test nbt should reborrow: {error}"));
472        let mob = bogged();
473        mob.load_additional((&borrowed).into());
474        mob
475    }
476
477    /// The shear flag has to survive a save, or a farm could shear the same
478    /// bogged again after every chunk reload.
479    #[test]
480    fn a_sheared_bogged_stays_sheared_across_a_save() {
481        let mob = bogged();
482        assert!(mob.ready_for_shearing());
483
484        mob.set_sheared(true);
485        assert!(!mob.ready_for_shearing());
486
487        let mut nbt = NbtCompound::new();
488        mob.save_additional(&mut nbt);
489        assert_eq!(nbt.byte("sheared"), Some(1));
490        assert!(reload(&nbt).is_sheared());
491    }
492
493    /// Vanilla's `getBooleanOr("sheared", false)` means an entity written
494    /// before the flag existed comes back unsheared, not sheared.
495    #[test]
496    fn a_bogged_saved_without_the_flag_comes_back_unsheared() {
497        assert!(!reload(&NbtCompound::new()).is_sheared());
498    }
