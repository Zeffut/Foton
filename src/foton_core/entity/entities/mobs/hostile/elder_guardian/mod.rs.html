2//!
3//! Vanilla parity: `ElderGuardian`. It is a guardian with a bigger beam, a
4//! slower wander and one thing entirely its own: every minute it lays mining
5//! fatigue on every player within fifty blocks, which is what an ocean monument
6//! feels like from the inside.

40/// Ticks an elder guardian's beam charges for.
41///
42/// Vanilla parity: `ElderGuardian.getAttackDuration`, three quarters of the
43/// ordinary guardian's.
44const ATTACK_DURATION: i32 = 60;
45
46/// Ticks between two rounds of mining fatigue.
47///
48/// Vanilla parity: `ElderGuardian.EFFECT_INTERVAL`.
49const EFFECT_INTERVAL: i32 = 1200;
50
51/// How far the mining fatigue reaches, in blocks.
52///
53/// Vanilla parity: `ElderGuardian.EFFECT_RADIUS`.
54const EFFECT_RADIUS: f64 = 50.0;
55
56/// How long the mining fatigue lasts, in ticks.
57///
58/// Vanilla parity: `ElderGuardian.EFFECT_DURATION`.
59const EFFECT_DURATION: i32 = 6000;
60
61/// Strength of the mining fatigue.
62///
63/// Vanilla parity: `ElderGuardian.EFFECT_AMPLIFIER`, which is mining fatigue
64/// III.
65const EFFECT_AMPLIFIER: i32 = 2;
66
67/// How little of the effect may be left before it is laid on again.
68///
69/// Vanilla parity: `ElderGuardian.EFFECT_DISPLAY_LIMIT`, the `displayEffectLimit`
70/// of `MobEffectUtil.addEffectToPlayersAround`.
71const EFFECT_DISPLAY_LIMIT: i32 = 1200;
72
73/// Ticks between two wander rolls.
74///
75/// Vanilla parity: the `this.randomStrollGoal.setInterval(400)` of the
76/// constructor.
77const STROLL_INTERVAL_TICKS: i32 = 400;
78
79/// How far from where it woke up an elder guardian will stray.
80///
81/// Vanilla parity: the `setHomeTo(this.blockPosition(), 16)` of
82/// `ElderGuardian.customServerAiStep`.
83const HOME_RADIUS: i32 = 16;

101impl ElderGuardianEntity {
102    /// Creates an elder guardian at runtime.
103    #[must_use]
104    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
105        Self::new_with_base(
106            EntityBase::new(id, position, entity_type.dimensions, world),
107            entity_type,
108        )
109    }
110
111    /// Creates an elder guardian from saved base data.
112    #[must_use]
113    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
114        Self::new_with_base(
115            EntityBase::from_load(load, entity_type.dimensions),
116            entity_type,
117        )
118    }
119
120    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
121        let living_base = LivingEntityBase::new(entity_type);
122        let mob_base = MobBase::new();
123        let mut entity_data = GuardianEntityData::new();
124        living_base.initialize_synced_data(&mut entity_data);
125        mob_base.set_xp_reward(XP_REWARD);
126        // Vanilla parity: the `setPersistenceRequired()` of the constructor.
127        // Monuments would empty out otherwise.
128        *mob_base.persistence_required().lock() = true;
129
130        mob_base
131            .pathfinding_malus()
132            .lock()
133            .set(PathType::Water, 0.0);
134
135        guardian_common::register_goals::<Self>(&mob_base, STROLL_INTERVAL_TICKS);
136
137        Self {
138            base,
139            entity_type,
140            living_base,
141            mob_base,
142            entity_data: SyncMutex::new(entity_data),
143            state: SyncMutex::new(GuardianState::default()),
144        }
145    }
146
147    /// Lays mining fatigue on every player in range, once a minute.
148    ///
149    /// Vanilla parity: `ElderGuardian.customServerAiStep` together with
150    /// `MobEffectUtil.addEffectToPlayersAround`. A player who already has the
151    /// effect at full strength and plenty of time left is skipped, so the
152    /// screen effect only replays when it is about to matter.
153    fn apply_mining_fatigue(&self) {
154        let Some(world) = self.level() else {
155            return;
156        };
157        let position = self.position();
158        let silent = self.is_silent();
159        let radius_sqr = EFFECT_RADIUS * EFFECT_RADIUS;
160
161        world.players.iter_players(|_uuid, player| {
162            if !matches!(player.game_mode(), GameType::Survival | GameType::Adventure) {
163                return true;
164            }
165            if self.is_allied_to(player.as_ref()) {
166                return true;
167            }
168            if position.distance_squared(player.position()) > radius_sqr {
169                return true;
170            }
171
172            let needs_effect = player
173                .mob_effect(vanilla_mob_effects::MINING_FATIGUE)
174                .is_none_or(|active| {
175                    active.amplifier() < EFFECT_AMPLIFIER
176                        || !active.is_infinite_duration()
177                            && active.duration() < EFFECT_DISPLAY_LIMIT
178                });
179            if !needs_effect {
180                return true;
181            }
182
183            player.add_mob_effect(MobEffectInstance::with_duration(
184                vanilla_mob_effects::MINING_FATIGUE,
185                EFFECT_DURATION,
186                EFFECT_AMPLIFIER,
187            ));
188            player.send_packet(CGameEvent {
189                event: GameEventType::GuardianElderEffect,
190                data: if silent { 0.0 } else { 1.0 },
191            });
192
193            true
194        });
195    }
196}
