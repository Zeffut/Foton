46/// Experience this mob drops.
47///
48/// Vanilla parity: the `this.xpReward = 5` of the `Monster` constructor, which
49/// every monster inherits and this one does not override.
50const XP_REWARD: i32 = 5;
51
52/// Speed multiplier while chasing.
53///
54/// Vanilla parity: `new MeleeAttackGoal(this, 1.0, false)`.
55const ATTACK_SPEED_MODIFIER: f64 = 1.0;
56
57/// Speed multiplier while wandering.
58///
59/// Vanilla parity: `SilverfishMergeWithStoneGoal`'s
60/// `super(silverfish, 1.0, 10)` call into `RandomStrollGoal`. Silverfish
61/// wander noticeably faster than most mobs, which use `0.8`.
62const MERGE_STROLL_SPEED_MODIFIER: f64 = 1.0;
63
64/// Ticks between two attempts to pick a new wander target.
65///
66/// Vanilla parity: the `10` of `super(silverfish, 1.0, 10)`, twelve times more
67/// eager than the `120`-tick default, which is why a silverfish never stands
68/// still for long.
69const MERGE_STROLL_INTERVAL_TICKS: i32 = 10;
70
71/// One attempt in this many ticks to burrow instead of wander.
72///
73/// Vanilla parity: the `reducedTickDelay(10)` bound of
74/// `SilverfishMergeWithStoneGoal.canUse`.
75const MERGE_ATTEMPT_INTERVAL_TICKS: i32 = 10;
76
77/// Ticks a silverfish waits after being hurt before it looks for a nearby
78/// infested block to wake up.
79///
80/// Vanilla parity: `SilverfishWakeUpFriendsGoal.notifyHurt`'s
81/// `adjustedTickDelay(20)`.
82const WAKE_UP_FRIENDS_DELAY_TICKS: i32 = 20;
83
84/// Walk-target value returned for a position with a compatible host block
85/// directly below it, pulling wandering silverfish toward stone they could
86/// (in vanilla) merge into.
87///
88/// Vanilla parity: the `10.0F` literal in `Silverfish.getWalkTargetValue`.
89const WALK_TARGET_HOST_BLOCK_VALUE: f32 = 10.0;

92///
93/// Vanilla parity: the `0.15F` literal in `Silverfish.playStepSound`.

101/// A silverfish.
102///
103/// TODO: vanilla spawns silverfish through `Silverfish.checkSilverfishSpawnRules`
104/// (`checkAnyLightMonsterSpawnRules`, then "no player within 5 blocks" unless
105/// spawner-spawned), registered externally as this entity type's natural-spawn
106/// placement predicate. Foton's natural spawner
107/// (`foton-core/src/world/natural_spawn.rs`) has no per-entity-type
108/// spawn-rule hook to register that against yet, so it is not implemented
109/// here.
110#[entity_behavior(class = "Silverfish")]
111pub struct SilverfishEntity {
112    base: EntityBase,
113    entity_type: EntityTypeRef,
114    living_base: LivingEntityBase,
115    mob_base: MobBase,
116    entity_data: SyncMutex<SilverfishEntityData>,
117    /// Ticks left before the wake-up-friends goal acts, or `0` when idle.
118    ///
119    /// Vanilla parity: `SilverfishWakeUpFriendsGoal.lookForFriends`.
120    look_for_friends: SyncMutex<i32>,
121}

128impl SilverfishEntity {
129    /// Creates a silverfish at runtime.
130    #[must_use]
131    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
132        Self::new_with_base(
133            EntityBase::new(id, position, entity_type.dimensions, world),
134            entity_type,
135        )
136    }
137
138    /// Creates a silverfish from saved base data.
139    #[must_use]
140    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
141        Self::new_with_base(
142            EntityBase::from_load(load, entity_type.dimensions),
143            entity_type,
144        )
145    }
146
147    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
148        let living_base = LivingEntityBase::new(entity_type);
149        let mob_base = MobBase::new();
150        mob_base.set_xp_reward(XP_REWARD);
151        let mut entity_data = SilverfishEntityData::new();
152        living_base.initialize_synced_data(&mut entity_data);
153
154        {
155            // Keep vanilla Silverfish goal priorities in the same order.
156            let mut goals = mob_base.goal_selector().lock();
157            goals.add_goal(1, FloatGoal::new(&mob_base));
158            goals.add_goal(1, ClimbOnTopOfPowderSnowGoal::new());
159            goals.add_goal(3, SilverfishWakeUpFriendsGoal);
160            goals.add_goal(4, MeleeAttackGoal::new(ATTACK_SPEED_MODIFIER, false));
161            goals.add_goal(
162                5,
163                SilverfishMergeWithStoneGoal::new(MERGE_STROLL_SPEED_MODIFIER),
164            );
165        }
166
167        {
168            let mut targets = mob_base.target_selector().lock();
169            targets.add_goal(1, HurtByTargetGoal::new().set_alert_others([]));
170            targets.add_goal(
171                2,
172                NearestAttackableTargetGoal::new_for_players(true, |_, _, _| true),
173            );
174        }
175
176        Self {
177            base,
178            entity_type,
179            living_base,
180            mob_base,
181            entity_data: SyncMutex::new(entity_data),
182            look_for_friends: SyncMutex::new(0),
183        }
184    }
185
186    /// Arms the wake-up-friends countdown if it is not already running.
187    ///
188    /// Vanilla parity: `SilverfishWakeUpFriendsGoal.notifyHurt`, called from
189    /// `Silverfish.hurtServer` whenever a real attacker (or a damage type
190    /// tagged `always_triggers_silverfish`, e.g. potions) hurts this
191    /// silverfish. See [`LivingEntity::before_actually_hurt`] below for where
192    /// Foton calls this.
193    fn notify_friends_hurt(&self) {
194        let mut counter = self.look_for_friends.lock();
195        if *counter == 0 {
196            *counter = WAKE_UP_FRIENDS_DELAY_TICKS;
197        }
198    }
199}

201/// Wanders, and now and then disappears into a block of stone instead.
202///
203/// Vanilla parity: `Silverfish.SilverfishMergeWithStoneGoal`, a `RandomStrollGoal`
204/// that first rolls a one-in-ten chance to check a random neighboring block and
205/// burrow into it, which is how a silverfish vanishes into a wall.
206struct SilverfishMergeWithStoneGoal {
207    stroll: RandomStrollGoal,
208    /// The neighbor picked for this attempt.
209    selected_direction: Option<Direction>,
210    /// Whether this run burrows instead of wandering.
211    do_merge: bool,
212}
213
214impl SilverfishMergeWithStoneGoal {
215    const fn new(speed_modifier: f64) -> Self {
216        Self {
217            stroll: RandomStrollGoal::with_interval(speed_modifier, MERGE_STROLL_INTERVAL_TICKS),
218            selected_direction: None,
219            do_merge: false,
220        }
221    }
222
223    /// Returns the block the silverfish would burrow into this attempt.
224    ///
225    /// Vanilla parity: the `BlockPos.containing(x, y + 0.5, z).relative(dir)` of
226    /// both `canUse` and `start`, which aims at the silverfish's own middle
227    /// rather than the block it stands on.
228    fn merge_target(mob: &dyn PathfinderMob, direction: Direction) -> BlockPos {
229        let position = mob.position();
230        BlockPos::from(position.with_y(position.y + 0.5)).relative(direction)
231    }
232}
233
234impl Goal for SilverfishMergeWithStoneGoal {
235    fn controls(&self) -> GoalControls {
236        GoalControls::MOVE
237    }
238
239    /// Vanilla parity: `SilverfishMergeWithStoneGoal.canUse`.
240    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
241        if mob.target().is_some() || !mob.mob_base().navigation().lock().is_done() {
242            return false;
243        }
244
245        if let Some(world) = mob.level()
246            && world.get_game_rule(&MOB_GRIEFING)
247            && rand::random_range(0..reduced_tick_delay(MERGE_ATTEMPT_INTERVAL_TICKS)) == 0
248        {
249            let direction = Direction::random();
250            self.selected_direction = Some(direction);
251            let target = Self::merge_target(mob, direction);
252            if is_compatible_host_block(world.get_block_state(target)) {
253                self.do_merge = true;
254                return true;
255            }
256        }
257
258        self.do_merge = false;
259        self.stroll.can_use(mob)
260    }
261
262    /// Vanilla parity: `SilverfishMergeWithStoneGoal.canContinueToUse`. A burrow
263    /// finishes the moment it starts, so it never continues.
264    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
265        !self.do_merge && self.stroll.can_continue_to_use(mob)
266    }
267
268    /// Vanilla parity: `SilverfishMergeWithStoneGoal.start`.
269    fn start(&mut self, mob: &dyn PathfinderMob) {
270        if !self.do_merge {
271            self.stroll.start(mob);
272            return;
273        }
274
275        let (Some(world), Some(direction)) = (mob.level(), self.selected_direction) else {
276            return;
277        };
278        let target = Self::merge_target(mob, direction);
279        let Some(infested) = infested_state_by_host(world.get_block_state(target)) else {
280            return;
281        };
282
283        world.set_block(target, infested, UpdateFlags::UPDATE_ALL);
284        // Vanilla parity: `spawnAnim`, the same puff of particles the silverfish
285        // makes coming out, played on the way in.
286        mob.broadcast_entity_event(EntityStatus::Poof);
287        mob.set_removed(RemovalReason::Discarded);
288    }
289
290    fn stop(&mut self, mob: &dyn PathfinderMob) {
291        self.stroll.stop(mob);
292    }
293}

295/// Counts down after the silverfish is hurt, then calls the walls for help.
296///
297/// Vanilla parity: `Silverfish.SilverfishWakeUpFriendsGoal`. This is what turns
298/// one silverfish in a stronghold into a swarm.
299struct SilverfishWakeUpFriendsGoal;
300
301/// Vertical reach of the search for infested blocks, in blocks.
302///
303/// Vanilla parity: the `5` bound of the outer loop in
304/// `SilverfishWakeUpFriendsGoal.tick`.
305const WAKE_UP_VERTICAL_RANGE: i32 = 5;
306
307/// Horizontal reach of the search for infested blocks, in blocks.
308///
309/// Vanilla parity: the `10` bound of the two inner loops.
310const WAKE_UP_HORIZONTAL_RANGE: i32 = 10;
311
312/// Walks outward from zero the way vanilla's loops do: `0, 1, -1, 2, -2, ...`.
313///
314/// Vanilla writes this as `for (int i = 0; i <= n && i >= -n; i = (i <= 0 ? 1 : 0) - i)`,
315/// which searches nearest-first so a silverfish wakes the closest wall.
316fn alternating_offsets(range: i32) -> impl Iterator<Item = i32> {
317    (0..=range).flat_map(move |step| {
318        if step == 0 {
319            vec![0]
320        } else {
321            vec![step, -step]
322        }
323    })
324}
325
326impl Goal for SilverfishWakeUpFriendsGoal {
327    fn controls(&self) -> GoalControls {
328        GoalControls::EMPTY
329    }
330
331    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
332        mob.downcast_ref::<SilverfishEntity>()
333            .is_some_and(|silverfish| *silverfish.look_for_friends.lock() > 0)
334    }
335
336    /// Vanilla parity: `SilverfishWakeUpFriendsGoal.tick`.
337    fn tick(&mut self, mob: &dyn PathfinderMob) {
338        let Some(silverfish) = mob.downcast_ref::<SilverfishEntity>() else {
339            return;
340        };
341
342        let elapsed = {
343            let mut counter = silverfish.look_for_friends.lock();
344            *counter -= 1;
345            *counter <= 0
346        };
347        if !elapsed {
348            return;
349        }
350        *silverfish.look_for_friends.lock() = 0;
351
352        let Some(world) = mob.level() else {
353            return;
354        };
355        let griefing = world.get_game_rule(&MOB_GRIEFING);
356        let origin = mob.block_position();
357
358        // Vanilla nests the loops Y, X, Z and stops on a coin flip after every
359        // block it wakes, so a hurt silverfish usually frees one or two friends
360        // rather than the whole wall at once.
361        for y in alternating_offsets(WAKE_UP_VERTICAL_RANGE) {
362            for x in alternating_offsets(WAKE_UP_HORIZONTAL_RANGE) {
363                for z in alternating_offsets(WAKE_UP_HORIZONTAL_RANGE) {
364                    let pos = origin.offset(x, y, z);
365                    let Some(host) = host_state_by_infested(world.get_block_state(pos)) else {
366                        continue;
367                    };
368
369                    if griefing {
370                        world.destroy_block_by_entity(pos, true, mob.as_entity_event_source());
371                    } else {
372                        world.set_block(pos, host, UpdateFlags::UPDATE_ALL);
373                    }
374
375                    if rand::random::<bool>() {
376                        return;
377                    }
378                }
379            }
380        }
381    }
382}

552impl PathfinderMob for SilverfishEntity {
553    /// Vanilla parity: `Silverfish.getWalkTargetValue`. Pulls wandering
554    /// silverfish toward a compatible host block so they linger near stone
555    /// they could (in vanilla) merge into.
556    ///
557    /// TODO: vanilla falls back to `super.getWalkTargetValue` (`Mob`'s
558    /// darkness-based formula) when the block below is not a host block.
559    /// Foton's `PathfinderMob::get_walk_target_value` default only
560    /// implements that formula for `Animal`s (see `animal_walk_target_value`)
561    /// and returns `0.0` for every other mob, so this falls back to `0.0`
562    /// too instead of vanilla's formula.
563    fn get_walk_target_value(&self, pos: BlockPos) -> f32 {
564        let Some(world) = self.level() else {
565            return 0.0;
566        };
567
568        if is_compatible_host_block(world.get_block_state(pos.below())) {
569            WALK_TARGET_HOST_BLOCK_VALUE
570        } else {
571            0.0
572        }
573    }
574}
