1//! The creaking's brain.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.monster.creaking.CreakingAi`.
4//!
5//! Two things here are unlike every other brain mob. The fight activity's melee
6//! attack is gated on `canMove`, so a creaking a player is staring at stops
7//! mid-swing; and `updateActivity` forces the *default* activity while frozen
8//! rather than picking the first valid one, which is what keeps a frozen
9//! creaking from quietly re-targeting.
10

28/// Vanilla parity: `Creaking.ATTACK_INTERVAL`.
29const ATTACK_INTERVAL: i64 = 40;
30/// Vanilla parity: `Creaking.SPEED_MULTIPLIER_WHEN_IDLING`.
31const SPEED_MULTIPLIER_WHEN_IDLING: f64 = 0.3;
32/// Vanilla parity: the `1.0F` of the fight activity's walk behavior.
33const SPEED_MULTIPLIER_WHEN_CHASING: f64 = 1.0;
34/// Vanilla parity: the `Swim<>(0.8F)` of the core activity.
35const SWIM_CHANCE: f32 = 0.8;
36/// Vanilla parity: the `LookAtTargetSink(45, 90)` of the core activity.
37const LOOK_AT_TARGET_MIN_DURATION: i32 = 45;
38const LOOK_AT_TARGET_MAX_DURATION: i32 = 90;
39/// Vanilla parity: the `SetEntityLookTargetSometimes.create(8.0F, UniformInt.of(30, 60))`
40/// of the idle activity.
41const GAZE_RANGE: f64 = 8.0;
42const GAZE_INTERVAL: UniformIntProvider = UniformIntProvider {
43    min_inclusive: 30,
44    max_inclusive: 60,
45};
46/// Vanilla parity: the `DoNothing(30, 60)` of the idle gate.
47const IDLE_DO_NOTHING_MIN: i32 = 30;
48const IDLE_DO_NOTHING_MAX: i32 = 60;
49/// Vanilla parity: the `SetWalkTargetFromLookTarget.create(0.3F, 3)` of the same gate.
50const IDLE_LOOK_WALK_CLOSE_ENOUGH: i32 = 3;
51
52/// The sensors a creaking runs.
53///
54/// Vanilla parity: the sensor list of `Creaking.BRAIN_PROVIDER`. The player
55/// sensor is not optional decoration here: both the freeze and the stuck-player
56/// check read `NEAREST_PLAYERS` straight out of the brain.
57pub const SENSORS: &[SensorType] = &[
58    SensorType::NearestLivingEntities,
59    SensorType::NearestPlayers,
60];
61
62/// Builds a creaking's brain.
63#[must_use]
64pub fn make_brain() -> Brain {
65    Brain::new(
66        SENSORS,
67        vec![core_activity(), idle_activity(), fight_activity()],
68    )
69}

71/// Vanilla parity: `CreakingAi.initCoreActivity`.
72fn core_activity() -> ActivityData {
73    ActivityData::create(
74        Activity::Core,
75        0,
76        vec![
77            Behavior::boxed(SwimWhileUnfrozen::new()),
78            Behavior::boxed(LookAtTargetSink::new(
79                LOOK_AT_TARGET_MIN_DURATION,
80                LOOK_AT_TARGET_MAX_DURATION,
81            )),
82            Behavior::boxed(MoveToTargetSink::new()),
83        ],
84    )
85}
86
87/// Vanilla parity: `CreakingAi.initIdleActivity`.
88fn idle_activity() -> ActivityData {
89    ActivityData::create(
90        Activity::Idle,
91        10,
92        vec![
93            OneShot::boxed(StartAttacking::conditional(
94                |ctx| with_creaking(ctx.mob(), CreakingEntity::is_active),
95                |ctx| {
96                    ctx.brain()
97                        .get_memory(memory_module_types::NEAREST_VISIBLE_ATTACKABLE_PLAYER)
98                        .and_then(|remembered| remembered.get())
99                },
100            )),
101            OneShot::boxed(SetEntityLookTargetSometimes::any_within(
102                GAZE_RANGE,
103                GAZE_INTERVAL,
104            )),
105            Box::new(RunOne::unconditional(vec![
106                (
107                    OneShot::boxed(RandomStroll::stroll(SPEED_MULTIPLIER_WHEN_IDLING)),
108                    2,
109                ),
110                (
111                    OneShot::boxed(SetWalkTargetFromLookTarget::new(
112                        SPEED_MULTIPLIER_WHEN_IDLING,
113                        IDLE_LOOK_WALK_CLOSE_ENOUGH,
114                    )),
115                    2,
116                ),
117                (
118                    Box::new(DoNothing::new(IDLE_DO_NOTHING_MIN, IDLE_DO_NOTHING_MAX)),
119                    1,
120                ),
121            ])),
122        ],
123    )
124}
125
126/// Vanilla parity: `CreakingAi.initFightActivity`.
127fn fight_activity() -> ActivityData {
128    ActivityData::create(
129        Activity::Fight,
130        10,
131        vec![
132            OneShot::boxed(SetWalkTargetFromAttackTargetIfTargetOutOfReach::new(
133                SPEED_MULTIPLIER_WHEN_CHASING,
134            )),
135            OneShot::boxed(MeleeAttack::conditional(
136                |mob| with_living_creaking(mob, CreakingEntity::can_move),
137                ATTACK_INTERVAL,
138            )),
139            OneShot::boxed(
140                StopAttackingIfTargetInvalid::new()
141                    .when(|ctx, target| !is_attack_target_still_reachable(ctx, target)),
142            ),
143        ],
144    )
145    .with_conditions(vec![(
146        memory_module_types::ATTACK_TARGET.id(),
147        MemoryStatus::ValuePresent,
148    )])
149}

151/// Vanilla parity: `CreakingAi.updateActivity`.
152///
153/// A frozen creaking is forced back to the default activity rather than being
154/// offered the first valid one, so it cannot pick a new fight while it is being
155/// stared at.
156pub fn update_activity(brain: &Brain, can_move: bool) {
157    if can_move {
158        brain.set_active_activity_to_first_valid(&[Activity::Fight, Activity::Idle]);
159    } else {
160        brain.use_default_activity();
161    }
162}

164/// Vanilla parity: the private `CreakingAi.isAttackTargetStillReachable`, which
165/// keeps a target only while the player sensor can still see it. That is what
166/// makes stepping out of sight enough to lose a creaking, without the usual
167/// hundred-tick memory.
168fn is_attack_target_still_reachable(ctx: &BrainContext<'_>, target: &SharedEntity) -> bool {
169    let Some(visible) = ctx
170        .brain()
171        .get_memory(memory_module_types::NEAREST_VISIBLE_ATTACKABLE_PLAYERS)
172    else {
173        return false;
174    };
175    visible
176        .iter()
177        .any(|remembered| remembered.id() == target.id())
178}

180fn with_creaking(mob: &dyn PathfinderMob, read: impl FnOnce(&CreakingEntity) -> bool) -> bool {
181    mob.downcast_ref::<CreakingEntity>().is_some_and(read)
182}

184fn with_living_creaking(
185    mob: &dyn LivingEntity,
186    read: impl FnOnce(&CreakingEntity) -> bool,
187) -> bool {
188    mob.downcast_ref::<CreakingEntity>().is_some_and(read)
189}

191/// Vanilla parity: the anonymous `Swim` subclass of `CreakingAi.initCoreActivity`,
192/// whose `checkExtraStartConditions` refuses while the creaking is frozen. A
193/// frozen creaking standing in water drowns rather than swimming out, which is
194/// the point.
195struct SwimWhileUnfrozen {
196    inner: Swim,
197}
198
199impl SwimWhileUnfrozen {
200    const fn new() -> Self {
201        Self {
202            inner: Swim::new(SWIM_CHANCE),
203        }
204    }
205}
206
207impl TimedBehavior for SwimWhileUnfrozen {
208    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
209        self.inner.entry_condition()
210    }
211
212    fn check_extra_start_conditions(&mut self, ctx: &BrainContext<'_>) -> bool {
213        with_creaking(ctx.mob(), CreakingEntity::can_move)
214            && self.inner.check_extra_start_conditions(ctx)
215    }
216
217    fn can_still_use(&mut self, ctx: &BrainContext<'_>) -> bool {
218        self.inner.can_still_use(ctx)
219    }
220
221    fn tick(&mut self, ctx: &BrainContext<'_>) {
222        self.inner.tick(ctx);
223    }
224
225    fn debug_name(&self) -> &'static str {
226        "Swim"
227    }
228}
