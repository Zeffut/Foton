1//! Vanilla `BreezeUtil`.
2//!
3//! The two geometry questions every breeze behavior asks: where would I like to
4//! be relative to my target, and can I see a spot from here.
5
6use foton_registry::vanilla_attributes;
7use glam::DVec3;
8
9use crate::entity::{LivingEntity, PathfinderMob};
10use crate::world::{ClipBlockShape, ClipFluid};
11
12/// Vanilla parity: `BreezeUtil.MAX_LINE_OF_SIGHT_TEST_RANGE`.
13const MAX_LINE_OF_SIGHT_TEST_RANGE: f64 = 50.0;
14
15/// Vanilla parity: the `90` degree spread of `randomPointBehindTarget`.
16const SPREAD_DEGREES: f32 = 90.0;
17
18/// Vanilla parity: the `Mth.lerp(nextFloat(), 4.0F, 8.0F)` radius of the same.
19const MIN_RADIUS: f32 = 4.0;
20const MAX_RADIUS: f32 = 8.0;
21
22/// Returns a point roughly behind `enemy`, four to eight blocks out.
23///
24/// Vanilla parity: `BreezeUtil.randomPointBehindTarget`. The angle is the
25/// target's own head rotation turned half around and then jittered by a
26/// gaussian, so a breeze tends to land behind whoever it is fighting and only
27/// sometimes off to one side.
28#[must_use]
29pub(super) fn random_point_behind_target(enemy: &dyn LivingEntity) -> DVec3 {
30    #[expect(
31        clippy::cast_possible_truncation,
32        reason = "a gaussian sample immediately used as an angle in degrees"
33    )]
34    let jitter = next_gaussian() as f32 * SPREAD_DEGREES / 2.0;
35    let view_angle = enemy.y_head_rot() + 180.0 + jitter;
36    // Vanilla parity: `Mth.lerp(random.nextFloat(), 4.0F, 8.0F)`.
37    let radius = (MAX_RADIUS - MIN_RADIUS).mul_add(rand::random::<f32>(), MIN_RADIUS);
38    enemy.position() + direction_from_yaw(view_angle) * f64::from(radius)
39}
40
41/// Vanilla parity: `Vec3.directionFromRotation(0.0F, yRot)`, whose pitch of
42/// zero leaves a purely horizontal unit vector.
43#[must_use]
44fn direction_from_yaw(y_rot: f32) -> DVec3 {
45    let yaw = -y_rot.to_radians();
46    DVec3::new(f64::from(yaw.sin()), 0.0, f64::from(yaw.cos()))
47}

49/// Returns whether nothing solid stands between the breeze and `target`.
50///
51/// Vanilla parity: `BreezeUtil.hasLineOfSight`. Anything past the test range
52/// counts as out of sight without a ray being cast at all.
53#[must_use]
54pub(super) fn has_line_of_sight(breeze: &dyn PathfinderMob, target: DVec3) -> bool {
55    let from = breeze.position();
56    if target.distance(from) > max_line_of_sight_test_range(breeze) {
57        return false;
58    }
59    let Some(world) = breeze.level() else {
60        return false;
61    };
62    world
63        .clip(from, target, ClipBlockShape::Collider, ClipFluid::None)
64        .is_miss()
65}
66
67/// Vanilla parity: the private `BreezeUtil.getMaxLineOfSightTestRange`.
68fn max_line_of_sight_test_range(breeze: &dyn PathfinderMob) -> f64 {
69    breeze
70        .attributes()
71        .lock()
72        .get_value(vanilla_attributes::FOLLOW_RANGE)
73        .unwrap_or(0.0)
74        .max(MAX_LINE_OF_SIGHT_TEST_RANGE)
75}

77/// Draws a standard normal sample.
78///
79/// Vanilla parity: `RandomSource.nextGaussian`, which is Java's Marsaglia polar
80/// method. Foton's seeded `Random` has one, but a breeze picking where to jump
81/// is incidental live randomness rather than anything a save has to reproduce,
82/// so this runs on the unseeded runtime RNG and only the distribution matters.
83fn next_gaussian() -> f64 {
84    loop {
85        let x = 2.0 * rand::random::<f64>() - 1.0;
86        let y = 2.0 * rand::random::<f64>() - 1.0;
87        let squared = x.mul_add(x, y * y);
88        if squared >= 1.0 || squared == 0.0 {
89            continue;
90        }
91        return x * (-2.0 * squared.ln() / squared).sqrt();
92    }
93}
