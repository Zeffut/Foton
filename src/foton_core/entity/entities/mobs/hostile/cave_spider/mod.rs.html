39/// Experience this mob drops.
40///
41/// Vanilla parity: the `this.xpReward = 5` of the `Monster` constructor, which
42/// every monster inherits and this one does not override.
43const XP_REWARD: i32 = 5;
44
45/// Bit of the synced flags byte that marks a climbing spider.
46///
47/// Vanilla parity: the `1` mask of `Spider.setClimbing`.
48const CLIMBING_FLAG: i8 = 1;
49
50/// Upward velocity of a spider's pounce.
51///
52/// Vanilla parity: `LeapAtTargetGoal(this, 0.4F)`.
53const LEAP_HEIGHT: f32 = 0.4;
54
55/// Speed multiplier while chasing.
56const ATTACK_SPEED_MODIFIER: f64 = 1.0;
57
58/// Seconds of poison a cave spider inflicts on normal difficulty.
59const POISON_SECONDS_NORMAL: i32 = 7;
60
61/// Seconds of poison a cave spider inflicts on hard difficulty.
62const POISON_SECONDS_HARD: i32 = 15;
63
64/// Speed multiplier while wandering.
65const STROLL_SPEED_MODIFIER: f64 = 0.8;
66
67/// Distance at which a spider watches a player.
68const LOOK_AT_PLAYER_RANGE: f64 = 8.0;
69
70/// A cave spider.
71#[entity_behavior(class = "CaveSpider")]
72pub struct CaveSpiderEntity {
73    base: EntityBase,
74    entity_type: EntityTypeRef,
75    living_base: LivingEntityBase,
76    mob_base: MobBase,
77    entity_data: SyncMutex<CaveSpiderEntityData>,
78}

85impl CaveSpiderEntity {
86    /// Creates a cave spider at runtime.
87    #[must_use]
88    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
89        Self::new_with_base(
90            EntityBase::new(id, position, entity_type.dimensions, world),
91            entity_type,
92        )
93    }
94
95    /// Creates a cave spider from saved base data.
96    #[must_use]
97    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
98        Self::new_with_base(
99            EntityBase::from_load(load, entity_type.dimensions),
100            entity_type,
101        )
102    }
103
104    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
105        let living_base = LivingEntityBase::new(entity_type);
106        let mob_base = MobBase::new();
107        mob_base.set_xp_reward(XP_REWARD);
108        let mut entity_data = CaveSpiderEntityData::new();
109        living_base.initialize_synced_data(&mut entity_data);
110
111        {
112            // A cave spider keeps the Spider goal set unchanged.
113            let mut goals = mob_base.goal_selector().lock();
114            goals.add_goal(1, FloatGoal::new(&mob_base));
115            goals.add_goal(3, LeapAtTargetGoal::new(LEAP_HEIGHT));
116            goals.add_goal(4, MeleeAttackGoal::new(ATTACK_SPEED_MODIFIER, true));
117            goals.add_goal(5, WaterAvoidingRandomStrollGoal::new(STROLL_SPEED_MODIFIER));
118            goals.add_goal(6, LookAtPlayerGoal::new(LOOK_AT_PLAYER_RANGE));
119            goals.add_goal(6, RandomLookAroundGoal::new());
120            // TODO: vanilla also avoids armadillos at priority 2.
121        }
122
123        {
124            let mut targets = mob_base.target_selector().lock();
125            targets.add_goal(1, HurtByTargetGoal::new());
126            targets.add_goal(2, SpiderTargetGoal::new_for_players());
127        }
128
129        Self {
130            base,
131            entity_type,
132            living_base,
133            mob_base,
134            entity_data: SyncMutex::new(entity_data),
135        }
136    }
137
138    /// Returns whether the spider is currently stuck to a wall.
139    #[must_use]
140    pub fn is_climbing(&self) -> bool {
141        *self.entity_data.lock().spider.flags.get() & CLIMBING_FLAG != 0
142    }
143
144    /// Marks the spider as climbing or not.
145    ///
146    /// Vanilla parity: `Spider.setClimbing`.
147    fn set_climbing(&self, climbing: bool) {
148        let mut data = self.entity_data.lock();
149        let flags = *data.spider.flags.get();
150        let updated = if climbing {
151            flags | CLIMBING_FLAG
152        } else {
153            flags & !CLIMBING_FLAG
154        };
155        data.spider.flags.set(updated);
156    }
157}

159impl Entity for CaveSpiderEntity {
160    fn base(&self) -> &EntityBase {
161        &self.base
162    }
163
164    fn entity_type(&self) -> EntityTypeRef {
165        self.entity_type
166    }
167
168    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
169        Some(&self.entity_data)
170    }
171
172    fn base_tick(&self) {
173        // Vanilla parity: a spider clings to whatever it just walked into, which is
174        // why walking a spider into a wall is enough to send it up.
175        self.set_climbing(self.horizontal_collision());
176        Mob::base_tick_mob(self);
177    }
178
179    fn sound_source(&self) -> SoundSource {
180        SoundSource::Hostile
181    }
182
183    /// Vanilla parity: `Spider.onClimbable`.
184    fn on_climbable(&self) -> bool {
185        self.is_climbing()
186    }
187
188    /// Vanilla parity: neither `CaveSpider` nor `Spider` overrides
189    /// `addAdditionalSaveData`, so the shared half is the whole of it.
190    fn save_additional(&self, nbt: &mut NbtCompound) {
191        self.save_mob(nbt);
192    }
193
194    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
195        self.load_mob(nbt);
196    }

274    /// Poisons the target after a successful hit.
275    ///
276    /// Vanilla parity: `CaveSpider.doHurtTarget`. Easy difficulty poisons nothing,
277    /// which is why a cave spider is merely annoying on peaceful settings.
278    fn do_hurt_target(&self, world: &World, target: &SharedEntity) -> bool {
279        if !Mob::mob_do_hurt_target(self, world, target) {
280            return false;
281        }
282        let Some(living) = target.as_living_entity() else {
283            return true;
284        };
285        let poison_seconds = match world.difficulty() {
286            Difficulty::Normal => POISON_SECONDS_NORMAL,
287            Difficulty::Hard => POISON_SECONDS_HARD,
288            Difficulty::Peaceful | Difficulty::Easy => 0,
289        };
290        if poison_seconds > 0 {
291            living.add_mob_effect(MobEffectInstance::with_duration(
292                vanilla_mob_effects::POISON,
293                poison_seconds * 20,
294                0,
295            ));
296        }
297        true
298    }
299
300    fn mob_flags(&self) -> i8 {
301        *self.entity_data.lock().mob().mob_flags.get()
302    }
303
304    fn set_mob_flags(&self, flags: i8) {
305        self.entity_data.lock().mob_mut().mob_flags.set(flags);
306    }
