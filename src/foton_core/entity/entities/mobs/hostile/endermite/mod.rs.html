37/// Experience an endermite drops.
38///
39/// Vanilla parity: the `this.xpReward = 3` of the constructor.
40const XP_REWARD: i32 = 3;
41
42/// Ticks an endermite lives before it vanishes.
43///
44/// Vanilla parity: `Endermite.MAX_LIFE`.
45const MAX_LIFE: i32 = 2400;
46
47/// Speed multiplier while chasing.
48///
49/// Vanilla parity: `new MeleeAttackGoal(this, 1.0, false)`.
50const ATTACK_SPEED_MODIFIER: f64 = 1.0;
51
52/// Speed multiplier while wandering.
53///
54/// Vanilla parity: `new WaterAvoidingRandomStrollGoal(this, 1.0)`.
55const STROLL_SPEED_MODIFIER: f64 = 1.0;
56
57/// Distance at which an endermite watches a player.
58///
59/// Vanilla parity: `new LookAtPlayerGoal(this, Player.class, 8.0F)`.
60const LOOK_AT_PLAYER_RANGE: f64 = 8.0;
61
62/// Volume of an endermite's footstep.
63///
64/// Vanilla parity: the `0.15F` of `Endermite.playStepSound`.
65const STEP_SOUND_VOLUME: f32 = 0.15;
66
67/// Pitch of an endermite's footstep.
68const STEP_SOUND_PITCH: f32 = 1.0;

70/// An endermite.
71#[entity_behavior(class = "Endermite")]
72pub struct EndermiteEntity {
73    base: EntityBase,
74    entity_type: EntityTypeRef,
75    living_base: LivingEntityBase,
76    mob_base: MobBase,
77    entity_data: SyncMutex<EndermiteEntityData>,
78    /// Ticks this endermite has been alive (vanilla `Endermite.life`).

80}

87impl EndermiteEntity {
88    /// Creates an endermite at runtime.
89    #[must_use]
90    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
91        Self::new_with_base(
92            EntityBase::new(id, position, entity_type.dimensions, world),
93            entity_type,
94        )
95    }
96
97    /// Creates an endermite from saved base data.
98    #[must_use]
99    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
100        Self::new_with_base(
101            EntityBase::from_load(load, entity_type.dimensions),
102            entity_type,
103        )
104    }
105
106    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
107        let living_base = LivingEntityBase::new(entity_type);
108        let mob_base = MobBase::new();
109        let mut entity_data = EndermiteEntityData::new();
110        living_base.initialize_synced_data(&mut entity_data);
111        mob_base.set_xp_reward(XP_REWARD);
112
113        {
114            // Keep vanilla Endermite goal priorities in the same order.
115            let mut goals = mob_base.goal_selector().lock();
116            goals.add_goal(1, FloatGoal::new(&mob_base));
117            goals.add_goal(1, ClimbOnTopOfPowderSnowGoal::new());
118            goals.add_goal(2, MeleeAttackGoal::new(ATTACK_SPEED_MODIFIER, false));
119            goals.add_goal(3, WaterAvoidingRandomStrollGoal::new(STROLL_SPEED_MODIFIER));
120            goals.add_goal(7, LookAtPlayerGoal::new(LOOK_AT_PLAYER_RANGE));
121            goals.add_goal(8, RandomLookAroundGoal::new());
122        }
123
124        {
125            let mut targets = mob_base.target_selector().lock();
126            targets.add_goal(1, HurtByTargetGoal::new().set_alert_others([]));
127            targets.add_goal(
128                2,
129                NearestAttackableTargetGoal::new_for_players(true, |_, _, _| true),
130            );
131        }
132
133        Self {
134            base,
135            entity_type,
136            living_base,
137            mob_base,
138            entity_data: SyncMutex::new(entity_data),
139            life: SyncMutex::new(0),
140        }
141    }
142
143    /// Returns how long this endermite has been alive, in ticks.
144    #[must_use]
145    pub fn life(&self) -> i32 {
146        *self.life.lock()
147    }
148
149    /// Advances the lifetime and discards the endermite once it runs out.
150    ///
151    /// Vanilla parity: the server half of `Endermite.aiStep`. A persistent
152    /// endermite stops aging entirely rather than aging and being spared at the
153    /// end, which is why one summoned with `PersistenceRequired` keeps forever.
154    fn tick_life(&self) {
155        if !self.is_persistence_required() {
156            *self.life.lock() += 1;
157        }
158
159        if self.life() >= MAX_LIFE {
160            self.set_removed(RemovalReason::Discarded);
161        }
162    }
163}

165impl Entity for EndermiteEntity {
166    fn base(&self) -> &EntityBase {
167        &self.base
168    }
169
170    fn entity_type(&self) -> EntityTypeRef {
171        self.entity_type
172    }
173
174    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
175        Some(&self.entity_data)
176    }
177
178    /// Vanilla parity: `Endermite.tick`, which pins the body to the head before
179    /// anything else runs; an endermite has no independent body turn.
180    fn tick(&self) {
181        self.set_y_body_rot(self.rotation().0);
182        self.tick_living_entity();
183    }
184
185    fn sound_source(&self) -> SoundSource {
186        SoundSource::Hostile
187    }
188
189    /// Vanilla parity: `Endermite.getMovementEmission`.
190    fn movement_emission(&self) -> EntityMovementEmission {
191        EntityMovementEmission::Events
192    }
193
194    /// Vanilla parity: `Endermite.playStepSound`, which ignores the block
195    /// stepped on.
196    fn play_step_sound(&self, _pos: BlockPos, _block_state: BlockStateId) {
197        self.play_sound(
198            &sound_events::ENTITY_ENDERMITE_STEP,
199            STEP_SOUND_VOLUME,
200            STEP_SOUND_PITCH,
201        );
202    }
203
204    fn save_additional(&self, nbt: &mut NbtCompound) {
205        self.save_mob(nbt);
206        nbt.insert("Lifetime", self.life());
207    }
208
209    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
210        self.load_mob(nbt);
211        if let Some(life) = nbt.int("Lifetime") {
212            *self.life.lock() = life;
213        }
214    }
215}

217impl LivingEntity for EndermiteEntity {
218    /// Returns synchronized data declared by vanilla `LivingEntity`.
219    fn living_synced_data(&self) -> Option<&dyn LivingEntitySyncedData> {
220        Some(&self.entity_data)
221    }
222
223    fn living_base(&self) -> &LivingEntityBase {
224        &self.living_base
225    }
226
227    /// Vanilla parity: `Mob.serverAiStep`, which is where a mob's goals run.
228    fn server_ai_step(&self) {
229        Mob::mob_server_ai_step(self);
230    }
231
232    /// Vanilla parity: `Endermite.aiStep`, which ages the endermite after the
233    /// base step rather than before it.
234    fn ai_step(&self) -> Option<MoveResult> {
235        let result = self.default_ai_step();
236        self.tick_life();
237        result
238    }
239
240    fn get_health(&self) -> f32 {
241        *self.entity_data.lock().living_entity().health.get()
242    }
243
244    fn set_health(&self, health: f32) {
245        let max_health = self.get_max_health();
246        let clamped = health.clamp(0.0, max_health);
247        self.entity_data
248            .lock()
249            .living_entity_mut()
250            .health
251            .set(clamped);
252    }
253
254    fn hurt_sound(&self, _source: &DamageSource) -> Option<SoundEventRef> {
255        Some(&sound_events::ENTITY_ENDERMITE_HURT)
256    }
257
258    fn death_sound(&self) -> Option<SoundEventRef> {
259        Some(&sound_events::ENTITY_ENDERMITE_DEATH)
260    }
261
262    /// Vanilla parity: `Endermite.setYBodyRot`, which snaps the head to match
263    /// whenever the body rotation is forced.
264    fn set_y_body_rot(&self, y_body_rot: f32) {
265        let (_, pitch) = self.rotation();
266        self.set_rotation((y_body_rot, pitch));
267        self.living_base().set_y_body_rot(y_body_rot);
268    }
269}
