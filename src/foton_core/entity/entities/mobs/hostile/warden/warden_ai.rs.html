1//! The warden's brain.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.monster.warden.WardenAi`. Eight
4//! activities, and the order they are tried in is the warden's whole personality: it
5//! emerges, then digs away if nothing has happened, then roars, then fights, then goes to
6//! look at whatever it heard, then sniffs, then wanders.

76/// Builds a warden's brain.
77///
78/// Vanilla parity: `Warden.BRAIN_PROVIDER` feeding `WardenAi.getActivities`.
79#[must_use]
80pub fn make_brain() -> Brain {
81    Brain::new_with_memories(
82        SENSORS,
83        EXTRA_MEMORIES,
84        vec![
85            core_activity(),
86            emerge_activity(),
87            digging_activity(),
88            idle_activity(),
89            roar_activity(),
90            fight_activity(),
91            investigate_activity(),
92            sniffing_activity(),
93        ],
94    )
95}

110/// Vanilla parity: `WardenAi.setDigCooldown`.
111///
112/// The cooldown is only refreshed, never created: a warden that never had one is a
113/// warden that is free to dig away as soon as it runs out of things to do.
114pub fn set_dig_cooldown(brain: &Brain) {
115    if brain.has_memory_value(memory_module_types::DIG_COOLDOWN.id()) {
116        brain.set_memory_with_expiry(
117            memory_module_types::DIG_COOLDOWN,
118            Unit,
119            DIGGING_COOLDOWN.into(),
120        );
121    }
122}

124/// Vanilla parity: `SonicBoom.setCooldown`.
125pub fn set_sonic_boom_cooldown(brain: &Brain, cooldown: i64) {
126    brain.set_memory_with_expiry(memory_module_types::SONIC_BOOM_COOLDOWN, Unit, cooldown);
127}

129/// Vanilla parity: `WardenAi.setDisturbanceLocation`.
130///
131/// Not implemented: the world-border bounds check vanilla guards this with, for the
132/// reason given on [`WardenEntity::can_target_entity`].
133pub fn set_disturbance_location(warden: &WardenEntity, position: BlockPos) {
134    let brain = warden.brain_ref();
135    if warden.entity_angry_at().is_some()
136        || brain.has_memory_value(memory_module_types::ATTACK_TARGET.id())
137    {
138        return;
139    }
140
141    set_dig_cooldown(brain);
142    brain.set_memory_with_expiry(
143        memory_module_types::SNIFF_COOLDOWN,
144        Unit,
145        DISTURBANCE_LOCATION_EXPIRY_TIME,
146    );
147    brain.set_memory_with_expiry(
148        memory_module_types::LOOK_TARGET,
149        PositionTracker::of_block(position),
150        DISTURBANCE_LOCATION_EXPIRY_TIME,
151    );
152    brain.set_memory_with_expiry(
153        memory_module_types::DISTURBANCE_LOCATION,
154        position,
155        DISTURBANCE_LOCATION_EXPIRY_TIME,
156    );
157    brain.erase_memory(memory_module_types::WALK_TARGET.id());
158}

164        0,

275/// Vanilla parity: `WardenAi.initFightActivity`.
276fn fight_activity() -> ActivityData {
277    ActivityData::create(
278        Activity::Fight,
279        10,
280        vec![
281            behaviors::dig_cooldown_setter(),
282            OneShot::boxed(
283                StopAttackingIfTargetInvalid::new()
284                    .when(|ctx, target| {
285                        use foton_utils::Downcast as _;
286
287                        // Vanilla stops when the warden has calmed down or the target has
288                        // stopped being something it could attack.
289                        ctx.mob()
290                            .downcast_ref::<WardenEntity>()
291                            .is_none_or(|warden| {
292                                !warden.anger_level().is_angry()
293                                    || !warden.can_target_entity(Some(target.as_ref()))
294                            })
295                    })
296                    .on_erased(on_target_invalid)
297                    .never_tiring(),
298            ),
299            OneShot::boxed(SetEntityLookTarget::matching_in_context(
300                is_attack_target,
301                FOLLOW_RANGE,
302            )),
303            OneShot::boxed(SetWalkTargetFromAttackTargetIfTargetOutOfReach::new(
304                SPEED_MULTIPLIER_WHEN_FIGHTING,
305            )),
306            Behavior::boxed(behaviors::SonicBoom),
307            OneShot::boxed(MeleeAttack::new(MELEE_ATTACK_COOLDOWN)),
308        ],
309    )
310    .gated_by(memory_module_types::ATTACK_TARGET.id())
311}

313/// Vanilla parity: `WardenAi.onTargetInvalid`.
314fn on_target_invalid(ctx: &BrainContext<'_>, target: &SharedEntity) {
315    use foton_utils::Downcast as _;
316
317    let Some(warden) = ctx.mob().downcast_ref::<WardenEntity>() else {
318        return;
319    };
320    if !warden.can_target_entity(Some(target.as_ref())) {
321        warden.clear_anger(target.as_ref());
322    }
323    set_dig_cooldown(warden.brain_ref());
324}

326/// Vanilla parity: the `entity -> isTarget(body, entity)` of the fight activity's look.
327fn is_attack_target(ctx: &BrainContext<'_>, entity: &dyn LivingEntity) -> bool {
328    ctx.brain()
329        .get_memory(memory_module_types::ATTACK_TARGET)
330        .is_some_and(|memory| memory.id() == entity.id())
