1//! Hoglin entity.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.monster.hoglin.Hoglin`. The
4//! Nether's only breedable animal and one of its nastier hostiles at once: it
5//! extends `Animal` and implements `Enemy`, so it falls in love over crimson
6//! fungus and gores anything else that comes near.
7//!
8//! Two things drive it away rather than toward: warped fungus and respawn
9//! anchors pacify it, and a crowd of piglins makes it run. Left in the
10//! overworld it turns into a [`crate::entity::entities::ZoglinEntity`], which
11//! keeps the charge and loses the fear.

161    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {

169        living_base.initialize_synced_data(&mut entity_data);
170
171        Self {
172            base,
173            entity_type,
174            living_base,
175            mob_base,
176            ageable_base,
177            animal_base,
178            entity_data: SyncMutex::new(entity_data),
179            state: SyncMutex::new(HoglinState {
180                attack_animation_remaining_ticks: 0,
181                time_in_overworld: 0,
182                cannot_be_hunted: false,

184            brain: hoglin_ai::make_brain(),
185        }

220    /// Whether this hoglin is standing somewhere that turns it into a zoglin.
221    ///
222    /// Vanilla parity: `Hoglin.isConverting`, which reads the
223    /// `EnvironmentAttributes.PIGLINS_ZOMBIFY` attribute. No vanilla biome or
224    /// timeline overrides that attribute, so the dimension type's own
225    /// `piglins_zombify` flag is the whole answer.
226    #[must_use]
227    pub fn is_converting(&self) -> bool {
228        let Some(world) = self.level() else {
229            return false;
230        };
231        !self.is_immune_to_zombification()
232            && !self.is_no_ai()
233            && world.dimension_type.piglins_zombify
234    }

249    /// Turns this hoglin into a zoglin.
250    ///
251    /// Vanilla parity: the private `Hoglin.finishConversion`, whose
252    /// `ConversionParams.single(this, true, false)` keeps the equipment -- a
253    /// hoglin wears none -- and drops `canPickUpLoot`.
254    pub fn finish_conversion(&self) -> Option<Arc<ZoglinEntity>> {
255        convert_to(
256            self,
257            ConversionParams::single(true, false).with_reason(PiglinZombification),
258            |id, position, world| ZoglinEntity::new(&vanilla_entities::ZOGLIN, id, position, world),
259            |zoglin| {
260                // The age comes across in `copy_common_state`, which runs
261                // before this callback.
262                zoglin
263                    .living_base()
264                    .add_mob_effect(MobEffectInstance::with_duration(
265                        vanilla_mob_effects::NAUSEA,
266                        CONVERSION_NAUSEA_TICKS,
267                        0,
268                    ));
269            },
270        )
271    }

273    /// Vanilla parity: `Hoglin.ageBoundaryReached`.
274    fn hoglin_age_boundary_reached(&self, baby: bool) {
275        let (xp_reward, attack_damage) = if baby {
276            (BABY_XP_REWARD, BABY_ATTACK_DAMAGE)
277        } else {
278            (XP_REWARD, ATTACK_DAMAGE)
279        };
280        self.mob_base.set_xp_reward(xp_reward);
281        self.attributes()
282            .lock()
283            .set_base_value(vanilla_attributes::ATTACK_DAMAGE, attack_damage);
284    }

286    /// Vanilla parity: `Hoglin.getAmbientSound`, whose sound follows the
287    /// activity: a retreat call while fleeing or converting, angry while
288    /// fighting, and the plain ambient otherwise.
289    fn activity_sound(&self) -> SoundEventRef {
290        if self.brain.is_active(Activity::Avoid) || self.is_converting() {
291            return &sound_events::ENTITY_HOGLIN_RETREAT;
292        }
293        if self.brain.is_active(Activity::Fight) {
294            return &sound_events::ENTITY_HOGLIN_ANGRY;
295        }
296        if self
297            .brain
298            .has_memory_value(memory_module_types::NEAREST_REPELLENT.id())
299        {
300            return &sound_events::ENTITY_HOGLIN_RETREAT;
301        }
302        &sound_events::ENTITY_HOGLIN_AMBIENT
303    }

376            state.time_in_overworld = nbt.int("TimeInOverworld").unwrap_or(0);
377            state.cannot_be_hunted = nbt.byte("CannotBeHunted").unwrap_or(0) != 0;
378        }
379        self.brain.load(nbt);

399    /// Vanilla parity: `Hoglin.aiStep`, which runs down the attack animation,
400    /// on top of the ageing and love clocks every animal ticks.
401    fn ai_step(&self) -> Option<MoveResult> {
402        {
403            let mut state = self.state.lock();
404            if state.attack_animation_remaining_ticks > 0 {
405                state.attack_animation_remaining_ticks -= 1;
406            }
407        }
408        let result = self.default_ai_step();
409        AgeableMob::tick_ageable_mob(self);
410        Animal::tick_animal_love(self);
411        result
412    }

419        let max_health = self.get_max_health();
420        let clamped = health.clamp(0.0, max_health);
421        self.entity_data
422            .lock()
423            .living_entity_mut()
424            .health
425            .set(clamped);
426    }

432    /// Vanilla parity: `Hoglin.hurtServer`, which tells the brain who hit it.
433    fn hurt_server(&self, world: &World, source: &DamageSource, amount: f32) -> bool {
434        let was_hurt = self.living_hurt_server(world, source, amount);
435        if !was_hurt {
436            return false;
437        }
438        let Some(attacker) = source
439            .causing_entity_id
440            .and_then(|id| world.get_entity_by_id(id))
441        else {
442            return true;
443        };
444        if attacker.as_living_entity().is_some() {
445            hoglin_ai::was_hurt_by(world, &self.brain, self, &attacker);
446        }
447        true
448    }

463    /// Vanilla parity: `Hoglin.blockedByItem`. A grown hoglin whose charge is
464    /// blocked launches the blocker anyway, which is why a shield does not save
465    /// you from being thrown.
466    fn blocked_by_item(&self, defender: &dyn LivingEntity) {
467        // Vanilla's override does not call super, so a baby's blocked charge
468        // produces no knockback at all rather than the default nudge.
469        if !self.is_adult() {
470            return;
471        }
472        let Some(world) = self.level() else {
473            return;
474        };
475        let Some(defender_entity) = world.get_entity_by_id(defender.id()) else {
476            return;
477        };
478        hoglin_base::throw_target(self, &defender_entity);
479    }

552    /// Vanilla parity: `Hoglin.customServerAiStep`, which is the brain tick and
553    /// the overworld conversion clock.
554    fn custom_server_ai_step(&self) {
555        let Some(world) = self.level() else {
556            return;
557        };
558        self.brain.tick(&world, self);
559        hoglin_ai::update_activity(&self.brain);
560
561        if !self.is_converting() {
562            self.state.lock().time_in_overworld = 0;
563            return;
564        }
565
566        let time_in_overworld = {
567            let mut state = self.state.lock();
568            state.time_in_overworld += 1;
569            state.time_in_overworld
570        };
571        if time_in_overworld > CONVERSION_TIME {
572            self.make_sound(Some(&sound_events::ENTITY_HOGLIN_CONVERTED_TO_ZOMBIFIED));
573            self.finish_conversion();
574        }
575    }

577    /// Vanilla parity: `Hoglin.doHurtTarget`, the gore-and-throw.
578    fn do_hurt_target(&self, world: &World, target: &SharedEntity) -> bool {
579        if target.as_living_entity().is_none() {
580            return false;
581        }
582        self.set_attack_animation_remaining_ticks(ATTACK_ANIMATION_DURATION);
583        self.broadcast_entity_event(EntityStatus::StartAttacking);
584        self.make_sound(Some(&sound_events::ENTITY_HOGLIN_ATTACK));
585        hoglin_ai::on_hit_target(&self.brain, self, target);
586        hoglin_base::hurt_and_throw_target(world, self, target)
587    }

608    /// Vanilla parity: `Hoglin.finalizeSpawn`, one in five of which is a baby.
609    fn finalize_spawn(
610        &self,
611        world: &Arc<World>,
612        spawn_reason: EntitySpawnReason,
613        group_data: Option<SpawnGroupData>,
614    ) -> Option<SpawnGroupData> {
615        if rand::random::<f32>() < PROBABILITY_OF_SPAWNING_AS_BABY {

639impl PathfinderMob for HoglinEntity {
640    /// Vanilla parity: `Hoglin.getWalkTargetValue`, which is what keeps a
641    /// hoglin on crimson nylium and off a respawn anchor.
642    fn get_walk_target_value(&self, pos: BlockPos) -> f32 {
643        use foton_registry::blocks::block_state_ext::BlockStateExt as _;
644
645        let Some(world) = self.level() else {
646            return 0.0;
647        };
648        if let Some(repellent) = self
649            .brain
650            .get_memory(memory_module_types::NEAREST_REPELLENT)
651            && utils::block_closer_than(repellent, pos, hoglin_ai::REPELLENT_AVOID_RANGE)
652        {
653            return REPELLENT_WALK_VALUE;
654        }
655        if world.get_block_state(pos.below()).get_block() == &vanilla_blocks::CRIMSON_NYLIUM {
656            return CRIMSON_NYLIUM_WALK_VALUE;
657        }
658        0.0
659    }
660}
