3//! Vanilla parity: `EnderMan`. Present in more biomes than any other mob, and

157    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
158        let living_base = LivingEntityBase::new(entity_type);
159        let mob_base = MobBase::new();
160        mob_base.set_xp_reward(XP_REWARD);
161        let mut entity_data = EndermanEntityData::new();
162        living_base.initialize_synced_data(&mut entity_data);
163
164        {
165            // Vanilla parity: the goal order of `EnderMan.registerGoals`.
166            let mut goals = mob_base.goal_selector().lock();
167            goals.add_goal(0, FloatGoal::new(&mob_base));
168            goals.add_goal(1, EndermanFreezeWhenLookedAt);
169            goals.add_goal(2, MeleeAttackGoal::new(1.0, false));
170            goals.add_goal(7, WaterAvoidingRandomStrollGoal::new(1.0));
171            goals.add_goal(8, LookAtPlayerGoal::new(8.0));
172            goals.add_goal(8, RandomLookAroundGoal::new());
173            goals.add_goal(10, EndermanLeaveBlockGoal);
174            goals.add_goal(11, EndermanTakeBlockGoal);
175        }
176
177        {
178            let mut targets = mob_base.target_selector().lock();
179            targets.add_goal(1, EndermanLookForPlayerGoal);
180            targets.add_goal(2, HurtByTargetGoal::new());
181            targets.add_goal(
182                3,
183                NearestAttackableTargetGoal::new(true, |_, target, _| {
184                    target.entity_type() == &vanilla_entities::ENDERMITE
185                }),
186            );
187        }
188
189        Self {
190            base,
191            entity_type,
192            living_base,
193            mob_base,
194            entity_data: SyncMutex::new(entity_data),
195            anger: PersistentAnger::new(),
196            target_change_time: SyncMutex::new(0),
197        }
198    }

243    /// Returns whether this player is looking the enderman in the eye.
244    ///
245    /// Vanilla parity: `isBeingStaredBy`. The gaze has to land on the
246    /// enderman's own eye height, not anywhere on its body, which is why
247    /// looking at its feet is safe.
248    fn is_being_stared_by(&self, player: &dyn LivingEntity) -> bool {
249        // TODO: vanilla exempts a player wearing a carved pumpkin via
250        // PLAYER_NOT_WEARING_DISGUISE_ITEM; equipment predicates are not wired.
251        is_looking_at(self, player, STARE_CONE, true, false, &[self.get_eye_y()])
252    }

254    /// Blinks to a random spot within sixty-four blocks.
255    ///
256    /// Vanilla parity: the no-argument `teleport`.
257    fn teleport_randomly(&self) -> bool {
258        if !Entity::is_alive(self) {
259            return false;
260        }
261        let position = self.position();
262        let target = DVec3::new(
263            (rand::random::<f64>() - 0.5).mul_add(64.0, position.x),
264            position.y + f64::from(rand::random_range(0..64) - 32),
265            (rand::random::<f64>() - 0.5).mul_add(64.0, position.z),
266        );
267        self.teleport_to_spot(target)
268    }
269
270    /// Blinks away from something, roughly sixteen blocks back.
271    ///
272    /// Vanilla parity: `teleportTowards`, which despite the name moves the
273    /// enderman away: it is how one escapes an arrow or a splash of water.
274    fn teleport_away_from(&self, from: DVec3, from_eye_y: f64) -> bool {
275        let position = self.position();
276        let away = DVec3::new(
277            position.x - from.x,
278            (position.y + 0.5) - from_eye_y,
279            position.z - from.z,
280        );
281        if away.length_squared() <= 0.0 {
282            return false;
283        }
284        let away = away.normalize();
285
286        let target = DVec3::new(
287            (rand::random::<f64>() - 0.5).mul_add(8.0, position.x) - away.x * 16.0,
288            position.y + f64::from(rand::random_range(0..16) - 8) - away.y * 16.0,
289            (rand::random::<f64>() - 0.5).mul_add(8.0, position.z) - away.z * 16.0,
290        );
291        self.teleport_to_spot(target)
292    }
293
294    /// Tries one teleport, refusing water and open air.
295    ///
296    /// Vanilla parity: the three-argument `teleport`.
297    fn teleport_to_spot(&self, target: DVec3) -> bool {
298        let Some(world) = self.level() else {
299            return false;
300        };
301
302        // Vanilla walks down to the first block that stops movement and refuses
303        // the spot if it is wet, so an enderman never lands in a lake.
304        let mut pos = foton_utils::BlockPos::containing(target.x, target.y, target.z);
305        while pos.y() > world.get_min_y() && !world.get_block_state(pos).blocks_motion() {
306            pos = pos.below();
307        }
308
309        let landing = world.get_block_state(pos);
310        if !landing.blocks_motion() || is_water_fluid(landing.get_fluid_state().fluid_id) {
311            return false;
312        }
313
314        if !self.random_teleport(target) {
315            return false;
316        }
317
318        self.play_sound(&sound_events::ENTITY_ENDERMAN_TELEPORT, 1.0, 1.0);
319        true
320    }

361            40.0,
362        );
363    }

366/// How far the goal looks for someone staring.
367///
368/// Vanilla reads `getFollowDistance()` here; Foton has used a flat 64 since the
369/// goal was written, and this only names it.
370const AGGRO_RANGE: f64 = 64.0;
371
372/// Takes as a target whoever is staring, and whoever it is already angry at.
373///
374/// Vanilla parity: `EnderMan.EndermanLookForPlayerGoal`.
375struct EndermanLookForPlayerGoal;
376
377impl Goal for EndermanLookForPlayerGoal {
378    fn controls(&self) -> GoalControls {
379        GoalControls::TARGET
380    }
381
382    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
383        let Some(enderman) = mob.downcast_ref::<EndermanEntity>() else {
384            return false;
385        };
386        let Some(world) = mob.level() else {
387            return false;
388        };
389
390        // Vanilla parity: the `startAggroTargetConditions` of
391        // `EndermanLookForPlayerGoal` -- a `TargetingConditions.forCombat()`
392        // carrying the stare test as its selector. The combat branch is what
393        // keeps a creative player out: it ends at `Player.canBeSeenAsEnemy`,
394        // false while their abilities say invulnerable. Testing the stare on
395        // its own skipped that, so looking at an enderman in creative angered
396        // it.
397        let conditions = TargetingConditions::for_combat().range(AGGRO_RANGE);
398        let staring = world.nearest_player(mob.position(), AGGRO_RANGE, |player| {
399            (enderman.is_being_stared_by(player) || enderman.is_angry_at(player, &world))
400                && conditions.test(&world, Some(mob), player)
401        });
402
403        let Some(player) = staring else {
404            return false;
405        };
406        let target: SharedEntity = player;
407        if let Some(living) = target.as_living_entity()
408            && enderman.is_being_stared_by(living)
409        {
410            enderman.set_been_stared_at();
411        }
412        mob.set_target(Some(&target))
413    }
414
415    fn can_continue_to_use(&mut self, mob: &dyn PathfinderMob) -> bool {
416        mob.target().is_some()
417    }
418}

433    fn base_tick(&self) {
434        Mob::base_tick_mob(self);
435
436        // Vanilla parity: `isSensitiveToWater`. Water and rain both burn, and
437        // the enderman blinks away rather than standing in it.
438        let Some(world) = self.level() else {
439            return;
440        };
441        let pos = self.block_position();
442        let wet = self.is_in_water() || world.is_raining_at(pos);
443        if wet && Entity::is_alive(self) {
444            let source = DamageSource::environment(&vanilla_damage_types::DROWN);
445            self.hurt_server(&world, &source, WATER_DAMAGE);
446            self.teleport_randomly();
447        }
448    }

451        SoundSource::Hostile
452    }

454    /// Vanilla parity: `EnderMan.addAdditionalSaveData`, whose own contribution
455    /// is the carried block and the grudge.
456    fn save_additional(&self, nbt: &mut NbtCompound) {
457        self.save_mob(nbt);
458        if let Some(state) = self.carried_block() {
459            nbt.insert(
460                TAG_CARRIED_BLOCK_STATE,
461                NbtTag::Compound(block_state_nbt::save(state)),
462            );
463        }
464        write_persistent_anger(self, nbt);
465    }
466
467    /// Vanilla parity: `EnderMan.readAdditionalSaveData`. The air filter is
468    /// vanilla's: a saved air state means "carrying nothing", not "carrying a
469    /// pocket of air".
470    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
471        self.load_mob(nbt);
472        let carried = nbt
473            .compound(TAG_CARRIED_BLOCK_STATE)
474            .and_then(block_state_nbt::load)
475            .filter(|state| !state.is_air());
476        self.set_carried_block(carried);
477
478        let angry_at = nbt
479            .int_array("angry_at")
480            .and_then(|values| <Uuid as foton_utils::UuidExt>::from_int_array(&values));
481        read_persistent_anger(
482            self,
483            nbt.long("anger_end_time"),
484            nbt.int("AngerTime"),
485            angry_at,
486        );
487        if let Some(world) = self.level()
488            && let Some(target) = resolve_anger_target(&world, angry_at)
489        {
490            self.set_target(Some(&target));
491        }
492    }
493}

539    /// Drops whatever the block it was carrying would have dropped.
540    ///
541    /// Vanilla parity: `EnderMan.dropCustomDeathLoot`, which runs the block's
542    /// own loot table against a fake diamond axe enchanted from the
543    /// `minecraft:enderman_loot_drop` provider. Foton has no
544    /// `EnchantmentProvider` registry, so the one enchantment that provider
545    /// grants is applied here instead; the provider is shipped at
546    /// `foton-utils/build_assets/builtin_datapacks/minecraft/enchantment_provider/enderman_loot_drop.json`
547    /// and this should read it once a build step for that registry exists.
548    /// Without the enchantment an enderman holding a grass block would drop
549    /// dirt, which is not what happens in game.
550    fn drop_custom_death_loot(&self, _source: &DamageSource, _killed_by_player: bool) {
551        let (Some(world), Some(carried)) = (self.level(), self.carried_block()) else {
552            return;
553        };
554
555        let mut fake_tool = ItemStack::new(&vanilla_items::DIAMOND_AXE);
556        fake_tool.set_enchantments(&[(SILK_TOUCH.key.clone(), 1)], false);
557
558        let position = self.position();
559        let drops = BlockLootContext::new(&world, self.block_position())
560            .with_entity(Some(self))
561            .with_tool(&fake_tool)
562            .get_drops(carried);
563        for drop in drops {
564            if !drop.is_empty() {
565                world.spawn_item(position, drop);
566            }
567        }
568    }

570    /// Blinks away from whatever hit it.
571    ///
572    /// Vanilla parity: the `hurtServer` override, which is why an enderman
573    /// cannot be shot: the arrow lands and it is already somewhere else.
574    fn before_actually_hurt(&self, source: &DamageSource, _amount: f32) {
575        let Some(world) = self.level() else {
576            return;
577        };
578        let Some(attacker) = source
579            .causing_entity_id
580            .and_then(|id| world.get_entity_by_id(id))
581        else {
582            self.teleport_randomly();
583            return;
584        };
585        self.teleport_away_from(attacker.position(), attacker.get_eye_y());
586    }

623    /// Marks the enderman as roused and speeds it up.
624    ///
625    /// Vanilla parity: the `setTarget` override. The speed modifier is
626    /// transient: it is added when a target is taken and removed when it is
627    /// dropped, so a calm enderman moves at its ordinary pace.
628    fn set_target(&self, target: Option<&SharedEntity>) -> bool {
629        let previous = self.target();
630        // Vanilla parity: the `Mob.asValidTarget` every other mob goes
631        // through. Passing `|_| true` here also kept a target that had since
632        // switched to creative or spectator.
633        let changed = self
634            .mob_base()
635            .set_target(target, |target| self.is_valid_target(target));
636        if !changed {
637            return false;
638        }
639        let accepted = self.finish_target_change(previous, target);
640        if !accepted {
641            return false;
642        }
643
644        if target.is_none() {
645            *self.target_change_time.lock() = 0;
646            self.set_creepy(false);
647            self.entity_data.lock().ender_man_mut().stared_at.set(false);
648            self.attributes().lock().set_base_value(
649                vanilla_attributes::MOVEMENT_SPEED,
650                self.entity_type
651                    .default_attributes
652                    .iter()
653                    .find(|(key, _)| *key == "minecraft:movement_speed")
654                    .map_or(0.3, |(_, value)| *value),
655            );
656        } else {
657            *self.target_change_time.lock() = self.tick_count();
658            self.set_creepy(true);
659            let base = self
660                .attributes()
661                .lock()
662                .required_value(vanilla_attributes::MOVEMENT_SPEED);
663            self.attributes().lock().set_base_value(
664                vanilla_attributes::MOVEMENT_SPEED,
665                base + ATTACKING_SPEED_BONUS,
666            );
667        }
668
669        true
670    }

672    /// Runs the anger clock, and flees the sun.
673    ///
674    /// Vanilla parity: `aiStep` plus `customServerAiStep`. Daylight only drives
675    /// an enderman off once its grudge has had thirty seconds to cool, which is
676    /// why one that has just been provoked stays and fights in the open.
677    fn custom_server_ai_step(&self) {
678        let Some(world) = self.level() else {
679            return;
680        };
681        self.update_persistent_anger(&world, true);
682
683        let grace_over =
684            self.tick_count() >= *self.target_change_time.lock() + DAYLIGHT_GRACE_TICKS;
685        if !world.is_bright_outside() || !grace_over {
686            return;
687        }
688
689        let pos = self.block_position();
690        let brightness = world.light_level_dependent_magic_value(pos);
691        if brightness > DAYLIGHT_BRIGHTNESS
692            && world.can_see_sky(pos)
693            && rand::random::<f32>() * 30.0 < (brightness - 0.4) * 2.0
694        {
695            self.set_target(None);
696            self.teleport_randomly();
697        }
698    }

709/// Puts the carried block down somewhere it will stand.
710///
711/// Vanilla parity: `EnderMan.EndermanLeaveBlockGoal`. Together with the take
712/// goal this is what quietly rearranges a landscape overnight.
713struct EndermanLeaveBlockGoal;
714
715impl Goal for EndermanLeaveBlockGoal {
716    fn controls(&self) -> GoalControls {
717        GoalControls::EMPTY
718    }
719
720    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
721        let Some(enderman) = mob.downcast_ref::<EndermanEntity>() else {
722            return false;
723        };
724        if enderman.carried_block().is_none() {
725            return false;
726        }
727        mob.level().is_some_and(|world| {
728            world.get_game_rule(&MOB_GRIEFING)
729                && rand::random_range(0..reduced_tick_delay(LEAVE_BLOCK_INTERVAL_TICKS)) == 0
730        })
731    }
732
733    fn tick(&mut self, mob: &dyn PathfinderMob) {
734        let (Some(enderman), Some(world)) = (mob.downcast_ref::<EndermanEntity>(), mob.level())
735        else {
736            return;
737        };
738        let Some(carried) = enderman.carried_block() else {
739            return;
740        };
741
742        let position = mob.position();
743        let target = BlockPos::new(
744            (position.x - 1.0 + rand::random::<f64>() * 2.0).floor() as i32,
745            (position.y + rand::random::<f64>() * 2.0).floor() as i32,
746            (position.z - 1.0 + rand::random::<f64>() * 2.0).floor() as i32,
747        );
748        let below = target.below();
749        let carried = update_from_neighbour_shapes(&world, carried, target);
750        if !can_place_block(&world, mob, target, carried, below) {
751            return;
752        }
753
754        world.set_block(target, carried, UpdateFlags::UPDATE_ALL);
755        world.game_event_at(
756            &vanilla_game_events::BLOCK_PLACE,
757            block_center(target),
758            &GameEventContext::new(Some(mob), Some(carried)),
759        );
760        enderman.set_carried_block(None);
761    }
762}

764/// Returns the middle of a block, the point vanilla's `Vec3.atCenterOf` gives.
765fn block_center(pos: BlockPos) -> DVec3 {
766    let (x, y, z) = pos.get_center();
767    DVec3::new(x, y, z)
768}

770/// Vanilla parity: `EndermanLeaveBlockGoal.canPlaceBlock`.
771fn can_place_block(
772    world: &Arc<World>,
773    enderman: &dyn PathfinderMob,
774    pos: BlockPos,
775    carried: BlockStateId,
776    below: BlockPos,
777) -> bool {
778    let target_state = world.get_block_state(pos);
779    let below_state = world.get_block_state(below);
780    if !target_state.is_air()
781        || below_state.is_air()
782        || below_state.get_block() == &vanilla_blocks::BEDROCK
783        || !world.is_collision_shape_full_block_at(below, below_state)
784    {
785        return false;
786    }
787    if !BLOCK_BEHAVIORS
788        .get_behavior(carried.get_block())
789        .can_survive(carried, world.as_ref(), pos)
790    {
791        return false;
792    }
793
794    // Vanilla parity: `level.getEntities(this.enderman, unitCube)`, which is
795    // what stops an enderman entombing you in the block it is holding.
796    let cube = WorldAabb::new(
797        f64::from(pos.x()),
798        f64::from(pos.y()),
799        f64::from(pos.z()),
800        f64::from(pos.x()) + 1.0,
801        f64::from(pos.y()) + 1.0,
802        f64::from(pos.z()) + 1.0,
803    );
804    world
805        .get_entities_in_aabb_matching(&cube, |entity| entity.id() != enderman.id())
806        .is_empty()
807}

809/// Takes a block out of the ground and walks off with it.
810///
811/// Vanilla parity: `EnderMan.EndermanTakeBlockGoal`.
812struct EndermanTakeBlockGoal;
813
814impl Goal for EndermanTakeBlockGoal {
815    fn controls(&self) -> GoalControls {
816        GoalControls::EMPTY
817    }
818
819    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
820        let Some(enderman) = mob.downcast_ref::<EndermanEntity>() else {
821            return false;
822        };
823        if enderman.carried_block().is_some() {
824            return false;
825        }
826        mob.level().is_some_and(|world| {
827            world.get_game_rule(&MOB_GRIEFING)
828                && rand::random_range(0..reduced_tick_delay(TAKE_BLOCK_INTERVAL_TICKS)) == 0
829        })
830    }
831
832    fn tick(&mut self, mob: &dyn PathfinderMob) {
833        let (Some(enderman), Some(world)) = (mob.downcast_ref::<EndermanEntity>(), mob.level())
834        else {
835            return;
836        };
837
838        let position = mob.position();
839        let target = BlockPos::new(
840            (position.x - 2.0 + rand::random::<f64>() * 4.0).floor() as i32,
841            (position.y + rand::random::<f64>() * 3.0).floor() as i32,
842            (position.z - 2.0 + rand::random::<f64>() * 4.0).floor() as i32,
843        );
844        let state = world.get_block_state(target);
845        if !REGISTRY
846            .blocks
847            .is_in_tag(state.get_block(), &BlockTag::ENDERMAN_HOLDABLE)
848        {
849            return;
850        }
851
852        // Vanilla parity: the clip that stops an enderman reaching through a
853        // wall for the block behind it.
854        let block_pos = mob.block_position();
855        let from = DVec3::new(
856            f64::from(block_pos.x()) + 0.5,
857            f64::from(target.y()) + 0.5,
858            f64::from(block_pos.z()) + 0.5,
859        );
860        let to = block_center(target);
861        let hit = world.clip(from, to, ClipBlockShape::Outline, ClipFluid::None);
862        if hit.block_pos != target {
863            return;
864        }
865
866        world.remove_block(target, false);
867        world.game_event_at(
868            &vanilla_game_events::BLOCK_DESTROY,
869            to,
870            &GameEventContext::new(Some(mob), Some(state)),
871        );
872        enderman.set_carried_block(Some(state.get_block().default_state()));
873    }
874}
