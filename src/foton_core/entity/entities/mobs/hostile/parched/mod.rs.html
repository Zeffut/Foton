41/// Experience this mob drops.
42///
43/// Vanilla parity: the `this.xpReward = 5` of the `Monster` constructor, which
44/// every monster inherits and this one does not override.
45const XP_REWARD: i32 = 5;
46
47/// Ticks between shots on Hard.
48///
49/// Vanilla parity: `Parched.getHardAttackInterval`.
50const HARD_ATTACK_INTERVAL_TICKS: i32 = 50;
51
52/// Ticks between shots on every other difficulty.
53///
54/// Vanilla parity: `Parched.getAttackInterval`.
55const ATTACK_INTERVAL_TICKS: i32 = 70;
56
57/// Range within which a parched will loose an arrow.
58///
59/// Vanilla parity: the `attackRadius` of `AbstractSkeleton`'s bow goal.
60const ATTACK_RADIUS: f64 = 15.0;
61
62/// Speed of the arrows a parched fires.
63///
64/// Vanilla parity: the `1.6F` velocity of `performRangedAttack`.
65const ARROW_POWER: f32 = 1.6;
66
67/// Spread of the arrows a parched fires on normal difficulty.
68///
69/// Vanilla parity: `14 - difficulty * 4`, with difficulty 2.
70const ARROW_UNCERTAINTY: f32 = 6.0;
71
72/// Ticks of weakness a parched's arrow carries.
73///
74/// Vanilla parity: the `MobEffectInstance(MobEffects.WEAKNESS, 600)` of
75/// `Parched.getArrow`.
76const WEAKNESS_TICKS: i32 = 600;
77
78/// Speed multiplier while repositioning.
79const STROLL_SPEED_MODIFIER: f64 = 1.0;
80
81/// Speed the archer closes the distance at.
82///
83/// Vanilla parity: the `1.0` speed modifier `AbstractSkeleton` builds its bow
84/// goal with.
85const BOW_APPROACH_SPEED: f64 = 1.0;
86
87/// Distance at which a parched watches a player.
88const LOOK_AT_PLAYER_RANGE: f64 = 8.0;

90/// A parched.
91#[entity_behavior(class = "Parched")]
92pub struct ParchedEntity {
93    base: EntityBase,
94    entity_type: EntityTypeRef,
95    living_base: LivingEntityBase,
96    mob_base: MobBase,
97    entity_data: SyncMutex<ParchedEntityData>,
98}

105impl ParchedEntity {
106    /// Creates a parched at runtime.
107    #[must_use]
108    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
109        Self::new_with_base(
110            EntityBase::new(id, position, entity_type.dimensions, world),
111            entity_type,
112        )
113    }
114
115    /// Creates a parched from saved base data.
116    #[must_use]
117    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
118        Self::new_with_base(
119            EntityBase::from_load(load, entity_type.dimensions),
120            entity_type,
121        )
122    }
123
124    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
125        let living_base = LivingEntityBase::new(entity_type);
126        let mob_base = MobBase::new();
127        mob_base.set_xp_reward(XP_REWARD);
128        let mut entity_data = ParchedEntityData::new();
129        living_base.initialize_synced_data(&mut entity_data);
130
131        {
132            // A parched keeps the AbstractSkeleton goal set unchanged.
133            let mut goals = mob_base.goal_selector().lock();
134            goals.add_goal(2, RestrictSunGoal::new());
135            goals.add_goal(3, FleeSunGoal::new(1.0));
136            goals.add_goal(
137                4,
138                RangedBowAttackGoal::by_difficulty(
139                    HARD_ATTACK_INTERVAL_TICKS,
140                    ATTACK_INTERVAL_TICKS,
141                    ATTACK_RADIUS,
142                    BOW_APPROACH_SPEED,
143                    fire_arrow,
144                ),
145            );
146            goals.add_goal(5, WaterAvoidingRandomStrollGoal::new(STROLL_SPEED_MODIFIER));
147            goals.add_goal(6, LookAtPlayerGoal::new(LOOK_AT_PLAYER_RANGE));
148            goals.add_goal(6, RandomLookAroundGoal::new());
149            // TODO: vanilla also flees wolves at priority 3.
150        }
151
152        {
153            let mut targets = mob_base.target_selector().lock();
154            targets.add_goal(1, HurtByTargetGoal::new());
155            targets.add_goal(
156                2,
157                NearestAttackableTargetGoal::new_for_players(true, |_, _, _| true),
158            );
159        }
160
161        Self {
162            base,
163            entity_type,
164            living_base,
165            mob_base,
166            entity_data: SyncMutex::new(entity_data),
167        }
168    }
169}

171/// Looses a weakening arrow at `target`.
172///
173/// Vanilla parity: `AbstractSkeleton.performRangedAttack` with the
174/// `Parched.getArrow` override.
175fn fire_arrow(mob: &dyn PathfinderMob, target: DVec3, power: f32) {
176    let Some(archer) = mob.downcast_ref::<ParchedEntity>() else {
177        return;
178    };
179    let Some(world) = archer.level() else {
180        return;
181    };
182    // Vanilla parity: `AbstractSkeleton.performRangedAttack` reads the bow out
183    // of whichever hand holds one and hands it to `ProjectileUtil.getMobArrow`,
184    // so the arrow can read Power and Flame off it when it lands. Nothing
185    // leaves a quiver: `Monster.getProjectile` conjures the arrow when the mob
186    // carries none, which is why a skeleton never runs out.
187    let bow = archer.get_item_in_hand(weapon_holding_hand(archer, &vanilla_items::BOW));
188    let arrow = ArrowEntity::shoot_at(&world, archer, target, ARROW_POWER, ARROW_UNCERTAINTY);
189    // Vanilla parity: the `ProjectileUtil.getMobArrow(.., power, ..)`
190    // of `performRangedAttack` -- a shallower draw hits softer.
191    arrow.set_base_damage_from_mob(power);
192    if bow.is(&vanilla_items::BOW) {
193        arrow.set_fired_from_weapon(Some(bow));
194    }
195    // Vanilla parity: `Parched.getArrow` weakens every arrow it fires.
196    arrow.add_effect(MobEffectInstance::with_duration(
197        vanilla_mob_effects::WEAKNESS,
198        WEAKNESS_TICKS,
199        0,
200    ));
201
202    world.play_sound_at(
203        &sound_events::ENTITY_SKELETON_SHOOT,
204        SoundSource::Hostile,
205        archer.position(),
206        1.0,
207        0.4f32.mul_add(rand::random::<f32>(), 0.8).recip(),
208        None,
209    );
210}

263    fn set_health(&self, health: f32) {
264        let max_health = self.get_max_health();
265        let clamped = health.clamp(0.0, max_health);
266        self.entity_data
267            .lock()
268            .living_entity_mut()
269            .health
270            .set(clamped);
271    }

273    /// Refuses weakness, and nothing else.
274    ///
275    /// Vanilla parity: `Parched.canBeAffected`. A parched hands out the effect
276    /// it is immune to, which is what stops two of them disarming each other.
277    fn can_be_affected(&self, effect: &MobEffectInstance) -> bool {
278        if effect.effect() == vanilla_mob_effects::WEAKNESS {
279            return false;
280        }
281        self.default_can_be_affected(effect)
282    }

293impl Mob for ParchedEntity {
294    /// Arms the skeleton with the bow it shoots with.
295    ///
296    /// Vanilla parity: `AbstractSkeleton.finalizeSpawn`, which runs the shared
297    /// `Mob.finalizeSpawn` and then `populateDefaultEquipmentSlots` -- and that
298    /// is the only place a skeleton's bow comes from.
299    fn finalize_spawn(
300        &self,
301        world: &Arc<World>,
302        spawn_reason: EntitySpawnReason,
303        group_data: Option<SpawnGroupData>,
304    ) -> Option<SpawnGroupData> {
305        let group_data = self.finalize_spawn_mob_base(world, spawn_reason, group_data);
306        // Vanilla parity: the `setCanPickUpLoot` of
307        // `AbstractSkeleton.finalizeSpawn`, which is what lets a skeleton pick
308        // your bow up off the ground.
309        self.roll_spawn_can_pick_up_loot(world);
310        self.set_item_in_hand(
311            InteractionHand::MainHand,
312            ItemStack::new(&vanilla_items::BOW),
313        );
314        group_data
315    }
316    /// Vanilla parity: `Parched` derives from `AbstractSkeleton`, and so from
317    /// `Monster`.
318    fn is_monster(&self) -> bool {
319        true
320    }
321
322    /// Returns whether this mob accepts where the spawner put it.
323    ///
324    /// Vanilla parity: the `Monster::checkSurfaceMonstersSpawnRules` a parched
325    /// is registered with in `SpawnPlacements`, which is why one only appears
326    /// under open sky.
327    fn check_spawn_rules(
328        &self,
329        world: &Arc<World>,
330        spawn_reason: EntitySpawnReason,
331        pos: BlockPos,
332    ) -> bool {
333        check_surface_monster_spawn_rules(world, spawn_reason, pos)
334    }
335
336    fn mob_base(&self) -> &MobBase {
337        &self.mob_base
338    }
339
340    fn tick_goal_selectors(&self) {
341        PathfinderMob::tick_pathfinder_goal_selectors(self);
342    }
343
344    fn tick_path_navigation(&self) {
345        PathfinderMob::tick_pathfinder_path_navigation(self);
346    }
347
348    fn ambient_sound(&self) -> Option<SoundEventRef> {
349        Some(&sound_events::ENTITY_PARCHED_AMBIENT)
350    }
351
352    fn mob_flags(&self) -> i8 {
353        *self.entity_data.lock().mob().mob_flags.get()
354    }
355
356    fn set_mob_flags(&self, flags: i8) {
357        self.entity_data.lock().mob_mut().mob_flags.set(flags);
358    }
359}

382    /// A parched shoots weakness and is immune to it. If the override were
383    /// dropped, two parched shooting at each other would disarm one another.
384    ///
385    /// Slowness is the control: `Parched.canBeAffected` refuses weakness and
386    /// then defers, so everything the undead tags allow still gets through.
387    /// Poison would be a useless control -- a parched is undead, so
388    /// `ignores_poison_and_regen` already refuses that one.
389    #[test]
390    fn a_parched_refuses_weakness_but_still_takes_slowness() {
391        let mob = parched();
392
393        assert!(!mob.can_be_affected(&MobEffectInstance::with_duration(
394            vanilla_mob_effects::WEAKNESS,
395            600,
396            0
397        )));
398        assert!(mob.can_be_affected(&MobEffectInstance::with_duration(
399            vanilla_mob_effects::SLOWNESS,
400            600,
401            0
402        )));
403    }
404
405    /// The undead immunities `Parched` inherits rather than declares. If the
406    /// override forgot to defer to `default_can_be_affected` a poison would get
407    /// through.
408    #[test]
409    fn a_parched_still_inherits_the_undead_poison_immunity() {
410        let mob = parched();
411        assert!(!mob.can_be_affected(&MobEffectInstance::with_duration(
412            vanilla_mob_effects::POISON,
413            100,
414            0
415        )));
416    }
