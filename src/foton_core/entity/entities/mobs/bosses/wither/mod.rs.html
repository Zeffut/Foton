1//! The wither.
2//!
3//! Vanilla parity: `WitherBoss`. The fight has three parts. It arrives
4//! invulnerable and healing, counts down for eleven seconds and then blows a
5//! hole in whatever it was built in. It fights with three heads that aim
6//! independently -- the middle one follows the mob's own target, the two side
7//! ones each pick their own from anything alive nearby -- and it chews through
8//! the blocks around it every time it is hurt. Below half health it is
9//! *powered*: arrows and wind charges stop working on it, and it stops trying
10//! to climb above its target.
11//!
12//! **Gaps**: vanilla also overrides `getBlockExplosionResistance` on the
13//! charged skull, which is what lets the blue skull eat obsidian; that lives on
14//! [`WitherSkullEntity`](crate::entity::entities::WitherSkullEntity) and is
15//! already documented there as a missing `World::explode` hook. The head
16//! rotation arrays (`xRotHeads`/`yRotHeads`) are not kept: their only readers
17//! are `getHeadYRots`/`getHeadXRots`, which the client renderer calls, and the
18//! client recomputes them from the synced targets in its own `aiStep`.
19

160/// Returns whether a wither would shoot at this entity.
161///
162/// Vanilla parity: `WitherBoss.LIVING_ENTITY_SELECTOR`. `attackable` is what
163/// keeps an armor stand out of it: the stand overrides it to `false`, so the
164/// wither ignores stands without naming the class.
165fn is_wither_prey(target: &dyn LivingEntity) -> bool {
166    !REGISTRY
167        .entity_types
168        .is_in_tag(target.entity_type(), &EntityTypeTag::WITHER_FRIENDS)
169        && target.attackable()
170}

172/// Vanilla parity: `Level.ExplosionInteraction.MOB`, which the level resolves
173/// through `mobGriefing` and the drop-decay rule.
174fn mob_explosion_interaction(world: &Arc<World>) -> ExplosionBlockInteraction {
175    if world.get_game_rule(&MOB_GRIEFING) {
176        world.explosion_destroy_type(&MOB_EXPLOSION_DROP_DECAY)

178        ExplosionBlockInteraction::Keep

240    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
241        let living_base = LivingEntityBase::new(entity_type);
242        let mob_base = MobBase::new();
243        let mut entity_data = WitherBossEntityData::new();
244        // Vanilla parity: the `setHealth(getMaxHealth())` of the constructor,
245        // which is what `initialize_synced_data` does for every mob here.
246        living_base.initialize_synced_data(&mut entity_data);
247        mob_base.set_xp_reward(XP_REWARD);
248
249        {
250            // Keep vanilla WitherBoss goal priorities in the same order.
251            let mut goals = mob_base.goal_selector().lock();
252            goals.add_goal(0, WitherDoNothingGoal);
253            goals.add_goal(2, RangedAttackGoal::new(1.0, 40, 20.0, fire_wither_skull));
254            goals.add_goal(5, WaterAvoidingRandomFlyingGoal::new(1.0));
255            goals.add_goal(6, LookAtPlayerGoal::new(8.0));
256            goals.add_goal(7, RandomLookAroundGoal::new());
257        }
258
259        {
260            let mut targets = mob_base.target_selector().lock();
261            targets.add_goal(1, HurtByTargetGoal::new());
262            // Vanilla parity: `new NearestAttackableTargetGoal<>(this,
263            // LivingEntity.class, 0, false, false, LIVING_ENTITY_SELECTOR)` --
264            // it rescans every tick, needs neither sight nor a path, and takes
265            // anything alive that is not a wither friend.
266            targets.add_goal(
267                2,
268                NearestAttackableTargetGoal::new_with_interval(0, false, false, |_, target, _| {
269                    is_wither_prey(target)
270                }),
271            );
272        }
273
274        let boss_event = ServerBossEvent::with_random_id(
275            display_name_of(&base, entity_type),
276            BossBarColor::Purple,
277            BossBarOverlay::Progress,
278        );
279        boss_event.set_darken_screen(true);
280
281        Self {
282            base,
283            entity_type,
284            living_base,
285            mob_base,
286            entity_data: SyncMutex::new(entity_data),
287            boss_event,
288            next_head_update: SyncMutex::new([0; 2]),
289            idle_head_updates: SyncMutex::new([0; 2]),
290            destroy_blocks_tick: SyncMutex::new(0),
291        }
292    }

358    /// Returns where one head sits.
359    ///
360    /// Vanilla parity: `WitherBoss.getHeadX` / `getHeadY` / `getHeadZ`. Head
361    /// `0` is the body position itself. Vanilla offsets the side heads by
362    /// `180 * (index - 1)` degrees, so heads `1` and `3` share an angle; that
363    /// is vanilla's own arithmetic and the skull spawn positions follow it.
364    #[must_use]
365    pub fn head_position(&self, head: usize) -> DVec3 {
366        let position = self.position();
367        let scale = f64::from(self.get_scale());
368        let height = if head == 0 {
369            CENTER_HEAD_HEIGHT
370        } else {
371            SIDE_HEAD_HEIGHT
372        };
373        let y = position.y + f64::from(height) * scale;
374        if head == 0 {
375            return DVec3::new(position.x, y, position.z);
376        }
377
378        #[expect(
379            clippy::cast_precision_loss,
380            reason = "the head index is 1 or 2, exactly representable"
381        )]
382        let head_angle = (self.y_body_rot() + 180.0 * (head as f32 - 1.0)).to_radians();
383        DVec3::new(
384            position.x + f64::from(head_angle.cos()) * SIDE_HEAD_OFFSET * scale,
385            y,
386            position.z + f64::from(head_angle.sin()) * SIDE_HEAD_OFFSET * scale,
387        )
388    }

390    /// Spits a skull from one head at a living target.
391    ///
392    /// Vanilla parity: the two-argument `WitherBoss.performRangedAttack`. Only
393    /// the middle head can roll the charged blue skull.
394    fn perform_ranged_attack(&self, head: usize, target: &SharedEntity) {
395        let position = target.position();
396        let dangerous = head == 0 && rand::random::<f32>() < DANGEROUS_SKULL_CHANCE;
397        self.perform_ranged_attack_at(
398            head,
399            DVec3::new(
400                position.x,
401                position.y + target.get_eye_height() * 0.5,
402                position.z,
403            ),
404            dangerous,
405        );
406    }

408    /// Spits a skull from one head at a point.
409    ///
410    /// Vanilla parity: the five-argument `WitherBoss.performRangedAttack`.
411    fn perform_ranged_attack_at(&self, head: usize, target: DVec3, dangerous: bool) {
412        let Some(world) = self.level() else {
413            return;
414        };
415
416        if !self.is_silent() {
417            world.level_event(
418                level_events::SOUND_WITHER_BOSS_SHOOT,
419                self.block_position(),
420                0,
421                None,
422            );
423        }
424
425        let muzzle = self.head_position(head);
426        let skull = Arc::new(WitherSkullEntity::new(
427            &vanilla_entities::WITHER_SKULL,
428            next_entity_id(),
429            muzzle,
430            Arc::downgrade(&world),
431        ));
432        if dangerous {
433            skull.set_dangerous(true);
434        }
435
436        let direction = target - muzzle;
437        if let Some(owner) = world.get_entity_by_id(self.id()) {
438            skull.shoot_from_owner(&owner, direction);
439        } else {
440            // Only reachable before the wither itself is in the world, which
441            // no goal can do; the skull still needs a heading.
442            skull.set_rotation(self.rotation());
443            skull.assign_directional_movement(direction);
444        }
445
446        let entity: SharedEntity = skull;
447        if let Err(error) = world.try_add_entity(entity) {
448            log::debug!("wither failed to spit a skull: {error}");
449        }
450    }

452    /// Runs the eleven seconds between the summon and the fight.
453    ///
454    /// Vanilla parity: the `getInvulnerableTicks() > 0` branch of
455    /// `customServerAiStep`.
456    fn tick_arrival(&self, world: &Arc<World>) {
457        let remaining = self.invulnerable_ticks() - 1;
458        #[expect(
459            clippy::cast_precision_loss,
460            reason = "the countdown is 220 ticks, exactly representable"
461        )]
462        let progress = 1.0 - remaining as f32 / INVULNERABLE_TICKS as f32;
463        self.boss_event.set_progress(progress);
464
465        if remaining <= 0 {
466            let position = self.position();
467            world.explode(
468                ExplosionSpec::new(
469                    Some(self.id()),
470                    Some(self.id()),
471                    None,
472                    SPAWN_EXPLOSION_RADIUS,
473                    false,
474                    mob_explosion_interaction(world),
475                ),
476                DVec3::new(position.x, self.get_eye_y(), position.z),
477            );
478            if !self.is_silent() {
479                world.global_level_event(
480                    level_events::SOUND_WITHER_BOSS_SPAWN,
481                    self.block_position(),
482                    0,
483                );
484            }
485        }
486
487        self.set_invulnerable_ticks(remaining);
488        if self.tick_count() % 10 == 0 {
489            self.heal(SPAWN_HEAL);
490        }
491    }

493    /// Lets one side head reconsider what it is shooting at.
494    ///
495    /// Vanilla parity: one iteration of the `for (int i = 1; i < 3; i++)` loop
496    /// of `customServerAiStep`. `slot` is vanilla's `i`, so the head it drives
497    /// is `slot + 1` and the counters it keeps are at `slot - 1`.
498    fn tick_side_head(&self, world: &Arc<World>, slot: usize) {
499        let tick = self.tick_count();
500        if tick < self.next_head_update.lock()[slot - 1] {
501            return;
502        }
503        self.next_head_update.lock()[slot - 1] = tick + 10 + rand::random_range(0..10);
504
505        if matches!(world.difficulty(), Difficulty::Normal | Difficulty::Hard) {
506            let idle = self.idle_head_updates.lock()[slot - 1];
507            self.idle_head_updates.lock()[slot - 1] = idle + 1;
508            if idle > IDLE_UPDATES_BEFORE_BLIND_SHOT {
509                let position = self.position();
510                let blind_target = DVec3::new(
511                    rand::random_range(
512                        position.x - BLIND_SHOT_HORIZONTAL_RANGE
513                            ..position.x + BLIND_SHOT_HORIZONTAL_RANGE,
514                    ),
515                    rand::random_range(
516                        position.y - BLIND_SHOT_VERTICAL_RANGE
517                            ..position.y + BLIND_SHOT_VERTICAL_RANGE,
518                    ),
519                    rand::random_range(
520                        position.z - BLIND_SHOT_HORIZONTAL_RANGE
521                            ..position.z + BLIND_SHOT_HORIZONTAL_RANGE,
522                    ),
523                );
524                self.perform_ranged_attack_at(slot + 1, blind_target, true);
525                self.idle_head_updates.lock()[slot - 1] = 0;
526            }
527        }
528
529        let head_target = self.alternative_target(slot);
530        if head_target > 0 {
531            self.tick_committed_head(world, slot, head_target);
532            return;
533        }
534
535        let search = self.bounding_box().inflate_xyz(
536            HEAD_SCAN_INFLATION.x,
537            HEAD_SCAN_INFLATION.y,
538            HEAD_SCAN_INFLATION.z,
539        );
540        let conditions = TargetingConditions::for_combat()
541            .range(HEAD_TARGET_RANGE)
542            .selector(|_, target, _| is_wither_prey(target));
543        let candidates = world.get_entities_in_aabb_matching(&search, |entity| {
544            entity
545                .as_living_entity()
546                .is_some_and(|living| conditions.test(world, Some(self), living))
547        });
548        if candidates.is_empty() {
549            return;
550        }
551        let chosen = &candidates[rand::random_range(0..candidates.len())];
552        self.set_alternative_target(slot, chosen.id());
553    }

555    /// Fires at the target a side head already has, or drops it.
556    ///
557    /// Vanilla parity: the `headTarget > 0` branch of the head loop.
558    fn tick_committed_head(&self, world: &Arc<World>, slot: usize, head_target: i32) {
559        let current = world.get_entity_by_id(head_target).filter(|entity| {
560            entity.as_living_entity().is_some_and(|living| {
561                Mob::can_attack(self, living)
562                    && entity.position().distance_squared(self.position()) <= HEAD_ATTACK_RANGE_SQR
563                    && self.has_line_of_sight(entity.as_ref())
564            })
565        });
566
567        let Some(current) = current else {
568            self.set_alternative_target(slot, 0);
569            return;
570        };
571
572        self.perform_ranged_attack(slot + 1, &current);
573        self.next_head_update.lock()[slot - 1] = self.tick_count() + 40 + rand::random_range(0..20);
574        self.idle_head_updates.lock()[slot - 1] = 0;
575    }

577    /// Eats the blocks a hurt wither is standing in.
578    ///
579    /// Vanilla parity: the `destroyBlocksTick` branch of `customServerAiStep`.
580    fn tick_block_destruction(&self, world: &Arc<World>) {
581        let remaining = {
582            let mut destroy_blocks_tick = self.destroy_blocks_tick.lock();
583            if *destroy_blocks_tick <= 0 {
584                return;
585            }
586            *destroy_blocks_tick -= 1;
587            *destroy_blocks_tick
588        };
589        if remaining != 0 || !world.get_game_rule(&MOB_GRIEFING) {
590            return;
591        }
592
593        let dimensions = self.base().dimensions();
594        #[expect(
595            clippy::cast_possible_truncation,
596            reason = "vanilla floors the same two dimensions into ints"
597        )]
598        let half_width = (dimensions.width / 2.0 + 1.0).floor() as i32;
599        #[expect(
600            clippy::cast_possible_truncation,
601            reason = "vanilla floors the same two dimensions into ints"
602        )]
603        let height = dimensions.height.floor() as i32;
604        let origin = self.block_position();
605
606        let mut destroyed = false;
607        for pos in BlockPos::between_closed(
608            BlockPos::new(origin.x() - half_width, origin.y(), origin.z() - half_width),
609            BlockPos::new(
610                origin.x() + half_width,
611                origin.y() + height,
612                origin.z() + half_width,
613            ),
614        ) {
615            if can_destroy(world.get_block_state(pos)) {
616                destroyed = world.destroy_block_by_entity(pos, true, self) || destroyed;
617            }
618        }
619
620        if destroyed {
621            world.level_event(
622                level_events::SOUND_WITHER_BLOCK_BREAK,
623                self.block_position(),
624                0,
625                None,
626            );
627        }
628    }
629
630    /// Returns whether a hit lands at all.
631    ///
632    /// Vanilla parity: the guard chain at the top of `WitherBoss.hurtServer`,
633    /// split out so the counters it would otherwise bump stay in one place.
634    fn rejects_damage(&self, world: &World, source: &DamageSource) -> bool {
635        if self.is_invulnerable_to(world, source) {
636            return true;
637        }
638        if source.is(&DamageTypeTag::WITHER_IMMUNE_TO) {
639            return true;
640        }
641        if self.invulnerable_ticks() > 0 && !source.is(&DamageTypeTag::BYPASSES_INVULNERABILITY) {
642            return true;
643        }
644        // Vanilla parity: `source.getEntity() instanceof WitherBoss`, then the
645        // separate `WITHER_FRIENDS` test on the same entity.
646        let causing = source
647            .causing_entity_id
648            .and_then(|id| world.get_entity_by_id(id));
649        if let Some(causing) = &causing {
650            if causing.downcast_ref::<Self>().is_some() {
651                return true;
652            }
653            if REGISTRY
654                .entity_types
655                .is_in_tag(causing.entity_type(), &EntityTypeTag::WITHER_FRIENDS)
656            {
657                return true;
658            }
659        }
660
661        // Vanilla parity: a powered wither shrugs off arrows and wind charges.
662        // Foton has no `AbstractArrow` layer, so its two concrete subclasses
663        // are named; a third would have to be added here.
664        if self.is_powered()
665            && let Some(direct) = source
666                .direct_entity_id
667                .and_then(|id| world.get_entity_by_id(id))
668            && (direct.downcast_ref::<ArrowEntity>().is_some()
669                || direct.downcast_ref::<ThrownTridentEntity>().is_some()
670                || direct.downcast_ref::<WindChargeEntity>().is_some())
671        {
672            return true;
673        }
674
675        false
676    }
677}

810    /// Vanilla parity: `WitherBoss.aiStep`, the half that moves the mob. The
811    /// rest of vanilla's override is the head rotation arrays and the smoke,
812    /// both of which the client owns.
813    fn ai_step(&self) -> Option<MoveResult> {
814        let mut movement = self.velocity() * DVec3::new(1.0, 0.6, 1.0);
815
816        let chased = self.level().and_then(|world| {
817            let target_id = self.alternative_target(0);
818            (target_id > 0)
819                .then(|| world.get_entity_by_id(target_id))
820                .flatten()
821        });
822        if let Some(chased) = chased {
823            let position = self.position();
824            let target_position = chased.position();
825            let mut vertical = movement.y;
826            if position.y < target_position.y
827                || !self.is_powered() && position.y < target_position.y + 5.0
828            {
829                vertical = vertical.max(0.0);
830                vertical += 0.3 - vertical * 0.6;
831            }
832            movement = DVec3::new(movement.x, vertical, movement.z);
833
834            let to_target = DVec3::new(
835                target_position.x - position.x,
836                0.0,
837                target_position.z - position.z,
838            );
839            if to_target.length_squared() > CHASE_DISTANCE_SQR {
840                let heading = to_target.normalize();
841                movement += DVec3::new(
842                    heading.x * 0.3 - movement.x * 0.6,
843                    0.0,
844                    heading.z * 0.3 - movement.z * 0.6,
845                );
846            }
847        }
848
849        self.set_velocity(movement);
850        if movement.x * movement.x + movement.z * movement.z > FACING_SPEED_SQR {
851            #[expect(
852                clippy::cast_possible_truncation,
853                reason = "vanilla casts the same angle to a float"
854            )]
855            let yaw = movement.z.atan2(movement.x).to_degrees() as f32 - 90.0;
856            self.set_rotation((yaw, self.rotation().1));
857        }
858
859        self.default_ai_step()
860    }

959        };

979        }
