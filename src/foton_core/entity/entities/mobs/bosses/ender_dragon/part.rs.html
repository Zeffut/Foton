1//! The dragon's hitboxes.
2//!
3//! Vanilla parity: `EnderDragonPart`. The dragon is not one box. It is eight,
4//! and every hit a player lands arrives addressed to one of them rather than to
5//! the dragon: the head takes full damage, everything else a quarter. A part is
6//! a real `Entity` -- it needs a position, a bounding box and a network ID,
7//! because the attack and interact packets carry an entity ID and the server
8//! range-checks against the box that ID names.
9//!
10//! Three things make a part unlike every other entity in the tree:
11//!
12//! * **It is never spawned.** No part is added to the world entity manager, so
13//!   nothing ticks it, saves it, or tracks it. Vanilla is the same: `ChunkMap`
14//!   skips `EnderDragonPart` in `addEntity`, and `getAddEntityPacket` throws.
15//!   The client builds all eight itself from the dragon's spawn packet.
16//! * **Its ID is not its own.** The client assigns `dragonId + 1 ..= + 8` in
17//!   `EnderDragon.recreateFromPacket`, so the server has to hand out exactly
18//!   that block. See [`reserve_entity_ids`](crate::entity::reserve_entity_ids).
19//! * **It cannot be hurt.** [`Entity::hurt`] forwards to the dragon, which is
20//!   the whole point of the part existing.
21//!
22//! **Gaps**: vanilla overrides `getPickResult` to return the dragon's, and `is`
23//! so that a part counts as its parent in identity tests. Foton implements
24//! neither hook yet -- there is no `ServerboundPickItemFromEntityPacket`
25//! handler, and no `Entity::is` -- so there is nothing to override.
26

38/// Vanilla `EntityDimensions.defaultEyeHeight`.
39const EYE_HEIGHT_FACTOR: f32 = 0.85;
40
41/// Hitboxes a dragon has.
42///
43/// Vanilla parity: the length of `EnderDragon.subEntities`.
44pub const PART_COUNT: usize = 8;
45
46/// One of the dragon's eight hitboxes.
47pub struct EnderDragonPart {
48    base: EntityBase,
49    entity_type: EntityTypeRef,
50    /// Network ID of the dragon this hitbox belongs to.
51    ///
52    /// Vanilla holds `parentMob` directly. Foton cannot: the dragon's `Arc`
53    /// does not exist yet while its constructor is building the parts, so the
54    /// link is stored as an ID and resolved through the world the same way a
55    /// projectile resolves its owner.
56    parent_id: i32,
57    /// Which of [`EnderDragon`]'s parts this is.
58    index: DragonPartIndex,
59}
60
61/// Which hitbox a part is.
62///
63/// Vanilla distinguishes them by `name` for the renderer and by reference
64/// identity in `EnderDragon.hurt`, where only `part != this.head` matters.
65#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
66pub enum DragonPartIndex {
67    /// Vanilla parity: `EnderDragon.head`.
68    Head,
69    /// Vanilla parity: `EnderDragon.neck`.
70    Neck,
71    /// Vanilla parity: `EnderDragon.body`.
72    Body,
73    /// Vanilla parity: `EnderDragon.tail1`.
74    Tail1,
75    /// Vanilla parity: `EnderDragon.tail2`.
76    Tail2,
77    /// Vanilla parity: `EnderDragon.tail3`.
78    Tail3,
79    /// Vanilla parity: `EnderDragon.wing1`.
80    Wing1,
81    /// Vanilla parity: `EnderDragon.wing2`.
82    Wing2,
83}
84
85impl DragonPartIndex {
86    /// The eight parts in the order vanilla builds them.
87    ///
88    /// Vanilla parity: the `subEntities` array. The order is load-bearing --
89    /// the client derives each part's ID from its index in it.
90    pub const ORDER: [Self; PART_COUNT] = [
91        Self::Head,
92        Self::Neck,
93        Self::Body,
94        Self::Tail1,
95        Self::Tail2,
96        Self::Tail3,
97        Self::Wing1,
98        Self::Wing2,
99    ];
100
101    /// Returns the part's slot in [`Self::ORDER`].
102    #[must_use]
103    pub const fn slot(self) -> usize {
104        match self {
105            Self::Head => 0,
106            Self::Neck => 1,
107            Self::Body => 2,
108            Self::Tail1 => 3,
109            Self::Tail2 => 4,
110            Self::Tail3 => 5,
111            Self::Wing1 => 6,
112            Self::Wing2 => 7,
113        }
114    }
115
116    /// Returns the part's vanilla name.
117    ///
118    /// Vanilla parity: the `name` each part is constructed with. The three tail
119    /// segments share one name, and so do the two wings.
120    #[must_use]
121    pub const fn name(self) -> &'static str {
122        match self {
123            Self::Head => "head",
124            Self::Neck => "neck",
125            Self::Body => "body",
126            Self::Tail1 | Self::Tail2 | Self::Tail3 => "tail",
127            Self::Wing1 | Self::Wing2 => "wing",
128        }
129    }
130
131    /// Returns the part's width and height.
132    ///
133    /// Vanilla parity: the two floats each `new EnderDragonPart` is given.
134    #[must_use]
135    pub const fn size(self) -> (f32, f32) {
136        match self {
137            Self::Head => (1.0, 1.0),
138            Self::Neck => (3.0, 3.0),
139            Self::Body => (5.0, 3.0),
140            Self::Tail1 | Self::Tail2 | Self::Tail3 => (2.0, 2.0),
141            Self::Wing1 | Self::Wing2 => (4.0, 2.0),
142        }
143    }
144
145    /// Returns the part's bounding-box dimensions.
146    ///
147    /// Vanilla parity: `EntityDimensions.scalable(w, h)`.
148    #[must_use]
149    pub fn dimensions(self) -> EntityDimensions {
150        let (width, height) = self.size();
151        EntityDimensions::new(width, height, height * EYE_HEIGHT_FACTOR)
152    }
153}

158}

160impl EnderDragonPart {
161    /// Builds one hitbox for a dragon.
162    ///
163    /// `id` must come from the block the dragon reserved, and `parent_id` must
164    /// be the dragon's own ID; the client reconstructs the relationship from
165    /// nothing but the arithmetic between them.
166    #[must_use]
167    pub fn new(
168        entity_type: EntityTypeRef,
169        id: i32,
170        parent_id: i32,
171        index: DragonPartIndex,
172        position: DVec3,
173        world: Weak<World>,
174    ) -> Self {
175        Self {
176            base: EntityBase::new(id, position, index.dimensions(), world),
177            entity_type,
178            parent_id,
179            index,
180        }
181    }
182
183    /// Returns which hitbox this is.
184    #[must_use]
185    pub const fn index(&self) -> DragonPartIndex {
186        self.index
187    }
188
189    /// Returns the network ID of the dragon this hitbox belongs to.
190    #[must_use]
191    pub const fn parent_id(&self) -> i32 {
192        self.parent_id
193    }
194
195    /// Moves the hitbox.
196    ///
197    /// Vanilla parity: the `part.setPos` of `EnderDragon.tickPart`. A part is
198    /// not registered with the world entity manager, so this is a local move --
199    /// there is no spatial index entry to keep in step.
200    pub fn set_part_position(&self, position: DVec3) {
201        self.base.set_position_local(position);
202    }
203
204    /// Returns a height `progress` of the way up the hitbox.
205    ///
206    /// Vanilla parity: `Entity.getY(double)`. The dragon's phases aim from
207    /// `head.getY(0.5)` -- the middle of the head, not its feet.
208    #[must_use]
209    pub fn y_at(&self, progress: f64) -> f64 {
210        let (_, height) = self.index.size();
211        self.position().y + f64::from(height) * progress
212    }
213
214    /// Runs `action` against the dragon this hitbox belongs to.
215    ///
216    /// Returns `None` when the dragon is no longer live, which is the state a
217    /// part is left in for the rest of the tick that removed its parent.
218    pub fn with_parent<R>(
219        &self,
220        world: &World,
221        action: impl FnOnce(&EnderDragon) -> R,
222    ) -> Option<R> {
223        let parent = world.get_entity_by_id(self.parent_id)?;
224        let dragon = parent.downcast_ref::<EnderDragon>()?;
225        Some(action(dragon))
226    }
227}
228
229impl Entity for EnderDragonPart {
230    fn base(&self) -> &EntityBase {
231        &self.base
232    }
233
234    /// Vanilla parity: `super(parentMob.getType(), parentMob.level())` -- a
235    /// part reports the dragon's own entity type.
236    fn entity_type(&self) -> EntityTypeRef {
237        self.entity_type
238    }
239
240    /// Vanilla parity: `EnderDragonPart.isPickable` returns true, which is what
241    /// makes the hitbox the thing a player's crosshair and attack packet find.
242    fn is_pickable(&self) -> bool {
243        true
244    }
245
246    /// Vanilla parity: `EnderDragonPart.hurtServer`. This is the routing the
247    /// whole part mechanism exists for -- the hit arrives on the hitbox and is
248    /// resolved against the dragon behind it.
249    fn hurt(&self, world: &World, source: &DamageSource, amount: f32) -> bool {
250        if self.is_invulnerable_to_base(source) {
251            return false;
252        }
253
254        self.with_parent(world, |dragon| {
255            dragon.hurt_part(world, self.index, source, amount)
256        })
257        .unwrap_or(false)
258    }
259}

261#[cfg(test)]
262mod tests {
263    use super::*;
264
265    use foton_registry::vanilla_entities;
266
267    #[test]
268    fn the_head_is_the_smallest_hitbox_and_the_body_the_widest() {
269        assert_eq!(DragonPartIndex::Head.size(), (1.0, 1.0));
270        assert_eq!(DragonPartIndex::Body.size(), (5.0, 3.0));
271    }
272
273    #[test]
274    fn part_slots_match_the_order_the_client_derives_ids_from() {
275        for (slot, part) in DragonPartIndex::ORDER.into_iter().enumerate() {
276            assert_eq!(part.slot(), slot);
277        }
278    }
279
280    #[test]
281    fn a_hitbox_reports_the_dragons_entity_type_rather_than_one_of_its_own() {
282        let part = EnderDragonPart::new(
283            &vanilla_entities::ENDER_DRAGON,
284            42,
285            41,
286            DragonPartIndex::Head,
287            DVec3::new(8.5, 64.0, 8.5),
288            Weak::new(),
289        );
290
291        assert_eq!(part.entity_type(), &vanilla_entities::ENDER_DRAGON);
292        assert_eq!(part.parent_id(), 41);
293    }
294
295    #[test]
296    fn a_hitbox_bounding_box_is_centered_on_its_position_and_sized_from_its_part() {
297        let part = EnderDragonPart::new(
298            &vanilla_entities::ENDER_DRAGON,
299            42,
300            41,
301            DragonPartIndex::Body,
302            DVec3::new(8.5, 64.0, 8.5),
303            Weak::new(),
304        );
305
306        let aabb = part.bounding_box();
307        assert!((aabb.max_x() - aabb.min_x() - 5.0).abs() < 1.0e-9);
308        assert!((aabb.max_y() - aabb.min_y() - 3.0).abs() < 1.0e-9);
309        assert!((aabb.min_y() - 64.0).abs() < 1.0e-9);
310    }
