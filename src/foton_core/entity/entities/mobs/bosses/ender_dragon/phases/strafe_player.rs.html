23/// Ticks of clean line-of-sight before the fireball is let go.
24///
25/// Vanilla parity: `DragonStrafePlayerPhase.FIREBALL_CHARGE_AMOUNT`.
26const FIREBALL_CHARGE_AMOUNT: i32 = 5;
27
28/// Squared distance inside which the dragon bothers charging a fireball.
29///
30/// Vanilla parity: the `distanceToSqr(this.dragon) < 4096.0` of `doServerTick`.
31const FIREBALL_RANGE_SQR: f64 = 4096.0;
32
33/// How far off aim the dragon may be and still fire, in degrees.
34///
35/// Vanilla parity: the `angleDegs >= 0.0F && angleDegs < 10.0F` of the same.
36const FIREBALL_AIM_TOLERANCE_DEGREES: f32 = 10.0;
37
38/// How far in front of the head the fireball starts.
39///
40/// Vanilla parity: the `1.0` the view vector is scaled by.
41const FIREBALL_SPAWN_OFFSET: f64 = 1.0;

43/// The dragon flies at one player and spits at them.
44///
45/// Vanilla parity: `DragonStrafePlayerPhase`.
46pub struct DragonStrafePlayerPhase {

50struct StrafeState {

53    target_location: Option<DVec3>,

56}

79    /// Aims the strafe at one player and paths to them.
80    ///
81    /// Vanilla parity: `DragonStrafePlayerPhase.setTarget`.
82    pub fn set_target(&self, dragon: &EnderDragon, target: &Arc<Player>) {
83        let Some(world) = dragon.level() else {
84            return;
85        };
86
87        let target_position = target.position();
88        let current_node = dragon.find_closest_node_to_self(&world);
89        let target_node = dragon.find_closest_node(
90            &world,
91            target_position.x,
92            target_position.y,
93            target_position.z,
94        );
95
96        let final_x = fast_floor(target_position.x);
97        let final_z = fast_floor(target_position.z);
98        let dragon_position = dragon.position();
99        let horizontal =
100            (f64::from(final_x) - dragon_position.x).hypot(f64::from(final_z) - dragon_position.z);
101        let height_offset = strafe_height_offset(horizontal);
102        let final_y = fast_floor(target_position.y + height_offset);
103        let final_node = Node::new(final_x, final_y, final_z);
104
105        let mut state = self.state.lock();
106        state.attack_target = Some(target.clone());
107        state.current_path = dragon.find_path(&world, current_node, target_node, Some(final_node));
108        if let Some(path) = state.current_path.as_mut() {
109            path.advance();
110            if let Some(location) = navigate_to_next_path_node(path) {
111                state.target_location = Some(location);
112            }
113        }
114    }
115
116    /// Vanilla parity: `DragonStrafePlayerPhase.findNewTarget`.
117    fn find_new_target(&self, dragon: &EnderDragon, world: &Arc<World>) {
118        let mut state = self.state.lock();
119        if state.current_path.as_ref().is_none_or(Path::is_done) {
120            let current_node = dragon.find_closest_node_to_self(world);
121            let mut target_node = current_node as i32;
122            if rand::random_range(0..8) == 0 {
123                state.holding_pattern_clockwise = !state.holding_pattern_clockwise;
124                target_node += 6;
125            }
126
127            if state.holding_pattern_clockwise {
128                target_node += 1;
129            } else {
130                target_node -= 1;
131            }
132
133            let target_node = wrap_ring_target(
134                target_node,
135                dragon.has_fight() && dragon.alive_crystals() > 0,
136            );
137            state.current_path = dragon.find_path(world, current_node, target_node, None);
138            if let Some(path) = state.current_path.as_mut() {
139                path.advance();
140            }
141        }
142
143        if let Some(path) = state.current_path.as_mut()
144            && let Some(location) = navigate_to_next_path_node(path)
145        {
146            state.target_location = Some(location);
147        }
148    }

150    /// Spits a fireball at the target and returns to the holding pattern.
151    ///
152    /// Vanilla parity: the innermost branch of `doServerTick`.
153    fn shoot_fireball(&self, dragon: &EnderDragon, world: &Arc<World>, target: &Arc<Player>) {
154        let view = dragon.look_angle();
155        let head = dragon.head();
156        let start = DVec3::new(
157            head.position().x - view.x * FIREBALL_SPAWN_OFFSET,
158            head.y_at(0.5) + 0.5,
159            head.position().z - view.z * FIREBALL_SPAWN_OFFSET,
160        );
161        let direction = DVec3::new(
162            target.position().x - start.x,
163            super::y_at(target.as_ref(), 0.5) - start.y,
164            target.position().z - start.z,
165        );
166
167        if !dragon.is_silent() {
168            world.level_event(
169                level_events::SOUND_DRAGON_FIREBALL,
170                dragon.block_position(),
171                0,
172                None,
173            );
174        }
175
176        let fireball = Arc::new(DragonFireballEntity::new(
177            &vanilla_entities::DRAGON_FIREBALL,
178            next_entity_id(),
179            start,
180            Arc::downgrade(world),
181        ));
182        // Vanilla parity: `new DragonFireball(level, this.dragon, direction)`
183        // then `snapTo(startingX, startingY, startingZ, 0, 0)`. The constructor
184        // is what sets the owner and the heading; Foton spawns at the muzzle and
185        // shoots from there, the same way the wither spits a skull.
186        if let Some(owner) = world.get_entity_by_id(dragon.id()) {
187            fireball.shoot_from_owner(&owner, direction);
188        } else {
189            fireball.set_rotation((0.0, 0.0));
190            fireball.assign_directional_movement(direction);
191        }
192
193        let entity: SharedEntity = fireball;
194        if let Err(error) = world.try_add_entity(entity) {
195            log::debug!("failed to spawn the dragon's fireball: {error}");
196        }
197
198        let mut state = self.state.lock();
199        state.fireball_charge = 0;
200        if let Some(path) = state.current_path.as_mut() {
201            while !path.is_done() {
202                path.advance();
203            }
204        }
205        drop(state);
206
207        dragon
208            .phase_manager()
209            .set_phase(dragon, EnderDragonPhase::HoldingPattern);
210    }
211}

213impl DragonPhaseInstance for DragonStrafePlayerPhase {
214    fn phase(&self) -> EnderDragonPhase {
215        EnderDragonPhase::StrafePlayer
216    }
217
218    fn do_server_tick(&self, dragon: &EnderDragon, world: &Arc<World>) {
219        let Some(target) = self.state.lock().attack_target.clone() else {
220            log::warn!("Skipping player strafe phase because no player was found");
221            dragon
222                .phase_manager()
223                .set_phase(dragon, EnderDragonPhase::HoldingPattern);
224            return;
225        };
226
227        let dragon_position = dragon.position();
228        let target_position = target.position();
229        {
230            let mut state = self.state.lock();
231            if state.current_path.as_ref().is_some_and(Path::is_done) {
232                let horizontal = (target_position.x - dragon_position.x)
233                    .hypot(target_position.z - dragon_position.z);
234                state.target_location = Some(DVec3::new(
235                    target_position.x,
236                    target_position.y + strafe_height_offset(horizontal),
237                    target_position.z,
238                ));
239            }
240        }
241
242        let dist_to_target = self
243            .state
244            .lock()
245            .target_location
246            .map_or(0.0, |location| location.distance_squared(dragon_position));
247        if dist_to_target < ARRIVED_DISTANCE_SQR || dist_to_target > LOST_DISTANCE_SQR {
248            self.find_new_target(dragon, world);
249        }
250
251        let in_range = target_position.distance_squared(dragon_position) < FIREBALL_RANGE_SQR;
252        if !in_range || !dragon.has_line_of_sight(target.as_ref()) {
253            let mut state = self.state.lock();
254            if state.fireball_charge > 0 {
255                state.fireball_charge -= 1;
256            }
257            return;
258        }
259
260        let charge = {
261            let mut state = self.state.lock();
262            state.fireball_charge += 1;
263            state.fireball_charge
264        };
265
266        let y_rot = f64::from(dragon.rotation().0).to_radians();
267        let aim = DVec3::new(
268            target_position.x - dragon_position.x,
269            0.0,
270            target_position.z - dragon_position.z,
271        )
272        .normalize_or_zero();
273        let dir = DVec3::new(
274            f64::from(trig::sin(y_rot)),
275            0.0,
276            -f64::from(trig::cos(y_rot)),
277        )
278        .normalize_or_zero();
279        let angle_degrees = (dir.dot(aim) as f32).acos().to_degrees() + 0.5;
280
281        if charge >= FIREBALL_CHARGE_AMOUNT
282            && (0.0..FIREBALL_AIM_TOLERANCE_DEGREES).contains(&angle_degrees)
283        {
284            self.shoot_fireball(dragon, world, &target);
285        }
286    }
287
288    fn begin(&self, _dragon: &EnderDragon) {
289        let mut state = self.state.lock();
290        state.fireball_charge = 0;
291        state.target_location = None;
292        state.current_path = None;
293        state.attack_target = None;
294    }
295
296    fn fly_target_location(&self) -> Option<DVec3> {
297        self.state.lock().target_location
298    }
299
300    fn as_strafe_player(&self) -> Option<&Self> {
301        Some(self)
302    }
303}

305/// How far above the target the dragon aims while strafing.
306///
307/// Vanilla parity: the `Math.min(0.4F + dist / 80.0 - 1.0, 10.0)` that both
308/// `setTarget` and `doServerTick` compute.
309fn strafe_height_offset(horizontal_distance: f64) -> f64 {
310    (f64::from(0.4_f32) + horizontal_distance / 80.0 - 1.0).min(10.0)
