1//! The ender dragon.
2//!
3//! Vanilla parity: `EnderDragon`. Almost nothing about it is a normal mob. It
4//! has no goals -- a [phase](phases) decides everything it does. It has no
5//! collision box of its own: it is [eight hitboxes](EnderDragonPart) that
6//! follow its body a few ticks behind, and every hit a player lands arrives
7//! addressed to one of them. It does not pathfind through the world but between
8//! [twenty-four fixed points](path) in the sky. And it does not die when its
9//! health runs out; it flies to the exit portal first.
10//!
11//! Two of those are why this file exists at all, and why the entity manager and
12//! the interact handler had to learn about parts. See [`part`] for the details.
13//!
14//! The dragon owns none of the fight around it. The boss bar, the crystal
15//! count, the exit portal and the experience all belong to
16//! [`EnderDragonFight`], which the End hangs on its [`World`]; a dragon
17//! summoned anywhere else has no bar and no crystals, exactly as in vanilla.
18//!
19//! **Gaps**, all of them things nothing in the tree can yet express:
20//!
21//! * `applyEffectsFromBlocks`, and the `interpolation` the client half of
22//!   `aiStep` drives, are not carried.
23//! * The growl and flap sounds, the death particles and the `onFlap` hook are
24//!   client-local in vanilla and have no server work.

218    /// Creates a dragon at runtime.
219    ///
220    /// **The `id` the entity registry hands in is discarded.** A dragon needs
221    /// [`ENDER_DRAGON_ID_BLOCK`] consecutive IDs, because the client derives its
222    /// eight hitboxes as `dragonId + 1 ..= dragonId + 8` and has nothing else to
223    /// go on; a single ID drawn by the caller cannot promise the eight after it
224    /// are free. So the dragon reserves its own block and takes the first of it,
225    /// and every caller reads the ID back off the entity, as they already do.
226    ///
227    /// Vanilla has the same shape for the same reason: an `EnderDragon`'s ID and
228    /// its parts' IDs all come from inside its own constructor, one after
229    /// another, never from the code that asked for a dragon.
230    #[must_use]
231    pub fn new(entity_type: EntityTypeRef, _id: i32, position: DVec3, world: Weak<World>) -> Self {
232        let id = reserve_entity_ids(ENDER_DRAGON_ID_BLOCK);
233        Self::new_with_base(
234            EntityBase::new(id, position, entity_type.dimensions, world.clone()),
235            entity_type,
236            position,
237            world,
238        )
239    }

244    /// are session-local and never persisted, so nothing is lost by it.

246    pub fn from_saved(entity_type: EntityTypeRef, mut load: EntityBaseLoad) -> Self {

249        let world = load.world.clone();

251            EntityBase::from_load(load, entity_type.dimensions),
252            entity_type,

254            world,
255        )
256    }

258    fn new_with_base(
259        base: EntityBase,
260        entity_type: EntityTypeRef,
261        position: DVec3,
262        world: Weak<World>,
263    ) -> Self {
264        let id = base.id();
265        let parts = DragonPartIndex::ORDER.map(|index| {
266            Arc::new(EnderDragonPart::new(
267                entity_type,
268                id + 1 + index.slot() as i32,
269                id,
270                index,
271                position,
272                world.clone(),
273            ))
274        });
275
276        let living_base = LivingEntityBase::new(entity_type);
277        let mut entity_data = EnderDragonEntityData::new();
278        // Vanilla parity: the `setHealth(getMaxHealth())` of the constructor.
279        living_base.initialize_synced_data(&mut entity_data);
280
281        Self {
282            base,
283            entity_type,
284            living_base,
285            mob_base: MobBase::new(),
286            entity_data: SyncMutex::new(entity_data),
287            parts,
288            flight_history: SyncMutex::new(DragonFlightHistory::new()),
289            phase_manager: EnderDragonPhaseManager::new(),
290            pathfinder: SyncMutex::new(DragonPathfinder::new()),
291            state: SyncMutex::new(DragonState {
292                dragon_death_time: 0,
293                y_rot_a: 0.0,
294                sitting_damage_received: 0.0,
295                in_wall: false,
296                fight_origin: BlockPos::new(0, 0, 0),
297                nearest_crystal: None,
298            }),
299        }

417    /// Returns which way the dragon's head is pointing.
418    ///
419    /// Vanilla parity: `EnderDragon.getHeadLookVector`. Vanilla writes a
420    /// temporary pitch onto the entity and reads the view vector back; the
421    /// pitch is substituted directly here instead, which avoids a visible
422    /// rotation flicker for anything reading the dragon mid-call.
423    #[must_use]
424    pub fn head_look_vector(&self, _partial_tick: f32) -> DVec3 {
425        let (y_rot, x_rot) = self.rotation();
426        let phase = self.phase_manager.current_phase();
427        let instance = self.phase_manager.current_instance();
428
429        let pitch = match phase {
430            EnderDragonPhase::Landing | EnderDragonPhase::Takeoff => {
431                let Some(world) = self.level() else {
432                    return self.calculate_view_vector(x_rot, y_rot);
433                };
434                let egg = world.heightmap_pos(
435                    HeightmapType::MotionBlockingNoLeaves,
436                    end_podium_location(self.fight_origin()),
437                );
438                let dist =
439                    (phases::dist_to_center_sqr(egg, self.position()).sqrt() as f32 / 4.0).max(1.0);
440                let y_offset = 6.0 / dist;
441                -y_offset * 1.5 * 5.0
442            }
443            _ if instance.is_sitting() => -45.0,
444            _ => return self.calculate_view_vector(x_rot, y_rot),
445        };
446
447        self.calculate_view_vector(pitch, y_rot)
448    }

489    /// Resolves a hit that arrived on one of the hitboxes.
490    ///
491    /// Vanilla parity: `EnderDragon.hurt(ServerLevel, EnderDragonPart,
492    /// DamageSource, float)`. This is where the parts pay off: anything that is
493    /// not the head has its damage cut to a quarter plus a point, which is why
494    /// killing a dragon means hitting it in the face.
495    pub fn hurt_part(
496        &self,
497        world: &World,
498        part: DragonPartIndex,
499        source: &DamageSource,
500        damage: f32,
501    ) -> bool {
502        if self.phase_manager.current_phase() == EnderDragonPhase::Dying {
503            return false;
504        }
505
506        let mut damage = self
507            .phase_manager
508            .current_instance()
509            .on_hurt(self, source, damage);
510        if part != DragonPartIndex::Head {
511            damage = damage / 4.0 + damage.min(1.0);
512        }
513
514        if damage < MINIMUM_DAMAGE {
515            return false;
516        }
517
518        let from_player = source
519            .causing_entity_id
520            .and_then(|id| world.get_entity_by_id(id))
521            .is_some_and(|causing| causing.entity_type() == &vanilla_entities::PLAYER);
522        if !from_player && !source.is(&DamageTypeTag::ALWAYS_HURTS_ENDER_DRAGONS) {
523            // Vanilla parity: `hurt` still reports true here. A hit from
524            // something that is neither a player nor tagged is swallowed, but
525            // the caller is told it landed.
526            return true;
527        }
528
529        let health_before = self.get_health();
530        self.living_hurt_server(world, source, damage);
531
532        if self.phase_manager.current_instance().is_sitting() {
533            let taken = health_before - self.get_health();
534            let takeoff = {
535                let mut state = self.state.lock();
536                state.sitting_damage_received += taken;
537                if state.sitting_damage_received
538                    > SITTING_ALLOWED_DAMAGE_PERCENTAGE * self.get_max_health()
539                {
540                    state.sitting_damage_received = 0.0;
541                    true
542                } else {
543                    false
544                }
545            };
546            if takeoff {
547                self.phase_manager
548                    .set_phase(self, EnderDragonPhase::Takeoff);
549            }
550        }
551
552        true
553    }

555    /// Reacts to one of the pillar crystals blowing up.
556    ///
557    /// Vanilla parity: `EnderDragon.onCrystalDestroyed`. The dragon takes the
558    /// blast in the face if it was the crystal it was healing from, then lets
559    /// the current phase decide whether to come after whoever did it.
560    pub fn on_crystal_destroyed(
561        &self,
562        world: &Arc<World>,
563        crystal: &EndCrystalEntity,
564        pos: BlockPos,
565        source: &DamageSource,
566        blamed: Option<&Arc<Player>>,
567    ) {
568        let player = blamed.cloned().or_else(|| {
569            world.nearest_player(
570                DVec3::new(f64::from(pos.x()), f64::from(pos.y()), f64::from(pos.z())),
571                CRYSTAL_DESTROY_RANGE,
572                LivingEntity::can_be_seen_as_enemy,
573            )
574        });
575
576        if self.state.lock().nearest_crystal == Some(crystal.id()) {
577            let mut explosion = DamageSource::environment(&vanilla_damage_types::EXPLOSION)
578                .with_direct_entity(crystal.id());
579            if let Some(player) = player.as_ref() {
580                explosion = explosion.with_causing_entity(player.id());
581            }
582            self.hurt_part(
583                world,
584                DragonPartIndex::Head,
585                &explosion,
586                CRYSTAL_EXPLOSION_DAMAGE,
587            );
588        }
589
590        self.phase_manager.current_instance().on_crystal_destroyed(
591            self,
592            world,
593            crystal,
594            pos,
595            source,
596            player.as_ref(),
597        );
598    }

600    /// Heals the dragon from the crystal it is tethered to.
601    ///
602    /// Vanilla parity: `EnderDragon.checkCrystals`. The beam is the client's
603    /// business; the effect is a point of health every ten ticks.
604    fn check_crystals(&self, world: &Arc<World>) {
605        let nearest = self.state.lock().nearest_crystal;
606        if let Some(crystal_id) = nearest {
607            let alive = world
608                .get_entity_by_id(crystal_id)
609                .is_some_and(|crystal| !crystal.is_removed());
610            if !alive {
611                self.state.lock().nearest_crystal = None;
612            } else if self.tick_count() % CRYSTAL_HEAL_INTERVAL == 0
613                && self.get_health() < self.get_max_health()
614            {
615                self.set_health(self.get_health() + 1.0);
616            }
617        }
618
619        if rand::random_range(0..10) != 0 {
620            return;
621        }
622

633            }

665    fn head_y_offset(&self) -> f32 {
666        if self.phase_manager.current_instance().is_sitting() {

674    /// Puts all eight hitboxes where this tick's body position implies.
675    ///
676    /// Vanilla parity: the geometry block in the middle of `EnderDragon.aiStep`.
677    /// The neck, head and tail read the flight history rather than the current
678    /// position, which is what makes the body flex through a turn.
679    fn tick_parts(&self) {
680        let (y_rot, _) = self.rotation();
681        let (tilt, sample_5) = {
682            let history = self.flight_history.lock();
683            let tilt = (history.get(5).y - history.get(10).y) as f32 * 10.0 * PI / 180.0;
684            (tilt, history.get(5))
685        };
686        let cc_tilt = trig::cos(f64::from(tilt));
687        let ss_tilt = trig::sin(f64::from(tilt));
688
689        let rot1 = f64::from(y_rot).to_radians();
690        let ss1 = trig::sin(rot1);
691        let cc1 = trig::cos(rot1);
692        self.tick_part(
693            DragonPartIndex::Body,
694            f64::from(ss1) * 0.5,
695            0.0,
696            f64::from(-cc1) * 0.5,
697        );
698        self.tick_part(
699            DragonPartIndex::Wing1,
700            f64::from(cc1) * 4.5,
701            2.0,
702            f64::from(ss1) * 4.5,
703        );
704        self.tick_part(
705            DragonPartIndex::Wing2,
706            f64::from(cc1) * -4.5,
707            2.0,
708            f64::from(ss1) * -4.5,
709        );
710
711        let head_rot = f64::from(y_rot).to_radians() - f64::from(self.y_rot_a()) * 0.01;
712        let ss2 = trig::sin(head_rot);
713        let cc2 = trig::cos(head_rot);
714        let y_offset = self.head_y_offset();
715        self.tick_part(
716            DragonPartIndex::Head,
717            f64::from(ss2 * 6.5 * cc_tilt),
718            f64::from(y_offset + ss_tilt * 6.5),
719            f64::from(-cc2 * 6.5 * cc_tilt),
720        );
721        self.tick_part(
722            DragonPartIndex::Neck,
723            f64::from(ss2 * 5.5 * cc_tilt),
724            f64::from(y_offset + ss_tilt * 5.5),
725            f64::from(-cc2 * 5.5 * cc_tilt),
726        );
727
728        for (segment, index) in [
729            DragonPartIndex::Tail1,
730            DragonPartIndex::Tail2,
731            DragonPartIndex::Tail3,
732        ]
733        .into_iter()
734        .zip(0..3)
735        {
736            let sample = self.flight_history.lock().get(12 + index * 2);
737            let rot = f64::from(y_rot).to_radians()
738                + f64::from(wrap_degrees(sample.y_rot - sample_5.y_rot)).to_radians();
739            let ss = trig::sin(rot);
740            let cc = trig::cos(rot);
741            let dd = (index + 1) as f32 * 2.0;
742            self.tick_part(
743                segment,
744                f64::from(-(ss1 * 1.5 + ss * dd) * cc_tilt),
745                sample.y - sample_5.y - f64::from((dd + 1.5) * ss_tilt) + 1.5,
746                f64::from((cc1 * 1.5 + cc * dd) * cc_tilt),
747            );
748        }
749    }

751    /// Shoves and hurts anything the wings sweep through.
752    ///
753    /// Vanilla parity: `EnderDragon.knockBack`.
754    fn knock_back(&self, world: &Arc<World>, wing: DragonPartIndex) {
755        let body = self.part(DragonPartIndex::Body).bounding_box();
756        let xm = f64::midpoint(body.min_x(), body.max_x());
757        let zm = f64::midpoint(body.min_z(), body.max_z());
758
759        let sweep = self
760            .part(wing)
761            .bounding_box()
762            .inflate_xyz(4.0, 2.0, 4.0)
763            .translate(DVec3::new(0.0, -2.0, 0.0));
764        let sitting = self.phase_manager.current_instance().is_sitting();
765
766        for entity in world.get_entities_in_aabb(&sweep) {
767            if entity.id() == self.id() || entity.as_living_entity().is_none() {
768                continue;
769            }
770            if is_creative_or_spectator(entity.as_ref()) {
771                continue;
772            }
773
774            let position = entity.position();
775            let xd = position.x - xm;
776            let zd = position.z - zm;
777            let dd = xd.mul_add(xd, zd * zd).max(0.1);
778            entity.push_impulse(DVec3::new(xd / dd * 4.0, f64::from(0.2_f32), zd / dd * 4.0));
779
780            if sitting {
781                continue;
782            }
783            let Some(living) = entity.as_living_entity() else {
784                continue;
785            };
786            if living.last_hurt_by_mob_timestamp() >= entity.tick_count() - 2 {
787                continue;
788            }
789            entity.hurt(world, &self.dragon_attack_damage_source(), WING_DAMAGE);
790        }
791    }

809    /// Eats the blocks the dragon is flying through.
810    ///
811    /// Vanilla parity: `EnderDragon.checkWalls`. Returns whether the dragon was
812    /// stopped by something it could not take, which is what makes it beat its
813    /// wings harder against bedrock.
814    fn check_walls(world: &Arc<World>, bb: WorldAabb) -> bool {
815        let x0 = bb.min_x().floor() as i32;
816        let y0 = bb.min_y().floor() as i32;
817        let z0 = bb.min_z().floor() as i32;
818        let x1 = bb.max_x().floor() as i32;
819        let y1 = bb.max_y().floor() as i32;
820        let z1 = bb.max_z().floor() as i32;
821
822        let mob_griefing = world.get_game_rule(&MOB_GRIEFING);
823        let mut hit_wall = false;
824        let mut destroyed_block = false;
825
826        for x in x0..=x1 {
827            for y in y0..=y1 {
828                for z in z0..=z1 {
829                    let pos = BlockPos::new(x, y, z);
830                    let state = world.get_block_state(pos);
831                    if state.is_air() || state.get_block().has_tag(&BlockTag::DRAGON_TRANSPARENT) {
832                        continue;
833                    }
834                    if mob_griefing && !state.get_block().has_tag(&BlockTag::DRAGON_IMMUNE) {
835                        destroyed_block = world.remove_block(pos, false) || destroyed_block;
836                    } else {
837                        hit_wall = true;
838                    }
839                }
840            }
841        }
842
843        if destroyed_block {
844            let pos = BlockPos::new(
845                x0 + rand::random_range(0..x1 - x0 + 1),
846                y0 + rand::random_range(0..y1 - y0 + 1),
847                z0 + rand::random_range(0..z1 - z0 + 1),
848            );
849            world.level_event(level_events::PARTICLES_DRAGON_BLOCK_BREAK, pos, 0, None);
850        }
851
852        hit_wall
853    }

855    /// Runs the death animation.
856    ///
857    /// Vanilla parity: `EnderDragon.tickDeath`. The dragon does not drop
858    /// experience through the mob path -- it awards the orbs here, in a trickle
859    /// over the last two and a half seconds and then a lump at the end.
860    fn run_death_animation(&self, world: &Arc<World>) {
861        if let Some(fight) = world.dragon_fight() {
862            fight.update_dragon(self);
863        }
864
865        let death_time = {
866            let mut state = self.state.lock();
867            state.dragon_death_time += 1;
868            state.dragon_death_time
869        };
870
871        // Vanilla parity: the fight substitutes twelve thousand for the first
872        // dragon of a world. With no fight, every dragon is worth five hundred.
873        let xp_count = if self
874            .dragon_fight(world)
875            .is_some_and(|fight| !fight.has_previously_killed_dragon())
876        {
877            FIRST_KILL_DEATH_XP
878        } else {
879            DEATH_XP
880        };
881
882        if world.get_game_rule(&MOB_DROPS)
883            && death_time > DEATH_XP_START_TICK
884            && death_time % DEATH_XP_INTERVAL == 0
885        {
886            ExperienceOrbEntity::award(
887                world,
888                self.position(),
889                (xp_count as f32 * DEATH_XP_TRICKLE_SHARE).floor() as i32,
890            );
891        }
892
893        if death_time == 1 && !self.is_silent() {
894            world.global_level_event(level_events::SOUND_DRAGON_DEATH, self.block_position(), 0);
895        }
896
897        let death_move = DVec3::new(0.0, f64::from(0.1_f32), 0.0);
898        self.move_entity(MoverType::SelfMovement, death_move);
899        for part in &self.parts {
900            part.set_old_position_to_current();
901            part.set_part_position(part.position() + death_move);
902        }
903
904        if death_time < DEATH_TIME {
905            return;
906        }
907
908        if world.get_game_rule(&MOB_DROPS) {
909            ExperienceOrbEntity::award(
910                world,
911                self.position(),
912                (xp_count as f32 * DEATH_XP_FINAL_SHARE).floor() as i32,
913            );
914        }
915
916        // Vanilla parity: `tickDeath` tells the fight before it removes itself.
917        // This is what opens the exit portal, drops the egg and hands out a
918        // gateway, and it is the only caller any of the three have.
919        if let Some(fight) = world.dragon_fight() {
920            fight.set_dragon_killed(world, self);
921        }
922
923        self.set_removed(RemovalReason::Killed);
924    }

1092    fn ai_step(&self) -> Option<MoveResult> {
1093        let world = self.level()?;
1094
1095        if self.is_dead_or_dying() {
1096            return None;
1097        }
1098
1099        self.check_crystals(&world);
1100
1101        // Vanilla parity: the flap-time bookkeeping is client-side animation
1102        // state; only `flightHistory.record` matters to the server, because the
1103        // hitboxes read it.
1104        self.set_rotation((wrap_degrees(self.rotation().0), self.rotation().1));
1105        if self.is_no_ai() {
1106            return None;
1107        }
1108
1109        self.flight_history
1110            .lock()
1111            .record(self.position().y, self.rotation().0);
1112
1113        let mut result = None;
1114        let current = self.phase_manager.current_instance();
1115        current.do_server_tick(self, &world);
1116        let current = if self.phase_manager.current_phase() == current.phase() {
1117            current
1118        } else {
1119            let switched = self.phase_manager.current_instance();
1120            switched.do_server_tick(self, &world);
1121            switched
1122        };
1123
1124        if let Some(target_location) = current.fly_target_location() {
1125            result = self.fly_towards(&world, target_location, current.as_ref());
1126        }
1127
1128        self.set_y_body_rot(self.rotation().0);
1129        let old_positions = self.parts.each_ref().map(|part| part.position());
1130        self.tick_parts();
1131
1132        // Vanilla parity: `this.hurtTime == 0`. Foton carries only
1133        // `invulnerableTime`, which is written on the same hit and decrements in
1134        // the same tick, so ten ticks in is exactly vanilla's `hurtTime` running out.
1135        if self.living_base().invulnerable_time() <= 10 {
1136            self.knock_back(&world, DragonPartIndex::Wing1);
1137            self.knock_back(&world, DragonPartIndex::Wing2);
1138            self.bite(&world, DragonPartIndex::Head);
1139            self.bite(&world, DragonPartIndex::Neck);
1140        }
1141
1142        let in_wall = Self::check_walls(&world, self.part(DragonPartIndex::Head).bounding_box())
1143            | Self::check_walls(&world, self.part(DragonPartIndex::Neck).bounding_box())
1144            | Self::check_walls(&world, self.part(DragonPartIndex::Body).bounding_box());
1145        self.state.lock().in_wall = in_wall;
1146        if let Some(fight) = world.dragon_fight() {
1147            fight.update_dragon(self);
1148        }
1149
1150        for (part, old) in self.parts.iter().zip(old_positions) {
1151            part.set_old_position(old);
1152        }
1153
1154        result
1155    }

1158impl EnderDragon {
1159    /// Steers and moves the dragon towards this tick's flight target.
1160    ///
1161    /// Vanilla parity: the `targetLocation != null` block of `aiStep`.
1162    fn fly_towards(
1163        &self,
1164        world: &Arc<World>,
1165        target: DVec3,
1166        phase: &dyn phases::DragonPhaseInstance,
1167    ) -> Option<MoveResult> {
1168        let _ = world;
1169        let position = self.position();
1170        let xdd = target.x - position.x;
1171        let ydd = target.y - position.y;
1172        let zdd = target.z - position.z;
1173        let dist_to_target = xdd.mul_add(xdd, ydd.mul_add(ydd, zdd * zdd));
1174        let max = f64::from(phase.fly_speed());
1175        let horizontal_dist = xdd.hypot(zdd);
1176        let ydd = if horizontal_dist > 0.0 {
1177            (ydd / horizontal_dist).clamp(-max, max)
1178        } else {
1179            ydd
1180        };
1181
1182        self.set_velocity(self.velocity() + DVec3::new(0.0, ydd * 0.01, 0.0));
1183        self.set_rotation((wrap_degrees(self.rotation().0), self.rotation().1));
1184
1185        let aim = (target - position).normalize_or_zero();
1186        let y_rot = f64::from(self.rotation().0).to_radians();
1187        let dir = DVec3::new(
1188            f64::from(trig::sin(y_rot)),
1189            self.velocity().y,
1190            f64::from(-trig::cos(y_rot)),
1191        )
1192        .normalize_or_zero();
1193        let dot = ((dir.dot(aim) as f32 + 0.5) / 1.5).max(0.0);
1194
1195        if xdd.abs() > f64::from(1.0e-5_f32) || zdd.abs() > f64::from(1.0e-5_f32) {
1196            let y_rot_d =
1197                wrap_degrees(180.0 - xdd.atan2(zdd).to_degrees() as f32 - self.rotation().0)
1198                    .clamp(-50.0, 50.0);
1199            let mut y_rot_a = self.y_rot_a() * 0.8;
1200            y_rot_a += y_rot_d * phase.turn_speed(self);
1201            self.set_y_rot_a(y_rot_a);
1202            self.set_rotation((self.rotation().0 + y_rot_a * 0.1, self.rotation().1));
1203        }
1204
1205        let span = (2.0 / (dist_to_target + 1.0)) as f32;
1206        self.move_relative(
1207            0.06 * dot.mul_add(span, 1.0 - span),
1208            DVec3::new(0.0, 0.0, -1.0),
1209        );
1210
1211        let in_wall = self.state.lock().in_wall;
1212        let movement = if in_wall {
1213            self.velocity() * f64::from(0.8_f32)
1214        } else {
1215            self.velocity()
1216        };
1217        let result = self.move_entity(MoverType::SelfMovement, movement);
1218
1219        let actual = self.velocity().normalize_or_zero();
1220        let slide = 0.8 + 0.15 * (actual.dot(dir) + 1.0) / 2.0;
1221        self.set_velocity(self.velocity() * DVec3::new(slide, f64::from(0.91_f32), slide));
1222
1223        result
1224    }
1225}
