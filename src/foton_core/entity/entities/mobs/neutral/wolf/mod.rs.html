463    /// Vanilla parity: the armor-repair branch of `Wolf.mobInteract`.
464    fn try_repair_body_armor(
465        &self,
466        player: &Player,
467        hand: InteractionHand,
468        item_stack: &ItemStack,
469    ) -> bool {
470        if !self.is_in_sitting_pose() || !self.is_owned_by(player) {
471            return false;
472        }
473
474        let mut repaired = false;
475        self.with_equipment_slot_mut(EquipmentSlot::Body, &mut |armor| {
476            if armor.is_empty() || !armor.is_damaged() || !armor.is_valid_repair_item(item_stack) {
477                return;
478            }
479            let repair_unit = (armor.get_max_damage() as f32 * ARMOR_REPAIR_UNIT) as i32;
480            let repaired_damage = (armor.get_damage_value() - repair_unit).max(0);
481            armor.set_damage_value(repaired_damage);
482            repaired = true;
483        });
484
485        if !repaired {
486            return false;
487        }
488
489        player.inventory.lock().shrink_item_in_hand(hand, 1);
490        self.play_sound(&sound_events::ITEM_WOLF_ARMOR_REPAIR, 1.0, 1.0);
491        true
492    }

494    /// Vanilla parity: the tamed half of `Wolf.mobInteract`.
495    fn tamed_interact(
496        &self,
497        player: &Player,
498        hand: InteractionHand,
499        item_stack: &ItemStack,
500    ) -> InteractionResult {
501        if Self::is_wolf_food(item_stack) && self.get_health() < self.get_max_health() {
502            self.feed(player, hand, item_stack, 2.0, 2.0);
503            return InteractionResult::Success;
504        }
505
506        let is_collar_dye = REGISTRY
507            .items
508            .is_in_tag(item_stack.item(), &ItemTag::WOLF_COLLAR_DYES);
509        if is_collar_dye && self.is_owned_by(player) {
510            return if self.try_dye_collar(player, hand, item_stack) {
511                InteractionResult::Success
512            } else {
513                Animal::mob_interact_animal(self, player, hand)
514            };
515        }
516
517        if self.try_equip_body_armor(player, hand, item_stack)
518            || self.try_repair_body_armor(player, hand, item_stack)
519        {
520            return InteractionResult::Success;
521        }
522
523        let interaction_result = Animal::mob_interact_animal(self, player, hand);
524        if interaction_result.consumes_action() || !self.is_owned_by(player) {
525            return interaction_result;
526        }
527
528        self.set_ordered_to_sit(!self.is_ordered_to_sit());
529        self.set_jumping(false);
530        self.mob_base.navigation().lock().stop();
531        self.set_target(None);
532        // Vanilla returns `InteractionResult.SUCCESS.withoutItem()`; Foton has
533        // no without-item variant, and the sit toggle consumes nothing anyway.
534        InteractionResult::Success
535    }

537    /// Runs the shake state machine.
538    ///
539    /// Vanilla parity: the `Wolf.tick` half that is not the interested-angle
540    /// interpolation. The splash particles are client-local; the sound, the
541    /// game event and the two entity events are not.
542    fn tick_shake(&self) {
543        if !Entity::is_alive(self) {
544            return;
545        }
546
547        if self.is_in_water_or_rain() {
548            let cancel = {
549                let mut shake = self.shake.lock();
550                shake.is_wet = true;
551                let was_shaking = shake.is_shaking;
552                if was_shaking {
553                    shake.cancel();
554                }
555                was_shaking
556            };
557            if cancel {
558                self.broadcast_entity_event(EntityStatus::CancelShakeWetness);
559            }
560            return;
561        }
562
563        let start_of_shake = {
564            let shake = self.shake.lock();
565            if !shake.is_shaking {
566                return;
567            }
568            shake.shake_anim == 0.0
569        };
570
571        if start_of_shake {
572            let pitch = (rand::random::<f32>() - rand::random::<f32>()).mul_add(0.2, 1.0);
573            self.play_sound(&sound_events::ENTITY_WOLF_SHAKE, WOLF_SOUND_VOLUME, pitch);
574            self.game_event(&vanilla_game_events::ENTITY_ACTION);
575        }
576
577        let mut shake = self.shake.lock();
578        shake.shake_anim_o = shake.shake_anim;
579        shake.shake_anim += SHAKE_ANIM_STEP;
580        if shake.shake_anim_o >= SHAKE_ANIM_END {
581            shake.is_wet = false;
582            shake.cancel();
583        }
584        // VANILLA CLIENT-LOCAL: `Wolf.tick` spawns the splash particles with
585        // `Level.addParticle` once `shakeAnim` passes 0.4.
586    }

588    /// Starts a shake once the wolf is out of the water and standing still.
589    ///
590    /// Vanilla parity: the first half of `Wolf.aiStep`.
591    fn start_shake_if_dry_land(&self) {
592        let should_start = {
593            let shake = self.shake.lock();
594            shake.is_wet && !shake.is_shaking
595        };
596        if !should_start || self.is_path_finding() || !self.on_ground() {
597            return;
598        }
599
600        {
601            let mut shake = self.shake.lock();
602            shake.is_shaking = true;
603            shake.shake_anim = 0.0;
604            shake.shake_anim_o = 0.0;
605        }
606        self.broadcast_entity_event(EntityStatus::ShakeWetness);
607    }

630///
631/// Vanilla parity: `Wolf.WolfAvoidEntityGoal`. Foton has no llama yet, so the
632/// goal is registered with the type and tame checks it can answer and never
633/// fires; the strength roll (`llama.getStrength() >= random.nextInt(5)`) and

635fn wolf_avoid_llama_goal() -> AvoidEntityGoal {

638            && !targeter.is_some_and(|wolf| is_tamed(wolf.as_entity_event_source()))
639    })

810    /// Vanilla parity: `Wolf.actuallyHurt`, where armor eats the whole hit
811    /// rather than reducing it.
812    fn actually_hurt(&self, world: &World, source: &DamageSource, amount: f32) {
813        if !self.can_armor_absorb(source) {
814            self.living_actually_hurt(world, source, amount);
815            return;
816        }
817
818        let mut cracked_further = false;
819        self.with_equipment_slot_mut(EquipmentSlot::Body, &mut |armor| {
820            let damage_before = armor.get_damage_value();
821            let max_damage = armor.get_max_damage();
822            armor.hurt_and_break(amount.ceil() as i32, false);
823            cracked_further = Crackiness::WOLF_ARMOR.by_damage(damage_before, max_damage)
824                != Crackiness::WOLF_ARMOR.by_stack(armor);
825        });
826
827        if !cracked_further {
828            return;
829        }
830
831        self.play_sound(&sound_events::ITEM_WOLF_ARMOR_CRACK, 1.0, 1.0);
832        let position = self.position();
833        world.send_particles(
834            ParticleData::new(
835                &vanilla_particle_types::ITEM,
836                ItemParticleOption::new(ItemStackTemplate::new(&vanilla_items::ARMADILLO_SCUTE)),
837            ),
838            DVec3::new(position.x, position.y + 1.0, position.z),
839            20,
840            DVec3::new(0.2, 0.1, 0.2),
841            0.1,
842        );
843    }

951                    self.collar_color(),

954            }
955        }
956
957        if let Some(sound_variant) = REGISTRY
958            .wolf_sound_variants
959            .pick_random(&mut LegacyRandom::from_seed(rand::random()))
960        {
961            pup.set_sound_variant(sound_variant);
962        }

1119            REGISTRY
