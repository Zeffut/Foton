1//! Copper golem entity.
2//!
3//! Vanilla parity: `CopperGolem`. New in 26.2. It oxidizes on a timer the way
4//! a copper block does, and once it is fully oxidized it eventually seizes up
5//! into a [`crate::block_entity::entities::CopperGolemStatueBlockEntity`].
6//! Honeycomb stops the clock, an axe winds it back.
7//!
8//! Its whole job -- carrying one stack at a time out of a copper chest and into
9//! an ordinary one -- is brain-driven, and lives in
10//! [`super::copper_golem_ai`]. This entity has no goals at all; the brain in
11//! [`Self::brain`] is its entire AI, exactly as in vanilla.

104/// What a copper golem is doing with its hands.
105///
106/// Vanilla parity: `CopperGolemState`. Every state but [`Self::Idle`] belongs
107/// to the chest behaviors Foton cannot run yet, but the value is synced, so the
108/// enum is here in full rather than reduced to what the server sets.
109#[derive(Debug, Clone, Copy, PartialEq, Eq)]
110pub enum CopperGolemState {
111    /// Standing about.
112    Idle,
113    /// Taking an item out of a container.
114    GettingItem,
115    /// Reaching into an empty container.
116    GettingNoItem,
117    /// Putting an item into a container.
118    DroppingItem,
119    /// Reaching into a container with nothing to put in it.
120    DroppingNoItem,
121}
122
123impl CopperGolemState {
124    /// Returns the protocol id of this state.
125    #[must_use]
126    pub const fn id(self) -> i32 {
127        match self {
128            Self::Idle => 0,
129            Self::GettingItem => 1,
130            Self::GettingNoItem => 2,
131            Self::DroppingItem => 3,
132            Self::DroppingNoItem => 4,
133        }
134    }
135
136    /// Returns the state with this protocol id.
137    ///
138    /// Vanilla parity: the `ByIdMap.continuous(..., ZERO)` of
139    /// `CopperGolemState.BY_ID`, which falls back to idle.
140    #[must_use]
141    pub const fn by_id(id: i32) -> Self {
142        match id {
143            1 => Self::GettingItem,
144            2 => Self::GettingNoItem,
145            3 => Self::DroppingItem,
146            4 => Self::DroppingNoItem,
147            _ => Self::Idle,
148        }
149    }
150}

207    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
208        let living_base = LivingEntityBase::new(entity_type);

217            navigation.set_can_open_doors(true);

359    /// Advances the oxidation clock, and freezes the golem when it runs out.
360    ///
361    /// Vanilla parity: `CopperGolem.updateWeathering`.
362    fn update_weathering(&self, world: &Arc<World>) {
363        let game_time = world.game_time();
364        let next_tick = self.state.lock().next_weathering_tick;
365        if next_tick == IGNORE_WEATHERING_TICK {
366            return;
367        }
368
369        if next_tick == UNSET_WEATHERING_TICK {
370            self.state.lock().next_weathering_tick =
371                game_time + rand::random_range(WEATHERING_TICK_FROM..=WEATHERING_TICK_TO);
372            return;
373        }
374
375        let weather_state = self.weather_state();
376        let is_fully_oxidized = weather_state == WeatherState::Oxidized;
377        if game_time >= next_tick && !is_fully_oxidized {
378            let new_state = weather_state.next();
379            self.set_weather_state(new_state);
380            self.state.lock().next_weathering_tick = if new_state == WeatherState::Oxidized {
381                0
382            } else {
383                next_tick + rand::random_range(WEATHERING_TICK_FROM..=WEATHERING_TICK_TO)
384            };
385        }
386
387        if is_fully_oxidized && self.can_turn_to_statue(world) {
388            self.turn_to_statue(world);
389        }
390    }

392    /// Vanilla parity: `CopperGolem.canTurnToStatue`.
393    fn can_turn_to_statue(&self, world: &Arc<World>) -> bool {
394        world.get_block_state(self.block_position()).is_air()
395            && rand::random::<f32>() <= TURN_TO_STATUE_CHANCE
396    }

398    /// Freezes the golem into a statue block.
399    ///
400    /// Vanilla parity: `CopperGolem.turnToStatue`.
401    fn turn_to_statue(&self, world: &Arc<World>) {
402        let pos = self.block_position();
403        let poses = COPPER_GOLEM_POSE.possible_values;
404        let pose = poses[rand::random_range(0..poses.len())].clone();
405        let statue = vanilla_blocks::OXIDIZED_COPPER_GOLEM_STATUE
406            .default_state()
407            .set_value(COPPER_GOLEM_POSE, pose)
408            .set_value(STATUE_FACING, Direction::from_yaw(self.rotation().0));
409        world.set_block(pos, statue, UpdateFlags::UPDATE_ALL);
410
411        let Some(block_entity) = world.get_block_entity(pos) else {
412            return;
413        };
414        let Some(statue_entity) = block_entity.downcast_ref::<CopperGolemStatueBlockEntity>()
415        else {
416            return;
417        };
418
419        statue_entity.create_statue(self.custom_name());
420        self.drop_preserved_equipment(world);
421        self.play_sound(&sound_events::ENTITY_COPPER_GOLEM_BECOME_STATUE, 1.0, 1.0);
422        if self.is_leashed() {
423            if world.get_game_rule(&vanilla_game_rules::ENTITY_DROPS) {
424                self.drop_leash();
425            } else {
426                self.remove_leash();
427            }
428        }
429        self.set_removed(RemovalReason::Discarded);
430    }

432    /// Drops the equipment a statue leaves behind.
433    ///
434    /// Vanilla parity: `Mob.dropPreservedEquipment`, restricted to the antenna
435    /// because that is the only slot a copper golem uses.
436    fn drop_preserved_equipment(&self, _world: &Arc<World>) {
437        if !self.is_equipment_drop_preserved(EQUIPMENT_SLOT_ANTENNA) {
438            return;
439        }
440
441        let mut antenna = ItemStack::empty();
442        self.with_equipment_slot_mut(EQUIPMENT_SLOT_ANTENNA, &mut |item| {
443            antenna = item.copy_with_count(item.count());
444            *item = ItemStack::empty();
445        });
446        if !antenna.is_empty() {
447            let _ = self.spawn_at_location(antenna, 0.0);
448        }
449    }

451    /// Returns the sounds this golem makes at its current oxidation.
452    ///
453    /// Vanilla parity: `CopperGolemOxidationLevels.getOxidationLevel`, minus
454    /// the two textures, which are client-side.
455    fn oxidation_sounds(&self) -> OxidationSounds {
456        match self.weather_state() {
457            WeatherState::Unaffected | WeatherState::Exposed => OxidationSounds {
458                hurt: &sound_events::ENTITY_COPPER_GOLEM_HURT,
459                death: &sound_events::ENTITY_COPPER_GOLEM_DEATH,
460                step: &sound_events::ENTITY_COPPER_GOLEM_STEP,
461            },
462            WeatherState::Weathered => OxidationSounds {
463                hurt: &sound_events::ENTITY_COPPER_GOLEM_WEATHERED_HURT,
464                death: &sound_events::ENTITY_COPPER_GOLEM_WEATHERED_DEATH,
465                step: &sound_events::ENTITY_COPPER_GOLEM_WEATHERED_STEP,
466            },
467            WeatherState::Oxidized => OxidationSounds {
468                hurt: &sound_events::ENTITY_COPPER_GOLEM_OXIDIZED_HURT,
469                death: &sound_events::ENTITY_COPPER_GOLEM_OXIDIZED_DEATH,

475    /// Handles the honeycomb and axe half of `CopperGolem.mobInteract`.
476    fn interact_with_tool(
477        &self,
478        world: &Arc<World>,
479        player: &Player,
480        hand: InteractionHand,
481        item_stack: &ItemStack,
482    ) -> InteractionResult {
483        let next_weathering_tick = self.state.lock().next_weathering_tick;
484
485        if item_stack.is(&vanilla_items::HONEYCOMB)
486            && next_weathering_tick != IGNORE_WEATHERING_TICK
487        {
488            world.level_event(
489                level_events::PARTICLES_AND_SOUND_WAX_ON,
490                self.block_position(),
491                0,
492                None,
493            );
494            self.state.lock().next_weathering_tick = IGNORE_WEATHERING_TICK;
495            self.use_player_item(player, hand);
496            return InteractionResult::SuccessServer;
497        }
498
499        if !REGISTRY.items.is_in_tag(item_stack.item(), &ItemTag::AXES) {
500            return InteractionResult::Pass;
501        }
502
503        if next_weathering_tick == IGNORE_WEATHERING_TICK {
504            world.play_sound_at(
505                &sound_events::ITEM_AXE_WAX_OFF,
506                self.sound_source(),
507                self.position(),
508                1.0,
509                1.0,
510                None,
511            );
512            world.level_event(
513                level_events::PARTICLES_WAX_OFF,
514                self.block_position(),
515                0,
516                None,
517            );
518            self.state.lock().next_weathering_tick = UNSET_WEATHERING_TICK;
519            player.hurt_item_in_hand(hand, 1);
520            return InteractionResult::SuccessServer;
521        }
522
523        let weather_state = self.weather_state();
524        if weather_state == WeatherState::Unaffected {
525            return InteractionResult::Pass;
526        }
527
528        world.play_sound_at(
529            &sound_events::ITEM_AXE_SCRAPE,
530            self.sound_source(),
531            self.position(),
532            1.0,
533            1.0,
534            None,
535        );
536        world.level_event(
537            level_events::PARTICLES_SCRAPE,
538            self.block_position(),
539            0,
540            None,
541        );
542        self.state.lock().next_weathering_tick = UNSET_WEATHERING_TICK;
543        self.set_weather_state(weather_state.previous());
544        player.hurt_item_in_hand(hand, 1);
545        InteractionResult::SuccessServer
546    }
547}

591        let bolt_uuid = bolt.uuid();

593            return;
594        }
595        self.state.lock().last_lightning_bolt_uuid = Some(bolt_uuid);
596
597        let weather_state = self.weather_state();
598        if weather_state == WeatherState::Unaffected {
599            return;

602        self.set_weather_state(weather_state.previous());
603    }

614    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
615        self.load_mob(nbt);
616        self.brain.load(nbt);
617        self.state.lock().next_weathering_tick = nbt
618            .long("next_weather_age")
619            .unwrap_or(UNSET_WEATHERING_TICK);
620        let weather_state = nbt
621            .string("weather_state")
622            .and_then(|name| weather_state_by_name(&name.to_str()))
623            .unwrap_or(WeatherState::Unaffected);
624        self.set_weather_state(weather_state);
625    }
626}
627
628/// Returns the name vanilla's `WeatherState.CODEC` writes.
629const fn weather_state_name(weather_state: WeatherState) -> &'static str {
630    match weather_state {
631        WeatherState::Unaffected => "unaffected",
632        WeatherState::Exposed => "exposed",
633        WeatherState::Weathered => "weathered",
634        WeatherState::Oxidized => "oxidized",
635    }
636}
637
638/// Returns the stage vanilla's `WeatherState.CODEC` reads.
639fn weather_state_by_name(name: &str) -> Option<WeatherState> {
640    match name {
641        "unaffected" => Some(WeatherState::Unaffected),
642        "exposed" => Some(WeatherState::Exposed),
643        "weathered" => Some(WeatherState::Weathered),
644        "oxidized" => Some(WeatherState::Oxidized),
645        _ => None,
646    }
647}

784impl CopperGolemEntity {
785    /// Tosses whatever the golem is holding towards `target`.
786    ///
787    /// Vanilla parity: the `BehaviorUtils.throwItem` branch of
788    /// `CopperGolem.mobInteract`.
789    fn throw_held_item_to(&self, world: &Arc<World>, target: DVec3) -> bool {
790        let mut held = ItemStack::empty();
791        self.with_equipment_slot_mut(EquipmentSlot::MainHand, &mut |item| {
792            held = item.copy_with_count(item.count());
793            *item = ItemStack::empty();
794        });
795        if held.is_empty() {
796            return false;
797        }
798
799        let position = self.position();
800        let hand_y = self.get_eye_y() - THROW_HAND_Y_OFFSET;
801        let velocity = (target - position).normalize_or_zero() * THROW_VELOCITY;
802        let _ = world.spawn_item_with_velocity(
803            DVec3::new(position.x, hand_y, position.z),
804            held,
805            velocity,
806        );
807        true
808    }
809}

815
816/// How hard the golem tosses an item back.
817///
818/// Vanilla parity: the `new Vec3(0.3F, 0.3F, 0.3F)` of `BehaviorUtils.throwItem`.
819const THROW_VELOCITY: f64 = 0.3;
