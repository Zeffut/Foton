1//! Allay entity.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.allay.Allay`. An allay is
4//! a flying `PathfinderMob` with no fight in it at all. Hand it an item and it
5//! remembers you, hunts the ground for more of the same, and brings them back
6//! -- to your hand, or to the note block it last heard, which is what turns a
7//! pair of them into a sorting machine. Play it a jukebox and it dances; give a
8//! dancing one an amethyst shard and there are two.

167    /// Attaches the two game-event listeners an allay carries.
168    ///
169    /// Vanilla builds them in the constructor from `this`. Foton cannot: the
170    /// constructor has no `Arc` to hand out and there is no `Arc<dyn Entity>`
171    /// downcast, so each listener holds the world and the entity id instead and
172    /// looks the allay back up. The world hands the world over the first time it
173    /// asks the allay to register.
174    fn ensure_listeners(&self, world: &Arc<World>) {
175        let mut listeners = self.listeners.lock();
176        if !listeners.is_empty() {
177            return;
178        }
179        let source = ListenerSource {
180            entity_id: self.id(),
181            world: Arc::downgrade(world),
182        };
183        let jukebox: SharedGameEventListener = Arc::new(JukeboxListener {
184            allay: source.clone(),
185        });
186        let vibration = Arc::new(VibrationListener::new(Arc::new(AllayVibrationUser {
187            allay: source,
188        })));
189        listeners.push(DynamicGameEventListener::new(jukebox));
190        listeners.push(DynamicGameEventListener::new(
191            Arc::clone(&vibration) as SharedGameEventListener
192        ));
193        *self.vibration_listener.lock() = Some(vibration);
194    }
195
196    fn vibration_listener(&self) -> Option<Arc<VibrationListener>> {
197        self.vibration_listener.lock().clone()
198    }

290    /// Vanilla parity: `Allay.setJukeboxPlaying`.
291    pub fn set_jukebox_playing(&self, jukebox: BlockPos, is_playing: bool) {
292        if is_playing {
293            if !self.is_dancing() {
294                *self.jukebox_pos.lock() = Some(jukebox);
295                self.set_dancing(true);
296            }
297            return;
298        }
299
300        let matches = {
301            let current = self.jukebox_pos.lock();
302            current.is_none_or(|pos| pos == jukebox)
303        };
304        if matches {
305            *self.jukebox_pos.lock() = None;
306            self.set_dancing(false);
307        }
308    }

311    fn should_stop_dancing(&self) -> bool {
312        let Some(jukebox_pos) = *self.jukebox_pos.lock() else {
313            return true;
314        };
315        let Some(world) = self.level() else {

341    /// Vanilla parity: the amethyst-shard branch of `Allay.mobInteract`.
342    fn try_duplicate(&self, player: &Player, hand: InteractionHand, held: &ItemStack) -> bool {
343        if !self.is_dancing() || !self.can_duplicate() {
344            return false;
345        }
346        if !REGISTRY
347            .items
348            .is_in_tag(held.item(), &ItemTag::DUPLICATES_ALLAYS)
349        {
350            return false;
351        }
352        let Some(world) = self.level() else {
353            return false;
354        };
355        if self.duplicate_allay(&world).is_none() {
356            return false;
357        }
358
359        self.broadcast_entity_event(DUPLICATION_HEARTS_EVENT);
360        world.play_sound_at(
361            &sound_events::BLOCK_AMETHYST_BLOCK_CHIME,
362            SoundSource::Neutral,
363            self.position(),
364            INTERACTION_SOUND_VOLUME,
365            1.0,
366            None,
367        );
368        player.inventory.lock().shrink_item_in_hand(hand, 1);
369        true
370    }

372    /// Vanilla parity: the "give the allay an item" branch of `mobInteract`.
373    fn take_item_from(&self, player: &Player, hand: InteractionHand, held: &ItemStack) {
374        self.living_base
375            .equipment()
376            .lock()
377            .set(EquipmentSlot::MainHand, held.copy_with_count(1));
378        player.inventory.lock().shrink_item_in_hand(hand, 1);
379        if let Some(world) = self.level() {
380            world.play_sound_at(
381                &sound_events::ENTITY_ALLAY_ITEM_GIVEN,
382                SoundSource::Neutral,
383                self.position(),
384                INTERACTION_SOUND_VOLUME,
385                1.0,
386                None,
387            );
388        }
389        self.brain
390            .set_memory(memory_module_types::LIKED_PLAYER, player.uuid());
391    }

393    /// Vanilla parity: the "take the allay's item back" branch of `mobInteract`.
394    ///
395    /// Everything the allay had gathered is thrown at the player's feet, not
396    /// handed over: only the one item it was holding goes into the inventory.
397    fn give_item_back_to(&self, player: &Player, held_by_allay: ItemStack) {
398        self.living_base
399            .equipment()
400            .lock()
401            .set(EquipmentSlot::MainHand, ItemStack::empty());
402        if let Some(world) = self.level() {
403            world.play_sound_at(
404                &sound_events::ENTITY_ALLAY_ITEM_TAKEN,
405                SoundSource::Neutral,
406                self.position(),
407                INTERACTION_SOUND_VOLUME,
408                1.0,
409                None,
410            );
411        }
412        self.swing(InteractionHand::MainHand, false);
413
414        let position = self.position();
415        for item in self.take_gathered_items() {
416            throw_item(self, item, position);
417        }
418
419        self.brain
420            .erase_memory(memory_module_types::LIKED_PLAYER.id());
421        player.add_item_or_drop(held_by_allay);
422    }
423
424    /// Empties the carried container and hands back what was in it.
425    ///
426    /// Vanilla parity: `SimpleContainer.removeAllItems`, which both
427    /// `mobInteract` and `dropEquipment` call.
428    fn take_gathered_items(&self) -> Vec<ItemStack> {
429        let mut inventory = self.inventory.lock();
430        inventory
431            .items_mut()
432            .iter_mut()
433            .filter(|item| !item.is_empty())
434            .map(take)
435            .collect()
436    }

439/// How a listener finds the allay it belongs to.
440///
441/// Vanilla parity: `EntityPositionSource(allay, allay.getEyeHeight())`, which
442/// is what makes both of an allay's listeners report its eye rather than its
443/// feet.
444#[derive(Clone)]
445struct ListenerSource {
446    entity_id: i32,
447    world: Weak<World>,
448}
449
450impl ListenerSource {
451    fn eye_position(&self) -> Option<DVec3> {
452        let world = self.world.upgrade()?;
453        let entity = world.get_entity_by_id(self.entity_id)?;
454        let allay = entity.downcast_ref::<AllayEntity>()?;
455        Some(DVec3::new(
456            allay.position().x,
457            allay.get_eye_y(),
458            allay.position().z,
459        ))
460    }
461
462    /// Runs `visit` on the allay while the world still holds it.
463    fn with_allay<R>(&self, visit: impl FnOnce(&AllayEntity) -> R) -> Option<R> {
464        let world = self.world.upgrade()?;
465        let entity = world.get_entity_by_id(self.entity_id)?;
466        let allay = entity.downcast_ref::<AllayEntity>()?;
467        Some(visit(allay))
468    }
469}

471/// Vanilla parity: `Allay.JukeboxListener`, a plain game-event listener rather
472/// than a vibration one -- an allay hears a jukebox directly.
473struct JukeboxListener {
474    allay: ListenerSource,
475}
476
477impl GameEventListener for JukeboxListener {
478    fn listener_pos(&self) -> Option<DVec3> {
479        self.allay.eye_position()
480    }
481
482    fn listener_radius(&self) -> i32 {
483        vanilla_game_events::JUKEBOX_PLAY.notification_radius
484    }
485
486    fn handle_game_event(
487        &self,
488        _world: &Arc<World>,
489        event: GameEventRef,
490        _context: &GameEventContext<'_>,
491        source_pos: DVec3,
492    ) -> bool {
493        let playing = if event.key == vanilla_game_events::JUKEBOX_PLAY.key {
494            true
495        } else if event.key == vanilla_game_events::JUKEBOX_STOP_PLAY.key {
496            false
497        } else {
498            return false;
499        };
500
501        self.allay
502            .with_allay(|allay| allay.set_jukebox_playing(BlockPos::from(source_pos), playing))
503            .is_some()
504    }
505}

507/// Vanilla `Allay.VibrationUser`.
508///
509/// An allay that has not been given a note block yet hears every one; once it likes one,
510/// it hears only that one, and only while it is still within a thousand blocks of it.
511struct AllayVibrationUser {
512    allay: ListenerSource,
513}
514
515impl VibrationUser for AllayVibrationUser {
516    fn listener_radius(&self) -> i32 {
517        NOTE_BLOCK_LISTENER_RADIUS
518    }
519
520    fn position_source(&self) -> VibrationPositionSource {
521        VibrationPositionSource::Entity {
522            world: Weak::clone(&self.allay.world),
523            entity_id: self.allay.entity_id,
524            y_offset: self
525                .allay
526                .with_allay(|allay| allay.get_eye_height() as f32)
527                .unwrap_or(0.0),
528        }
529    }
530
531    fn listenable_events(&self) -> Identifier {
532        GameEventTag::ALLAY_CAN_LISTEN
533    }
534
535    fn can_receive_vibration(
536        &self,
537        world: &Arc<World>,
538        pos: BlockPos,
539        _event: GameEventRef,
540        _context: &GameEventContext<'_>,
541    ) -> bool {
542        self.allay
543            .with_allay(|allay| {
544                if allay.is_no_ai() {
545                    return false;
546                }
547                let Some(liked) = allay
548                    .brain
549                    .get_memory(memory_module_types::LIKED_NOTEBLOCK_POSITION)
550                else {
551                    return true;
552                };
553                liked.dimension == world.key
554                    && block_closer_than(liked.pos, allay.block_position(), LIKED_NOTEBLOCK_RANGE)
555                    && liked.pos == pos
556            })
557            .unwrap_or(false)
558    }
559
560    fn on_receive_vibration(
561        &self,
562        world: &Arc<World>,
563        pos: BlockPos,
564        event: GameEventRef,
565        _source_entity: Option<&dyn Entity>,
566        _projectile_owner: Option<&dyn Entity>,
567        _receiving_distance: f32,
568    ) {
569        if event.key != vanilla_game_events::NOTE_BLOCK_PLAY.key {
570            return;
571        }
572        self.allay.with_allay(|allay| {
573            allay_ai::hear_noteblock(&allay.brain, world, pos);
574        });
575    }
576}

578/// Vanilla `GlobalPos.isCloseEnough`, which measures the block positions themselves.
579fn block_closer_than(from: BlockPos, to: BlockPos, distance: f64) -> bool {
580    let delta = from.0 - to.0;
581    let distance_squared = f64::from(delta.x).mul_add(
582        f64::from(delta.x),
583        f64::from(delta.y).mul_add(f64::from(delta.y), f64::from(delta.z) * f64::from(delta.z)),
584    );
585    distance_squared < distance * distance
586}

605    /// Vanilla parity: the server half of `Allay.tick`, which stops a panicking
606    /// allay dancing.
607    fn tick(&self) {
608        // Vanilla parity: the `VibrationSystem.Ticker.tick` of `Allay.tick`, which is what
609        // makes a note block heard across the room arrive a moment after it was struck.
610        if let Some(world) = self.level() {
611            self.ensure_listeners(&world);
612            if let Some(listener) = self.vibration_listener() {
613                listener.tick(&world);
614            }
615        }
616        LivingEntity::tick_living_entity(self);
617        if self.is_panicking() {
618            self.set_dancing(false);
619        }
620    }

872    /// Vanilla parity: `Allay.mobInteract`.
873    fn mob_interact(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
874        let held = {
875            let inventory = player.inventory.lock();
876            let held = inventory.get_item_in_hand(hand);
877            held.copy_with_count(held.count())
878        };
879        let item_in_hand = self.get_item_in_hand(InteractionHand::MainHand);
880
881        if self.try_duplicate(player, hand, &held) {
882            return InteractionResult::Success;
883        }
884
885        if item_in_hand.is_empty() && !held.is_empty() {
886            self.take_item_from(player, hand, &held);
887            return InteractionResult::Success;
888        }
889
890        if !item_in_hand.is_empty() && hand == InteractionHand::MainHand && held.is_empty() {
891            self.give_item_back_to(player, item_in_hand);
892            return InteractionResult::Success;
893        }
894
895        InteractionResult::Pass
896    }

912/// Spawns an allay for a test or a command.
913///
914/// Vanilla parity: the plain `EntityType.ALLAY.create` an allay is otherwise
915/// only made by, which the generated factory table already covers.
916#[must_use]
917pub fn spawn_allay(world: &Arc<World>, position: DVec3) -> Option<SharedEntity> {
918    let entity = ENTITIES.create(
919        &vanilla_entities::ALLAY,
920        next_entity_id(),
921        position,
922        Arc::downgrade(world),
923    )?;
924    entity.set_old_position_to_current();
925    if let Some(mob) = entity.as_mob() {
926        let _ = mob.finalize_spawn(world, EntitySpawnReason::Command, None);
927    }
928    world.try_add_entity(Arc::clone(&entity)).ok()?;
929    Some(entity)
930}
