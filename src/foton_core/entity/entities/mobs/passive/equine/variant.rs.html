1//! The coat and markings a horse is born with.
2//!
3//! Vanilla parity: `animal.equine.Variant` and `animal.equine.Markings`. Both
4//! travel as one synchronized int, the coat in the low byte and the markings in
5//! the high one, which is why they are defined together.
6
7/// The seven horse coats.
8///
9/// Vanilla parity: `animal.equine.Variant`.
10#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
11pub enum HorseVariant {
12    /// Vanilla `Variant.WHITE`.
13    #[default]
14    White,
15    /// Vanilla `Variant.CREAMY`.
16    Creamy,
17    /// Vanilla `Variant.CHESTNUT`.
18    Chestnut,
19    /// Vanilla `Variant.BROWN`.
20    Brown,
21    /// Vanilla `Variant.BLACK`.
22    Black,
23    /// Vanilla `Variant.GRAY`.
24    Gray,
25    /// Vanilla `Variant.DARK_BROWN`.
26    DarkBrown,
27}
28
29impl HorseVariant {
30    /// Every coat in vanilla id order.
31    pub const ALL: [Self; 7] = [
32        Self::White,
33        Self::Creamy,
34        Self::Chestnut,
35        Self::Brown,
36        Self::Black,
37        Self::Gray,
38        Self::DarkBrown,
39    ];
40
41    /// Returns the vanilla synchronized id.
42    #[must_use]
43    pub const fn id(self) -> i32 {
44        match self {
45            Self::White => 0,
46            Self::Creamy => 1,
47            Self::Chestnut => 2,
48            Self::Brown => 3,
49            Self::Black => 4,
50            Self::Gray => 5,
51            Self::DarkBrown => 6,
52        }
53    }
54
55    /// Returns the coat for a synchronized id.
56    ///
57    /// Vanilla parity: `Variant.BY_ID`, which wraps rather than clamps.
58    #[must_use]
59    pub const fn by_id(id: i32) -> Self {
60        let len = Self::ALL.len() as i32;
61        Self::ALL[id.rem_euclid(len) as usize]
62    }
63
64    /// Picks a coat at random, as vanilla's `Util.getRandom` does.
65    #[must_use]
66    pub fn random() -> Self {
67        Self::ALL[rand::random_range(0..Self::ALL.len())]
68    }
69}

127/// Packs a coat and its markings into vanilla's single synchronized int.
128///
129/// Vanilla parity: `Horse.setVariantAndMarkings`.
130#[must_use]
131pub const fn pack_type_variant(variant: HorseVariant, markings: HorseMarkings) -> i32 {
132    variant.id() & 0xFF | markings.id() << 8 & 0xFF00
133}
134
135/// Replaces only the coat of a packed value.
136///
137/// Vanilla parity: `Horse.setVariant`.
138#[must_use]
139pub const fn with_variant(type_variant: i32, variant: HorseVariant) -> i32 {
140    variant.id() & 0xFF | type_variant & -256
141}

143#[cfg(test)]
144mod tests {
145    use super::*;
146
147    #[test]
148    fn a_packed_coat_and_marking_survive_the_round_trip() {
149        // The two halves share one int, so an off-by-one shift would silently
150        // repaint every horse on the server.
151        for variant in HorseVariant::ALL {
152            for markings in HorseMarkings::ALL {
153                let packed = pack_type_variant(variant, markings);
154                assert_eq!(HorseVariant::by_id(packed & 0xFF), variant);
155                assert_eq!(HorseMarkings::by_id((packed & 0xFF00) >> 8), markings);
156            }
157        }
158    }
159
160    #[test]
161    fn replacing_the_coat_leaves_the_markings_alone() {
162        let packed = pack_type_variant(HorseVariant::Black, HorseMarkings::WhiteDots);
163        let repainted = with_variant(packed, HorseVariant::Creamy);
164
165        assert_eq!(HorseVariant::by_id(repainted & 0xFF), HorseVariant::Creamy);
166        assert_eq!(
167            HorseMarkings::by_id((repainted & 0xFF00) >> 8),
168            HorseMarkings::WhiteDots
169        );
170    }
171
172    #[test]
173    fn out_of_range_ids_wrap_like_vanillas_continuous_map() {
174        assert_eq!(HorseVariant::by_id(7), HorseVariant::White);
175        assert_eq!(HorseVariant::by_id(-1), HorseVariant::DarkBrown);
176        assert_eq!(HorseMarkings::by_id(5), HorseMarkings::None);
177    }
