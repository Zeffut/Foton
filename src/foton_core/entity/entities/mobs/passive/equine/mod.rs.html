51/// Registers the goals every horse shares.
52///
53/// Vanilla parity: `AbstractHorse.registerGoals`, minus the `addBehaviourGoals`
54/// call each subclass supplies itself.
55pub(super) fn add_abstract_horse_goals(goals: &mut GoalSelector, can_perform_rearing: bool) {
56    goals.add_goal(1, RunAroundLikeCrazyGoal::new(1.2));
57    goals.add_goal(2, BreedGoal::new(1.0));
58    goals.add_goal(4, FollowParentGoal::new(1.0));
59    goals.add_goal(6, WaterAvoidingRandomStrollGoal::new(0.7));
60    goals.add_goal(7, LookAtPlayerGoal::new(6.0));
61    goals.add_goal(8, RandomLookAroundGoal::new());
62    if can_perform_rearing {
63        goals.add_goal(9, RandomStandGoal::new());
64    }
65}

67/// Registers the goals a horse that has not replaced them keeps.
68///
69/// Vanilla parity: `AbstractHorse.addBehaviourGoals`. The panic goal is vanilla's
70/// inner `MountPanicGoal`, which sits out while a mob is steering the horse.
71pub(super) fn add_default_horse_behaviour_goals(goals: &mut GoalSelector, mob_base: &MobBase) {
72    goals.add_goal(0, FloatGoal::new(mob_base));
73    goals.add_goal(
74        1,
75        PanicGoal::new(1.2).with_panic_filter(|mob| {
76            mob.as_abstract_horse()
77                .is_none_or(|horse| !horse.is_mob_controlled())
78        }),
79    );
80    goals.add_goal(
81        3,
82        TemptGoal::new(
83            1.25,
84            |item_stack| {
85                REGISTRY
86                    .items
87                    .is_in_tag(item_stack.item(), &ItemTag::HORSE_TEMPT_ITEMS)
88            },
89            false,
90        ),
91    );
92}

94/// Pushes freshly changed mob-effect state into synchronized entity data.
95///
96/// Vanilla parity: the `LivingEntity.tick` effect sync. Every Foton mob repeats
97/// this; the horse family shares one copy because eight of them would not.
98pub(super) fn sync_mob_effect_entity_data<E, D>(entity: &E, data: &SyncMutex<D>)
99where
100    E: LivingEntity + ?Sized,
101    D: VanillaLivingEntityData + VanillaEntityData + Send + Sync,
102{
103    if !entity.living_base().take_effects_dirty() {
104        return;
105    }
106
107    let display = entity.living_base().mob_effect_display_state();
108    {
109        let mut entity_data = data.lock();
110        let living = entity_data.living_entity_mut();
111        living.effect_particles.set(display.particles);
112        living.effect_ambience.set(display.ambient);
113    }
114
115    data.set_base_invisible_flag(display.invisible);
116    data.set_base_glowing_flag(entity.has_glowing_tag() || display.glowing);
117}
