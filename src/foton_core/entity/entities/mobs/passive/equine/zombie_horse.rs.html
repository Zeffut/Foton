44/// Where an undead foal carries its rider.
45///
46/// Vanilla parity: `ZombieHorse.BABY_DIMENSIONS`, attached at
47/// `ZOMBIE_HORSE.getHeight() - 0.25F` and scaled by `0.7F`.
48const ZOMBIE_HORSE_BABY_PASSENGER_ATTACHMENTS: [EntityAttachmentPoint; 1] =
49    [EntityAttachmentPoint::new(
50        0.0,
51        (1.6 - 0.25) * BABY_SCALE as f64,
52        0.0,
53    )];
54
55/// An undead foal's hitbox.
56const ZOMBIE_HORSE_BABY_DIMENSIONS: EntityDimensions = EntityDimensions::new_with_attachments(
57    1.396_484_4 * BABY_SCALE,
58    1.6 * BABY_SCALE,
59    1.52 * BABY_SCALE,
60    EntityAttachments::new(&ZOMBIE_HORSE_BABY_PASSENGER_ATTACHMENTS, &[], &[], &[]),
61);
62
63/// Divisor that turns the zombie horse's raw speed roll into an attribute.
64///
65/// Vanilla parity: `ZombieHorse.SPEED_FACTOR`.
66const SPEED_FACTOR: f64 = 42.16;
67
68/// Vanilla parity: `ZombieHorse.BASE_JUMP_STRENGTH`.
69const BASE_JUMP_STRENGTH: f64 = 0.5;
70
71/// Vanilla parity: `ZombieHorse.PER_RANDOM_JUMP_STRENGTH`.
72const PER_RANDOM_JUMP_STRENGTH: f64 = 0.066_666_666_666_666_67;
73
74/// Vanilla parity: `ZombieHorse.BASE_SPEED`.
75const BASE_SPEED: f64 = 9.0;
76
77/// Vanilla parity: `ZombieHorse.PER_RANDOM_SPEED`.
78const PER_RANDOM_SPEED: f64 = 1.0;
79
80/// Vanilla parity: `ZombieHorse.generateZombieHorseJumpStrength`.
81fn generate_zombie_horse_jump_strength(probability: &mut dyn FnMut() -> f64) -> f64 {
82    BASE_JUMP_STRENGTH
83        + probability() * PER_RANDOM_JUMP_STRENGTH
84        + probability() * PER_RANDOM_JUMP_STRENGTH
85        + probability() * PER_RANDOM_JUMP_STRENGTH
86}
87
88/// Vanilla parity: `ZombieHorse.generateZombieHorseSpeed`.
89fn generate_zombie_horse_speed(probability: &mut dyn FnMut() -> f64) -> f64 {
90    (BASE_SPEED
91        + probability() * PER_RANDOM_SPEED
92        + probability() * PER_RANDOM_SPEED
93        + probability() * PER_RANDOM_SPEED)
94        / f64::from(SPEED_FACTOR as f32)
95}

115impl ZombieHorseEntity {
116    /// Creates a new zombie horse.
117    #[must_use]
118    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
119        Self::new_with_base(
120            EntityBase::new(id, position, entity_type.dimensions, world),
121            entity_type,
122        )
123    }
124
125    /// Creates a zombie horse from saved base data.
126    #[must_use]
127    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
128        Self::new_with_base(
129            EntityBase::from_load(load, entity_type.dimensions),
130            entity_type,
131        )
132    }
133
134    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
135        let living_base = LivingEntityBase::new(entity_type);
136        let mob_base = MobBase::new();
137        let ageable_base = AgeableMobBase::new();
138        let animal_base = AnimalBase::new();
139        AnimalBase::initialize_pathfinding_malus(&mob_base);
140        let mut entity_data = ZombieHorseEntityData::new();
141        living_base.initialize_synced_data(&mut entity_data);
142
143        {
144            let mut goals = mob_base.goal_selector().lock();
145            add_abstract_horse_goals(&mut goals, true);
146            // Vanilla parity: `ZombieHorse.addBehaviourGoals` keeps the float and
147            // the tempt goal but drops the mount panic goal.
148            goals.add_goal(0, FloatGoal::new(&mob_base));
149            goals.add_goal(
150                3,
151                TemptGoal::new(
152                    1.25,
153                    |item_stack| {
154                        REGISTRY
155                            .items
156                            .is_in_tag(item_stack.item(), &ItemTag::ZOMBIE_HORSE_FOOD)
157                    },
158                    false,
159                ),
160            );
161        }
162
163        Self {
164            base,
165            entity_type,
166            living_base,
167            mob_base,
168            ageable_base,
169            animal_base,
170            horse_base: AbstractHorseBase::new(0),
171            entity_data: SyncMutex::new(entity_data),
172        }
173    }
174}

194    fn dimensions_for_pose(&self, _pose: EntityPose) -> EntityDimensions {
195        let scale = LivingEntity::get_scale(self);
196        if AgeableMob::is_baby(self) {
