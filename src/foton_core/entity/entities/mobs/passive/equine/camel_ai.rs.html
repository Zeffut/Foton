1//! The camel's brain, shared by the camel and the camel husk.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.camel.CamelAi`. Two
4//! activities, and what makes it a camel rather than a horse is `RandomSitting`
5//! -- an idle camel that has been in one pose for twenty seconds flips to the
6//! other, which is why a wild camel is usually found sitting down.
7//!
8//! `CamelHusk` inherits this whole brain unchanged, `AnimalMakeLove` on
9//! `EntityType.CAMEL` included. That behavior is inert for a husk, which cannot
10//! fall in love at all, exactly as it is in vanilla.
11

30/// Vanilla parity: `CamelAi.SPEED_MULTIPLIER_WHEN_PANICKING`.
31const SPEED_MULTIPLIER_WHEN_PANICKING: f64 = 4.0;
32/// Vanilla parity: `CamelAi.SPEED_MULTIPLIER_WHEN_IDLING`.
33const SPEED_MULTIPLIER_WHEN_IDLING: f64 = 2.0;
34/// Vanilla parity: `CamelAi.SPEED_MULTIPLIER_WHEN_TEMPTED`.
35const SPEED_MULTIPLIER_WHEN_TEMPTED: f64 = 2.5;
36/// Vanilla parity: `CamelAi.SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT`.
37const SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT: f64 = 2.5;
38/// Vanilla parity: `CamelAi.SPEED_MULTIPLIER_WHEN_MAKING_LOVE`, the default of
39/// the one-argument `AnimalMakeLove`.
40const SPEED_MULTIPLIER_WHEN_MAKING_LOVE: f64 = 1.0;
41/// Vanilla parity: the `1` close-enough of the same constructor.
42const MAKE_LOVE_CLOSE_ENOUGH: i32 = 1;
43/// Vanilla parity: the two `closeEnoughDist` of the camel's `FollowTemptation`.
44const BABY_CLOSE_ENOUGH_DIST: f64 = 2.5;
45const ADULT_CLOSE_ENOUGH_DIST: f64 = 3.5;
46/// Vanilla parity: `CamelAi.ADULT_FOLLOW_RANGE`.
47const ADULT_FOLLOW_RANGE: UniformIntProvider = UniformIntProvider {
48    min_inclusive: 5,
49    max_inclusive: 16,
50};
51/// Vanilla parity: the `0.8F` of the core activity's `Swim`.
52const SWIM_CHANCE: f32 = 0.8;
53/// Vanilla parity: the `LookAtTargetSink(45, 90)` of the core activity.
54const LOOK_AT_TARGET_MIN_DURATION: i32 = 45;
55const LOOK_AT_TARGET_MAX_DURATION: i32 = 90;
56/// Vanilla parity: the `SetEntityLookTargetSometimes.create(PLAYER, 6.0F, UniformInt.of(30, 60))`.
57const GAZE_RANGE: f64 = 6.0;
58const GAZE_INTERVAL: UniformIntProvider = UniformIntProvider {
59    min_inclusive: 30,
60    max_inclusive: 60,
61};

63const LOOK_AROUND_INTERVAL: UniformIntProvider = UniformIntProvider {
64    min_inclusive: 150,
65    max_inclusive: 250,
66};
67const LOOK_AROUND_MAX_YAW: f32 = 30.0;
68const LOOK_AROUND_MIN_PITCH: f32 = 0.0;
69const LOOK_AROUND_MAX_PITCH: f32 = 0.0;

71const WALK_TO_LOOK_TARGET_CLOSE_ENOUGH: i32 = 3;

73const DO_NOTHING_MIN: i32 = 30;
74const DO_NOTHING_MAX: i32 = 60;
75/// Vanilla parity: the `new RandomSitting(20)` of the gate -- twenty seconds in
76/// one pose before it will consider the other.
77const RANDOM_SITTING_MINIMAL_POSE_SECONDS: i64 = 20;
78const TICKS_PER_SECOND: i64 = 20;

88/// Vanilla parity: `Camel.BRAIN_PROVIDER` plus `CamelAi.getActivities`.
89#[must_use]
90pub(super) fn make_brain(hooks: CamelHooks) -> Brain {
91    Brain::new(SENSORS, vec![core_activity(hooks), idle_activity(hooks)])
92}

94/// Vanilla parity: `CamelAi.initCoreActivity`.
95fn core_activity(hooks: CamelHooks) -> ActivityData {
96    ActivityData::create(
97        Activity::Core,
98        0,
99        vec![
100            Behavior::boxed(Swim::new(SWIM_CHANCE)),
101            // Vanilla parity: `CamelAi.CamelPanic`, which stands the camel up
102            // instantly before it runs and refuses to fire while a player is
103            // steering.
104            Behavior::boxed(CamelPanic::new(hooks)),
105            Behavior::boxed(LookAtTargetSink::new(
106                LOOK_AT_TARGET_MIN_DURATION,
107                LOOK_AT_TARGET_MAX_DURATION,
108            )),
109            Behavior::boxed(MoveToTargetSink::new()),
110            Behavior::boxed(CountDownCooldownTicks::new(
111                memory_module_types::TEMPTATION_COOLDOWN_TICKS,
112            )),
113            Behavior::boxed(CountDownCooldownTicks::new(
114                memory_module_types::GAZE_COOLDOWN_TICKS,
115            )),
116        ],
117    )
118}

120/// Vanilla parity: `CamelAi.initIdleActivity`.
121fn idle_activity(hooks: CamelHooks) -> ActivityData {
122    ActivityData::with_priorities(
123        Activity::Idle,
124        vec![
125            (
126                0,
127                OneShot::boxed(SetEntityLookTargetSometimes::of_type(
128                    &vanilla_entities::PLAYER,
129                    GAZE_RANGE,
130                    GAZE_INTERVAL,
131                )),
132            ),
133            (
134                1,
135                Behavior::boxed(AnimalMakeLove::new(
136                    &vanilla_entities::CAMEL,
137                    SPEED_MULTIPLIER_WHEN_MAKING_LOVE,
138                    MAKE_LOVE_CLOSE_ENOUGH,
139                )),
140            ),
141            (
142                2,
143                Box::new(RunOne::unconditional(vec![
144                    (
145                        Behavior::boxed(
146                            FollowTemptation::new(|_| SPEED_MULTIPLIER_WHEN_TEMPTED)
147                                .with_close_enough_distance(|mob| {
148                                    if mob.as_ageable_mob().is_some_and(AgeableMob::is_baby) {
149                                        BABY_CLOSE_ENOUGH_DIST
150                                    } else {
151                                        ADULT_CLOSE_ENOUGH_DIST
152                                    }
153                                }),
154                        ),
155                        1,
156                    ),
157                    (
158                        while_camel_will_move(
159                            hooks,
160                            OneShot::boxed(BabyFollowAdult::new(
161                                ADULT_FOLLOW_RANGE,
162                                SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT,
163                            )),
164                        ),
165                        1,
166                    ),
167                ])),
168            ),
169            (
170                3,
171                Behavior::boxed(RandomLookAround::new(
172                    LOOK_AROUND_INTERVAL,
173                    LOOK_AROUND_MAX_YAW,
174                    LOOK_AROUND_MIN_PITCH,
175                    LOOK_AROUND_MAX_PITCH,
176                )),
177            ),
178            (
179                4,
180                Box::new(RunOne::gated(
181                    vec![(
182                        memory_module_types::WALK_TARGET.id(),
183                        MemoryStatus::ValueAbsent,
184                    )],
185                    vec![
186                        (
187                            while_camel_will_move(
188                                hooks,
189                                OneShot::boxed(RandomStroll::stroll(SPEED_MULTIPLIER_WHEN_IDLING)),
190                            ),
191                            1,
192                        ),
193                        (
194                            while_camel_will_move(
195                                hooks,
196                                OneShot::boxed(SetWalkTargetFromLookTarget::new(
197                                    SPEED_MULTIPLIER_WHEN_IDLING,
198                                    WALK_TO_LOOK_TARGET_CLOSE_ENOUGH,
199                                )),
200                            ),
201                            1,
202                        ),
203                        (
204                            Behavior::boxed(RandomSitting::new(
205                                hooks,
206                                RANDOM_SITTING_MINIMAL_POSE_SECONDS,
207                            )),
208                            1,
209                        ),
210                        (Box::new(DoNothing::new(DO_NOTHING_MIN, DO_NOTHING_MAX)), 1),
211                    ],
212                )),
213            ),
214        ],
215    )
216}

223/// Vanilla parity: the `BehaviorBuilder.triggerIf(Predicate.not(Camel::refuseToMove), ...)`
224/// three of the camel's idle behaviors are wrapped in -- a sitting camel is not
225/// asked to walk anywhere in the first place.
226struct WhileCamelWillMove {
227    hooks: CamelHooks,
228    inner: Box<dyn BehaviorControl>,
229}
230
231fn while_camel_will_move(
232    hooks: CamelHooks,
233    inner: Box<dyn BehaviorControl>,
234) -> Box<dyn BehaviorControl> {
235    Box::new(WhileCamelWillMove { hooks, inner })
236}
237
238impl BehaviorControl for WhileCamelWillMove {
239    fn status(&self) -> BehaviorStatus {
240        self.inner.status()
241    }
242
243    fn required_memories(&self) -> Vec<MemoryModuleId> {
244        self.inner.required_memories()
245    }
246
247    fn try_start(&mut self, ctx: &BrainContext<'_>) -> bool {
248        !(self.hooks.refuses_to_move)(ctx.mob()) && self.inner.try_start(ctx)
249    }
250
251    fn tick_or_stop(&mut self, ctx: &BrainContext<'_>) {
252        self.inner.tick_or_stop(ctx);
253    }
254
255    fn do_stop(&mut self, ctx: &BrainContext<'_>) {
256        self.inner.do_stop(ctx);
257    }
258
259    fn debug_name(&self) -> &'static str {
260        self.inner.debug_name()
261    }
262}

264/// Vanilla parity: `CamelAi.CamelPanic`, which will not fire while a player is
265/// steering and stands the camel up before it runs.
266struct CamelPanic {
267    hooks: CamelHooks,
268    inner: AnimalPanic,
269}
270
271impl CamelPanic {
272    fn new(hooks: CamelHooks) -> Self {
273        Self {
274            hooks,
275            inner: AnimalPanic::new(SPEED_MULTIPLIER_WHEN_PANICKING),
276        }
277    }
278}
279
280impl TimedBehavior for CamelPanic {
281    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
282        self.inner.entry_condition()
283    }
284
285    fn duration(&self) -> (i32, i32) {
286        self.inner.duration()
287    }
288
289    fn times_out(&self) -> bool {
290        self.inner.times_out()
291    }
292
293    fn check_extra_start_conditions(&mut self, ctx: &BrainContext<'_>) -> bool {
294        !(self.hooks.is_mob_controlled)(ctx.mob()) && self.inner.check_extra_start_conditions(ctx)
295    }
296
297    fn can_still_use(&mut self, ctx: &BrainContext<'_>) -> bool {
298        self.inner.can_still_use(ctx)
299    }
300
301    fn start(&mut self, ctx: &BrainContext<'_>) {
302        (self.hooks.stand_up_instantly)(ctx.mob());
303        self.inner.start(ctx);
304    }
305
306    fn tick(&mut self, ctx: &BrainContext<'_>) {
307        self.inner.tick(ctx);
308    }
309
310    fn stop_with_timeout(&mut self, ctx: &BrainContext<'_>, timed_out: bool) {
311        self.inner.stop_with_timeout(ctx, timed_out);
312    }
313
314    fn debug_name(&self) -> &'static str {
315        "CamelPanic"
316    }
317}

319/// Vanilla parity: `CamelAi.RandomSitting`.
320///
321/// One behavior for both directions: a camel that has held a pose for twenty
322/// seconds flips to the other one, which is why an undisturbed camel spends
323/// most of its life sitting down.
324struct RandomSitting {
325    hooks: CamelHooks,
326    minimal_pose_ticks: i64,
327}
328
329impl RandomSitting {
330    const fn new(hooks: CamelHooks, minimal_pose_seconds: i64) -> Self {
331        Self {
332            hooks,
333            minimal_pose_ticks: minimal_pose_seconds * TICKS_PER_SECOND,
334        }
335    }
336}
337
338impl TimedBehavior for RandomSitting {
339    fn entry_condition(&self) -> &[(MemoryModuleId, MemoryStatus)] {
340        &[]
341    }
342
343    fn check_extra_start_conditions(&mut self, ctx: &BrainContext<'_>) -> bool {
344        (self.hooks.can_random_sit)(ctx.mob(), self.minimal_pose_ticks)
345    }
346
347    fn start(&mut self, ctx: &BrainContext<'_>) {
348        (self.hooks.random_sit)(ctx.mob());
349    }
350
351    fn debug_name(&self) -> &'static str {
352        "RandomSitting"
353    }
