1//! The camel husk.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.camel.CamelHusk`, which
4//! extends `Camel` and adds no state at all: it swaps every sound, eats a
5//! different tag, will not breed and will not be a baby, and it despawns like a
6//! monster rather than sticking around like an animal. Everything else is
7//! [`super::camel_common`].
8//!
9//! One vanilla override has no reader in Foton yet and is deliberately absent:
10//! `chargeSpeedModifier`, which returns `4.0F` so a rider charging with a spear
11//! is carried four times as fast. Foton has no `SpearUseGoal`, `SpearAttack`,
12//! `SpearApproach` or `SpearRetreat`, so nothing would ever read the value; it
13//! arrives with the spear behaviors that want it.
14

67impl CamelHuskEntity {
68    /// Creates a camel husk at runtime.
69    #[must_use]
70    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
71        Self::new_with_base(
72            EntityBase::new(id, position, entity_type.dimensions, world),
73            entity_type,
74        )
75    }
76
77    /// Creates a camel husk from saved base data.
78    #[must_use]
79    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
80        Self::new_with_base(
81            EntityBase::from_load(load, entity_type.dimensions),
82            entity_type,
83        )
84    }
85
86    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
87        let living_base = LivingEntityBase::new(entity_type);
88        let mob_base = MobBase::new();
89        let ageable_base = AgeableMobBase::new();
90        let animal_base = AnimalBase::new();
91        AnimalBase::initialize_pathfinding_malus(&mob_base);
92        {
93            // Vanilla parity: the two navigation calls the `Camel` constructor
94            // makes, which `CamelHusk` inherits.
95            let mut navigation = mob_base.navigation().lock();
96            navigation.set_can_float(true);
97            navigation.set_can_walk_over_fences(true);
98        }
99        let mut entity_data = CamelHuskEntityData::new();
100        living_base.initialize_synced_data(&mut entity_data);
101
102        Self {
103            base,
104            entity_type,
105            living_base,
106            mob_base,
107            ageable_base,
108            animal_base,
109            abstract_horse_base: AbstractHorseBase::new(0),
110            camel_base: CamelBase::new(),
111            brain: camel_ai::make_brain(hooks_for::<Self>()),
112            entity_data: SyncMutex::new(entity_data),
113        }
114    }
115}

587            &vanilla_entities::CAMEL_HUSK,
588            next_entity_id(),
589            SPAWN,

598    /// The husk and the camel eat from different tags. Sharing one base is only
599    /// safe if the hook the base reads is the subclass's, so this is the check
600    /// that the override is reached rather than shadowed by the camel's.
601    #[test]
602    fn a_camel_husk_eats_from_its_own_tag_and_a_camel_from_the_camels() {
603        let husk = detached_husk();
604        let camel = CamelEntity::new(
605            &vanilla_entities::CAMEL,
606            next_entity_id(),
607            SPAWN,
608            Weak::new(),
609        );
610
611        let cactus = ItemStack::new(&vanilla_items::CACTUS);
612        let rabbit_foot = ItemStack::new(&vanilla_items::RABBIT_FOOT);
613
614        assert!(
615            Animal::is_food(&camel, &cactus),
616            "`#camel_food` is the cactus"
617        );
618        assert!(
619            !Animal::is_food(&camel, &rabbit_foot),
620            "and a camel wants nothing to do with a rabbit's foot"
621        );
622        assert!(
623            Animal::is_food(&husk, &rabbit_foot),
624            "`#camel_husk_food` is the rabbit's foot"
625        );
626        assert!(
627            !Animal::is_food(&husk, &cactus),
628            "a husk does not eat what a camel eats"

661    /// Vanilla's `CamelHusk.canBeLeashed` refuses while a mob is in the saddle,
