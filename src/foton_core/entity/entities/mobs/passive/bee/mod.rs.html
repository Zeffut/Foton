1//! Bee entity.
2//!
3//! Vanilla parity: `Bee`. A bee is built around two block positions it
4//! remembers -- a hive and a flower -- and everything it does is a trip between
5//! them. It leaves the hive, finds a flower, hovers over it until it is carrying
6//! nectar, grows whatever crop it happens to fly over on the way, and comes back
7//! to raise the hive's honey level. Provoke it and it stings once, which fills
8//! the target with poison and kills the bee within the minute.
9//!
10//! Foton already had beehives, bee nests, honey levels, comparator output and
11//! harvesting; this is the occupant those blocks were written for.

140/// Runtime bee fields vanilla keeps on the entity.
141///
142/// Two of these -- `pollinating` and `blacklisted_hives` -- live on the goals in
143/// vanilla, where an inner class reaches its sibling's fields directly. Foton's
144/// goals are boxed behind the goal selector's mutex, so a goal that read another
145/// goal's field would have to re-lock the selector from inside its own tick.
146/// They are shared state, so they sit here instead.
147#[derive(Debug, Clone, PartialEq, Eq)]
148struct BeeState {
149    /// Vanilla `Bee.hivePos`.
150    hive_pos: Option<BlockPos>,
151    /// Vanilla `Bee.savedFlowerPos`.
152    saved_flower_pos: Option<BlockPos>,
153    /// Vanilla `Bee.timeSinceSting`.
154    time_since_sting: i32,
155    /// Vanilla `Bee.ticksWithoutNectarSinceExitingHive`.
156    ticks_without_nectar_since_exiting_hive: i32,
157    /// Vanilla `Bee.stayOutOfHiveCountdown`.
158    stay_out_of_hive_countdown: i32,
159    /// Vanilla `Bee.numCropsGrownSincePollination`.
160    num_crops_grown_since_pollination: i32,
161    /// Vanilla `Bee.remainingCooldownBeforeLocatingNewHive`.
162    remaining_cooldown_before_locating_new_hive: i32,
163    /// Vanilla `Bee.remainingCooldownBeforeLocatingNewFlower`.
164    remaining_cooldown_before_locating_new_flower: i32,
165    /// Vanilla `Bee.underWaterTicks`.
166    under_water_ticks: i32,
167    /// Vanilla `Bee.BeePollinateGoal.pollinating`.
168    pollinating: bool,
169    /// Vanilla `Bee.BeeGoToHiveGoal.blacklistedTargets`.
170    blacklisted_hives: Vec<BlockPos>,
171}

173impl BeeState {
174    fn new() -> Self {
175        Self {
176            hive_pos: None,
177            saved_flower_pos: None,
178            time_since_sting: 0,
179            ticks_without_nectar_since_exiting_hive: 0,
180            stay_out_of_hive_countdown: 0,
181            num_crops_grown_since_pollination: 0,
182            remaining_cooldown_before_locating_new_hive: 0,

184            // does not go looking for the same flower on the same tick.
185            remaining_cooldown_before_locating_new_flower: rand::random_range(
186                MIN_FIND_FLOWER_RETRY_COOLDOWN..MAX_FIND_FLOWER_RETRY_COOLDOWN,
187            ),
188            under_water_ticks: 0,
189            pollinating: false,
190            blacklisted_hives: Vec::new(),
191        }
192    }

233    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
234        let living_base = LivingEntityBase::new(entity_type);
235        let mob_base = MobBase::new();
236        let ageable_base = AgeableMobBase::new();
237        let animal_base = AnimalBase::new();
238        AnimalBase::initialize_pathfinding_malus(&mob_base);
239        {
240            // Vanilla parity: the five `setPathfindingMalus` calls of the `Bee`
241            // constructor. Fire and water are walls, the waterline is merely
242            // expensive, and a bee will not squeeze past cocoa or a fence.
243            let mut malus = mob_base.pathfinding_malus().lock();
244            malus.set(PathType::Fire, -1.0);
245            malus.set(PathType::Water, -1.0);
246            malus.set(PathType::WaterBorder, 16.0);
247            malus.set(PathType::Cocoa, -1.0);
248            malus.set(PathType::Fence, -1.0);
249        }
250        {
251            // Vanilla parity: `Bee.createNavigation`.
252            let mut navigation = mob_base.navigation().lock();
253            navigation.set_can_open_doors(false);
254            navigation.set_can_float(false);
255            navigation
256                .set_required_path_length(REQUIRED_PATH_LENGTH, f64::from(REQUIRED_PATH_LENGTH));
257        }
258
259        let mut entity_data = BeeEntityData::new();
260        living_base.initialize_synced_data(&mut entity_data);
261
262        {
263            // Vanilla parity: `Bee.registerGoals`.
264            let mut goals = mob_base.goal_selector().lock();
265            goals.add_goal(0, BeeAttackGoal::new(ATTACK_SPEED_MOD));
266            goals.add_goal(1, BeeEnterHiveGoal::new());
267            goals.add_goal(2, BreedGoal::new(BREED_SPEED_MOD));
268            goals.add_goal(3, TemptGoal::new(TEMPT_SPEED_MOD, Self::is_bee_food, false));
269            goals.add_goal(3, ValidateHiveGoal::new());
270            goals.add_goal(3, ValidateFlowerGoal::new());
271            goals.add_goal(4, BeePollinateGoal::new());
272            goals.add_goal(5, FollowParentGoal::new(FOLLOW_PARENT_SPEED_MOD));
273            goals.add_goal(5, BeeLocateHiveGoal::new());
274            goals.add_goal(5, BeeGoToHiveGoal::new());
275            goals.add_goal(6, BeeGoToKnownFlowerGoal::new());
276            goals.add_goal(7, BeeGrowCropGoal::new());
277            goals.add_goal(8, BeeWanderGoal::new());
278            goals.add_goal(9, FloatGoal::new(&mob_base));
279        }
280        {
281            let mut targets = mob_base.target_selector().lock();
282            targets.add_goal(1, BeeHurtByOtherGoal::new());
283            targets.add_goal(2, BeeBecomeAngryTargetGoal::new());
284            targets.add_goal(3, ResetUniversalAngerTargetGoal::new(true));
285        }
286
287        Self {
288            base,
289            entity_type,
290            living_base,
291            mob_base,
292            ageable_base,
293            animal_base,
294            anger: PersistentAnger::new(),
295            state: SyncMutex::new(BeeState::new()),
296            entity_data: SyncMutex::new(entity_data),
297        }
298    }

403    /// Vanilla parity: `Bee.BeeGoToHiveGoal.blacklistTarget`, which remembers the
404    /// last three hives that disappointed this bee.
405    pub(super) fn blacklist_hive(&self, max_blacklisted: usize) {
406        let mut state = self.state.lock();
407        let Some(hive_pos) = state.hive_pos else {
408            return;
409        };
410        state.blacklisted_hives.push(hive_pos);
411        while state.blacklisted_hives.len() > max_blacklisted {
412            state.blacklisted_hives.remove(0);
413        }
414    }

416    /// Takes the closest hive that is not blacklisted, clearing the blacklist if
417    /// they all are.
418    ///
419    /// Vanilla parity: the body of `Bee.BeeLocateHiveGoal.start`.
420    pub(super) fn adopt_first_unblacklisted_hive(&self, hives: &[BlockPos]) {
421        let mut state = self.state.lock();
422        for candidate in hives {
423            if !state.blacklisted_hives.contains(candidate) {
424                state.hive_pos = Some(*candidate);
425                return;
426            }
427        }
428
429        // Everything nearby has already disappointed this bee once. Vanilla wipes
430        // the slate rather than leaving it homeless.
431        state.blacklisted_hives.clear();
432        state.hive_pos = hives.first().copied();
433    }

518    /// Vanilla parity: the private `Bee.closerThan`, which measures block to
519    /// block rather than from the bee's exact position.
520    pub(super) fn closer_than(&self, target: BlockPos, distance: i32) -> bool {
521        let mob_pos = self.block_position();
522        let dx = f64::from(target.x() - mob_pos.x());
523        let dy = f64::from(target.y() - mob_pos.y());
524        let dz = f64::from(target.z() - mob_pos.z());
525        let distance = f64::from(distance);
526        dx.mul_add(dx, dy.mul_add(dy, dz * dz)) < distance * distance
527    }

529    /// Runs `read` against this bee's hive, if it still has a reachable one.
530    ///
531    /// Vanilla parity: `Bee.getBeehiveBlockEntity`, which refuses a hive the bee
532    /// has drifted too far from rather than reaching into an unloaded chunk.
533    /// Foton hands the block entity to a closure rather than returning it,
534    /// because a `SharedBlockEntity` can only be downcast by reference.
535    pub(super) fn with_beehive<R>(&self, read: impl FnOnce(&BeehiveBlockEntity) -> R) -> Option<R> {
536        let hive_pos = self.hive_pos()?;
537        if self.is_too_far_away(hive_pos) {
538            return None;
539        }
540
541        let world = self.level()?;
542        let block_entity = world.get_block_entity(hive_pos)?;

573    /// Vanilla parity: `Bee.wantsToEnterHive`.
574    ///
575    /// In 26.2 the "is it night or raining" test the bee used to make itself is
576    /// the `gameplay/bees_stay_in_hive` environment attribute, so a datapack
577    /// timeline decides the bees' curfew rather than the entity.
578    pub(super) fn wants_to_enter_hive(&self) -> bool {
579        let (stay_out, pollinating) = {
580            let state = self.state.lock();
581            (state.stay_out_of_hive_countdown, state.pollinating)
582        };
583        if stay_out > 0 || pollinating || self.has_stung() || self.target().is_some() {
584            return false;
585        }

588        let wants = self.has_nectar() || self.is_tired_of_looking_for_nectar() || curfew;
589        wants && !self.is_hive_near_fire()
590    }

592    /// Vanilla parity: `Bee.attractsBees`, the block test behind both the
593    /// pollination search and the flower the bee remembers.
594    #[must_use]
595    pub fn attracts_bees(state: BlockStateId) -> bool {
596        if !state.get_block().has_tag(&BlockTag::BEE_ATTRACTIVE) {
597            return false;
598        }
599        if state
600            .try_get_value(&BlockStateProperties::WATERLOGGED)
601            .unwrap_or(false)
602        {
603            return false;
604        }
605        if state.get_block() == &vanilla_blocks::SUNFLOWER {
606            // Only the upper half of a sunflower is a flower to a bee.
607            return state
608                .try_get_value(&BlockStateProperties::DOUBLE_BLOCK_HALF)
609                .is_some_and(|half| half == DoubleBlockHalf::Upper);
610        }
611        true
612    }

897    /// Vanilla parity: `Bee.customServerAiStep` -- drowning, the sting clock, the
898    /// hunger clock, and the anger reconciliation.
899    fn custom_server_ai_step(&self) {
900        Animal::custom_server_ai_step_animal(self);
901
902        let Some(world) = self.level() else {
903            return;
904        };
905
906        let under_water_ticks = {
907            let mut state = self.state.lock();
908            state.under_water_ticks = if self.is_in_water() {
909                state.under_water_ticks + 1
910            } else {
911                0
912            };
913            state.under_water_ticks
914        };
915        if under_water_ticks > UNDERWATER_DROWN_TICKS {
916            self.hurt_server(
917                &world,
918                &DamageSource::environment(&vanilla_damage_types::DROWN),
919                DROWN_DAMAGE,
920            );
921        }
922
923        if self.has_stung() {
924            let time_since_sting = {
925                let mut state = self.state.lock();
926                state.time_since_sting += 1;
927                state.time_since_sting
928            };
929            // Vanilla parity: a bee that has stung rolls every fifth tick against
930            // an ever-shortening window, so it dies somewhere inside the minute
931            // rather than exactly at the end of one.
932            let window = (STING_DEATH_COUNTDOWN - time_since_sting).clamp(1, STING_DEATH_COUNTDOWN);
933            if time_since_sting % STING_DEATH_ROLL_INTERVAL == 0
934                && rand::random_range(0..window) == 0
935            {
936                self.hurt_server(
937                    &world,
938                    &DamageSource::environment(&vanilla_damage_types::GENERIC),
939                    self.get_health(),
940                );
941            }
942        }
943
944        if !self.has_nectar() {
945            self.state.lock().ticks_without_nectar_since_exiting_hive += 1;
946        }
947
948        self.update_persistent_anger(&world, false);
949    }

951    /// Vanilla parity: `Bee.doHurtTarget`, the sting: poison scaled by difficulty
952    /// and the bee's own death sentence.
953    fn do_hurt_target(&self, world: &World, target: &SharedEntity) -> bool {
954        let damage = self
955            .attributes()
956            .lock()
957            .required_value(vanilla_attributes::ATTACK_DAMAGE) as f32;
958        let source = DamageSource::environment(&vanilla_damage_types::STING)
959            .with_causing_entity(self.id())
960            .with_direct_entity(self.id());
961
962        let Some(living) = target.as_living_entity() else {
963            return false;
964        };
965        if !living.hurt_server(world, &source, damage) {
966            return false;
967        }
968
969        let poison_seconds = match world.difficulty() {
970            Difficulty::Normal => POISON_SECONDS_NORMAL,
971            Difficulty::Hard => POISON_SECONDS_HARD,
972            Difficulty::Peaceful | Difficulty::Easy => 0,
973        };
974        if poison_seconds > 0 {
975            living.add_mob_effect(MobEffectInstance::with_duration(
976                vanilla_mob_effects::POISON,
977                poison_seconds * 20,
978                0,
979            ));
980        }
981
982        self.set_has_stung(true);
983        self.stop_being_angry();
984        self.play_sound(&sound_events::ENTITY_BEE_STING, 1.0, 1.0);
985        true
986    }
