1//! Where a parrot chooses to fly next.
2//!
3//! Vanilla parity: `Parrot.ParrotWanderGoal`, which is a
4//! `WaterAvoidingRandomFlyingGoal` that first looks for a branch to land on.
5
6use foton_registry::blocks::block_state_ext::BlockStateExt as _;
7use foton_registry::vanilla_block_tags::BlockTag;
8use foton_utils::BlockPos;
9use glam::DVec3;
10
11use crate::entity::PathfinderMob;
12use crate::entity::ai::goal::{
13    WATER_AVOIDING_RANDOM_STROLL_PROBABILITY, flying_stroll_position, land_random_pos,
14};
15
16/// How far a swimming parrot looks for dry land.
17///
18/// Vanilla parity: the `LandRandomPos.getPos(mob, 15, 15)` of `getPosition`.
19const WATER_ESCAPE_RANGE: i32 = 15;
20
21/// How far to the side a parrot looks for a branch.
22///
23/// Vanilla parity: the `+/- 3.0` of `ParrotWanderGoal.getTreePos`.
24const TREE_SEARCH_HORIZONTAL: i32 = 3;
25
26/// How far up and down a parrot looks for a branch.
27///
28/// Vanilla parity: the `+/- 6.0` of the same method.
29const TREE_SEARCH_VERTICAL: i32 = 6;
30
31/// Vanilla parity: `Parrot.ParrotWanderGoal.getPosition`.
32#[must_use]
33pub(super) fn parrot_wander_position(mob: &dyn PathfinderMob) -> Option<DVec3> {
34    let mut position = if mob.is_in_water() {
35        land_random_pos(mob, WATER_ESCAPE_RANGE, WATER_ESCAPE_RANGE)
36    } else {
37        None
38    };
39
40    if rand::random::<f32>() >= WATER_AVOIDING_RANDOM_STROLL_PROBABILITY {
41        position = tree_position(mob);
42    }
43
44    position.or_else(|| flying_stroll_position(mob))
45}
46
47/// Finds a perch: an empty block, empty above it, sitting on leaves or a log.
48///
49/// Vanilla parity: `ParrotWanderGoal.getTreePos`.
50fn tree_position(mob: &dyn PathfinderMob) -> Option<DVec3> {
51    let world = mob.level()?;
52    let mob_pos = mob.block_position();
53    let origin = mob.position();
54    let min = BlockPos::containing(
55        origin.x - f64::from(TREE_SEARCH_HORIZONTAL),
56        origin.y - f64::from(TREE_SEARCH_VERTICAL),
57        origin.z - f64::from(TREE_SEARCH_HORIZONTAL),
58    );
59    let max = BlockPos::containing(
60        origin.x + f64::from(TREE_SEARCH_HORIZONTAL),
61        origin.y + f64::from(TREE_SEARCH_VERTICAL),
62        origin.z + f64::from(TREE_SEARCH_HORIZONTAL),
63    );
64
65    for x in min.x()..=max.x() {
66        for y in min.y()..=max.y() {
67            for z in min.z()..=max.z() {
68                let pos = BlockPos::new(x, y, z);
69                if pos == mob_pos {
70                    continue;
71                }
72
73                let below = world.get_block_state(pos.below()).get_block();
74                let can_sit_on = below.has_tag(&BlockTag::LEAVES) || below.has_tag(&BlockTag::LOGS);
75                if !can_sit_on
76                    || !world.get_block_state(pos).is_air()
77                    || !world.get_block_state(pos.above()).is_air()
78                {
79                    continue;
80                }
81
82                let (bottom_x, bottom_y, bottom_z) = pos.get_bottom_center();
83                return Some(DVec3::new(bottom_x, bottom_y, bottom_z));
84            }
85        }
86    }
87
88    None
89}
