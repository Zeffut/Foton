1//! The goat's brain.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.goat.GoatAi`. Four
4//! activities: the shared core, an idle set, the long jump, and the ram.
5//!
6//! A goat is the one vanilla animal with no `registerGoals` at all -- every
7//! thing it does lives here, which is why Foton's goat stood still until this
8//! landed.

112/// Vanilla parity: `Goat.BRAIN_PROVIDER` plus `GoatAi.getActivities`.

126/// Vanilla parity: `GoatAi.initMemories`, which staggers the first jump and the
127/// first ram so a herd does not act in lockstep.
128pub fn init_memories(brain: &Brain) {
129    brain.set_memory(
130        memory_module_types::LONG_JUMP_COOLDOWN_TICKS,
131        rand::random_range(
132            TIME_BETWEEN_LONG_JUMPS.min_inclusive..=TIME_BETWEEN_LONG_JUMPS.max_inclusive,
133        ),
134    );
135    brain.set_memory(
136        memory_module_types::RAM_COOLDOWN_TICKS,
137        rand::random_range(TIME_BETWEEN_RAMS.min_inclusive..=TIME_BETWEEN_RAMS.max_inclusive),
138    );
139}

141/// Vanilla parity: `GoatAi.initCoreActivity`.
142fn core_activity() -> ActivityData {
143    ActivityData::create(
144        Activity::Core,
145        0,
146        vec![
147            Behavior::boxed(Swim::new(SWIM_CHANCE)),
148            Behavior::boxed(AnimalPanic::new(SPEED_MULTIPLIER_WHEN_PANICKING)),
149            Behavior::boxed(LookAtTargetSink::new(
150                LOOK_AT_TARGET_MIN_DURATION,
151                LOOK_AT_TARGET_MAX_DURATION,
152            )),
153            Behavior::boxed(MoveToTargetSink::new()),
154            Behavior::boxed(CountDownCooldownTicks::new(
155                memory_module_types::TEMPTATION_COOLDOWN_TICKS,
156            )),
157            Behavior::boxed(CountDownCooldownTicks::new(
158                memory_module_types::LONG_JUMP_COOLDOWN_TICKS,
159            )),
160            Behavior::boxed(CountDownCooldownTicks::new(
161                memory_module_types::RAM_COOLDOWN_TICKS,
162            )),
163        ],
164    )
165}

167/// Vanilla parity: `GoatAi.initIdleActivity`.
168fn idle_activity() -> ActivityData {
169    ActivityData::with_priorities(
170        Activity::Idle,
171        vec![
172            (0, gaze_at_players()),
173            (
174                0,
175                Behavior::boxed(AnimalMakeLove::new(
176                    &vanilla_entities::GOAT,
177                    SPEED_MULTIPLIER_WHEN_IDLING,
178                    1,
179                )),
180            ),
181            (
182                1,
183                Behavior::boxed(FollowTemptation::new(|_| SPEED_MULTIPLIER_WHEN_TEMPTED)),
184            ),
185            (
186                2,
187                OneShot::boxed(BabyFollowAdult::new(
188                    ADULT_FOLLOW_RANGE,
189                    SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT,
190                )),
191            ),
192            (
193                3,
194                Box::new(RunOne::unconditional(vec![
195                    (
196                        OneShot::boxed(RandomStroll::stroll(SPEED_MULTIPLIER_WHEN_IDLING)),
197                        2,
198                    ),
199                    (
200                        OneShot::boxed(SetWalkTargetFromLookTarget::new(
201                            SPEED_MULTIPLIER_WHEN_IDLING,
202                            WALK_TO_LOOK_TARGET_CLOSE_ENOUGH,
203                        )),
204                        2,
205                    ),
206                    (
207                        Box::new(DoNothing::new(IDLE_DO_NOTHING_MIN, IDLE_DO_NOTHING_MAX)),
208                        1,
209                    ),
210                ])),
211            ),
212        ],
213    )
214    .with_conditions(vec![
215        (
216            memory_module_types::RAM_TARGET.id(),
217            MemoryStatus::ValueAbsent,
218        ),
219        (
220            memory_module_types::LONG_JUMP_MID_JUMP.id(),
221            MemoryStatus::ValueAbsent,
222        ),
223    ])
224}

270/// Vanilla parity: `GoatAi.initRamActivity`.
271fn ram_activity() -> ActivityData {
272    ActivityData::with_priorities(
273        Activity::Ram,
274        vec![
275            (0, Behavior::boxed(RamTarget::new(time_between_rams))),

284                )),

288    .with_conditions(vec![
289        (
290            memory_module_types::TEMPTING_PLAYER.id(),

294            memory_module_types::BREED_TARGET.id(),

296        ),
297        (
298            memory_module_types::RAM_COOLDOWN_TICKS.id(),

300        ),

302}

304/// Vanilla parity: `GoatAi.updateActivity`.
305pub fn update_activity(brain: &Brain) {
306    brain.set_active_activity_to_first_valid(&[Activity::Ram, Activity::LongJump, Activity::Idle]);
307}

309/// Vanilla parity: the `TIME_BETWEEN_RAMS` a screaming goat halves.
310fn time_between_rams(goat: &GoatEntity) -> (i32, i32) {
311    let range = if goat.is_screaming_goat() {
312        TIME_BETWEEN_RAMS_SCREAMER
313    } else {
314        TIME_BETWEEN_RAMS
315    };
316    (range.min_inclusive, range.max_inclusive)
317}
318
319/// Vanilla parity: the `minInclusive` a failed ram is punished with.
320fn cooldown_on_failed_ram(goat: &GoatEntity) -> i32 {
321    time_between_rams(goat).0
322}

324/// Vanilla parity: the `SetEntityLookTargetSometimes.create(PLAYER, 6.0F, UniformInt.of(30, 60))`.
325fn gaze_at_players() -> Box<dyn BehaviorControl> {
326    OneShot::boxed(SetEntityLookTargetSometimes::of_type(
327        &vanilla_entities::PLAYER,
328        GAZE_RANGE,
329        GAZE_INTERVAL,
330    ))
