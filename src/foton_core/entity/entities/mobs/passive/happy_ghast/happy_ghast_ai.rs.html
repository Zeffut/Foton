1//! The ghastling's brain.
2//!
3//! Vanilla parity:
4//! `net.minecraft.world.entity.animal.happyghast.HappyGhastAi`. Only the baby
5//! runs it -- an adult happy ghast is driven by goals -- so everything here is
6//! about a ghastling drifting after whoever it has decided to follow.

19/// Vanilla parity: `HappyGhast.SPEED_MULTIPLIER_WHEN_PANICKING`.
20const SPEED_MULTIPLIER_WHEN_PANICKING: f64 = 2.0;
21/// Vanilla parity: `HappyGhastAi.SPEED_MULTIPLIER_WHEN_IDLING`.
22const SPEED_MULTIPLIER_WHEN_IDLING: f64 = 1.0;
23/// Vanilla parity: `HappyGhastAi.SPEED_MULTIPLIER_WHEN_TEMPTED`.
24const SPEED_MULTIPLIER_WHEN_TEMPTED: f64 = 1.25;
25/// Vanilla parity: `HappyGhastAi.SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT`.
26const SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT: f64 = 1.1;
27/// Vanilla parity: `HappyGhastAi.BABY_GHAST_CLOSE_ENOUGH_DIST`.
28const BABY_GHAST_CLOSE_ENOUGH_DIST: f64 = 3.0;
29/// Vanilla parity: `HappyGhastAi.ADULT_FOLLOW_RANGE`.
30const ADULT_FOLLOW_RANGE: UniformIntProvider = UniformIntProvider {
31    min_inclusive: 3,
32    max_inclusive: 16,
33};
34/// Vanilla parity: the `0` flight height of the core activity's `AnimalPanic`,
35/// which sends a fleeing ghastling sideways rather than up.
36const PANIC_FLYING_HEIGHT: i32 = 0;
37/// Vanilla parity: the `0.8F` of the core activity's `Swim`.
38const SWIM_CHANCE: f32 = 0.8;
39/// Vanilla parity: the `LookAtTargetSink(45, 90)` of the core activity.
40const LOOK_AT_TARGET_MIN_DURATION: i32 = 45;
41const LOOK_AT_TARGET_MAX_DURATION: i32 = 90;
42/// Vanilla parity: the `SetWalkTargetFromLookTarget.create(1.0F, 3)` of the
43/// idle gate.
44const WALK_TO_LOOK_TARGET_CLOSE_ENOUGH: i32 = 3;

46/// Vanilla parity: the sensor list of `HappyGhast.BRAIN_PROVIDER`.
47const SENSORS: &[SensorType] = &[
48    SensorType::NearestLivingEntities,
49    SensorType::HurtBy,
50    SensorType::FoodTemptations,
51    SensorType::NearestAdultAnyType,

55/// Vanilla parity: `HappyGhast.BRAIN_PROVIDER` plus `HappyGhastAi.getActivities`.
56#[must_use]
57pub fn make_brain() -> Brain {
58    Brain::new(
59        SENSORS,
60        vec![core_activity(), idle_activity(), panic_activity()],
61    )
62}

64/// Vanilla parity: `HappyGhastAi.initCoreActivity`.
65fn core_activity() -> ActivityData {
66    ActivityData::create(
67        Activity::Core,
68        0,
69        vec![
70            Behavior::boxed(Swim::new(SWIM_CHANCE)),
71            Behavior::boxed(AnimalPanic::flying(
72                SPEED_MULTIPLIER_WHEN_PANICKING,
73                PANIC_FLYING_HEIGHT,
74            )),
75            Behavior::boxed(LookAtTargetSink::new(
76                LOOK_AT_TARGET_MIN_DURATION,
77                LOOK_AT_TARGET_MAX_DURATION,
78            )),
79            Behavior::boxed(MoveToTargetSink::new()),
80            Behavior::boxed(CountDownCooldownTicks::new(
81                memory_module_types::TEMPTATION_COOLDOWN_TICKS,
82            )),
83        ],
84    )
85}

87/// Vanilla parity: `HappyGhastAi.initIdleActivity`.
88///
89/// The two `BabyFollowAdult` entries are the whole of a ghastling's social
90/// life: the first trails the nearest visible player, the second any adult on
91/// the `followable_friendly_mobs` list, and a player outranks the adult.
92fn idle_activity() -> ActivityData {
93    ActivityData::with_priorities(
94        Activity::Idle,
95        vec![
96            (
97                1,
98                Behavior::boxed(
99                    FollowTemptation::new(|_| SPEED_MULTIPLIER_WHEN_TEMPTED)
100                        .with_close_enough_distance(|_| BABY_GHAST_CLOSE_ENOUGH_DIST)
101                        .looking_in_the_eyes(),
102                ),
103            ),
104            (
105                2,
106                OneShot::boxed(
107                    BabyFollowAdult::variable(ADULT_FOLLOW_RANGE, |_| {
108                        SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT
109                    })
110                    .following(memory_module_types::NEAREST_VISIBLE_PLAYER)
111                    .targeting_eye(),
112                ),
113            ),
114            (
115                3,
116                OneShot::boxed(
117                    BabyFollowAdult::variable(ADULT_FOLLOW_RANGE, |_| {
118                        SPEED_MULTIPLIER_WHEN_FOLLOWING_ADULT
119                    })
120                    .following(memory_module_types::NEAREST_VISIBLE_ADULT)
121                    .targeting_eye(),
122                ),
123            ),
124            (
125                4,
126                Box::new(RunOne::unconditional(vec![
127                    (
128                        OneShot::boxed(RandomStroll::fly(SPEED_MULTIPLIER_WHEN_IDLING)),
129                        1,
130                    ),
131                    (
132                        OneShot::boxed(SetWalkTargetFromLookTarget::new(
133                            SPEED_MULTIPLIER_WHEN_IDLING,
134                            WALK_TO_LOOK_TARGET_CLOSE_ENOUGH,
135                        )),
136                        1,
137                    ),
138                ])),
139            ),
140        ],
141    )
142}

144/// Vanilla parity: `HappyGhastAi.initPanicActivity`, which holds no behaviors
145/// at all. Its whole job is to be the activity a panicking ghastling is in, so
146/// the idle set stops running while `AnimalPanic` from the core set drives.
147fn panic_activity() -> ActivityData {
148    ActivityData::with_priorities(Activity::Panic, Vec::new()).with_conditions(vec![(
149        memory_module_types::IS_PANICKING.id(),
150        MemoryStatus::ValuePresent,
151    )])
152}
153
154/// Vanilla parity: `HappyGhastAi.updateActivity`.
155pub fn update_activity(brain: &Brain) {
156    brain.set_active_activity_to_first_valid(&[Activity::Panic, Activity::Idle]);
