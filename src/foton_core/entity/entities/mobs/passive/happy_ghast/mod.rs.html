1//! Happy ghast entity.
2//!
3//! Vanilla parity: `net.minecraft.world.entity.animal.happyghast.HappyGhast`.
4//! A happy ghast is two mobs in one body. The ghastling is a brain mob that
5//! drifts after whoever it has taken to; the adult is a goal mob, a four-seat
6//! flying vehicle, and the second holder in the game a leashable can hang four
7//! ropes off. Growing up swaps the whole control set over.
8//!
9//! **Gaps**: `getDismountLocationForPassenger` puts a rider who gets off on top
10//! of the ghast, and Foton has no dismount-location hook to hang that on -- the
11//! `DismountHelper` foundation is landed but wired to nothing, so a rider
12//! leaves from where it was sitting. `getMaxSpawnClusterSize` caps a natural
13//! spawn at one, which Foton's spawner has no per-entity hook for; that is the
14//! same gap the ghast carries. `BabyFlyingPathNavigation.setRequiredPathLength`
15//! has no equivalent, so a ghastling's paths are as long as any flier's. And
16//! `setServerStillTimeout` pushes a position packet the instant the ghast is
17//! told to hold still: Foton has no path from an entity to its tracker, so the
18//! full position `requires_precise_position` forces arrives on the next tracker
19//! pass instead of inside the same tick. `getWalkTargetValue` is written here
20//! against vanilla, but nothing in Foton reads `PathfinderMob::get_walk_target_value`
21//! yet -- the hook is landed and unwired for every mob that overrides it, this
22//! one included.
23//!
24//! Two overrides are missing here because there is nothing under them yet.
25//! `shouldStayCloseToLeashHolder` returns false on a happy ghast, and Foton has
26//! no `PathfinderMob.closeRangeLeashBehaviour` for it to gate -- no leashed mob
27//! in Foton walks back toward its holder, so the answer is already false for
28//! everybody. `PathfinderMob.whenLeashedTo` sets a home at the holder, which is
29//! what `checkRestriction` here is skipping around while leashed; Foton's
30//! `when_leashed_to` only notifies the holder, so a leashed happy ghast keeps
31//! whatever home it had.

144/// Returns whether two block positions are within `distance` of each other.
145///
146/// Vanilla parity: `Vec3i.closerThan(Vec3i, double)`.
147fn block_pos_closer_than(from: BlockPos, to: BlockPos, distance: i32) -> bool {
148    let dx = f64::from(from.x() - to.x());
149    let dy = f64::from(from.y() - to.y());
150    let dz = f64::from(from.z() - to.z());
151    let limit = f64::from(distance);
152    dx.mul_add(dx, dy.mul_add(dy, dz * dz)) < limit * limit
153}

342    /// Vanilla parity: `HappyGhast.checkRestriction`, which is what keeps a
343    /// loose happy ghast circling where it was found.
344    fn check_restriction(&self) {
345        if self.is_leashed() || self.is_vehicle() {
346            return;
347        }
348
349        let radius = self.restriction_radius();
350        let position = self.block_position();
351        let home_is_close = self.has_home()
352            && block_pos_closer_than(
353                self.home_position(),
354                position,
355                radius + RESTRICTION_RADIUS_BUFFER,
356            );
357        if !home_is_close || radius != self.home_radius() {
358            self.set_home_to(position, radius);
359        }
360    }

362    /// Vanilla parity: `HappyGhast.continuousHeal`, the slow mend that is
363    /// twenty times faster inside a cloud or under rain.
364    fn continuous_heal(&self) {
365        let Some(world) = self.level() else {
366            return;
367        };
368        if !Entity::is_alive(self)
369            || self.living_base().death_time() != 0
370            || (self.get_max_health() - self.get_health()).abs() < f32::EPSILON
371        {
372            return;
373        }
374
375        let fast = self.is_in_clouds()
376            || world.precipitation_at(self.block_position()) != Precipitation::None;
377        let interval = if fast {
378            FAST_HEALING_TICKS
379        } else {
380            SLOW_HEALING_TICKS
381        };
382        if self.tick_count() % interval == 0 {
383            self.heal(1.0);
384        }
385    }

387    /// Vanilla parity: `HappyGhast.scanPlayerAboveGhast`, which is what makes a
388    /// happy ghast freeze the moment somebody steps onto its back.
389    fn scan_player_above_ghast(&self) -> bool {
390        let Some(world) = self.level() else {
391            return false;
392        };
393        let bounding_box = self.bounding_box();
394        let detection_box = WorldAabb::new(
395            bounding_box.min_x() - PLAYER_SCAN_MARGIN,
396            bounding_box.max_y() - PLAYER_SCAN_TOP_EPSILON,
397            bounding_box.min_z() - PLAYER_SCAN_MARGIN,
398            bounding_box.max_x() + PLAYER_SCAN_MARGIN,
399            bounding_box.max_y() + bounding_box.height() / 2.0,
400            bounding_box.max_z() + PLAYER_SCAN_MARGIN,
401        );
402
403        let mut found = false;
404        world.players.iter_players(|_, player| {
405            if player.is_spectator() {
406                return true;
407            }
408            let (root_position, root_is_happy_ghast) = player.root_vehicle().map_or_else(
409                || (player.position(), false),
410                |root| (root.position(), root.downcast_ref::<Self>().is_some()),
411            );
412            if !root_is_happy_ghast && detection_box.contains(root_position) {
413                found = true;
414                return false;
415            }
416            true
417        });
418        found
419    }

490impl TemptNavigation for HappyGhastTemptNavigation {
491    fn stop_navigation(&self, mob: &dyn PathfinderMob) {

495    fn navigate_towards(&self, mob: &dyn PathfinderMob, player: &Arc<Player>, speed_modifier: f64) {
496        let position = mob.position();
497        let player_position = player.position();
498        let player_eyes = DVec3::new(player_position.x, player.get_eye_y(), player_position.z);
499        let toward_eyes = player_eyes - position;
500        let target = toward_eyes * rand::random::<f64>() + position;
501        mob.set_wanted_position(target, speed_modifier);
502    }

643
644        {
645            let mut leash_holder_time = self.leash_holder_time.lock();
646            if *leash_holder_time > 0 {
647                *leash_holder_time -= 1;
648            }
649        }
650        self.set_leash_holder(*self.leash_holder_time.lock() > 0);

655                still_timeout - 1
656            } else {
657                still_timeout
658            };
659            self.set_server_still_timeout(ticked);
660        }

664        }
665    }

790    /// Vanilla parity: `HappyGhast.getRiddenInput`, which is the whole of
791    /// steering one: the rider's pitch decides how much of the push is climb.
792    fn ridden_input(&self, controller: &Player, _self_input: DVec3) -> DVec3 {
793        let input = controller.travel_input();
794        let strafe = input.sideways();
795        let mut forward = 0.0;
796        let mut up = 0.0;
797        if input.forward() != 0.0 {
798            let pitch_radians = controller.rotation().1.to_radians();
799            let mut forward_look = pitch_radians.cos();
800            let mut up_look = -pitch_radians.sin();
801            if input.forward() < 0.0 {
802                forward_look *= RIDDEN_REVERSE_SCALE;
803                up_look *= RIDDEN_REVERSE_SCALE;
804            }
805            up = up_look;
806            forward = forward_look;
807        }
808        if controller.is_jumping() {
809            up += RIDDEN_JUMP_CLIMB;
810        }
811
812        let flying_speed = self
813            .attributes()
814            .lock()
815            .required_value(vanilla_attributes::FLYING_SPEED);
816        DVec3::new(f64::from(strafe), f64::from(up), f64::from(forward))
817            * (RIDDEN_INPUT_SCALE * flying_speed)
818    }

820    /// Vanilla parity: `HappyGhast.tickRidden`.
821    fn tick_ridden(&self, controller: &Player, _ridden_input: DVec3) {
822        let (wanted_yaw, wanted_pitch) = Self::ridden_rotation(controller);
823        let yaw = self.rotation().0;
824        let difference = wrap_degrees(wanted_yaw - yaw);
825        let yaw = difference.mul_add(RIDDEN_TURN_SPEED, yaw);
826        self.set_rotation((yaw, wanted_pitch));
827        self.base.set_old_yaw_to_current();
828        self.set_y_body_rot(yaw);
829        self.set_y_head_rot(yaw);
830    }

943            return;
944        }
945
946        if self.is_on_still_timeout() {

964            face_movement_direction(self);
965            return;
966        };

970            .atan2(wanted_position.z - position.z)
971            .to_degrees();
972        #[expect(
973            clippy::cast_possible_truncation,
974            reason = "vanilla computes the same angle as a float"
975        )]
976        let yaw = yaw as f32;
977        self.set_rotation((yaw, self.rotation().1));
978        self.set_y_body_rot(yaw);
979        self.set_y_head_rot(yaw);
980    }
