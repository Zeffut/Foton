1//! The mannequin.
2//!
3//! Vanilla parity: `Mannequin`. A player-shaped statue new in 26.2: it wears a
4//! profile's skin, holds a pose, can be nailed in place, and carries a label
5//! above its head. It has no AI at all -- like the armor stand it is a
6//! `LivingEntity` that is not a mob -- and its whole behaviour is the state it
7//! remembers.
8//!
9//! Vanilla splits it from the player through a shared `Avatar` superclass.
10//! Foton's `Player` predates that split, so the two are separate here; the
11//! `AvatarEntityData` layer both share is generated, and this entity is the
12//! only thing on it so far.
13//!
14//! Not implemented: `applyImplicitComponents` for `minecraft:profile`, which is
15//! how a placed mannequin item carries a skin onto the entity. Foton's spawn
16//! path has no item component to read, the same gap the command block's
17//! `BLOCK_ENTITY_DATA` has -- a summoned mannequin still loads and saves its
18//! profile through NBT.
19

43/// The seven parts of a player skin that can be hidden.
44///
45/// Vanilla parity: `PlayerModelPart`, whose mask is `1 << bit` and whose
46/// serialized name is what the `hidden_layers` list stores.
47const MODEL_PARTS: [(&str, i8); 7] = [
48    ("cape", 1 << 0),
49    ("jacket", 1 << 1),
50    ("left_sleeve", 1 << 2),
51    ("right_sleeve", 1 << 3),
52    ("left_pants_leg", 1 << 4),
53    ("right_pants_leg", 1 << 5),
54    ("hat", 1 << 6),
55];
56
57/// Every model part shown.
58///
59/// Vanilla parity: `Mannequin.ALL_LAYERS`. The synced byte lists the parts that
60/// are *shown*, so "all layers" is every bit set and the saved `hidden_layers`
61/// list is its complement.
62const ALL_LAYERS: i8 = 0b0111_1111;
63
64/// The poses a mannequin may be put into.
65///
66/// Vanilla parity: `Mannequin.VALID_POSES`. Anything else in the `pose` tag is
67/// rejected by the codec and falls back to standing.
68const VALID_POSES: [EntityPose; 5] = [
69    EntityPose::Standing,
70    EntityPose::Sneaking,
71    EntityPose::Swimming,
72    EntityPose::FallFlying,
73    EntityPose::Sleeping,
74];
75
76/// State a mannequin keeps that is not mirrored to clients.
77///
78/// The synced description is the *effective* one -- absent when hidden -- so
79/// the description itself has to be kept separately or hiding it would lose it.
80struct MannequinState {
81    description: TextComponent,
82    hide_description: bool,
83}

85/// A mannequin.
86#[entity_behavior(class = "Mannequin")]
87pub struct MannequinEntity {
88    base: EntityBase,
89    entity_type: EntityTypeRef,
90    living_base: LivingEntityBase,
91    entity_data: SyncMutex<MannequinEntityData>,
92    state: SyncMutex<MannequinState>,
93    /// Whether the mannequin refuses to be moved at all.
94    immovable: AtomicBool,
95}

102impl MannequinEntity {
103    /// Creates a mannequin at runtime.
104    #[must_use]
105    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
106        Self::new_with_base(
107            EntityBase::new(id, position, entity_type.dimensions, world),
108            entity_type,
109        )
110    }
111
112    /// Creates a mannequin from saved base data.
113    #[must_use]
114    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
115        Self::new_with_base(
116            EntityBase::from_load(load, entity_type.dimensions),
117            entity_type,
118        )
119    }
120
121    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
122        let living_base = LivingEntityBase::new(entity_type);
123        let mut entity_data = MannequinEntityData::new();
124        living_base.initialize_synced_data(&mut entity_data);
125        // Vanilla parity: the constructor sets every model layer visible.
126        entity_data
127            .avatar_mut()
128            .player_mode_customisation
129            .set(ALL_LAYERS);
130
131        Self {
132            base,
133            entity_type,
134            living_base,
135            entity_data: SyncMutex::new(entity_data),
136            state: SyncMutex::new(MannequinState {
137                description: default_description(),
138                hide_description: false,
139            }),
140            immovable: AtomicBool::new(false),
141        }
142    }
143
144    /// Returns the profile whose skin this mannequin wears.
145    #[must_use]
146    pub fn profile(&self) -> ResolvableProfile {
147        self.entity_data.lock().profile.get().clone()
148    }
149
150    /// Sets the profile whose skin this mannequin wears.
151    pub fn set_profile(&self, profile: ResolvableProfile) {
152        self.entity_data.lock().profile.set(profile);
153    }
154
155    /// Returns whether this mannequin refuses to be moved.
156    ///
157    /// Vanilla parity: `Mannequin.getImmovable`.
158    #[must_use]
159    pub fn is_immovable(&self) -> bool {
160        self.immovable.load(Ordering::Relaxed)
161    }
162
163    /// Sets whether this mannequin refuses to be moved.
164    pub fn set_immovable(&self, immovable: bool) {
165        self.immovable.store(immovable, Ordering::Relaxed);
166        self.entity_data.lock().immovable.set(immovable);
167    }

169    /// Returns the bit set of model parts this mannequin shows.
170    #[must_use]
171    pub fn shown_model_parts(&self) -> i8 {
172        *self
173            .entity_data
174            .lock()
175            .avatar()
176            .player_mode_customisation
177            .get()
178    }
179
180    /// Sets the bit set of model parts this mannequin shows.
181    pub fn set_shown_model_parts(&self, parts: i8) {
182        self.entity_data
183            .lock()
184            .avatar_mut()
185            .player_mode_customisation
186            .set(parts);
187    }
188
189    /// Returns which hand this mannequin holds items in.
190    #[must_use]
191    pub fn main_arm(&self) -> HumanoidArm {
192        *self.entity_data.lock().avatar().player_main_hand.get()
193    }
194
195    /// Sets which hand this mannequin holds items in.
196    pub fn set_main_arm(&self, arm: HumanoidArm) {
197        self.entity_data
198            .lock()
199            .avatar_mut()
200            .player_main_hand
201            .set(arm);
202    }
203
204    /// Returns the label shown above this mannequin, if it is shown at all.
205    ///
206    /// Vanilla parity: `Mannequin.getDescription`, which reads the synced
207    /// optional rather than the stored one -- a hidden description reads as
208    /// nothing even though the text is still remembered.
209    #[must_use]
210    pub fn description(&self) -> Option<TextComponent> {
211        self.entity_data
212            .lock()
213            .description
214            .get()
215            .as_ref()
216            .map(|description| (**description).clone())
217    }
218
219    /// Sets the label text, without changing whether it is shown.
220    pub fn set_description(&self, description: TextComponent) {
221        self.state.lock().description = description;
222        self.update_description();
223    }
224
225    /// Sets whether the label is shown, without forgetting its text.
226    pub fn set_hide_description(&self, hide: bool) {
227        self.state.lock().hide_description = hide;
228        self.update_description();
229    }
230
231    /// Vanilla parity: `Mannequin.updateDescription`.
232    fn update_description(&self) {
233        let state = self.state.lock();
234        let effective = if state.hide_description {
235            None
236        } else {
237            Some(Box::new(state.description.clone()))
238        };
239        drop(state);
240        self.entity_data.lock().description.set(effective);
241    }
242
243    /// Returns the pose this mannequin is holding.
244    #[must_use]
245    pub fn pose(&self) -> EntityPose {
246        *self.entity_data.lock().base().pose.get()
247    }
248
249    /// Puts this mannequin into `pose`, if it is one a mannequin may hold.
250    ///
251    /// Vanilla parity: `Mannequin.POSE_CODEC`, which rejects anything outside
252    /// [`VALID_POSES`] rather than letting a map put a mannequin into, say, a
253    /// warden's digging pose.
254    pub fn set_pose_checked(&self, pose: EntityPose) {
255        let pose = if VALID_POSES.contains(&pose) {
256            pose
257        } else {
258            EntityPose::Standing
259        };
260        self.entity_data.lock().base_mut().pose.set(pose);
261    }
262}

264/// Returns the label a mannequin shows unless one is set.
265///
266/// Vanilla parity: `Mannequin.DEFAULT_DESCRIPTION`.
267fn default_description() -> TextComponent {
268    TextComponent::translated(translations::ENTITY_MINECRAFT_MANNEQUIN_LABEL.msg())
269}
270
271/// Returns the model-part mask the `hidden_layers` list describes.
272///
273/// Vanilla parity: `Mannequin.LAYERS_CODEC`, which stores the *hidden* parts
274/// and reconstructs the shown mask by clearing each one from `ALL_LAYERS`.
275fn shown_parts_from_hidden(names: &[String]) -> i8 {
276    let mut shown = ALL_LAYERS;
277    for name in names {
278        if let Some((_, mask)) = MODEL_PARTS.iter().find(|(part, _)| part == name) {
279            shown &= !mask;
280        }
281    }
282    shown
283}
284
285/// Returns the `hidden_layers` list for a shown-part mask.
286fn hidden_parts_from_shown(shown: i8) -> Vec<String> {
287    MODEL_PARTS
288        .iter()
289        .filter(|(_, mask)| shown & mask == 0)
290        .map(|(part, _)| (*part).to_owned())
291        .collect()
292}
293
294/// Returns the serialized name of a pose.
295const fn pose_name(pose: EntityPose) -> &'static str {
296    match pose {
297        EntityPose::Sneaking => "crouching",
298        EntityPose::Swimming => "swimming",
299        EntityPose::FallFlying => "fall_flying",
300        EntityPose::Sleeping => "sleeping",
301        _ => "standing",
302    }
303}
304
305/// Returns the pose a serialized name describes, if a mannequin may hold it.
306fn pose_from_name(name: &str) -> EntityPose {
307    match name {
308        "crouching" => EntityPose::Sneaking,
309        "swimming" => EntityPose::Swimming,
310        "fall_flying" => EntityPose::FallFlying,
311        "sleeping" => EntityPose::Sleeping,
312        _ => EntityPose::Standing,
313    }
314}

316impl Entity for MannequinEntity {
317    fn base(&self) -> &EntityBase {
318        &self.base
319    }
320
321    fn entity_type(&self) -> EntityTypeRef {
322        self.entity_type
323    }
324
325    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
326        Some(&self.entity_data)
327    }
328
329    /// Vanilla parity: `Mannequin.isEffectiveAi`, which an immovable mannequin
330    /// refuses -- that is what stops knockback and pushing moving it.
331    fn is_effective_ai(&self) -> bool {
332        !self.is_immovable()
333    }
334
335    fn save_additional(&self, nbt: &mut NbtCompound) {
336        nbt.insert("profile", self.profile().to_nbt_tag());
337
338        let hidden = hidden_parts_from_shown(self.shown_model_parts());
339        nbt.insert(
340            "hidden_layers",
341            NbtTag::List(NbtList::String(
342                hidden.into_iter().map(Into::into).collect(),
343            )),
344        );
345
346        nbt.insert(
347            "main_hand",
348            match self.main_arm() {
349                HumanoidArm::Left => "left",
350                HumanoidArm::Right => "right",
351            },
352        );
353        nbt.insert("pose", pose_name(self.pose()));
354        nbt.insert("immovable", self.is_immovable());
355
356        // Vanilla parity: the default label is not written, and a hidden one is
357        // recorded as a flag rather than as an absent description -- the two
358        // cases have to stay distinguishable across a save.
359        let state = self.state.lock();
360        if state.hide_description {
361            nbt.insert("hide_description", true);
362        } else if state.description != default_description() {
363            nbt.insert("description", state.description.to_codec_nbt());
364        }
365    }
366
367    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
368        if let Some(profile) = nbt.get("profile").and_then(ResolvableProfile::from_nbt_tag) {
369            self.set_profile(profile);
370        }
371
372        let hidden: Vec<String> = nbt.list("hidden_layers").map_or_else(Vec::new, |list| {
373            list.strings()
374                .map(|values| values.iter().map(ToString::to_string).collect())
375                .unwrap_or_default()
376        });
377        self.set_shown_model_parts(shown_parts_from_hidden(&hidden));
378
379        let arm = match nbt.string("main_hand").map(ToString::to_string).as_deref() {
380            Some("left") => HumanoidArm::Left,
381            _ => HumanoidArm::Right,
382        };
383        self.set_main_arm(arm);
384
385        let pose = nbt.string("pose").map_or(EntityPose::Standing, |value| {
386            pose_from_name(&value.to_str())
387        });
388        self.set_pose_checked(pose);
389
390        self.set_immovable(nbt.byte("immovable").is_some_and(|value| value != 0));
391        self.set_hide_description(nbt.byte("hide_description").is_some_and(|value| value != 0));
392        let description = nbt
393            .get("description")
394            .map(|tag| tag.to_owned())
395            .as_ref()
396            .and_then(TextComponent::from_nbt)
397            .unwrap_or_else(default_description);
398        self.set_description(description);
399    }
400}

402impl LivingEntity for MannequinEntity {
403    /// Returns synchronized data declared by vanilla `LivingEntity`.
404    fn living_synced_data(&self) -> Option<&dyn LivingEntitySyncedData> {
405        Some(&self.entity_data)
406    }
407
408    fn living_base(&self) -> &LivingEntityBase {
409        &self.living_base
410    }
411
412    fn get_health(&self) -> f32 {
413        *self.entity_data.lock().living_entity().health.get()
414    }
415
416    fn set_health(&self, health: f32) {
417        let clamped = health.clamp(0.0, self.get_max_health());
418        self.entity_data
419            .lock()
420            .living_entity_mut()
421            .health
422            .set(clamped);
423    }
424
425    /// Vanilla parity: `Mannequin.isImmobile`, which is what nails an immovable
426    /// mannequin down.
427    fn is_immobile(&self) -> bool {
428        self.is_immovable() || self.get_health() <= 0.0
429    }

444    fn mannequin() -> MannequinEntity {
445        init_vanilla_registry();
446        MannequinEntity::new(
447            &vanilla_entities::MANNEQUIN,
448            next_entity_id(),
449            DVec3::new(8.5, 64.0, 8.5),

454    fn reload(nbt: &NbtCompound) -> MannequinEntity {
455        let mut bytes = Vec::new();
456        nbt.write(&mut bytes);
457        let borrowed = read_borrowed_compound(&mut Cursor::new(&bytes))
458            .unwrap_or_else(|error| panic!("test nbt should reborrow: {error}"));
459        let entity = mannequin();
460        entity.load_additional((&borrowed).into());
461        entity
462    }

464    /// The saved list holds the *hidden* layers while the synced byte holds the
465    /// shown ones, so the two are complements. Getting that backwards would
466    /// make every mannequin load inside-out.
467    #[test]
468    fn hidden_layers_are_the_complement_of_the_shown_mask() {
469        assert_eq!(shown_parts_from_hidden(&[]), ALL_LAYERS);
470        assert!(hidden_parts_from_shown(ALL_LAYERS).is_empty());
471
472        let hat_hidden = shown_parts_from_hidden(&["hat".to_owned()]);
473        assert_eq!(hat_hidden, ALL_LAYERS & !(1 << 6));
474        assert_eq!(hidden_parts_from_shown(hat_hidden), vec!["hat".to_owned()]);
475    }

477    /// A mannequin may only hold five of the eighteen poses; anything else
478    /// falls back to standing rather than putting it into a warden's crouch.
479    #[test]
480    fn an_invalid_pose_falls_back_to_standing() {
481        let entity = mannequin();
482        entity.set_pose_checked(EntityPose::Swimming);
483        assert_eq!(entity.pose(), EntityPose::Swimming);
484
485        entity.set_pose_checked(EntityPose::Digging);
486        assert_eq!(entity.pose(), EntityPose::Standing);
487    }

489    /// Hiding the label must not lose it: vanilla keeps the text and syncs an
490    /// absent description, so unhiding brings the same text back.
491    #[test]
492    fn hiding_the_label_keeps_its_text() {
493        let entity = mannequin();
494        entity.set_description(TextComponent::plain("Shopkeeper"));
495        assert_eq!(
496            entity.description(),
497            Some(TextComponent::plain("Shopkeeper"))
498        );

501        assert_eq!(entity.description(), None);
502
503        entity.set_hide_description(false);
504        assert_eq!(
505            entity.description(),
506            Some(TextComponent::plain("Shopkeeper"))

508    }

510    /// Everything a map-maker sets has to come back after a chunk reload.
511    #[test]
512    fn a_posed_immovable_mannequin_round_trips() {
513        let entity = mannequin();
514        entity.set_pose_checked(EntityPose::Sleeping);
515        entity.set_immovable(true);
516        entity.set_main_arm(HumanoidArm::Left);
517        entity.set_shown_model_parts(shown_parts_from_hidden(&["cape".to_owned()]));
518        entity.set_description(TextComponent::plain("Guard"));
519
520        let mut nbt = NbtCompound::new();
521        entity.save_additional(&mut nbt);
522        assert_eq!(nbt.byte("immovable"), Some(1));
523        assert_eq!(
524            nbt.string("pose").map(ToString::to_string),
525            Some("sleeping".to_owned())
526        );
527        assert_eq!(
528            nbt.string("main_hand").map(ToString::to_string),
529            Some("left".to_owned())
530        );
531
532        let reloaded = reload(&nbt);
533        assert_eq!(reloaded.pose(), EntityPose::Sleeping);
534        assert!(reloaded.is_immovable());
535        assert_eq!(reloaded.main_arm(), HumanoidArm::Left);
536        assert_eq!(
537            reloaded.shown_model_parts(),
538            shown_parts_from_hidden(&["cape".to_owned()])
539        );
540        assert_eq!(reloaded.description(), Some(TextComponent::plain("Guard")));
541    }

543    /// A hidden label survives as a flag, not as a missing description -- the
544    /// two have to stay distinguishable or unhiding would show the default.
545    #[test]
546    fn a_hidden_label_stays_hidden_across_a_save() {
547        let entity = mannequin();
548        entity.set_description(TextComponent::plain("Guard"));
549        entity.set_hide_description(true);
550
551        let mut nbt = NbtCompound::new();
552        entity.save_additional(&mut nbt);
553        assert_eq!(nbt.byte("hide_description"), Some(1));
554        assert!(nbt.get("description").is_none());
555
556        assert_eq!(reload(&nbt).description(), None);
557    }

559    /// An immovable mannequin is nailed down: vanilla expresses that as
