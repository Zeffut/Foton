40/// An item frame.
41///
42/// Vanilla parity: `ItemFrame`. A frame holds one item, turns it eight ways,
43/// and reports the rotation to a comparator. Broken frames drop their contents
44/// according to Vanilla's fixed, creative and drop-chance rules; support is
45/// checked periodically by the entity tick.
46#[entity_behavior(class = "ItemFrame")]
47pub struct ItemFrameEntity {
48    base: EntityBase,
49    entity_type: EntityTypeRef,
50    entity_data: SyncMutex<ItemFrameEntityData>,
51    block_pos: SyncMutex<BlockPos>,
52    state: SyncMutex<FrameState>,
53}
54
55/// The two saved fields a frame keeps outside its synced data.
56///
57/// Vanilla parity: `ItemFrame.fixed` and `ItemFrame.dropChance`.
58pub(super) struct FrameState {
59    pub(super) fixed: bool,
60    pub(super) drop_chance: f32,
61}
62
63impl Default for FrameState {
64    fn default() -> Self {
65        Self {
66            fixed: false,
67            drop_chance: 1.0,
68        }
69    }

72/// The half of `ItemFrame` that the glow frame inherits in vanilla.
73///
74/// Foton's two frames are separate structs rather than a subclass pair, so what
75/// vanilla gets from `extends ItemFrame` lives here: the fixed flag, the drop
76/// chance, and the damage rules built on them. An implementation supplies the
77/// five accessors its own storage decides.
78pub(super) trait FrameLike: BlockAttached {
79    fn frame_direction(&self) -> Direction;
80    fn frame_box(&self) -> WorldAabb;
81    /// The saved fields outside the synced data.
82    fn frame_state(&self) -> &SyncMutex<FrameState>;
83
84    /// Vanilla parity: `ItemFrame.getItem`.
85    fn framed_item(&self) -> ItemStack;
86
87    /// Vanilla parity: the `setItem(ItemStack.EMPTY)` of `ItemFrame.dropItem`.
88    fn clear_framed_item(&self);
89
90    /// Vanilla parity: `ItemFrame.getFrameItemStack`, which the glow frame
91    /// overrides with the glowing one.
92    fn frame_item(&self) -> ItemRef;
93
94    /// Plays one of the frame's own sounds at the frame.
95    fn play_frame_sound(&self, sound: SoundEventRef);
96
97    /// Returns whether this frame refuses to be emptied or broken.
98    ///
99    /// Vanilla parity: the `fixed` field, set by structures so a frame that is
100    /// part of the scenery survives a wandering skeleton.
101    fn is_fixed(&self) -> bool {
102        self.frame_state().lock().fixed
103    }
104
105    fn survives_frame(&self, world: &Arc<World>) -> bool {
106        let pop_box = self.frame_box();
107        if has_block_collision(&WorldCollisionProvider::new(world), pop_box) {
108            return false;
109        }
110        let support = pop_box
111            .translate(self.frame_direction().offset_vec().as_dvec3() * -0.5)
112            .deflate(1.0e-7);
113        let min = BlockPos::new(
114            support.min_x().floor() as i32,
115            support.min_y().floor() as i32,
116            support.min_z().floor() as i32,
117        );
118        let max = BlockPos::new(
119            support.max_x().floor() as i32,
120            support.max_y().floor() as i32,
121            support.max_z().floor() as i32,
122        );
123        BlockPos::between_closed(min, max).all(|pos| {
124            let state = world.get_block_state(pos);
125            state.is_solid()
126                || state.get_block() == &vanilla_blocks::REPEATER
127                || state.get_block() == &vanilla_blocks::COMPARATOR
128        })
129    }
130
131    /// Returns how often the framed item survives the frame breaking.
132    ///
133    /// Vanilla parity: the `dropChance` field, one by default.
134    fn drop_chance(&self) -> f32 {
135        self.frame_state().lock().drop_chance
136    }
137
138    /// Drops what a broken or emptied frame leaves behind.
139    ///
140    /// Vanilla parity: the private `ItemFrame.dropItem(level, causedBy, withFrame)`.
141    /// A fixed frame drops nothing and keeps its item; a creative player gets
142    /// nothing back; and the framed item rolls against its own drop chance,
143    /// which is how a map in a structure frame stays put.
144    ///
145    /// Not implemented: `removeFramedMap`, the map-tracking bookkeeping. Foton
146    /// tracks no frames on a map, so there is nothing to remove.
147    fn drop_frame_contents(
148        &self,
149        world: &World,
150        caused_by: Option<&SharedEntity>,
151        with_frame: bool,
152    ) {
153        if self.is_fixed() {
154            return;
155        }
156
157        let item = self.framed_item();
158        self.clear_framed_item();
159
160        if !world.get_game_rule(&ENTITY_DROPS) || drop_would_be_wasted(caused_by) {
161            return;
162        }
163        if with_frame {
164            self.spawn_at_location(ItemStack::new(self.frame_item()), 0.0);
165        }
166        if !item.is_empty() && rand::random::<f32>() < self.drop_chance() {
167            self.spawn_at_location(item, 0.0);
168        }
169    }
170
171    /// Vanilla parity: `ItemFrame.dropItem(level, causedBy)`, the public one,
172    /// which is the break rather than the emptying.
173    fn drop_broken_frame(&self, world: &World, caused_by: Option<&SharedEntity>) {
174        self.play_frame_sound(&sound_events::ENTITY_ITEM_FRAME_BREAK);
175        self.drop_frame_contents(world, caused_by, true);
176        self.frame_block_change(caused_by);
177    }
178
179    /// Vanilla parity: `ItemFrame.hurtServer`.
180    ///
181    /// A punch on a frame holding something empties it and leaves the frame on
182    /// the wall; a punch on an empty one takes the frame down. An explosion
183    /// skips the emptying and always breaks it.
184    fn hurt_item_frame(&self, world: &World, source: &DamageSource) -> bool {
185        if self.is_fixed() {
186            return can_hurt_when_fixed(world, source) && self.hurt_block_attached(world, source);
187        }
188        if self.is_invulnerable_to_base(source) {
189            return false;
190        }
191        if source.is(&DamageTypeTag::IS_EXPLOSION) || self.framed_item().is_empty() {
192            return self.hurt_block_attached(world, source);
193        }
194
195        let caused_by = caused_by_entity(world, source);
196        self.drop_frame_contents(world, caused_by.as_ref(), false);
197        self.frame_block_change(caused_by.as_ref());
198        self.play_frame_sound(&sound_events::ENTITY_ITEM_FRAME_REMOVE_ITEM);
199        true
200    }
201
202    /// Vanilla parity: the `gameEvent(BLOCK_CHANGE, causedBy)` both drop paths
203    /// end with.
204    fn frame_block_change(&self, caused_by: Option<&SharedEntity>) {
205        let Some(world) = self.level() else {
206            return;
207        };
208        world.game_event_at(
209            &vanilla_game_events::BLOCK_CHANGE,
210            self.position(),
211            &GameEventContext::new(caused_by.map(AsRef::as_ref), None),
212        );
213    }
214}
215
216/// Vanilla parity: `ItemFrame.canHurtWhenFixed`.
217fn can_hurt_when_fixed(world: &World, source: &DamageSource) -> bool {
218    if source.bypasses_invulnerability() {
219        return true;
220    }
221    caused_by_entity(world, source).is_some_and(|entity| {
222        entity
223            .as_player()
224            .is_some_and(|player| player.game_mode() == GameType::Creative)
225    })
226}

240    /// Creates a fresh item frame from the generic entity factory path.
241    #[must_use]
242    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
243        Self::new_attached(
244            entity_type,
245            id,
246            BlockPos::new(
247                position.x.floor() as i32,
248                position.y.floor() as i32,
249                position.z.floor() as i32,
250            ),
251            Direction::South,
252            world,

256    /// Creates a fresh item frame attached to `block_pos`.
257    #[must_use]
258    pub fn new_attached(
259        entity_type: EntityTypeRef,
260        id: i32,
261        block_pos: BlockPos,
262        direction: Direction,
263        world: Weak<World>,
264    ) -> Self {
265        let entity = Self {
266            base: EntityBase::new_with_state(
267                id,
268                EntityBaseState::new_with_bounding_box(
269                    Self::frame_center(block_pos, direction),
270                    entity_type.dimensions,
271                    Self::frame_bounding_box(block_pos, direction, false),
272                )
273                .with_rotation(Self::rotation_for_direction(direction)),
274                world,
275            ),
276            entity_type,
277            entity_data: SyncMutex::new(ItemFrameEntityData::new()),
278            block_pos: SyncMutex::new(block_pos),
279            state: SyncMutex::new(FrameState::default()),
280        };
281        entity
282            .entity_data
283            .lock()
284            .hanging_entity
285            .direction
286            .set(direction);
287        entity
288    }

290    /// Creates an item frame from persistent entity data.

293        let position = load.position;

300                position.y.floor() as i32,
301                position.z.floor() as i32,
302            )),
303            state: SyncMutex::new(FrameState::default()),
304        }

307    /// Returns the item currently displayed in the frame.
308    pub fn framed_item(&self) -> ItemStack {
309        self.entity_data.lock().item.get().clone()
310    }
311
312    /// Sets the framed item, matching vanilla by storing a single item.
313    pub fn set_item(&self, item: ItemStack) {
314        self.set_item_with_update(item, true);
315    }
316
317    /// Sets the framed item and optionally notifies nearby comparators.
318    pub(crate) fn set_item_with_update(&self, mut item: ItemStack, update_comparators: bool) {
319        if !item.is_empty() {
320            item.set_count(1);
321        }
322        self.entity_data.lock().item.set(item);
323        self.recalculate_position();
324        if update_comparators && let Some(world) = self.level() {
325            world.update_neighbor_for_output_signal(*self.block_pos.lock(), &vanilla_blocks::AIR);
326        }
327    }

329    /// Returns whether the frame still has valid support and no collision.
330    #[must_use]
331    pub fn survives(&self) -> bool {
332        let Some(world) = self.level() else {
333            return false;
334        };
335        self.survives_frame(&world)
336    }

338    /// Turns the frame to face `direction`, updating both the synchronized
339    /// facing and the bounding box that follows it.
340    pub fn set_direction(&self, direction: Direction) {
341        self.entity_data
342            .lock()
343            .hanging_entity
344            .direction
345            .set(direction);
346        self.base
347            .set_rotation(Self::rotation_for_direction(direction));
348        self.recalculate_position();
349    }

351    /// Plays one of the frame's own sounds at the frame.
352    fn play_sound_at_frame(&self, sound: SoundEventRef) {
353        let Some(world) = self.level() else {
354            return;
355        };
356        world.play_sound_at(sound, SoundSource::Neutral, self.position(), 1.0, 1.0, None);
357    }

359    /// Tells the world the frame changed.
360    ///
361    /// Vanilla parity: the `gameEvent(BLOCK_CHANGE, player)` of
362    /// `ItemFrame.interact`, plus the comparator update -- a frame's signal is
363    /// its rotation, so turning the item is a redstone change.
364    fn frame_changed(&self, player: &Player) {
365        let Some(world) = self.level() else {
366            return;
367        };
368        world.game_event_at(
369            &vanilla_game_events::BLOCK_CHANGE,
370            self.position(),
371            &GameEventContext::new(Some(player as &dyn Entity), None),
372        );
373        world.update_neighbor_for_output_signal(*self.block_pos.lock(), &vanilla_blocks::AIR);
374    }
375
376    fn recalculate_position(&self) {
377        let block_pos = *self.block_pos.lock();
378        let direction = *self.entity_data.lock().hanging_entity.direction.get();
379        let position = Self::frame_center(block_pos, direction);
380        if let Err(error) = self.base.try_set_position(position) {
381            panic!(
382                "failed to commit item frame {} position recalculation: {error}",
383                self.base.id()
384            );
385        }
386        self.base.set_bounding_box(Self::frame_bounding_box(
387            block_pos,
388            direction,
389            self.has_framed_map(),
390        ));
391    }
392
393    fn has_framed_map(&self) -> bool {
394        self.entity_data.lock().item.get().has(MAP_ID)
395    }
396
397    /// Returns where a frame attached to `block_pos` sits.
398    ///
399    /// Shared with [`super::GlowItemFrameEntity`], which is the same geometry.
400    pub(super) fn frame_center(block_pos: BlockPos, direction: Direction) -> DVec3 {
401        let off = direction.offset_vec().as_dvec3() * 0.46875;
402        block_pos.0.as_dvec3() + DVec3::splat(0.5) - off
403    }
404
405    /// Returns the rotation a frame facing `direction` is drawn at.
406    pub(super) fn rotation_for_direction(direction: Direction) -> (f32, f32) {
407        if direction.is_horizontal() {
408            (f32::from(direction_2d_data_value(direction)) * 90.0, 0.0)
409        } else {
410            let pitch = match direction {
411                Direction::Up => -90.0,
412                Direction::Down => 90.0,
413                Direction::North | Direction::South | Direction::West | Direction::East => 0.0,
414            };
415            (0.0, pitch)
416        }
417    }
418
419    /// Returns the hitbox of a frame attached to `block_pos`.
420    pub(super) fn frame_bounding_box(
421        block_pos: BlockPos,
422        direction: Direction,
423        has_framed_map: bool,
424    ) -> WorldAabb {
425        let center = Self::frame_center(block_pos, direction);
426        let size = if has_framed_map { 1.0 } else { 0.75 };
427        let x_size = if direction.axis() == Axis::X {
428            0.0625
429        } else {
430            size
431        };
432        let y_size = if direction.axis() == Axis::Y {
433            0.0625
434        } else {
435            size
436        };
437        let z_size = if direction.axis() == Axis::Z {
438            0.0625
439        } else {
440            size
441        };
442        WorldAabb::new(
443            center.x - x_size / 2.0,
444            center.y - y_size / 2.0,
445            center.z - z_size / 2.0,
446            center.x + x_size / 2.0,
447            center.y + y_size / 2.0,
448            center.z + z_size / 2.0,
449        )
450    }
451}

459        let entity_data = self.entity_data.lock();

468impl Entity for ItemFrameEntity {
469    fn base(&self) -> &EntityBase {
470        &self.base
471    }
472
473    fn entity_type(&self) -> EntityTypeRef {
474        self.entity_type
475    }
476
477    /// Vanilla parity: `ItemFrame.getSlot`, whose one slot is what is hung in
478    /// the frame.
479    fn slot_item(&self, slot: i32) -> Option<ItemStack> {
480        if slot == CONTENTS_SLOT {
481            return Some(self.framed_item());
482        }
483        self.entity_slot_item(slot)
484    }
485
486    fn spawn_data(&self) -> i32 {
487        direction_3d_data_value(*self.entity_data.lock().hanging_entity.direction.get())
488    }
489
490    /// Vanilla parity: `BlockAttachedEntity.thunderHit`, an empty override.
491    /// Lightning passes straight through anything hung on a block.
492    fn thunder_hit(&self, _world: &World, _bolt: &dyn Entity) {}
493
494    fn hurt(&self, world: &World, source: &DamageSource, _amount: f32) -> bool {
495        self.hurt_item_frame(world, source)
496    }
497
498    fn tick(&self) {
499        let Some(world) = self.level() else {
500            return;
501        };
502        if self.tick_count() % 100 != 0 || self.is_removed() || self.survives_frame(&world) {
503            return;
504        }
505        let mut event = HangingBreakEvent::new(self.uuid(), "PHYSICS");
506        world.fire_event(&mut event);
507        if !event.is_cancelled() {
508            self.set_removed(Discarded);
509            self.drop_item(&world, None);
510        }
511    }
512
513    fn spawn_position(&self) -> DVec3 {
514        let block_pos = *self.block_pos.lock();
515        DVec3::new(
516            f64::from(block_pos.x()),
517            f64::from(block_pos.y()),
518            f64::from(block_pos.z()),
519        )
520    }
521
522    fn is_pickable(&self) -> bool {
523        true
524    }
525
526    /// Fills the frame, or turns what is already in it.
527    ///
528    /// Vanilla parity: `ItemFrame.interact`. The same click does two different
529    /// things depending on whether the frame is empty, which is what lets a
530    /// player set a rotation without a second gesture -- and what a comparator
531    /// beside the frame is reading.
532    fn interact(
533        &self,
534        player: &Player,
535        hand: InteractionHand,
536        _location: DVec3,
537    ) -> InteractionResult {
538        let frame_is_empty = self.entity_data.lock().item.get().is_empty();
539
540        if frame_is_empty {
541            let held = {
542                let inventory = player.inventory.lock();
543                let stack = inventory.get_item_in_hand(hand);
544                stack.copy_with_count(1)
545            };
546            if held.is_empty() || self.is_removed() {
547                return InteractionResult::Pass;
548            }
549
550            self.set_item(held);
551            self.play_sound_at_frame(&sound_events::ENTITY_ITEM_FRAME_ADD_ITEM);
552            self.frame_changed(player);
553
554            if !player.has_infinite_materials() {
555                let mut inventory = player.inventory.lock();
556                inventory.get_item_in_hand_mut(hand).shrink(1);
557            }
558            return InteractionResult::Success;
559        }
560
561        {
562            let mut entity_data = self.entity_data.lock();
563            let next = (*entity_data.rotation.get() + 1).rem_euclid(ROTATION_STEPS);
564            entity_data.rotation.set(next);
565        }
566        self.play_sound_at_frame(&sound_events::ENTITY_ITEM_FRAME_ROTATE_ITEM);
567        self.frame_changed(player);
568        InteractionResult::Success
569    }
570
571    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
572        Some(&self.entity_data)
573    }
574
575    fn save_additional(&self, nbt: &mut NbtCompound) {
576        let block_pos = *self.block_pos.lock();
577        nbt.insert(
578            "block_pos",
579            NbtTag::IntArray(vec![block_pos.x(), block_pos.y(), block_pos.z()]),
580        );
581
582        let entity_data = self.entity_data.lock();
583        let item = entity_data.item.get();
584        if !item.is_empty() {
585            nbt.insert("Item", item.to_nbt_tag_ref());
586        }
587        nbt.insert("ItemRotation", *entity_data.rotation.get() as i8);
588        nbt.insert("ItemDropChance", self.drop_chance());
589        nbt.insert(
590            "Facing",
591            direction_3d_data_value(*entity_data.hanging_entity.direction.get()) as i8,
592        );
593        nbt.insert("Invisible", 0_i8);
594        nbt.insert("Fixed", i8::from(self.is_fixed()));
595    }
596
597    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
598        if let Some(block_pos) = nbt.int_array("block_pos")
599            && block_pos.len() == 3
600        {
601            *self.block_pos.lock() = BlockPos::new(block_pos[0], block_pos[1], block_pos[2]);
602        }
603
604        if let Some(item_tag) = nbt.compound("Item")
605            && let Some(item) = ItemStack::from_borrowed_compound(&item_tag)
606        {
607            self.set_item_with_update(item, false);
608        }
609
610        {
611            let mut state = self.state.lock();
612            if let Some(drop_chance) = nbt.float("ItemDropChance") {
613                state.drop_chance = drop_chance;
614            }
615            if let Some(fixed) = nbt.byte("Fixed") {
616                state.fixed = fixed != 0;
617            }
618        }
619
620        if let Some(item_rotation) = nbt.byte("ItemRotation") {
621            self.entity_data
622                .lock()
623                .rotation
624                .set(i32::from(item_rotation).rem_euclid(8));
625        }
626
627        let facing = nbt
628            .byte("Facing")
629            .and_then(|value| direction_from_3d_data_value(i32::from(value)))
630            .or_else(|| nbt.int("Facing").and_then(direction_from_3d_data_value));
631        if let Some(direction) = facing {
632            self.set_direction(direction);
633        }
634
635        self.recalculate_position();
636    }
637}
