40/// An item frame.
41///
42/// Vanilla parity: `ItemFrame`. A frame holds one item, turns it eight ways,
43/// and reports the rotation to a comparator. Broken frames drop their contents
44/// according to Vanilla's fixed, creative and drop-chance rules; support is
45/// checked periodically by the entity tick.
46#[entity_behavior(class = "ItemFrame")]
47pub struct ItemFrameEntity {
48    base: EntityBase,
49    entity_type: EntityTypeRef,
50    entity_data: SyncMutex<ItemFrameEntityData>,
51    block_pos: SyncMutex<BlockPos>,
52    state: SyncMutex<FrameState>,
53}

56///
57/// Vanilla parity: `ItemFrame.fixed` and `ItemFrame.dropChance`.
58pub(super) struct FrameState {
59    pub(super) fixed: bool,
60    pub(super) drop_chance: f32,
61}

72/// The half of `ItemFrame` that the glow frame inherits in vanilla.
73///
74/// Foton's two frames are separate structs rather than a subclass pair, so what
75/// vanilla gets from `extends ItemFrame` lives here: the fixed flag, the drop
76/// chance, and the damage rules built on them. An implementation supplies the
77/// five accessors its own storage decides.
78pub(super) trait FrameLike: BlockAttached {
79    fn frame_direction(&self) -> Direction;
80    fn frame_box(&self) -> WorldAabb;
81    /// The saved fields outside the synced data.
82    fn frame_state(&self) -> &SyncMutex<FrameState>;

84    /// Vanilla parity: `ItemFrame.getItem`.
85    fn framed_item(&self) -> ItemStack;
86
87    /// Vanilla parity: the `setItem(ItemStack.EMPTY)` of `ItemFrame.dropItem`.
88    fn clear_framed_item(&self);
89
90    /// Vanilla parity: `ItemFrame.getFrameItemStack`, which the glow frame
91    /// overrides with the glowing one.
92    fn frame_item(&self) -> ItemRef;
93
94    /// Plays one of the frame's own sounds at the frame.
95    fn play_frame_sound(&self, sound: SoundEventRef);
96
97    /// Returns whether this frame refuses to be emptied or broken.
98    ///
99    /// Vanilla parity: the `fixed` field, set by structures so a frame that is
100    /// part of the scenery survives a wandering skeleton.
101    fn is_fixed(&self) -> bool {
102        self.frame_state().lock().fixed
103    }
104
105    fn survives_frame(&self, world: &Arc<World>) -> bool {
106        let pop_box = self.frame_box();
107        if has_block_collision(&WorldCollisionProvider::new(world), pop_box) {
108            return false;
109        }
110        let support = pop_box
111            .translate(self.frame_direction().offset_vec().as_dvec3() * -0.5)
112            .deflate(1.0e-7);
113        let min = BlockPos::new(
114            support.min_x().floor() as i32,
115            support.min_y().floor() as i32,
116            support.min_z().floor() as i32,
117        );
118        let max = BlockPos::new(
119            support.max_x().floor() as i32,
120            support.max_y().floor() as i32,
121            support.max_z().floor() as i32,
122        );
123        BlockPos::between_closed(min, max).all(|pos| {
124            let state = world.get_block_state(pos);
125            state.is_solid()
126                || state.get_block() == &vanilla_blocks::REPEATER
127                || state.get_block() == &vanilla_blocks::COMPARATOR
128        })
129    }
130
131    /// Returns how often the framed item survives the frame breaking.
132    ///
133    /// Vanilla parity: the `dropChance` field, one by default.
134    fn drop_chance(&self) -> f32 {
135        self.frame_state().lock().drop_chance
136    }
137
138    /// Drops what a broken or emptied frame leaves behind.
139    ///
140    /// Vanilla parity: the private `ItemFrame.dropItem(level, causedBy, withFrame)`.
141    /// A fixed frame drops nothing and keeps its item; a creative player gets
142    /// nothing back; and the framed item rolls against its own drop chance,
143    /// which is how a map in a structure frame stays put.
144    ///
145    /// Not implemented: `removeFramedMap`, the map-tracking bookkeeping. Foton
146    /// tracks no frames on a map, so there is nothing to remove.
147    fn drop_frame_contents(
148        &self,
149        world: &World,
150        caused_by: Option<&SharedEntity>,
151        with_frame: bool,
152    ) {
153        if self.is_fixed() {
154            return;
155        }
156
157        let item = self.framed_item();
158        self.clear_framed_item();
159
160        if !world.get_game_rule(&ENTITY_DROPS) || drop_would_be_wasted(caused_by) {
161            return;
162        }
163        if with_frame {
164            self.spawn_at_location(ItemStack::new(self.frame_item()), 0.0);
165        }
166        if !item.is_empty() && rand::random::<f32>() < self.drop_chance() {
167            self.spawn_at_location(item, 0.0);
168        }
169    }
170
171    /// Vanilla parity: `ItemFrame.dropItem(level, causedBy)`, the public one,
172    /// which is the break rather than the emptying.
173    fn drop_broken_frame(&self, world: &World, caused_by: Option<&SharedEntity>) {
174        self.play_frame_sound(&sound_events::ENTITY_ITEM_FRAME_BREAK);
175        self.drop_frame_contents(world, caused_by, true);
176        self.frame_block_change(caused_by);
177    }
178
179    /// Vanilla parity: `ItemFrame.hurtServer`.
180    ///
181    /// A punch on a frame holding something empties it and leaves the frame on
182    /// the wall; a punch on an empty one takes the frame down. An explosion
183    /// skips the emptying and always breaks it.
184    fn hurt_item_frame(&self, world: &World, source: &DamageSource, amount: f32) -> bool {
185        if self.is_fixed() {
186            return can_hurt_when_fixed(world, source) && self.hurt_block_attached(world, source);
187        }
188        if self.is_invulnerable_to_base(source) {
189            return false;
190        }
191        if source.is(&DamageTypeTag::IS_EXPLOSION) || self.framed_item().is_empty() {
192            return self.hurt_block_attached(world, source);
193        }
194        // Paper parity: plugins may keep the item in its frame; the hit
195        // still counts as landed.
196        if !EntityDamageEvent::allows_non_living_hurt(
197            self.as_entity_event_source(),
198            world,
199            source,
200            amount,
201            false,
202        ) || self.is_removed()
203        {
204            return true;
205        }
206
207        let caused_by = caused_by_entity(world, source);
208        self.drop_frame_contents(world, caused_by.as_ref(), false);
209        self.frame_block_change(caused_by.as_ref());
210        self.play_frame_sound(&sound_events::ENTITY_ITEM_FRAME_REMOVE_ITEM);
211        true
212    }
213
214    /// Vanilla parity: the `gameEvent(BLOCK_CHANGE, causedBy)` both drop paths
215    /// end with.
216    fn frame_block_change(&self, caused_by: Option<&SharedEntity>) {
217        let Some(world) = self.level() else {
218            return;
219        };
220        world.game_event_at(
221            &vanilla_game_events::BLOCK_CHANGE,
222            self.position(),
223            &GameEventContext::new(caused_by.map(AsRef::as_ref), None),
224        );
225    }
226}

228/// Vanilla parity: `ItemFrame.canHurtWhenFixed`.
229fn can_hurt_when_fixed(world: &World, source: &DamageSource) -> bool {
230    if source.bypasses_invulnerability() {
231        return true;
232    }
233    caused_by_entity(world, source).is_some_and(|entity| {
234        entity
235            .as_player()
236            .is_some_and(|player| player.game_mode() == GameType::Creative)
237    })
238}

343    /// Sets the framed item and optionally notifies nearby comparators.
344    pub(crate) fn set_item_with_update(&self, mut item: ItemStack, update_comparators: bool) {
345        if !item.is_empty() {

385    /// Tells the world the frame changed.
386    ///
387    /// Vanilla parity: the `gameEvent(BLOCK_CHANGE, player)` of
388    /// `ItemFrame.interact`, plus the comparator update -- a frame's signal is
389    /// its rotation, so turning the item is a redstone change.
390    fn frame_changed(&self, player: &Player) {
391        let Some(world) = self.level() else {
392            return;
393        };
394        world.game_event_at(
395            &vanilla_game_events::BLOCK_CHANGE,
396            self.position(),
397            &GameEventContext::new(Some(player as &dyn Entity), None),

399        world.update_neighbor_for_output_signal(*self.block_pos.lock(), &vanilla_blocks::AIR);

402    #[cfg_attr(
403        not(test),
404        expect(
405            clippy::panic,
406            reason = "the value was constructed, locked or registered by this same call a few lines up"
407        )
408    )]
409    fn recalculate_position(&self) {
410        let block_pos = *self.block_pos.lock();
411        let direction = *self.entity_data.lock().hanging_entity.direction.get();
412        let position = Self::frame_center(block_pos, direction);
413        if let Err(error) = self.base.try_set_position(position) {
414            panic!(
415                "failed to commit item frame {} position recalculation: {error}",
416                self.base.id()
417            );
418        }
419        self.base.set_bounding_box(Self::frame_bounding_box(
420            block_pos,
421            direction,
422            self.has_framed_map(),
423        ));
424    }

426    fn has_framed_map(&self) -> bool {

430    /// Returns where a frame attached to `block_pos` sits.
431    ///
432    /// Shared with [`super::GlowItemFrameEntity`], which is the same geometry.
433    pub(super) fn frame_center(block_pos: BlockPos, direction: Direction) -> DVec3 {
434        let off = direction.offset_vec().as_dvec3() * 0.46875;
435        block_pos.0.as_dvec3() + DVec3::splat(0.5) - off
436    }
437
438    /// Returns the rotation a frame facing `direction` is drawn at.
439    pub(super) fn rotation_for_direction(direction: Direction) -> (f32, f32) {
440        if direction.is_horizontal() {
441            (f32::from(direction_2d_data_value(direction)) * 90.0, 0.0)
442        } else {
443            let pitch = match direction {
444                Direction::Up => -90.0,
445                Direction::Down => 90.0,
446                Direction::North | Direction::South | Direction::West | Direction::East => 0.0,
447            };
448            (0.0, pitch)
449        }
450    }
451
452    /// Returns the hitbox of a frame attached to `block_pos`.
453    pub(super) fn frame_bounding_box(
454        block_pos: BlockPos,
455        direction: Direction,
456        has_framed_map: bool,
457    ) -> WorldAabb {
458        let center = Self::frame_center(block_pos, direction);
459        let size = if has_framed_map { 1.0 } else { 0.75 };
460        let x_size = if direction.axis() == Axis::X {
461            0.0625
462        } else {
463            size
464        };
465        let y_size = if direction.axis() == Axis::Y {
466            0.0625
467        } else {
468            size
469        };
470        let z_size = if direction.axis() == Axis::Z {
471            0.0625
472        } else {
473            size
474        };
475        WorldAabb::new(
476            center.x - x_size / 2.0,
477            center.y - y_size / 2.0,
478            center.z - z_size / 2.0,
479            center.x + x_size / 2.0,
480            center.y + y_size / 2.0,
481            center.z + z_size / 2.0,
482        )
483    }

540    fn tick(&self) {
541        let Some(world) = self.level() else {
542            return;
543        };
544        if self.tick_count() % 100 != 0 || self.is_removed() || self.survives_frame(&world) {
545            return;
546        }
547        let mut event = HangingBreakEvent::new(self.uuid(), "PHYSICS");
548        world.fire_event(&mut event);
549        if !event.is_cancelled() {
550            self.set_removed(Discarded);
551            self.drop_item(&world, None);
552        }
553    }

568    /// Fills the frame, or turns what is already in it.
569    ///
570    /// Vanilla parity: `ItemFrame.interact`. The same click does two different
571    /// things depending on whether the frame is empty, which is what lets a
572    /// player set a rotation without a second gesture -- and what a comparator
573    /// beside the frame is reading.
574    fn interact(
575        &self,
576        player: &Player,
577        hand: InteractionHand,
578        _location: DVec3,
579    ) -> InteractionResult {
580        let frame_is_empty = self.entity_data.lock().item.get().is_empty();
581
582        if frame_is_empty {
583            let held = {
584                let inventory = player.inventory.lock();
585                let stack = inventory.get_item_in_hand(hand);
586                stack.copy_with_count(1)
587            };
588            if held.is_empty() || self.is_removed() {
589                return InteractionResult::Pass;
590            }
591
592            self.set_item(held);
593            self.play_sound_at_frame(&sound_events::ENTITY_ITEM_FRAME_ADD_ITEM);
594            self.frame_changed(player);
595
596            if !player.has_infinite_materials() {
597                let mut inventory = player.inventory.lock();
598                inventory.get_item_in_hand_mut(hand).shrink(1);
599            }
600            return InteractionResult::Success;
601        }
602
603        {
604            let mut entity_data = self.entity_data.lock();
605            let next = (*entity_data.rotation.get() + 1).rem_euclid(ROTATION_STEPS);
606            entity_data.rotation.set(next);
607        }
608        self.play_sound_at_frame(&sound_events::ENTITY_ITEM_FRAME_ROTATE_ITEM);
609        self.frame_changed(player);
610        InteractionResult::Success
611    }

617    fn save_additional(&self, nbt: &mut NbtCompound) {
618        let block_pos = *self.block_pos.lock();
619        nbt.insert(
620            "block_pos",
621            NbtTag::IntArray(vec![block_pos.x(), block_pos.y(), block_pos.z()]),
622        );
623
624        let entity_data = self.entity_data.lock();
625        let item = entity_data.item.get();
626        if !item.is_empty() {
627            nbt.insert("Item", item.to_nbt_tag_ref());
628        }
629        nbt.insert("ItemRotation", *entity_data.rotation.get() as i8);
630        nbt.insert("ItemDropChance", self.drop_chance());
631        nbt.insert(
632            "Facing",
633            direction_3d_data_value(*entity_data.hanging_entity.direction.get()) as i8,
634        );
635        nbt.insert("Invisible", 0_i8);
636        nbt.insert("Fixed", i8::from(self.is_fixed()));
637    }

639    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
640        if let Some(block_pos) = nbt.int_array("block_pos")
641            && block_pos.len() == 3
642        {
643            *self.block_pos.lock() = BlockPos::new(block_pos[0], block_pos[1], block_pos[2]);
644        }
645
646        if let Some(item_tag) = nbt.compound("Item")
647            && let Some(item) = ItemStack::from_borrowed_compound(&item_tag)
648        {
649            self.set_item_with_update(item, false);
650        }
651
652        {
653            let mut state = self.state.lock();
654            if let Some(drop_chance) = nbt.float("ItemDropChance") {
655                state.drop_chance = drop_chance;
656            }
657            if let Some(fixed) = nbt.byte("Fixed") {
658                state.fixed = fixed != 0;
659            }
660        }
661
662        if let Some(item_rotation) = nbt.byte("ItemRotation") {
663            self.entity_data
664                .lock()
665                .rotation
666                .set(i32::from(item_rotation).rem_euclid(8));
667        }
668
669        let facing = nbt
670            .byte("Facing")
671            .and_then(|value| direction_from_3d_data_value(i32::from(value)))
672            .or_else(|| nbt.int("Facing").and_then(direction_from_3d_data_value));
673        if let Some(direction) = facing {
674            self.set_direction(direction);
675        }
676
677        self.recalculate_position();
678    }
679}

717/// Vanilla parity: `Direction.get3DDataValue`.
718pub(super) const fn direction_3d_data_value(direction: Direction) -> i32 {
719    match direction {
720        Direction::Down => 0,
721        Direction::Up => 1,
722        Direction::North => 2,
723        Direction::South => 3,
724        Direction::West => 4,
725        Direction::East => 5,
726    }
727}

729/// Vanilla parity: `Direction.from3DDataValue`, with the out-of-range case
730/// rejected rather than wrapped.
731pub(super) const fn direction_from_3d_data_value(value: i32) -> Option<Direction> {
732    match value {
733        0 => Some(Direction::Down),
734        1 => Some(Direction::Up),
735        2 => Some(Direction::North),
736        3 => Some(Direction::South),
737        4 => Some(Direction::West),
738        5 => Some(Direction::East),
739        _ => None,
740    }
741}

743/// Vanilla parity: `Direction.get2DDataValue`.
744pub(super) const fn direction_2d_data_value(direction: Direction) -> u8 {
745    match direction {
746        Direction::South | Direction::Down | Direction::Up => 0,
747        Direction::West => 1,
748        Direction::North => 2,
749        Direction::East => 3,
750    }
