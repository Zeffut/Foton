37/// A painting hanging on a wall.
38///
39/// Vanilla parity: `Painting`. The variant carries the size, so every change
40/// of variant or facing resizes the entity. Support is checked periodically and
41/// a physics break event is emitted when the wall no longer supports it.
42#[entity_behavior(class = "Painting")]
43pub struct PaintingEntity {
44    base: EntityBase,
45    entity_type: EntityTypeRef,
46    entity_data: SyncMutex<PaintingEntityData>,
47    block_pos: SyncMutex<BlockPos>,
48    check_interval: SyncMutex<i32>,
49}

53    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:entity/painting");
54}

56/// How far the canvas sits from the middle of its block, toward the wall.
57///
58/// Vanilla parity: the `shiftToBlockWall` local of
59/// `Painting.calculateBoundingBox`.
60const SHIFT_TO_BLOCK_WALL: f64 = 0.46875;
61
62/// Vanilla parity: `Painting.DEPTH`.
63const DEPTH: f64 = 0.0625;
64
65/// Vanilla parity: the `deflate(1.0E-7)` of `HangingEntity.calculateSupportBox`.
66///
67/// Without it the support region would spill into the block past each edge,
68/// and a painting would demand a wall one block wider than it is.
69const SUPPORT_BOX_EPSILON: f64 = 1.0e-7;

88    /// Creates a fresh painting attached to `block_pos`, carrying the default
89    /// variant.
90    ///
91    /// Vanilla parity: `Painting(Level, BlockPos)` plus the
92    /// `VariantUtils.getAny` seed of `Painting.defineSynchedData`. The
93    /// generated synced data already defaults to the first registry entry, so
94    /// the seed is inherited rather than repeated here.
95    #[must_use]
96    pub fn new_attached(
97        entity_type: EntityTypeRef,
98        id: i32,
99        block_pos: BlockPos,
100        direction: Direction,
101        world: Weak<World>,
102    ) -> Self {
103        let entity_data = PaintingEntityData::new();
104        let variant = entity_data.painting_variant.get().value();
105        let bounding_box = Self::calculate_bounding_box(block_pos, direction, variant);
106        let entity = Self {
107            base: EntityBase::new_with_state(
108                id,
109                EntityBaseState::new_with_bounding_box(
110                    bounding_box.center(),
111                    entity_type.dimensions,
112                    bounding_box,
113                )
114                .with_rotation(Self::rotation_for_direction(direction)),
115                world,
116            ),
117            entity_type,
118            entity_data: SyncMutex::new(entity_data),
119            block_pos: SyncMutex::new(block_pos),
120            check_interval: SyncMutex::new(0),
121        };
122        entity
123            .entity_data
124            .lock()
125            .hanging_entity
126            .direction
127            .set(direction);
128        entity
129    }

148    /// Builds a painting for `block_pos`, wearing the biggest picture that fits.
149    ///
150    /// Vanilla parity: `Painting.create`. Returns `None` when nothing in the
151    /// `placeable` tag survives on that wall -- that is what makes a painting
152    /// item do nothing against a lone block instead of hanging a canvas in
153    /// mid-air.
154    #[must_use]
155    pub fn create(
156        entity_type: EntityTypeRef,
157        id: i32,
158        world: &Arc<World>,
159        block_pos: BlockPos,
160        direction: Direction,
161    ) -> Option<Self> {
162        let candidate =
163            Self::new_attached(entity_type, id, block_pos, direction, Arc::downgrade(world));
164
165        let mut fitting: Vec<PaintingVariantRef> = REGISTRY
166            .painting_variants
167            .iter_tag(&PaintingVariantTag::PLACEABLE)
168            .filter(|variant| {
169                candidate.set_variant(variant);
170                candidate.survives()
171            })
172            .collect();
173
174        // Vanilla keeps only the joint-largest by area and rolls between them,
175        // so a wide wall never gets a postage stamp.
176        let largest_area = fitting.iter().map(|variant| variant_area(variant)).max()?;
177        fitting.retain(|variant| variant_area(variant) >= largest_area);
178        let selected = *fitting.get(rand::random_range(0..fitting.len()))?;
179
180        candidate.set_variant(selected);
181        Some(candidate)
182    }

209    /// Returns whether the painting can stay where it is.
210    ///
211    /// Vanilla parity: `HangingEntity.survives`. Foton's item frame has no
212    /// such check at all today, and this change deliberately leaves it that
213    /// way; the painting needs one because its size is chosen by whether it
214    /// fits.
215    ///
216    /// Deviation: vanilla's `hasLevelCollision` also refuses a pop box that
217    /// crosses the world border. Foton's `WorldCollisionProvider` only reports
218    /// border shapes for a moving entity, so only the block half is tested.
219    #[must_use]
220    pub fn survives(&self) -> bool {
221        let Some(world) = self.level() else {
222            return false;
223        };
224        let pop_box = self.bounding_box();
225        if has_block_collision(&WorldCollisionProvider::new(&world), pop_box) {
226            return false;
227        }
228        self.is_supported(&world, pop_box) && self.can_coexist(&world, pop_box)
229    }
230
231    /// Vanilla parity: the support half of `HangingEntity.survives`, walked
232    /// over `calculateSupportBox`.
233    fn is_supported(&self, world: &World, pop_box: WorldAabb) -> bool {
234        let support = pop_box
235            .translate(self.direction().offset_vec().as_dvec3() * -0.5)
236            .deflate(SUPPORT_BOX_EPSILON);
237        let min = BlockPos::new(
238            support.min_x().floor() as i32,
239            support.min_y().floor() as i32,
240            support.min_z().floor() as i32,
241        );
242        let max = BlockPos::new(
243            support.max_x().floor() as i32,
244            support.max_y().floor() as i32,
245            support.max_z().floor() as i32,
246        );
247        BlockPos::between_closed(min, max).all(|pos| {
248            let state = world.get_block_state(pos);
249            state.is_solid() || is_diode(state)
250        })
251    }
252
253    /// Vanilla parity: `HangingEntity.canCoexist(false)`.
254    ///
255    /// Deviation: vanilla tests `EntityTypeTest.forClass(HangingEntity.class)`.
256    /// Foton has no `HangingEntity` supertype, so the concrete hanging
257    /// entities are recognized one by one in [`hanging_entity_direction`].
258    fn can_coexist(&self, world: &World, pop_box: WorldAabb) -> bool {
259        let own_id = self.base.id();
260        let own_direction = self.direction();
261        let own_type = self.entity_type;
262        !world.has_entity_in_aabb_matching(&pop_box, |entity| {
263            let Some(direction) = hanging_entity_direction(entity) else {
264                return false;
265            };
266            entity.id() != own_id
267                && (entity.entity_type() == own_type || direction == own_direction)
268        })
269    }

271    /// Vanilla parity: `HangingEntity.setDirection`.
272    pub fn set_direction(&self, direction: Direction) {
273        self.entity_data
274            .lock()
275            .hanging_entity
276            .direction
277            .set(direction);
278        self.base
279            .set_rotation(Self::rotation_for_direction(direction));
280        self.recalculate_position();
281    }

283    /// Vanilla parity: `HangingEntity.recalculateBoundingBox`.
284    fn recalculate_position(&self) {
285        let block_pos = *self.block_pos.lock();
286        let (direction, variant) = {
287            let entity_data = self.entity_data.lock();
288            (
289                *entity_data.hanging_entity.direction.get(),
290                entity_data.painting_variant.get().value(),
291            )
292        };
293        let bounding_box = Self::calculate_bounding_box(block_pos, direction, variant);
294        if let Err(error) = self.base.try_set_position(bounding_box.center()) {
295            panic!(
296                "failed to commit painting {} position recalculation: {error}",
297                self.base.id()
298            );
299        }
300        self.base.set_bounding_box(bounding_box);
301    }
302
303    fn rotation_for_direction(direction: Direction) -> (f32, f32) {
304        (f32::from(direction_2d_data_value(direction)) * 90.0, 0.0)
305    }
306
307    /// Vanilla parity: `Painting.calculateBoundingBox`.
308    ///
309    /// The canvas hangs off the middle of the block, then slides half a block
310    /// left and up for every even dimension: an even-sized painting straddles
311    /// the seam between blocks, an odd-sized one is centered on one.
312    fn calculate_bounding_box(
313        block_pos: BlockPos,
314        direction: Direction,
315        variant: PaintingVariantRef,
316    ) -> WorldAabb {
317        let value = variant.value();
318        let attached_to_wall = block_pos.0.as_dvec3() + DVec3::splat(0.5)
319            - direction.offset_vec().as_dvec3() * SHIFT_TO_BLOCK_WALL;
320        let left = direction.rotate_y_counter_clockwise();
321        let position = attached_to_wall
322            + left.offset_vec().as_dvec3() * offset_for_painting_size(value.width)
323            + DVec3::Y * offset_for_painting_size(value.height);
324
325        let width = f64::from(value.width);
326        let x_size = if direction.axis() == Axis::X {
327            DEPTH
328        } else {
329            width
330        };
331        let z_size = if direction.axis() == Axis::Z {
332            DEPTH
333        } else {
334            width
335        };
336        WorldAabb::of_size(position, x_size, f64::from(value.height), z_size)
337    }
338}

363    fn tick(&self) {
364        let Some(world) = self.level() else {
365            return;
366        };
367        let mut interval = self.check_interval.lock();
368        if *interval < 100 {
369            *interval += 1;
370            return;
371        }
372        *interval = 0;
373        if self.is_removed() || self.survives() {
374            return;
375        }
376        let mut event = HangingBreakEvent::new(self.uuid(), "PHYSICS");
377        world.fire_event(&mut event);
378        if !event.is_cancelled() {
379            self.set_removed(RemovalReason::Discarded);
380            self.drop_item(&world, None);
381        }
382    }

403    /// Vanilla parity: `Painting.addAdditionalSaveData`.
404    ///
405    /// The facing is the 2-D legacy value, not the 3-D one the item frame
406    /// writes: `Painting` stores it through `Direction.LEGACY_ID_CODEC_2D`.
407    fn save_additional(&self, nbt: &mut NbtCompound) {
408        nbt.insert("facing", direction_2d_data_value(self.direction()) as i8);
409
410        let block_pos = *self.block_pos.lock();
411        nbt.insert(
412            "block_pos",
413            NbtTag::IntArray(vec![block_pos.x(), block_pos.y(), block_pos.z()]),
414        );
415
416        nbt.insert("variant", self.variant().key.to_string());
417    }
418
419    /// Vanilla parity: `Painting.readAdditionalSaveData`.
420    ///
421    /// The facing is read before `block_pos` but applied after it, because
422    /// applying it recalculates a box anchored on that block.
423    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
424        let facing = nbt.byte("facing").map_or(Direction::South, |value| {
425            direction_from_2d_data_value(i32::from(value))
426        });
427
428        if let Some(block_pos) = nbt.int_array("block_pos")
429            && block_pos.len() == 3
430        {
431            *self.block_pos.lock() = BlockPos::new(block_pos[0], block_pos[1], block_pos[2]);
432        }
433
434        self.set_direction(facing);
435
436        if let Some(variant) = nbt.string("variant")
437            && let Ok(key) = Identifier::from_str(variant.to_str().as_ref())
438            && let Some(variant) = REGISTRY.painting_variants.by_key(&key)
439        {
440            self.set_variant(variant);
441        }
442
443        self.recalculate_position();
444    }
445}

447/// Vanilla parity: `PaintingVariant.area`, which Foton's registry value does
448/// not carry.
449const fn variant_area(variant: PaintingVariantRef) -> i32 {

453/// Vanilla parity: `Painting.offsetForPaintingSize`.
454const fn offset_for_painting_size(size: i32) -> f64 {
455    if size % 2 == 0 { 0.5 } else { 0.0 }
456}

458/// Vanilla parity: `DiodeBlock.isDiode`. `RepeaterBlock` and `ComparatorBlock`
459/// are the only two `DiodeBlock` subclasses, so the class test becomes a pair
460/// of block comparisons.
461fn is_diode(state: BlockStateId) -> bool {
462    let block = state.get_block();
463    block == &vanilla_blocks::REPEATER || block == &vanilla_blocks::COMPARATOR
464}

466/// The facing of `entity` when it is one of Foton's hanging entities.
467///
468/// Vanilla parity: the `EntityTypeTest.forClass(HangingEntity.class)` of
469/// `HangingEntity.canCoexist`. Foton has no `HangingEntity` supertype, so the
470/// painting is matched by its concrete type and both item frames through the
471/// `ItemFrame` capability trait -- which is what stops a glow frame being hung
472/// inside a plain one.
473fn hanging_entity_direction(entity: &dyn Entity) -> Option<Direction> {
474    if let Some(painting) = entity.downcast_ref::<PaintingEntity>() {
475        return Some(painting.direction());
476    }
477    entity.as_item_frame().map(ItemFrame::direction)
478}

480/// Vanilla parity: `Direction.get2DDataValue`.
481const fn direction_2d_data_value(direction: Direction) -> u8 {
482    match direction {
483        Direction::South | Direction::Down | Direction::Up => 0,
484        Direction::West => 1,
485        Direction::North => 2,
486        Direction::East => 3,
487    }
488}
489
490/// Vanilla parity: `Direction.from2DDataValue`, wrap included.
491const fn direction_from_2d_data_value(value: i32) -> Direction {
492    let remainder = value % 4;
493    let index = if remainder < 0 { -remainder } else { remainder };
494    match index {
495        1 => Direction::West,
496        2 => Direction::North,
497        3 => Direction::East,
498        _ => Direction::South,
499    }
500}

502impl BlockAttached for PaintingEntity {
503    /// Vanilla parity: `Painting.dropItem`. Every painting drops the same
504    /// plain item -- the variant is rolled again when it is hung back up.
505    fn drop_item(&self, world: &World, caused_by: Option<&SharedEntity>) {
506        if !world.get_game_rule(&ENTITY_DROPS) {
507            return;
508        }
509        self.play_sound(&sound_events::ENTITY_PAINTING_BREAK, 1.0, 1.0);
510        if drop_would_be_wasted(caused_by) {
511            return;
512        }
513        self.spawn_at_location(ItemStack::new(&vanilla_items::PAINTING), 0.0);
514    }
