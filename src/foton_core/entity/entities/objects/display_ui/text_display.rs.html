24/// Draws a drop shadow behind the glyphs.
25///
26/// Vanilla parity: `Display.TextDisplay.FLAG_SHADOW`.
27pub const FLAG_SHADOW: i8 = 1;
28
29/// Draws the text through blocks, like a glowing name tag.
30///
31/// Vanilla parity: `Display.TextDisplay.FLAG_SEE_THROUGH`.
32pub const FLAG_SEE_THROUGH: i8 = 2;
33
34/// Uses the viewer's own text-background color instead of `background`.
35///
36/// Vanilla parity: `Display.TextDisplay.FLAG_USE_DEFAULT_BACKGROUND`.
37pub const FLAG_USE_DEFAULT_BACKGROUND: i8 = 4;
38
39/// Alignment bit for left-aligned text.
40///
41/// Vanilla parity: `Display.TextDisplay.FLAG_ALIGN_LEFT`.
42pub const FLAG_ALIGN_LEFT: i8 = 8;
43
44/// Alignment bit for right-aligned text.
45///
46/// Vanilla parity: `Display.TextDisplay.FLAG_ALIGN_RIGHT`.
47pub const FLAG_ALIGN_RIGHT: i8 = 16;
48
49/// Wrap width in pixels before a line breaks.
50///
51/// Vanilla parity: `Display.TextDisplay.INITIAL_LINE_WIDTH`.
52const INITIAL_LINE_WIDTH: i32 = 200;
53
54/// Default background, a quarter-opaque black in ARGB.
55///
56/// Vanilla parity: `Display.TextDisplay.INITIAL_BACKGROUND`.
57const INITIAL_BACKGROUND: i32 = 1_073_741_824;
58
59/// Default opacity byte. `-1` is vanilla's "fully opaque" sentinel.
60///
61/// Vanilla parity: `Display.TextDisplay.INITIAL_TEXT_OPACITY`.
62const INITIAL_TEXT_OPACITY: i8 = -1;
63
64/// Where each wrapped line sits inside the block of text.
65///
66/// Vanilla parity: `Display.TextDisplay.Align`. Vanilla keeps this in two bits
67/// of the style-flags byte rather than a field of its own, so the NBT name and
68/// the synced byte disagree on shape and both have to be converted by hand.
69#[derive(Debug, Clone, Copy, PartialEq, Eq)]
70pub enum TextAlign {
71    /// Lines are centered on the entity.
72    Center,
73    /// Lines share a left edge.
74    Left,
75    /// Lines share a right edge.
76    Right,
77}
78
79impl TextAlign {
80    /// Reads the alignment out of a style-flags byte.
81    ///
82    /// Vanilla parity: `Display.TextDisplay.getAlign`. Left wins when both
83    /// alignment bits are set, exactly as vanilla orders the checks.
84    #[must_use]
85    pub const fn from_flags(flags: i8) -> Self {
86        if flags & FLAG_ALIGN_LEFT != 0 {
87            Self::Left
88        } else if flags & FLAG_ALIGN_RIGHT != 0 {
89            Self::Right
90        } else {
91            Self::Center
92        }
93    }
94
95    /// Returns the flag bits this alignment contributes to the style byte.
96    #[must_use]
97    const fn flag_bits(self) -> i8 {
98        match self {
99            Self::Center => 0,
100            Self::Left => FLAG_ALIGN_LEFT,
101            Self::Right => FLAG_ALIGN_RIGHT,
102        }
103    }
104
105    /// Returns the vanilla NBT name for this alignment.
106    ///
107    /// Vanilla parity: `Display.TextDisplay.Align.getSerializedName`.
108    #[must_use]
109    pub const fn serialized_name(self) -> &'static str {
110        match self {
111            Self::Center => "center",
112            Self::Left => "left",
113            Self::Right => "right",
114        }
115    }
116
117    /// Parses a vanilla NBT alignment name.
118    #[must_use]
119    pub fn from_serialized_name(name: &str) -> Option<Self> {
120        match name {
121            "center" => Some(Self::Center),
122            "left" => Some(Self::Left),
123            "right" => Some(Self::Right),
124            _ => None,
125        }
126    }
127}

129/// A text display entity.
130///
131/// Vanilla parity: `Display.TextDisplay`. Like its `BlockDisplay` sibling this
132/// carries the subclass state only. The shared `Display` layer -- transformation,
133/// billboard mode, brightness, view range, shadow, culling size and glow color --
134/// exists in the synced data but is neither persisted nor interpolated here,
135/// because Foton has no display render-state system; see `BlockDisplayEntity`
136/// for the same gap.
137#[entity_behavior(class = "TextDisplay")]
138pub struct TextDisplayEntity {
139    /// Common entity fields (id, uuid, position, etc.).
140    base: EntityBase,
141    /// Vanilla entity type registered for this implementation.
142    entity_type: EntityTypeRef,
143    /// Synced entity data for network serialization.
144    entity_data: SyncMutex<TextDisplayEntityData>,
145}

281    /// Vanilla parity: `Display.TextDisplay.addAdditionalSaveData`.
282    ///
283    /// The shared `Display.addAdditionalSaveData` half is not written, matching
284    /// `BlockDisplayEntity`.
285    fn save_additional(&self, nbt: &mut NbtCompound) {
286        let data = self.entity_data.lock();
287        let flags = *data.style_flags.get();
288        nbt.insert("text", data.text.get().to_codec_nbt());
289        nbt.insert("line_width", *data.line_width.get());
290        nbt.insert("background", *data.background_color.get());
291        nbt.insert("text_opacity", *data.text_opacity.get());
292        drop(data);
293
294        // Vanilla parity: `Display.TextDisplay.storeFlag`, one boolean per bit.
295        nbt.insert("shadow", i8::from(flags & FLAG_SHADOW != 0));
296        nbt.insert("see_through", i8::from(flags & FLAG_SEE_THROUGH != 0));
297        nbt.insert(
298            "default_background",
299            i8::from(flags & FLAG_USE_DEFAULT_BACKGROUND != 0),
300        );
301        nbt.insert("alignment", TextAlign::from_flags(flags).serialized_name());
302    }
303
304    /// Vanilla parity: `Display.TextDisplay.readAdditionalSaveData`.
305    ///
306    /// Deviation: vanilla resolves selectors and scores in the saved component
307    /// through `ComponentUtils.resolve` with a gamemaster command source. Foton
308    /// has no component resolution context, so the component is restored as
309    /// written.
310    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
311        let mut data = self.entity_data.lock();
312        data.line_width
313            .set(nbt.int("line_width").unwrap_or(INITIAL_LINE_WIDTH));
314        data.text_opacity
315            .set(nbt.byte("text_opacity").unwrap_or(INITIAL_TEXT_OPACITY));
316        data.background_color
317            .set(nbt.int("background").unwrap_or(INITIAL_BACKGROUND));
318
319        // Vanilla parity: `Display.TextDisplay.loadFlag`, absent means unset.
320        let mut flags = 0_i8;
321        flags |= load_flag(nbt, "shadow", FLAG_SHADOW);
322        flags |= load_flag(nbt, "see_through", FLAG_SEE_THROUGH);
323        flags |= load_flag(nbt, "default_background", FLAG_USE_DEFAULT_BACKGROUND);
324        if let Some(align) = nbt
325            .string("alignment")
326            .and_then(|name| TextAlign::from_serialized_name(&name.to_str()))
327        {
328            flags |= align.flag_bits();
329        }
330        data.style_flags.set(flags);
331
332        if let Some(text) = nbt
333            .get("text")
334            .map(|tag| tag.to_owned())
335            .as_ref()
336            .and_then(TextComponent::from_nbt)
337        {
338            data.text.set(Box::new(text));
339        }
340    }
341}

343/// Turns one saved boolean into its style-flag bit.
344///
345/// Vanilla parity: `Display.TextDisplay.loadFlag`.
346fn load_flag(nbt: BorrowedNbtCompoundView<'_, '_>, key: &str, mask: i8) -> i8 {
347    if nbt.byte(key).is_some_and(|value| value != 0) {
348        mask
349    } else {
350        0
351    }

355mod tests {
356    use std::io::Cursor;
357
358    use foton_registry::{init_vanilla_registry, vanilla_entities};
359    use simdnbt::borrow::read_compound;
360
361    use super::*;
362    use crate::entity::{EntityBaseSaveData, EntityFireFreezeState};
363
364    fn reload(entity: &TextDisplayEntity) -> TextDisplayEntity {
365        let mut saved = NbtCompound::new();
366        entity.save_additional(&mut saved);
367        let mut bytes = Vec::new();
368        saved.write(&mut bytes);
369        let Ok(borrowed) = read_compound(&mut Cursor::new(bytes.as_slice())) else {
370            panic!("saved text display NBT should reborrow");
371        };
372        let loaded = TextDisplayEntity::from_saved(
373            &vanilla_entities::TEXT_DISPLAY,
374            EntityBaseLoad {
375                id: 2,
376                position: DVec3::ZERO,
377                uuid: Uuid::nil(),
378                velocity: DVec3::ZERO,
379                rotation: (0.0, 0.0),
380                fall_distance: 0.0,
381                fire_freeze: EntityFireFreezeState::new(),
382                on_ground: false,
383                save_data: EntityBaseSaveData::new(),
384                world: Weak::new(),
385            },
386        );
387        loaded.load_additional((&borrowed).into());
388        loaded
389    }
390
391    #[test]
392    fn a_saved_text_display_comes_back_with_its_text_layout_and_style_bits() {
393        init_vanilla_registry();
394        let entity = TextDisplayEntity::new(
395            &vanilla_entities::TEXT_DISPLAY,
396            1,
397            DVec3::new(0.5, 65.0, 0.5),
398            Weak::new(),
399        );
400        entity.set_text(TextComponent::plain("hello hologram"));
401        entity.set_line_width(72);
402        entity.set_background_color(0x40FF_00FF_u32 as i32);
403        entity.set_text_opacity(96);
404        entity.set_style_flags(FLAG_SHADOW | FLAG_SEE_THROUGH | FLAG_ALIGN_RIGHT);
405
406        let loaded = reload(&entity);
407
408        assert_eq!(loaded.text(), TextComponent::plain("hello hologram"));
409        assert_eq!(loaded.line_width(), 72);
410        assert_eq!(loaded.background_color(), 0x40FF_00FF_u32 as i32);
411        assert_eq!(loaded.text_opacity(), 96);
412        assert_eq!(
413            loaded.style_flags(),
414            FLAG_SHADOW | FLAG_SEE_THROUGH | FLAG_ALIGN_RIGHT
415        );
416        assert_eq!(loaded.align(), TextAlign::Right);
417    }

419    #[test]
420    fn a_text_display_saved_with_no_style_reloads_on_vanilla_defaults() {
421        init_vanilla_registry();
422        let entity =

428        assert_eq!(loaded.background_color(), INITIAL_BACKGROUND);
429        assert_eq!(loaded.text_opacity(), INITIAL_TEXT_OPACITY);
430        assert_eq!(loaded.style_flags(), 0);
431        assert_eq!(loaded.align(), TextAlign::Center);
432    }
