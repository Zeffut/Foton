1//! The furnace minecart.
2//!
3//! Vanilla parity: `MinecartFurnace`. A cart that drives itself: feed it coal
4//! and it pushes along the rails in whatever direction it was fed from, for a
5//! minute per piece of fuel. It is slower than an ordinary cart, which is the
6//! price of not needing a powered rail.
7//!
8//! Everything about rolling is [`super::minecart_common`]; what is here is the
9//! fuel, the push, and the slowdown the push replaces.
10

34/// Vanilla parity: `AbstractMinecart.getDefaultGravity`.
35const MINECART_GRAVITY: f64 = 0.04;
36
37/// And the same in water.
38const MINECART_GRAVITY_IN_WATER: f64 = 0.005;
39
40/// How long one piece of fuel lasts.
41///
42/// Vanilla parity: `MinecartFurnace.FUEL_TICKS_PER_ITEM` -- three minutes.
43const FUEL_TICKS_PER_ITEM: i32 = 3600;
44
45/// How much fuel the cart will hold.
46///
47/// Vanilla parity: `MinecartFurnace.MAX_FUEL_TICKS`.
48const MAX_FUEL_TICKS: i32 = 32000;
49
50/// How fast a fueled cart runs, against an ordinary one.
51///
52/// Vanilla parity: the `* 0.5` of `MinecartFurnace.getMaxSpeed`.
53const SPEED_ON_LAND: f64 = 0.5;
54
55/// And in water, where it loses less.
56///
57/// Vanilla parity: the `* 0.75` of the same method.
58const SPEED_IN_WATER: f64 = 0.75;
59
60/// The drag a pushing cart feels, against the `0.98` of a coasting one.
61///
62/// Vanilla parity: the `multiply(0.8, 0.0, 0.8)` of `applyNaturalSlowdown`.
63const PUSHING_DRAG: f64 = 0.8;
64
65/// And the drag with no fuel left.
66const COASTING_DRAG: f64 = 0.98;
67
68/// How much of its push a cart keeps under water.
69///
70/// Vanilla parity: the `scale(0.1)` of `applyNaturalSlowdown`.
71const UNDERWATER_PUSH: f64 = 0.1;
72
73/// Below this, the push counts as nothing.
74///
75/// Vanilla parity: the `1.0E-7` of `applyNaturalSlowdown`.
76const PUSH_EPSILON: f64 = 1.0e-7;
77
78/// Vanilla parity: the `epsilonPushCheck` of `calculateNewPushAlong`.
79const PUSH_REALIGN_EPSILON: f64 = 1.0e-4;
80
81/// Vanilla parity: the `epsilonMovementCheck` of the same method.
82const MOVEMENT_REALIGN_EPSILON: f64 = 0.001;
83
84/// A furnace minecart.
85#[entity_behavior(class = "MinecartFurnace")]
86pub struct FurnaceMinecartEntity {
87    base: EntityBase,
88    entity_type: EntityTypeRef,
89    minecart: SyncMutex<MinecartState>,
90    entity_data: SyncMutex<FurnaceMinecartEntityData>,
91    /// Ticks of fuel left.
92    fuel: AtomicI32,
93    /// The direction the cart drives itself in, horizontal only.
94    push: SyncMutex<DVec3>,
95}

102impl FurnaceMinecartEntity {
103    /// Creates one at runtime.
104    #[must_use]
105    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
106        Self::new_with_base(
107            EntityBase::new(id, position, entity_type.dimensions, world),
108            entity_type,
109        )
110    }
111
112    /// Creates one from saved base data.
113    #[must_use]
114    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
115        Self::new_with_base(
116            EntityBase::from_load(load, entity_type.dimensions),
117            entity_type,
118        )
119    }
120
121    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
122        Self {
123            base,
124            entity_type,
125            minecart: SyncMutex::new(MinecartState::default()),
126            entity_data: SyncMutex::new(FurnaceMinecartEntityData::new()),
127            fuel: AtomicI32::new(0),
128            push: SyncMutex::new(DVec3::ZERO),
129        }
130    }
131
132    /// Returns whether the furnace is burning.
133    #[must_use]
134    pub fn has_fuel(&self) -> bool {
135        self.fuel.load(Ordering::Relaxed) > 0
136    }
137
138    /// Returns the ticks of fuel left.
139    #[must_use]
140    pub fn fuel(&self) -> i32 {
141        self.fuel.load(Ordering::Relaxed)
142    }
143
144    /// Feeds the cart, reporting whether the item was taken.
145    ///
146    /// Vanilla parity: `MinecartFurnace.addFuel`. The push points away from
147    /// whoever fed it, which is how a player chooses the direction: you stand
148    /// behind the cart and feed it forwards.
149    pub fn add_fuel(&self, from: DVec3, item: &ItemStack) -> bool {
150        if !REGISTRY
151            .items
152            .is_in_tag(item.item(), &ItemTag::FURNACE_MINECART_FUEL)
153        {
154            return false;
155        }
156        let fuel = self.fuel.load(Ordering::Relaxed);
157        if fuel + FUEL_TICKS_PER_ITEM > MAX_FUEL_TICKS {
158            return false;
159        }
160
161        self.fuel
162            .store(fuel + FUEL_TICKS_PER_ITEM, Ordering::Relaxed);
163        let away = self.position() - from;
164        *self.push.lock() = DVec3::new(away.x, 0.0, away.z);
165        true
166    }
167
168    /// Keeps the push pointing the way the cart is actually going.
169    ///
170    /// Vanilla parity: `MinecartFurnace.calculateNewPushAlong`. A rail corner
171    /// turns the cart, and without this the push would keep shoving it at the
172    /// wall it just turned away from.
173    fn realign_push(&self, movement: DVec3) -> DVec3 {
174        let push = *self.push.lock();
175        let push_xz = push.x.mul_add(push.x, push.z * push.z);
176        let movement_xz = movement.x.mul_add(movement.x, movement.z * movement.z);
177        if push_xz <= PUSH_REALIGN_EPSILON || movement_xz <= MOVEMENT_REALIGN_EPSILON {
178            return push;
179        }
180
181        // The projection of the push onto the direction of travel, scaled back
182        // to the push's own length.
183        let projected = movement * (push.dot(movement) / movement.length_squared());
184        if projected.length_squared() <= 0.0 {
185            return push;
186        }
187        projected.normalize() * push.length()
188    }
189}

191impl Entity for FurnaceMinecartEntity {
192    fn is_minecart(&self) -> bool {
193        true
194    }
195
196    fn base(&self) -> &EntityBase {
197        &self.base
198    }
199
200    fn entity_type(&self) -> EntityTypeRef {
201        self.entity_type
202    }
203
204    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
205        Some(&self.entity_data)
206    }
207
208    /// Vanilla parity: `MinecartFurnace.tick`.
209    fn tick(&self) {
210        minecart_common::tick_minecart(self);
211
212        let fuel = self.fuel.load(Ordering::Relaxed);
213        if fuel > 0 {
214            self.fuel.store(fuel - 1, Ordering::Relaxed);
215        }
216        if self.fuel.load(Ordering::Relaxed) <= 0 {
217            *self.push.lock() = DVec3::ZERO;
218        }
219
220        // The synced flag is what lights the furnace texture on the client.
221        let burning = self.has_fuel();
222        let mut data = self.entity_data.lock();
223        if *data.minecart_furnace.id_fuel.get() != burning {
224            data.minecart_furnace.id_fuel.set(burning);
225        }
226    }
227
228    fn get_default_gravity(&self) -> f64 {
229        if self.is_in_water() {
230            MINECART_GRAVITY_IN_WATER
231        } else {
232            MINECART_GRAVITY
233        }
234    }
235
236    fn blocks_building(&self) -> bool {
237        true
238    }
239
240    fn is_pushable(&self) -> bool {
241        true
242    }
243
244    fn is_pickable(&self) -> bool {
245        !self.is_removed()
246    }
247
248    fn movement_emission(&self) -> EntityMovementEmission {
249        EntityMovementEmission::Events
250    }
251
252    fn sound_source(&self) -> SoundSource {
253        SoundSource::Neutral
254    }
255
256    /// Vanilla parity: `MinecartFurnace.interact`, which answers `SUCCESS`
257    /// whether or not the item was fuel -- there is nothing else to do with a
258    /// furnace cart, so a failed feed is not a fall-through.
259    fn interact(
260        &self,
261        player: &Player,
262        hand: InteractionHand,
263        _location: DVec3,
264    ) -> InteractionResult {
265        let fed = {
266            let inventory = player.inventory.lock();
267            let held = inventory.get_item_in_hand(hand);
268            self.add_fuel(player.position(), held)
269        };
270        if fed && !player.has_infinite_materials() {
271            player.inventory.lock().get_item_in_hand_mut(hand).shrink(1);
272        }
273        InteractionResult::Success
274    }
275
276    fn save_additional(&self, nbt: &mut NbtCompound) {
277        let push = *self.push.lock();
278        nbt.insert("PushX", push.x);
279        nbt.insert("PushZ", push.z);
280        nbt.insert("Fuel", self.fuel.load(Ordering::Relaxed) as i16);
281    }
282
283    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
284        let x = nbt.double("PushX").unwrap_or(0.0);
285        let z = nbt.double("PushZ").unwrap_or(0.0);
286        *self.push.lock() = DVec3::new(x, 0.0, z);
287        if let Some(fuel) = nbt.short("Fuel") {
288            self.fuel.store(i32::from(fuel), Ordering::Relaxed);
289        }
290    }
291}

293impl MinecartLike for FurnaceMinecartEntity {
294    fn minecart_state(&self) -> &SyncMutex<MinecartState> {
295        &self.minecart
296    }
297
298    /// Vanilla parity: `MinecartFurnace.getMaxSpeed`, which is half an ordinary
299    /// cart's on land and three quarters of it in water.
300    fn max_speed_factor(&self) -> f64 {
301        if self.is_in_water() {
302            SPEED_IN_WATER
303        } else {
304            SPEED_ON_LAND
305        }
306    }
307
308    /// Vanilla parity: `MinecartFurnace.applyNaturalSlowdown`. A pushing cart
309    /// sheds speed faster than a coasting one and then adds the push back, so
310    /// it settles at a steady pace rather than accelerating forever.
311    fn apply_natural_slowdown(&self, movement: DVec3) -> DVec3 {
312        let push = *self.push.lock();
313        if push.length_squared() <= PUSH_EPSILON {
314            return DVec3::new(movement.x * COASTING_DRAG, 0.0, movement.z * COASTING_DRAG);
315        }
316
317        let realigned = self.realign_push(movement);
318        *self.push.lock() = realigned;
319        let mut pushed =
320            DVec3::new(movement.x * PUSHING_DRAG, 0.0, movement.z * PUSHING_DRAG) + realigned;
321        if self.is_in_water() {
322            pushed *= UNDERWATER_PUSH;
323        }
324        pushed
325    }
