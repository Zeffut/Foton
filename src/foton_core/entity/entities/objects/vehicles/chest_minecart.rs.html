1//! The chest minecart.
2//!
3//! Vanilla parity: `MinecartChest` and `AbstractMinecartContainer`. A rolling
4//! chest: it runs on rails like any cart, and a right-click opens it rather
5//! than seating anybody, because there is nowhere to sit.
6//!
7//! Mineshafts generate these with a loot table attached. The table is rolled
8//! the first time somebody opens the cart, matching
9//! `ContainerEntity.unpackChestVehicleLootTable`.
10

46/// Slots a chest minecart carries.
47///
48/// Vanilla parity: `MinecartChest.getContainerSize`.
49const CHEST_MINECART_SLOTS: usize = 27;
50
51/// Rows the menu shows.
52const CHEST_MINECART_ROWS: usize = 3;
53
54/// Vanilla parity: `AbstractMinecart.getDefaultGravity`.
55const MINECART_GRAVITY: f64 = 0.04;
56
57/// And the same in water.
58const MINECART_GRAVITY_IN_WATER: f64 = 0.005;
59
60/// How much speed a loaded cart keeps each tick before its cargo is counted.
61///
62/// Vanilla parity: the `0.98F` of `AbstractMinecartContainer.applyNaturalSlowdown`.
63const LOADED_SLOWDOWN: f32 = 0.98;
64
65/// How much each unfilled step of the container gives back.
66///
67/// Vanilla parity: the `emptiness * 0.001F` of the same method: an empty chest
68/// cart rolls noticeably further than a full one.
69const EMPTINESS_BONUS: f32 = 0.001;
70
71/// The largest comparator reading a container can give.
72const FULL_SIGNAL: i32 = 15;
73
74/// Extra drag while under water.
75const WATER_SLOWDOWN: f32 = 0.95;
76
77/// A minecart with a chest in it.
78#[entity_behavior(class = "MinecartChest")]
79pub struct ChestMinecartEntity {
80    base: EntityBase,
81    entity_type: EntityTypeRef,
82    state: SyncMutex<ChestMinecartState>,
83    minecart: SyncMutex<MinecartState>,
84    container: Shared<SimpleContainer>,
85    container_ref: ContainerRef,
86    /// Vanilla parity: the `lootTable`/`lootTableSeed` pair of
87    /// `AbstractMinecartContainer`.
88    loot: Arc<ContainerLoot>,
89}

107impl ChestMinecartEntity {
108    /// Creates a new chest minecart entity.
109    #[must_use]
110    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
111        Self::with_state(
112            EntityBase::new(id, position, entity_type.dimensions, world),
113            entity_type,
114            ChestMinecartState::new(true),
115        )
116    }
117
118    /// Creates a chest minecart entity from saved data.
119    #[must_use]
120    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
121        Self::with_state(
122            EntityBase::from_load(load, entity_type.dimensions),
123            entity_type,
124            ChestMinecartState::new(false),
125        )
126    }
127
128    fn with_state(base: EntityBase, entity_type: EntityTypeRef, state: ChestMinecartState) -> Self {
129        let container: Shared<SimpleContainer> =
130            Arc::new(SyncMutex::new(SimpleContainer::new(CHEST_MINECART_SLOTS)));
131        let shared: SharedContainer = container.clone();
132        Self {
133            base,
134            entity_type,
135            state: SyncMutex::new(state),
136            minecart: SyncMutex::new(MinecartState::default()),
137            container_ref: ContainerRef::from(shared),
138            container,
139            loot: Arc::new(ContainerLoot::new()),
140        }
141    }
142
143    /// Rolls a still-packed loot table into the cart.
144    ///
145    /// Vanilla parity: `ContainerEntity.unpackChestVehicleLootTable`, whose
146    /// `ORIGIN` is the cart's own position because a cart moves.
147    fn unpack_loot_table(&self, player: Option<&Player>) {
148        let Some(world) = self.level() else {
149            return;
150        };
151        let container: SharedContainer = self.container.clone();
152        self.loot
153            .unpack_at(&world, self.position(), &container, player);
154    }
155
156    /// Opens the chest for `player`.
157    ///
158    /// Vanilla parity: `MinecartChest.interact`, which goes straight to the
159    /// container -- there is no seat to compete with, so no sneak is needed.
160    fn open_chest(&self, player: &Player) -> InteractionResult {
161        // Vanilla parity: `AbstractMinecartContainer.createMenu` unpacks
162        // with the opening player, whose luck the roll uses.
163        self.unpack_loot_table(Some(player));
164        let inventory = player.inventory.clone();
165        let container = self.container_ref.clone();
166        player.open_menu(self.name(), move |context| {
167            chest(
168                inventory,
169                context.container_id,
170                container,
171                CHEST_MINECART_ROWS,
172            )
173        });
174
175        if let Some(world) = self.level() {
176            world.game_event_at(
177                &vanilla_game_events::CONTAINER_OPEN,
178                self.position(),
179                &GameEventContext::new(Some(player as &dyn Entity), None),
180            );
181        }
182
183        InteractionResult::Success
184    }
185
186    /// Sets the deferred loot table used when the container is first opened.
187    ///
188    /// Vanilla parity: `AbstractMinecartContainer.setLootTable`.
189    pub fn set_loot_table(&self, loot_table: Identifier, seed: i64) {
190        self.loot.set_loot_table(loot_table, seed);
191    }
192
193    const fn nbt_bool(value: bool) -> i8 {
194        if value { 1 } else { 0 }
195    }
196}

277    fn save_additional(&self, nbt: &mut NbtCompound) {
278        nbt.insert(
279            "FlippedRotation",
280            Self::nbt_bool(self.minecart.lock().flipped),
281        );
282
283        nbt.insert("HasTicked", Self::nbt_bool(self.state.lock().first_tick));
284
285        // Vanilla parity: `ContainerEntity.addChestVehicleSaveData`, which
286        // writes the loot table *instead of* the items when one is packed.
287        if self.loot.try_save_loot_table(nbt) {
288            return;
289        }
290        let container = self.container.lock();
291        let mut items: Vec<NbtCompound> = Vec::new();
292        for (slot, item) in container.items().iter().enumerate() {
293            if !item.is_empty()
294                && let NbtTag::Compound(mut item_nbt) = item.clone().to_nbt_tag()
295            {
296                item_nbt.insert("Slot", slot as i8);
297                items.push(item_nbt);
298            }
299        }
300        nbt.insert("Items", NbtList::Compound(items));
301    }
302
303    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
304        {
305            let mut state = self.state.lock();
306            if let Some(first_tick) = nbt.byte("HasTicked") {
307                state.first_tick = first_tick != 0;
308            }
309        }
310
311        if let Some(flipped) = nbt.byte("FlippedRotation") {
312            self.minecart.lock().flipped = flipped != 0;
313        }
314
315        // Vanilla parity: `ContainerEntity.readChestVehicleSaveData` clears
316        // the slots, then reads the items only when no table is packed.
317        let packed = self.loot.try_load_loot_table(&nbt);
318        let mut container = self.container.lock();
319        container.items_mut().fill(ItemStack::empty());
320        if packed {
321            return;
322        }
323        let Some(items_list) = nbt.list("Items") else {
324            return;
325        };
326        let Some(compounds) = items_list.compounds() else {
327            return;
328        };
329        for compound in compounds {
330            let Some(slot) = compound.byte("Slot") else {
331                continue;
332            };
333            let slot = slot as usize;
334            if slot < CHEST_MINECART_SLOTS
335                && let Some(item) = ItemStack::from_borrowed_compound(&compound)
336            {
337                container.items_mut()[slot] = item;
338            }
339        }
340    }
341}

348    /// Vanilla parity: `AbstractMinecartContainer.applyNaturalSlowdown`.
349    ///
350    /// A loaded cart rolls less far than an empty one, measured by what a
351    /// comparator would read off the container. It is a small effect -- an
352    /// empty cart keeps 0.995 of its speed and a full one 0.98 -- but over a
353    /// long line it is the difference between arriving and stopping short.
354    fn apply_natural_slowdown(&self, movement: DVec3) -> DVec3 {
355        let mut keep = LOADED_SLOWDOWN;
356
357        // Vanilla skips the cargo bonus while a loot table is still packed,
358        // because the cart does not yet know what it is carrying.
359        if !self.loot.is_packed() {
360            let filled = calculate_redstone_signal_from_container(&*self.container.lock());
361            keep += (FULL_SIGNAL - filled) as f32 * EMPTINESS_BONUS;
362        }
363
364        if self.is_in_water() {
365            keep *= WATER_SLOWDOWN;
366        }
367
368        let keep = f64::from(keep);
369        DVec3::new(movement.x * keep, 0.0, movement.z * keep)
370    }
371}

394    fn load_from_owned_nbt(minecart: &ChestMinecartEntity, nbt: &NbtCompound) {
395        let mut bytes = Vec::new();
396        nbt.write(&mut bytes);
397        let base = read_borrowed_compound(&mut Cursor::new(bytes.as_slice()))
398            .expect("test nbt should reborrow");
399        minecart.load_additional((&base).into());
400    }

402    #[test]
403    fn chest_minecart_saves_structure_loot_table_state() {
404        let minecart = test_minecart();
405        minecart.set_loot_table(
406            Identifier::new_static("minecraft", "chests/abandoned_mineshaft"),
407            42,
408        );
409
410        let mut nbt = NbtCompound::new();
411        minecart.save_additional(&mut nbt);
412
413        assert_eq!(
414            nbt.string("LootTable").map(ToString::to_string),
415            Some(MINESHAFT.to_owned())
416        );
417        assert_eq!(nbt.long("LootTableSeed"), Some(42));
418        assert_eq!(nbt.byte("HasTicked"), Some(1));
419        assert_eq!(nbt.byte("FlippedRotation"), Some(0));
420        assert!(
421            nbt.list("Items").is_none(),
422            "vanilla writes the table instead of the items, never both"
423        );
424    }

426    /// A mineshaft cart comes back off disk still packed, and the stale `Items`
427    /// list a hand-edited save might carry does not become free loot on top.
428    #[test]
429    fn a_packed_chest_minecart_reloads_its_table_and_ignores_stale_items() {
430        let minecart = test_minecart();
431        let mut nbt = NbtCompound::new();
432        nbt.insert("LootTable", MINESHAFT);
433        nbt.insert("LootTableSeed", 1234_i64);
434        let mut stale = NbtCompound::new();
435        if let NbtTag::Compound(mut item) = ItemStack::new(&vanilla_items::DIAMOND).to_nbt_tag() {
436            item.insert("Slot", 0_i8);
437            stale = item;
438        }
439        nbt.insert("Items", NbtList::Compound(vec![stale]));
440
441        load_from_owned_nbt(&minecart, &nbt);
442
443        assert!(minecart.loot.is_packed());
444        assert!(
445            minecart
446                .container
447                .lock()
448                .items()
449                .iter()
450                .all(ItemStack::is_empty),
451            "a packed cart must not also carry the items it saved before"
452        );
453
454        let mut saved = NbtCompound::new();
455        minecart.save_additional(&mut saved);
456        assert_eq!(saved.long("LootTableSeed"), Some(1234));
457    }
