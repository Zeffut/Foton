1//! What every boat and raft shares.
2//!
3//! Vanilla parity: `AbstractBoat`. A boat is not a mob with a speed: it is a
4//! box that floats, and everything interesting about it comes from where the
5//! water surface is relative to its own hull. The five states below are that
6//! question answered, and the float step is the answer applied.
7//!
8//! A ridden boat is driven by the client -- vanilla runs `controlBoat` there
9//! and sends the result as a vehicle move, which Foton already accepts. What
10//! the server owns is the boat nobody is riding: it still has to float, drift
11//! and fall, or a boat left on a lake would sink through it.
12

29/// How long a submerged boat carries its passengers before throwing them out.
30///
31/// Vanilla parity: `AbstractBoat.TIME_TO_EJECT`.
32const TIME_TO_EJECT_TICKS: f32 = 60.0;
33
34/// Vanilla parity: `AbstractBoat.getDefaultGravity`.
35pub(super) const BOAT_GRAVITY: f64 = 0.04;
36
37/// How high a rider sits in a boat, as a share of its height.
38///
39/// Vanilla parity: `Boat.rideHeight`.
40pub(super) const BOAT_RIDE_HEIGHT: f64 = 1.0 / 3.0;
41
42/// How high a rider sits on a raft.
43///
44/// Vanilla parity: `Raft.rideHeight`. A raft has no hull, so the rider sits
45/// almost on top of it.
46pub(super) const RAFT_RIDE_HEIGHT: f64 = 0.888_888_9;
47
48/// How much of its speed a boat keeps each tick in water.
49const WATER_DRAG: f64 = 0.9;
50
51/// How much it keeps while fully submerged, where it is being pushed up.
52const UNDER_WATER_DRAG: f64 = 0.45;
53
54/// Upward push a submerged boat gets.
55const UNDER_WATER_BUOYANCY: f64 = 0.01;
56
57/// The tiny downward drift of a boat under flowing water.
58const UNDER_FLOWING_WATER_FALL: f64 = -7.0e-4;
59
60/// How much of the buoyancy survives each tick.
61///
62/// Vanilla parity: the `* 0.75` of `floatBoat`.
63const BUOYANCY_DAMPING: f64 = 0.75;
64
65/// Divisor turning the water gap into an upward push.
66///
67/// Vanilla parity: the `getDefaultGravity() / 0.65` of `floatBoat`.
68const BUOYANCY_SCALE: f64 = 0.65;
69
70/// How many riders a boat carries.
71///
72/// Vanilla parity: `AbstractBoat.getMaxPassengers`.
73pub(super) const MAX_PASSENGERS: usize = 2;

75/// Where a boat sits relative to the water.
76///
77/// Vanilla parity: `AbstractBoat.Status`.
78#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
79pub(super) enum BoatStatus {
80    /// Floating with its deck above the surface.
81    InWater,
82    /// Fully submerged in still water.
83    UnderWater,
84    /// Fully submerged in flowing water.
85    UnderFlowingWater,
86    /// Resting on a block.
87    OnLand,
88    /// Falling.
89    #[default]
90    InAir,
91}
92
93impl BoatStatus {
94    /// Returns whether the boat is fully under.
95    ///
96    /// Vanilla parity: `AbstractBoat.isUnderWater`.
97    pub(super) const fn is_under_water(self) -> bool {
98        matches!(self, Self::UnderWater | Self::UnderFlowingWater)
99    }
100}

102/// The floating state vanilla keeps on the boat itself.
103#[derive(Debug, Default)]
104pub(super) struct BoatState {
105    /// Where the boat sits right now.
106    pub status: BoatStatus,
107    /// Where it sat last tick, which is how a splashdown is detected.
108    pub old_status: BoatStatus,
109    /// Height of the water the hull is resting in.
110    pub water_level: f64,
111    /// Friction of whatever it is sitting on.
112    pub land_friction: f32,
113    /// Ticks it has spent submerged.
114    pub out_of_control_ticks: f32,
115    /// Vertical speed last tick, used by the fall check.
116    pub last_yd: f64,
117}

119/// What a concrete boat exposes to the shared code.
120pub(super) trait BoatLike: Entity {
121    /// Returns the floating state.
122    fn boat_state(&self) -> &SyncMutex<BoatState>;
123
124    /// Returns how high a rider sits.
125    ///
126    /// Vanilla parity: `AbstractBoat.rideHeight`, the one thing a raft changes.
127    fn ride_height(&self, dimensions: EntityDimensions) -> f64;
128}

130/// Boards a player, or refuses to.
131///
132/// Vanilla parity: `AbstractBoat.interact`. Sneaking passes rather than
133/// boarding, which is what leaves room for a chest boat to be opened instead,
134/// and a boat that has spent three seconds upside down will not take a rider
135/// at all -- otherwise a player could climb into a boat that is already
136/// throwing its passengers out.
137pub(super) fn interact_boat<B: BoatLike>(boat: &B, player: &Player) -> InteractionResult {
138    if player.is_secondary_use_active() {
139        return InteractionResult::Pass;
140    }
141    if boat.boat_state().lock().out_of_control_ticks >= TIME_TO_EJECT_TICKS {
142        return InteractionResult::Pass;
143    }
144    let Some(world) = boat.level() else {
145        return InteractionResult::Pass;
146    };
147    let Some(vehicle) = world.get_entity_by_id(boat.id()) else {
148        return InteractionResult::Pass;
149    };
150    if player.start_riding(&vehicle) {
151        InteractionResult::Success
152    } else {
153        InteractionResult::Pass
154    }
155}

157/// Runs one tick of a boat's own physics.
158///
159/// Vanilla parity: the server half of `AbstractBoat.tick`. The client half --
160/// reading the rider's keys and steering -- belongs to the client, which sends
161/// the result back as a vehicle move.
162pub(super) fn tick_boat<B: BoatLike>(boat: &B) {
163    let Some(world) = boat.level() else {
164        return;
165    };
166
167    let status = compute_status(boat, &world);
168    let submerged = {
169        let mut state = boat.boat_state().lock();
170        state.old_status = state.status;
171        state.status = status;
172        if status.is_under_water() {
173            state.out_of_control_ticks += 1.0;
174        } else {
175            state.out_of_control_ticks = 0.0;
176        }
177        state.out_of_control_ticks >= TIME_TO_EJECT_TICKS
178    };
179
180    // Vanilla parity: a boat held under for three seconds throws everyone out,
181    // which is what stops a player drowning inside one.
182    if submerged {
183        for passenger in boat.passengers() {
184            passenger.stop_riding();
185        }
186    }
187
188    float_boat(boat, &world);
189
190    // A ridden boat is moved by its rider's client; an empty one drifts here.
191    if boat.controlling_passenger().is_none() {
192        boat.move_entity(MoverType::SelfMovement, boat.velocity());
193    }
194}

196/// Applies buoyancy, drag and gravity for this tick.
197///
198/// Vanilla parity: `AbstractBoat.floatBoat`.
199fn float_boat<B: BoatLike>(boat: &B, world: &Arc<World>) {
200    let (status, old_status, water_level, land_friction) = {
201        let state = boat.boat_state().lock();
202        (
203            state.status,
204            state.old_status,
205            state.water_level,
206            state.land_friction,
207        )
208    };
209
210    // Vanilla parity: the splashdown branch, which drops a falling boat onto
211    // the surface rather than letting it plunge through.
212    if old_status == BoatStatus::InAir
213        && status != BoatStatus::InAir
214        && status != BoatStatus::OnLand
215    {
216        let target_y =
217            f64::from(water_level_above(boat, world)) - boat.bounding_box().height() + 0.101;
218        let position = boat.position();
219        let _ = boat.try_set_position(DVec3::new(position.x, target_y, position.z));
220        boat.set_velocity(boat.velocity() * DVec3::new(1.0, 0.0, 1.0));
221
222        let mut state = boat.boat_state().lock();
223        state.last_yd = 0.0;
224        state.status = BoatStatus::InWater;
225        return;
226    }
227
228    let mut fall_speed = -BOAT_GRAVITY;
229    let mut buoyancy = 0.0;
230    let drag = match status {
231        BoatStatus::InWater => {
232            buoyancy = (water_level - boat.position().y) / boat.bounding_box().height();
233            WATER_DRAG
234        }
235        BoatStatus::UnderFlowingWater => {
236            fall_speed = UNDER_FLOWING_WATER_FALL;
237            WATER_DRAG
238        }
239        BoatStatus::UnderWater => {
240            buoyancy = UNDER_WATER_BUOYANCY;
241            UNDER_WATER_DRAG
242        }
243        BoatStatus::InAir => WATER_DRAG,
244        BoatStatus::OnLand => f64::from(land_friction),
245    };
246
247    let movement = boat.velocity();
248    let mut next = DVec3::new(
249        movement.x * drag,
250        movement.y + fall_speed,
251        movement.z * drag,
252    );
253
254    if buoyancy > 0.0 {
255        next.y = (next.y + buoyancy * (BOAT_GRAVITY / BUOYANCY_SCALE)) * BUOYANCY_DAMPING;
256    }
257
258    boat.set_velocity(next);
259}

261/// Works out where the boat sits, and records the water level it found.
262///
263/// Vanilla parity: `AbstractBoat.getStatus`.
264fn compute_status<B: BoatLike>(boat: &B, world: &Arc<World>) -> BoatStatus {
265    if let Some(submerged) = submerged_status(boat, world) {
266        boat.boat_state().lock().water_level = boat.bounding_box().max_y();
267        return submerged;
268    }
269
270    if check_in_water(boat, world) {
271        return BoatStatus::InWater;
272    }
273
274    let friction = ground_friction(boat, world);
275    if friction > 0.0 {
276        boat.boat_state().lock().land_friction = friction;
277        return BoatStatus::OnLand;
278    }
279
280    BoatStatus::InAir
281}

283/// Returns the submerged status, if the boat's deck is under water.
284///
285/// Vanilla parity: `AbstractBoat.isUnderwater`.
286fn submerged_status<B: BoatLike>(boat: &B, world: &Arc<World>) -> Option<BoatStatus> {
287    let aabb = boat.bounding_box();
288    let deck = aabb.max_y() + 0.001;
289    let mut still = false;
290
291    for x in floor(aabb.min_x())..ceil(aabb.max_x()) {
292        for y in floor(aabb.max_y())..ceil(deck) {
293            for z in floor(aabb.min_z())..ceil(aabb.max_z()) {
294                let pos = BlockPos::new(x, y, z);
295                let fluid = world.get_block_state(pos).get_fluid_state();
296                if !fluid.is_water() {
297                    continue;
298                }
299                let surface = f64::from(y) + f64::from(get_height(world, pos, fluid));
300                if deck >= surface {
301                    continue;
302                }
303                if !fluid.is_source() {
304                    return Some(BoatStatus::UnderFlowingWater);
305                }
306                still = true;
307            }
308        }
309    }
310
311    still.then_some(BoatStatus::UnderWater)
312}

314/// Returns whether the hull is touching water, recording the surface height.
315///
316/// Vanilla parity: `AbstractBoat.checkInWater`.
317fn check_in_water<B: BoatLike>(boat: &B, world: &Arc<World>) -> bool {
318    let aabb = boat.bounding_box();
319    let mut in_water = false;
320    let mut water_level = f64::MIN;
321
322    for x in floor(aabb.min_x())..ceil(aabb.max_x()) {
323        for y in floor(aabb.min_y())..ceil(aabb.min_y() + 0.001) {
324            for z in floor(aabb.min_z())..ceil(aabb.max_z()) {
325                let pos = BlockPos::new(x, y, z);
326                let fluid = world.get_block_state(pos).get_fluid_state();
327                if !fluid.is_water() {
328                    continue;
329                }
330                let surface = f64::from(y) + f64::from(get_height(world, pos, fluid));
331                water_level = water_level.max(surface);
332                in_water |= aabb.min_y() < surface;
333            }
334        }
335    }
336
337    boat.boat_state().lock().water_level = water_level;
338    in_water
339}

341/// Returns the water surface just above the boat.
342///
343/// Vanilla parity: `AbstractBoat.getWaterLevelAbove`, which is what a falling
344/// boat lands on.
345fn water_level_above<B: BoatLike>(boat: &B, world: &Arc<World>) -> f32 {
346    let aabb = boat.bounding_box();
347    let last_yd = boat.boat_state().lock().last_yd;
348    let top = ceil(aabb.max_y() - last_yd);
349
350    for y in floor(aabb.max_y())..top {
351        let mut height = 0.0_f32;
352        for x in floor(aabb.min_x())..ceil(aabb.max_x()) {
353            for z in floor(aabb.min_z())..ceil(aabb.max_z()) {
354                let pos = BlockPos::new(x, y, z);
355                let fluid = world.get_block_state(pos).get_fluid_state();
356                if fluid.is_water() {
357                    height = height.max(get_height(world, pos, fluid));
358                }
359            }
360        }
361        if height < 1.0 {
362            #[expect(
363                clippy::cast_precision_loss,
364                reason = "a build height, well inside f32"
365            )]
366            return y as f32 + height;
367        }
368    }
369
370    #[expect(clippy::cast_precision_loss, reason = "a build height")]
371    let above = top as f32 + 1.0;
372    above
373}

375/// Returns the average friction of the blocks under the hull.
376///
377/// Vanilla parity: `AbstractBoat.getGroundFriction`, which is how a boat slides
378/// forever on ice and stops dead on grass.
379fn ground_friction<B: BoatLike>(boat: &B, world: &Arc<World>) -> f32 {
380    let aabb = boat.bounding_box();
381    let x0 = floor(aabb.min_x()) - 1;
382    let x1 = ceil(aabb.max_x()) + 1;
383    let y0 = floor(aabb.min_y() - 0.001) - 1;
384    let y1 = ceil(aabb.min_y()) + 1;
385    let z0 = floor(aabb.min_z()) - 1;
386    let z1 = ceil(aabb.max_z()) + 1;
387
388    let mut total = 0.0;
389    let mut count = 0;
390
391    for x in x0..x1 {
392        for z in z0..z1 {
393            let edges = i32::from(x == x0 || x == x1 - 1) + i32::from(z == z0 || z == z1 - 1);
394            if edges == 2 {
395                continue;
396            }
397            for y in y0..y1 {
398                if edges > 0 && (y == y0 || y == y1 - 1) {
399                    continue;
400                }
401                let pos = BlockPos::new(x, y, z);
402                let state = world.get_block_state(pos);
403                // Vanilla parity: a lily pad is not something a boat rests on,
404                // which is why a boat runs one over instead of grinding to a
405                // halt on it.
406                if state.get_block() == &vanilla_blocks::LILY_PAD || state.is_air() {
407                    continue;
408                }
409                total += state.get_block().config.friction;
410                count += 1;
411            }
412        }
413    }
414
415    if count == 0 {
416        0.0
417    } else {
418        #[expect(clippy::cast_precision_loss, reason = "a small block count")]
419        let average = total / count as f32;
420        average
421    }
422}
