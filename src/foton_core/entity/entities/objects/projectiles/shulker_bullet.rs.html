1//! Shulker bullet.
2//!
3//! Vanilla parity: `ShulkerBullet`. The odd one out of the fireball family: it
4//! is a plain `Projectile`, not an `AbstractHurtingProjectile`, and it does not
5//! fly straight. It picks one of the six axis directions, commits to it for a
6//! stretch, and picks again when it runs out of steps, when the block ahead of
7//! it is something it could stand on, or when it draws level with its target on
8//! the axis it is traveling. That is what makes it turn corners and follow a
9//! player around a pillar instead of thudding into the wall.
10//!
11//! Losing the target does not stop it: without one it simply falls, which is why
12//! a bullet whose shulker died drops out of the air.
13

44/// How fast a steering bullet travels.
45///
46/// Vanilla parity: `ShulkerBullet.SPEED`. The steering delta is normalized to
47/// this every time a new direction is chosen, and then eased toward each tick.
48const SPEED: f64 = 0.15;
49
50/// How far a bullet falls per tick once it has nothing to chase.
51///
52/// Vanilla parity: `ShulkerBullet.getDefaultGravity`.
53const DEFAULT_GRAVITY: f64 = 0.04;
54
55/// How much the steering delta grows each tick while a target is held.
56///
57/// Vanilla parity: the `* 1.025` of `ShulkerBullet.tick`. The bullet speeds up
58/// the longer it is locked on, which is what makes a long chase land.
59const STEERING_ACCELERATION: f64 = 1.025;
60
61/// Cap on each component of the steering delta.
62const MAX_STEERING_DELTA: f64 = 1.0;
63
64/// How much of the gap to the steering delta is closed each tick.
65///
66/// Vanilla parity: the `0.2` of the `setDeltaMovement` in `ShulkerBullet.tick`.
67/// Turning is eased rather than instant, which is what rounds off the corners.
68const STEERING_EASE: f64 = 0.2;
69
70/// How close the bullet has to be to its target to stop weaving toward it.
71///
72/// Vanilla parity: the `closerToCenterThan(this.position(), 2.0)` of
73/// `selectNextMoveDirection`. Inside two blocks the bullet aims straight at the
74/// target instead of picking another axis to travel down.
75const DIRECT_APPROACH_RANGE: f64 = 2.0;
76
77/// How many times a cornered bullet re-rolls for an unobstructed direction.
78///
79/// Vanilla parity: the `attempts = 5` loop of `selectNextMoveDirection`.
80const BLIND_DIRECTION_ATTEMPTS: i32 = 5;
81
82/// Damage a bullet does on contact.
83///
84/// Vanilla parity: the `4.0F` of `ShulkerBullet.onHitEntity`.
85const HIT_DAMAGE: f32 = 4.0;
86
87/// How long the levitation from a hit lasts, in ticks.
88///
89/// Vanilla parity: the `new MobEffectInstance(LEVITATION, 200)` of `onHitEntity`.
90const LEVITATION_TICKS: i32 = 200;
91
92/// How far the bullet turns toward its heading each tick.
93///
94/// Vanilla parity: the `rotateTowardsMovement(this, 0.5F)` of the tick loop.
95const ROTATION_SPEED: f32 = 0.5;

97/// State a bullet keeps that is not mirrored to clients.
98struct BulletState {
99    /// Who the bullet is chasing (vanilla `finalTarget`).
100    target: Option<Uuid>,
101    /// The axis direction it is currently traveling (vanilla `currentMoveDirection`).
102    move_direction: Option<Direction>,
103    /// Ticks left before it picks a new direction (vanilla `flightSteps`).
104    flight_steps: i32,
105    /// The velocity it is easing toward (vanilla `targetDeltaX/Y/Z`).
106    target_delta: DVec3,
107}

128        let base = EntityBase::new(id, position, entity_type.dimensions, world);
129        // Vanilla parity: the `ShulkerBullet` constructor sets `noPhysics`; the
130        // bullet steers itself around blocks rather than colliding with them.

132        Self {
133            base,
134            entity_type,
135            entity_data: SyncMutex::new(ShulkerBulletEntityData::new()),

139                move_direction: None,
140                flight_steps: 0,
141                target_delta: DVec3::ZERO,

143        }
144    }

165    /// Aims a freshly fired bullet at `target`.
166    ///
167    /// Vanilla parity: the `ShulkerBullet(level, owner, target, invalidStartAxis)`
168    /// constructor. The shulker passes the axis it is attached along so the
169    /// bullet never opens by flying into its own wall.
170    pub fn fire_at(
171        &self,
172        world: &Arc<World>,
173        owner: &SharedEntity,
174        target: &SharedEntity,
175        invalid_start_axis: Option<Axis>,
176    ) {
177        self.set_owner_entity(Some(owner));
178        {
179            let mut state = self.state.lock();
180            state.target = Some(target.uuid());
181            state.move_direction = Some(Direction::Up);
182        }
183        self.select_next_move_direction(world, invalid_start_axis, Some(target));
184    }
185
186    /// Returns the entity this bullet is chasing, if it is still around.
187    fn resolve_target(&self, world: &Arc<World>) -> Option<SharedEntity> {
188        let uuid = self.state.lock().target?;
189        world.get_entity_by_uuid(&uuid)
190    }

192    /// Returns whether the bullet could rest on the block at `pos`.
193    ///
194    /// Vanilla parity: `LevelReader.loadedAndEntityCanStandOn`. Foton checks the
195    /// full-face sturdiness of the top face, which is what vanilla's collision
196    /// shape test comes to for every block a bullet meets.
197    fn can_stand_on(world: &Arc<World>, pos: BlockPos) -> bool {
198        world.has_full_chunk(ChunkPos::from_block_pos(pos))
199            && world.is_face_sturdy(world.get_block_state(pos), pos, Direction::Up)
200    }
201
202    /// Returns whether the block at `pos` is open air.
203    ///
204    /// Vanilla parity: `Level.isEmptyBlock`.
205    fn is_empty_block(world: &Arc<World>, pos: BlockPos) -> bool {
206        world.get_block_state(pos).is_air()
207    }
208
209    /// Chooses the next leg of the bullet's flight.
210    ///
211    /// Vanilla parity: `ShulkerBullet.selectNextMoveDirection`. Far from the
212    /// target the bullet travels one axis at a time, preferring an unobstructed
213    /// axis that closes the gap and falling back to a blind re-roll when it is
214    /// boxed in; within two blocks it drops the weaving and aims straight.
215    fn select_next_move_direction(
216        &self,
217        world: &Arc<World>,
218        avoid_axis: Option<Axis>,
219        target: Option<&SharedEntity>,
220    ) {
221        let position = self.position();
222        let (y_offset, target_pos) = match target {
223            None => (0.5, self.block_position().below()),
224            Some(target) => {
225                let offset = target.bounding_box().height() * 0.5;
226                let target_position = target.position();
227                (
228                    offset,
229                    BlockPos::containing(
230                        target_position.x,
231                        target_position.y + offset,
232                        target_position.z,
233                    ),
234                )
235            }
236        };
237
238        let mut aim = DVec3::new(
239            f64::from(target_pos.x()) + 0.5,
240            f64::from(target_pos.y()) + y_offset,
241            f64::from(target_pos.z()) + 0.5,
242        );
243
244        let center = DVec3::new(
245            f64::from(target_pos.x()) + 0.5,
246            f64::from(target_pos.y()) + 0.5,
247            f64::from(target_pos.z()) + 0.5,
248        );
249        let mut selection = None;
250        if center.distance_squared(position) >= DIRECT_APPROACH_RANGE * DIRECT_APPROACH_RANGE {
251            let current = self.block_position();
252            let mut options = Vec::new();
253
254            if avoid_axis != Some(Axis::X) {
255                if current.x() < target_pos.x() && Self::is_empty_block(world, current.east()) {
256                    options.push(Direction::East);
257                } else if current.x() > target_pos.x()
258                    && Self::is_empty_block(world, current.west())
259                {
260                    options.push(Direction::West);
261                }
262            }
263            if avoid_axis != Some(Axis::Y) {
264                if current.y() < target_pos.y() && Self::is_empty_block(world, current.above()) {
265                    options.push(Direction::Up);
266                } else if current.y() > target_pos.y()
267                    && Self::is_empty_block(world, current.below())
268                {
269                    options.push(Direction::Down);
270                }
271            }
272            if avoid_axis != Some(Axis::Z) {
273                if current.z() < target_pos.z() && Self::is_empty_block(world, current.south()) {
274                    options.push(Direction::South);
275                } else if current.z() > target_pos.z()
276                    && Self::is_empty_block(world, current.north())
277                {
278                    options.push(Direction::North);
279                }
280            }
281
282            let chosen = if options.is_empty() {
283                let mut blind = Direction::random();
284                let mut attempts = BLIND_DIRECTION_ATTEMPTS;
285                while attempts > 0 && !Self::is_empty_block(world, current.relative(blind)) {
286                    blind = Direction::random();
287                    attempts -= 1;
288                }
289                blind
290            } else {
291                options[rand::random_range(0..options.len())]
292            };
293
294            let (step_x, step_y, step_z) = chosen.offset();
295            aim = position + DVec3::new(f64::from(step_x), f64::from(step_y), f64::from(step_z));
296            selection = Some(chosen);
297        }
298
299        let delta = aim - position;
300        let distance = delta.length();
301        let target_delta = if distance == 0.0 {
302            DVec3::ZERO
303        } else {
304            delta / distance * SPEED
305        };
306
307        let mut state = self.state.lock();
308        state.move_direction = selection;
309        state.target_delta = target_delta;
310        // Vanilla picks 10, 20, 30, 40 or 50 steps, so the leg lengths vary and
311        // two bullets fired at once do not fly in lockstep.
312        state.flight_steps = 10 + rand::random_range(0..5) * 10;
313        drop(state);
314        self.mark_velocity_sync();
315    }
316
317    /// Eases the bullet's velocity toward its steering delta.
318    ///
319    /// Vanilla parity: the target branch of `ShulkerBullet.tick`.
320    fn steer(&self) {
321        let mut state = self.state.lock();
322        state.target_delta = (state.target_delta * STEERING_ACCELERATION).clamp(
323            DVec3::splat(-MAX_STEERING_DELTA),
324            DVec3::splat(MAX_STEERING_DELTA),
325        );
326        let target_delta = state.target_delta;
327        drop(state);
328
329        let movement = self.velocity();
330        self.set_velocity(movement + (target_delta - movement) * STEERING_EASE);
331    }
332
333    /// Re-picks the direction when the current leg has run its course.
334    ///
335    /// Vanilla parity: the tail of `ShulkerBullet.tick`, which turns the bullet
336    /// when its step budget runs out, when it is about to fly into something it
337    /// could stand on, or when it has drawn level with the target on the axis it
338    /// is traveling and has nothing more to gain from this leg.
339    fn advance_flight_plan(&self, world: &Arc<World>, target: &SharedEntity) {
340        let (steps, direction) = {
341            let mut state = self.state.lock();
342            if state.flight_steps > 0 {
343                state.flight_steps -= 1;
344            }
345            (state.flight_steps, state.move_direction)
346        };
347
348        if steps == 0 {
349            let axis = direction.map(Direction::get_axis);
350            self.select_next_move_direction(world, axis, Some(target));
351            return;
352        }
353
354        let Some(direction) = self.state.lock().move_direction else {
355            return;
356        };
357        let current = self.block_position();
358        let axis = direction.get_axis();
359        if Self::can_stand_on(world, current.relative(direction)) {
360            self.select_next_move_direction(world, Some(axis), Some(target));
361            return;
362        }
363
364        let target_pos = target.block_position();
365        let level_with_target = match axis {
366            Axis::X => current.x() == target_pos.x(),
367            Axis::Y => current.y() == target_pos.y(),
368            Axis::Z => current.z() == target_pos.z(),
369        };
370        if level_with_target {
371            self.select_next_move_direction(world, Some(axis), Some(target));
372        }
373    }
374
375    /// Removes the bullet and reports the impact.
376    ///
377    /// Vanilla parity: `ShulkerBullet.destroy`.
378    fn destroy(&self) {
379        self.set_removed(RemovalReason::Discarded);
380        if let Some(world) = self.level() {
381            world.game_event_at(
382                &vanilla_game_events::ENTITY_DAMAGE,
383                self.position(),
384                &GameEventContext::new(Some(self.as_entity_event_source()), None),
385            );
386        }
387    }

453    /// Vanilla parity: `ShulkerBullet.tick`.
454    fn tick(&self) {
455        // Vanilla's level captures `Entity.setOldPosAndRot()` before ticking.
456        self.set_old_position_to_current();
457        self.base().set_old_rotation_to_current();
458        self.projectile_base_tick();
459
460        let Some(world) = self.level() else {
461            return;
462        };
463
464        let target = self.resolve_target(&world);
465        if target.is_none() {
466            self.state.lock().target = None;
467        }
468
469        // Vanilla parity: a dead target, or one that has gone into spectator, is
470        // treated as no target at all -- the bullet drops instead of chasing.
471        let chase = target
472            .as_ref()
473            .filter(|target| Entity::is_alive(target.as_ref()) && !target.is_spectator());
474        if chase.is_some() {
475            self.steer();
476        } else {
477            self.apply_gravity();
478        }
479
480        let hit = self.get_hit_result_on_move_vector();
481
482        let movement = self.velocity();
483        if let Err(error) = self.try_set_position(self.position() + movement) {
484            log::debug!("failed to advance shulker bullet {}: {error}", self.id());
485            self.set_removed(RemovalReason::Discarded);
486            return;
487        }
488        self.apply_effects_from_blocks();
489
490        // TODO: vanilla also runs `handlePortal` here; Foton's projectiles do
491        // their portal handling in the shared entity tick.
492
493        if let Some(result) = hit
494            && self.is_alive()
495            && !self.is_world_change_pending()
496        {
497            self.hit_target_or_deflect_self(&result);
498        }
499
500        rotate_towards_movement(self, ROTATION_SPEED);
501
502        // VANILLA CLIENT-LOCAL: the end-rod trail behind the bullet.
503
504        if let Some(target) = target
505            && self.is_alive()
506        {
507            self.advance_flight_plan(&world, &target);
508        }
509    }
510
511    fn save_additional(&self, nbt: &mut NbtCompound) {
512        self.save_projectile(nbt);
513        let state = self.state.lock();
514        if let Some(target) = state.target {
515            nbt.insert("Target", NbtTag::IntArray(target.to_int_array().to_vec()));
516        }
517        if let Some(direction) = state.move_direction {
518            nbt.insert("Dir", direction.get_3d_data_value() as i8);
519        }
520        nbt.insert("Steps", state.flight_steps);
521        nbt.insert("TXD", state.target_delta.x);
522        nbt.insert("TYD", state.target_delta.y);
523        nbt.insert("TZD", state.target_delta.z);
524    }
525
526    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
527        self.load_projectile(nbt);
528        let mut state = self.state.lock();
529        state.flight_steps = nbt.int("Steps").unwrap_or(0);
530        state.target_delta = DVec3::new(
531            nbt.double("TXD").unwrap_or(0.0),
532            nbt.double("TYD").unwrap_or(0.0),
533            nbt.double("TZD").unwrap_or(0.0),
534        );
535        state.move_direction = nbt
536            .byte("Dir")
537            .map(|value| Direction::from_3d_data_value(i32::from(value)));
538        state.target = nbt
539            .int_array("Target")
540            .and_then(|array| Uuid::from_int_array(&array));
541    }
542}

544impl Projectile for ShulkerBulletEntity {

549    /// Vanilla parity: `ShulkerBullet.onHitEntity`.
550    fn on_hit_entity(&self, entity: &SharedEntity, _location: DVec3) {
551        let Some(world) = self.level() else {
552            return;
553        };
554
555        let owner = self.get_owner();
556        let mut source = DamageSource::environment(&vanilla_damage_types::MOB_PROJECTILE)
557            .with_direct_entity(self.id());
558        if let Some(owner) = owner
559            .as_ref()
560            .filter(|owner| owner.as_living_entity().is_some())
561        {
562            source = source.with_causing_entity(owner.id());
563        }
564
565        if entity.hurt(&world, &source, HIT_DAMAGE)
566            && let Some(living) = entity.as_living_entity()
567        {
568            living.add_mob_effect(MobEffectInstance::with_duration(
569                vanilla_mob_effects::LEVITATION,
570                LEVITATION_TICKS,
571                0,
572            ));
573        }
574
575        // TODO: vanilla also runs `EnchantmentHelper.doPostAttackEffects`; Foton
576        // has no post-attack enchantment dispatch for projectiles yet.
577    }
578
579    /// Vanilla parity: `ShulkerBullet.onHitBlock`.
580    fn on_hit_block(&self, hit: &ClipHitResult) {
581        self.projectile_on_hit_block(hit);
582        if let Some(world) = self.level() {
583            world.send_particles(
584                ParticleData::simple(&vanilla_particle_types::EXPLOSION),
585                self.position(),
586                2,
587                DVec3::splat(0.2),
588                0.0,
589            );
590        }
591        self.play_sound(&sound_events::ENTITY_SHULKER_BULLET_HIT, 1.0, 1.0);
592    }
593
594    /// Vanilla parity: `ShulkerBullet.onHit`.
595    fn on_hit(&self, hit: &ProjectileHit) {
596        self.projectile_on_hit(hit);
597        self.destroy();
598    }
