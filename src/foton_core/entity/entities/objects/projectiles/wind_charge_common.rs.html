1//! What the two wind charges share.
2//!
3//! Vanilla parity: `AbstractWindCharge`, the base a hand-thrown `WindCharge`
4//! and a breeze's `BreezeWindCharge` sit on. Both fly dead straight -- no
5//! gravity, no drag, no acceleration -- and burst on the first thing they
6//! touch, shoving whatever stands nearby without breaking a block or lighting
7//! a fire. Only the size of that burst, how hard it shoves and the noise it
8//! makes differ, so those three are what [`AbstractWindCharge`] asks a concrete
9//! charge for.
10

27/// What a direct hit costs the entity it lands on.
28///
29/// Vanilla parity: the `1.0F` of `AbstractWindCharge.onHitEntity`. In vanilla
30/// this is the only damage a wind charge deals; the burst itself deals none.
31const IMPACT_DAMAGE: f32 = 1.0;
32
33/// How far off the struck face the burst is centered.
34///
35/// Vanilla parity: the `multiply(0.25, 0.25, 0.25)` applied to the hit normal
36/// in `AbstractWindCharge.onHitBlock`, which lifts the burst out of the block
37/// so it pushes along the surface rather than from inside it.
38const BLOCK_HIT_NORMAL_OFFSET: f64 = 0.25;
39
40/// How far above the world a wind charge may climb before it bursts on its own.
41///
42/// Vanilla parity: the `getBlockY() > getMaxY() + 30` of
43/// `AbstractWindCharge.tick`.
44pub(super) const MAX_Y_OVERSHOOT: i32 = 30;
45
46/// Vanilla parity: the `0.1` an `AbstractHurtingProjectile` is born with and
47/// that `onDeflection` restores after a deflecting attack. Every
48/// `AbstractWindCharge` constructor zeroes it, so a wind charge carries it only
49/// once something has knocked the charge off course.
50pub(super) const DEFLECTED_ACCELERATION_POWER: f64 = 0.1;
51
52/// Vanilla parity: `AbstractHurtingProjectile.DEFLECTION_SCALE`, the halving
53/// `onDeflection` applies to a deflection that was not an attack.
54const PASSIVE_DEFLECTION_SCALE: f64 = 0.5;
55
56/// Vanilla parity: `AbstractWindCharge.getInertia`, which returns 1 where every
57/// other hurting projectile returns 0.95. `getLiquidInertia` returns the same
58/// value, so water does not slow a wind charge either.
59const INERTIA: f64 = 1.0;
60
61/// The one field `AbstractWindCharge` keeps that clients never see.
62pub(super) struct AbstractWindChargeBase {
63    /// Vanilla parity: `AbstractHurtingProjectile.accelerationPower`, which
64    /// every `AbstractWindCharge` constructor zeroes and only a deflection
65    /// brings back.
66    acceleration_power: SyncMutex<f64>,
67}
68
69impl AbstractWindChargeBase {
70    /// Creates the base of a charge that has not been deflected.
71    #[must_use]
72    pub(super) const fn new() -> Self {
73        Self {
74            // Vanilla parity: the `this.accelerationPower = 0.0` every
75            // `AbstractWindCharge` constructor ends with.
76            acceleration_power: SyncMutex::new(0.0),
77        }
78    }
79}

81/// Returns the velocity a wind charge carries into the next tick.
82///
83/// Vanilla parity: `AbstractHurtingProjectile.applyInertia`. With the wind
84/// charge's acceleration power of zero and its inertia of one this is the
85/// identity, which is exactly why a wind charge flies in a straight line at a
86/// constant speed; a deflected charge picks acceleration back up and the first
87/// term starts to matter.
88#[must_use]
89pub(super) fn inertia_applied(velocity: DVec3, acceleration_power: f64) -> DVec3 {
90    (velocity + velocity.normalize_or_zero() * acceleration_power) * INERTIA
91}
92
93/// Returns where a burst set off by a block hit is centered.
94///
95/// Vanilla parity: the `hitResult.getLocation().add(normal * 0.25)` of
96/// `AbstractWindCharge.onHitBlock`.
97#[must_use]
98pub(super) fn block_burst_center(location: DVec3, face_normal: DVec3) -> DVec3 {
99    location + face_normal * BLOCK_HIT_NORMAL_OFFSET
100}

102/// Returns whether `entity` is a wind charge of either kind.
103///
104/// Vanilla parity: the `instanceof AbstractWindCharge` of
105/// `AbstractWindCharge.canCollideWith` and `canHitEntity`. One wind charge must
106/// not shoot another down, whoever threw either of them.
107#[must_use]
108pub(super) fn is_wind_charge(entity: &dyn Entity) -> bool {
109    entity.entity_type() == &vanilla_entities::WIND_CHARGE
110        || entity.entity_type() == &vanilla_entities::BREEZE_WIND_CHARGE
111}

113/// A wind charge of either kind.
114///
115/// Vanilla parity: `net.minecraft.world.entity.projectile.hurtingprojectile.windcharge.AbstractWindCharge`.
116pub(super) trait AbstractWindCharge: Projectile {
117    /// Returns the acceleration a deflection has left this charge with.
118    fn wind_charge_base(&self) -> &AbstractWindChargeBase;
119
120    /// Returns how far this charge's burst reaches.
121    ///
122    /// Vanilla parity: the radius each subclass hands `Level.explode`.
123    fn burst_radius(&self) -> f32;
124
125    /// Returns how much harder than an ordinary blast this charge shoves.
126    ///
127    /// Vanilla parity: the `knockbackMultiplier` of the subclass's
128    /// `SimpleExplosionDamageCalculator`. `Optional.empty()` there means the
129    /// `ExplosionDamageCalculator.getKnockbackMultiplier` default of one.
130    fn burst_knockback(&self) -> f64;
131
132    /// Returns the noise the burst makes.
133    fn burst_sound(&self) -> SoundEventRef;
134
135    /// Vanilla parity: `AbstractHurtingProjectile.accelerationPower`.
136    fn acceleration_power(&self) -> f64 {
137        *self.wind_charge_base().acceleration_power.lock()
138    }
139
140    /// Writes the acceleration a deflection or a load has produced.
141    fn set_acceleration_power(&self, acceleration_power: f64) {
142        *self.wind_charge_base().acceleration_power.lock() = acceleration_power;
143    }
144
145    /// Bursts at `position`.
146    ///
147    /// Vanilla parity: the `explode` each subclass overrides, which calls
148    /// `Level.explode` with the subclass's radius, no fire,
149    /// `ExplosionInteraction.TRIGGER` and a
150    /// `SimpleExplosionDamageCalculator(explodesBlocks = true,
151    /// damagesEntities = false, immuneBlocks = #blocks_wind_charge_explosions)`.
152    ///
153    /// One part of that is still out of reach and is worth naming rather than
154    /// approximating in silence: Foton has no
155    /// `Explosion.BlockInteraction.TRIGGER_BLOCK`. [`ExplosionBlockInteraction`]
156    /// offers only `Keep` and `Destroy`, so `Keep` is used -- blocks survive,
157    /// which is the half that matters, but the per-block
158    /// `BlockState.onExplosionHit` that lets a vanilla wind charge slam a door,
159    /// flip a trapdoor or ring a bell never runs. That block callback is the
160    /// missing system.
161    ///
162    /// The `#blocks_wind_charge_explosions` immunity is moot while every block
163    /// is kept. The burst sound and the gust emitter both travel inside the
164    /// explosion packet, which is why they are named here rather than played
165    /// separately: a wind charge is the one blast whose presentation is not
166    /// vanilla's default.
167    fn explode(&self, world: &Arc<World>, position: DVec3) {
168        world.explode_sparing(
169            ExplosionSpec {
170                direct_entity_id: Some(self.id()),
171                causing_entity_id: self.get_owner().map(|owner| owner.id()),
172                // Vanilla parity: both `explode` overrides pass a null damage source.
173                damage_source: None,
174                radius: self.burst_radius(),
175                fire: false,
176                interaction: ExplosionBlockInteraction::Keep,
177                // This is the whole point of a wind charge: it shoves what it
178                // reaches and hurts none of it.
179                damages_entities: false,
180                knockback_multiplier: self.burst_knockback(),
181                small_particle: ParticleData::simple(&vanilla_particle_types::GUST_EMITTER_SMALL),
182                large_particle: ParticleData::simple(&vanilla_particle_types::GUST_EMITTER_LARGE),
183                // Vanilla parity: `WeightedList.of()`. A wind charge breaks
184                // nothing, so it has no debris to throw.
185                block_particles: WeightedList::empty(),
186                sound: self.burst_sound(),
187            },
188            position,
189            &|_pos| true,
190        );
191    }
192
193    /// Runs one tick of flight.
194    ///
195    /// Vanilla parity: `AbstractHurtingProjectile.tick`, minus the parts the
196    /// wind charge switches off -- it never burns (`shouldBurn` is false) and
197    /// leaves no trail particle (`getTrailParticle` is null).
198    ///
199    /// Vanilla also discards the projectile when its owner has been removed or
200    /// its block position is in an unloaded chunk. Neither half is reachable
201    /// here: [`Projectile::get_owner`] cannot tell "no owner" from "owner
202    /// removed" -- both come back as `None`, which vanilla reads as reason to
203    /// keep flying -- and Foton's entity manager only ticks entities in loaded
204    /// chunks.
205    fn hurting_projectile_tick(&self) {
206        // Vanilla `Entity.setOldPosAndRot()` runs before the tick; capture it
207        // here so the rotation has a base to unwind against.
208        self.set_old_position_to_current();
209        self.base().set_old_rotation_to_current();
210
211        self.set_velocity(inertia_applied(self.velocity(), self.acceleration_power()));
212
213        let hit = self.get_hit_result_on_move_vector();
214        let new_position = match &hit {
215            Some(result) => result.location(),
216            None => self.position() + self.velocity(),
217        };
218
219        if let Err(error) = self.try_set_position(new_position) {
220            log::debug!("failed to advance wind charge {}: {error}", self.id());
221            self.set_removed(RemovalReason::Discarded);
222            return;
223        }
224
225        self.rotate_towards_movement();
226        self.apply_effects_from_blocks();
227        self.projectile_base_tick();
228
229        if let Some(result) = hit
230            && self.is_alive()
231            && !self.is_world_change_pending()
232        {
233            self.hit_target_or_deflect_self(&result);
234        }
235    }
236
237    /// Points the charge along its velocity.
238    ///
239    /// Vanilla parity: `ProjectileUtil.rotateTowardsMovement(this, 0.2F)`, which
240    /// snaps the rotation onto the movement vector rather than lerping toward it
241    /// the way `Projectile.updateRotation` does. The `0.2F` vanilla passes is
242    /// unused by that method, and the old-rotation unwinding that follows it
243    /// only keeps client-side interpolation from taking the long way round, so
244    /// neither is reproduced.
245    fn rotate_towards_movement(&self) {
246        let movement = self.velocity();
247        if movement.length_squared() == 0.0 {
248            return;
249        }
250        let horizontal = movement.x.hypot(movement.z);
251        #[expect(
252            clippy::cast_possible_truncation,
253            reason = "an angle in degrees, immediately used as a rotation"
254        )]
255        let yaw = movement.z.atan2(movement.x).to_degrees() as f32 + 90.0;
256        #[expect(
257            clippy::cast_possible_truncation,
258            reason = "an angle in degrees, immediately used as a rotation"
259        )]
260        let pitch = horizontal.atan2(movement.y).to_degrees() as f32 - 90.0;
261        self.set_rotation((yaw, pitch));
262    }
263
264    /// Vanilla parity: `AbstractWindCharge.tick`, which bursts a charge that has
265    /// climbed far enough above the world rather than letting it fly forever.
266    fn abstract_wind_charge_tick(&self) {
267        let Some(world) = self.level() else {
268            self.hurting_projectile_tick();
269            return;
270        };
271        if self.block_position().y() > world.get_max_y() + MAX_Y_OVERSHOOT {
272            self.explode(&world, self.position());
273            self.set_removed(RemovalReason::Discarded);
274        } else {
275            self.hurting_projectile_tick();
276        }
277    }

279    /// Vanilla parity: `AbstractWindCharge.canCollideWith`.
280    fn wind_charge_can_collide_with(&self, other: &dyn Entity) -> bool {
281        !is_wind_charge(other)
282            && other.can_be_collided_with(Some(self.as_entity_event_source()))
283            && !self.is_passenger_of_same_vehicle(other)
284    }

286    /// Vanilla parity: `AbstractWindCharge.canHitEntity`. An end crystal is
287    /// spared, so a charge cannot be used to pop one from a distance.
288    ///
289    /// Foton reaches this test through `Entity::can_be_hit_by_projectile`,
290    /// which no living entity answers yes to yet -- `Entity::is_pickable` is
291    /// still false for every mob -- so today a wind charge can only score a
292    /// direct hit on a player. Mobs are reached by the burst alone. That is a
293    /// gap in Foton's living entities rather than in the charge.
294    fn wind_charge_can_hit_entity(&self, entity: &dyn Entity) -> bool {
295        if is_wind_charge(entity) || entity.entity_type() == &vanilla_entities::END_CRYSTAL {
296            return false;
297        }
298        self.projectile_can_hit_entity(entity)
299    }

301    /// Vanilla parity: `AbstractHurtingProjectile.onDeflection`.
302    fn wind_charge_on_deflection(&self, by_attack: bool) {
303        if by_attack {
304            self.set_acceleration_power(DEFLECTED_ACCELERATION_POWER);
305        } else {
306            self.set_acceleration_power(self.acceleration_power() * PASSIVE_DEFLECTION_SCALE);
307        }
308    }
309
310    /// Vanilla parity: `AbstractWindCharge.onHitEntity`. One point of damage to
311    /// what it struck, then the burst at the charge's own position.
312    fn wind_charge_on_hit_entity(&self, entity: &SharedEntity) {
313        let owner = self.get_owner();
314        let mut source = DamageSource::environment(&vanilla_damage_types::WIND_CHARGE)
315            .with_direct_entity(self.id());
316        if let Some(owner) = &owner {
317            source = source.with_causing_entity(owner.id());
318            if let Some(living) = owner.as_living_entity() {
319                living.set_last_hurt_mob(Some(entity));
320            }
321        }
322
323        let Some(world) = self.level() else {
324            return;
325        };
326        // TODO: vanilla runs `EnchantmentHelper.doPostAttackEffects` on a living
327        // target the hit actually damaged; Foton has no projectile enchantment
328        // dispatch yet.
329        entity.hurt(&world, &source, IMPACT_DAMAGE);
330        self.explode(&world, self.position());
331    }
332
333    /// Vanilla parity: `AbstractWindCharge.onHitBlock`, which nudges the burst
334    /// out of the face it struck before setting it off.
335    fn wind_charge_on_hit_block(&self, hit: &ClipHitResult) {
336        self.projectile_on_hit_block(hit);
337        let Some(world) = self.level() else {
338            return;
339        };
340        let normal = hit.direction.offset_vec();
341        let center = block_burst_center(
342            hit.location,
343            DVec3::new(
344                f64::from(normal.x),
345                f64::from(normal.y),
346                f64::from(normal.z),
347            ),
348        );
349        self.explode(&world, center);
350        self.set_removed(RemovalReason::Discarded);
351    }

353    /// Vanilla parity: `AbstractHurtingProjectile.addAdditionalSaveData`.
354    fn save_wind_charge(&self, nbt: &mut NbtCompound) {
355        self.save_projectile(nbt);
356        nbt.insert("acceleration_power", self.acceleration_power());
357    }
358
359    /// Vanilla parity: `AbstractHurtingProjectile.readAdditionalSaveData`,
360    /// which falls back to the class default of 0.1 rather than to the zero an
361    /// unflicked wind charge carries.
362    fn load_wind_charge(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
363        self.load_projectile(nbt);
364        self.set_acceleration_power(
365            nbt.double("acceleration_power")
366                .unwrap_or(DEFLECTED_ACCELERATION_POWER),
367        );
368    }
