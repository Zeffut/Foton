1//! A breeze's wind charge.
2//!
3//! Vanilla parity: `BreezeWindCharge`. The same projectile a player throws by
4//! hand, with a burst two and a half times as wide and no extra shove behind
5//! it: `WindCharge` passes a `1.22F` knockback multiplier, this one passes the
6//! `AbstractWindCharge` calculator whose multiplier is empty and so defaults to
7//! one. It also has no deflection grace period -- a breeze fires from a
8//! distance, so there is no point-blank throw to protect against.
9//!
10//! Everything else is [`super::wind_charge_common`].
11

34/// Vanilla parity: `BreezeWindCharge.RADIUS`, the `3.0F` handed to
35/// `Level.explode`.
36const BURST_RADIUS: f32 = 3.0;
37
38/// Vanilla parity: the empty `knockbackMultiplier` of
39/// `AbstractWindCharge.EXPLOSION_DAMAGE_CALCULATOR`, which
40/// `SimpleExplosionDamageCalculator.getKnockbackMultiplier` resolves to the
41/// `ExplosionDamageCalculator` default of one.
42const BURST_KNOCKBACK: f64 = 1.0;
43
44/// A wind charge a breeze fired.
45#[entity_behavior(class = "BreezeWindCharge")]
46pub struct BreezeWindChargeEntity {
47    base: EntityBase,
48    entity_type: EntityTypeRef,
49    entity_data: SyncMutex<BreezeWindChargeEntityData>,
50    projectile_base: ProjectileBase,
51    wind_charge_base: AbstractWindChargeBase,
52}

190mod tests {

202    /// Vanilla gives the breeze's charge a three-block burst against the
203    /// player's 1.2, and takes away the 1.22 knockback multiplier that makes a
204    /// hand-thrown charge shove harder than an ordinary blast. Swapping either
205    /// of the two would turn a breeze fight into a very different fight.
206    #[test]
207    fn a_breeze_charge_bursts_wider_and_shoves_no_harder_than_an_ordinary_blast() {
208        init_vanilla_registry();
209        let breeze_charge = BreezeWindChargeEntity::new(
210            &vanilla_entities::BREEZE_WIND_CHARGE,
211            next_entity_id(),
212            DVec3::ZERO,
213            Weak::<World>::new(),
214        );
215        let thrown = WindChargeEntity::new(
216            &vanilla_entities::WIND_CHARGE,
217            next_entity_id(),
218            DVec3::ZERO,
219            Weak::<World>::new(),
220        );
221
222        assert!(breeze_charge.burst_radius() > thrown.burst_radius());
223        assert!((breeze_charge.burst_radius() - 3.0).abs() < f32::EPSILON);
224        assert!((breeze_charge.burst_knockback() - 1.0).abs() < f64::EPSILON);
225        assert!(breeze_charge.burst_knockback() < thrown.burst_knockback());
226    }
227
228    /// A breeze standing in its own gust would otherwise shoot its own charges
229    /// down, and two breezes in a trial chamber would shoot each other's.
230    #[test]
231    fn a_breeze_charge_will_not_shoot_a_hand_thrown_charge_out_of_the_air() {
232        init_vanilla_registry();
233        let breeze_charge = BreezeWindChargeEntity::new(
234            &vanilla_entities::BREEZE_WIND_CHARGE,
235            next_entity_id(),
236            DVec3::ZERO,
237            Weak::<World>::new(),
238        );
239        let thrown = WindChargeEntity::new(
240            &vanilla_entities::WIND_CHARGE,
241            next_entity_id(),
242            DVec3::ZERO,
243            Weak::<World>::new(),
244        );
245
246        assert!(!breeze_charge.can_hit_entity(&thrown));
247    }

249    /// Unlike the hand-thrown charge, a breeze's has no `noDeflectTicks`: a
250    /// shield or a sword swing turns it around on the tick it arrives.
251    #[test]
252    fn a_breeze_charge_can_be_deflected_the_moment_it_is_fired() {
253        init_vanilla_registry();
254        let charge = BreezeWindChargeEntity::new(
255            &vanilla_entities::BREEZE_WIND_CHARGE,
256            next_entity_id(),
257            DVec3::ZERO,
258            Weak::<World>::new(),
259        );
260        charge.set_velocity(DVec3::new(0.0, 0.0, 1.0));
261
262        assert!(charge.deflect(ProjectileDeflection::Reverse, None, None, None, true));
263    }
264
265    /// The burst is the breeze's whole attack: it has to move a bystander and
266    /// leave its health alone.
267    #[test]
268    fn the_burst_shoves_a_bystander_without_hurting_it() {
269        init_vanilla_registry();
270        init_behaviors();
271        let world = fresh_test_world("breeze_wind_charge_burst");
272        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
273
274        let pig: SharedEntity = Arc::new(PigEntity::new(
275            &vanilla_entities::PIG,
276            next_entity_id(),
277            DVec3::new(9.5, 64.0, 8.5),
278            Arc::downgrade(&world),
279        ));
280        world
281            .try_add_entity(Arc::clone(&pig))
282            .expect("pig should attach to the loaded test chunk");
283        let health_before = pig
284            .as_living_entity()
285            .expect("a pig is a living entity")
286            .get_health();
287
288        let charge = BreezeWindChargeEntity::new(
289            &vanilla_entities::BREEZE_WIND_CHARGE,
290            next_entity_id(),
291            DVec3::new(8.5, 64.0, 8.5),
292            Arc::downgrade(&world),
293        );
294        charge.explode(&world, DVec3::new(8.5, 64.0, 8.5));
295
296        let living = pig.as_living_entity().expect("a pig is a living entity");
297        assert!(
298            (living.get_health() - health_before).abs() < f32::EPSILON,
299            "a breeze's burst took health off a bystander"
300        );
301        assert!(
302            pig.velocity().length_squared() > 0.0,
303            "a breeze's burst did not move the bystander at all"
304        );
305    }
