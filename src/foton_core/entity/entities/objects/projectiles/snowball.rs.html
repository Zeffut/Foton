1//! Thrown snowball.
2//!
3//! Vanilla parity: `Snowball`. The simplest thing on the
4//! `Projectile → ThrowableProjectile → ThrowableItemProjectile` stack: it flies,
5//! it hits, it breaks. The only thing it hurts is a blaze, and even then only
6//! for three -- against everything else it deals zero damage and exists to
7//! knock things about and to trip a target block.
8

31/// What a snowball does to a blaze.
32///
33/// Vanilla parity: the `entity instanceof Blaze ? 3 : 0` of
34/// `Snowball.onHitEntity`.
35const BLAZE_DAMAGE: f32 = 3.0;
36
37/// A thrown snowball.
38#[entity_behavior(class = "Snowball")]
39pub struct SnowballEntity {
40    base: EntityBase,
41    entity_type: EntityTypeRef,
42    entity_data: SyncMutex<SnowballEntityData>,
43    projectile_base: ProjectileBase,
44}
45
46// SAFETY: This key is owned by Foton and uniquely identifies `SnowballEntity`.
47unsafe impl DowncastType for SnowballEntity {
48    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:entity/snowball");
49}

51impl SnowballEntity {
52    /// Creates a snowball.
53    #[must_use]
54    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
55        Self {
56            base: EntityBase::new(id, position, entity_type.dimensions, world),
57            entity_type,
58            entity_data: SyncMutex::new(SnowballEntityData::new()),
59            projectile_base: ProjectileBase::new(),
60        }
61    }
62
63    /// Creates a snowball from saved base data.
64    #[must_use]
65    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
66        Self {
67            base: EntityBase::from_load(load, entity_type.dimensions),
68            entity_type,
69            entity_data: SyncMutex::new(SnowballEntityData::new()),
70            projectile_base: ProjectileBase::new(),
71        }
72    }
73}

97impl Projectile for SnowballEntity {
98    fn projectile_base(&self) -> &ProjectileBase {
99        &self.projectile_base
100    }
101
102    /// Vanilla parity: `Snowball.onHitEntity`.
103    fn on_hit_entity(&self, entity: &SharedEntity, _location: DVec3) {
104        let mut damage =
105            DamageSource::environment(&vanilla_damage_types::THROWN).with_direct_entity(self.id());
106        if let Some(owner) = self.get_owner() {
107            damage = damage.with_causing_entity(owner.id());
108        }
109
110        let amount = if entity.entity_type() == &vanilla_entities::BLAZE {
111            BLAZE_DAMAGE
112        } else {
113            0.0
114        };
115        if let Some(world) = entity.level() {
116            entity.hurt(&world, &damage, amount);
117        }
118    }
119
120    /// Vanilla parity: `Snowball.onHit`, which breaks the snowball wherever it
121    /// landed. The entity event is what puts the puff of particles on screen.
122    fn on_hit(&self, hit: &ProjectileHit) {
123        self.projectile_on_hit(hit);
124        if self.is_removed() {
125            return;
126        }
127        self.broadcast_entity_event(EntityStatus::Death);
128        self.set_removed(RemovalReason::Discarded);
129    }
130}

142            .throwable_item_projectile

147    fn get_item(&self) -> ItemStack {
148        self.entity_data
149            .lock()
150            .throwable_item_projectile
151            .item_stack
152            .get()
153            .clone()
154    }
