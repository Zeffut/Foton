1//! Wither skull.
2//!
3//! Vanilla parity: `WitherSkull`, on `AbstractHurtingProjectile`. It comes in
4//! two forms. The plain skull is what the wither's side heads spit; the
5//! dangerous one is the blue skull the middle head charges up, and it differs in
6//! two ways: it keeps less of its speed each tick, so it travels slower and
7//! visibly homes less, and it eats through blocks a plain skull cannot.
8//!
9//! Both forms wither what they hit, and a kill heals the wither that threw it,
10//! which is the whole reason the fight has a healing phase.
11

34/// Velocity a charged skull keeps each tick.
35///
36/// Vanilla parity: `WitherSkull.getInertia` for a dangerous skull. Lower than
37/// the 0.95 of everything else on this branch, so the blue skull crawls.
38const DANGEROUS_INERTIA: f64 = 0.73;
39
40/// Damage a skull thrown by a living shooter does.
41///
42/// Vanilla parity: the `8.0F` of `WitherSkull.onHitEntity`.
43const OWNED_DAMAGE: f32 = 8.0;
44
45/// Damage an ownerless skull does.
46///
47/// Vanilla parity: the `5.0F` magic fallback of `WitherSkull.onHitEntity`, used
48/// when the wither that threw it is gone.
49const UNOWNED_DAMAGE: f32 = 5.0;
50
51/// Health the shooter gets back for a kill.
52///
53/// Vanilla parity: the `livingOwner.heal(5.0F)` of `WitherSkull.onHitEntity`.
54const KILL_HEAL: f32 = 5.0;
55
56/// Seconds of wither a skull inflicts on normal difficulty.
57const NORMAL_WITHER_SECONDS: i32 = 10;
58
59/// Seconds of wither a skull inflicts on hard difficulty.
60const HARD_WITHER_SECONDS: i32 = 40;
61
62/// Amplifier of the wither effect a skull inflicts.
63///
64/// Vanilla parity: the `1` of the `MobEffectInstance` in `onHitEntity`.
65const WITHER_AMPLIFIER: i32 = 1;
66
67/// Radius of the blast a skull leaves.
68///
69/// Vanilla parity: the `1.0F` of `WitherSkull.onHit`, the same for both forms.
70const EXPLOSION_RADIUS: f32 = 1.0;
71
72/// A wither's skull.
73#[entity_behavior(class = "WitherSkull")]
74pub struct WitherSkullEntity {
75    base: EntityBase,
76    entity_type: EntityTypeRef,
77    entity_data: SyncMutex<WitherSkullEntityData>,
78    projectile_base: ProjectileBase,
79    hurting_projectile_base: HurtingProjectileBase,
80}

87impl WitherSkullEntity {
88    /// Creates a plain wither skull.
89    #[must_use]
90    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
91        Self {
92            base: EntityBase::new(id, position, entity_type.dimensions, world),
93            entity_type,
94            entity_data: SyncMutex::new(WitherSkullEntityData::new()),
95            projectile_base: ProjectileBase::new(),
96            hurting_projectile_base: HurtingProjectileBase::new(),
97        }
98    }
99
100    /// Creates a wither skull from saved base data.
101    #[must_use]
102    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
103        Self {
104            base: EntityBase::from_load(load, entity_type.dimensions),
105            entity_type,
106            entity_data: SyncMutex::new(WitherSkullEntityData::new()),
107            projectile_base: ProjectileBase::new(),
108            hurting_projectile_base: HurtingProjectileBase::new(),
109        }
110    }
111
112    /// Returns whether this is the charged blue skull.
113    ///
114    /// Vanilla parity: `WitherSkull.isDangerous`.
115    #[must_use]
116    pub fn is_dangerous(&self) -> bool {
117        *self.entity_data.lock().dangerous.get()
118    }
119
120    /// Charges or discharges the skull.
121    ///
122    /// Vanilla parity: `WitherSkull.setDangerous`, called by the wither's middle
123    /// head when it has finished charging.
124    pub fn set_dangerous(&self, dangerous: bool) {
125        self.entity_data.lock().dangerous.set(dangerous);
126    }
127
128    /// Withers a living target for as long as the difficulty says.
129    ///
130    /// Vanilla parity: the difficulty branch of `WitherSkull.onHitEntity`. Easy
131    /// and peaceful inflict nothing at all, which is why a wither on easy is a
132    /// damage fight rather than an attrition one.
133    fn apply_wither(world: &World, target: &SharedEntity) {
134        let seconds = match world.difficulty() {
135            Difficulty::Normal => NORMAL_WITHER_SECONDS,
136            Difficulty::Hard => HARD_WITHER_SECONDS,
137            Difficulty::Peaceful | Difficulty::Easy => return,
138        };
139        let Some(living) = target.as_living_entity() else {
140            return;
141        };
142        living.add_mob_effect(MobEffectInstance::with_duration(
143            vanilla_mob_effects::WITHER,
144            seconds * 20,
145            WITHER_AMPLIFIER,
146        ));
147    }
148}

195impl Projectile for WitherSkullEntity {
196    fn projectile_base(&self) -> &ProjectileBase {
197        &self.projectile_base
198    }
199
200    fn on_deflection(&self, by_attack: bool) {
201        self.hurting_projectile_on_deflection(by_attack);
202    }
203
204    /// Vanilla parity: `WitherSkull.onHitEntity`.
205    fn on_hit_entity(&self, entity: &SharedEntity, _location: DVec3) {
206        let Some(world) = self.level() else {
207            return;
208        };
209
210        let owner = self.get_owner();
211        let living_owner = owner
212            .as_ref()
213            .filter(|owner| owner.as_living_entity().is_some());
214
215        let was_hurt = if let Some(owner) = living_owner {
216            let source = DamageSource::environment(&vanilla_damage_types::WITHER_SKULL)
217                .with_direct_entity(self.id())
218                .with_causing_entity(owner.id());
219            let hurt = entity.hurt(&world, &source, OWNED_DAMAGE);
220            if hurt && !Entity::is_alive(entity.as_ref()) {
221                // Vanilla parity: the kill heals the wither, not the skull.
222                if let Some(living) = owner.as_living_entity() {
223                    living.heal(KILL_HEAL);
224                }
225            }
226            // TODO: on a hit that did not kill, vanilla also runs
227            // `EnchantmentHelper.doPostAttackEffects`; Foton has no post-attack
228            // enchantment dispatch for projectiles yet.
229            hurt
230        } else {
231            entity.hurt(
232                &world,
233                &DamageSource::environment(&vanilla_damage_types::MAGIC),
234                UNOWNED_DAMAGE,
235            )
236        };
237
238        if was_hurt {
239            Self::apply_wither(&world, entity);
240        }
241    }
242
243    /// Vanilla parity: `WitherSkull.onHit`.
244    ///
245    /// Deviation: vanilla also overrides `getBlockExplosionResistance` so a
246    /// charged skull caps the resistance of anything `WitherBoss.canDestroy`
247    /// allows at 0.8, which is how the blue skull chews through obsidian-grade
248    /// blocks a plain one bounces off. Foton's `World::explode` has no
249    /// per-source resistance hook, so both forms here break exactly the same
250    /// blocks.
251    fn on_hit(&self, hit: &ProjectileHit) {
252        self.projectile_on_hit(hit);
253
254        let Some(world) = self.level() else {
255            return;
256        };
257        if self.is_removed() {
258            return;
259        }
260
261        // Vanilla passes `ExplosionInteraction.MOB`, which the level resolves
262        // through the `mobGriefing` rule; the fire flag is hard false, so a
263        // skull never leaves flames the way a ghast's fireball does.
264        let interaction = if world.get_game_rule(&MOB_GRIEFING) {
265            world.explosion_destroy_type(&MOB_EXPLOSION_DROP_DECAY)
266        } else {
267            ExplosionBlockInteraction::Keep
268        };
269        world.explode(
270            ExplosionSpec::new(
271                Some(self.id()),
272                self.get_owner().map(|owner| owner.id()),
273                None,
274                EXPLOSION_RADIUS,
275                false,
276                interaction,
277            ),
278            self.position(),
279        );
280        self.set_removed(RemovalReason::Discarded);
281    }
282}

331    #[test]
332    fn a_charged_skull_gives_up_more_speed_each_tick_than_a_plain_one() {
333        let skull = skull(Weak::new());
334
335        assert!((skull.get_inertia() - 0.95).abs() < 1.0e-9);
336
337        skull.set_dangerous(true);
338        assert!((skull.get_inertia() - DANGEROUS_INERTIA).abs() < 1.0e-9);
339    }

341    #[test]
342    fn a_skull_never_shows_flames_even_though_it_flies_out_of_the_nether() {
343        assert!(!skull(Weak::new()).is_on_fire());
344        assert!(!skull(Weak::new()).should_burn());
345    }

347    #[test]
348    fn a_hit_on_normal_difficulty_withers_the_target_for_ten_seconds() {
349        init_vanilla_registry();
350        init_behaviors();
351
352        let world = fresh_test_world("test_wither_skull_effect");
353        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
354        let target: SharedEntity = Arc::new(PigEntity::new(
355            &vanilla_entities::PIG,
356            next_entity_id(),
357            DVec3::new(8.5, 64.0, 8.5),
358            Arc::downgrade(&world),
359        ));
360        world
361            .try_add_entity(Arc::clone(&target))
362            .expect("the test chunk is loaded");
363
364        let skull = skull(Arc::downgrade(&world));
365        skull.on_hit_entity(&target, target.position());
366
367        let living = target.as_living_entity().expect("a pig is a living entity");
368        let effect = living
369            .mob_effect(vanilla_mob_effects::WITHER)
370            .expect("a wither skull withers what it hits");
371        assert_eq!(effect.duration(), 200);
372        assert_eq!(effect.amplifier(), 1);
373    }
