1//! Evoker fangs entity.
2//!
3//! Vanilla parity: `EvokerFangs`. A pair of jaws that rise out of the ground,
4//! bite once, and sink. Vanilla files it under projectiles but it is a plain
5//! entity: it does not move, it cannot be hurt, and it exists for a little
6//! over a second. The two numbers that matter are the warmup -- which is how
7//! the evoker's line of fangs ripples outwards instead of firing at once --
8//! and the eight-tick offset at which the bite lands, well after the jaws are
9//! visible.
10

34/// NBT key vanilla stores the warmup under.
35const TAG_WARMUP: &str = "Warmup";
36/// NBT key vanilla stores the owner under.
37const TAG_OWNER: &str = "Owner";
38
39/// Ticks the jaws stay up once they have risen.
40///
41/// Vanilla parity: the `lifeTicks = 22` field.
42const LIFE_TICKS: i32 = 22;
43
44/// Warmup value at which the bite lands.
45///
46/// Vanilla parity: the `warmupDelayTicks == -8` of `tick`. The jaws are drawn
47/// from the moment the warmup runs out, so the bite is eight ticks behind what
48/// a player sees -- which is the window to walk out of them.
49const BITE_AT_WARMUP: i32 = -8;
50
51/// How far past the jaws the bite reaches.
52///
53/// Vanilla parity: the `inflate(0.2, 0.0, 0.2)` of `tick`.
54const BITE_REACH: f64 = 0.2;
55
56/// Damage one bite deals.
57///
58/// Vanilla parity: the `6.0F` of `dealDamageTo`.
59const BITE_DAMAGE: f32 = 6.0;
60
61/// A pair of evoker fangs.
62#[entity_behavior(class = "EvokerFangs")]
63pub struct EvokerFangsEntity {
64    base: EntityBase,
65    entity_type: EntityTypeRef,
66    entity_data: SyncMutex<EvokerFangsEntityData>,
67    state: SyncMutex<FangsState>,
68}
69
70/// The three counters a pair of fangs carries.
71#[derive(Debug)]
72struct FangsState {
73    /// Ticks before the jaws rise; counts down past zero to time the bite.
74    warmup_delay_ticks: i32,
75    /// Whether the client has been told to draw the jaws.
76    sent_spike_event: bool,
77    /// Ticks the jaws stay up for once risen.
78    life_ticks: i32,
79    /// Who cast the spell, so the fangs do not bite their own side.
80    owner: Option<Uuid>,
81}
82
83impl FangsState {
84    const fn new() -> Self {
85        Self {
86            warmup_delay_ticks: 0,
87            sent_spike_event: false,
88            life_ticks: LIFE_TICKS,
89            owner: None,
90        }
91    }
92}

99impl EvokerFangsEntity {
100    /// Creates a pair of fangs at runtime.
101    #[must_use]
102    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
103        Self {
104            base: EntityBase::new(id, position, entity_type.dimensions, world),
105            entity_type,
106            entity_data: SyncMutex::new(EvokerFangsEntityData::new()),
107            state: SyncMutex::new(FangsState::new()),
108        }
109    }
110
111    /// Creates a pair of fangs from saved base data.
112    #[must_use]
113    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
114        Self {
115            base: EntityBase::from_load(load, entity_type.dimensions),
116            entity_type,
117            entity_data: SyncMutex::new(EvokerFangsEntityData::new()),
118            state: SyncMutex::new(FangsState::new()),
119        }
120    }
121
122    /// Aims a fresh pair of fangs, the way the evoker's spell does.
123    ///
124    /// Vanilla parity: the `EvokerFangs(level, x, y, z, rotation, warmup, owner)`
125    /// constructor. The rotation arrives in radians, as the spell computes it.
126    pub fn place(&self, position: DVec3, rotation_radians: f32, warmup_delay_ticks: i32) {
127        {
128            let mut state = self.state.lock();
129            state.warmup_delay_ticks = warmup_delay_ticks;
130        }
131        self.base().set_position_local(position);
132        self.set_rotation((rotation_radians.to_degrees(), 0.0));
133        self.base().set_old_rotation_to_current();
134    }
135
136    /// Sets who cast the spell.
137    pub fn set_owner_uuid(&self, owner: Option<Uuid>) {
138        self.state.lock().owner = owner;
139    }
140
141    /// Returns who cast the spell.
142    #[must_use]
143    pub fn owner_uuid(&self) -> Option<Uuid> {
144        self.state.lock().owner
145    }
146
147    /// Resolves the caster in the current world.
148    ///
149    /// Vanilla parity: `EvokerFangs.getOwner`.
150    #[must_use]
151    pub fn get_owner(&self) -> Option<SharedEntity> {
152        let uuid = self.owner_uuid()?;
153        self.level()?.get_entity_by_uuid(&uuid)
154    }
155
156    /// Bites everything standing in the jaws.
157    ///
158    /// Vanilla parity: the `warmupDelayTicks == -8` branch of `tick`.
159    fn bite(&self, world: &World) {
160        let owner = self.get_owner();
161        let owner_id = owner.as_ref().map(|owner| owner.id());
162        let source = match owner_id {
163            // Vanilla parity: `damageSources().indirectMagic(this, owner)`.
164            Some(owner_id) => DamageSource::environment(&vanilla_damage_types::INDIRECT_MAGIC)
165                .with_causing_entity(owner_id)
166                .with_direct_entity(self.id()),
167            // Vanilla parity: `damageSources().magic()` when the caster is gone.
168            None => DamageSource::environment(&vanilla_damage_types::MAGIC),
169        };
170
171        let bite_box = self.bounding_box().inflate_xyz(BITE_REACH, 0.0, BITE_REACH);
172        let caught = world.get_entities_in_aabb_matching(&bite_box, |entity| {
173            let Some(living) = entity.as_living_entity() else {
174                return false;
175            };
176            LivingEntity::is_alive(living)
177                && !entity.is_invulnerable()
178                && Some(entity.id()) != owner_id
179        });
180
181        for entity in caught {
182            // Vanilla parity: an owned pair of fangs never bites the caster's
183            // allies, which is what keeps an evoker's fangs off its vexes.
184            if let Some(owner) = owner.as_ref()
185                && owner.is_allied_to(entity.as_ref())
186            {
187                continue;
188            }
189            let Some(living) = entity.as_living_entity() else {
190                continue;
191            };
192            living.hurt(world, &source, BITE_DAMAGE);
193        }
194    }
195
196    /// Runs vanilla `EvokerFangs.tick` on the server side.
197    fn tick_server(&self, world: &World) {
198        let (should_bite, should_spike, expired) = {
199            let mut state = self.state.lock();
200            state.warmup_delay_ticks -= 1;
201            if state.warmup_delay_ticks >= 0 {
202                return;
203            }
204
205            let should_bite = state.warmup_delay_ticks == BITE_AT_WARMUP;
206            let should_spike = !state.sent_spike_event;
207            state.sent_spike_event = true;
208            state.life_ticks -= 1;
209            (should_bite, should_spike, state.life_ticks < 0)
210        };
211
212        if should_bite {
213            self.bite(world);
214        }
215        if should_spike {
216            self.broadcast_entity_event(EntityStatus::StartAttacking);
217        }
218        if expired {
219            self.set_removed(RemovalReason::Killed);
220        }
221    }
222}

263        PushReaction::Ignore

266    fn save_additional(&self, nbt: &mut NbtCompound) {
267        let state = self.state.lock();
268        nbt.insert(TAG_WARMUP, state.warmup_delay_ticks);
269        if let Some(owner) = state.owner {
270            nbt.insert(TAG_OWNER, NbtTag::IntArray(owner.to_int_array().to_vec()));
271        }
272    }
273
274    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
275        let mut state = self.state.lock();
276        state.warmup_delay_ticks = nbt.int(TAG_WARMUP).unwrap_or(0);
277        state.owner = nbt
278            .int_array(TAG_OWNER)
279            .and_then(|owner| Uuid::from_int_array(&owner));
280    }

283#[cfg(test)]
284mod tests {
285    use std::sync::Arc;
286
287    use foton_registry::{init_vanilla_registry, vanilla_entities};
288    use foton_utils::ChunkPos;
289
290    use super::*;
291    use crate::behavior::init_behaviors;
292    use crate::entity::next_entity_id;
293    use crate::test_support::{fresh_test_world, insert_ready_full_chunk};
294
295    const TEST_POSITION: DVec3 = DVec3::new(8.5, 64.0, 8.5);
296
297    /// The bite is eight ticks behind the jaws a player sees, and the whole
298    /// pair lives for the warmup plus thirty ticks. Both numbers are what make
299    /// an evoker's fang line dodgeable rather than instant.
300    #[test]
301    fn fangs_bite_eight_ticks_after_they_appear_and_sink_thirty_ticks_later() {
302        init_vanilla_registry();
303        init_behaviors();
304        let world = fresh_test_world("evoker_fangs_lifetime");
305        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
306
307        let fangs = Arc::new(EvokerFangsEntity::new(
308            &vanilla_entities::EVOKER_FANGS,
309            next_entity_id(),
310            TEST_POSITION,
311            Arc::downgrade(&world),
312        ));
313        fangs.place(TEST_POSITION, 0.0, 0);
314        world
315            .try_add_entity(Arc::clone(&fangs) as SharedEntity)
316            .expect("the fangs should attach to the loaded chunk");
317
318        // One tick takes the warmup to -1, which is when the jaws are drawn.
319        fangs.tick();
320        assert!(fangs.state.lock().sent_spike_event);
321        assert_eq!(fangs.state.lock().life_ticks, LIFE_TICKS - 1);
322
323        for _ in 0..(-BITE_AT_WARMUP - 1) {
324            fangs.tick();
325        }
326        assert_eq!(fangs.state.lock().warmup_delay_ticks, BITE_AT_WARMUP);
327
328        for _ in 0..LIFE_TICKS {
329            fangs.tick();
330        }
331        assert!(fangs.is_removed(), "the jaws sink once their life runs out");
332    }
