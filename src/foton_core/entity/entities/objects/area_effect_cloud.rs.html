1//! Area effect cloud.
2//!
3//! Vanilla parity: `AreaEffectCloud`. The lingering half of alchemy: a patch of
4//! ground that keeps working after the bottle is gone. What makes it a distinct
5//! thing rather than a slow splash is that it charges for use -- every entity it
6//! catches shrinks it -- so one cloud can serve a crowd once or one target
7//! several times, but not both.

135/// State a cloud keeps that is not synced.
136struct CloudState {
137    /// Ticks the cloud runs for once it starts, or -1 for forever.
138    duration: i32,
139    /// Ticks before it starts.
140    wait_time: i32,
141    /// Ticks before one entity may be dosed twice.
142    reapplication_delay: i32,
143    /// How the radius changes each tick.
144    radius_per_tick: f32,
145    /// How the radius changes each time somebody is dosed.
146    radius_on_use: f32,
147    /// Effects this cloud carries.
148    ///
149    /// Keep complete `MobEffectInstance` metadata so cloud dosing preserves
150    /// vanilla ambient, particle and icon visibility flags.
151    effects: Vec<MobEffectInstance>,
152    /// Entity id to the tick it may be dosed again on.
153    victims: FxHashMap<i32, i32>,
154    /// UUID of the projectile owner, when this cloud came from a projectile.
155    owner_uuid: Option<Uuid>,
156    /// Base potion carried by the cloud, when created from a potion item.
157    base_potion: Option<PotionRef>,
158}

249    /// Sets this cloud up the way a lingering potion does.
250    ///
251    /// Vanilla parity: `ThrownLingeringPotion.onHitAsPotion`. The per-tick
252    /// shrink is derived from the duration, so a cloud fades out exactly as it
253    /// runs out rather than vanishing at full size.
254    pub fn configure_as_lingering(&self, effects: Vec<(MobEffectRef, i32, i32)>) {
255        self.set_radius(DEFAULT_LINGERING_RADIUS);
256
257        let mut state = self.state.lock();
258        state.duration = DEFAULT_LINGERING_DURATION;
259        state.wait_time = DEFAULT_LINGERING_WAIT_TIME;
260        state.radius_on_use = DEFAULT_LINGERING_RADIUS_ON_USE;
261        #[expect(
262            clippy::cast_precision_loss,
263            reason = "the lingering duration is six hundred ticks"
264        )]
265        let per_tick = -DEFAULT_LINGERING_RADIUS / DEFAULT_LINGERING_DURATION as f32;
266        state.radius_per_tick = per_tick;
267        state.effects = effects
268            .into_iter()
269            .map(|(effect, duration, amplifier)| {
270                MobEffectInstance::with_duration(effect, duration, amplifier)
271            })
272            .collect();
273    }

275    /// Sets this cloud up the way a creeper's death does.
276    ///
277    /// Vanilla parity: `Creeper.spawnLingeringCloud`. It is smaller and
278    /// shorter-lived than a lingering potion's, and it hands out a quarter of
279    /// each effect it carries.
280    ///
281    /// Deviation: vanilla keeps that quarter on the cloud as
282    /// `potionDurationScale` and applies it per victim. Foton stores plain

290        state.wait_time = CREEPER_CLOUD_WAIT_TIME;
291        state.radius_on_use = CREEPER_CLOUD_RADIUS_ON_USE;
292        #[expect(
293            clippy::cast_precision_loss,
294            reason = "the creeper cloud's duration is three hundred ticks"
295        )]
296        let per_tick = -CREEPER_CLOUD_RADIUS / CREEPER_CLOUD_DURATION as f32;
297        state.radius_per_tick = per_tick;

306    /// Sets this cloud up the way a dragon fireball does.
307    ///
308    /// Vanilla parity: the cloud built by `DragonFireball.onHit`. Unlike a
309    /// lingering potion's cloud this one grows, from three blocks to seven over
310    /// its half-minute, and costs nothing per victim, so it keeps working on
311    /// everything standing in it for the whole duration.
312    ///
313    /// Two vanilla settings are dropped because Foton has nowhere to put them.
314    /// `cloud.setOwner(livingEntity)` has no equivalent -- Foton's cloud has no
315    /// owner, so its damage is credited to nobody. `setPotionDurationScale(0.25F)`
316    /// likewise: Foton stores plain durations on the cloud, and the scale only
317    /// matters for effects that last, which a dragon's breath does not carry.
318    pub fn configure_as_dragon_breath(&self) {
319        self.set_radius(DRAGON_BREATH_RADIUS);
320        self.entity_data.lock().particle.set(ParticleData::new(
321            &vanilla_particle_types::DRAGON_BREATH,
322            PowerParticleOption::new(1.0),
323        ));
324
325        let mut state = self.state.lock();
326        state.duration = DRAGON_BREATH_DURATION;
327        state.wait_time = 10;
328        state.radius_on_use = 0.0;
329        state.radius_per_tick =
330            (DRAGON_BREATH_FINAL_RADIUS - DRAGON_BREATH_RADIUS) / DRAGON_BREATH_DURATION as f32;
331        state.effects = vec![MobEffectInstance::with_duration(
332            vanilla_mob_effects::INSTANT_DAMAGE,
333            1,
334            DRAGON_BREATH_AMPLIFIER,
335        )];
336    }

444    /// Adds a custom effect, following Bukkit override semantics.
445    pub fn add_custom_effect(&self, effect: MobEffectInstance, override_existing: bool) -> bool {
446        let mut state = self.state.lock();
447        if let Some(existing) = state
448            .effects
449            .iter_mut()
450            .find(|existing| existing.effect() == effect.effect())
451        {
452            if !override_existing {
453                return false;
454            }
455            *existing = effect;
456            return true;
457        }
458        state.effects.push(effect);
459        true
460    }

476    /// Doses everything standing in the cloud that is not on cooldown.
477    ///
478    /// Vanilla parity: the victim sweep of `AreaEffectCloud.serverTick`.
479    /// Returns how much the radius should change for the entities it dosed.
480    fn dose_victims(&self, world: &Arc<World>, radius: f32) -> f32 {
481        let tick = self.tick_count();
482        let position = self.position();
483
484        let (effects, reapplication_delay, radius_on_use) = {
485            let mut state = self.state.lock();
486            state.victims.retain(|_, ready_at| tick < *ready_at);
487            if state.effects.is_empty() {
488                state.victims.clear();
489                return 0.0;
490            }
491            (
492                state.effects.clone(),
493                state.reapplication_delay,
494                state.radius_on_use,
495            )
496        };
497
498        let aabb = WorldAabb::new(
499            position.x - f64::from(radius),
500            position.y - 0.5,
501            position.z - f64::from(radius),
502            position.x + f64::from(radius),
503            position.y + 0.5,
504            position.z + f64::from(radius),
505        );
506
507        let mut radius_change = 0.0;
508        for entity in world.get_entities_in_aabb(&aabb) {
509            let Some(living) = entity.as_living_entity() else {
510                continue;
511            };
512            if self.state.lock().victims.contains_key(&entity.id()) {
513                continue;
514            }
515
516            // Vanilla measures the horizontal distance only, so a cloud reaches
517            // across a floor but not up a ladder.
518            let entity_position = entity.position();
519            let dx = entity_position.x - position.x;
520            let dz = entity_position.z - position.z;
521            if dx.mul_add(dx, dz * dz) > f64::from(radius * radius) {
522                continue;
523            }
524
525            let mut dosed = false;
526            for effect_instance in &effects {
527                let effect = effect_instance.effect();
528                // Vanilla parity: the `isInstantaneous` branch of
529                // `AreaEffectCloud.tick`. A cloud does not afflict anyone with
530                // an instant effect -- it applies it on the spot, and at half
531                // strength, which is why standing in a dragon's breath costs
532                // three points a dose rather than six.
533                if is_instantaneous(effect) {
534                    apply_instantaneous_mob_effect(
535                        world,
536                        living,
537                        effect,
538                        effect_instance.amplifier(),
539                        CLOUD_INSTANT_SCALE,
540                    );
541                    dosed = true;
542                    continue;
543                }
544                let instance = effect_instance.clone();
545                if !living.can_be_affected(&instance) {
546                    continue;
547                }
548                living.add_mob_effect(instance);
549                dosed = true;
550            }
551
552            if dosed {
553                self.state
554                    .lock()
555                    .victims
556                    .insert(entity.id(), tick + reapplication_delay);
557                radius_change += radius_on_use;
558            }
559        }
560
561        radius_change
562    }

578    /// Vanilla parity: `AreaEffectCloud.serverTick`.
579    fn tick(&self) {
580        let Some(world) = self.level() else {
581            return;
582        };
583        let tick = self.tick_count();
584
585        let (duration, wait_time, radius_per_tick) = {
586            let state = self.state.lock();
587            (state.duration, state.wait_time, state.radius_per_tick)
588        };
589
590        if duration != -1 && tick - wait_time >= duration {
591            self.set_removed(RemovalReason::Discarded);
592            return;
593        }
594
595        let should_wait = tick < wait_time;
596        if self.is_waiting() != should_wait {
597            self.set_waiting(should_wait);
598        }
599        if should_wait {
600            return;
601        }
602
603        let mut radius = self.radius();
604        if radius_per_tick != 0.0 {
605            radius += radius_per_tick;
606            if radius < MINIMAL_RADIUS {
607                self.set_removed(RemovalReason::Discarded);
608                return;
609            }
610            self.set_radius(radius);
611        }
612
613        if tick % TIME_BETWEEN_APPLICATIONS != 0 {
614            return;
615        }
616
617        let radius_change = self.dose_victims(&world, radius);
618        if radius_change != 0.0 {
619            radius += radius_change;
620            if radius < MINIMAL_RADIUS {
621                self.set_removed(RemovalReason::Discarded);
622                return;
623            }
624            self.set_radius(radius);
625        }
626    }

635    fn save_additional(&self, nbt: &mut NbtCompound) {
636        let state = self.state.lock();
637        nbt.insert("Duration", state.duration);
638        nbt.insert("WaitTime", state.wait_time);
639        nbt.insert("ReapplicationDelay", state.reapplication_delay);
640        nbt.insert("RadiusPerTick", state.radius_per_tick);
641        nbt.insert("RadiusOnUse", state.radius_on_use);
642        nbt.insert("Radius", self.radius());
643        if !state.effects.is_empty() {
644            nbt.insert(
645                "Effects",
646                NbtList::Compound(
647                    state
648                        .effects
649                        .iter()
650                        .map(MobEffectInstance::to_vanilla_nbt)
651                        .collect(),
652                ),
653            );
654        }
655        if let Some(potion) = state.base_potion {
656            nbt.insert("Potion", potion.key.to_string());
657        }
658        if let Some(owner) = state.owner_uuid {
659            nbt.insert("Owner", NbtTag::IntArray(owner.to_int_array().to_vec()));
660        }
661    }
662
663    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
664        let mut state = self.state.lock();
665        state.duration = nbt.int("Duration").unwrap_or(DEFAULT_LINGERING_DURATION);
666        state.wait_time = nbt.int("WaitTime").unwrap_or(20);
667        state.reapplication_delay = nbt
668            .int("ReapplicationDelay")
669            .unwrap_or(DEFAULT_REAPPLICATION_DELAY);
670        state.radius_per_tick = nbt.float("RadiusPerTick").unwrap_or(0.0);
671        state.radius_on_use = nbt.float("RadiusOnUse").unwrap_or(0.0);
672
673        state.effects = nbt
674            .list("Effects")
675            .and_then(|list| list.compounds())
676            .map(|effects| {
677                effects
678                    .into_iter()
679                    .filter_map(|effect| MobEffectInstance::from_vanilla_nbt(&effect))
680                    .collect()
681            })
682            .unwrap_or_default();
683        state.base_potion = nbt
684            .string("Potion")
685            .and_then(|key| key.to_str().parse().ok())
686            .and_then(|key| foton_registry::REGISTRY.potions.by_key(&key));
687        state.owner_uuid = nbt
688            .int_array("Owner")
689            .and_then(|values| Uuid::from_int_array(&values));
690        drop(state);
691        self.set_radius(nbt.float("Radius").unwrap_or(DEFAULT_LINGERING_RADIUS));
692    }

701    use crate::entity::entities::CowEntity;

704
705    /// A dragon's breath carries `INSTANT_DAMAGE`, which a cloud applies on the
706    /// spot rather than afflicting anyone with. Before that branch existed the
707    /// cloud handed out an effect instance nothing ever ticked, and standing in
708    /// a dragon's breath was free.
709    #[test]
710    fn a_dragon_breath_cloud_hurts_what_stands_in_it() {
711        init_vanilla_registry();
712        let world = fresh_test_world("area_effect_cloud_dragon_breath");
713        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
714
715        let cow = Arc::new(CowEntity::new(
716            &vanilla_entities::COW,
717            next_entity_id(),
718            DVec3::new(8.5, 64.0, 8.5),
719            Arc::downgrade(&world),
720        ));
721        world
722            .try_add_entity(cow.clone())
723            .expect("the cow's chunk is loaded");
724        let full_health = cow.get_health();
725
726        let cloud = Arc::new(AreaEffectCloudEntity::new(
727            &vanilla_entities::AREA_EFFECT_CLOUD,
728            next_entity_id(),
729            DVec3::new(8.5, 64.0, 8.5),
730            Arc::downgrade(&world),
731        ));
732        world
733            .try_add_entity(cloud.clone())
734            .expect("the cloud's chunk is loaded");
735        cloud.configure_as_dragon_breath();
736
737        // The cloud waits ten ticks before it starts, and only sweeps every
738        // fifth tick after that.
739        for _ in 0..=(DEFAULT_LINGERING_WAIT_TIME + TIME_BETWEEN_APPLICATIONS) {
740            cloud.advance_tick_count();
741            cloud.tick();
742        }
743
744        // Amplifier one at the cloud's half scale: `(0.5 * (6 << 1) + 0.5)`.
745        assert_f32_close(full_health - cow.get_health(), 6.0);
746    }
747
748    fn assert_f32_close(left: f32, right: f32) {
749        assert!(
750            (left - right).abs() <= f32::EPSILON,
751            "expected {left} to equal {right}"
752        );
753    }
