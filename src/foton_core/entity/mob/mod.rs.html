121/// Picks the slot an item rolled by an equipment table goes into.
122///
123/// Vanilla parity: `EquipmentUser.resolveSlot`.
124fn resolve_equipment_slot(
125    to_equip: &ItemStack,
126    already_inserted: &[EquipmentSlot],
127) -> Option<EquipmentSlot> {
128    if to_equip.is_empty() {
129        return None;
130    }
131    match to_equip.get_equippable_slot() {
132        Some(slot) if !already_inserted.contains(&slot) => Some(slot),
133        Some(_) => None,
134        None => (!already_inserted.contains(&EquipmentSlot::MainHand))
135            .then_some(EquipmentSlot::MainHand),
136    }
137}

139#[derive(Debug, Clone, Copy, PartialEq)]
140struct DropChances {
141    by_equipment: [f32; EquipmentSlot::ALL.len()],
142}
143
144impl DropChances {
145    const DEFAULT: Self = Self {
146        by_equipment: [DEFAULT_EQUIPMENT_DROP_CHANCE; EquipmentSlot::ALL.len()],
147    };
148
149    #[must_use]
150    const fn by_equipment(self, slot: EquipmentSlot) -> f32 {
151        self.by_equipment[slot.index()]
152    }
153
154    const fn set_guaranteed_drop(&mut self, slot: EquipmentSlot) {
155        self.by_equipment[slot.index()] = PRESERVE_ITEM_DROP_CHANCE;
156    }
157
158    fn set_equipment_chance(&mut self, slot: EquipmentSlot, chance: f32) -> bool {
159        if chance < 0.0 {
160            return false;
161        }
162
163        self.by_equipment[slot.index()] = chance;
164        true
165    }
166
167    #[must_use]
168    fn is_preserved(self, slot: EquipmentSlot) -> bool {
169        self.by_equipment(slot) > PRESERVE_ITEM_DROP_CHANCE_THRESHOLD
170    }
171
172    fn save(self, nbt: &mut NbtCompound) {
173        if self == Self::DEFAULT {
174            return;
175        }
176
177        let mut drop_chances = NbtCompound::new();
178        for slot in EquipmentSlot::ALL {
179            let chance = self.by_equipment(slot);
180            if chance.to_bits() != DEFAULT_EQUIPMENT_DROP_CHANCE.to_bits() {
181                drop_chances.insert(slot.name(), chance);
182            }
183        }
184
185        nbt.insert("drop_chances", NbtTag::Compound(drop_chances));
186    }
187
188    fn load(nbt: BorrowedNbtCompoundView<'_, '_>) -> Self {
189        let Some(drop_chances) = nbt.compound("drop_chances") else {
190            return Self::DEFAULT;
191        };
192
193        let mut loaded = Self::DEFAULT;
194        for slot in EquipmentSlot::ALL {
195            let Some(chance) = drop_chances.float(slot.name()) else {
196                continue;
197            };
198            if !loaded.set_equipment_chance(slot, chance) {
199                return Self::DEFAULT;
200            }
201        }
202
203        loaded
204    }
205}

297    ///
298    /// Vanilla parity: `Mob.getTarget` through `asValidTarget`. The target is
299    /// held weakly, so a target that has since been removed reads as `None` and

322        &self,
323        target: Option<&SharedEntity>,
324        is_valid: impl Fn(&dyn LivingEntity) -> bool,
325    ) -> bool {
326        let Some(target) = target else {
327            *self.target.lock() = None;
328            return true;
329        };
330        if !target.is_living_entity() {
331            return false;
332        }
333        let Some(living_target) = target.as_living_entity() else {
334            return false;
335        };
336        if !is_valid(living_target) {
337            *self.target.lock() = None;
338            return false;
339        }
340
341        *self.target.lock() = Some(Arc::downgrade(target));
342        true
343    }

462/// A living entity that vanilla derives from `Mob`: it has goals, a
463/// navigation, and a target.
464///
465/// Every method here names the vanilla method it stands for. The default
466/// bodies are `Mob`'s own, so a mob overrides only what its Java class
467/// overrides; where a vanilla override would call `super`, the default body is
468/// exposed separately under a `mob_`-prefixed name, because Rust has no
469/// `super`.

471    /// The state this mob holds on `Mob`'s behalf. See [`MobBase`].
472    fn mob_base(&self) -> &MobBase;
473
474    /// The packed flag byte vanilla synchronizes as `DATA_MOB_FLAGS_ID`.
475    ///
476    /// Bit 1 is "no AI", bit 2 "left-handed", bit 4 "aggressive". Read them
477    /// through [`Self::is_no_ai`], [`Self::is_left_handed`] and
478    /// [`Self::is_aggressive`] rather than masking here.
479    fn mob_flags(&self) -> i8;
480
481    /// Writes the whole flag byte. See [`Self::mob_flags`].
482    fn set_mob_flags(&self, flags: i8);
483
484    /// Returns vanilla `Mob.isSaddled`.
485    fn is_saddled(&self) -> bool {
486        let mut is_saddled = false;
487        self.with_equipment_slot(EquipmentSlot::Saddle, &mut |item_stack| {
488            is_saddled = self.is_equippable_in_slot(item_stack, EquipmentSlot::Saddle);
489        });
490        is_saddled
491    }

493    /// Returns whether this mob is a vanilla `Monster`.
494    ///
495    /// Vanilla parity: the `instanceof Monster` tests that goals such as
496    /// `Rabbit.RabbitAvoidEntityGoal` and `Fox` use. Foton has no entity class
497    /// hierarchy, so each mob that vanilla derives from `Monster` says so here.
498    /// A slime is deliberately not one: vanilla derives it from `AbstractCubeMob`
499    /// and only tags it `Enemy`, which is why a rabbit never flees a slime.

501        false

504    /// Returns this mob's brain, when it has one.
505    ///

507    /// `LivingEntity` a brain and lets `makeBrain` return an empty one, which
508    /// `Brain.isBrainDead` then has to detect. Foton's brain owns three mutexes
509    /// and two maps, and every goal-driven mob in the tree would pay for one it

548    /// Returns whatever this mob's brain is attacking.
549    ///
550    /// Vanilla parity: `Mob.getTargetFromBrain`. A brain mob has no `target`
551    /// field of its own -- the memory is the target -- so a mob whose `getTarget`
552    /// is `getTargetFromBrain` overrides [`Self::target`] with this.
553    fn target_from_brain(&self) -> Option<SharedEntity> {
554        let target = self
555            .brain()?
556            .get_memory(memory_module_types::ATTACK_TARGET)
557            .and_then(|memory| memory.get())?;
558        let living = target.as_living_entity()?;
559        if !self.is_valid_target(living) {
560            return None;
561        }
562        Some(target)

565    /// Sets vanilla `Mob.target`.
566    ///
567    /// Returns `false` when the supplied entity is not a living entity.
568    fn set_target(&self, target: Option<&SharedEntity>) -> bool {
569        let previous = self.target();
570        let changed = self
571            .mob_base()
572            .set_target(target, |target| self.is_valid_target(target));
573        if !changed {
574            return false;
575        }
576        self.finish_target_change(previous, target)
577    }

579    /// Announces a target change and undoes it if a plugin cancels.
580    ///
581    /// Split out of [`Self::set_target`] because a mob that sets its target
582    /// through the brain reaches the same event from a different path. A
583    /// change to the same entity fires nothing, matching vanilla, which only
584    /// notifies on an actual transition.
585    fn finish_target_change(
586        &self,
587        previous: Option<SharedEntity>,
588        target: Option<&SharedEntity>,
589    ) -> bool {
590        let next_id = target.map(|entity| entity.uuid());
591        if previous.as_ref().map(|entity| entity.uuid()) == next_id {
592            return true;
593        }
594        let Some(world) = self.level() else {

637    ///

639    /// guaranteed drop adds `1 + random(0..3)`, on top of the mob's own

641    /// make a zero-experience mob pay out.

658            if should_increase {

661        }

717    /// Sets undead mobs alight when the sun reaches them.
718    ///
719    /// Vanilla parity: `Mob.burnUndead`, gated by the `burn_in_daylight` entity
720    /// tag, so the behavior is data-driven rather than hard-coded per mob.
721    fn burn_in_daylight(&self) {
722        if !REGISTRY
723            .entity_types
724            .is_in_tag(self.entity_type(), &EntityTypeTag::BURN_IN_DAYLIGHT)

771    /// brightness is why a zombie in the shade of a tree survives, and why one
772    /// standing on open ground can last a few ticks before catching.
773    fn is_sun_burn_tick(&self) -> bool {
774        let Some(world) = self.level() else {

776        };
777        // Deviation: 26.2 reads the `gameplay/monsters_burn` environment

794        }

798            position.x,

933    /// Handles the two items that mean the same thing on every mob, whatever
934    /// its own `mob_interact` would rather do with a right click.
935    ///
936    /// Vanilla parity: `Mob.checkAndHandleImportantInteractions`. Running first
937    /// is the whole point of it: a villager's `mob_interact` opens its trades
938    /// and a horse's puts you in the saddle, so a name tag held out to either
939    /// would never be read if this came second.
940    fn check_and_handle_important_interactions(
941        &self,
942        player: &Player,
943        hand: InteractionHand,
944    ) -> InteractionResult {
945        let this = self.as_mob_source();
946        let held = InventoryAccess::new(player.inventory.clone(), hand);
947        let (is_name_tag, item) =
948            held.with_item(|stack| (stack.is(&vanilla_items::NAME_TAG), stack.item()));
949        let behavior = ITEM_BEHAVIORS.get_behavior(item);
950
951        if is_name_tag {
952            let result =
953                held.with_item(|stack| behavior.interact_living_entity(stack, player, this, hand));
954            if result.consumes_action() {
955                return result;
956            }
957        }
958
959        if !behavior.is_spawn_egg() {
960            return InteractionResult::Pass;
961        }
962
963        // Vanilla's `level() instanceof ServerLevel` guard. Foton only ever runs
964        // the server side, so the only way past it is a mob with no world at
965        // all, which vanilla's client branch answers the same way.
966        let Some(world) = self.level() else {
967            return InteractionResult::SuccessServer;
968        };
969
970        let offspring = held.with_item(|stack| {
971            SpawnEggItem::spawn_offspring_from_spawn_egg(player, this, &world, stack)
972        });
973        let Some(offspring) = offspring else {
974            // A wrong egg for this mob falls through to the ordinary
975            // interaction rather than eating the click.
976            return InteractionResult::Pass;
977        };
978        if let Some(baby) = offspring.as_mob() {
979            self.on_offspring_spawned_from_egg(player, baby);
980        }
981
982        InteractionResult::SuccessServer
983    }

1608    /// Runs one tick of leash physics: resolve a delayed holder, snap or
1609    /// pull, and spend the spin.
1610    ///
1611    /// Vanilla parity: `Leashable.tickLeash`.
1612    fn tick_leash(&self) {
1613        if let Some(holder) = self.leash_holder() {
1614            if !self.can_interact_with_level() || !holder.can_interact_with_level() {
1615                if let Some(world) = self.level()
1616                    && world.get_game_rule(&ENTITY_DROPS)
1617                {
1618                    self.drop_leash();
1619                } else {
1620                    self.remove_leash();
1621                }
1622                return;
1623            }
1624
1625            // Vanilla parity: `Leashable.tickLeash` wraps everything below in
1626            // `leashHolder.level() == entity.level()`. Without it a mob left
1627            // behind at a portal keeps doing leash physics against a holder in
1628            // another dimension: it measures a 3D distance between coordinates
1629            // from two different worlds, plays LEAD_BREAK at the holder's
1630            // position but in its own world, and -- worst -- reaches into the
1631            // holder's velocity from this world's tick thread while the other
1632            // world is ticking it on its own.
1633            //
1634            // Only a demonstrated mismatch stops the tick. Vanilla entities
1635            // always have a level; Foton's is a `Weak` that may be gone, and a
1636            // detached entity is not evidence of a different dimension.
1637            if let (Some(world), Some(holder_world)) = (self.level(), holder.level())
1638                && !Arc::ptr_eq(&world, &holder_world)
1639            {
1640                return;
1641            }
1642
1643            let distance_to = self.leash_distance_to(holder.as_ref());
1644            self.when_leashed_to(holder.as_ref());
1645            let angular_momentum_before_distance_action = self.leash_angular_momentum();
1646            if distance_to > self.leash_snap_distance() {
1647                if let Some(world) = self.level() {
1648                    world.play_sound_at(
1649                        &sound_events::ITEM_LEAD_BREAK,
1650                        SoundSource::Neutral,
1651                        holder.position(),
1652                        1.0,
1653                        1.0,
1654                        None,
1655                    );
1656                }
1657                self.leash_too_far_behavior();
1658            } else if distance_to
1659                > self.leash_elastic_distance()
1660                    - f64::from(holder.base().dimensions().width)
1661                    - f64::from(self.base().dimensions().width)
1662                && self.check_elastic_interactions(holder.as_ref())
1663            {
1664                self.on_elastic_leash_pull();
1665            } else {
1666                self.close_range_leash_behavior(holder.as_ref());
1667            }
1668            if !self.apply_leash_angular_momentum()
1669                && let Some(angular_momentum) = angular_momentum_before_distance_action
1670            {
1671                self.rotate_by_leash_angular_momentum(angular_momentum);
1672            }
1673            return;
1674        }
1675
1676        let Some(attachment) = self.leash_attachment() else {
1677            return;
1678        };
1679
1680        let Some(world) = self.level() else {
1681            return;
1682        };
1683
1684        match attachment {
1685            LeashAttachment::Entity(uuid) => {
1686                if let Some(holder) = world.get_entity_by_uuid(&uuid) {
1687                    let _ = self.set_leashed_to(&holder);
1688                    return;
1689                }
1690
1691                if self.tick_count() > DELAYED_LEASH_DROP_TICKS {
1692                    let _ = self.spawn_at_location(ItemStack::new(&vanilla_items::LEAD), 0.0);
1693                    self.remove_leash_state();
1694                }
1695            }
1696            LeashAttachment::FenceKnot(pos) => {
1697                if let Some(holder) = LeashFenceKnotEntity::get_or_create_knot(&world, pos) {
1698                    let _ = self.set_leashed_to(&holder);
1699                    return;
1700                }
1701
1702                if self.tick_count() > DELAYED_LEASH_DROP_TICKS {
1703                    let _ = self.spawn_at_location(ItemStack::new(&vanilla_items::LEAD), 0.0);
1704                    self.remove_leash_state();
1705                }
1706            }
1707        }
1708    }

1799    /// Decides whether this mob is removed for being unwatched.
1800    ///
1801    /// Vanilla parity: `Mob.checkDespawn`. Four outcomes in order: a mob not
1802    /// allowed in peaceful goes immediately; a persistent one only has its
1803    /// idle clock reset; one past its category's despawn distance goes; and
1804    /// one past the no-despawn distance that has idled for 600 ticks goes with
1805    /// probability 1/800 per tick, which is what makes distant despawning
1806    /// gradual rather than a cliff.
1807    fn check_mob_despawn(&self) {
1808        if self
1809            .level()
1810            .is_some_and(|world| world.difficulty() == Difficulty::Peaceful)
1811            && !self.entity_type().allowed_in_peaceful
1812        {
1813            self.set_removed(RemovalReason::Discarded);
1814            return;
1815        }
1816
1817        if self.is_persistence_required() || self.requires_custom_persistence() {
1818            self.set_no_action_time(0);
1819            return;
1820        }
1821
1822        let Some(nearest_player_dist_sqr) = self.nearest_player_distance_sqr() else {
1823            return;
1824        };
1825
1826        let mob_category = self.entity_type().mob_category;
1827        let despawn_distance = mob_category.despawn_distance();
1828        let despawn_distance_sqr = despawn_distance * despawn_distance;
1829        if nearest_player_dist_sqr > f64::from(despawn_distance_sqr)
1830            && self.remove_when_far_away(nearest_player_dist_sqr)
1831        {
1832            self.set_removed(RemovalReason::Discarded);
1833            return;
1834        }
1835
1836        let no_despawn_distance = mob_category.no_despawn_distance();
1837        let no_despawn_distance_sqr = no_despawn_distance * no_despawn_distance;
1838        if self.no_action_time() > 600
1839            && nearest_player_dist_sqr > f64::from(no_despawn_distance_sqr)
1840            && self.remove_when_far_away(nearest_player_dist_sqr)
1841        {
1842            let should_discard = rand::random_range(0..800) == 0;
1843            if should_discard {
1844                self.set_removed(RemovalReason::Discarded);
1845            }
1846        } else if nearest_player_dist_sqr < f64::from(no_despawn_distance_sqr) {
1847            self.set_no_action_time(0);
1848        }
1849    }

2349    /// Vanilla parity: the private `Mob.compareArmor`.
2350    #[expect(
2351        clippy::float_cmp,
2352        reason = "vanilla compares the two attribute values exactly; an epsilon here \
2353                  would change which armor a mob swaps for"
2354    )]
2355    fn compare_armor(
2356        &self,
2357        new_item_stack: &ItemStack,
2358        current_item_stack: &ItemStack,
2359        slot: EquipmentSlot,
2360    ) -> bool {
2361        if current_item_stack.has_enchantment_effect(EnchantmentEffectComponent::PreventArmorChange)
2362        {
2363            return false;
2364        }
2365
2366        let new_defense =
2367            self.approximate_attribute_with(new_item_stack, vanilla_attributes::ARMOR, slot);
2368        let old_defense =
2369            self.approximate_attribute_with(current_item_stack, vanilla_attributes::ARMOR, slot);
2370        if new_defense != old_defense {
2371            return new_defense > old_defense;
2372        }
2373
2374        let new_toughness = self.approximate_attribute_with(
2375            new_item_stack,
2376            vanilla_attributes::ARMOR_TOUGHNESS,
2377            slot,
2378        );
2379        let old_toughness = self.approximate_attribute_with(
2380            current_item_stack,
2381            vanilla_attributes::ARMOR_TOUGHNESS,
2382            slot,
2383        );
2384        if new_toughness != old_toughness {
2385            return new_toughness > old_toughness;
2386        }
2387        self.can_replace_equal_item(new_item_stack, current_item_stack)
2388    }

2465    /// equipped, so an empty stack means the mob turned it down. Armor the mob

2467    /// carrying a helmet it is not wearing. Vanilla also takes the level, which

2723    /// Steers a flier toward its wanted position.
2724    ///
2725    /// Vanilla parity: `FlyingMoveControl.tick`. The gravity flag is the part
2726    /// that matters: a flier only stays up because the move control turns
2727    /// gravity off for as long as it has somewhere to be.
2728    fn tick_flying_move_to_control(
2729        &self,
2730        wanted_position: DVec3,
2731        speed_modifier: f64,
2732        max_turn: f32,
2733    ) {
2734        self.set_no_gravity(true);
2735
2736        let position = self.position();
2737        let xd = wanted_position.x - position.x;
2738        let yd = wanted_position.y - position.y;
2739        let zd = wanted_position.z - position.z;
2740        if xd.mul_add(xd, yd.mul_add(yd, zd * zd)) < MOVE_CONTROL_MIN_SPEED_SQR {
2741            self.set_travel_input(LivingTravelInput::new(0.0, 0.0, 0.0));
2742            return;
2743        }
2744
2745        let y_rot = (zd.atan2(xd) as f32).to_degrees() - 90.0;
2746        let (yaw, pitch) = self.rotation();
2747        self.set_rotation((rotlerp(yaw, y_rot, MOVE_CONTROL_MAX_TURN), pitch));
2748
2749        let attribute = if self.on_ground() {
2750            vanilla_attributes::MOVEMENT_SPEED
2751        } else {
2752            vanilla_attributes::FLYING_SPEED
2753        };
2754        let speed = (speed_modifier * self.attributes().lock().required_value(attribute)) as f32;
2755        self.set_mob_speed(speed);
2756
2757        let horizontal = xd.hypot(zd);
2758        if yd.abs() <= f64::from(MOVE_CONTROL_MIN_FLYING_DELTA)
2759            && horizontal <= f64::from(MOVE_CONTROL_MIN_FLYING_DELTA)
2760        {
2761            return;
2762        }
2763
2764        let x_rot = -(yd.atan2(horizontal) as f32).to_degrees();
2765        let (yaw, pitch) = self.rotation();
2766        self.set_rotation((yaw, rotlerp(pitch, x_rot, max_turn)));
2767        let vertical = if yd > 0.0 { speed } else { -speed };
2768        let input = self.travel_input();
2769        self.set_travel_input(LivingTravelInput::new(
2770            input.sideways(),
2771            vertical,
2772            input.forward(),
2773        ));
2774    }

2813    /// Runs one tick of sideways movement, vanilla's `STRAFE` arm of
2814    /// `MoveControl.tick`.
2815    ///
2816    /// If the strafed direction is not walkable the mob falls back to walking
2817    /// straight forward, which is what stops a strafing skeleton from backing
2818    /// itself off a ledge.
2819    fn tick_strafe_control(&self, forward: f32, right: f32) {
2820        let movement_speed = self
2821            .attributes()
2822            .lock()
2823            .required_value(vanilla_attributes::MOVEMENT_SPEED) as f32;
2824        let speed = movement_speed * 0.25;
2825        let mut strafe_forward = forward;
2826        let mut strafe_right = right;
2827
2828        let mut distance = strafe_forward
2829            .mul_add(strafe_forward, strafe_right * strafe_right)
2830            .sqrt();
2831        if distance < 1.0 {
2832            distance = 1.0;
2833        }
2834        distance = speed / distance;
2835        let xa = strafe_forward * distance;
2836        let za = strafe_right * distance;
2837        let yaw_radians = self.rotation().0 * PI / 180.0;
2838        let sin = yaw_radians.sin();
2839        let cos = yaw_radians.cos();
2840        let dx = xa.mul_add(cos, -(za * sin));
2841        let dz = za.mul_add(cos, xa * sin);
2842        if !self.is_strafe_walkable(dx, dz) {
2843            strafe_forward = 1.0;
2844            strafe_right = 0.0;
2845        }
2846
2847        self.set_speed(speed);
2848        self.set_travel_input(LivingTravelInput::new(strafe_right, 0.0, strafe_forward));
2849        self.mob_base().controls().lock().move_control.set_wait();
2850    }
