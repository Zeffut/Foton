1//! One entity as a self-contained blob, for plugins that keep an entity
2//! outside the world -- a stabled horse in a database -- and bring it back.
3//!
4//! The format is Paper's `UnsafeValues.serializeEntity`: vanilla's entity
5//! compound (`id` included, passengers left out), written as a named root
6//! compound and gzipped the way `NbtIo.writeCompressed` does. Foton performs no
7//! data fixing, so the compound carries no `DataVersion`; a blob written by a
8//! Paper server of the same Minecraft version still reads back.
9
10use std::io::{Cursor, Read as _, Write as _};
11use std::sync::Arc;

14use flate2::read::GzDecoder;
15use flate2::write::GzEncoder;
16use glam::DVec3;
17use simdnbt::borrow::read_compound as read_borrowed_compound;
18use simdnbt::owned::{Nbt, NbtCompound, NbtList, read as read_owned};
19use uuid::Uuid;

21use super::nbt_load::read_entity_nbt;
22use super::{ENTITIES, Entity, EntityLoadRequest, SharedEntity};
23use crate::world::World;

25/// The entity's save data, compressed; `None` for an entity vanilla never
26/// saves (a player, a lightning bolt, anything already discarded for good).
27#[must_use]
28pub fn serialize_entity(entity: &dyn Entity) -> Option<Vec<u8>> {
29    let mut compound = entity.nbt_for_passenger_save()?;
30    let _ = compound.remove("Passengers");
31    let mut raw = Vec::new();
32    Nbt::new("".into(), compound).write(&mut raw);
33    let mut encoder = GzEncoder::new(Vec::new(), Compression::default());
34    encoder.write_all(&raw).ok()?;
35    encoder.finish().ok()
36}

38/// Builds the entity a blob describes, in `world` but not added to it; the
39/// caller decides where and whether it spawns.
40///
41/// Vanilla parity: `EntityType.create(CompoundTag, Level, LOAD)`, which reads
42/// no passengers. Without `preserve_uuid` the entity gets a fresh UUID, so a
43/// copy can stand beside the original it was taken from.
44#[must_use]
45pub fn deserialize_entity(
46    bytes: &[u8],
47    world: &Arc<World>,
48    preserve_uuid: bool,
49) -> Option<SharedEntity> {
50    let mut raw = Vec::new();
51    GzDecoder::new(bytes).read_to_end(&mut raw).ok()?;
52    let Nbt::Some(root) = read_owned(&mut Cursor::new(raw.as_slice())).ok()? else {
53        return None;
54    };
55    let compound: NbtCompound = root.as_compound();
56    let mut unnamed = Vec::new();
57    compound.write(&mut unnamed);
58    let borrowed = read_borrowed_compound(&mut Cursor::new(unnamed.as_slice())).ok()?;
59    let view = (&borrowed).into();
60    let loaded = read_entity_nbt(&view)?;
61    if !ENTITIES.has_load_factory(loaded.entity_type) {
62        return None;
63    }
64    let position = compound
65        .list("Pos")
66        .and_then(NbtList::doubles)
67        .filter(|values| values.len() >= 3)
68        .map_or(DVec3::ZERO, |values| {
69            DVec3::new(values[0], values[1], values[2])
70        });
71
72    let mut remainder_bytes = Vec::new();
73    loaded.remainder.write(&mut remainder_bytes);
74    let remainder = read_borrowed_compound(&mut Cursor::new(remainder_bytes.as_slice())).ok()?;
75    let uuid = if preserve_uuid {
76        loaded.uuid.unwrap_or_else(Uuid::new_v4)
77    } else {
78        Uuid::new_v4()
79    };
80    Some(ENTITIES.create_and_load_or_raw(
81        EntityLoadRequest {
82            entity_type: loaded.entity_type,
83            position,
84            uuid,
85            velocity: loaded.velocity,
86            rotation: loaded.rotation,
87            fall_distance: loaded.fall_distance,
88            fire_freeze: loaded.fire_freeze,
89            on_ground: loaded.on_ground,
90            save_data: loaded.save_data,
91            world: Arc::downgrade(world),
92        },
93        &remainder,
94    ))
95}

97#[cfg(test)]
98mod tests {
99    use std::sync::Arc;
100
101    use foton_registry::{init_vanilla_registry, vanilla_entities};
102    use glam::DVec3;
103
104    use super::{deserialize_entity, serialize_entity};
105    use crate::entity::entities::PigEntity;
106    use crate::entity::{Entity, Mob, init_entities};
107    use crate::test_support::test_world;
108
109    #[test]
110    fn an_entity_comes_back_with_its_state_and_a_new_uuid() {
111        init_vanilla_registry();
112        init_entities();
113        let world = test_world();
114        let pig = PigEntity::new(
115            &vanilla_entities::PIG,
116            1,
117            DVec3::new(1.0, 64.0, 2.0),
118            Arc::downgrade(world),
119        );
120        pig.set_no_ai(true);
121        assert!(pig.add_tag("stabled".to_owned()));
122
123        let Some(bytes) = serialize_entity(&pig) else {
124            panic!("a pig is saveable");
125        };
126        let Some(copy) = deserialize_entity(&bytes, world, false) else {
127            panic!("the blob should read back");
128        };
129        assert_eq!(copy.entity_type().key, vanilla_entities::PIG.key);
130        assert_ne!(copy.uuid(), pig.uuid());
131        assert_eq!(copy.position(), pig.position());
132        assert_eq!(copy.tags(), vec!["stabled".to_owned()]);
133        assert!(copy.as_mob().is_some_and(Mob::is_no_ai));
134
135        let Some(same) = deserialize_entity(&bytes, world, true) else {
136            panic!("the blob should read back twice");
137        };
138        assert_eq!(same.uuid(), pig.uuid());
139    }
140}
