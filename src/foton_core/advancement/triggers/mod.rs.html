1//! Firing a criterion's trigger for one player.
2//!
3//! Vanilla parity: `CriteriaTriggers` and `SimpleCriterionTrigger.trigger`.
4//! Vanilla keeps a per-player listener map and adds or removes entries as
5//! criteria are met; Foton's advancement set never changes, so the criteria a
6//! trigger *could* award are indexed once for the whole server in
7//! [`super::TRIGGER_INDEX`], and the player narrows that to the ones still
8//! listening.
9//!
10//! A trigger nothing calls awards nothing, which is what vanilla does before
11//! the trigger is invoked -- it is not the same as a criterion that always
12//! passes. The vanilla call sites Foton has not reached are marked with a
13//! `// Not implemented:` comment naming the trigger rather than granting it.
14

27/// Awards every criterion of `trigger_id` whose own conditions `matches`
28/// accepts and whose `player` predicate this player satisfies.
29///
30/// Vanilla parity: `SimpleCriterionTrigger.trigger`, order included -- the
31/// trigger's conditions are tested first, the `player` predicate second, and
32/// nothing is awarded until every candidate has been tested.
33///
34/// The candidates are taken while the progress lock is held and the predicates
35/// are evaluated after it is dropped: a predicate reads the world and the
36/// player's equipment, and an award takes the lock again.
37pub fn fire(
38    player: &Player,
39    trigger_id: &'static str,
40    mut matches: impl FnMut(&'static TriggerInstance) -> bool,
41) {
42    let pending = player.pending_advancement_criteria(trigger_id);
43    if pending.is_empty() {
44        return;
45    }
46
47    // Vanilla: `EntityPredicate.createContext(player, player)` -- the player as
48    // both `THIS_ENTITY` and the source of `ORIGIN`.
49    let context = PredicateContext {
50        player,
51        origin: player.position(),
52        subject: Subject::Player,
53        block_state: None,
54        tool: None,
55    };
56
57    let mut awarded = Vec::new();
58    for reference in pending {
59        let advancement = ADVANCEMENT_TREE.node(reference.node).advancement;
60        let criterion = &advancement.criteria[reference.criterion];
61        if matches(&criterion.trigger) && context.matches_conditions(criterion.trigger.player()) {
62            awarded.push((reference.node, criterion.name));
63        }
64    }
65
66    for (node, criterion) in awarded {
67        player.award_advancement_criterion(node, criterion);
68    }
69}
