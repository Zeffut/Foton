13/// Fired once per player tick.
14///
15/// Vanilla parity: `CriteriaTriggers.TICK`, invoked from `ServerPlayer.tick`.
16/// `PlayerTrigger` carries nothing but its `player` predicate, so that
17/// predicate is the whole test -- which is how `nether/all_effects` and the
18/// other "be in this state right now" advancements are written.
19pub fn tick(player: &Player) {
20    fire(player, "minecraft:tick", |_| true);
21}
22
23/// Fired once every twenty player ticks.
24///
25/// Vanilla parity: `CriteriaTriggers.LOCATION`, invoked from
26/// `ServerPlayer.doTick` under `this.tickCount % 20 == 0`. It shares
27/// `PlayerTrigger` with [`tick`]; the twenty-tick spacing is the only
28/// difference, and it is what keeps the biome and structure checks off the
29/// per-tick path.
30pub fn location(player: &Player) {
31    fire(player, "minecraft:location", |_| true);
32}

34/// Vanilla parity: `CriteriaTriggers.ENTER_BLOCK`, fired from the
35/// `onInsideBlock` of every block an entity is standing in.
36pub fn enter_block(player: &Player, state: BlockStateId) {
37    block_state_trigger(player, "minecraft:enter_block", state);
38}

46/// Vanilla parity: `EnterBlockTrigger.TriggerInstance.matches`. A named block
47/// that does not match fails; an absent state predicate accepts every state.
48fn block_state_trigger(player: &Player, trigger_id: &'static str, state: BlockStateId) {
49    fire(player, trigger_id, |instance| {
50        let (wanted_block, wanted_state) = match instance {
51            TriggerInstance::EnterBlock { block, state, .. }
52            | TriggerInstance::SlideDownBlock { block, state, .. } => (block, *state),
53            _ => return false,
54        };
55        if let Some(wanted_block) = wanted_block
56            && state.get_block().key != *wanted_block
57        {
58            return false;
59        }
60        state_properties_match(wanted_state, state)
61    });
62}

74/// Vanilla parity: `ItemUsedOnLocationTrigger.trigger`, whose loot context is
75/// the block center as `ORIGIN`, the block state as `BLOCK_STATE` and the item
76/// as `TOOL` -- the three things a `location` predicate reads. The `player`
77/// predicate is checked by [`fire`] against a *different* context, the player
78/// own one, exactly as vanilla does.
79fn item_used_on_location(
80    player: &Player,
81    trigger_id: &'static str,
82    pos: BlockPos,
83    state: BlockStateId,
84    tool: &ItemStack,
85) {
86    let context = PredicateContext {
87        player,
88        origin: DVec3::new(
89            f64::from(pos.x()) + 0.5,
90            f64::from(pos.y()) + 0.5,
91            f64::from(pos.z()) + 0.5,
92        ),
93        subject: Subject::Player,
94        block_state: Some(state),
95        tool: Some(tool),
96    };
97    fire(player, trigger_id, |instance| {
98        let location = match instance {
99            TriggerInstance::PlacedBlock { location, .. }
100            | TriggerInstance::ItemUsedOnBlock { location, .. }
101            | TriggerInstance::AllayDropItemOnBlock { location, .. } => *location,
102            _ => return false,
103        };
104        context.matches_conditions(location)
105    });
106}

108/// Vanilla parity: `CriteriaTriggers.CHANGED_DIMENSION`.
109pub fn changed_dimension(player: &Player, from: &Identifier, to: &Identifier) {
110    fire(player, "minecraft:changed_dimension", |instance| {
111        let TriggerInstance::ChangedDimension {
112            from: wanted_from,
113            to: wanted_to,
114            ..
115        } = instance
116        else {
117            return false;
118        };
119        wanted_from.as_ref().is_none_or(|wanted| wanted == from)
120            && wanted_to.as_ref().is_none_or(|wanted| wanted == to)
121    });
122}

129/// Vanilla parity: `CriteriaTriggers.GENERATE_LOOT`, whose registry name is
130/// `minecraft:player_generates_container_loot`.
131pub fn player_generates_container_loot(player: &Player, loot_table: &Identifier) {
132    fire(
133        player,
134        "minecraft:player_generates_container_loot",
135        |instance| {
136            let TriggerInstance::PlayerGeneratesContainerLoot {
137                loot_table: wanted, ..
138            } = instance
139            else {
140                return false;
141            };
142            wanted == loot_table
143        },
144    );
145}

148pub fn construct_beacon(player: &Player, levels: i32) {
