1//! Evaluating the predicates an advancement criterion asks about.
2//!
3//! Vanilla parity: `net.minecraft.advancements.predicates` plus the
4//! `LootItemCondition`s a `ContextAwarePredicate` is built from.
5//! [`foton_registry::advancement::predicate`] holds the plain data; this is the
6//! half that needs a world, a player and the entity a trigger fired for.
7//!
8//! Vanilla evaluates every one of these against a `LootContext` built by
9//! `EntityPredicate.createContext` or by the trigger itself, so
10//! [`PredicateContext`] carries exactly the `LootContextParams` an advancement
11//! ever reads: `THIS_ENTITY`, `ORIGIN`, `BLOCK_STATE` and `TOOL`, plus the
12//! player whose level and position vanilla takes them from.
13//!
14//! Anything Foton cannot answer fails closed. A predicate that silently passes
15//! hands out an advancement vanilla gates behind a state nobody checked, which
16//! is strictly worse than one that never fires -- every such gap is marked with
17//! a `// Not implemented:` comment naming the vanilla check it stands in for.
18

61/// Vanilla parity: `Mth.floor(double)`, which is how `BlockPos.containing`
62/// turns a predicate's raw coordinates into the block that contains them.
63///
64/// Flooring rather than truncating is what puts `-0.5` in the block at `-1`.
65const fn block_coord(value: f64) -> i32 {
66    value.floor() as i32
67}

69/// Vanilla parity: what `LootContextParams.THIS_ENTITY` resolves to.
70///
71/// [`Subject::None`] is vanilla's null, which fails every entity predicate --
72/// `EntityPredicate.matches` opens on `entity == null ? false : ...`, so even an
73/// empty predicate rejects it.
74pub enum Subject<'a> {
75    /// No subject at all, vanilla's null `THIS_ENTITY`.
76    None,
77    /// The player the trigger fired for.
78    Player,
79    /// Some other entity the trigger handed over.
80    Entity(&'a dyn Entity),
81}

83/// Vanilla parity: the `LootContext` an advancement predicate is evaluated with.
84pub struct PredicateContext<'a> {
85    /// The player the trigger fired for. Vanilla takes the level and, for most
86    /// triggers, the origin from them.
87    pub player: &'a Player,
88    /// Vanilla's `LootContextParams.ORIGIN`. The player's own position for most
89    /// triggers; the block center for the location triggers.
90    pub origin: DVec3,
91    /// Vanilla's `LootContextParams.THIS_ENTITY`.
92    pub subject: Subject<'a>,
93    /// Vanilla's `LootContextParams.BLOCK_STATE`, set by the location triggers.
94    pub block_state: Option<BlockStateId>,
95    /// Vanilla's `LootContextParams.TOOL`, set by the location triggers.
96    pub tool: Option<&'a ItemStack>,
97}

99impl PredicateContext<'_> {
100    /// The entity a [`Subject`] names, `None` for vanilla's null.
101    fn subject_entity<'s>(&'s self, subject: &'s Subject<'_>) -> Option<&'s dyn Entity> {
102        match subject {
103            Subject::None => None,
104            Subject::Player => Some(self.player as &dyn Entity),
105            Subject::Entity(entity) => Some(*entity),
106        }
107    }

109    /// Vanilla parity: `ContextAwarePredicate.matches`.
110    ///
111    /// Every term must pass; an empty list passes, which is what makes an absent
112    /// `player` field on a criterion accept everyone.
113    #[must_use]
114    pub fn matches_conditions(&self, terms: ContextAwarePredicate) -> bool {
115        terms.iter().all(|term| self.matches_condition(term))
116    }

118    /// Vanilla parity: one `LootItemCondition.test`.
119    fn matches_condition(&self, term: &ConditionTerm) -> bool {
120        match term {
121            // Vanilla `LootItemEntityPropertyCondition.test`, whose entity
122            // target is always `THIS` in advancement data.
123            ConditionTerm::EntityProperties(predicate) => {
124                self.matches_entity(predicate, &self.subject)
125            }
126            // Vanilla `LocationCheck.test`, which adds the block offset to the
127            // raw `ORIGIN` doubles before flooring.
128            ConditionTerm::LocationCheck {
129                offset_x,
130                offset_y,
131                offset_z,
132                predicate,
133            } => self.matches_location(
134                predicate,
135                self.origin.x + f64::from(*offset_x),
136                self.origin.y + f64::from(*offset_y),
137                self.origin.z + f64::from(*offset_z),
138            ),
139            // Vanilla `MatchTool.test` is `tool != null && predicate.test(tool)`:
140            // a missing TOOL fails outright, even for an empty predicate.
141            ConditionTerm::MatchTool(predicate) => {
142                self.tool.is_some_and(|tool| item_matches(predicate, tool))
143            }
144            // Vanilla `LootItemBlockStatePropertyCondition.test`: no BLOCK_STATE
145            // fails, then the state's block and its properties must both match.
146            ConditionTerm::BlockStateProperty { block, properties } => {
147                self.block_state.is_some_and(|state| {
148                    state.get_block().key == *block && state_properties_match(properties, state)
149                })
150            }
151            ConditionTerm::AnyOf(terms) => terms.iter().any(|term| self.matches_condition(term)),
152            ConditionTerm::AllOf(terms) => terms.iter().all(|term| self.matches_condition(term)),
153            ConditionTerm::Inverted(term) => !self.matches_condition(term),
154        }
155    }

157    /// Vanilla parity: `EntityPredicate.matches(level, position, entity)`.
158    #[must_use]
159    pub fn matches_entity(&self, predicate: &EntityPredicate, subject: &Subject<'_>) -> bool {
160        self.entity_matches(predicate, self.subject_entity(subject))
161    }
162
163    /// The body of [`Self::matches_entity`], on a resolved entity.
164    ///
165    /// Vanilla's sub-predicate map is a pure conjunction, and a null subject
166    /// fails all of it -- `EntityPredicate.matches` never even reaches the
167    /// combined part.
168    fn entity_matches(&self, predicate: &EntityPredicate, entity: Option<&dyn Entity>) -> bool {
169        let Some(entity) = entity else {
170            return false;
171        };
172
173        if let Some(types) = &predicate.entity_type
174            && !entity_type_in_set(types, entity.entity_type())
175        {
176            return false;
177        }
178
179        let position = entity.position();
180        if let Some(location) = predicate.location
181            && !self.matches_location(location, position.x, position.y, position.z)
182        {
183            return false;
184        }
185
186        if let Some(stepping_on) = predicate.stepping_on
187            && !self.matches_stepping_on(stepping_on, entity)
188        {
189            return false;
190        }
191
192        if let Some(distance) = &predicate.distance
193            && !distance_matches(distance, self.origin, position)
194        {
195            return false;
196        }
197
198        if let Some(flags) = &predicate.flags
199            && !flags_match(*flags, entity)
200        {
201            return false;
202        }
203
204        if let Some(equipment) = predicate.equipment
205            && !equipment_matches(equipment, entity)
206        {
207            return false;
208        }
209
210        if !predicate
211            .components
212            .iter()
213            .all(|component| entity_component_matches(component, entity))
214        {
215            return false;
216        }
217
218        self.entity_relations_match(predicate, entity)
219    }

221    /// The half of [`Self::entity_matches`] that walks off the subject: its
222    /// vehicle, its passengers, what it is looking at and, for a bolt, how many
223    /// blocks it lit.
224    fn entity_relations_match(&self, predicate: &EntityPredicate, entity: &dyn Entity) -> bool {
225        // Vanilla `VehiclePredicate` feeds `entity.getVehicle()` straight into
226        // `EntityPredicate.matches`, so no vehicle is a null subject: false.
227        if let Some(vehicle) = predicate.vehicle {
228            let Some(ridden) = entity.vehicle() else {
229                return false;
230            };
231            if !self.entity_matches(vehicle, Some(ridden.as_ref())) {
232                return false;
233            }
234        }
235
236        // Vanilla `PassengerPredicate` returns on the first direct passenger
237        // that matches, and false when there are none.
238        if let Some(passenger) = predicate.passenger
239            && !entity
240                .passengers()
241                .iter()
242                .any(|rider| self.entity_matches(passenger, Some(rider.as_ref())))
243        {
244            return false;
245        }
246
247        // Not implemented: `PlayerPredicate.lookingAt`, which needs
248        // `ProjectileUtil.getEntityHitResult` along 100 blocks of the player's
249        // view vector plus `ServerPlayer.hasLineOfSight`. Foton's equivalent
250        // (`entity::projectile::get_entity_hit_result`) is private to that
251        // module, so this cannot be answered from here and must not pass.
252        if predicate.looking_at.is_some() {
253            return false;
254        }
255
256        if let Some(bounds) = predicate.lightning_blocks_set_on_fire {
257            // Vanilla `LightningBoltPredicate` rejects anything that is not a
258            // `LightningBolt` before it looks at the count.
259            let Some(bolt) = entity.downcast_ref::<LightningBoltEntity>() else {
260                return false;
261            };
262            if !bounds.matches(bolt.blocks_set_on_fire()) {
263                return false;
264            }
265        }
266
267        true
268    }

270    /// Vanilla parity: `SteppingOnPredicate.matches`.
271    ///
272    /// The position is `Vec3.atCenterOf(entity.getOnPos())`. Vanilla's
273    /// `getOnPos()` is `getOnPos(1.0E-5F)`, which is exactly the offset Foton's
274    /// [`Entity::on_pos`] treats as "the supporting block itself".
275    fn matches_stepping_on(&self, predicate: &LocationPredicate, entity: &dyn Entity) -> bool {
276        if !entity.on_ground() {
277            return false;
278        }
279        let Some(pos) = entity.on_pos(1.0e-5) else {
280            return false;
281        };
282        self.matches_location(
283            predicate,
284            f64::from(pos.x()) + 0.5,
285            f64::from(pos.y()) + 0.5,
286            f64::from(pos.z()) + 0.5,
287        )
288    }

290    /// Vanilla parity: `LocationPredicate.matches(level, x, y, z)`.
291    ///
292    /// The coordinate bounds are checked on the raw doubles, before anything is
293    /// floored into a block position. `biomes`, `structures` and `smokey`
294    /// additionally require the chunk to be loaded and are false when it is not.
295    #[must_use]
296    pub fn matches_location(&self, predicate: &LocationPredicate, x: f64, y: f64, z: f64) -> bool {
297        if !predicate.x.matches(x) || !predicate.y.matches(y) || !predicate.z.matches(z) {
298            return false;
299        }
300
301        let needs_level = predicate.dimension.is_some()
302            || predicate.biomes.is_some()
303            || predicate.structures.is_some()
304            || predicate.smokey.is_some()
305            || predicate.block.is_some();
306        if !needs_level {
307            return true;
308        }
309
310        // Vanilla always has a `ServerLevel` here; a player without one cannot
311        // answer any of the remaining keys, so they all fail.
312        let Some(level) = self.player.level() else {
313            return false;
314        };
315
316        if let Some(dimension) = &predicate.dimension
317            && level.key != *dimension
318        {
319            return false;
320        }
321
322        // Not implemented: `level.structureManager().getStructureWithPieceAt`.
323        // Foton has no structure-at-position lookup at all, so a `structures`
324        // check cannot be answered and must not pass.
325        if predicate.structures.is_some() {
326            return false;
327        }
328
329        let pos = BlockPos::new(block_coord(x), block_coord(y), block_coord(z));
330        // `loaded_block_state` is `None` for exactly the positions vanilla's
331        // `level.isLoaded(pos)` rejects.
332        let loaded_state = level.loaded_block_state(pos.x(), pos.y(), pos.z());
333
334        if let Some(biomes) = &predicate.biomes {
335            let Some(biome) = level.loaded_biome(pos.x(), pos.y(), pos.z()) else {
336                return false;
337            };
338            if !biome_in_set(biomes, biome) {
339                return false;
340            }
341        }
342
343        if let Some(smokey) = predicate.smokey
344            && (loaded_state.is_none() || smokey != is_smokey_pos(&level, pos))
345        {
346            return false;
347        }
348
349        if let Some(block) = &predicate.block {
350            let Some(state) = loaded_state else {
351                return false;
352            };
353            if !block_state_matches(block, state) {
354                return false;
355            }
356        }
357
358        true
359    }

361    /// Vanilla parity: `DamageSourcePredicate.matches(level, position, source)`.
362    #[must_use]
363    pub fn matches_damage_source(
364        &self,
365        predicate: &DamageSourcePredicate,
366        source: &DamageSource,
367    ) -> bool {
368        // Vanilla `TagPredicate.matches` is `holder.is(tag) == expected`.
369        if !predicate
370            .tags
371            .iter()
372            .all(|tag| source.is(&tag.id) == tag.expected)
373        {
374            return false;
375        }
376
377        if let Some(direct) = predicate.direct_entity
378            && !self.matches_entity_by_id(direct, source.direct_entity_id)
379        {
380            return false;
381        }
382
383        if let Some(causing) = predicate.source_entity
384            && !self.matches_entity_by_id(causing, source.causing_entity_id)
385        {
386            return false;
387        }
388
389        true
390    }

392    /// Vanilla parity: `DamagePredicate.matches(player, source, dealt, taken, blocked)`.
393    ///
394    /// Vanilla's `source_entity` field is not modeled by
395    /// [`foton_registry::advancement::predicate::DamagePredicate`] because no
396    /// vanilla advancement uses it; the nested `type.source_entity` is what the
397    /// data reaches for instead.
398    #[must_use]
399    pub fn matches_damage(
400        &self,
401        predicate: &DamagePredicate,
402        source: &DamageSource,
403        dealt: f32,
404        taken: f32,
405        blocked: bool,
406    ) -> bool {
407        if !predicate.dealt.matches(f64::from(dealt)) || !predicate.taken.matches(f64::from(taken))
408        {
409            return false;
410        }
411        if let Some(expected) = predicate.blocked
412            && expected != blocked
413        {
414            return false;
415        }
416        predicate
417            .source
418            .is_none_or(|source_predicate| self.matches_damage_source(source_predicate, source))
419    }

421    /// Resolves a [`DamageSource`]'s stored entity id and tests it.
422    ///
423    /// Foton stores ids rather than references, so an id that no longer resolves
424    /// is vanilla's null entity: it fails the predicate.
425    fn matches_entity_by_id(&self, predicate: &EntityPredicate, id: Option<i32>) -> bool {
426        let Some(id) = id else {
427            return false;
428        };
429        let Some(level) = self.player.level() else {
430            return false;
431        };
432        let Some(entity) = level.get_entity_by_id(id) else {
433            return false;
434        };
435        self.entity_matches(predicate, Some(entity.as_ref()))
436    }
437}

439/// Vanilla parity: `EntityFlagsPredicate.matches`.
440///
441/// Note the asymmetry vanilla actually has: `is_flying` reads false for a
442/// non-living entity and is then *compared*, so `is_flying: false` accepts an
443/// arrow -- while `is_baby` is skipped entirely for a non-living entity and so
444/// passes whatever it asked for.
445fn flags_match(flags: EntityFlagsPredicate, entity: &dyn Entity) -> bool {
446    let living = entity.as_living_entity();
447
448    if let Some(expected) = flags.is_on_fire
449        && entity.is_on_fire() != expected
450    {
451        return false;
452    }
453    if let Some(expected) = flags.is_sneaking
454        && entity.is_crouching() != expected
455    {
456        return false;
457    }
458    // Vanilla reads `Entity.isSprinting()` off the shared entity flags; Foton
459    // only tracks sprinting on living entities, so anything else is not
460    // sprinting -- which is what the shared flag would say for it anyway.
461    if let Some(expected) = flags.is_sprinting
462        && living.is_some_and(LivingEntity::is_sprinting) != expected
463    {
464        return false;
465    }
466    if let Some(expected) = flags.is_swimming
467        && entity.is_swimming() != expected
468    {
469        return false;
470    }
471    // Vanilla: `entity instanceof LivingEntity living && (living.isFallFlying()
472    // || living instanceof Player player && player.getAbilities().flying)`.
473    if let Some(expected) = flags.is_flying {
474        let flying = living.is_some_and(LivingEntity::is_fall_flying) || entity.is_flying_player();
475        if flying != expected {
476            return false;
477        }
478    }
479    // Vanilla skips the whole check for a non-living entity rather than
480    // comparing against a default, so a non-living subject passes it.
481    if let Some(expected) = flags.is_baby
482        && let Some(living) = living
483        && living.is_baby() != expected
484    {
485        return false;
486    }
487
488    true
489}

491/// Vanilla parity: `EntityEquipmentPredicate.matches`.
492///
493/// A non-living entity fails, even for an all-empty predicate -- vanilla's
494/// `instanceof LivingEntity` check has an `else return false`.
495fn equipment_matches(predicate: &EntityEquipmentPredicate, entity: &dyn Entity) -> bool {
496    let Some(living) = entity.as_living_entity() else {
497        return false;
498    };
499
500    let wanted = [
501        &predicate.head,
502        &predicate.chest,
503        &predicate.legs,
504        &predicate.feet,
505        &predicate.body,
506        &predicate.mainhand,
507        &predicate.offhand,
508    ];
509
510    wanted.iter().zip(EQUIPMENT_SLOTS).all(|(item, slot)| {
511        item.as_ref()
512            .is_none_or(|item| item_matches(item, &living.get_item_by_slot(slot)))
513    })
514}

516/// Vanilla parity: one entry of the `minecraft:components` entity sub-predicate.
517///
518/// Vanilla compares the entity's data component against the value the predicate
519/// carries; Foton reaches the three variants vanilla advancement data actually
520/// asks for and fails closed on anything else.
521fn entity_component_matches(component: &EntityComponentMatch, entity: &dyn Entity) -> bool {
522    match component.component {
523        "minecraft:cat/variant" => entity
524            .downcast_ref::<CatEntity>()
525            .is_some_and(|cat| cat.variant().key == component.value),
526        "minecraft:wolf/variant" => entity
527            .downcast_ref::<WolfEntity>()
528            .is_some_and(|wolf| wolf.variant().key == component.value),
529        "minecraft:frog/variant" => entity
530            .as_living_entity()
531            .and_then(LivingEntity::frog_loot_variant)
532            .is_some_and(|variant| *variant == component.value),
533        // Not implemented: every other entity data component. Vanilla's
534        // `EntityPartialComponentsPredicate` is open-ended and Foton has no
535        // entity-wide component map to compare against, so an unmodeled key
536        // must fail rather than pass unchecked.
537        _ => false,
538    }
539}

541/// Vanilla parity: `DistancePredicate.matches`.
542///
543/// Vanilla narrows every delta to `float` before comparing and sums the squares
544/// in `float` too, so the rounding is part of the contract: a check written as
545/// `absolute: {min: 30}` accepts positions a `double` computation would reject.
546fn distance_matches(predicate: &DistancePredicate, origin: DVec3, target: DVec3) -> bool {
547    let dx = (origin.x - target.x) as f32;
548    let dy = (origin.y - target.y) as f32;
549    let dz = (origin.z - target.z) as f32;
550
551    if !predicate.x.matches(f64::from(dx.abs()))
552        || !predicate.y.matches(f64::from(dy.abs()))
553        || !predicate.z.matches(f64::from(dz.abs()))
554    {
555        return false;
556    }
557
558    predicate
559        .horizontal
560        .matches_sqr(f64::from(dx * dx + dz * dz))
561        && predicate
562            .absolute
563            .matches_sqr(f64::from(dx * dx + dy * dy + dz * dz))
564}

598/// Vanilla parity: `ItemPredicate.test`.
599///
600/// An empty stack reports as `minecraft:air` with count `0`, so an
601/// [`ItemPredicate`] that asks nothing matches [`ItemStack::empty`] -- which is
602/// what lets an absent `item` field on a criterion accept a bare hand.
603#[must_use]
604pub fn item_matches(predicate: &ItemPredicate, stack: &ItemStack) -> bool {
605    if let Some(items) = &predicate.items
606        && !item_in_set(items, stack.item())
607    {
608        return false;
609    }
610    if !predicate.count.matches(stack.count()) {
611        return false;
612    }
613    item_components_match(predicate, stack)
614}

616/// Vanilla parity: `DataComponentMatchers.test`, narrowed to the checks vanilla
617/// advancement data uses.
618///
619/// The exact half compares against the stack's *effective* component value --
620/// its patch, or the item prototype when the patch says nothing -- and an
621/// absent component fails, because `Objects.equals(expected, null)` is false.
622fn item_components_match(predicate: &ItemPredicate, stack: &ItemStack) -> bool {
623    let components = &predicate.components;
624
625    if let Some(damage) = components.damage
626        && stack.get(DAMAGE).copied() != Some(damage)
627    {
628        return false;
629    }
630
631    if let Some(layers) = components.banner_patterns {
632        let Some(actual) = stack.get(BANNER_PATTERNS) else {
633            return false;
634        };
635        if !banner_patterns_equal(layers, actual) {
636            return false;
637        }
638    }
639
640    if let Some(key) = components.item_name_translate {
641        let Some(name) = stack.get(ITEM_NAME) else {
642            return false;
643        };
644        if !is_translatable_with_key(name, key) {
645            return false;
646        }
647    }
648
649    // Vanilla `JukeboxPlayablePredicate` with no `song` is a bare presence
650    // check on the component, which is the only shape vanilla data uses.
651    if predicate.jukebox_playable && !stack.has(JUKEBOX_PLAYABLE) {
652        return false;
653    }
654
655    predicate
656        .enchantments
657        .iter()
658        .all(|enchantment| enchantment_contained_in(enchantment, stack))
659}
660
661/// Vanilla parity: comparing a `minecraft:banner_patterns` component for exact
662/// equality, layer by layer and in order.
663fn banner_patterns_equal(expected: &[BannerPatternLayer], actual: &BannerPatternLayers) -> bool {
664    let actual = actual.layers();
665    actual.len() == expected.len()
666        && expected.iter().zip(actual).all(|(want, have)| {
667            have.color().serialized_name() == want.color
668                && have
669                    .pattern()
670                    .as_reference()
671                    .is_some_and(|pattern| pattern.key == want.pattern)
672        })
673}
674
675/// Whether `component` is a translatable component with key `key`.
676///
677/// Vanilla compares the whole `minecraft:item_name` value, style included; the
678/// generator only ever sees a bare translate key, so the key is the whole check
679/// Foton can express. The one vanilla use (the ominous banner's name) is unique
680/// enough for that to be the same answer.
681fn is_translatable_with_key(component: &TextComponent, key: &str) -> bool {
682    matches!(&component.content, Content::Translate(message) if message.key.as_ref() == key)
683}
684
685/// Vanilla parity: `EnchantmentPredicate.containedIn`.
686fn enchantment_contained_in(predicate: &EnchantmentPredicate, stack: &ItemStack) -> bool {
687    let Some(enchantments) = stack.get_enchantments() else {
688        return false;
689    };
690
691    match &predicate.enchantments {
692        // Vanilla `matchesEnchantment`: an enchantment at level zero is absent,
693        // and any other level has to fall inside the bounds.
694        Some(RegistrySet::Entries(keys)) => keys.iter().any(|key| {
695            let level = enchantments.get_level(key);
696            level != 0 && predicate.levels.matches(level as i32)
697        }),
698        Some(RegistrySet::Tag(tag)) => enchantments.iter().any(|(key, level)| {
699            *level != 0
700                && predicate.levels.matches(*level as i32)
701                && REGISTRY
702                    .enchantments
703                    .by_key(key)
704                    .is_some_and(|enchantment| REGISTRY.enchantments.is_in_tag(enchantment, tag))
705        }),
706        // No `enchantments` field: bounded levels ask whether anything on the
707        // stack sits inside them, and unbounded ones ask only for any
708        // enchantment at all.
709        None if predicate.levels.is_any() => !enchantments.is_empty(),
710        None => enchantments
711            .iter()
712            .any(|(_, level)| predicate.levels.matches(*level as i32)),
713    }
714}

716/// Vanilla parity: `BlockPredicate.matchesState`, the half that needs no world.
717///
718/// The `nbt` and `components` halves of vanilla's predicate are not modeled --
719/// no vanilla advancement uses them, and the generator refuses one that does.
720#[must_use]
721pub fn block_state_matches(predicate: &BlockPredicate, state: BlockStateId) -> bool {
722    if let Some(blocks) = &predicate.blocks
723        && !block_in_set(blocks, state.get_block())
724    {
725        return false;
726    }
727    state_properties_match(predicate.state, state)
728}
729
730/// Vanilla parity: `StatePropertiesPredicate.matches`.
731///
732/// A property the block does not have fails: vanilla's `PropertyMatcher.match`
733/// opens on `property != null`, so an unknown name is never vacuously true.
734#[must_use]
735pub fn state_properties_match(properties: &[StatePropertyMatch], state: BlockStateId) -> bool {
736    properties
737        .iter()
738        .all(|property| state.get_property_str(property.name).as_deref() == Some(property.value))
739}
