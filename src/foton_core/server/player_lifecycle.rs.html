27    pub(super) async fn load_join_domain(&self, player: &Player) -> Result<String, String> {
28        match self
29            .player_data_storage
30            .load_global(player.gameprofile.id)
31            .await
32        {
33            Ok(Some(global)) if self.worlds.has_domain(&global.last_active_domain) => {
34                self.publish_global_player_data(player.gameprofile.id, global.clone());
35                Ok(global.last_active_domain)
36            }
37            Ok(Some(global)) => {
38                self.publish_global_player_data(player.gameprofile.id, global.clone());
39                log::warn!(
40                    "Player {} last active domain {} no longer exists, using default domain",
41                    player.gameprofile.name,
42                    global.last_active_domain
43                );
44                Ok(self.worlds.default_domain().to_owned())
45            }
46            Ok(None) => {
47                let now = chrono::Utc::now().timestamp_millis();
48                self.publish_global_player_data(
49                    player.gameprofile.id,
50                    super::GlobalPlayerData {
51                        last_active_domain: self.worlds.default_domain().to_owned(),
52                        first_played: now,
53                        last_played: now,
54                        statistics: Vec::new(),
55                        whitelisted: false,
56                    },
57                );
58                Ok(self.worlds.default_domain().to_owned())
59            }
60            Err(e) => Err(format!("failed to load global player data: {e}")),
61        }
62    }

76    pub(in crate::server) async fn load_unprepared_domain_player_state(
77        &self,
78        player: &Player,
79        target_domain: &str,
80        explicit_target_world: Option<Arc<World>>,
81    ) -> Result<UnpreparedDomainPlayerState, String> {
82        let has_explicit_target = explicit_target_world.is_some();
83        let default_world = self
84            .worlds
85            .default_world(target_domain)
86            .ok_or_else(|| format!("domain {target_domain} has no default world"))?;
87
88        match self
89            .player_data_storage
90            .load_domain(target_domain, player.gameprofile.id)
91            .await
92        {
93            Ok(Some(saved_data)) => {
94                let saved_world = self.resolve_saved_world(
95                    &saved_data.world,
96                    target_domain,
97                    explicit_target_world.as_ref().map(|world| &world.key),
98                    &player.gameprofile.name,
99                );
100                let (world, data) = if let Some(saved_world) = saved_world {
101                    (
102                        saved_world,
103                        UnpreparedDomainPlayerData::SavedRestored {
104                            data: Box::new(saved_data),
105                        },
106                    )
107                } else {
108                    (
109                        explicit_target_world.unwrap_or(default_world),
110                        UnpreparedDomainPlayerData::SavedWithoutLocation {
111                            data: Box::new(saved_data),
112                        },
113                    )
114                };
115                log::info!("Loaded saved data for player {}", player.gameprofile.name);
116                Ok(UnpreparedDomainPlayerState {
117                    world,
118                    data,
119                    explicit_target: has_explicit_target,
120                })
121            }
122            Ok(None) => {
123                log::debug!(
124                    "No saved data for player {} in domain {}, using defaults",
125                    player.gameprofile.name,
126                    target_domain
127                );
128                Ok(UnpreparedDomainPlayerState {
129                    world: explicit_target_world.unwrap_or(default_world),
130                    data: UnpreparedDomainPlayerData::FirstVisit,
131                    explicit_target: has_explicit_target,
132                })
133            }
134            Err(e) => Err(format!(
135                "failed to load domain player data for {} in domain {}: {e}",
136                player.gameprofile.name, target_domain
137            )),
138        }
139    }

141    async fn prepare_domain_player_state(
142        &self,
143        target_domain: &str,
144        state: UnpreparedDomainPlayerState,
145    ) -> Result<DomainPlayerState, String> {
146        let UnpreparedDomainPlayerState {
147            mut world,
148            explicit_target,
149            data,
150        } = state;
151        let (data, spawn_position) = match data {
152            UnpreparedDomainPlayerData::SavedRestored { data } => {
153                let spawn_position = DVec3::new(data.pos[0], data.pos[1], data.pos[2]);
154                (DomainPlayerData::SavedRestored { data }, spawn_position)
155            }
156            UnpreparedDomainPlayerData::SavedWithoutLocation { data } => {
157                let (spawn_world, spawn) = self
158                    .prepare_target_spawn(target_domain, explicit_target, &world)
159                    .await?;
160                world = spawn_world;
161                (
162                    DomainPlayerData::SavedWithoutLocation { data, spawn },
163                    spawn.position,
164                )
165            }
166            UnpreparedDomainPlayerData::FirstVisit => {
167                let (spawn_world, spawn) = self
168                    .prepare_target_spawn(target_domain, explicit_target, &world)
169                    .await?;
170                world = spawn_world;
171                (DomainPlayerData::FirstVisit { spawn }, spawn.position)
172            }
173        };
174        let spawn_chunk_request = world.prepare_player_spawn_chunks(spawn_position).await?;
175        Ok(DomainPlayerState {
176            world,
177            data,
178            spawn_chunk_request,
179        })
180    }

194        Ok((world, spawn))

227    fn resolve_saved_world(
228        &self,
229        saved_world: &str,
230        target_domain: &str,
231        explicit_target_world: Option<&Identifier>,
232        player_name: &str,
233    ) -> Option<Arc<World>> {
234        let Ok(saved_world_key) = saved_world.parse::<Identifier>() else {
235            log::warn!(
236                "Saved world {saved_world} for player {player_name} is invalid, using target spawn"
237            );
238            return None;
239        };
240        if saved_world_key.namespace.as_ref() != target_domain {
241            log::warn!(
242                "Saved world {saved_world_key} for player {player_name} is outside target domain {target_domain}, using target spawn"
243            );
244            return None;
245        }
246        if let Some(explicit_target_world) = explicit_target_world
247            && explicit_target_world != &saved_world_key
248        {
249            return None;
250        }
251        let Some(saved_world) = self.worlds.get_owned(&saved_world_key) else {
252            log::warn!(
253                "Saved world {saved_world_key} for player {player_name} is missing, using target spawn"
254            );
255            return None;
256        };
257        Some(saved_world)
258    }

260    pub(super) fn apply_domain_player_state(player: &Arc<Player>, state: &DomainPlayerState) {
261        // Vanilla parity: the `player.getStats().markAllDirty()` of
262        // `PlayerList.placeNewPlayer`. Without it the statistics screen would
263        // only ever show what moved since this login.
264        player.mark_all_statistics_dirty();
265        match &state.data {
266            DomainPlayerData::SavedRestored { data } => {
267                data.apply_to_player(player);
268            }
269            DomainPlayerData::SavedWithoutLocation { data, spawn } => {
270                apply_default_spawn(player, &state.world, *spawn);
271                data.apply_to_player_without_location(player);
272            }
273            DomainPlayerData::FirstVisit { spawn } => {
274                player.reset_domain_data_for_first_visit();
275                apply_default_spawn(player, &state.world, *spawn);
276            }
277        }
278        let mut spawn_event = PlayerSpawnLocationEvent::new(
279            player.gameprofile.id,
280            state.world.key.to_string(),
281            player.position().to_array(),
282            player.rotation(),
283        );
284        player.fire_event(&mut spawn_event);
285        if spawn_event.world() == state.world.key.to_string() {
286            player
287                .base()
288                .set_position_local(glam::DVec3::from_array(spawn_event.position()));
289            player.set_rotation(spawn_event.rotation());
290        }
291    }

294        &self,
295        player: &Player,
296        state: &DomainPlayerState,
297    ) -> PreparedDomainRestores {
298        let target_domain = state.world.domain();
299        let ender_pearls = Self::ender_pearls_to_restore(state)

301            .filter_map(|payload| {
302                self.resolve_pearl_world(&payload.world, target_domain, player)

304            })
305            .collect();
306        PreparedDomainRestores {
307            target_world: Arc::clone(&state.world),
308            root_vehicle: Self::root_vehicle_to_restore(state),
309            ender_pearls,
310        }
311    }

330    pub(super) fn schedule_domain_restores(
331        &self,
332        player: &Arc<Player>,
333        residence_token: DomainResidenceToken,
334        restores: PreparedDomainRestores,
335    ) {
336        if let Some(root_vehicle) = restores.root_vehicle {
337            if let Some(job) = RootVehicleRestoreJob::new(
338                Arc::clone(player),
339                Arc::clone(&restores.target_world),
340                &root_vehicle,
341                residence_token,
342            ) {
343                self.jobs.spawn(job);
344            } else {
345                player.take_matching_pending_root_vehicle(
346                    residence_token,
347                    &restores.target_world,
348                    root_vehicle.attach,
349                    root_vehicle.entity.uuid,
350                );
351            }
352        }
353
354        for restore in restores.ender_pearls {
355            let pearl_uuid = Uuid::from_bytes(restore.payload.entity.uuid);
356            if let Some(job) = EnderPearlRestoreJob::new(
357                Arc::clone(player),
358                restore.world,
359                restore.payload.entity,
360                residence_token,
361            ) {
362                self.jobs.spawn(job);
363            } else {
364                player.discard_pending_ender_pearl(residence_token, pearl_uuid);
365            }
366        }
367    }

407        };

411                 {expected_domain}, skipping",
412                world.domain()
413            );
414            return None;
415        }
416        Some(world.clone())
417    }

509    /// Returns a sample of up to 12 online players for the server list ping.
510    #[must_use]
511    pub fn player_sample(&self) -> Vec<(String, String)> {
512        const MAX_SAMPLE: usize = 12;
513
514        let players = self.get_players();
515        if players.is_empty() {
516            return vec![];
517        }
518
519        let sample_size = players.len().min(MAX_SAMPLE);
520        // Random starting offset into the player list
521        let offset = if players.len() > sample_size {
522            (rand::random::<u64>() as usize) % (players.len() - sample_size + 1)
523        } else {
524            0
525        };
526
527        let mut sample: Vec<(String, String)> = players[offset..offset + sample_size]
528            .iter()
529            .map(|p| {
530                (
531                    p.gameprofile.name.clone(),
532                    p.gameprofile.id.hyphenated().to_string(),
533                )
534            })
535            .collect();
536
537        // Shuffle using Fisher-Yates with random indices
538        for i in (1..sample.len()).rev() {
539            let j = (rand::random::<u64>() as usize) % (i + 1);
540            sample.swap(i, j);
541        }
542
543        sample
544    }

560    /// Resolves the default respawn world and data for a domain.
561    pub fn respawn_world_and_data_for_domain(
562        &self,
563        domain: &str,
564    ) -> Result<(Arc<World>, RespawnData), String> {
565        let default_world = self
566            .worlds
567            .default_world(domain)
568            .ok_or_else(|| format!("domain {domain} has no default world"))?;
569        let respawn_data = {
570            let level_data = default_world.level_data.read();
571            level_data.data().respawn_data_or_local(&default_world.key)
572        };
573
574        let Some(target_world) = self
575            .worlds
576            .get(respawn_data.dimension())
577            .filter(|world| world.domain() == domain)
578        else {
579            let respawn_data = default_world
580                .world_border_adjusted_respawn_data(local_respawn_data_for_world(&default_world));
581            return Ok((default_world.clone(), respawn_data));
582        };
583
584        let respawn_data = target_world.world_border_adjusted_respawn_data(respawn_data);
585        Ok((target_world, respawn_data))
586    }

594    /// Sets the default respawn data for the respawn data's domain and broadcasts it.
595    pub fn set_respawn_data(&self, respawn_data: RespawnData) -> Result<(), String> {
596        let domain = respawn_data.dimension().namespace.as_ref();
597        let default_world = self
598            .worlds
599            .default_world(domain)
600            .ok_or_else(|| format!("domain {domain} has no default world"))?;
601        let target_world = self
602            .worlds
603            .get(respawn_data.dimension())
604            .filter(|world| world.domain() == domain)
605            .ok_or_else(|| {
606                format!(
607                    "respawn dimension {} is not loaded in domain {domain}",
608                    respawn_data.dimension()
609                )
610            })?;
611
612        if Arc::ptr_eq(&default_world, &target_world) {
613            let mut level_data = default_world.level_data.write();
614            let data = level_data.data_mut();
615            data.set_spawn_pos(respawn_data.pos());
616            data.spawn.angle = respawn_data.yaw;
617            data.set_respawn_data(respawn_data.clone());
618        } else {
619            default_world
620                .level_data
621                .write()
622                .data_mut()
623                .set_respawn_data(respawn_data.clone());
624
625            let mut level_data = target_world.level_data.write();
626            let data = level_data.data_mut();
627            data.set_spawn_pos(respawn_data.pos());
628            data.spawn.angle = respawn_data.yaw;
629        }
630
631        let packet = CSetDefaultSpawnPosition {
632            global_pos: respawn_data.global_pos.clone(),
633            yaw: respawn_data.yaw,
634            pitch: respawn_data.pitch,
635        };
636        for snapshot in self.worlds.snapshots() {
637            let world = snapshot.world();
638            if world.domain() == domain {
639                world.broadcast_to_all(packet.clone());
640            }
641        }
642
643        Ok(())
644    }
