21    /// Runs gameplay packets, game ticks, and chunk sending. Game-tick boundaries
22    /// fork background chunk-scheduling epochs through each world's task tracker.
23    pub async fn run(self: Arc<Self>, cancel_token: CancellationToken) {
24        // A world is built before the server that owns it, so the link back is
25        // filled in here, before anything can tick.
26        self.attach_worlds();
27        // Functions are compiled against a command source, and a source needs
28        // the server, so the datapack load cannot happen before this point.
29        let report = self.reload_functions();
30        for error in &report.errors {
31            log::error!("Datapack load: {error}");
32        }
33        log::info!(
34            "Loaded {} function(s) and {} function tag(s) from {}",
35            report.functions,
36            report.tags,
37            self.functions.root().display()
38        );
39        self.packet_processor.open_after_tick();
40        let packet_worker_count = configured_packet_workers(self.config.packet_workers);
41        let mut packet_handles = Vec::with_capacity(packet_worker_count);
42        for worker_id in 0..packet_worker_count {
43            let s = self.clone();
44            let t = cancel_token.clone();
45            packet_handles.push(tokio::spawn(async move {
46                if let Err(error) = spawn_blocking(move || s.packet_processor.run(&s)).await {
47                    log::error!("Gameplay packet worker {worker_id} failed: {error}");
48                    t.cancel();
49                }
50            }));
51        }
52        let packet_supervisor_cancel = cancel_token.clone();
53        let packet_workers = async move {
54            for handle in packet_handles {
55                if let Err(error) = handle.await {
56                    log::error!("Gameplay packet supervisor failed: {error}");
57                    packet_supervisor_cancel.cancel();
58                }
59            }
60        };
61        let game_handle = {
62            let s = self.clone();
63            let t = cancel_token.clone();
64            let task_guard = GameTickTaskGuard::new(self.clone(), cancel_token.clone());
65            tokio::spawn(async move {
66                let _task_guard = task_guard;
67                s.run_game_tick(t).await;
68            })
69        };
70        let chunk_send_handle = {
71            let s = self.clone();
72            let t = cancel_token.clone();
73            tokio::spawn(async move { s.run_chunk_sending_tick(t).await })
74        };
75        let ((), game_result, chunk_send_result) =
76            tokio::join!(packet_workers, game_handle, chunk_send_handle);
77        for (task, result) in [
78            ("Game tick", game_result),
79            ("Chunk sending tick", chunk_send_result),
80        ] {
81            if let Err(error) = result {
82                log::error!("{task} task failed: {error}");
83            }
84        }
85    }

89        clippy::too_many_lines,
90        reason = "the ordered tick phases and their shutdown joins remain easier to audit together"

92    async fn run_game_tick(self: Arc<Self>, cancel_token: CancellationToken) {
93        let mut world_tick_workers = match WorldTickWorkers::spawn(self.worlds.snapshots()) {
94            Ok(workers) => workers,
95            Err(error) => {
96                log::error!("Failed to start world tick workers: {error}");
97                cancel_token.cancel();
98                return;
99            }
100        };
101        let mut next_tick_time = Instant::now();
102        // `None` is vanilla's zero-initialized `lastOverloadWarningNanos`: the
103        // first overload is always allowed to report and to drop its debt.
104        let mut last_overload_warning: Option<Instant> = None;
105        let mut next_command_data_autosave = Instant::now() + AUTOSAVE_INTERVAL;
106        let mut player_info_ticks = 0_u64;
107        let mut pending_command_executions = PendingCommandExecutionQueue::<CommandSource>::new();
108        let mut player_disconnect_saves = JoinSet::new();
109        let mut command_data_autosaves = JoinSet::new();
110
111        loop {
112            if cancel_token.is_cancelled() {
113                break;
114            }
115
116            let (nanoseconds_per_tick, should_sprint_this_tick) = {
117                let mut tick_manager = self.tick_rate_manager.write();
118                let nanoseconds_per_tick = tick_manager.nanoseconds_per_tick;
119                let (should_sprint, sprint_report) = tick_manager.check_should_sprint_this_tick();
120                drop(tick_manager);
121
122                if let Some(report) = sprint_report {
123                    self.broadcast_sprint_report(&report);
124                    self.broadcast_ticking_state();
125                }
126
127                (nanoseconds_per_tick, should_sprint)
128            };
129
130            if should_sprint_this_tick {
131                next_tick_time = Instant::now();
132                last_overload_warning = Some(next_tick_time);
133            } else {
134                let tick_duration = Duration::from_nanos(nanoseconds_per_tick);
135                let now = Instant::now();
136
137                // A tick that overran leaves `next_tick_time` in the past, and
138                // the loop then runs flat out until it has paid the whole debt
139                // back -- replaying every missed tick, so mobs, redstone,
140                // growth and the day cycle all fast-forward, and on a server
141                // that is merely slow the debt never stops growing. Vanilla
142                // gives up on the debt instead, and says so.
143                let behind = now.saturating_duration_since(next_tick_time);
144                let may_warn = last_overload_warning.is_none_or(|last| {
145                    next_tick_time.saturating_duration_since(last)
146                        >= OVERLOADED_WARNING_INTERVAL + 100 * tick_duration
147                });
148                if behind > OVERLOADED_THRESHOLD + 20 * tick_duration && may_warn {
149                    let ticks_behind = behind.as_nanos() / u128::from(nanoseconds_per_tick.max(1));
150                    log::warn!(
151                        "Can't keep up! Is the server overloaded? Running {}ms or {ticks_behind} ticks behind",
152                        behind.as_millis()
153                    );
154                    next_tick_time +=
155                        tick_duration * u32::try_from(ticks_behind).unwrap_or(u32::MAX);
156                    last_overload_warning = Some(next_tick_time);
157                }
158
159                if now < next_tick_time {
160                    tokio::select! {
161                        () = cancel_token.cancelled() => break,
162                        () = sleep(next_tick_time - now) => {}
163                    }
164                }
165                next_tick_time += tick_duration;
166            }
167
168            if cancel_token.is_cancelled() {
169                break;
170            }
171
172            let tick_start = Instant::now();
173            self.packet_processor.close_for_tick().await;
174            self.advance_chunk_scheduling();
175            self.start_player_disconnect_saves(&mut player_disconnect_saves);
176
177            let (tick_count, runs_normally) = {
178                let mut tick_manager = self.tick_rate_manager.write();
179                tick_manager.tick();
180                let runs_normally = tick_manager.runs_normally();
181                tick_manager.increment_tick_count();
182                (tick_manager.tick_count, runs_normally)
183            };
184
185            self.tick_pending_command_executions(&mut pending_command_executions);
186            self.tick_command_requests(&mut pending_command_executions);
187            self.tick_functions(&mut pending_command_executions, runs_normally);
188            if let Err(error) = self
189                .tick_worlds_game(&world_tick_workers, tick_count, runs_normally)
190                .await
191            {
192                log::error!("World game tick failed: {error}");
193                cancel_token.cancel();
194                break;
195            }
196            self.process_world_removals(&mut world_tick_workers);
197            self.tick_player_connections();
198            player_info_ticks += 1;
199            if player_info_ticks > SEND_PLAYER_INFO_INTERVAL {
200                let _span = tracing::trace_span!("broadcast_latency").entered();
201                self.broadcast_player_latency_updates();
202                player_info_ticks = 0;
203            }
204            self.tick_jobs(tick_count, runs_normally);
205            self.process_player_joins();
206
207            {
208                let server = self.clone();
209                let _ =
210                    spawn_blocking(move || server.process_world_changes(tick_count, runs_normally))
211                        .await;
212            }
213
214            self.process_world_additions(&mut world_tick_workers);
215            self.process_domain_switches();
216
217            self.tick_command_data_autosave(
218                &mut next_command_data_autosave,
219                &mut command_data_autosaves,
220            );
221
222            let tab_list_tick_stats = self.record_tick_and_capture_tab_stats(
223                tick_count,
224                tick_start.elapsed().as_nanos() as u64,
225            );
226
227            if let Some(tick_stats) = tab_list_tick_stats {
228                self.broadcast_tab_list(tick_stats);
229            }
230
231            if should_sprint_this_tick {
232                let mut tick_manager = self.tick_rate_manager.write();
233                tick_manager.end_tick_work();
234            }
235
236            // Last, so that anything listening sees a tick that has already
237            // done its own work rather than one still in the middle of it.
238            self.events.fire(&mut ServerTickEvent::new(tick_count));
239
240            self.packet_processor.open_after_tick();
241            if should_sprint_this_tick || Instant::now() >= next_tick_time {
242                self.packet_processor.wait_for_overload_progress().await;
243            }
244        }
245
246        self.jobs.cancel_all();
247        pending_command_executions.cancel_all();
248        self.command_requests.clear();
249        // A compiled function holds the source it was parsed with, and that
250        // source holds the server; dropping the library breaks the cycle.
251        self.functions.unload();
252        self.packet_processor.stop();
253        self.start_player_disconnect_saves(&mut player_disconnect_saves);
254        self.pending_player_disconnects.clear();
255        while let Some(result) = player_disconnect_saves.join_next().await {
256            if let Err(error) = result {
257                log::error!("Player disconnect save task failed during shutdown: {error}");
258            }
259        }
260        while let Some(result) = command_data_autosaves.join_next().await {
261            if let Err(error) = result {
262                log::error!("Command data autosave task failed during shutdown: {error}");
263            }
264        }
265    }

267    /// Ticks every online player's connection, world membership or not.
268    ///
269    /// Vanilla parity: `MinecraftServer.tickConnection` ->
270    /// `ServerConnectionListener.tick`, which walks its own connection list and
271    /// not the player list. That distinction is the whole point of the call.
272    /// Foton ticked the connection from `Player::tick`, which only runs for a
273    /// player some world still ticks -- so a player who reached the End credits,
274    /// and whom `show_end_credits` removes from the world, stopped being sent
275    /// keep-alives and stopped being timed out. A client whose link then died
276    /// without a FIN left its socket, both halves of the connection task and the
277    /// whole `Player` alive until restart, with the name still in the tab list;
278    /// and because a duplicate login is refused while the ghost holds the UUID,
279    /// the person behind it could never get back in either.
280    fn tick_player_connections(&self) {
281        self.online_players().iter_players(|_, player| {
282            player.connection.tick();
283            true
284        });
285    }

299    async fn autosave_command_data(self: &Arc<Self>) {
300        tracing::debug!("Autosave started");
301
302        // Vanilla parity: `MinecraftServer.autoSave` runs
303        // `saveEverything(true, false, false)` -- chunks, level data and players
304        // -- every 6000 ticks. Foton's timer only ever covered the command-owned
305        // data below, so nothing else was written until a chunk unloaded, a
306        // player disconnected, or the server shut down cleanly. Around spawn
307        // (`keep_spawn_in_memory`) and inside any player's radius chunks never
308        // unload at all, so a long session could reach the end of the day having
309        // written nothing, and any unclean stop took the lot.
310        for world in self.worlds.values() {
311            world.request_save();
312        }
313        self.request_save_players();
314
315        let results = self.save_command_data().await;
316        match results.scoreboards {
317            Ok(saved) => tracing::debug!(saved, "Domain scoreboard autosave completed"),
318            Err(error) => tracing::error!(%error, "Domain scoreboard autosave failed"),
319        }
320        match results.storage {
321            Ok(saved) => tracing::debug!(saved, "Domain command-storage autosave completed"),
322            Err(error) => tracing::error!(%error, "Domain command-storage autosave failed"),
323        }
324        match results.maps {
325            Ok(saved) => tracing::debug!(saved, "Domain map autosave completed"),
326            Err(error) => tracing::error!(%error, "Domain map autosave failed"),
327        }
328        match results.boss_bars {
329            Ok(saved) => tracing::debug!(saved, "Domain boss-bar autosave completed"),
330            Err(error) => tracing::error!(%error, "Domain boss-bar autosave failed"),
331        }
332    }

334    fn tick_command_data_autosave(
335        self: &Arc<Self>,
336        next_autosave: &mut Instant,
337        saves: &mut JoinSet<()>,
338    ) {
339        while let Some(result) = saves.try_join_next() {
340            if let Err(error) = result {
341                log::error!("Command data autosave task failed: {error}");
342            }
343        }
344        if Instant::now() < *next_autosave {
345            return;
346        }
347        if saves.is_empty() {
348            let server = Arc::clone(self);
349            saves.spawn(async move {
350                server.autosave_command_data().await;
351            });
352        } else {
353            tracing::warn!("Skipping command data autosave while the previous save runs");
354        }
355        *next_autosave = Instant::now() + AUTOSAVE_INTERVAL;
356    }

373    /// Runs the `#minecraft:load` and `#minecraft:tick` function tags.
374    ///
375    /// Vanilla parity: `ServerFunctionManager.tick`, which skips the tags
376    /// entirely while the tick rate manager is frozen or stepping.
377    fn tick_functions(
378        self: &Arc<Self>,
379        pending: &mut PendingCommandExecutionQueue<CommandSource>,
380        runs_normally: bool,
381    ) {
382        if !runs_normally {
383            return;
384        }
385        let load = self.functions.take_load_functions();
386        let tick = self.functions.ticking_functions();
387        if load.is_empty() && tick.is_empty() {
388            return;
389        }
390        for function in load.into_iter().chain(tick) {
391            self.run_function_now(pending, &function);
392        }
393    }

395    /// Starts one function as its own top-level execution.
396    fn run_function_now(
397        self: &Arc<Self>,
398        pending: &mut PendingCommandExecutionQueue<CommandSource>,
399        function: &CommandFunction,
400    ) {
401        let owner = CommandExecutionOwner::capture(CommandSender::Console, self);
402        let source = self.function_source();
403        // A macro function in a tag has no arguments to fill in, so it fails
404        // here the way vanilla's does rather than running half-substituted.
405        let entries = match self
406            .with_command_dispatcher(|dispatcher| function.instantiate(None, dispatcher))
407        {
408            Ok(entries) => entries,
409            Err(reason) => {
410                log::error!("Failed to instantiate function {}: {reason}", function.id());
411                return;
412            }
413        };
414        let mut execution = CommandExecutionContext::for_source(&source);
415        execution.queue_initial_function_call(entries, source, CommandResultCallback::empty());
416        if execution.run() == ExecutionStop::Suspended && !pending.push_suspended(owner, execution)
417        {
418            tracing::error!(
419                function = %function.id(),
420                "suspended function execution could not be retained"
421            );
422        }
423    }

425    fn tick_command_requests(
426        self: &Arc<Self>,
427        pending: &mut PendingCommandExecutionQueue<CommandSource>,
428    ) {
429        let mut handled = 0;
430        for _ in 0..COMMAND_REQUESTS_PER_TICK {
431            let Some(request) = self
432                .command_requests
433                .pop_front_runnable(|owner| !pending.blocks(owner.key()))
434            else {
435                break;
436            };
437            handled += 1;
438
439            match request {
440                CommandRequest::Execute { owner, command } => {
441                    if !owner.is_current(self) {
442                        continue;
443                    }
444                    self.execute_command_request(pending, owner, &command);
445                }
446                CommandRequest::Suggestions {
447                    owner,
448                    transaction_id,
449                    input,
450                } => {
451                    if !owner.is_current(self) {
452                        continue;
453                    }
454                    let Some(player) = owner.sender().get_player() else {
455                        tracing::error!("command suggestion request has a non-player owner");
456                        continue;
457                    };
458                    self.send_command_suggestions(player, transaction_id, &input);
459                }
460            }
461        }
462
463        if handled == COMMAND_REQUESTS_PER_TICK {
464            tracing::debug!(handled, "Command request tick reached its processing limit");
465        }
466    }

468    fn execute_command_request(
469        self: &Arc<Self>,
470        pending: &mut PendingCommandExecutionQueue<CommandSource>,
471        owner: CommandExecutionOwner,
472        command: &str,
473    ) {
474        let command = command.strip_prefix('/').unwrap_or(command);
475
476        // Before the dispatcher, because a plugin's command is not in the
477        // Brigadier tree and never will be: the tree is built at startup from
478        // types the server knows. A listener that claims the name runs it
479        // itself, and the server does not go looking for a command it has
480        // never heard of only to report that it does not exist.
481        let mut event = CommandEvent::new(owner.sender().get_player().cloned(), command);
482        self.events.fire(&mut event);
483        if event.is_handled() {
484            return;
485        }
486
487        let source = CommandSource::new(owner.sender().clone(), Arc::clone(self));
488        let chain = {
489            let dispatcher = self.command_dispatcher.read();
490            let parse = dispatcher.parse(command, source.clone());
491            dispatcher.context_chain(parse)
492        };
493        let chain = match chain {
494            Ok(chain) => chain,
495            Err(error) => {
496                source.handle_error(&error, false);
497                return;
498            }
499        };
500
501        let mut execution = CommandExecutionContext::for_source(&source);
502        execution.queue_initial_command(chain, source, CommandResultCallback::empty());
503        if execution.run() == ExecutionStop::Suspended && !pending.push_suspended(owner, execution)
504        {
505            tracing::error!("suspended command execution could not be retained");
506        }
507    }

509    fn send_command_suggestions(
510        self: &Arc<Self>,
511        player: &Arc<Player>,
512        transaction_id: i32,
513        input: &str,
514    ) {
515        // Paper's `AsyncTabCompleteEvent`, fired before the tree is consulted:
516        // a plugin's command is not in the Brigadier tree and never will be, so
517        // completing one is only possible ahead of the server's own answer.
518        let mut event = AsyncTabCompleteEvent::new(Some(Arc::clone(player)), input.to_owned());
519        self.events.fire(&mut event);
520        if event.is_cancelled() {
521            player.send_packet(CCommandSuggestions::new(transaction_id, 0, 0, Vec::new()));
522            return;
523        }
524        if event.is_handled() {
525            player.send_packet(plugin_suggestions_packet(
526                transaction_id,
527                input,
528                event.completions(),
529            ));
530            return;
531        }
532
533        let suggestions =
534            self.build_command_suggestions(CommandSender::Player(Arc::clone(player)), input);
535        match suggestions {
536            Ok(suggestions) => {
537                player.send_packet(command_suggestions_packet(transaction_id, &suggestions));
538            }
539            Err(error) => {
540                tracing::warn!(%error, "failed to build command suggestions");
541                player.send_packet(CCommandSuggestions::new(transaction_id, 0, 0, Vec::new()));
542            }
543        }
544    }

546    pub(super) fn build_command_suggestions(
547        self: &Arc<Self>,
548        sender: CommandSender,
549        input: &str,
550    ) -> Result<Suggestions, SuggestionError> {
551        let source = CommandSource::new(sender, Arc::clone(self));
552        let mut reader = StringReader::new(input);
553        if reader.peek() == Some('/') {
554            reader.skip();
555        }
556        let dispatcher = self.command_dispatcher.read();
557        let parse = dispatcher.parse_reader(reader, source);
558        dispatcher.completion_suggestions(&parse)
559    }

561    /// Chunk sending tick loop — encodes and sends chunks to players independently.
562    async fn run_chunk_sending_tick(self: Arc<Self>, cancel_token: CancellationToken) {
563        let nanos_per_tick = 1_000_000_000 / CHUNK_SENDING_TPS;
564        let mut next_tick_time = Instant::now();
565
566        loop {
567            if cancel_token.is_cancelled() {
568                break;
569            }
570
571            let now = Instant::now();
572            if now < next_tick_time {
573                tokio::select! {
574                    () = cancel_token.cancelled() => break,
575                    () = sleep(next_tick_time - now) => {}
576                }
577            }
578            next_tick_time += Duration::from_nanos(nanos_per_tick);
579
580            if cancel_token.is_cancelled() {
581                break;
582            }
583
584            let server = self.clone();
585            let _ = spawn_blocking(move || {
586                server.tick_chunk_sending();
587            })
588            .await;
589        }
590    }

592    /// Executes one chunk sending tick across all worlds and players.
593    ///
594    /// A per-world per-tick encode cache is used so overlapping view areas
595    /// don't re-encode the same chunk within a single tick.
596    fn tick_chunk_sending(&self) {
597        let tick_start = Instant::now();
598        for snapshot in self.worlds.snapshots() {
599            let world = snapshot.world();
600            let mut encode_cache = rustc_hash::FxHashMap::default();
601            world.players.iter_players(|_uuid, player| {
602                Self::send_chunks_for_player(
603                    player,
604                    world,
605                    &mut encode_cache,
606                    self.chunk_encoding_pool.as_ref(),
607                );
608                true
609            });
610        }
611
612        let elapsed = tick_start.elapsed();
613        if elapsed >= SLOW_CHUNK_TICK_THRESHOLD {
614            tracing::warn!(?elapsed, "Chunk sending tick slow");
615        }
616    }

618    /// Three-phase chunk send for a single player: prepare (lock briefly),
619    /// encode (no lock), commit (lock briefly + generation check).
620    fn send_chunks_for_player(
621        player: &Arc<Player>,
622        world: &Arc<World>,
623        encode_cache: &mut rustc_hash::FxHashMap<ChunkPos, EncodedChunk>,
624        encoding_pool: &ThreadPool,
625    ) {
626        let chunk_pos = *player.last_chunk_pos.lock();
627        let connection = &player.connection;
628
629        // Phase 1: prepare (brief lock)
630        let prepared = {
631            let mut sender = player.chunk_sender.lock();
632            sender.prepare_batch(world, chunk_pos, &player.chunk_send_epoch)
633        };
634
635        let Some(batch) = prepared else {
636            return;
637        };
638
639        // Phase 2: encode (no lock held — uses per-tick local cache)
640        let compression = connection.compression();
641        let encoded = ChunkSender::encode_batch(&batch, encode_cache, compression, encoding_pool);
642
643        // Phase 3: commit while holding the tracking view that world detachment invalidates.
644        // This makes membership validation, packet commit, and tracker refresh one side of
645        // the same synchronization boundary.
646        let tracking_view = player.last_tracking_view.lock();
647        let Some(view) = *tracking_view else {
648            return;
649        };
650        if !world.contains_player(player) {
651            return;
652        }
653        let sent_chunks = {
654            let mut sender = player.chunk_sender.lock();
655            sender.commit_batch(&batch, encoded, connection, &player.chunk_send_epoch)
656        };
657
658        if sent_chunks.is_empty() {
659            return;
660        }
661
662        let sent_chunks = player.chunk_sender.lock().sent_chunks_snapshot();
663        world
664            .entity_tracker()
665            .update_player(player, &view, |chunk| sent_chunks.contains(&chunk));
666    }

668    /// Commits ready chunk lifecycle epochs and forks the next background work.
669    fn advance_chunk_scheduling(&self) {
670        for (i, snapshot) in self.worlds.snapshots().into_iter().enumerate() {
671            let world = snapshot.world();
672            let timings = world.chunk_map.advance_scheduling();
673
674            let background_elapsed = timings.ticket_updates
675                + timings.schedule_generation
676                + timings.run_generation
677                + timings.process_unloads;
678            let boundary_elapsed = timings.block_entity_unloads
679                + timings.readiness_demotions
680                + timings.lifecycle_commit
681                + timings.readiness_reconcile
682                + timings.ticking_snapshot_rebuild;
683            let work_elapsed = background_elapsed + boundary_elapsed;
684
685            if work_elapsed >= SLOW_CHUNK_TICK_THRESHOLD {
686                tracing::warn!(
687                    world = i,
688                    work_elapsed = ?work_elapsed,
689                    background_elapsed = ?background_elapsed,
690                    boundary_elapsed = ?boundary_elapsed,
691                    ticket_updates = ?timings.ticket_updates,
692                    block_entity_unloads = ?timings.block_entity_unloads,
693                    readiness_demotions = ?timings.readiness_demotions,
694                    lifecycle_commit = ?timings.lifecycle_commit,
695                    readiness_reconcile = ?timings.readiness_reconcile,
696                    post_process_generation = ?timings.post_process_generation,
697                    post_process_chunk_count = timings.post_process_chunk_count,
698                    post_process_position_count = timings.post_process_position_count,
699                    readiness_candidate_count = timings.readiness_candidate_count,
700                    ticking_snapshot_rebuild = ?timings.ticking_snapshot_rebuild,
701                    rebuilt_ticking_chunk_count = timings.rebuilt_ticking_chunk_count,
702                    lookup_cache_holder_hits = timings.lookup_cache.holder_hits,
703                    lookup_cache_missing_hits = timings.lookup_cache.missing_hits,
704                    lookup_cache_scc_lookups = timings.lookup_cache.scc_lookups,
705                    lookup_cache_foreign_map_bypasses = timings.lookup_cache.foreign_map_bypasses,
706                    lookup_cache_evictions = timings.lookup_cache.evictions,
707                    schedule_generation = ?timings.schedule_generation,
708                    scheduled_count = timings.scheduled_count,
709                    // A non-zero count is the lifecycle boundary and the
710                    // scheduler disagreeing about which chunks are loaded --
711                    // the state that used to end the epoch in a panic.
712                    deferred_schedule_count = timings.deferred_schedule_count,
713                    run_generation = ?timings.run_generation,
714                    process_unloads = ?timings.process_unloads,
715                    "Chunk scheduling epoch slow"
716                );
717            }
718        }
719    }

721    /// Attaches worlds constructed asynchronously at a tick safe-point.
722    fn process_world_additions(&self, workers: &mut WorldTickWorkers) {
723        let worlds = take(&mut *self.pending_world_additions.lock());
724        for (world, completion) in worlds {
725            let key = world.key.clone();
726            if let Err(error) = self.worlds.insert(key.clone(), world) {
727                let _ = completion.send(Err(error));
728                continue;
729            }
730            let Some(world) = self.worlds.get(&key) else {
731                let _ = completion.send(Err(format!("world {key} was not attached")));
732                continue;
733            };
734            if let Err(error) = workers.add(&world) {
735                log::error!("Failed to start tick worker for {key}: {error}");
736                let _ = self.worlds.remove(&key);
737                let _ = completion.send(Err(error.to_string()));
738            } else {
739                let _ = completion.send(Ok(()));
740            }
741        }
742    }

792    /// Detaches worlds only after the current tick has completed.
793    fn process_world_removals(&self, workers: &mut WorldTickWorkers) {
794        let requests = take(&mut *self.pending_world_removals.lock());
795        for request in requests {
796            let WorldRemovalRequest {
797                key,
798                save,
799                completion,
800            } = request;
801            let Some(world) = self.worlds.get_owned(&key) else {
802                continue;
803            };
804            if !workers.remove(&key) {
805                log::error!("Requested world {key} has no tick worker");
806                if let Some(sender) = completion {
807                    let _ = sender.send(Err("world has no tick worker".to_owned()));
808                }
809                continue;
810            }
811            if let Err(error) = self.worlds.remove(&key) {
812                log::debug!("World removal deferred for {key}: {error:?}");
813                if let Some(sender) = completion {
814                    let _ = sender.send(Err(format!("{error:?}")));
815                }
816                continue;
817            }
818            if !save {
819                if let Some(sender) = completion {
820                    let _ = sender.send(Ok(0));
821                }
822                continue;
823            }
824            // Persistence is I/O and must not stall the serialized game tick.
825            let runtime = Arc::clone(&self.chunk_runtime);
826            let _cleanup = runtime.spawn(async move {
827                let mut saved = 0;
828                world.cleanup(&mut saved).await;
829                tracing::debug!(world = %key, saved_chunks = saved, "World cleanup finished");
830                if let Some(sender) = completion {
831                    let _ = sender.send(Ok(saved));
832                }
833            });
834        }
835    }

837    #[tracing::instrument(level = "trace", skip(self, workers), name = "tick_worlds")]
838    async fn tick_worlds_game(
839        &self,
840        workers: &WorldTickWorkers,
841        tick_count: u64,
842        runs_normally: bool,
843    ) -> Result<(), WorldTickWorkerError> {
844        let all_timings = workers.tick_all(tick_count, runs_normally).await?;
845        for (i, timings) in all_timings.iter().enumerate() {
846            if timings.elapsed.as_millis() < 50 {
847                continue;
848            }
849            let cm = &timings.chunk_map;
850            tracing::warn!(
851                world = i,
852                elapsed = ?timings.elapsed,
853                tick_count,
854                entity_tick = ?timings.entity_tick,
855                broadcast_changes = ?cm.broadcast_changes,
856                collect_tickable = ?cm.collect_tickable,
857                tick_chunks = ?cm.tick_chunks,
858                tick_block_entities = ?cm.tick_block_entities,
859                tickable_count = cm.tickable_count,
860                total_chunks = cm.total_chunks,
861                lookup_cache_holder_hits = cm.lookup_cache.holder_hits,
862                lookup_cache_missing_hits = cm.lookup_cache.missing_hits,
863                lookup_cache_scc_lookups = cm.lookup_cache.scc_lookups,
864                lookup_cache_foreign_map_bypasses = cm.lookup_cache.foreign_map_bypasses,
865                lookup_cache_evictions = cm.lookup_cache.evictions,
866                "Game tick slow"
867            );
868        }
869        Ok(())
870    }

873        let stats = self
874            .jobs

878                polled = stats.polled,
879                finished = stats.finished,
880                pending = stats.pending,
881                "Server jobs pending"
