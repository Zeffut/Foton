14    /// Sends one scoreboard-team operation to players in the owning domain.
15    ///
16    /// Unlike vanilla's server-wide scoreboard, each Foton domain has its own
17    /// scoreboard, so leaking this packet across domains would install teams
18    /// that do not exist for the receiving player.
19    pub fn broadcast_scoreboard_packet(&self, domain: &str, packet: CSetPlayerTeam) {
20        self.online_players.iter_players(|_, player| {
21            if player.get_world().domain() == domain {
22                player.send_packet(packet.clone());
23            }
24            true
25        });
26    }

53    /// Builds the tab list header/footer with recent and five-second tick statistics.
54    pub(super) fn tab_list_components(
55        tick_stats: TabListTickStats,
56    ) -> (TextComponent, TextComponent) {
57        // Color TPS based on value
58        let tps_color = if tick_stats.tps >= 19.5 {
59            Color::Green
60        } else if tick_stats.tps >= 15.0 {
61            Color::Yellow
62        } else {
63            Color::Red
64        };
65
66        let mspt_color = |mspt: f32| {
67            if mspt <= 50.0 {
68                Color::Aqua
69            } else {
70                Color::Red
71            }
72        };
73
74        let header = TextComponent::plain("\n").add_children(vec![
75            TextComponent::plain("Foton Dev Build").color(Color::Yellow),
76            TextComponent::plain("\n"),
77        ]);
78        let footer = TextComponent::plain("\n").add_children(vec![
79            TextComponent::plain("TPS: ").color(Color::Gray),
80            TextComponent::plain(format!("{:.1}", tick_stats.tps)).color(tps_color),
81            TextComponent::plain(" | ").color(Color::DarkGray),
82            TextComponent::plain("MSPT: ").color(Color::Gray),
83            TextComponent::plain(format!("{:.2}", tick_stats.recent_mspt))
84                .color(mspt_color(tick_stats.recent_mspt)),
85            TextComponent::plain(" recent | ").color(Color::Gray),
86            TextComponent::plain(format!("{:.2}", tick_stats.average_mspt))
87                .color(mspt_color(tick_stats.average_mspt)),
88            TextComponent::plain(" avg (5s) | ").color(Color::Gray),
89            TextComponent::plain(format!("{:.2}", tick_stats.p95_mspt))
90                .color(mspt_color(tick_stats.p95_mspt)),
91            TextComponent::plain(" p95").color(Color::Gray),
92            TextComponent::plain("\n"),
93        ]);
94
95        (header, footer)
96    }

142    pub(super) fn broadcast_player_join_message(
143        &self,
144        player: &Arc<Player>,
145        previous_name: Option<&str>,
146    ) {
147        let display_name = player.display_name();
148        // Fallback to the current name when the cache has no prior entry.
149        let old_name = previous_name.unwrap_or(player.gameprofile.name.as_str());
150        let message: TextComponent = if player.gameprofile.name.eq_ignore_ascii_case(old_name) {
151            translations::MULTIPLAYER_PLAYER_JOINED
152                .message([display_name])
153                .into()
154        } else {
155            translations::MULTIPLAYER_PLAYER_JOINED_RENAMED
156                .message([display_name, TextComponent::plain(old_name.to_owned())])
157                .into()
158        };
159        let mut event =
160            PlayerJoinEvent::new(Arc::clone(player), Some(message.color(Color::Yellow)));
161        self.events.fire(&mut event);
162        let uuid = player.gameprofile.id;
163        if let Some(message) = event.into_message() {
164            self.broadcast_system_chat(&message, Some(uuid));
165        }
166        // The server's pack is answered in configuration, before there is a
167        // player to tell a listener about.
168        player.announce_configuration_resource_pack();
169    }

226    /// Resends the command tree and vanilla client permission-level projection.
227    pub fn resend_player_permission_context(self: &Arc<Self>, player: &Arc<Player>) {
228        let world = player.get_world();
229        player.send_packet(CEntityEvent {
230            entity_id: player.id(),
231            event: client_permission_event(player, &world),
232        });
233
234        let server = player.server();
235        if !Arc::ptr_eq(&server, self) {
236            tracing::error!(
237                player = %player.gameprofile.name,
238                "cannot project commands from a different server"
239            );
240            return;
241        }
242        let Some(shared_player) = self.online_players.get_by_uuid(&player.gameprofile.id) else {
243            tracing::error!(
244                player = %player.gameprofile.name,
245                "cannot project commands for a player outside the online player map"
246            );
247            return;
248        };
249        if !Arc::ptr_eq(&shared_player, player) {
250            tracing::error!(
251                player = %player.gameprofile.name,
252                "cannot project commands for a stale player handle"
253            );
254            return;
255        }
256        let source = CommandSource::new(CommandSender::Player(shared_player), server);
257        let commands = {
258            let dispatcher = self.command_dispatcher.read();
259            // Bukkit's `PlayerCommandSendEvent`: a listener removes root names
260            // to keep a command out of this player's list. Hiding is not
261            // forbidding -- a command dropped here still runs if typed, which
262            // is what permission plugins rely on to shorten the list without
263            // changing who may use what.
264            let names = match visible_root_names(&dispatcher, &source) {
265                Ok(names) => names,
266                Err(error) => {
267                    tracing::error!(
268                        player = %player.gameprofile.name,
269                        %error,
270                        "failed to list the player's root commands"
271                    );
272                    return;
273                }
274            };
275            let offered = names.len();
276            let mut event = PlayerCommandSendEvent::new(Arc::clone(player), names.clone());
277            self.events().fire(&mut event);
278            if event.commands().len() == offered {
279                command_tree_packet(&dispatcher, &source)
280            } else {
281                let kept = event.commands().iter().cloned().collect::<FxHashSet<_>>();
282                let hidden = names
283                    .into_iter()
284                    .filter(|name| !kept.contains(name))
285                    .collect::<FxHashSet<_>>();
286                command_tree_packet_without(&dispatcher, &source, &hidden)
287            }
288        };
289        match commands {
290            Ok(commands) => player.send_packet(commands),
291            Err(error) => tracing::error!(
292                player = %player.gameprofile.name,
293                %error,
294                "failed to project the player's command tree"
295            ),
296        }
297    }

299    /// Sends each domain's queued scoreboard packets to the players in that domain.
300    ///
301    /// Vanilla parity: `ServerScoreboard` broadcasting its team, objective and
302    /// score changes to `PlayerList.broadcastAll`. Foton's scoreboards are per
303    /// domain, so a player sees the scoreboard of the domain they are in.
304    pub(super) fn flush_team_updates(&self) {
305        for (domain, _) in self.scoreboards.iter() {
306            self.flush_domain_scoreboard(domain, None);
307        }
308    }
309
310    /// Sends one domain's queued team and objective packets to its players,
311    /// leaving out `skip`.
312    fn flush_domain_scoreboard(&self, domain: &str, skip: Option<Uuid>) {
313        let Some(scoreboard) = self.scoreboards.get(domain) else {
314            return;
315        };
316        let teams = scoreboard.take_team_updates();
317        let objectives = scoreboard.take_objective_updates();
318        if teams.is_empty() && objectives.is_empty() {
319            return;
320        }
321        self.online_players.iter_players(|uuid, player| {
322            if Some(*uuid) != skip && player.get_world().domain() == domain {
323                for update in &teams {
324                    player.send_packet(update.clone());
325                }
326                for update in &objectives {
327                    send_scoreboard_packet(player, update);
328                }
329            }
330            true
331        });
332    }

334    /// Hands a player what their client keeps of the domain they have just
335    /// entered: their recipe book, then the domain's teams.
336    ///
337    /// Vanilla parity: `sendInitialRecipeBook` followed by
338    /// `updateEntireScoreboard` in `PlayerList.placeNewPlayer`. Both are sent
339    /// again on a domain switch, because the book is saved per domain and the
340    /// client keeps its old copy across the respawn packet.
341    pub(super) fn send_domain_state(&self, player: &Player, left: Option<&str>, entered: &str) {
342        player.send_initial_recipe_book();
343        // What is queued is already in the snapshot below; everyone else in
344        // the domain still needs it, and in order.
345        self.flush_domain_scoreboard(entered, Some(player.gameprofile.id));
346        self.send_domain_scoreboard(player, left, entered);
347    }

349    /// Sends a player every team and displayed objective of a domain they
350    /// have just entered, after taking off those of the one they left.
351    ///
352    /// Vanilla parity: `PlayerList.updateEntireScoreboard`, sent on login; on a
353    /// domain switch the client still holds the previous domain's teams and
354    /// objectives, which would otherwise keep coloring names and drawing
355    /// scores there.
356    fn send_domain_scoreboard(&self, player: &Player, left: Option<&str>, entered: &str) {
357        if let Some(scoreboard) = left.and_then(|domain| self.scoreboards.get(domain)) {
358            for removal in scoreboard.team_removals() {
359                player.send_packet(removal);
360            }
361            for removal in scoreboard.objective_removals() {
362                send_scoreboard_packet(player, &removal);
363            }
364        }
365        if let Some(scoreboard) = self.scoreboards.get(entered) {
366            for team in scoreboard.team_snapshot_packets() {
367                player.send_packet(team);
368            }
369            for packet in scoreboard.objective_snapshot_packets() {
370                send_scoreboard_packet(player, &packet);
371            }
372        }
373    }

376fn send_scoreboard_packet(player: &Player, packet: &ScoreboardPacket) {
377    match packet {
378        ScoreboardPacket::Objective(packet) => player.send_packet(packet.clone()),
379        ScoreboardPacket::Score(packet) => player.send_packet(packet.clone()),
380        ScoreboardPacket::Reset(packet) => player.send_packet(packet.clone()),
381        ScoreboardPacket::Display(packet) => player.send_packet(packet.clone()),
382    }

385fn scoreboard_membership_packets(
386    holder: &str,
387    previous_team: Option<&str>,
388    current_team: Option<&str>,
389) -> Vec<CSetPlayerTeam> {
390    let mut packets = Vec::with_capacity(2);
391    if let Some(previous_team) = previous_team {
392        packets.push(CSetPlayerTeam::remove_player(previous_team, holder));
393    }
394    if let Some(current_team) = current_team {
395        packets.push(CSetPlayerTeam::add_player(current_team, holder));
396    }
397    packets
398}
