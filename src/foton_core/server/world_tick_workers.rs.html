12struct WorldTickRequest {
13    tick_count: u64,
14    runs_normally: bool,
15    response: oneshot::Sender<WorldGameTickTimings>,
16}
17
18struct WorldTickWorker {
19    world_key: Arc<str>,
20    requests: Option<Sender<WorldTickRequest>>,
21    thread: Option<thread::JoinHandle<()>>,
22}
23
24impl WorldTickWorker {
25    fn spawn(index: usize, snapshot: WorldMapSnapshot) -> io::Result<Self> {
26        let world = Arc::clone(snapshot.world());
27        let world_key = Arc::<str>::from(world.key.to_string());
28        let (request_sender, request_receiver) = channel::bounded::<WorldTickRequest>(1);
29        let thread = thread::Builder::new()
30            .name(format!("world-tick-{index}"))
31            .spawn(move || {
32                while let Ok(request) = request_receiver.recv() {
33                    if request.runs_normally {
34                        world.chunk_map.tick_timed_tickets();
35                    }
36                    let timings = world.tick_game(request.tick_count, request.runs_normally);
37                    let _ = request.response.send(timings);
38                }
39            })?;
40
41        Ok(Self {
42            world_key,
43            requests: Some(request_sender),
44            thread: Some(thread),
45        })
46    }
47
48    fn start_tick(
49        &self,
50        tick_count: u64,
51        runs_normally: bool,
52    ) -> Result<oneshot::Receiver<WorldGameTickTimings>, WorldTickWorkerError> {
53        let (response, receiver) = oneshot::channel();
54        let Some(requests) = &self.requests else {
55            return Err(WorldTickWorkerError::Unavailable {
56                world: Arc::clone(&self.world_key),
57            });
58        };
59        requests
60            .send(WorldTickRequest {
61                tick_count,
62                runs_normally,
63                response,
64            })
65            .map_err(|_| WorldTickWorkerError::Unavailable {
66                world: Arc::clone(&self.world_key),
67            })?;
68        Ok(receiver)
69    }
70}
71
72impl Drop for WorldTickWorker {
73    fn drop(&mut self) {
74        drop(self.requests.take());
75        let Some(thread) = self.thread.take() else {
76            return;
77        };
78        if thread.join().is_err() {
79            log::error!(
80                "World tick worker for {} panicked during execution",
81                self.world_key
82            );
83        }
84    }
85}

87#[derive(Debug, Error)]
88pub(super) enum WorldTickWorkerError {
89    #[error("world tick worker for {world} is unavailable")]
90    Unavailable { world: Arc<str> },
91    #[error("world tick worker for {world} stopped without returning timings")]
92    MissingResponse { world: Arc<str> },
93}

95pub(super) struct WorldTickWorkers {
96    workers: Vec<WorldTickWorker>,
97}
98
99impl WorldTickWorkers {
100    pub(super) fn spawn<S>(worlds: impl IntoIterator<Item = S>) -> io::Result<Self>
101    where
102        S: Into<WorldMapSnapshot>,
103    {
104        let mut workers = Vec::new();
105        for (index, world) in worlds.into_iter().map(Into::into).enumerate() {
106            workers.push(WorldTickWorker::spawn(index, world)?);
107        }
108        Ok(Self { workers })
109    }
110
111    /// Adds a worker at a tick safe-point.
112    pub(super) fn add<S>(&mut self, world: S) -> io::Result<()>
113    where
114        S: Into<WorldMapSnapshot>,
115    {
116        let index = self.workers.len();
117        self.workers
118            .push(WorldTickWorker::spawn(index, world.into())?);
119        Ok(())
120    }
121
122    /// Removes a worker after the caller has completed the current tick boundary.
123    ///
124    /// Dropping the worker closes its request channel and joins the thread, so no
125    /// world tick can still be executing when this returns. The world itself is
126    /// intentionally not removed here; the server must first persist it and
127    /// detach it from the loaded-world map.
128    pub(super) fn remove(&mut self, key: &Identifier) -> bool {
129        let Some(index) = self
130            .workers
131            .iter()
132            .position(|worker| worker.world_key.as_ref() == key.to_string())
133        else {
134            return false;
135        };
136        drop(self.workers.remove(index));
137        true
138    }
139
140    pub(super) async fn tick_all(
141        &self,
142        tick_count: u64,
143        runs_normally: bool,
144    ) -> Result<Vec<WorldGameTickTimings>, WorldTickWorkerError> {
145        let mut responses = Vec::with_capacity(self.workers.len());
146        for worker in &self.workers {
147            responses.push(worker.start_tick(tick_count, runs_normally)?);
148        }
149
150        let mut timings = Vec::with_capacity(responses.len());
151        for (worker, response) in self.workers.iter().zip(responses) {
152            timings.push(
153                response
154                    .await
155                    .map_err(|_| WorldTickWorkerError::MissingResponse {
156                        world: Arc::clone(&worker.world_key),
157                    })?,
158            );
159        }
160        Ok(timings)
161    }
162}

164#[cfg(test)]
165mod tests {
166    use futures::executor::block_on;
167
168    use super::WorldTickWorkers;
169    use crate::test_support::fresh_test_world;
170
171    #[test]
172    fn removing_worker_joins_it_and_keeps_remaining_worlds_ticking() {
173        let first = fresh_test_world("removable_worker_first");
174        let second = fresh_test_world("removable_worker_second");
175        let Ok(mut workers) = WorldTickWorkers::spawn([&first, &second]) else {
176            panic!("world tick workers should start");
177        };
178        assert!(block_on(workers.tick_all(1, true)).is_ok());
179
180        assert!(workers.remove(&first.key));
181        assert!(!workers.remove(&first.key));
182        assert!(workers.add(&first).is_ok());
183        let Ok(timings) = block_on(workers.tick_all(2, true)) else {
184            panic!("remaining worker should finish its tick");
185        };
186        assert_eq!(timings.len(), 2);
187        assert_eq!(first.game_time(), 2);
188        assert_eq!(second.game_time(), 2);
189    }
190
191    #[test]
192    fn persistent_workers_tick_every_world_across_boundaries() {
193        let first = fresh_test_world("persistent_worker_first");
194        let second = fresh_test_world("persistent_worker_second");
195        let Ok(workers) = WorldTickWorkers::spawn([&first, &second]) else {
196            panic!("world tick workers should start");
197        };
198
199        let Ok(first_tick) = block_on(workers.tick_all(1, true)) else {
200            panic!("world tick workers should finish the first tick");
201        };
202        assert_eq!(first_tick.len(), 2);
203        assert_eq!(first.game_time(), 1);
204        assert_eq!(second.game_time(), 1);
205
206        let Ok(second_tick) = block_on(workers.tick_all(2, true)) else {
207            panic!("world tick workers should finish the second tick");
208        };
209        assert_eq!(second_tick.len(), 2);
210        assert_eq!(first.game_time(), 2);
211        assert_eq!(second.game_time(), 2);
212    }
