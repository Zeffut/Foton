1//! Ticks per second over the last one, five and fifteen minutes.
2//!
3//! Not vanilla: vanilla reports tick *time*, which says nothing once the
4//! server falls behind and every tick starts late. This is what Paper's
5//! `Server.getTPS()` measures -- wall-clock time between tick starts, sampled
6//! once a second -- because plugins read it as a load signal and compare it to
7//! the numbers a Paper server would give.
8
9use std::time::Instant;
10
11/// Ticks between samples: one second at the nominal rate.
12const SAMPLE_INTERVAL: u64 = 20;
13
14const NANOS_PER_SECOND: f64 = 1_000_000_000.0;
15
16/// A time-weighted average of the last `N` one-second samples.
17///
18/// Weighted by each sample's real duration, so a ten-second stall counts ten
19/// times as much as a normal second rather than once.
20struct RollingAverage<const N: usize> {
21    samples: [f64; N],
22    durations: [f64; N],
23    index: usize,
24    /// Sum of `durations`, kept alongside so reading is O(1).
25    total_duration: f64,
26    /// Sum of `samples[i] * durations[i]`.
27    weighted_total: f64,
28}
29
30impl<const N: usize> RollingAverage<N> {
31    /// Starts full of perfect seconds, as Paper does, so the first minute does
32    /// not read as a server that was idle before it booted.
33    const fn new(nominal: f64) -> Self {
34        Self {
35            samples: [nominal; N],
36            durations: [NANOS_PER_SECOND; N],
37            index: 0,
38            total_duration: NANOS_PER_SECOND * N as f64,
39            weighted_total: nominal * NANOS_PER_SECOND * N as f64,
40        }
41    }
42
43    fn add(&mut self, tps: f64, duration: f64) {
44        self.total_duration -= self.durations[self.index];
45        self.weighted_total -= self.samples[self.index] * self.durations[self.index];
46        self.samples[self.index] = tps;
47        self.durations[self.index] = duration;
48        self.total_duration += duration;
49        self.weighted_total += tps * duration;
50        self.index = (self.index + 1) % N;
51    }
52
53    fn average(&self) -> f64 {
54        if self.total_duration <= 0.0 {
55            return 0.0;
56        }
57        self.weighted_total / self.total_duration
58    }
59}
60
61/// The three averages `getTPS()` returns, fed once per tick.
62pub struct TpsAverages {
63    one_minute: RollingAverage<60>,
64    five_minutes: RollingAverage<300>,
65    fifteen_minutes: RollingAverage<900>,
66    /// The tick the current sample started at, and when, or `None` before
67    /// the first tick.
68    section_start: Option<(u64, Instant)>,
69}
70
71impl TpsAverages {
72    /// Averages that read as a steady `nominal` until real samples arrive.
73    #[must_use]
74    pub const fn new(nominal: f64) -> Self {
75        Self {
76            one_minute: RollingAverage::new(nominal),
77            five_minutes: RollingAverage::new(nominal),
78            fifteen_minutes: RollingAverage::new(nominal),
79            section_start: None,
80        }
81    }
82
83    /// Records that tick number `tick` started at `now`.
84    pub fn tick_started(&mut self, tick: u64, now: Instant) {
85        let Some((start_tick, start)) = self.section_start else {
86            self.section_start = Some((tick, now));
87            return;
88        };
89        if !tick.is_multiple_of(SAMPLE_INTERVAL) || tick <= start_tick {
90            return;
91        }
92        let elapsed = now.duration_since(start).as_nanos() as f64;
93        self.section_start = Some((tick, now));
94        if elapsed <= 0.0 {
95            return;
96        }
97        let tps = NANOS_PER_SECOND * (tick - start_tick) as f64 / elapsed;
98        self.one_minute.add(tps, elapsed);
99        self.five_minutes.add(tps, elapsed);
100        self.fifteen_minutes.add(tps, elapsed);
101    }
102
103    /// Ticks per second over the last 1, 5 and 15 minutes, in that order.
104    #[must_use]
105    pub fn averages(&self) -> [f64; 3] {
106        [
107            self.one_minute.average(),
108            self.five_minutes.average(),
109            self.fifteen_minutes.average(),
110        ]
111    }
112}

114#[cfg(test)]
115mod tests {
116    use std::time::Duration;
117
118    use super::*;
119
120    #[test]
121    fn a_stall_lowers_the_short_average_more_than_the_long_one() {
122        let mut tps = TpsAverages::new(20.0);
123        let mut now = Instant::now();
124        tps.tick_started(0, now);
125        // Forty ticks at half speed: two samples of 10 TPS, two seconds each.
126        for tick in 1..=40 {
127            now += Duration::from_millis(100);
128            tps.tick_started(tick, now);
129        }
130        let [one, five, fifteen] = tps.averages();
131        assert!(one < five && five < fifteen, "{one} {five} {fifteen}");
132        // 58 seconds at 20 and 4 seconds at 10, weighted by duration.
133        let expected = (58.0 * 20.0 + 4.0 * 10.0) / 62.0;
134        assert!((one - expected).abs() < 1e-9, "{one} != {expected}");
135    }
