202    packet_workers_for_available(configured_workers, available_worker_threads())

205fn available_worker_threads() -> usize {
206    thread::available_parallelism().map_or(4, NonZero::get)
207}
208
209fn cap_positive_thread_count(
210    configured_threads: Option<usize>,
211    available_threads: usize,
212) -> Option<usize> {
213    let configured_threads = configured_threads.filter(|&threads| threads > 0)?;
214    Some(configured_threads.min(available_threads.max(1)))
215}
216
217fn packet_workers_for_available(
218    configured_workers: Option<usize>,
219    available_threads: usize,
220) -> usize {
221    let available_threads = available_threads.max(1);
222    if let Some(configured_workers) = configured_workers.filter(|&workers| workers > 0) {
223        return configured_workers.min(available_threads);
224    }
225
226    ((available_threads / 2).max(2)).min(available_threads)
227}

295fn can_entity_return_from_end_to_overworld(
296    entity: &dyn Entity,
297    projectile_owner_seen_credits: impl Fn(&uuid::Uuid) -> Option<bool>,
298) -> bool {
299    if entity.entity_type() == &vanilla_entities::ENDER_PEARL
300        && entity
301            .projectile_owner_uuid()
302            .and_then(|uuid| projectile_owner_seen_credits(&uuid))
303            == Some(false)
304    {
305        return false;
306    }
307
308    direct_passengers_allow_end_return(entity)
309}
310
311fn direct_passengers_allow_end_return(entity: &dyn Entity) -> bool {
312    for passenger in entity.passengers() {
313        if passenger
314            .as_player()
315            .is_some_and(|player| !player.has_seen_credits())
316        {
317            return false;
318        }
319    }
320
321    true
322}

542/// A handle on a world being built off-thread.
543///
544/// Poll it from a safe-point with [`WorldCreationRequest::poll`], or await it
545/// from outside the game tick; never block the tick on one.
546pub struct WorldCreationRequest {
547    id: u64,
548    receiver: oneshot::Receiver<Result<(), String>>,
549}
550
551impl WorldCreationRequest {
552    /// This request's identifier, unique for the lifetime of the server.
553    ///
554    /// Lets a caller that fired several creations tell the answers apart.
555    #[must_use]
556    pub const fn id(&self) -> u64 {
557        self.id
558    }
559
560    /// Polls the request without waiting for asynchronous world construction.
561    pub fn poll(&mut self) -> WorldCreationState {
562        match self.receiver.try_recv() {
563            Ok(Ok(())) => WorldCreationState::Ready,
564            Ok(Err(error)) => WorldCreationState::Failed(error),
565            Err(Empty) => WorldCreationState::Pending,
566            Err(Closed) => WorldCreationState::Failed("world creation task closed".to_owned()),
567        }
568    }
569
570    /// Awaits completion for callers already outside the game tick.
571    pub async fn wait(self) -> WorldCreationState {
572        match self.receiver.await {
573            Ok(Ok(())) => WorldCreationState::Ready,
574            Ok(Err(error)) => WorldCreationState::Failed(error),
575            Err(_) => WorldCreationState::Failed("world creation task closed".to_owned()),
576        }
577    }
578}

607impl Server {
608    /// Schedules creation of a plugin-requested world without blocking the game
609    /// tick. Completion is reported only after safe-point attachment.
610    pub fn request_world_creation(
611        self: &Arc<Self>,
612        name: String,
613        generator: Identifier,
614        seed: i64,
615        bonus_chest: bool,
616    ) -> Result<WorldCreationRequest, String> {
617        if name.is_empty() || !Identifier::validate_path(&name) {
618            return Err(format!("invalid world name {name}"));
619        }
620        let domain = self.worlds.default_domain().to_owned();
621        let key = Identifier::new(domain.clone(), name.clone());
622        if self.worlds.get(&key).is_some() {
623            return Err(format!("world {key} is already loaded"));
624        }
625        let generator_config = self
626            .world_generator_registry
627            .validate_config(&generator, &toml::Value::Table(Map::new()))?;
628        let world_entry = ResolvedWorldConfig {
629            key,
630            domain,
631            name,
632            generator,
633            generator_config,
634            seed,
635            default_gamemode: GameType::Survival,
636            difficulty: Normal,
637            bonus_chest,
638            storage: StorageSelection::default_world_disk(),
639            nether_portal_target: None,
640            end_portal_target: None,
641        };
642        let (sender, receiver) = oneshot::channel();
643        let server = Arc::clone(self);
644        self.chunk_runtime.spawn(async move {
645            let result = match server.load_world_from_config_tracked(world_entry).await {
646                Ok(request) => match request.wait().await {
647                    WorldCreationState::Ready => Ok(()),
648                    WorldCreationState::Failed(error) => Err(error),
649                    WorldCreationState::Pending => {
650                        Err("world creation request ended pending".to_owned())
651                    }
652                },
653                Err(error) => Err(error),
654            };
655            let _ = sender.send(result);
656        });
657        let id = NEXT_WORLD_CREATION_ID.fetch_add(1, Ordering::Relaxed);
658        Ok(WorldCreationRequest { id, receiver })
659    }

759    /// Creates a new server with only Foton's built-in commands.
760    /// Builds a world using the same validated generator/storage pipeline as startup.
761    ///
762    /// The caller must insert the returned world into `WorldMap` and attach it at a
763    /// tick safe-point.
764    pub(crate) async fn build_world_from_config(
765        &self,
766        chunk_runtime: Arc<Runtime>,
767        world_entry: &ResolvedWorldConfig,
768    ) -> Result<Arc<World>, String> {
769        let world_path = self
770            .world_save_path
771            .join(&world_entry.domain)
772            .join("worlds")
773            .join(&world_entry.name);
774        let storage_output = self
775            .world_storage_registry
776            .create(&world_entry.storage, &self.world_save_path, &world_path)
777            .map_err(|error| {
778                format!("failed to create storage for {}: {error}", world_entry.key)
779            })?;
780        let world_seed = LevelDataManager::load_seed_or_default(
781            storage_output.level_data_path.as_deref(),
782            world_entry.seed,
783        )
784        .await
785        .map_err(|error| {
786            format!(
787                "failed to load level data seed for {}: {error}",
788                world_entry.key
789            )
790        })?;
791        let generator_output = self
792            .world_generator_registry
793            .create(
794                storage_output.level_data_path.as_deref(),
795                &world_entry.generator_config,
796                world_seed,
797                Arc::clone(&self.generation_pool),
798            )
799            .map_err(|error| {
800                format!(
801                    "failed to create generator for {}: {error}",
802                    world_entry.key
803                )
804            })?;
805        let generation_settings = generation_settings_for_world(world_entry, &generator_output);
806        let world = World::new_with_config_and_encoding_pool(
807            chunk_runtime,
808            world_entry.key.clone(),
809            generator_output.dimension_type,
810            world_seed,
811            WorldConfig {
812                storage: storage_output.storage,
813                level_data_path: storage_output
814                    .level_data_path
815                    .map(|path| path.to_string_lossy().into_owned()),
816                generator: Arc::new(generator_output.generator),
817                generation_settings,
818                view_distance: self.config.view_distance,
819                simulation_distance: self.config.simulation_distance,
820                max_chained_neighbor_updates: self.config.max_chained_neighbor_updates,
821                compression: self.config.compression,
822                is_flat: generator_output.is_flat,
823                sea_level: generator_output.sea_level,
824                default_gamemode: world_entry.default_gamemode,
825                difficulty: world_entry.difficulty,
826                bonus_chest: world_entry.bonus_chest,
827            },
828            Arc::clone(&self.generation_pool),
829            Arc::clone(&self.chunk_encoding_pool),
830        )
831        .await
832        .map_err(|error| format!("failed to create world {}: {error}", world_entry.key))?;
833        world
834            .initialize_spawn_if_needed_with_bonus_chest(world_entry.bonus_chest)
835            .await
836            .map_err(|error| {
837                format!(
838                    "failed to initialize spawn for {}: {error}",
839                    world_entry.key
840                )
841            })?;
842        Ok(world)
843    }

845    pub(crate) async fn load_world_from_config_tracked(
846        self: &Arc<Self>,
847        world_entry: ResolvedWorldConfig,
848    ) -> Result<WorldCreationRequest, String> {
849        if self.worlds.get(&world_entry.key).is_some() {
850            return Err(format!("world {} is already loaded", world_entry.key));
851        }
852        let world = self
853            .build_world_from_config(Arc::clone(&self.chunk_runtime), &world_entry)
854            .await?;
855        world.attach_server(self);
856        let id = NEXT_WORLD_CREATION_ID.fetch_add(1, Ordering::Relaxed);
857        let (sender, receiver) = oneshot::channel();
858        let mut pending = self.pending_world_additions.lock();
859        if pending
860            .iter()
861            .any(|(pending_world, _)| pending_world.key == world_entry.key)
862        {
863            return Err(format!(
864                "world {} is already pending attachment",
865                world_entry.key
866            ));
867        }
868        pending.push((world, sender));
869        Ok(WorldCreationRequest { id, receiver })
870    }

899    /// Creates a new server and atomically merges startup command extensions after built-ins.

901        clippy::too_many_lines,
902        reason = "server initialization is a single cohesive flow"
903    )]
904    pub async fn new_with_commands(
905        chunk_runtime: Arc<Runtime>,
906        cancel_token: CancellationToken,
907        config: RuntimeConfig,
908        worlds_config: WorldsConfig,
909        permission_groups: PermissionGroupManager,
910        command_registry: CommandRegistry,
911    ) -> Result<Self, String> {
912        validate_login_security(config.online_mode, config.encryption).map_err(str::to_owned)?;
913        let config = Arc::new(config);
914        init_globals()?;
915        log::info!(
916            "Foton is not affiliated with Mojang or Microsoft. Use is subject to the Minecraft EULA: https://aka.ms/MinecraftEULA"
917        );
918
919        // Authlib starts this fetch alongside server initialization and waits on first use.
920        // Foton completes the same initial attempt before opening its listener.
921        let service_keys = Arc::new(
922            ServiceKeyStore::new(config.services_server.as_deref())
923                .map_err(|error| format!("failed to configure Minecraft services keys: {error}"))?,
924        );
925        let service_keys_ready = service_keys.start(cancel_token.clone());
926
927        let registry_cache = RegistryCache::new(config.compression);
928
929        let (generator_registry, storage_registry) = world_config_registries()?;
930        let resolved_worlds = worlds_config
931            .validate_and_resolve(&generator_registry, &storage_registry)
932            .map_err(|e| format!("failed to validate worlds.toml: {e}"))?;
933
934        let generation_pool: Arc<ThreadPool> = Arc::new({
935            let mut builder = ThreadPoolBuilder::new().thread_name(|i| format!("rayon-gen-{i}"));
936            if let Some(chunk_generation_threads) =
937                configured_chunk_generation_threads(config.chunk_generation_threads)
938            {
939                builder = builder.num_threads(chunk_generation_threads);
940            }
941            // Debug builds have deep call chains in density functions that overflow the default 2 MB stack
942            if cfg!(debug_assertions) {
943                builder = builder.stack_size(8 * 1024 * 1024);
944            }
945            builder
946                .build()
947                .map_err(|e| format!("failed to create generation thread pool: {e}"))?
948        });
949        let chunk_encoding_pool = Arc::new({
950            let mut builder =
951                ThreadPoolBuilder::new().thread_name(|i| format!("rayon-chunk-encode-{i}"));
952            if let Some(chunk_encoding_threads) =
953                configured_chunk_encoding_threads(config.chunk_encoding_threads)
954            {
955                builder = builder.num_threads(chunk_encoding_threads);
956            }
957            builder
958                .build()
959                .map_err(|e| format!("failed to create chunk encoding thread pool: {e}"))?
960        });
961
962        let player_data_storage = PlayerDataStorage::new(
963            resolved_worlds.save_path.clone(),
964            resolved_worlds.player_storage.clone(),
965        )
966        .await
967        .map_err(|e| format!("failed to create player data storage: {e}"))?;
968        let player_permission_states = player_data_storage
969            .load_permission_subjects()
970            .await
971            .map_err(|error| format!("failed to load player permissions: {error}"))?;
972        let known_players = player_data_storage
973            .load_known_players()
974            .await
975            .map_err(|error| format!("failed to load known players: {error}"))?;
976        let mut global_player_data = FxHashMap::default();
977        for known in known_players.entries() {
978            if let Some(data) = player_data_storage
979                .load_global(known.uuid())
980                .await
981                .map_err(|error| format!("failed to load global data: {error}"))?
982            {
983                let mut data = data;
984                if !data.last_active_domain.is_empty()
985                    && let Some(domain_data) = player_data_storage
986                        .load_domain(&data.last_active_domain, known.uuid())
987                        .await
988                        .map_err(|error| format!("failed to load player statistics: {error}"))?
989                {
990                    data.statistics = domain_data.statistics;
991                }
992                global_player_data.insert(known.uuid(), data);
993            }
994        }
995        let worlds = WorldMap::new(
996            resolved_worlds.default_domain.clone(),
997            &resolved_worlds.domains,
998            &resolved_worlds.worlds,
999        );
1000
1001        for world_entry in &resolved_worlds.worlds {
1002            let default_world_path = resolved_worlds
1003                .save_path
1004                .join(&world_entry.domain)
1005                .join("worlds")
1006                .join(&world_entry.name);
1007            let storage_output = storage_registry
1008                .create(
1009                    &world_entry.storage,
1010                    &resolved_worlds.save_path,
1011                    Path::new(&default_world_path),
1012                )
1013                .map_err(|e| format!("failed to create storage for {}: {e}", world_entry.key))?;
1014            let world_seed = LevelDataManager::load_seed_or_default(
1015                storage_output.level_data_path.as_deref(),
1016                world_entry.seed,
1017            )
1018            .await
1019            .map_err(|e| {
1020                format!(
1021                    "failed to load level data seed for {}: {e}",
1022                    world_entry.key
1023                )
1024            })?;
1025            let generator_output = generator_registry
1026                .create(
1027                    storage_output.level_data_path.as_deref(),
1028                    &world_entry.generator_config,
1029                    world_seed,
1030                    generation_pool.clone(),
1031                )
1032                .map_err(|e| format!("failed to create generator for {}: {e}", world_entry.key))?;
1033            let generation_settings = generation_settings_for_world(world_entry, &generator_output);
1034            let world = World::new_with_config_and_encoding_pool(
1035                chunk_runtime.clone(),
1036                world_entry.key.clone(),
1037                generator_output.dimension_type,
1038                world_seed,
1039                WorldConfig {
1040                    storage: storage_output.storage,
1041                    level_data_path: storage_output
1042                        .level_data_path
1043                        .map(|path| path.to_string_lossy().into_owned()),
1044                    generator: Arc::new(generator_output.generator),
1045                    generation_settings,
1046                    view_distance: config.view_distance,
1047                    simulation_distance: config.simulation_distance,
1048                    max_chained_neighbor_updates: config.max_chained_neighbor_updates,
1049                    compression: config.compression,
1050                    is_flat: generator_output.is_flat,
1051                    sea_level: generator_output.sea_level,
1052                    default_gamemode: world_entry.default_gamemode,
1053                    difficulty: world_entry.difficulty,
1054                    bonus_chest: world_entry.bonus_chest,
1055                },
1056                generation_pool.clone(),
1057                Arc::clone(&chunk_encoding_pool),
1058            )
1059            .await
1060            .map_err(|e| format!("failed to create world {}: {e}", world_entry.key))?;
1061            world
1062                .initialize_spawn_if_needed_with_bonus_chest(world_entry.bonus_chest)
1063                .await
1064                .map_err(|e| format!("failed to initialize spawn for {}: {e}", world_entry.key))?;
1065            worlds
1066                .insert(world_entry.key.clone(), world)
1067                .map_err(|error| format!("failed to publish world {}: {error}", world_entry.key))?;
1068        }
1069
1070        let scoreboards = DomainScoreboards::load(&worlds)
1071            .await
1072            .map_err(|error| format!("failed to load domain scoreboards: {error}"))?;
1073        let command_storage = DomainCommandStorage::load(&worlds)
1074            .await
1075            .map_err(|error| format!("failed to load domain command storage: {error}"))?;
1076        let map_data = DomainMapData::load(&worlds)
1077            .map_err(|error| format!("failed to load domain map data: {error}"))?;
1078        let boss_bars = DomainCustomBossEvents::load(&worlds)
1079            .await
1080            .map_err(|error| format!("failed to load domain boss bars: {error}"))?;
1081        let registered_commands = create_registered_dispatcher(command_registry)
1082            .map_err(|error| format!("failed to register commands: {error}"))?;
1083        let command_permission_keys = registered_commands
1084            .permissions
1085            .into_iter()
1086            .map(|permission| permission.as_str().to_owned())
1087            .collect();
1088
1089        if service_keys_ready.await.is_err() {
1090            log::error!("Minecraft services key fetch task stopped before its initial attempt");
1091        }
1092
1093        Ok(Server {
1094            config,
1095            chunk_runtime,
1096            permission_groups,
1097            cancel_token,
1098            key_store: KeyStore::create()
1099                .map_err(|error| format!("Failed to create the server encryption keys: {error}"))?,
1100            worlds,
1101            world_save_path: resolved_worlds.save_path.clone(),
1102            online_players: PlayerMap::new(),
1103            player_admissions: SyncMutex::new(FxHashMap::default()),
1104            registry_cache,
1105            tick_rate_manager: SyncRwLock::new(TickRateManager::new()),
1106            scoreboards,
1107            command_storage,
1108            map_data,
1109            boss_bars,
1110            command_dispatcher: SyncRwLock::new(registered_commands.dispatcher),
1111            functions: FunctionManager::new(resolved_worlds.save_path.join("datapacks")),
1112            command_permission_keys,
1113            command_requests: CommandRequestQueue::new(),
1114            packet_processor: PacketProcessor::new(),
1115            world_generator_registry: generator_registry,
1116            world_storage_registry: storage_registry,
1117            generation_pool,
1118            chunk_encoding_pool,
1119            jobs: ServerJobQueue::new(),
1120            player_data_storage,
1121            global_player_data: SyncRwLock::new(global_player_data),
1122            player_permission_states: SyncRwLock::new(player_permission_states),
1123            player_permission_updates: AsyncMutex::new(()),
1124            known_players: SyncMutex::new(KnownPlayerCacheState::new(known_players)),

1213    /// Runs one command to completion inside the current tick.
1214    ///
1215    /// Vanilla parity: the `commands.performPrefixedCommand` of
1216    /// `BaseCommandBlock.performCommand`, which runs synchronously because the
1217    /// command block reads its own success count in the same tick -- that count
1218    /// is what a comparator reports and what gates a conditional chain.
1219    ///
1220    /// [`Self::submit_command`] is the queued path used by chat and the
1221    /// console; it cannot serve a command block, because the queue is drained
1222    /// before the world tick that would read the result.
1223    ///
1224    /// A command that suspends -- today only a structure search waiting on
1225    /// chunk generation -- is cancelled rather than carried into the next tick,
1226    /// and counts as no successes. Vanilla blocks the server thread there
1227    /// instead; a game tick may not wait.
1228    pub(crate) fn run_command_now(self: &Arc<Self>, source: CommandSource, command: &str) -> i32 {
1229        let successes = Arc::new(AtomicI32::new(0));
1230        let counter = Arc::clone(&successes);
1231        let source = source.with_callback(CommandResultCallback::new(move |success, _result| {
1232            if success {
1233                counter.fetch_add(1, AtomicOrdering::Relaxed);
1234            }
1235        }));
1236
1237        let command = command.strip_prefix('/').unwrap_or(command);
1238        let chain = {
1239            let dispatcher = self.command_dispatcher.read();
1240            let parse = dispatcher.parse(command, source.clone());
1241            dispatcher.context_chain(parse)
1242        };
1243        let chain = match chain {
1244            Ok(chain) => chain,
1245            Err(error) => {
1246                source.handle_error(&error, false);
1247                return 0;
1248            }
1249        };
1250
1251        let mut execution = CommandExecutionContext::for_source(&source);
1252        let callback = source.callback();
1253        execution.queue_initial_command(chain, source, callback);
1254        if execution.run() == ExecutionStop::Suspended {
1255            log::warn!(
1256                "command block command `{command}` suspended on chunk work and was cancelled"
1257            );
1258            execution.cancel();
1259        }
1260
1261        successes.load(AtomicOrdering::Relaxed)
1262    }

1276    /// Queues one Rcon command and hands back the reply its client waits on.
1277    ///
1278    /// Vanilla parity: `DedicatedServer.runCommand`, which clears the shared
1279    /// Rcon buffer, blocks the Rcon thread on the server thread with
1280    /// `executeBlocking`, and reads the buffer back. A Foton tick may not be
1281    /// blocked on, so the wait moves to the caller: the reply arrives when the
1282    /// last handle to the command's output sink is dropped, which happens
1283    /// whether the command completed, failed to parse, hit the command limit,
1284    /// overflowed the execution queue, or was cancelled at shutdown.
1285    ///
1286    /// # Errors
1287    /// Returns [`CommandQueueFull`] when the command request queue is full,
1288    /// in which case no reply will ever be sent.
1289    pub fn submit_rcon_command(
1290        &self,
1291        connection: u64,
1292        command: String,
1293    ) -> Result<oneshot::Receiver<String>, CommandQueueFull> {
1294        let (output, reply) = RconOutput::new(connection);
1295        self.submit_command(CommandSender::Rcon(Arc::new(output)), command)?;
1296        Ok(reply)
1297    }

1323    /// Returns Brigadier completions visible to a command sender.
1324    pub fn command_completions(
1325        self: &Arc<Self>,
1326        sender: CommandSender,
1327        input: &str,
1328    ) -> Vec<CommandCompletion> {
1329        if !CommandExecutionOwner::capture(sender.clone(), self).is_current(self) {
1330            return Vec::new();
1331        }
1332        match self.build_command_suggestions(sender, input) {
1333            Ok(suggestions) => {
1334                let range = suggestions.range();
1335                suggestions
1336                    .list()
1337                    .iter()
1338                    .map(|suggestion| {
1339                        CommandCompletion::new(
1340                            range.start(),
1341                            range.len(),
1342                            suggestion.text().to_owned(),
1343                        )
1344                    })
1345                    .collect()
1346            }
1347            Err(error) => {
1348                tracing::warn!(%error, "failed to build command suggestions");
1349                Vec::new()
1350            }
1351        }
1352    }
