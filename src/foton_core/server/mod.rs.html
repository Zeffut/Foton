202fn configured_packet_workers(configured_workers: Option<usize>) -> usize {
203    packet_workers_for_available(configured_workers, available_worker_threads())
204}
205
206fn available_worker_threads() -> usize {
207    thread::available_parallelism().map_or(4, NonZero::get)
208}
209
210fn cap_positive_thread_count(
211    configured_threads: Option<usize>,
212    available_threads: usize,
213) -> Option<usize> {
214    let configured_threads = configured_threads.filter(|&threads| threads > 0)?;
215    Some(configured_threads.min(available_threads.max(1)))
216}
217
218fn packet_workers_for_available(
219    configured_workers: Option<usize>,
220    available_threads: usize,
221) -> usize {
222    let available_threads = available_threads.max(1);
223    if let Some(configured_workers) = configured_workers.filter(|&workers| workers > 0) {
224        return configured_workers.min(available_threads);
225    }
226
227    ((available_threads / 2).max(2)).min(available_threads)
228}

296fn can_entity_return_from_end_to_overworld(
297    entity: &dyn Entity,
298    projectile_owner_seen_credits: impl Fn(&uuid::Uuid) -> Option<bool>,
299) -> bool {
300    if entity.entity_type() == &vanilla_entities::ENDER_PEARL
301        && entity
302            .projectile_owner_uuid()
303            .and_then(|uuid| projectile_owner_seen_credits(&uuid))
304            == Some(false)
305    {
306        return false;
307    }
308
309    direct_passengers_allow_end_return(entity)
310}
311
312fn direct_passengers_allow_end_return(entity: &dyn Entity) -> bool {
313    for passenger in entity.passengers() {
314        if passenger
315            .as_player()
316            .is_some_and(|player| !player.has_seen_credits())
317        {
318            return false;
319        }
320    }
321
322    true
323}

578/// A handle on a world being built off-thread.
579///
580/// Poll it from a safe-point with [`WorldCreationRequest::poll`], or await it
581/// from outside the game tick; never block the tick on one.
582pub struct WorldCreationRequest {
583    id: u64,
584    receiver: oneshot::Receiver<Result<(), String>>,
585}
586
587impl WorldCreationRequest {
588    /// This request's identifier, unique for the lifetime of the server.
589    ///
590    /// Lets a caller that fired several creations tell the answers apart.
591    #[must_use]
592    pub const fn id(&self) -> u64 {
593        self.id
594    }
595
596    /// Polls the request without waiting for asynchronous world construction.
597    pub fn poll(&mut self) -> WorldCreationState {
598        match self.receiver.try_recv() {
599            Ok(Ok(())) => WorldCreationState::Ready,
600            Ok(Err(error)) => WorldCreationState::Failed(error),
601            Err(Empty) => WorldCreationState::Pending,
602            Err(Closed) => WorldCreationState::Failed("world creation task closed".to_owned()),
603        }
604    }
605
606    /// Awaits completion for callers already outside the game tick.
607    pub async fn wait(self) -> WorldCreationState {
608        match self.receiver.await {
609            Ok(Ok(())) => WorldCreationState::Ready,
610            Ok(Err(error)) => WorldCreationState::Failed(error),
611            Err(_) => WorldCreationState::Failed("world creation task closed".to_owned()),
612        }
613    }
614}

645    /// tick. Completion is reported only after safe-point attachment.
646    pub fn request_world_creation(
647        self: &Arc<Self>,
648        name: String,
649        generator: Identifier,
650        seed: i64,
651        bonus_chest: bool,
652    ) -> Result<WorldCreationRequest, String> {
653        if name.is_empty() || !Identifier::validate_path(&name) {
654            return Err(format!("invalid world name {name}"));
655        }
656        let domain = self.worlds.default_domain().to_owned();
657        let key = Identifier::new(domain.clone(), name.clone());
658        let generator_config = self
659            .world_generator_registry
660            .validate_config(&generator, &toml::Value::Table(Map::new()))?;
661        let world_entry = ResolvedWorldConfig {
662            key: key.clone(),
663            domain,
664            name,
665            generator,
666            generator_config,
667            seed,
668            default_gamemode: GameType::Survival,
669            difficulty: Normal,
670            bonus_chest,
671            storage: StorageSelection::default_world_disk(),
672            nether_portal_target: None,
673            end_portal_target: None,
674        };
675        let (sender, receiver) = oneshot::channel();
676        let server = Arc::clone(self);
677        self.spawn_tracked_world_creation(&key, move |reservation| async move {
678            let result = match server
679                .load_world_from_config_tracked(world_entry, reservation)
680                .await
681            {
682                Ok(request) => match request.wait().await {
683                    WorldCreationState::Ready => Ok(()),
684                    WorldCreationState::Failed(error) => Err(error),
685                    WorldCreationState::Pending => {
686                        Err("world creation request ended pending".to_owned())
687                    }
688                },
689                Err(error) => Err(error),
690            };
691            let _ = sender.send(result);
692        })?;
693        let id = NEXT_WORLD_CREATION_ID.fetch_add(1, Ordering::Relaxed);
694        Ok(WorldCreationRequest { id, receiver })
695    }

697    fn spawn_tracked_world_creation<F, Fut>(
698        self: &Arc<Self>,
699        key: &Identifier,
700        task: F,
701    ) -> Result<(), String>
702    where
703        F: FnOnce(WorldKeyReservation) -> Fut,
704        Fut: Future<Output = ()> + Send + 'static,
705    {
706        let _admission = self.pending_world_additions.lock();
707        if self.world_creation_closed.load(Ordering::Acquire) {
708            return Err("world creation is closed".to_owned());
709        }
710        let reservation = self.reserve_world_key_without_admission(key)?;
711        self.world_tasks
712            .creations
713            .spawn_on(task(reservation), self.chunk_runtime.handle());
714        Ok(())
715    }

815    /// Creates a new server with only Foton's built-in commands.
816    /// Builds a world using the same validated generator/storage pipeline as startup.
817    ///
818    /// The caller must insert the returned world into `WorldMap` and attach it at a
819    /// tick safe-point.
820    pub(crate) async fn build_world_from_config(
821        &self,
822        chunk_runtime: Arc<Runtime>,
823        world_entry: &ResolvedWorldConfig,
824    ) -> Result<Arc<World>, String> {
825        let world_path = self
826            .world_save_path
827            .join(&world_entry.domain)
828            .join("worlds")
829            .join(&world_entry.name);
830        let storage_output = self
831            .world_storage_registry
832            .create(&world_entry.storage, &self.world_save_path, &world_path)
833            .map_err(|error| {
834                format!("failed to create storage for {}: {error}", world_entry.key)
835            })?;
836        let world_seed = LevelDataManager::load_seed_or_default(
837            storage_output.level_data_path.as_deref(),
838            world_entry.seed,
839        )
840        .await
841        .map_err(|error| {
842            format!(
843                "failed to load level data seed for {}: {error}",
844                world_entry.key
845            )
846        })?;
847        let generator_output = self
848            .world_generator_registry
849            .create(
850                storage_output.level_data_path.as_deref(),
851                &world_entry.generator_config,
852                world_seed,
853                Arc::clone(&self.generation_pool),
854            )
855            .map_err(|error| {
856                format!(
857                    "failed to create generator for {}: {error}",
858                    world_entry.key
859                )
860            })?;
861        let generation_settings = generation_settings_for_world(world_entry, &generator_output);
862        let world = World::new_with_config_and_encoding_pool(
863            chunk_runtime,
864            world_entry.key.clone(),
865            generator_output.dimension_type,
866            world_seed,
867            WorldConfig {
868                storage: storage_output.storage,
869                level_data_path: storage_output
870                    .level_data_path
871                    .map(|path| path.to_string_lossy().into_owned()),
872                generator: Arc::new(generator_output.generator),
873                generation_settings,
874                view_distance: self.config.view_distance,
875                simulation_distance: self.config.simulation_distance,
876                max_chained_neighbor_updates: self.config.max_chained_neighbor_updates,
877                compression: self.config.compression,
878                is_flat: generator_output.is_flat,
879                sea_level: generator_output.sea_level,
880                default_gamemode: world_entry.default_gamemode,
881                difficulty: world_entry.difficulty,
882                bonus_chest: world_entry.bonus_chest,
883            },
884            Arc::clone(&self.generation_pool),
885            Arc::clone(&self.chunk_encoding_pool),
886        )
887        .await
888        .map_err(|error| format!("failed to create world {}: {error}", world_entry.key))?;
889        if let Err(error) = world
890            .initialize_spawn_if_needed_with_bonus_chest(world_entry.bonus_chest)
891            .await
892        {
893            world.cleanup_without_save().await;
894            return Err(format!(
895                "failed to initialize spawn for {}: {error}",
896                world_entry.key
897            ));
898        }
899        Ok(world)
900    }

902    async fn load_world_from_config_tracked(
903        self: &Arc<Self>,
904        world_entry: ResolvedWorldConfig,
905        reservation: WorldKeyReservation,
906    ) -> Result<WorldCreationRequest, String> {
907        if self.worlds.get(&world_entry.key).is_some() {
908            return Err(format!("world {} is already loaded", world_entry.key));
909        }
910        let world = self
911            .build_world_from_config(Arc::clone(&self.chunk_runtime), &world_entry)
912            .await?;
913        world.attach_server(self);
914        let id = NEXT_WORLD_CREATION_ID.fetch_add(1, Ordering::Relaxed);
915        let (sender, receiver) = oneshot::channel();
916        let insertion = {
917            let mut pending = self.pending_world_additions.lock();
918            if self.world_creation_closed.load(Ordering::Acquire) {
919                Err("world creation is closed".to_owned())
920            } else if pending
921                .iter()
922                .any(|(pending_world, _, _)| pending_world.key == world_entry.key)
923            {
924                Err(format!(
925                    "world {} is already pending attachment",
926                    world_entry.key
927                ))
928            } else {
929                pending.push((Arc::clone(&world), sender, reservation));
930                Ok(())
931            }
932        };
933        if let Err(error) = insertion {
934            world.cleanup_without_save().await;
935            return Err(error);
936        }
937        Ok(WorldCreationRequest { id, receiver })
938    }

967    /// Creates a new server and atomically merges startup command extensions after built-ins.
968    #[expect(
969        clippy::too_many_lines,
970        reason = "server initialization is a single cohesive flow"
971    )]
972    pub async fn new_with_commands(
973        chunk_runtime: Arc<Runtime>,
974        cancel_token: CancellationToken,
975        config: RuntimeConfig,
976        worlds_config: WorldsConfig,
977        permission_groups: PermissionGroupManager,
978        command_registry: CommandRegistry,
979    ) -> Result<Self, String> {
980        validate_login_security(config.online_mode, config.encryption).map_err(str::to_owned)?;
981        let config = Arc::new(config);
982        init_globals()?;
983        log::info!(
984            "Foton is not affiliated with Mojang or Microsoft. Use is subject to the Minecraft EULA: https://aka.ms/MinecraftEULA"
985        );
986
987        // Authlib starts this fetch alongside server initialization and waits on first use.
988        // Foton completes the same initial attempt before opening its listener.
989        let service_keys = Arc::new(
990            ServiceKeyStore::new(config.services_server.as_deref())
991                .map_err(|error| format!("failed to configure Minecraft services keys: {error}"))?,
992        );
993        let service_keys_ready = service_keys.start(cancel_token.clone());
994
995        let registry_cache = RegistryCache::new(config.compression);
996
997        let (generator_registry, storage_registry) = world_config_registries()?;
998        let resolved_worlds = worlds_config
999            .validate_and_resolve(&generator_registry, &storage_registry)
1000            .map_err(|e| format!("failed to validate worlds.toml: {e}"))?;
1001
1002        let generation_pool: Arc<ThreadPool> = Arc::new({
1003            let mut builder = ThreadPoolBuilder::new().thread_name(|i| format!("rayon-gen-{i}"));
1004            if let Some(chunk_generation_threads) =
1005                configured_chunk_generation_threads(config.chunk_generation_threads)
1006            {
1007                builder = builder.num_threads(chunk_generation_threads);
1008            }
1009            // Debug builds have deep call chains in density functions that overflow the default 2 MB stack
1010            if cfg!(debug_assertions) {
1011                builder = builder.stack_size(8 * 1024 * 1024);
1012            }
1013            builder
1014                .build()
1015                .map_err(|e| format!("failed to create generation thread pool: {e}"))?
1016        });
1017        let chunk_encoding_pool = Arc::new({
1018            let mut builder =
1019                ThreadPoolBuilder::new().thread_name(|i| format!("rayon-chunk-encode-{i}"));
1020            if let Some(chunk_encoding_threads) =
1021                configured_chunk_encoding_threads(config.chunk_encoding_threads)
1022            {
1023                builder = builder.num_threads(chunk_encoding_threads);
1024            }
1025            builder
1026                .build()
1027                .map_err(|e| format!("failed to create chunk encoding thread pool: {e}"))?
1028        });
1029
1030        let player_data_storage = PlayerDataStorage::new(
1031            resolved_worlds.save_path.clone(),
1032            resolved_worlds.player_storage.clone(),
1033        )
1034        .await
1035        .map_err(|e| format!("failed to create player data storage: {e}"))?;
1036        let player_permission_states = player_data_storage
1037            .load_permission_subjects()
1038            .await
1039            .map_err(|error| format!("failed to load player permissions: {error}"))?;
1040        let known_players = player_data_storage
1041            .load_known_players()
1042            .await
1043            .map_err(|error| format!("failed to load known players: {error}"))?;
1044        let mut global_player_data = FxHashMap::default();
1045        for known in known_players.entries() {
1046            if let Some(data) = player_data_storage
1047                .load_global(known.uuid())
1048                .await
1049                .map_err(|error| format!("failed to load global data: {error}"))?
1050            {
1051                let mut data = data;
1052                if !data.last_active_domain.is_empty()
1053                    && let Some(domain_data) = player_data_storage
1054                        .load_domain(&data.last_active_domain, known.uuid())
1055                        .await
1056                        .map_err(|error| format!("failed to load player statistics: {error}"))?
1057                {
1058                    data.statistics = domain_data.statistics;
1059                }
1060                global_player_data.insert(known.uuid(), data);
1061            }
1062        }
1063        let worlds = WorldMap::new(
1064            resolved_worlds.default_domain.clone(),
1065            &resolved_worlds.domains,
1066            &resolved_worlds.worlds,
1067        );
1068
1069        for world_entry in &resolved_worlds.worlds {
1070            let default_world_path = resolved_worlds
1071                .save_path
1072                .join(&world_entry.domain)
1073                .join("worlds")
1074                .join(&world_entry.name);
1075            let storage_output = storage_registry
1076                .create(
1077                    &world_entry.storage,
1078                    &resolved_worlds.save_path,
1079                    Path::new(&default_world_path),
1080                )
1081                .map_err(|e| format!("failed to create storage for {}: {e}", world_entry.key))?;
1082            let world_seed = LevelDataManager::load_seed_or_default(
1083                storage_output.level_data_path.as_deref(),
1084                world_entry.seed,
1085            )
1086            .await
1087            .map_err(|e| {
1088                format!(
1089                    "failed to load level data seed for {}: {e}",
1090                    world_entry.key
1091                )
1092            })?;
1093            let generator_output = generator_registry
1094                .create(
1095                    storage_output.level_data_path.as_deref(),
1096                    &world_entry.generator_config,
1097                    world_seed,
1098                    generation_pool.clone(),
1099                )
1100                .map_err(|e| format!("failed to create generator for {}: {e}", world_entry.key))?;
1101            let generation_settings = generation_settings_for_world(world_entry, &generator_output);
1102            let world = World::new_with_config_and_encoding_pool(
1103                chunk_runtime.clone(),
1104                world_entry.key.clone(),
1105                generator_output.dimension_type,
1106                world_seed,
1107                WorldConfig {
1108                    storage: storage_output.storage,
1109                    level_data_path: storage_output
1110                        .level_data_path
1111                        .map(|path| path.to_string_lossy().into_owned()),
1112                    generator: Arc::new(generator_output.generator),
1113                    generation_settings,
1114                    view_distance: config.view_distance,
1115                    simulation_distance: config.simulation_distance,
1116                    max_chained_neighbor_updates: config.max_chained_neighbor_updates,
1117                    compression: config.compression,
1118                    is_flat: generator_output.is_flat,
1119                    sea_level: generator_output.sea_level,
1120                    default_gamemode: world_entry.default_gamemode,
1121                    difficulty: world_entry.difficulty,
1122                    bonus_chest: world_entry.bonus_chest,
1123                },
1124                generation_pool.clone(),
1125                Arc::clone(&chunk_encoding_pool),
1126            )
1127            .await
1128            .map_err(|e| format!("failed to create world {}: {e}", world_entry.key))?;
1129            world
1130                .initialize_spawn_if_needed_with_bonus_chest(world_entry.bonus_chest)
1131                .await
1132                .map_err(|e| format!("failed to initialize spawn for {}: {e}", world_entry.key))?;
1133            worlds
1134                .insert(world_entry.key.clone(), world)
1135                .map_err(|error| format!("failed to publish world {}: {error}", world_entry.key))?;
1136        }
1137
1138        let scoreboards = DomainScoreboards::load(&worlds)
1139            .await
1140            .map_err(|error| format!("failed to load domain scoreboards: {error}"))?;
1141        let command_storage = DomainCommandStorage::load(&worlds)
1142            .await
1143            .map_err(|error| format!("failed to load domain command storage: {error}"))?;
1144        let map_data = DomainMapData::load(&worlds)
1145            .map_err(|error| format!("failed to load domain map data: {error}"))?;
1146        let boss_bars = DomainCustomBossEvents::load(&worlds)
1147            .await
1148            .map_err(|error| format!("failed to load domain boss bars: {error}"))?;
1149        let registered_commands = create_registered_dispatcher(command_registry)
1150            .map_err(|error| format!("failed to register commands: {error}"))?;
1151        let command_permission_keys = registered_commands
1152            .permissions
1153            .into_iter()
1154            .map(|permission| permission.as_str().to_owned())
1155            .collect();
1156
1157        if service_keys_ready.await.is_err() {
1158            log::error!("Minecraft services key fetch task stopped before its initial attempt");
1159        }
1160
1161        Ok(Server {
1162            config,
1163            chunk_runtime,
1164            world_tasks: WorldTaskTrackers::new(),
1165            world_key_reservations: Arc::new(SyncMutex::new(FxHashSet::default())),
1166            permission_groups,
1167            cancel_token,
1168            key_store: KeyStore::create()
1169                .map_err(|error| format!("Failed to create the server encryption keys: {error}"))?,
1170            worlds,
1171            world_save_path: resolved_worlds.save_path.clone(),
1172            online_players: PlayerMap::new(),
1173            player_admissions: SyncMutex::new(FxHashMap::default()),
1174            registry_cache,
1175            tick_rate_manager: SyncRwLock::new(TickRateManager::new()),
1176            scoreboards,
1177            command_storage,
1178            map_data,
1179            boss_bars,
1180            command_dispatcher: SyncRwLock::new(registered_commands.dispatcher),
1181            functions: FunctionManager::new(resolved_worlds.save_path.join("datapacks")),
1182            command_permission_keys,
1183            command_requests: CommandRequestQueue::new(),
1184            packet_processor: PacketProcessor::new(),
1185            world_generator_registry: generator_registry,
1186            world_storage_registry: storage_registry,
1187            generation_pool,
1188            chunk_encoding_pool,
1189            jobs: ServerJobQueue::new(),
1190            player_data_storage,
1191            global_player_data: SyncRwLock::new(global_player_data),
1192            player_permission_states: SyncRwLock::new(player_permission_states),
1193            player_permission_updates: AsyncMutex::new(()),
1194            known_players: SyncMutex::new(KnownPlayerCacheState::new(known_players)),

1196            profile_lookup_client: reqwest::Client::new(),

1199            pending_player_disconnects: PlayerDisconnectQueue::new(),

1238    ///
1239    /// A world is built before the server that owns it, so the link cannot be

1248    /// Closes admission of detached world cleanups and waits for those already
1249    /// queued so shutdown cannot race a removed world's final save. Worlds
1250    /// still waiting for attachment are cleaned up here as well.
1251    pub async fn wait_for_world_cleanups(&self) {
1252        let pending = {
1253            let mut pending = self.pending_world_additions.lock();
1254            self.world_creation_closed.store(true, Ordering::Release);
1255            self.world_tasks.creations.close();
1256            mem::take(&mut *pending)
1257        };
1258        for (world, completion, reservation) in pending {
1259            self.world_tasks.cleanups.spawn_on(
1260                async move {
1261                    world.cleanup_without_save().await;
1262                    drop(reservation);
1263                    let _ = completion.send(Err(
1264                        "world creation stopped before safe-point attachment".to_owned(),
1265                    ));
1266                },
1267                self.chunk_runtime.handle(),
1268            );
1269        }
1270        self.world_tasks.cleanups.close();
1271        tokio::join!(
1272            self.world_tasks.creations.wait(),
1273            self.world_tasks.cleanups.wait()
1274        );
1275    }

1347    /// Runs one command to completion inside the current tick.
1348    ///
1349    /// Vanilla parity: the `commands.performPrefixedCommand` of
1350    /// `BaseCommandBlock.performCommand`, which runs synchronously because the
1351    /// command block reads its own success count in the same tick -- that count
1352    /// is what a comparator reports and what gates a conditional chain.
1353    ///
1354    /// [`Self::submit_command`] is the queued path used by chat and the
1355    /// console; it cannot serve a command block, because the queue is drained
1356    /// before the world tick that would read the result.
1357    ///
1358    /// A command that suspends -- today only a structure search waiting on
1359    /// chunk generation -- is cancelled rather than carried into the next tick,
1360    /// and counts as no successes. Vanilla blocks the server thread there
1361    /// instead; a game tick may not wait.
1362    pub(crate) fn run_command_now(self: &Arc<Self>, source: CommandSource, command: &str) -> i32 {
1363        let successes = Arc::new(AtomicI32::new(0));
1364        let counter = Arc::clone(&successes);
1365        let source = source.with_callback(CommandResultCallback::new(move |success, _result| {
1366            if success {
1367                counter.fetch_add(1, AtomicOrdering::Relaxed);
1368            }
1369        }));
1370
1371        let command = command.strip_prefix('/').unwrap_or(command);
1372        let chain = {
1373            let dispatcher = self.command_dispatcher.read();
1374            let parse = dispatcher.parse(command, source.clone());
1375            dispatcher.context_chain(parse)
1376        };
1377        let chain = match chain {
1378            Ok(chain) => chain,
1379            Err(error) => {
1380                source.handle_error(&error, false);
1381                return 0;
1382            }
1383        };
1384
1385        let mut execution = CommandExecutionContext::for_source(&source);
1386        let callback = source.callback();
1387        execution.queue_initial_command(chain, source, callback);
1388        if execution.run() == ExecutionStop::Suspended {
1389            log::warn!(
1390                "command block command `{command}` suspended on chunk work and was cancelled"
1391            );
1392            execution.cancel();
1393        }
1394
1395        successes.load(AtomicOrdering::Relaxed)
1396    }

1410    /// Queues one Rcon command and hands back the reply its client waits on.
1411    ///
1412    /// Vanilla parity: `DedicatedServer.runCommand`, which clears the shared
1413    /// Rcon buffer, blocks the Rcon thread on the server thread with
1414    /// `executeBlocking`, and reads the buffer back. A Foton tick may not be
1415    /// blocked on, so the wait moves to the caller: the reply arrives when the
1416    /// last handle to the command's output sink is dropped, which happens
1417    /// whether the command completed, failed to parse, hit the command limit,
1418    /// overflowed the execution queue, or was cancelled at shutdown.
1419    ///
1420    /// # Errors
1421    /// Returns [`CommandQueueFull`] when the command request queue is full,
1422    /// in which case no reply will ever be sent.
1423    pub fn submit_rcon_command(
1424        &self,
1425        connection: u64,
1426        command: String,
1427    ) -> Result<oneshot::Receiver<String>, CommandQueueFull> {
1428        let (output, reply) = RconOutput::new(connection);
1429        self.submit_command(CommandSender::Rcon(Arc::new(output)), command)?;
1430        Ok(reply)
1431    }

1457    /// Returns Brigadier completions visible to a command sender.
1458    pub fn command_completions(
1459        self: &Arc<Self>,
1460        sender: CommandSender,
1461        input: &str,
1462    ) -> Vec<CommandCompletion> {
1463        if !CommandExecutionOwner::capture(sender.clone(), self).is_current(self) {
1464            return Vec::new();
1465        }
1466        match self.build_command_suggestions(sender, input) {
1467            Ok(suggestions) => {
1468                let range = suggestions.range();
1469                suggestions
1470                    .list()
1471                    .iter()
1472                    .map(|suggestion| {
1473                        CommandCompletion::new(
1474                            range.start(),
1475                            range.len(),
1476                            suggestion.text().to_owned(),
1477                        )
1478                    })
1479                    .collect()
1480            }
1481            Err(error) => {
1482                tracing::warn!(%error, "failed to build command suggestions");
1483                Vec::new()
1484            }
1485        }
1486    }
