1//! The items that hang an entity on a wall.
2//!
3//! Vanilla parity: `HangingEntityItem`. Foton maps only the painting to this
4//! behavior -- the two frame items go through `ItemFrameItem`, which predates
5//! it -- so the frame branches of `useOn` have no counterpart here.
6
7use std::sync::Arc;
8
9use foton_macros::item_behavior;
10use foton_registry::data_components::vanilla_components::PAINTING_VARIANT;
11use foton_registry::entity_type::EntityTypeRef;
12use foton_registry::{sound_events, vanilla_game_events};
13
14use crate::behavior::{InteractionResult, ItemBehavior, UseOnContext};
15use crate::entity::entities::PaintingEntity;
16use crate::entity::{Entity as _, next_entity_id};
17use crate::event::HangingPlaceEvent;
18use crate::world::game_event::GameEventContext;
19
20/// Behavior for the items that hang an entity on a wall.
21#[item_behavior]
22pub struct HangingEntityItem {
23    /// The entity this item hangs.
24    #[json_arg(vanilla_entities, json = "type")]
25    entity_type: EntityTypeRef,
26}
27
28impl HangingEntityItem {
29    /// Creates a hanging entity item behavior.
30    #[must_use]
31    pub const fn new(entity_type: EntityTypeRef) -> Self {
32        Self { entity_type }
33    }
34}
35
36impl ItemBehavior for HangingEntityItem {
37    /// Vanilla parity: `HangingEntityItem.useOn`.
38    ///
39    /// A painting needs a wall, so a click on the top or bottom of a block
40    /// places nothing. Which picture goes up is decided by what fits, and a
41    /// wall too small for any of them consumes the click rather than failing
42    /// it -- vanilla returns `CONSUME` there.
43    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
44        let face = context.hit_result.direction;
45        // Vanilla parity: `HangingEntityItem.mayPlace`. Its second half,
46        // `Player.mayUseItemAt`, has no counterpart -- Foton has no
47        // adventure-mode placement permission on an item stack yet.
48        if !face.is_horizontal() {
49            return InteractionResult::Fail;
50        }
51
52        let painting_pos = context.hit_result.block_pos.relative(face);
53        let Some(painting) = PaintingEntity::create(
54            self.entity_type,
55            next_entity_id(),
56            context.world,
57            painting_pos,
58            face,
59        ) else {
60            return InteractionResult::Consume;
61        };
62        let painting = Arc::new(painting);
63
64        // Vanilla parity: the `EntityType.createDefaultStackConfig` of `useOn`,
65        // which copies the stack's implicit components onto the entity. Only
66        // `PAINTING_VARIANT` is copied here: a stack that names its picture
67        // overrides the one `create` rolled, so a player who crafted a
68        // specific painting hangs that one.
69        let stack_variant = context.inv.with_item(|item| {
70            item.get(PAINTING_VARIANT)
71                .and_then(|component| component.variant().as_reference())
72        });
73        if let Some(variant) = stack_variant {
74            painting.set_variant(variant);
75        }
76
77        if !painting.survives() {
78            return InteractionResult::Consume;
79        }
80
81        let position = painting.position();
82        let mut event = HangingPlaceEvent::new(
83            painting.uuid(),
84            context.player.uuid(),
85            context.world.key.to_string(),
86            context.hit_result.block_pos,
87            format!("{face:?}"),
88        );
89        context.world.fire_event(&mut event);
90        if event.is_cancelled() {
91            return InteractionResult::Fail;
92        }
93        painting.play_sound(&sound_events::ENTITY_PAINTING_PLACE, 1.0, 1.0);
94        context.world.game_event_at(
95            &vanilla_game_events::ENTITY_PLACE,
96            position,
97            &GameEventContext::new(Some(context.player), None),
98        );
99
100        if context.world.try_add_entity(painting).is_err() {
101            return InteractionResult::Fail;
102        }
103
104        // Deviation: vanilla shrinks unconditionally and lets the creative
105        // inventory put the stack back. Foton has no such restore, so the
106        // creative case is skipped here, as it is for the item frame.
107        if !context.player.has_infinite_materials() {
108            context.inv.with_item(|item| item.shrink(1));
109        }
110
111        InteractionResult::Success
112    }
113}
