1//! Wind charge item.
2//!
3//! Vanilla parity: `WindChargeItem`. Throwing one spawns a [`WindChargeEntity`]
4//! from the thrower's eye along their look. The half-second cooldown that keeps
5//! a stack from being emptied in one burst comes from the item's own
6//! `minecraft:use_cooldown` component, not from anything written here.
7
8use std::sync::Arc;
9
10use foton_macros::item_behavior;
11use foton_protocol::packets::game::SoundSource;
12use foton_registry::{sound_events, vanilla_entities};
13use glam::DVec3;
14
15use crate::behavior::context::{InteractionResult, UseItemContext};
16use crate::behavior::item::ItemBehavior;
17use crate::entity::entities::WindChargeEntity;
18use crate::entity::{Entity, Projectile, SharedEntity, next_entity_id};

20/// How hard a wind charge is thrown.
21///
22/// Vanilla parity: `WindChargeItem.PROJECTILE_SHOOT_POWER`.
23const SHOOT_POWER: f32 = 1.5;
24
25/// How wide the throw scatters.
26///
27/// Vanilla parity: the `1.0F` uncertainty of the
28/// `Projectile.spawnProjectileFromRotation` call in `WindChargeItem.use`.
29const SHOOT_UNCERTAINTY: f32 = 1.0;
30
31/// Behavior for the wind charge item.
32#[item_behavior]
33pub struct WindChargeItem;
34
35impl ItemBehavior for WindChargeItem {
36    /// Vanilla parity: `WindChargeItem.use`.
37    ///
38    /// Vanilla returns `InteractionResult.SUCCESS` and the caller then applies
39    /// the `Item.Properties.useCooldown(0.5F)` the item was registered with.
40    /// Foton does the same: `use_item` applies the stack's
41    /// `minecraft:use_cooldown` for any result that
42    /// `should_apply_item_use_side_effects`, so the half second is data, not
43    /// code, and the only thing owed here is the `Success`.
44    ///
45    /// Not implemented: the dispenser path. Vanilla's `WindChargeItem` is a
46    /// `ProjectileItem` whose `asProjectile` and `createDispenseConfig` let a
47    /// dispenser fire one, with its own triangular spread and the 1051 dispense
48    /// event. Foton's dispense table has no entry for it yet, so a dispenser
49    /// throws the item instead.
50    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
51        let player = context.player;
52        let world = context.world;
53
54        // Vanilla parity: the deliberately jittery pitch of `WindChargeItem.use`.
55        // Vanilla plays this after spawning but unconditionally, so it is played
56        // first here rather than being lost to a spawn Foton could not complete.
57        let sound_pitch = 0.4 / rand::random::<f32>().mul_add(0.4, 0.8);
58        world.play_sound_at(
59            &sound_events::ENTITY_WIND_CHARGE_THROW,
60            SoundSource::Neutral,
61            player.position(),
62            0.5,
63            sound_pitch,
64            None,
65        );
66
67        // Vanilla parity: `WindChargeItem.use` spawns at the thrower's eye
68        // height over their feet's horizontal position, and unlike a snowball
69        // it does not drop that spawn by 0.1.
70        let player_pos = player.position();
71        let charge = Arc::new(WindChargeEntity::new(
72            &vanilla_entities::WIND_CHARGE,
73            next_entity_id(),
74            DVec3::new(player_pos.x, player.get_eye_y(), player_pos.z),
75            Arc::downgrade(world),
76        ));
77
78        if let Some(owner) = world.players.get_by_uuid(&player.gameprofile.id) {
79            let owner: SharedEntity = owner;
80            charge.set_owner_entity(Some(&owner));
81        } else {
82            charge.set_owner_uuid(Some(player.gameprofile.id));
83        }
84
85        let (yaw, pitch) = player.rotation();
86        charge.shoot_from_rotation(player, pitch, yaw, 0.0, SHOOT_POWER, SHOOT_UNCERTAINTY);
87
88        let charge: SharedEntity = charge;
89        if let Err(error) = world.try_add_entity(charge) {
90            log::debug!("failed to spawn wind charge: {error}");
91            return InteractionResult::Fail;
92        }
93
94        // TODO: award the ITEM_USED stat once a stats system exists.
95        if !player.has_infinite_materials() {
96            context.inv.with_item(|item| item.shrink(1));
97        }
98
99        InteractionResult::Success
100    }
101}

103#[cfg(test)]
104mod tests {
105    use foton_registry::item_stack::ItemStack;
106    use foton_registry::{init_vanilla_registry, vanilla_items};
107    use foton_utils::types::InteractionHand;
108    use foton_utils::{ChunkPos, WorldAabb};
109
110    use crate::behavior::init_behaviors;
111    use crate::player::game_mode::use_item;
112    use crate::test_support::{TestPlayerBuilder, fresh_test_world, insert_ready_full_chunk};
113
114    use super::*;
115
116    #[test]
117    fn throwing_a_wind_charge_sends_one_out_at_eye_height_and_spends_it() {
118        init_vanilla_registry();
119        init_behaviors();
120        let world = fresh_test_world("wind_charge_item_throw");
121        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
122        let player =
123            TestPlayerBuilder::new(Arc::clone(&world), "Thrower", next_entity_id()).build();
124        player
125            .inventory
126            .lock()
127            .set_selected_item(ItemStack::with_count(&vanilla_items::WIND_CHARGE, 3));
128
129        let result = use_item(&player, &world, InteractionHand::MainHand);
130
131        assert_eq!(result, InteractionResult::Success);
132        let thrown = world
133            .get_entities_in_aabb(&WorldAabb::new(-8.0, -8.0, -8.0, 8.0, 24.0, 8.0))
134            .into_iter()
135            .find(|entity| entity.entity_type() == &vanilla_entities::WIND_CHARGE)
136            .expect("using a wind charge should put one in the world");
137
138        assert!((thrown.position().y - player.get_eye_y()).abs() < 1.0e-9);
139        assert_eq!(thrown.projectile_owner_uuid(), Some(player.gameprofile.id));
140        assert!(
141            thrown.velocity().length() > 0.0,
142            "the charge should leave the hand already moving"
143        );
144        assert_eq!(player.inventory.lock().get_selected_item_mut().count, 2);
145    }
146
147    #[test]
148    fn throwing_a_wind_charge_puts_the_stack_on_its_half_second_cooldown() {
149        init_vanilla_registry();
150        init_behaviors();
151        let world = fresh_test_world("wind_charge_item_cooldown");
152        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
153        let player =
154            TestPlayerBuilder::new(Arc::clone(&world), "Thrower", next_entity_id()).build();
155        let stack = ItemStack::with_count(&vanilla_items::WIND_CHARGE, 2);
156        player.inventory.lock().set_selected_item(stack.clone());
157
158        assert!(!player.is_item_on_cooldown(&stack));
159        assert_eq!(
160            use_item(&player, &world, InteractionHand::MainHand),
161            InteractionResult::Success
162        );
163
164        // Vanilla parity: `Item.Properties.useCooldown(0.5F)` on the registered
165        // item, which the use path applies for a successful result.
166        assert!(player.is_item_on_cooldown(&stack));
167        assert_eq!(
168            use_item(&player, &world, InteractionHand::MainHand),
169            InteractionResult::Pass,
170            "a second throw is refused while the cooldown runs"
171        );
172    }
