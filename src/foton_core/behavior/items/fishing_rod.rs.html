1//! Fishing rod item behavior (`FishingRodItem`).
2//!
3//! One right-click casts a [`FishingHookEntity`]; the next reels it in, taking
4//! the durability the hook reports back. Lure and Luck of the Sea are read off
5//! the rod at cast time and travel with the bobber, exactly as vanilla does --
6//! swapping to a different rod mid-cast does not change the odds.
7
8use std::sync::Arc;
9
10use foton_macros::item_behavior;
11use foton_protocol::packets::game::SoundSource;
12use foton_registry::data_components::vanilla_components::USE_EFFECTS;
13use foton_registry::game_events::GameEventRef;
14use foton_registry::{sound_events, vanilla_entities, vanilla_game_events};
15use foton_utils::Downcast as _;
16
17use crate::behavior::context::{InteractionResult, UseItemContext};
18use crate::behavior::item::ItemBehavior;
19use crate::enchantment_helper;
20use crate::entity::entities::FishingHookEntity;
21use crate::entity::{Entity as _, Projectile as _, SharedEntity, next_entity_id};
22
23/// Vanilla turns the `fishing_time_reduction` value effect, which Lure declares
24/// in seconds, into ticks shaved off the lure timer.
25const SECONDS_TO_TICKS: f32 = 20.0;
26
27/// Behavior for the fishing rod item.
28#[item_behavior(class = "FishingRodItem")]
29pub struct FishingRodItem;
30
31impl FishingRodItem {
32    /// Vanilla `ItemStack.causeUseVibration`, which a warden can hear.
33    fn cause_use_vibration(context: &mut UseItemContext, event: GameEventRef) {
34        let interact_vibrations = context.inv.with_item(|item| {
35            item.get(USE_EFFECTS)
36                .is_some_and(|effects| effects.interact_vibrations)
37        });
38        if interact_vibrations {
39            context.player.game_event(event);
40        }
41    }
42}
43
44impl ItemBehavior for FishingRodItem {
45    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
46        let player = context.player;
47        let world = context.world;
48        let pitch = 0.4 / (rand::random::<f32>() * 0.4 + 0.8);
49
50        if let Some(hook) = player.fishing_hook() {
51            let rod = context.inv.with_item(|item| item.clone());
52            let damage = hook
53                .as_ref()
54                .downcast_ref::<FishingHookEntity>()
55                .map_or(0, |hook| hook.retrieve(&rod));
56            player.hurt_item_in_hand(context.hand, damage);
57
58            world.play_sound_at(
59                &sound_events::ENTITY_FISHING_BOBBER_RETRIEVE,
60                SoundSource::Neutral,
61                player.position(),
62                1.0,
63                pitch,
64                None,
65            );
66            Self::cause_use_vibration(context, &vanilla_game_events::ITEM_INTERACT_FINISH);
67            return InteractionResult::Success;
68        }
69
70        world.play_sound_at(
71            &sound_events::ENTITY_FISHING_BOBBER_THROW,
72            SoundSource::Neutral,
73            player.position(),
74            0.5,
75            pitch,
76            None,
77        );
78
79        let rod = context.inv.with_item(|item| item.clone());
80        let lure_speed =
81            (enchantment_helper::get_fishing_time_reduction(&rod) * SECONDS_TO_TICKS) as i32;
82        let luck = enchantment_helper::get_fishing_luck_bonus(&rod);
83
84        let hook = Arc::new(FishingHookEntity::new(
85            &vanilla_entities::FISHING_BOBBER,
86            next_entity_id(),
87            player.position(),
88            Arc::downgrade(world),
89        ));
90        if let Some(owner) = world.players.get_by_uuid(&player.gameprofile.id) {
91            let owner: SharedEntity = owner;
92            hook.set_owner_entity(Some(&owner));
93        } else {
94            hook.set_owner_uuid(Some(player.gameprofile.id));
95        }
96        hook.cast_from(player, luck, lure_speed);
97
98        let entity: SharedEntity = hook;
99        if let Err(error) = world.try_add_entity(Arc::clone(&entity)) {
100            log::debug!("failed to cast a fishing hook: {error}");
101            return InteractionResult::Fail;
102        }
103        player.set_fishing_hook(Some(&entity));
104
105        // TODO: award the ITEM_USED stat once a stats system exists.
106        Self::cause_use_vibration(context, &vanilla_game_events::ITEM_INTERACT_START);
107        InteractionResult::Success
108    }
109}
