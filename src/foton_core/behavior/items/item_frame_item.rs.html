1//! Item frame items.
2//!
3//! Vanilla parity: `HangingEntityItem` as it is used for `ItemFrame` and
4//! `GlowItemFrame`. Without this a frame can only be summoned by a command,
5//! which is not something a player has -- and the frame entity, comparator
6//! output and all, has been in the tree with no way to reach it.
7
8use std::sync::Arc;
9
10use foton_macros::item_behavior;
11use foton_registry::entity_type::EntityTypeRef;
12use foton_registry::vanilla_game_events;
13
14use crate::behavior::{InteractionResult, ItemBehavior, UseOnContext};
15use crate::entity::entities::ItemFrameEntity;
16use crate::entity::{Entity as _, next_entity_id};
17use crate::event::HangingPlaceEvent;
18use crate::world::LevelReader as _;
19use crate::world::game_event::GameEventContext;
20
21/// Behavior for the item frame items.
22#[item_behavior]
23pub struct ItemFrameItem {
24    /// The frame this item hangs.
25    #[json_arg(vanilla_entities, json = "type")]
26    entity_type: EntityTypeRef,
27}
28
29impl ItemFrameItem {
30    /// Creates an item frame item behavior.
31    #[must_use]
32    pub const fn new(entity_type: EntityTypeRef) -> Self {
33        Self { entity_type }
34    }
35}
36
37impl ItemBehavior for ItemFrameItem {
38    /// Vanilla parity: `HangingEntityItem.useOn`.
39    ///
40    /// A frame hangs on the side of something, so a click on the top or bottom
41    /// of a block places nothing, and the block clicked has to be solid enough
42    /// to hold it.
43    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
44        let clicked = context.hit_result.block_pos;
45        let face = context.hit_result.direction;
46        if !face.is_horizontal() {
47            return InteractionResult::Fail;
48        }
49
50        // Vanilla parity: the support half of `HangingEntity.survives`. The
51        // collision and the rule against two frames sharing a square are not
52        // checked yet, so frames can be stacked where vanilla would refuse.
53        if !context
54            .world
55            .is_face_sturdy(context.world.get_block_state(clicked), clicked, face)
56        {
57            return InteractionResult::Fail;
58        }
59
60        let frame_pos = clicked.relative(face);
61        let frame = Arc::new(ItemFrameEntity::new_attached(
62            self.entity_type,
63            next_entity_id(),
64            frame_pos,
65            face,
66            Arc::downgrade(context.world),
67        ));
68        let position = frame.position();
69
70        let mut event = HangingPlaceEvent::new(
71            frame.uuid(),
72            context.player.uuid(),
73            context.world.key.to_string(),
74            clicked,
75            format!("{face:?}"),
76        );
77        context.world.fire_event(&mut event);
78        if event.is_cancelled() {
79            return InteractionResult::Fail;
80        }
81
82        if context.world.try_add_entity(frame).is_err() {
83            return InteractionResult::Fail;
84        }
85
86        context.world.game_event_at(
87            &vanilla_game_events::ENTITY_PLACE,
88            position,
89            &GameEventContext::new(Some(context.player), None),
90        );
91
92        if !context.player.has_infinite_materials() {
93            context.inv.with_item(|item| item.shrink(1));
94        }
95
96        InteractionResult::Success
97    }
98}
