1//! Minecart items.
2//!
3//! Vanilla parity: `MinecartItem`. Six items, one class, and the reason the
4//! minecart entity is reachable at all: without this a cart can only be
5//! summoned by a command, which is not something a player has.
6//!
7//! Which cart each item makes comes from the extracted `type`, so there is no
8//! table of variants here.
9

14use foton_registry::vanilla_game_events;

19use crate::entity::{ENTITIES, next_entity_id};

23/// How far above the rail block a cart sits.
24///
25/// Vanilla parity: the `0.0625` of `MinecartItem.useOn`, one pixel.
26const RAIL_HEIGHT: f64 = 0.0625;
27
28/// And how much higher on a slope.
29const SLOPE_LIFT: f64 = 0.5;
30
31/// Behavior for every minecart item.
32#[item_behavior]
33pub struct MinecartItem {
34    /// The cart this item puts on the rail.
35    #[json_arg(vanilla_entities, json = "type")]
36    entity_type: EntityTypeRef,
37}
38
39impl MinecartItem {
40    /// Creates a minecart item behavior.
41    #[must_use]
42    pub const fn new(entity_type: EntityTypeRef) -> Self {
43        Self { entity_type }
44    }
45}
46
47impl ItemBehavior for MinecartItem {
48    /// Vanilla parity: `MinecartItem.useOn`.
49    ///
50    /// A minecart goes on a rail and nowhere else, which is the whole of the
51    /// check: clicking anything that is not a rail fails outright rather than
52    /// falling through to placing something.
53    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
54        let pos = context.hit_result.block_pos;
55        let Some(shape) = rail_shape_at(context.world.get_block_state(pos)) else {
56            return InteractionResult::Fail;
57        };
58
59        let lift = if shape.is_slope() { SLOPE_LIFT } else { 0.0 };
60        let location = DVec3::new(
61            f64::from(pos.x()) + 0.5,
62            f64::from(pos.y()) + RAIL_HEIGHT + lift,
63            f64::from(pos.z()) + 0.5,
64        );
65
66        let Some(cart) = ENTITIES.create(
67            self.entity_type,
68            next_entity_id(),
69            location,
70            Arc::downgrade(context.world),
71        ) else {
72            // The item exists for every variant, but only the plain cart has an
73            // entity yet; using one of the others places nothing rather than
74            // crashing.
75            return InteractionResult::Fail;
76        };
77
78        if context.world.try_add_entity(cart).is_err() {
79            return InteractionResult::Fail;
80        }
81
82        context.world.game_event(
83            &vanilla_game_events::ENTITY_PLACE,
84            pos,
85            &GameEventContext::new(Some(context.player), None),
86        );
87
88        if !context.player.has_infinite_materials() {
89            context.inv.with_item(|item| item.shrink(1));
90        }
91
92        InteractionResult::Success
93    }
94}

96#[cfg(test)]
97mod tests {
98    use foton_registry::{init_vanilla_registry, vanilla_entities};
99
100    use super::*;
101
102    /// The item knows which cart it makes.
103    ///
104    /// The mapping is extracted data rather than a table here, so this is the
105    /// check that the codegen really passes it through -- an item pointing at
106    /// the wrong variant would put the wrong cart on the rail.
107    #[test]
108    fn a_minecart_item_names_its_own_cart() {
109        init_vanilla_registry();
110        let item = MinecartItem::new(&vanilla_entities::MINECART);
111        assert_eq!(item.entity_type, &vanilla_entities::MINECART);
112    }
