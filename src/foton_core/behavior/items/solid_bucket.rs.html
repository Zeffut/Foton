25/// A bucket that places a block instead of a fluid.
26///
27/// Vanilla parity: `SolidBucketItem`, a `BlockItem` with its own place sound
28/// that hands back an empty bucket.
29#[item_behavior]
30pub struct SolidBucketItem {
31    #[json_arg(vanilla_blocks, json = "block")]
32    block: BlockRef,
33    #[json_arg(sound_events, json = "place_sound")]
34    place_sound: SoundEventRef,
35    base: BlockItem,
36}
37
38impl SolidBucketItem {
39    /// Creates a solid bucket behavior.
40    #[must_use]
41    pub const fn new(block: BlockRef, place_sound: SoundEventRef) -> Self {
42        Self {
43            block,
44            place_sound,
45            base: BlockItem::new(block).with_place_sound(place_sound),
46        }
47    }
48}
49
50impl DispensibleContainerItem for SolidBucketItem {
51    /// Vanilla parity: `SolidBucketItem.emptyContents`, which is what lets a
52    /// dispenser lay powder snow. Unlike a fluid bucket it insists on an empty
53    /// block and never retries at the neighbour, so the hit result goes unread.
54    fn empty_contents(
55        &self,
56        user: Option<&Player>,
57        world: &Arc<World>,
58        pos: BlockPos,
59        _hit: Option<BucketHit>,
60    ) -> bool {
61        if !world.is_in_valid_bounds(pos) || !world.get_block_state(pos).is_air() {
62            return false;
63        }
64
65        world.set_block(pos, self.block.default_state(), UpdateFlags::UPDATE_ALL);
66        world.game_event(
67            &vanilla_game_events::FLUID_PLACE,
68            pos,
69            &GameEventContext::new(user.map(|player| player as &dyn Entity), None),
70        );
71        world.play_block_sound(self.place_sound, pos, 1.0, 1.0, None);
72        true
73    }
74}
75
76impl ItemBehavior for SolidBucketItem {
77    fn as_dispensible_container(&self) -> Option<&dyn DispensibleContainerItem> {
78        Some(self)
79    }
80
81    /// Vanilla parity: `SolidBucketItem.useOn`.
82    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
83        let result = self.base.place(context.build_place_context());
84        if !result.consumes_action() {
85            return result;
86        }
87
88        // Vanilla parity: `BucketItem.getEmptySuccessItem` -- an empty bucket,
89        // or the untouched stack in creative.
90        let empty = if context.player.has_infinite_materials() {
91            context
92                .inv
93                .with_item(|item| item.copy_with_count(item.count()))
94        } else {
95            ItemStack::new(&vanilla_items::BUCKET)
96        };
97        context.inv.with_item(|item| *item = empty);
98
99        result
100    }
