22/// Behavior for the sixteen dye items (`DyeItem`).
23///
24/// Ports vanilla `DyeItem.interactLivingEntity`: dying a sheep plays the `DYE_USE`
25/// sound, sets the sheep's wool color, and consumes one dye from the stack.
26/// Vanilla `DyeItem` is also a `SignApplicator`, which is what recolors sign text.
27#[item_behavior(class = "DyeItem")]
28pub struct DyeItem;
29
30impl ItemBehavior for DyeItem {
31    fn interact_living_entity(
32        &self,
33        stack: &mut ItemStack,
34        _player: &Player,
35        target: &dyn LivingEntity,
36        _hand: InteractionHand,
37    ) -> InteractionResult {
38        let Some(dye_color) = stack.get(DYE).copied() else {
39            return InteractionResult::Pass;
40        };
41        let Some(sheep) = target.downcast_ref::<SheepEntity>() else {
42            return InteractionResult::Pass;
43        };
44
45        if !Entity::is_alive(sheep) || sheep.is_sheared() || sheep.color() == dye_color {
46            return InteractionResult::Pass;
47        }
48
49        sheep.play_sound(&ITEM_DYE_USE, 1.0, 1.0);
50        sheep.set_color(dye_color);
51        stack.shrink(1);
52
53        InteractionResult::Success
54    }
55
56    fn as_sign_applicator(&self) -> Option<&dyn SignApplicator> {
57        Some(self)
58    }
59}
60
61impl SignApplicator for DyeItem {
62    /// Vanilla parity: `DyeItem.tryApplyToSign`. The color comes from the stack's
63    /// `minecraft:dye` component, which is what tells the sixteen dyes apart.
64    fn try_apply_to_sign(
65        &self,
66        world: &Arc<World>,
67        sign: &SignBlockEntity,
68        is_front_text: bool,
69        stack: &ItemStack,
70        _player: &Player,
71    ) -> bool {
72        let Some(dye) = stack.get(DYE).copied() else {
73            return false;
74        };
75        if !sign.update_text(|text| text.set_color(dye), is_front_text) {
76            return false;
77        }
78        world.play_sound(
79            &ITEM_DYE_USE,
80            SoundSource::Blocks,
81            sign.get_block_pos(),
82            1.0,
83            1.0,
84            None,
85        );
86        true
87    }
88}
