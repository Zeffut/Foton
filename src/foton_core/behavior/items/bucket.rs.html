41/// Handles all bucket variants (empty, water, lava).
42#[item_behavior]
43pub struct BucketItem {
44    #[json_arg(vanilla_blocks, json = "content", optional = "empty")]
45    fluid_block: Option<BlockRef>,
46}
47
48impl BucketItem {
49    /// Creates a new bucket behavior. `None` = empty bucket, `Some(block)` = filled.
50    #[must_use]
51    pub const fn new(fluid_block: Option<BlockRef>) -> Self {
52        Self { fluid_block }
53    }
54}
55
56impl ItemBehavior for BucketItem {
57    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
58        if self.fluid_block.is_none() {
59            return use_empty_bucket(context);
60        }
61        use_filled_bucket(self, self.fluid_block, context)
62    }
63
64    fn as_dispensible_container(&self) -> Option<&dyn DispensibleContainerItem> {
65        Some(self)
66    }
67}

69impl DispensibleContainerItem for BucketItem {
70    fn empty_contents(
71        &self,
72        user: Option<&Player>,
73        world: &Arc<World>,
74        pos: BlockPos,
75        hit: Option<BucketHit>,
76    ) -> bool {
77        // Vanilla parity: the `!(this.content instanceof FlowingFluid)` guard,
78        // which is what makes an empty bucket empty nothing.
79        let Some(fluid_block) = self.fluid_block else {
80            return false;
81        };
82        empty_contents(fluid_block, user, world, pos, hit, EmptySound::Fluid)
83    }
84}

86/// Which sound and side effects emptying a bucket produces.
87///
88/// Vanilla parity: `BucketItem.playEmptySound`, which `MobBucketItem` replaces
89/// with a neutral-category sound and no `FLUID_PLACE` game event.
90#[derive(Clone, Copy)]
91pub(super) enum EmptySound {
92    /// The fluid sounds of a plain water or lava bucket.
93    Fluid,
94    /// A mob bucket's own emptying sound.
95    Mob(SoundEventRef),
96}

98pub(super) fn filled_bucket_success_stack(context: &UseItemContext) -> ItemStack {
99    if context.player.has_infinite_materials() {
100        context
101            .inv
102            .with_item(|item| item.copy_with_count(item.count()))
103    } else {
104        ItemStack::new(&vanilla_items::BUCKET)
105    }
106}

108fn use_empty_bucket(context: &mut UseItemContext) -> InteractionResult {
109    // Vanilla parity: `getPlayerPOVHitResult(.., ClipContext.Fluid.SOURCE_ONLY)`.
110    // Vanilla returns PASS when the clip misses (allows other handlers to try).
111    let Some(hit_pos) = player_pov_hit_source_fluid(context) else {
112        return InteractionResult::Pass;
113    };
114
115    let hit_state = context.world.get_block_state(hit_pos);
116    let mut event = PlayerBucketFillEvent::new(
117        context.player.gameprofile.id,
118        context.world.key.to_string(),
119        hit_pos,
120        "BUCKET",
121    );
122    context.world.fire_event(&mut event);
123    if event.is_cancelled() {
124        return InteractionResult::Fail;
125    }
126    let block_behavior = BLOCK_BEHAVIORS.get_behavior(hit_state.get_block());
127
128    if let Some(result) =
129        block_behavior.pickup_block(context.world, hit_pos, hit_state, Some(context.player))
130    {
131        // Apply sound
132        if let Some(sound) = result.sound {
133            context
134                .world
135                .play_block_sound(sound, hit_pos, 1.0, 1.0, None);
136        }
137
138        // Give filled bucket
139        create_filled_result(context, result.filled_bucket, true);
140        context.world.game_event(
141            &vanilla_game_events::FLUID_PICKUP,
142            hit_pos,
143            &GameEventContext::new(Some(context.player), None),
144        );
145
146        return InteractionResult::Success;
147    }
148
149    // TODO: Remove fallback once all waterloggable blocks implement pickup_block.
150    if let Some(result) = pickup_waterlogged_block(
151        block_behavior,
152        context.world,
153        hit_pos,
154        hit_state,
155        Some(context.player),
156    ) {
157        if let Some(sound) = result.sound {
158            context
159                .world
160                .play_block_sound(sound, hit_pos, 1.0, 1.0, None);
161        }
162
163        create_filled_result(context, result.filled_bucket, true);
164        context.world.game_event(
165            &vanilla_game_events::FLUID_PICKUP,
166            hit_pos,
167            &GameEventContext::new(Some(context.player), None),
168        );
169
170        return InteractionResult::Success;
171    }
172
173    // Nothing was picked up — no fluid source block and no waterlogged block found.
174    // Vanilla returns FAIL here so the client knows no item change occurred.
175    InteractionResult::Fail
176}

178/// The block a filled bucket was aimed at.
179pub(super) struct FilledBucketTarget {
180    pub(super) clicked_pos: BlockPos,
181    pub(super) direction: Direction,
182    pub(super) clicked_state: BlockStateId,
183}
184
185/// Runs the clip that opens `BucketItem.use` for a filled bucket.
186///
187/// Vanilla parity: `getPlayerPOVHitResult(.., getFluidContext())`, which is
188/// `ClipContext.Fluid.NONE` for anything holding contents. Returns the refusal
189/// Vanilla would have produced when nothing usable was hit.
190pub(super) fn filled_bucket_target(
191    context: &UseItemContext<'_>,
192) -> Result<FilledBucketTarget, InteractionResult> {
193    let (start, end) = context.player.get_ray_endpoints();
194    let (ray_block, ray_dir) = context.world.raytrace(start, end, |pos, world| {
195        let state = world.get_block_state(pos);
196        let block = state.get_block();
197        // Filled buckets use ClipContext.Fluid.NONE: ignore fluid shapes, but
198        // still test the block shape of waterlogged/container blocks.
199        if block == &vanilla_blocks::AIR {
200            return RaytraceAction::Pass;
201        }
202        RaytraceAction::CheckShape
203    });
204
205    // Vanilla returns PASS when raytrace misses (allows other handlers to try)
206    let (Some(clicked_pos), Some(direction)) = (ray_block, ray_dir) else {
207        return Err(InteractionResult::Pass);
208    };
209
210    // If the block is out of bounds, return fail
211    if !context.world.is_in_valid_bounds(clicked_pos) {
212        return Err(InteractionResult::Fail);
213    }
214
215    Ok(FilledBucketTarget {
216        clicked_pos,
217        direction,
218        clicked_state: context.world.get_block_state(clicked_pos),
219    })
220}

222/// Uses a bucket that carries something, from the hand.
223///
224/// Vanilla parity: `BucketItem.use` for anything whose content is not
225/// `Fluids.EMPTY` -- the clip, then the item's own `emptyContents`, then
226/// `checkExtraContent` and `createFilledResult`.
227///
228/// `fluid_block` is `None` for a mob bucket that carries no fluid, which still
229/// takes this path: vanilla's `MobBucketItem.getFluidContext` is `NONE`
230/// whatever it holds, and its `emptyContents` succeeds on the sound alone.
231pub(super) fn use_filled_bucket(
232    item: &dyn DispensibleContainerItem,
233    fluid_block: Option<BlockRef>,
234    context: &mut UseItemContext,
235) -> InteractionResult {
236    let target = match filled_bucket_target(context) {
237        Ok(target) => target,
238        Err(result) => return result,
239    };
240    let FilledBucketTarget {
241        clicked_pos,
242        direction,
243        clicked_state,
244    } = target;
245
246    let is_water_bucket = fluid_block == Some(&vanilla_blocks::WATER);
247    let place_pos =
248        filled_bucket_primary_pos(clicked_state, clicked_pos, direction, is_water_bucket);
249    let hit = BucketHit {
250        block_pos: clicked_pos,
251        direction,
252    };
253
254    let bucket = if is_water_bucket {
255        "WATER_BUCKET"
256    } else if fluid_block == Some(&vanilla_blocks::LAVA) {
257        "LAVA_BUCKET"
258    } else {
259        "POWDER_SNOW_BUCKET"
260    };
261    let mut event = PlayerBucketEmptyEvent::new(
262        context.player.gameprofile.id,
263        context.world.key.to_string(),
264        place_pos,
265        bucket,
266    );
267    context.world.fire_event(&mut event);
268    if event.is_cancelled() {
269        return InteractionResult::Fail;
270    }
271
272    if !item.empty_contents(Some(context.player), context.world, place_pos, Some(hit)) {
273        return InteractionResult::Fail;
274    }
275
276    // Vanilla hands `checkExtraContent` the position it asked to be emptied,
277    // even when the fluid itself ended up one block further along.
278    let stack = context.inv.with_item(|item| item.copy_with_count(1));
279    item.check_extra_content(Some(context.player), context.world, &stack, place_pos);
280
281    let result_stack = filled_bucket_success_stack(context);
282    create_filled_result(context, result_stack, true);
283    InteractionResult::Success
284}

286/// Empties `fluid_block` into the world at `pos`.
287///
288/// Vanilla parity: `BucketItem.emptyContents`. Both `user` and `hit` are
289/// nullable in vanilla, and a dispenser supplies neither -- so nothing below
290/// this line may reach for a player.
291pub(super) fn empty_contents(
292    fluid_block: BlockRef,
293    user: Option<&Player>,
294    world: &Arc<World>,
295    pos: BlockPos,
296    hit: Option<BucketHit>,
297    empty_sound: EmptySound,
298) -> bool {
299    // The sneak check only applies while a hit result is in hand; vanilla's
300    // `!shiftKeyDown || hitResult == null` is what switches it off on the retry
301    // and for a dispenser.
302    if try_place_fluid(fluid_block, user, world, pos, hit.is_some(), empty_sound) {
303        return true;
304    }
305
306    // Vanilla parity: the single `emptyContents(user, level, hitResult
307    // .getBlockPos().relative(hitResult.getDirection()), null)` recursion. With
308    // no hit result there is nowhere to retry, which is why a dispenser gets
309    // exactly one attempt.
310    let Some(hit) = hit else {
311        return false;
312    };
313    try_place_fluid(
314        fluid_block,
315        user,
316        world,
317        hit.direction.relative(hit.block_pos),
318        false,
319        empty_sound,
320    )
321}

323/// Tries to put the bucket's fluid into one block.
324///
325/// `check_sneak` is vanilla's `!shiftKeyDown || hitResult == null`, false once
326/// there is no hit result left to redirect with.
327fn try_place_fluid(
328    fluid_block: BlockRef,
329    user: Option<&Player>,
330    world: &Arc<World>,
331    pos: BlockPos,
332    check_sneak: bool,
333    empty_sound: EmptySound,
334) -> bool {
335    if !world.is_in_valid_bounds(pos) {
336        return false;
337    }
338
339    let state = world.get_block_state(pos);
340    let fluid_state = state.get_fluid_state();
341
342    // Vanilla parity (bl4): when sneaking, only air allows placement at this position.
343    // Non-air blocks redirect to the neighbor — handled by the secondary call.
344    // The secondary call bypasses this check (hitResult == null in vanilla).
345    let is_sneaking = user.is_some_and(Player::is_crouching);
346    if check_sneak && is_sneaking && !state.get_block().config.is_air {
347        return false;
348    }
349
350    let is_water_bucket = fluid_block == &vanilla_blocks::WATER;
351    let behavior = BLOCK_BEHAVIORS.get_behavior(state.get_block());
352    let is_liquid_container = state.is_liquid_container();
353    let can_place_liquid = is_water_bucket
354        && is_liquid_container
355        && behavior.can_place_liquid_with_player(
356            state,
357            FluidState::source(&vanilla_fluids::WATER).fluid_id,
358            user,
359        );
360    let can_replace = state.can_be_replaced_by_fluid(fluid_block);
361
362    // Vanilla parity: block must be replaceable or liquid-container-admissible for placement.
363    if !can_replace && !can_place_liquid {
364        return false;
365    }
366
367    // Vanilla parity: in worlds where water evaporates (e.g. the Nether),
368    // water buckets fizz out without placing any fluid.
369    // TODO: Per-position environment attributes (vanilla uses EnvironmentAttributes.WATER_EVAPORATES per-pos)
370    if is_water_bucket && world.dimension_type.water_evaporates {
371        world.level_event(level_events::PARTICLES_WATER_EVAPORATING, pos, 0, None);
372        return true;
373    }
374
375    // 1. Try LiquidBlockContainer handling (only if Water bucket).
376    if is_water_bucket && is_liquid_container {
377        let source_water = FluidState::source(&vanilla_fluids::WATER);
378        behavior.place_liquid(world, pos, state, source_water);
379        play_empty_sound_and_event(world, user, pos, true, empty_sound);
380        return true;
381    }
382
383    // 2. Try Standard Placement (Replaceable block)
384    if can_replace {
385        // If same fluid already exists and is source, just consume bucket (parity)
386        let is_same_fluid = if is_water_bucket {
387            fluid_state.is_water()
388        } else {
389            fluid_state.is_lava()
390        };
391
392        if is_same_fluid && fluid_state.is_source() {
393            play_empty_sound_and_event(world, user, pos, is_water_bucket, empty_sound);
394            return true;
395        }
396
397        // Vanilla parity: destroy non-liquid replaceable blocks first so they
398        // drop their items (e.g. tall grass, flowers, snow layers).
399        if !state.get_block().config.liquid && !state.get_block().config.is_air {
400            world.destroy_block(pos, true);
401        }
402
403        // Place fluid block
404        let fluid_state_to_place = fluid_block.default_state();
405        if world.set_block(pos, fluid_state_to_place, UpdateFlags::UPDATE_ALL_IMMEDIATE) {
406            let fluid_ref = if is_water_bucket {
407                &vanilla_fluids::WATER
408            } else {
409                &vanilla_fluids::LAVA
410            };
411            let tick_delay = FLUID_BEHAVIORS.get_behavior(fluid_ref).tick_delay(world);
412            world.schedule_fluid_tick_default(pos, fluid_ref, tick_delay);
413
414            play_empty_sound_and_event(world, user, pos, is_water_bucket, empty_sound);
415
416            return true;
417        }
418    }
419    false
420}

422pub(super) fn play_empty_sound_and_event(
423    world: &Arc<World>,
424    user: Option<&Player>,
425    pos: BlockPos,
426    is_water_bucket: bool,
427    empty_sound: EmptySound,
428) {
429    match empty_sound {
430        EmptySound::Fluid => {
431            let sound_event = if is_water_bucket {
432                &sound_events::ITEM_BUCKET_EMPTY
433            } else {
434                &sound_events::ITEM_BUCKET_EMPTY_LAVA
435            };
436            world.play_block_sound(sound_event, pos, 1.0, 1.0, None);
437            world.game_event(
438                &vanilla_game_events::FLUID_PLACE,
439                pos,
440                &GameEventContext::new(user.map(|player| player as &dyn Entity), None),
441            );
442        }
443        // Vanilla's `MobBucketItem.playEmptySound` replaces the whole method:
444        // a neutral-category sound, and no fluid-place game event.
445        EmptySound::Mob(sound) => {
446            world.play_sound(sound, SoundSource::Neutral, pos, 1.0, 1.0, None);
447        }
448    }
449}

451fn filled_bucket_primary_pos(
452    clicked_state: BlockStateId,
453    clicked_pos: BlockPos,
454    direction: Direction,
455    is_water_bucket: bool,
456) -> BlockPos {
457    if is_water_bucket && clicked_state.is_liquid_container() {
458        clicked_pos
459    } else {
460        direction.relative(clicked_pos)
461    }
