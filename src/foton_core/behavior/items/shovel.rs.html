21const FLATTENABLES: [&Block; 6] = [
22    &vanilla_blocks::GRASS_BLOCK,
23    &vanilla_blocks::DIRT,
24    &vanilla_blocks::PODZOL,
25    &vanilla_blocks::COARSE_DIRT,
26    &vanilla_blocks::MYCELIUM,
27    &vanilla_blocks::ROOTED_DIRT,
28];

30const LIT_PROPERTY: BoolProperty = BlockStateProperties::LIT;
31
32/// Behavior for Shovels, extinguishes campfires and turns grass blocks into paths
33#[item_behavior]
34pub struct ShovelItem;
35
36impl ItemBehavior for ShovelItem {
37    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
38        if context.hit_result.direction == Direction::Down {
39            return InteractionResult::Pass;
40        }
41
42        let block_state = context.world.get_block_state(context.hit_result.block_pos);
43        let block = block_state.get_block();
44
45        let pos = context.hit_result.block_pos;
46
47        // Vanilla builds an `updatedState` in one branch or the other and then
48        // shares the tail: the block change, the game event and the durability.
49        // Keeping that shape is what stops the campfire branch quietly skipping
50        // the wear the path branch pays.
51        let updated_state = if FLATTENABLES.contains(&block) {
52            if !context.world.get_block_state(pos.above()).is_air() {
53                return InteractionResult::Pass;
54            }
55            // Vanilla `level.playSound(player, ..)` excludes the acting player,
56            // who already played the sound locally.
57            context.world.play_block_sound(
58                &sound_events::ITEM_SHOVEL_FLATTEN,
59                pos,
60                1.0,
61                1.0,
62                Some(context.player.id()),
63            );
64            vanilla_blocks::DIRT_PATH.default_state()
65        } else if block.has_tag(&BlockTag::CAMPFIRES) {
66            if !block_state.get_value(&LIT_PROPERTY) {
67                return InteractionResult::Pass;
68            }
69            context
70                .world
71                .level_event(level_events::SOUND_EXTINGUISH_FIRE, pos, 0, None);
72            // TODO: CampfireBlock::dowse() — eject cooking items
73            block_state.set_value(&LIT_PROPERTY, false)
74        } else {
75            return InteractionResult::Pass;
76        };
77
78        context
79            .world
80            .set_block(pos, updated_state, UpdateFlags::UPDATE_ALL_IMMEDIATE);
81        context.world.game_event(
82            &vanilla_game_events::BLOCK_CHANGE,
83            pos,
84            &GameEventContext::new(Some(context.player), Some(updated_state)),
85        );
86        context.player.hurt_item_in_hand(context.hand, 1);
87
88        InteractionResult::Success
89    }
90}

109    const TARGET: BlockPos = BlockPos::new(8, 64, 8);
110
111    fn world_with(key: &'static str, state: foton_utils::BlockStateId) -> Arc<World> {
112        let world = fresh_test_world(key);
113        insert_ready_full_chunk(&world, ChunkPos::from_block_pos(TARGET));
114        world.set_block(TARGET, state, UpdateFlags::UPDATE_NONE);
115        world
116    }
117
118    fn player_with_a_shovel(world: &Arc<World>) -> Arc<Player> {
119        let player = TestPlayerBuilder::new(Arc::clone(world), "Digger", 1).build();
120        player
121            .inventory
122            .lock()
123            .set_selected_item(ItemStack::new(&vanilla_items::IRON_SHOVEL));
124        player
125    }
126
127    fn dig(world: &Arc<World>, player: &Player) -> InteractionResult {
128        let mut context = UseOnContext::new(
129            player,
130            InteractionHand::MainHand,
131            BlockHitResult {
132                location: DVec3::new(8.5, 65.0, 8.5),
133                direction: Direction::Up,
134                block_pos: TARGET,
135                miss: false,
136                inside: false,
137                world_border_hit: false,
138            },
139            world,
140            player.inventory.clone(),
141        );
142        ShovelItem.use_on(&mut context)
143    }
144
145    fn shovel_damage(player: &Player) -> i32 {
146        player
147            .inventory
148            .lock()
149            .get_selected_item()
150            .get_damage_value()
151    }
152
153    #[test]
154    fn making_a_path_wears_the_shovel() {
155        init_globals_once();
156        let world = world_with(
157            "shovel_flatten_costs_durability",
158            vanilla_blocks::GRASS_BLOCK.default_state(),
159        );
160        let player = player_with_a_shovel(&world);
161
162        assert_eq!(dig(&world, &player), InteractionResult::Success);
163        assert_eq!(
164            world.get_block_state(TARGET).get_block(),
165            &vanilla_blocks::DIRT_PATH
166        );
167        assert_eq!(shovel_damage(&player), 1);
168    }
169
170    /// Putting a campfire out costs the same point the path does. Vanilla pays
171    /// it in the tail both branches share; Foton's campfire branch used to
172    /// return before reaching it, so a shovel could dowse every campfire in a
173    /// village for free.
174    #[test]
175    fn putting_out_a_campfire_wears_the_shovel_too() {
176        init_globals_once();
177        let world = world_with(
178            "shovel_dowse_costs_durability",
179            vanilla_blocks::CAMPFIRE.default_state(),
180        );
181        let player = player_with_a_shovel(&world);
182        assert!(
183            world.get_block_state(TARGET).get_value(&LIT_PROPERTY),
184            "a fresh campfire is lit"
185        );
186
187        assert_eq!(dig(&world, &player), InteractionResult::Success);
188        assert!(!world.get_block_state(TARGET).get_value(&LIT_PROPERTY));
189        assert_eq!(shovel_damage(&player), 1);
190    }
191
192    /// An unlit campfire and a block with no shovel use both leave the tool
193    /// alone -- the shared tail must not run for a click that did nothing.
194    #[test]
195    fn a_click_that_changes_nothing_is_free() {
196        init_globals_once();
197        let world = world_with(
198            "shovel_unlit_campfire_is_free",
199            vanilla_blocks::CAMPFIRE
200                .default_state()
201                .set_value(&LIT_PROPERTY, false),
202        );
203        let player = player_with_a_shovel(&world);
204
205        assert_eq!(dig(&world, &player), InteractionResult::Pass);
206        assert_eq!(shovel_damage(&player), 0);
207
208        world.set_block(
209            TARGET,
210            vanilla_blocks::STONE.default_state(),
211            UpdateFlags::UPDATE_NONE,
212        );
213        assert_eq!(dig(&world, &player), InteractionResult::Pass);
214        assert_eq!(shovel_damage(&player), 0);
215    }
