38/// The filled map.
39#[item_behavior]
40pub struct MapItem;
41
42impl ItemBehavior for MapItem {
43    /// Vanilla parity: `MapItem.inventoryTick`.
44    ///
45    /// Deviation: this also runs `onCraftedPostProcess`. Vanilla runs it from
46    /// `Slot.onTake`, which hands the taken stack to the item; Foton's
47    /// `Slot::on_take` only sees the stack by reference, so a map that leaves a
48    /// cartography table still carrying its `minecraft:map_post_processing`
49    /// marker resolves it on the first tick after it lands in a real slot. The
50    /// marker is what the tooltip reads while the map sits in the result slot,
51    /// so resolving it any earlier would be wrong.
52    fn inventory_tick(
53        &self,
54        stack: &mut ItemStack,
55        world: &Arc<World>,
56        owner: &dyn LivingEntity,
57        slot: Option<EquipmentSlot>,
58    ) {
59        let Some(player) = owner.as_player() else {
60            return;
61        };
62        apply_post_processing(stack, world, player);
63        let Some(map_id) = stack.get(MAP_ID).copied() else {
64            return;
65        };
66        let Some(data) = saved_data(world, player, map_id) else {
67            return;
68        };
69
70        let decorations = stack
71            .get(MAP_DECORATIONS)
72            .cloned()
73            .unwrap_or(MapDecorations::EMPTY);
74        tick_carried_by(world, player, map_id, &data, &decorations, None);
75
76        let locked = data.lock().locked;
77        if !locked && slot.is_some_and(|slot| slot.slot_type() == EquipmentSlotType::Hand) {
78            update(world, player, &data);
79        }
80    }
81
82    /// Vanilla parity: `MapItem.useOn`, which marks the banner it clicked.
83    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
84        let pos = context.hit_result.block_pos;
85        if !context
86            .world
87            .get_block_state(pos)
88            .get_block()
89            .has_tag(&BlockTag::BANNERS)
90        {
91            return InteractionResult::Pass;
92        }
93
94        let map_id = context.inv.with_item(|stack| stack.get(MAP_ID).copied());
95        let Some(map_id) = map_id else {
96            return InteractionResult::Success;
97        };
98        let Some(data) = saved_data(context.world, context.player, map_id) else {
99            return InteractionResult::Success;
100        };
101        if !data.lock().toggle_banner(context.world, pos) {
102            return InteractionResult::Fail;
103        }
104        InteractionResult::Success
105    }
106}

108/// The blank map.
109#[item_behavior]
110pub struct EmptyMapItem;
111
112impl ItemBehavior for EmptyMapItem {
113    /// Vanilla parity: `EmptyMapItem.use`.
114    ///
115    /// Deviation: vanilla hands the new map straight back through
116    /// `InteractionResult.heldItemTransformedTo` when the last blank map was
117    /// spent, and otherwise adds it to the inventory. Foton has no
118    /// "transformed to" result, so both cases go through
119    /// `ItemUtils.createFilledResult`, which puts the map in the emptied hand
120    /// when it can and in the inventory or on the ground when it cannot.
121    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
122        let Some(storage) = map_storage(context.world, context.player) else {
123            return InteractionResult::Fail;
124        };
125
126        context.world.play_sound_at(
127            &sound_events::UI_CARTOGRAPHY_TABLE_TAKE_RESULT,
128            SoundSource::Players,
129            context.player.position(),
130            1.0,
131            1.0,
132            None,
133        );
134
135        let position = context.player.position();
136        let map = create_map(
137            context.world,
138            &storage,
139            position.x.floor() as i32,
140            position.z.floor() as i32,
141            0,
142            true,
143            false,
144        );
145        create_filled_result(context, map, true);
146        InteractionResult::Success
147    }
148}

150/// Vanilla parity: `MapItem.create`.
151pub fn create_map(
152    world: &Arc<World>,
153    storage: &Arc<MapStorage>,
154    origin_x: i32,
155    origin_z: i32,
156    scale: u8,
157    tracking_position: bool,
158    unlimited_tracking: bool,
159) -> ItemStack {
160    let mut map = ItemStack::new(&vanilla_items::FILLED_MAP);
161    let data = MapItemSavedData::create_fresh(
162        f64::from(origin_x),
163        f64::from(origin_z),
164        scale,
165        tracking_position,
166        unlimited_tracking,
167        world.key.clone(),
168        world.is_nether(),
169    );
170    let id = storage.next_id();
171    storage.set(id, data);
172    map.set(MAP_ID, id);
173    map
174}

176/// Vanilla parity: `MapItem.onCraftedPostProcess`, which is what turns the
177/// cartography table's `MAP_POST_PROCESSING` marker into a real new map.
178pub fn apply_post_processing(stack: &mut ItemStack, world: &Arc<World>, player: &Player) {
179    let Some(post_processing) = stack.get(MAP_POST_PROCESSING).copied() else {
180        return;
181    };
182    stack.remove(MAP_POST_PROCESSING);
183
184    let Some(storage) = map_storage(world, player) else {
185        return;
186    };
187    let Some(map_id) = stack.get(MAP_ID).copied() else {
188        return;
189    };
190    let Some(original) = storage.get(map_id) else {
191        return;
192    };
193
194    let replacement = match post_processing {
195        MapPostProcessing::Lock => original.lock().locked_copy(),
196        MapPostProcessing::Scale => original.lock().scaled(),
197    };
198    let id = storage.next_id();
199    storage.set(id, replacement);
200    stack.set(MAP_ID, id);
201}

213/// Resolves the other players a map is tracking, anywhere on the server.
214///
215/// Vanilla holds live `Player` references in `carriedBy`, so a holder who walks
216/// into another dimension is still resolvable. Foton looks them up by UUID
217/// across every online player for the same reason: scoping the lookup to one
218/// world would drop and re-add such a holder every tick, and each re-add would
219/// resend the whole 128x128 image.
220struct OnlineMapHolders<'a> {
221    player: &'a Player,
222    map_id: MapId,
223}
224
225impl MapPlayerSource for OnlineMapHolders<'_> {
226    fn holder(&self, uuid: Uuid) -> Option<MapPlayerState> {
227        let holder = self.player.server().get_player_by_uuid(uuid)?;
228        Some(player_state(
229            &holder,
230            self.map_id,
231            holds_map(&holder, self.map_id),
232        ))
233    }
234}

236/// Vanilla parity: `MapItemSavedData.mapMatcher` applied to the whole inventory.
237fn holds_map(player: &Player, map_id: MapId) -> bool {
238    let inventory = player.inventory.lock();
239    inventory
240        .get_items()
241        .iter()
242        .any(|stack| stack.is(&vanilla_items::FILLED_MAP) && stack.get(MAP_ID) == Some(&map_id))
243}
244
245fn player_state(player: &Player, _map_id: MapId, holds_map: bool) -> MapPlayerState {
246    let position = player.position();
247    let (yaw, _) = player.rotation();
248    MapPlayerState {
249        uuid: player.gameprofile.id,
250        name: player.gameprofile.name.clone(),
251        x: position.x,
252        z: position.z,
253        y_rot: yaw,
254        dimension: player.get_world().key.clone(),
255        holds_map,
256        map_invisible: map_invisibility_equipped(player),
257    }
258}

260/// Vanilla parity: `MapItemSavedData.hasMapInvisibilityItemEquipped`.
261///
262/// The `minecraft:map_invisibility_equipment` item tag is empty in vanilla data
263/// for this version, so nothing can currently satisfy it; the check is kept so
264/// the behavior appears the moment the tag gains a member.
265fn map_invisibility_equipped(player: &Player) -> bool {
266    use crate::inventory::equipment::EntityEquipment as _;
267    use foton_registry::vanilla_item_tags::ItemTag;
268
269    let inventory = player.inventory.lock();
270    EquipmentSlot::ALL
271        .into_iter()
272        .filter(|slot| !matches!(slot, EquipmentSlot::MainHand | EquipmentSlot::OffHand))
273        .any(|slot| {
274            inventory
275                .get_ref(slot)
276                .item()
277                .has_tag(&ItemTag::MAP_INVISIBILITY_EQUIPMENT)
278        })
279}

281/// Runs one holder's `tickCarriedBy` against the shared map.
282pub fn tick_carried_by(
283    world: &Arc<World>,
284    player: &Player,
285    map_id: MapId,
286    data: &SharedMapData,
287    decorations: &MapDecorations,
288    frame: Option<MapFrame>,
289) {
290    // The ticking holder always matches: vanilla's `mapMatcher` short-circuits
291    // on stack identity, and Foton has lifted the stack out of the slot for the
292    // duration of the tick, so an inventory scan would answer wrongly.
293    let state = player_state(player, map_id, true);
294    let source = OnlineMapHolders { player, map_id };
295    data.lock()
296        .tick_carried_by(&state, &source, frame, decorations, world.game_time());
297}

299/// Vanilla parity: `MapItem.update`, the pass that reads the world under the
300/// map and writes one stripe of pixels.
301///
302/// Deviation: vanilla calls `level.getChunk`, which generates the chunk if it
303/// has to. Foton skips columns whose chunk is not loaded rather than driving
304/// worldgen from an item tick; those pixels keep their previous color and are
305/// filled in once the chunk is there.
306#[expect(
307    clippy::too_many_lines,
308    reason = "one vanilla method whose loops only make sense together"
309)]
310pub fn update(world: &Arc<World>, player: &Player, data: &SharedMapData) {
311    let mut map = data.lock();
312    if map.dimension != world.key {
313        return;
314    }
315
316    let scale = 1i32 << map.scale;
317    let center_x = map.center_x;
318    let center_z = map.center_z;
319    let position = player.position();
320    let player_img_x = (position.x.floor() as i32 - center_x) / scale + 64;
321    let player_img_y = (position.z.floor() as i32 - center_z) / scale + 64;
322    let mut radius = IMAGE_SIZE / scale;
323    let has_ceiling = world.dimension_type.has_ceiling;
324    if has_ceiling {
325        radius /= 2;
326    }
327
328    let step = {
329        let holder = map.holding_player_mut(player.gameprofile.id, &player.gameprofile.name);
330        holder.step = holder.step.wrapping_add(1);
331        holder.step
332    };
333
334    let min_y = world.get_min_y();
335    let mut found_consecutive_changes = false;
336
337    for img_x in (player_img_x - radius + 1)..(player_img_x + radius) {
338        if (img_x & 15) != (step & 15) && !found_consecutive_changes {
339            continue;
340        }
341        found_consecutive_changes = false;
342        let mut previous_average_area_height = 0.0f64;
343
344        for img_y in (player_img_y - radius - 1)..(player_img_y + radius) {
345            if img_x < 0 || img_y < -1 || img_x >= IMAGE_SIZE || img_y >= IMAGE_SIZE {
346                continue;
347            }
348
349            let distance_sqr = (img_x - player_img_x).pow(2) + (img_y - player_img_y).pow(2);
350            let dither_black = distance_sqr > (radius - 2) * (radius - 2);
351            let area_min_x = (center_x / scale + img_x - 64) * scale;
352            let area_min_z = (center_z / scale + img_y - 64) * scale;
353            let chunk = ChunkPos::new(
354                SectionPos::block_to_section_coord(area_min_x),
355                SectionPos::block_to_section_coord(area_min_z),
356            );
357            if world.chunk_map.with_full_chunk(chunk, |_| ()).is_none() {
358                continue;
359            }
360
361            let mut counts: Vec<(MapColor, u32)> = Vec::new();
362            let mut water_depth = 0i32;
363            let mut average_area_height = 0.0f64;
364
365            if has_ceiling {
366                // Vanilla's deterministic Nether fog pattern, with Java's
367                // wrapping integer arithmetic.
368                let mut noise = area_min_x.wrapping_add(area_min_z.wrapping_mul(231_871));
369                noise = noise
370                    .wrapping_mul(noise)
371                    .wrapping_mul(31_287_121)
372                    .wrapping_add(noise.wrapping_mul(11));
373                if (noise >> 20) & 1 == 0 {
374                    add_color(&mut counts, default_map_color(&vanilla_blocks::DIRT), 10);
375                } else {
376                    add_color(&mut counts, default_map_color(&vanilla_blocks::STONE), 100);
377                }
378                average_area_height = 100.0;
379            } else {
380                for delta_x in 0..scale {
381                    for delta_z in 0..scale {
382                        let x = area_min_x + delta_x;
383                        let z = area_min_z + delta_z;
384                        let Some(first_available) =
385                            world.height_at(HeightmapType::WorldSurface, x, z)
386                        else {
387                            continue;
388                        };
389
390                        let mut column_y = first_available;
391                        let state = if column_y <= min_y {
392                            REGISTRY
393                                .blocks
394                                .get_default_state_id(&vanilla_blocks::BEDROCK)
395                        } else {
396                            let mut pos;
397                            let mut state;
398                            loop {
399                                column_y -= 1;
400                                pos = BlockPos::new(x, column_y, z);
401                                state = world.get_block_state(pos);
402                                if state.get_map_color() != MapColor::NONE || column_y <= min_y {
403                                    break;
404                                }
405                            }
406
407                            if column_y > min_y && !state.get_fluid_state().is_empty() {
408                                let mut solid_y = column_y - 1;
409                                loop {
410                                    let below = world.get_block_state(BlockPos::new(x, solid_y, z));
411                                    solid_y -= 1;
412                                    water_depth += 1;
413                                    if solid_y <= min_y || below.get_fluid_state().is_empty() {
414                                        break;
415                                    }
416                                }
417                                state = correct_state_for_fluid_block(state, pos);
418                            }
419                            state
420                        };
421
422                        map.check_banners(world, x, z);
423                        average_area_height += f64::from(column_y) / f64::from(scale * scale);
424                        add_color(&mut counts, state.get_map_color(), 1);
425                    }
426                }
427            }
428
429            water_depth /= scale * scale;
430            let color = highest_count_first(&counts);
431            let checker = f64::from((img_x + img_y) & 1);
432            let brightness = if color == MapColor::WATER {
433                let diff = f64::from(water_depth).mul_add(0.1, checker * 0.2);
434                if diff < 0.5 {
435                    Brightness::High
436                } else if diff > 0.9 {
437                    Brightness::Low
438                } else {
439                    Brightness::Normal
440                }
441            } else {
442                let diff = (average_area_height - previous_average_area_height) * 4.0
443                    / f64::from(scale + 4)
444                    + (checker - 0.5) * 0.4;
445                if diff > 0.6 {
446                    Brightness::High
447                } else if diff < -0.6 {
448                    Brightness::Low
449                } else {
450                    Brightness::Normal
451                }
452            };
453
454            previous_average_area_height = average_area_height;
455            if img_y >= 0
456                && distance_sqr < radius * radius
457                && (!dither_black || (img_x + img_y) & 1 != 0)
458            {
459                found_consecutive_changes |=
460                    map.update_color(img_x as usize, img_y as usize, color.packed_id(brightness));
461            }
462        }
463    }
464}

466/// Vanilla parity: `MapItem.getCorrectStateForFluidBlock`.
467fn correct_state_for_fluid_block(state: BlockStateId, pos: BlockPos) -> BlockStateId {
468    let fluid_state = state.get_fluid_state();
469    if fluid_state.is_empty() || state.is_face_sturdy_at(pos, Direction::Up) {
470        return state;
471    }
472    fluid_state_to_block(fluid_state)
473}

479/// Vanilla parity: `LinkedHashMultiset.add`, which keeps first-seen order.
480fn add_color(counts: &mut Vec<(MapColor, u32)>, color: MapColor, amount: u32) {
481    if let Some(entry) = counts.iter_mut().find(|(seen, _)| *seen == color) {
482        entry.1 += amount;
483        return;
484    }
485    counts.push((color, amount));
486}

488/// Vanilla parity: `Iterables.getFirst(Multisets.copyHighestCountFirst(..), NONE)`.
489///
490/// Guava's sort is stable, so a tie is broken by which color was counted first.
491fn highest_count_first(counts: &[(MapColor, u32)]) -> MapColor {
492    let mut best = MapColor::NONE;
493    let mut best_count = 0;
494    for (color, count) in counts {
495        if *count > best_count {
496            best = *color;
497            best_count = *count;
498        }
499    }
500    best
