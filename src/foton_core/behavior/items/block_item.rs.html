25/// Behavior for items that place blocks.
26#[item_behavior]
27pub struct BlockItem {
28    /// The block this item places.
29    #[json_arg(vanilla_blocks, json = "block")]
30    pub block: BlockRef,
31    /// Vanilla parity: `BlockItem.getPlaceSound`, which only `SolidBucketItem`
32    /// overrides. `None` keeps the placed block's own sound type.
33    place_sound: Option<SoundEventRef>,
34    /// Vanilla parity: `BlockItem.mustSurvive`, which only scaffolding turns off.
35    must_survive: bool,
36}
37
38impl BlockItem {
39    const PLACE_BLOCK_FLAGS: UpdateFlags = UpdateFlags::UPDATE_ALL_IMMEDIATE;
40
41    /// Creates a new block item behavior for the given block.
42    #[must_use]
43    pub const fn new(block: BlockRef) -> Self {
44        Self {
45            block,
46            place_sound: None,
47            must_survive: true,
48        }
49    }
50
51    /// Returns this behavior with `BlockItem.getPlaceSound` overridden.
52    #[must_use]
53    pub const fn with_place_sound(mut self, place_sound: SoundEventRef) -> Self {
54        self.place_sound = Some(place_sound);
55        self
56    }
57
58    /// Returns this behavior with `BlockItem.mustSurvive` turned off.
59    #[must_use]
60    pub const fn without_must_survive(mut self) -> Self {
61        self.must_survive = false;
62        self
63    }
64
65    /// Runs vanilla's `BlockItem.place` with the two steps subclasses replace.
66    ///
67    /// `placement_state` stands in for `getPlacementState`, which the
68    /// standing-and-wall item overrides to choose between two blocks;
69    /// `place_block` stands in for `placeBlock`, which the double-high item
70    /// overrides to clear the space above.
71    pub(super) fn place_with(
72        &self,
73        mut context: BlockPlaceContext<'_>,
74        placement_state: impl FnOnce(&BlockPlaceContext<'_>) -> Option<BlockStateId>,
75        place_block: impl FnOnce(&BlockPlaceContext<'_>, BlockStateId) -> bool,
76    ) -> InteractionResult {
77        if !context.can_place() {
78            return InteractionResult::Fail;
79        }
80        let place_pos = context.place_pos();
81
82        let Some(new_state) = placement_state(&context) else {
83            return InteractionResult::Fail;
84        };
85
86        let behavior = BLOCK_BEHAVIORS.get_behavior(new_state.get_block());
87        if self.must_survive && !behavior.can_survive(new_state, context.world, place_pos) {
88            return InteractionResult::Fail;
89        }
90
91        let collision_shape = new_state.get_collision_shape_at(place_pos);
92        if !context.world.is_unobstructed(collision_shape, place_pos) {
93            return InteractionResult::Fail;
94        }
95
96        // Taken before anything is written, so a listener that vetoes the
97        // placement finds the world as the player left it.
98        let before = context
99            .player()
100            .map(|_| PlacementSnapshot::capture(context.world, place_pos));
101
102        if !place_block(&context, new_state) {
103            return InteractionResult::Fail;
104        }
105
106        let mut placed_state = context.world.get_block_state(place_pos);
107        let landed = placed_state.get_block() == new_state.get_block();
108        if landed {
109            placed_state = Self::update_block_state_from_tag(&context, place_pos, placed_state);
110            Self::update_block_entity_components(&context, place_pos);
111            let placed_behavior = BLOCK_BEHAVIORS.get_behavior(placed_state.get_block());
112            placed_behavior.set_placed_by(placed_state, context.world, place_pos, context.source());
113        }
114
115        // Paper parity: `CraftEventFactory.callBlockPlaceEvent` runs once the
116        // block is in the world, so a listener reads what landed and what it
117        // replaced; a veto restores the blocks. A dispenser firing a block has
118        // no player and does not reach this.
119        if let Some(player) = context.player()
120            && let Some(shared) = player.shared()
121            && let Some(before) = before
122        {
123            let item = context.with_item(Clone::clone);
124            let mut event = BlockPlaceEvent::new(
125                shared,
126                place_pos,
127                placed_state,
128                before.state_at(place_pos).unwrap_or(placed_state),
129                context.hit_pos(),
130                item,
131            );
132            player.fire_event(&mut event);
133            if event.is_cancelled() {
134                before.restore(context.world);
135                return InteractionResult::Fail;
136            }
137        }
138
139        if landed {
140            // Vanilla parity: the `CriteriaTriggers.PLACED_BLOCK` of
141            // `BlockItem.place`, which sits inside this same
142            // "the block that landed is the one we asked for" branch and runs
143            // before the stack is shrunk below.
144            if let Some(player) = context.player() {
145                let tool = context.with_item(Clone::clone);
146                triggers::world::placed_block(player, place_pos, placed_state, &tool);
147            }
148        }
149
150        // Play place sound (exclude the placing player, they hear it
151        // client-side). Vanilla reads the sound off the state that ended up in
152        // the world, which is how a wall banner sounds like the wall form.
153        let sound_type = &placed_state.get_block().config.sound_type;
154        context.world.play_block_sound(
155            self.place_sound.unwrap_or(sound_type.place_sound),
156            place_pos,
157            sound_type.volume,
158            sound_type.pitch,
159            context.player().map(Entity::id),
160        );
161        context.world.game_event(
162            &vanilla_game_events::BLOCK_PLACE,
163            place_pos,
164            &GameEventContext::new(
165                context.player().map(|player| player as &dyn Entity),
166                Some(placed_state),
167            ),
168        );
169
170        context.with_item_mut(|item| item.shrink(1));
171
172        InteractionResult::Success
173    }
174
175    /// Places this block using an already constructed placement context.
176    pub fn place(&self, context: BlockPlaceContext<'_>) -> InteractionResult {
177        self.place_with(context, |c| self.placement_state(c), Self::place_block)
178    }
179
180    /// Vanilla parity: `BlockItem.getPlacementState`.
181    pub(super) fn placement_state(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
182        BLOCK_BEHAVIORS
183            .get_behavior(self.block)
184            .get_state_for_placement(context)
185    }
186
187    pub(super) fn place_block(context: &BlockPlaceContext<'_>, state: BlockStateId) -> bool {
188        context
189            .world
190            .set_block(context.place_pos(), state, Self::PLACE_BLOCK_FLAGS)
191    }
192
193    /// Re-applies the block properties the item was carrying.
194    ///
195    /// Vanilla parity: `BlockItem.updateBlockStateFromTag`, the other half of
196    /// the `minecraft:copy_state` loot function. Without it a picked-up hive
197    /// carries its `honey_level` and then forgets it the moment it goes back
198    /// down.
199    fn update_block_state_from_tag(
200        context: &BlockPlaceContext<'_>,
201        pos: BlockPos,
202        placed_state: BlockStateId,
203    ) -> BlockStateId {
204        let properties = context.with_item(|item| item.get(BLOCK_STATE).cloned());
205        let Some(properties) = properties.filter(|properties| !properties.is_empty()) else {
206            return placed_state;
207        };
208
209        let modified_state = properties.apply(placed_state);
210        if modified_state != placed_state {
211            context
212                .world
213                .set_block(pos, modified_state, UpdateFlags::UPDATE_CLIENTS);
214        }
215        modified_state
216    }
217
218    /// Hands the placed block entity the components the item carried.
219    ///
220    /// Vanilla parity: `BlockItem.updateBlockEntityComponents`.
221    fn update_block_entity_components(context: &BlockPlaceContext<'_>, pos: BlockPos) {
222        let Some(block_entity) = context.world.get_block_entity(pos) else {
223            return;
224        };
225        // The stack is copied out first: `with_item` holds the placing player's
226        // inventory lock for the whole closure, and a block entity taking its
227        // components has no business running under it.
228        let stack = context.with_item(Clone::clone);
229        block_entity.apply_components_from_item_stack(&stack);
230        block_entity.set_changed();
231    }
232}

234/// The blocks a placement can touch, as they stood before it.
235///
236/// Paper reverts a vetoed placement from the block states it captured while
237/// placing. Foton keeps the cube around the target instead: a double-high or
238/// bed half lands in the next cell, and shape updates reach the six
239/// neighbours, which is the whole reach of one placement.
240struct PlacementSnapshot {
241    cells: Vec<(BlockPos, BlockStateId)>,
242}
243
244impl PlacementSnapshot {
245    fn capture(world: &Arc<World>, center: BlockPos) -> Self {
246        let mut cells = Vec::with_capacity(27);
247        for dx in -1..=1 {
248            for dy in -1..=1 {
249                for dz in -1..=1 {
250                    let pos = BlockPos::new(center.x() + dx, center.y() + dy, center.z() + dz);
251                    cells.push((pos, world.get_block_state(pos)));
252                }
253            }
254        }
255        Self { cells }
256    }
257
258    fn state_at(&self, pos: BlockPos) -> Option<BlockStateId> {
259        self.cells
260            .iter()
261            .find_map(|(cell, state)| (*cell == pos).then_some(*state))
262    }
263
264    /// Puts back every cell that changed, without waking its neighbours: the
265    /// blocks around it are being restored too.
266    fn restore(&self, world: &Arc<World>) {
267        for (pos, state) in &self.cells {
268            if world.get_block_state(*pos) != *state {
269                world.set_block(*pos, *state, UpdateFlags::UPDATE_CLIENTS);
270            }
271        }
272    }
273}

275impl ItemBehavior for BlockItem {
276    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
277        self.place(context.build_place_context())
278    }
279
280    fn placed_block(&self) -> Option<BlockRef> {
281        Some(self.block)
282    }

284    /// Vanilla parity: `BlockItem.canFitInsideContainerItems`. Foton has no
285    /// `ShulkerBoxBlock` behavior to test against, so the block tag holding
286    /// exactly those seventeen blocks stands in, as it already does for banners
287    /// in the loom.
288    fn can_fit_inside_container_items(&self) -> bool {
289        !REGISTRY
290            .blocks
291            .is_in_tag(self.block, &BlockTag::SHULKER_BOXES)
292    }
293}

295/// Behavior for double-high block items (doors, tall flowers, etc.).
296///
297/// Vanilla's `DoubleHighBlockItem` extends `BlockItem` and overrides `placeBlock`
298/// to place the upper half block above the lower half.
299///
300/// The `_block` field is read by the build script via `#[json_arg]` to generate constructor
301/// calls from `classes.json`. The actual value is forwarded into `base`.
302#[item_behavior]
303pub struct DoubleHighBlockItem {
304    #[json_arg(vanilla_blocks, json = "block")]
305    _block: BlockRef,
306    base: BlockItem,
307}
308
309impl DoubleHighBlockItem {
310    const PREPARE_UPPER_FLAGS: UpdateFlags =
311        UpdateFlags::UPDATE_ALL_IMMEDIATE.union(UpdateFlags::UPDATE_KNOWN_SHAPE);
312
313    /// Creates a new double-high block item behavior for the given block.
314    #[must_use]
315    pub const fn new(block: BlockRef) -> Self {
316        Self {
317            _block: block,
318            base: BlockItem::new(block),
319        }
320    }
321
322    fn place_block(context: &BlockPlaceContext<'_>, state: BlockStateId) -> bool {
323        let above = context.place_pos().above();
324        let above_state = if get_fluid_state(context.world, above).is_water() {
325            vanilla_blocks::WATER.default_state()
326        } else {
327            vanilla_blocks::AIR.default_state()
328        };
329        let _ = context
330            .world
331            .set_block(above, above_state, Self::PREPARE_UPPER_FLAGS);
332
333        BlockItem::place_block(context, state)
334    }
335}
336
337impl ItemBehavior for DoubleHighBlockItem {
338    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
339        self.base.place_with(
340            context.build_place_context(),
341            |c| self.base.placement_state(c),
342            Self::place_block,
343        )
344    }
345}

347#[cfg(test)]
348mod tests {
349    use foton_registry::init_vanilla_registry;
350    use foton_utils::ChunkPos;

352    use super::*;
353    use crate::behavior::init_behaviors;
354    use crate::test_support::{fresh_test_world, insert_ready_full_chunk};
355
356    /// A vetoed door has two halves in two cells, and the replaced grass under
357    /// the lower one has to come back with them.
358    #[test]
359    fn a_restored_snapshot_takes_back_both_halves_and_what_they_replaced() {
360        init_vanilla_registry();
361        init_behaviors();
362        let world = fresh_test_world("block_item_snapshot");
363        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
364        let pos = BlockPos::new(8, 64, 8);
365        let grass = vanilla_blocks::SHORT_GRASS.default_state();
366        assert!(world.set_block(pos, grass, UpdateFlags::UPDATE_NONE));
367
368        let before = PlacementSnapshot::capture(&world, pos);
369        assert_eq!(before.state_at(pos), Some(grass));
370        let door = vanilla_blocks::OAK_DOOR.default_state();
371        assert!(world.set_block(pos, door, UpdateFlags::UPDATE_NONE));
372        assert!(world.set_block(pos.above(), door, UpdateFlags::UPDATE_NONE));
373
374        before.restore(&world);
375
376        assert_eq!(world.get_block_state(pos), grass);
377        assert!(world.get_block_state(pos.above()).is_air());
378    }
