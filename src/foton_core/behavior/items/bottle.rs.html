1//! Vanilla `BottleItem` -- the empty glass bottle.
2//!
3//! Two of vanilla's three ways to fill a bottle are unreachable in Foton today:
4//!
5//! - The dragon-breath branch of `BottleItem.use` collects area effect clouds
6//!   whose `getOwner()` is an `EnderDragon`. Foton has an area effect cloud
7//!   entity but no ender dragon (see `dev/parity-gaps.txt`) and no owner on the
8//!   cloud, so the filter can never match and the branch is omitted rather than
9//!   approximated.
10//! - Filling from a water cauldron is not this class's job in vanilla either: it
11//!   lives in the `CauldronInteraction.WATER` map, which Foton has not ported
12//!   (`blocks/building/cauldron_block.rs` says as much). Nothing here can stand
13//!   in for it.
14//!
15//! What is implemented is the branch that is reachable: clipping against a water
16//! source and turning the bottle into a water bottle.
17

36/// Behavior for the empty glass bottle.
37#[item_behavior]
38pub struct BottleItem;
39
40impl ItemBehavior for BottleItem {
41    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
42        let Some(pos) = player_pov_hit_source_fluid(context) else {
43            return InteractionResult::Pass;
44        };
45
46        if !context.world.may_interact(context.player, pos) {
47            return InteractionResult::Pass;
48        }
49
50        if !context
51            .world
52            .get_block_state(pos)
53            .get_fluid_state()
54            .is_water()
55        {
56            return InteractionResult::Pass;
57        }
58
59        // Vanilla `level.playSound(player, ..)` excludes the acting player, who
60        // already played the sound locally.
61        context.world.play_sound_at(
62            &sound_events::ITEM_BOTTLE_FILL,
63            SoundSource::Neutral,
64            context.player.position(),
65            1.0,
66            1.0,
67            Some(context.player.id()),
68        );
69        context.world.game_event(
70            &vanilla_game_events::FLUID_PICKUP,
71            pos,
72            &GameEventContext::new(Some(context.player), None),
73        );
74
75        // Vanilla `BottleItem.turnBottleIntoItem`, which is
76        // `ItemUtils.createFilledResult` plus a statistic.
77        let used = context.inv.with_item(|item| item.item);
78        context
79            .player
80            .award_stat(Stat::new(&vanilla_stat_types::USED, used));
81        create_filled_result(context, water_bottle(), true);
82        InteractionResult::Success
83    }
84}
85
86/// Builds vanilla `PotionContents.createItemStack(Items.POTION, Potions.WATER)`.
87fn water_bottle() -> ItemStack {
88    let mut stack = ItemStack::new(&vanilla_items::POTION);
89    stack.set(
90        POTION_CONTENTS,
91        PotionContents::new(
92            Some(RegistryReference::new(&vanilla_potions::WATER)),
93            None,
94            Vec::new(),
95            None,
96        ),
97    );
98    stack
99}

101#[cfg(test)]
102mod tests {
103    use std::sync::Arc;
104
105    use foton_registry::blocks::BlockRef;
106    use foton_registry::vanilla_blocks;
107    use foton_utils::types::{InteractionHand, UpdateFlags};
108    use foton_utils::{BlockPos, ChunkPos};
109    use glam::DVec3;
110
111    use super::*;
112    use crate::bootstrap::init_globals_once;
113    use crate::player::Player;
114    use crate::test_support::{TestPlayerBuilder, fresh_test_world, insert_ready_full_chunk};
115
116    const FLUID_POS: BlockPos = BlockPos::new(8, 64, 8);
117
118    /// Stands a bottle-holding player two blocks above `fluid` and points them
119    /// straight down at it, so `use_item` clips the way the client would.
120    fn use_bottle_over(key: &'static str, fluid: BlockRef) -> Arc<Player> {
121        init_globals_once();
122        let world = fresh_test_world(key);
123        insert_ready_full_chunk(&world, ChunkPos::from_block_pos(FLUID_POS));
124        world.set_block(FLUID_POS, fluid.default_state(), UpdateFlags::UPDATE_NONE);
125
126        let player = TestPlayerBuilder::new(Arc::clone(&world), "BottleDipper", 1).build();
127        player.base().set_position_local(DVec3::new(8.5, 66.0, 8.5));
128        player.set_rotation((0.0, 90.0));
129        player
130            .inventory
131            .lock()
132            .set_selected_item(ItemStack::new(&vanilla_items::GLASS_BOTTLE));
133
134        let mut context = UseItemContext::new(
135            &player,
136            InteractionHand::MainHand,
137            &world,
138            player.inventory.clone(),
139        );
140        let expected = if fluid == &vanilla_blocks::WATER {
141            InteractionResult::Success
142        } else {
143            InteractionResult::Pass
144        };
145        assert_eq!(BottleItem.use_item(&mut context), expected);
146
147        drop(context);
148        player
149    }
150
151    #[test]
152    fn dipping_a_glass_bottle_in_water_fills_it_with_a_water_potion() {
153        let player = use_bottle_over("bottle_fills_from_water", &vanilla_blocks::WATER);
154
155        let inventory = player.inventory.lock();
156        let filled = inventory.get_selected_item();
157        assert!(filled.is(&vanilla_items::POTION));
158        assert!(
159            filled
160                .get(POTION_CONTENTS)
161                .is_some_and(|contents| contents.is(&vanilla_potions::WATER)),
162            "a bottle filled from water holds the water potion, not empty contents"
163        );
164    }
165
166    #[test]
167    fn a_glass_bottle_cannot_be_filled_from_lava() {
168        let player = use_bottle_over("bottle_ignores_lava", &vanilla_blocks::LAVA);
169
170        assert!(
171            player
172                .inventory
173                .lock()
174                .get_selected_item()
175                .is(&vanilla_items::GLASS_BOTTLE)
176        );
177    }
