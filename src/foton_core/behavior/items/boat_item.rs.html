1//! Boat items.
2//!
3//! Vanilla parity: `BoatItem`. Twenty items, one class, and the reason the boat
4//! entity is reachable at all: without this a boat can only be summoned by a
5//! command, which is not something a player has.
6//!
7//! Which boat each item makes comes from the extracted `entity_type`, so there
8//! is no table of woods here.
9

25/// Behavior for every boat item.
26#[item_behavior]
27pub struct BoatItem {
28    /// The boat this item puts in the world.
29    #[json_arg(vanilla_entities, json = "entity_type")]
30    entity_type: EntityTypeRef,
31}
32
33impl BoatItem {
34    /// Creates a boat item behavior.
35    #[must_use]
36    pub const fn new(entity_type: EntityTypeRef) -> Self {
37        Self { entity_type }
38    }
39}

41impl ItemBehavior for BoatItem {
42    /// Vanilla parity: `BoatItem.use`.
43    ///
44    /// The ray stops on fluid as well as blocks, which is what lets a boat be
45    /// placed on the surface of water rather than only on the bank.
46    ///
47    /// TODO: vanilla also refuses when the player's eye is inside a pickable
48    /// entity, so a boat cannot be placed through one. Foton has no entity ray
49    /// query yet, so that check is missing and a boat may be placed while
50    /// standing in another.
51    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
52        let Some(location) = looked_at_point(context.world, context.player) else {
53            return InteractionResult::Pass;
54        };
55
56        let Some(boat) = ENTITIES.create(
57            self.entity_type,
58            next_entity_id(),
59            location,
60            Arc::downgrade(context.world),
61        ) else {
62            // The item exists for every wood, but the chest variants have no
63            // entity yet; using one places nothing rather than crashing.
64            return InteractionResult::Fail;
65        };
66
67        boat.set_rotation((context.player.rotation().0, 0.0));
68
69        // Vanilla parity: a boat that would be inside a block is not placed.
70        if has_collision(
71            &WorldCollisionProvider::new(context.world),
72            boat.bounding_box(),
73        ) {
74            return InteractionResult::Fail;
75        }
76
77        if context.world.try_add_entity(boat).is_err() {
78            return InteractionResult::Fail;
79        }
80
81        context.world.game_event(
82            &vanilla_game_events::ENTITY_PLACE,
83            BlockPos::from(location),
84            &GameEventContext::new(Some(context.player), None),
85        );
86
87        if !context.player.has_infinite_materials() {
88            context.inv.with_item(|item| item.shrink(1));
89        }
90
91        InteractionResult::Success
92    }
93}

95/// Returns where the player is looking, stopping at fluid as well as blocks.
96///
97/// Vanilla parity: `Item.getPlayerPOVHitResult` with `ClipContext.Fluid.ANY`.
98fn looked_at_point(world: &Arc<World>, player: &Player) -> Option<DVec3> {
99    let from = player.position().with_y(player.get_eye_y());
100    let (yaw, pitch) = player.rotation();
101    let to = from + player.calculate_view_vector(pitch, yaw) * player.block_interaction_range();
102
103    let hit = world.clip(from, to, ClipBlockShape::Outline, ClipFluid::Any);
104    (!hit.miss).then_some(hit.location)
105}

107#[cfg(test)]
108mod tests {
109    use foton_registry::{init_vanilla_registry, vanilla_entities};
110
111    use super::*;
112
113    /// The item knows which boat it makes.
114    ///
115    /// The mapping is extracted data rather than a table here, so this is the
116    /// check that the codegen really passes it through -- an item pointing at
117    /// the wrong wood would put the wrong boat on the water.
118    #[test]
119    fn a_boat_item_names_its_own_boat() {
120        init_vanilla_registry();
121
122        let oak = BoatItem::new(&vanilla_entities::OAK_BOAT);
123        assert_eq!(oak.entity_type.key, vanilla_entities::OAK_BOAT.key);
124
125        let raft = BoatItem::new(&vanilla_entities::BAMBOO_RAFT);
126        assert_eq!(raft.entity_type.key, vanilla_entities::BAMBOO_RAFT.key);
127    }
