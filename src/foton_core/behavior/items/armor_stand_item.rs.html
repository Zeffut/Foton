2//!
3//! Vanilla parity: `ArmorStandItem`. The stand entity has been in the tree with
4//! no way for a player to reach it -- without this item one can only be

26/// Vanilla parity: the `0.75F` of the `playSound` in `useOn`.
27const PLACE_VOLUME: f32 = 0.75;
28
29/// Vanilla parity: the `0.8F` of the same call.
30const PLACE_PITCH: f32 = 0.8;
31
32/// The angle between the eight facings a placed stand can take.
33///
34/// Vanilla parity: the `45.0F` of `useOn`.
35const FACING_STEP: f32 = 45.0;
36
37/// How far the stand is allowed to drop onto its support.
38///
39/// Vanilla parity: the `-2.0` `EntityType.getYOffset` passes to `Shapes.collide`
40/// when the spawn was moved up.
41const MAX_DROP: f64 = 2.0;
42
43/// Behavior for the armor stand item.
44#[item_behavior]
45pub struct ArmorStandItem;
46
47impl ItemBehavior for ArmorStandItem {
48    /// Vanilla parity: `ArmorStandItem.useOn`.
49    ///
50    /// A stand is nearly two blocks tall, so the whole of the placement check
51    /// is whether that column is empty: vanilla asks for no block collision and
52    /// no entity at all in the box the stand would occupy, and refuses rather
53    /// than nudging the stand somewhere it does fit.
54    ///
55    /// Note that only a click on the underside is refused outright, not every
56    /// click that is not on a top face -- a stand placed against a wall lands
57    /// in the block beside it, which is what `BlockPlaceContext` resolves.
58    fn use_on(&self, context: &mut UseOnContext) -> InteractionResult {
59        if context.hit_result.direction == Direction::Down {
60            return InteractionResult::Fail;
61        }
62
63        // Vanilla parity: the `new BlockPlaceContext(context)` of `useOn`. The
64        // stand goes where a block from the same click would have gone, so
65        // clicking tall grass replaces it rather than standing on top of it.
66        let (place_pos, player_yaw) = {
67            let placement = context.build_place_context();
68            (placement.place_pos(), placement.rotation())
69        };
70
71        let dimensions = vanilla_entities::ARMOR_STAND.dimensions;
72        let (center_x, floor_y, center_z) = place_pos.get_bottom_center();
73        let stand_box = WorldAabb::entity_box(
74            center_x,
75            floor_y,
76            center_z,
77            f64::from(dimensions.half_width()),
78            f64::from(dimensions.height),
79        );
80
81        // Vanilla parity: the `level.noCollision(null, box)` of `useOn`. Passing
82        // no source entity is what makes vanilla skip the world border here, and
83        // Foton's provider skips it for the same reason -- it only reports border
84        // shapes for a moving entity.
85        if has_collision(&WorldCollisionProvider::new(context.world), stand_box) {
86            return InteractionResult::Fail;
87        }
88
89        // Vanilla parity: the `level.getEntities(null, box).isEmpty()` of
90        // `useOn`, which goes through `EntitySelector.NO_SPECTATORS`. This is a
91        // stricter test than the collision one above: a stand refuses to share
92        // its square with a dropped item or another stand, neither of which
93        // collides with anything.
94        if context
95            .world
96            .has_entity_in_aabb_matching(&stand_box, |entity| !entity.is_spectator())
97        {
98            return InteractionResult::Fail;
99        }
100
101        let raised_box = stand_box.translate(DVec3::new(0.0, 1.0, 0.0));
102        let stand_y = floor_y + resting_y_offset(context.world, place_pos, raised_box);
103        let stand = Arc::new(ArmorStandEntity::new(
104            &vanilla_entities::ARMOR_STAND,
105            next_entity_id(),
106            DVec3::new(center_x, stand_y, center_z),
107            Arc::downgrade(context.world),
108        ));
109
110        // Vanilla parity: the `snapTo` of `useOn`, which overwrites the random
111        // yaw `EntityType.create` rolled a moment earlier -- a placed stand is
112        // never randomly turned, it faces the player.
113        stand.set_rotation((placement_yaw(player_yaw), 0.0));
114
115        // Not implemented: vanilla's `EntityType.createDefaultStackConfig` also
116        // copies the stack's components and its `ENTITY_DATA` tag onto the new
117        // stand. Foton has no `applyComponentsFromItemStack` for entities, so a
118        // stand item that carries a custom name or a command-written pose
119        // places a plain stand.
120        let position = stand.position();
121        if context.world.try_add_entity(stand).is_err() {
122            return InteractionResult::Fail;
123        }
124
125        context.world.play_sound_at(
126            &sound_events::ENTITY_ARMOR_STAND_PLACE,
127            SoundSource::Blocks,
128            position,
129            PLACE_VOLUME,
130            PLACE_PITCH,
131            None,
132        );
133        context.world.game_event_at(
134            &vanilla_game_events::ENTITY_PLACE,
135            position,
136            &GameEventContext::new(Some(context.player), None),
137        );
138
139        // Deviation: vanilla shrinks unconditionally and lets the creative
140        // inventory put the stack back. Foton has no such restore, so the
141        // creative case is skipped here, as it is for the item frame.
142        if !context.player.has_infinite_materials() {
143            context.inv.with_item(|item| item.shrink(1));
144        }
145
146        InteractionResult::Success
147    }
148}

150/// Snaps a player yaw to the eight facings a placed stand can take.
151///
152/// Vanilla parity: the
153/// `Mth.floor((Mth.wrapDegrees(context.getRotation() - 180.0F) + 22.5F) / 45.0F) * 45.0F`
154/// of `ArmorStandItem.useOn`. The half turn is what makes the stand look back
155/// at whoever placed it.
156fn placement_yaw(player_yaw: f32) -> f32 {
157    let steps = (wrap_degrees(player_yaw - 180.0) + FACING_STEP / 2.0) / FACING_STEP;
158    steps.floor() * FACING_STEP
159}
160
161/// How far above the placement block's floor the stand comes to rest.
162///
163/// Vanilla parity: `EntityType.getYOffset`, reached through the
164/// `EntityType.create(..., true, true)` of `useOn`. The stand is lifted a block
165/// and dropped back onto whatever is under the placement position, which is
166/// what leaves one placed on a slab standing on the slab rather than half a
167/// block above it. With nothing at all below, the drop runs out at `MAX_DROP`
168/// and the stand starts a block low -- vanilla does the same and lets gravity
169/// finish.
170fn resting_y_offset(world: &Arc<World>, place_pos: BlockPos, raised_box: WorldAabb) -> f64 {
171    let x = f64::from(place_pos.x());
172    let y = f64::from(place_pos.y());
173    let z = f64::from(place_pos.z());
174    // Vanilla parity: the `new AABB(spawnPos).expandTowards(0.0, -1.0, 0.0)`
175    // that bounds the search, so only the placement block and the one under it
176    // can catch the stand.
177    let search = WorldAabb::new(x, y - 1.0, z, x + 1.0, y + 1.0, z + 1.0);
178
179    let provider = WorldCollisionProvider::new(world);
180    let mut shapes = provider.get_block_collisions(&search);
181    shapes.extend(provider.get_entity_collisions(&search));
182
183    1.0 + collide(Axis::Y, &raised_box, &shapes, -MAX_DROP)
184}

207    /// A world with one solid block to click on and the chunk around it loaded.
208    fn ground_world(key: &'static str) -> Arc<World> {
209        init_vanilla_registry();
210        init_globals_once();
211        let world = fresh_test_world(key);
212        insert_ready_full_chunk(&world, ChunkPos::new(0, 0));
213        assert!(world.set_block(
214            GROUND,
215            vanilla_blocks::STONE.default_state(),
216            UpdateFlags::UPDATE_NONE,
217        ));
218        world

221    /// A survival player holding a stack of two stands, so a shrink shows.
222    fn stand_holder(world: &Arc<World>) -> Arc<Player> {
223        let player =
224            TestPlayerBuilder::new(Arc::clone(world), "StandTester", TEST_PLAYER_ENTITY_ID).build();
225        let mut stack = ItemStack::new(&vanilla_items::ARMOR_STAND);
226        stack.count = 2;
227        player
228            .inventory
229            .lock()
230            .set_item_in_hand(InteractionHand::MainHand, stack);
231        player
232    }

234    fn hit(pos: BlockPos, face: Direction) -> BlockHitResult {
235        let (step_x, step_y, step_z) = face.offset();
236        BlockHitResult {
237            location: DVec3::new(
238                f64::from(pos.x()) + 0.5 + f64::from(step_x) * 0.5,
239                f64::from(pos.y()) + 0.5 + f64::from(step_y) * 0.5,
240                f64::from(pos.z()) + 0.5 + f64::from(step_z) * 0.5,
241            ),
242            direction: face,
243            block_pos: pos,
244            miss: false,
245            inside: false,
246            world_border_hit: false,
247        }
248    }
249
250    /// Every stand the world holds, wherever the click put it.
251    fn stands(world: &Arc<World>) -> Vec<DVec3> {
252        let everywhere = WorldAabb::new(0.0, 0.0, 0.0, 16.0, 128.0, 16.0);
253        world
254            .get_entities_in_aabb_matching(&everywhere, |entity| {
255                entity.entity_type().key == vanilla_entities::ARMOR_STAND.key
256            })
257            .iter()
258            .map(|entity| entity.position())
259            .collect()
260    }
261
262    fn click(
263        world: &Arc<World>,
264        player: &Arc<Player>,
265        hit_result: BlockHitResult,
266    ) -> (InteractionResult, i32) {
267        let mut context = UseOnContext::new(
268            player,
269            InteractionHand::MainHand,
270            hit_result,
271            world,
272            player.inventory.clone(),
273        );
274        let result = ArmorStandItem.use_on(&mut context);
275        let left = context.inv.with_item(|item| item.count);
276        (result, left)
277    }

279    /// The whole point of the item: a click on something solid leaves a stand
280    /// standing on it, one block up from the block that was clicked, and takes
281    /// one stand out of the stack.
282    #[test]
283    fn a_click_on_a_block_top_leaves_a_stand_on_the_block() {
284        let world = ground_world("armor_stand_item_place");
285        let player = stand_holder(&world);
286
287        let (result, left) = click(&world, &player, hit(GROUND, Direction::Up));
288
289        assert_eq!(result, InteractionResult::Success);
290        assert_eq!(left, 1, "placing a stand should take one from the stack");
291
292        let placed = stands(&world);
293        assert_eq!(placed.len(), 1, "exactly one stand should have been placed");
294        assert!(
295            (placed[0].y - STAND_FLOOR).abs() < 1.0e-9,
296            "the stand should rest on the clicked block, not float: {}",
297            placed[0].y
298        );
299        assert!(
300            (placed[0].x - 8.5).abs() < 1.0e-9 && (placed[0].z - 8.5).abs() < 1.0e-9,
301            "the stand should stand in the middle of its block: {placed:?}"
302        );
303    }

358    /// Compares two yaws as directions: vanilla's arithmetic can land on -180
359    /// or 180 for the same facing, and either is right.
360    fn assert_facing(actual: f32, expected: f32) {
361        let difference = (actual - expected).rem_euclid(360.0);
362        assert!(
363            difference < 1.0e-4 || (360.0 - difference) < 1.0e-4,
364            "expected the stand to face {expected}, got {actual}"
365        );
366    }

368    /// Vanilla parity: the yaw arithmetic of `useOn`. Eight facings, each one
369    /// turned around from the player, so a stand placed by somebody looking
370    /// north looks south at them.
371    #[test]
372    fn the_stand_faces_back_at_the_player_in_eight_steps() {
373        assert_facing(placement_yaw(0.0), 180.0);
374        assert_facing(placement_yaw(180.0), 0.0);
375        assert_facing(placement_yaw(90.0), 270.0);
376        assert_facing(placement_yaw(270.0), 90.0);
377
378        // A yaw a little either side of a facing still snaps onto it.
379        assert_facing(placement_yaw(20.0), 180.0);
380        assert_facing(placement_yaw(-20.0), 180.0);
381
382        // And every yaw lands on one of the eight, never between them.
383        for degrees in -360_i16..=360 {
384            let yaw = placement_yaw(f32::from(degrees));
385            assert!(
386                (yaw / FACING_STEP).fract().abs() < 1.0e-4,
387                "a yaw of {degrees} gave {yaw}, which is not one of the eight facings"
388            );
389        }
390    }
