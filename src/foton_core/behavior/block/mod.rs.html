55/// Returns `state` after every neighbour has had its say about its shape.
56///
57/// Vanilla parity: `Block.updateFromNeighbourShapes`. It is what gives a grass
58/// block its `snowy` value and a fence its connections at the moment it is
59/// placed, rather than waiting for the next neighbour update.
60#[must_use]
61pub(crate) fn update_from_neighbour_shapes(
62    world: &Arc<World>,
63    state: BlockStateId,
64    pos: BlockPos,
65) -> BlockStateId {
66    let mut updated = state;
67    for direction in Direction::UPDATE_SHAPE_ORDER {
68        let neighbor_pos = pos.relative(direction);
69        let neighbor_state = world.get_block_state(neighbor_pos);
70        let behavior = BLOCK_BEHAVIORS.get_behavior(updated.get_block());
71        updated =
72            behavior.update_shape(updated, world, pos, direction, neighbor_pos, neighbor_state);
73    }
74    updated
75}

88/// Gets random loot from a given loot table reference and other factors, and returns
89/// each item from it in a [`Vec`].
90#[must_use]
91pub(crate) fn drop_from_block_interact_loot_table(
92    key: LootTableRef,
93    interacted_block_state: BlockStateId,
94    _interacted_block_entity: Option<SharedBlockEntity>,
95    tool: Option<&ItemStack>,
96    interacting_entity: Option<&dyn Entity>,
97    rng: &mut ThreadRng,
98) -> Vec<ItemStack> {
99    let mut ctx = LootContext::new(rng).with_block_state(interacted_block_state);
100
101    // TODO: Add the block entity to the context when it can be done.
102
103    if let Some(interacting_entity) = interacting_entity {
104        ctx = ctx.with_interacting_entity(entity_loot_ref(interacting_entity));
105    }
106
107    if let Some(tool) = tool {
108        ctx = ctx.with_tool(tool);
109    }
110
111    key.get_random_items(&mut ctx)
112}

114/// Lets plugins change or refuse what a player picks from a block.
115///
116/// Paper parity: `CraftEventFactory.callPlayerHarvestBlockEvent`, which
117/// sweet berry bushes and cave vines go through with the main hand. Returns
118/// the stacks to drop, or `None` when a plugin refused -- the caller then
119/// leaves the block as it was.
120pub(crate) fn offer_harvest(
121    player: &Player,
122    world: &Arc<World>,
123    pos: BlockPos,
124    items: Vec<ItemStack>,
125) -> Option<Vec<ItemStack>> {
126    let mut event = PlayerHarvestBlockEvent::new(
127        player.gameprofile.id,
128        world.key.to_string(),
129        pos,
130        InteractionHand::MainHand,
131        items,
132    );
133    player.fire_event(&mut event);
134    if event.is_cancelled() {
135        return None;
136    }
137    Some(event.items().to_vec())
138}

140/// Samples a block's experience drop and applies the tool's enchantments to it.
141///
142/// Mirrors the sampling half of vanilla `Block.tryDropExperience`. Mining
143/// experience is incidental live-gameplay randomness, so Foton samples it from
144/// an unseeded runtime source.
145pub(crate) fn sample_block_experience(tool: &ItemStack, experience: &IntProvider) -> i32 {
146    let mut random = LegacyRandom::from_seed(rand::random());
147    let base_experience = experience.sample(&mut random);
148    tool.apply_unconditional_enchantment_value_effects(
149        EnchantmentEffectComponent::BlockExperience,
150        base_experience as f32,
151    ) as i32
152}

154/// Drops sampled block experience, after plugins had their say.
155///
156/// Mirrors the dropping half of vanilla `Block.tryDropExperience`, with
157/// Bukkit's `BlockExpEvent` in between.
158fn pop_block_experience(world: &Arc<World>, pos: BlockPos, experience: i32) {
159    if experience > 0 {
160        let mut event = BlockExpEvent::new(world.key.to_string(), pos, experience);
161        world.fire_event(&mut event);
162        if !event.is_cancelled() && event.exp_to_drop() > 0 {
163            world.pop_experience(pos, event.exp_to_drop());
164        }
165    }
166}
