1//! What a sapling grows into.
2//!
3//! Vanilla parity: `TreeGrower`, the table that turns a sapling into a
4//! configured tree feature. Every entry here is transcribed from the static
5//! constants of `TreeGrower.java`; the sapling that points at each one comes
6//! from the extracted `tree_grower_name`.
7//!
8//! The interesting part is not the table but that it can run at all: a tree is
9//! a worldgen feature, and worldgen features write through a `WorldGenRegion`
10//! that only exists while a chunk is being generated. The tree feature is
11//! written against [`LevelAccessor`] instead, so the same code that grows a
12//! forest during generation grows one from a sapling in a live world -- through
13//! `World::set_block`, so the client sees it.
14

33/// How far around a sapling vanilla looks for flowers.
34///
35/// Vanilla parity: the `below().north(2).west(2)` to `above().south(2).east(2)`
36/// box of `TreeGrower.hasFlowers`. Flowers near an oak are what turn it into a
37/// bee-nest oak, which is the whole reason the check exists.
38const FLOWER_SEARCH_RADIUS: i32 = 2;
39
40/// One sapling's worth of tree choices.
41///
42/// Vanilla parity: `TreeGrower`. The keys are vanilla configured-feature paths,
43/// resolved through the registry when a tree is actually grown.
44pub struct TreeGrower {
45    /// Chance of preferring the secondary variant of each pair.
46    secondary_chance: f32,
47    /// The 2x2 tree, if this sapling can make one.
48    mega_tree: Option<&'static str>,
49    /// The other 2x2 tree.
50    secondary_mega_tree: Option<&'static str>,
51    /// The ordinary tree.
52    tree: Option<&'static str>,
53    /// The other ordinary tree.
54    secondary_tree: Option<&'static str>,
55    /// The tree grown when there are flowers nearby.
56    flowers: Option<&'static str>,
57    /// The other flowering tree.
58    secondary_flowers: Option<&'static str>,
59}
60
61impl TreeGrower {
62    const fn simple(
63        mega_tree: Option<&'static str>,
64        tree: Option<&'static str>,
65        flowers: Option<&'static str>,
66    ) -> Self {
67        Self {
68            secondary_chance: 0.0,
69            mega_tree,
70            secondary_mega_tree: None,
71            tree,
72            secondary_tree: None,
73            flowers,
74            secondary_flowers: None,
75        }
76    }
77
78    /// Returns the grower a sapling names, if it is one Foton knows.
79    ///
80    /// Vanilla parity: the `GROWERS` map `TreeGrower`'s constructor fills.
81    #[must_use]
82    pub fn by_name(name: &str) -> Option<&'static Self> {
83        Some(match name {
84            "oak" => &OAK,
85            "spruce" => &SPRUCE,
86            "birch" => &BIRCH,
87            "jungle" => &JUNGLE,
88            "acacia" => &ACACIA,
89            "cherry" => &CHERRY,
90            "dark_oak" => &DARK_OAK,
91            "pale_oak" => &PALE_OAK,
92            "mangrove" => &MANGROVE,
93            "azalea" => &AZALEA,
94            _ => return None,
95        })
96    }
97
98    /// Vanilla parity: `TreeGrower.getConfiguredFeature`.
99    fn configured_feature(&self, rng: &mut dyn Rng, has_flowers: bool) -> Option<&'static str> {
100        if rng.random::<f32>() < self.secondary_chance {
101            if has_flowers && self.secondary_flowers.is_some() {
102                return self.secondary_flowers;
103            }
104            if self.secondary_tree.is_some() {
105                return self.secondary_tree;
106            }
107        }
108
109        if has_flowers && self.flowers.is_some() {
110            self.flowers
111        } else {
112            self.tree
113        }
114    }
115
116    /// Vanilla parity: `TreeGrower.getConfiguredMegaFeature`.
117    fn configured_mega_feature(&self, rng: &mut dyn Rng) -> Option<&'static str> {
118        if self.secondary_mega_tree.is_some() && rng.random::<f32>() < self.secondary_chance {
119            self.secondary_mega_tree
120        } else {
121            self.mega_tree
122        }
123    }
124
125    /// Grows the tree, replacing the sapling.
126    ///
127    /// Vanilla parity: `TreeGrower.growTree`. Returns whether a tree was placed;
128    /// on failure the sapling is put back exactly as it was, which is what lets
129    /// a player try again rather than losing it.
130    pub fn grow_tree(
131        &self,
132        world: &Arc<World>,
133        pos: BlockPos,
134        state: BlockStateId,
135        rng: &mut dyn Rng,
136    ) -> bool {
137        if let Some(mega) = self.configured_mega_feature(rng)
138            && Self::grow_mega_tree(world, pos, state, rng, mega)
139        {
140            return true;
141        }
142
143        let has_flowers = has_flowers_near(world, pos);
144        let Some(key) = self.configured_feature(rng, has_flowers) else {
145            return false;
146        };
147
148        // Vanilla clears the sapling first so the trunk placer sees an empty
149        // column, and restores it when the tree does not fit.
150        let cleared = fluid_state_to_block(world.get_block_state(pos).get_fluid_state());
151        world.set_block(pos, cleared, UpdateFlags::UPDATE_CLIENTS);
152
153        if place_tree(world, pos, rng, key) {
154            true
155        } else {
156            world.set_block(pos, state, UpdateFlags::UPDATE_CLIENTS);
157            false
158        }
159    }
160
161    /// Grows a 2x2 tree if this sapling is one corner of a 2x2 patch.
162    ///
163    /// Vanilla parity: the mega-tree half of `TreeGrower.growTree`, which tries
164    /// each of the four corners the sapling could be.
165    fn grow_mega_tree(
166        world: &Arc<World>,
167        pos: BlockPos,
168        state: BlockStateId,
169        rng: &mut dyn Rng,
170        key: &'static str,
171    ) -> bool {
172        for dx in [0, -1] {
173            for dz in [0, -1] {
174                if !is_two_by_two_sapling(world, state, pos, dx, dz) {
175                    continue;
176                }
177
178                let corners = [
179                    pos.offset(dx, 0, dz),
180                    pos.offset(dx + 1, 0, dz),
181                    pos.offset(dx, 0, dz + 1),
182                    pos.offset(dx + 1, 0, dz + 1),
183                ];
184                let air = vanilla_blocks::AIR.default_state();
185                for corner in corners {
186                    world.set_block(corner, air, UpdateFlags::UPDATE_CLIENTS);
187                }
188
189                if place_tree(world, corners[0], rng, key) {
190                    return true;
191                }
192
193                for corner in corners {
194                    world.set_block(corner, state, UpdateFlags::UPDATE_CLIENTS);
195                }
196                return false;
197            }
198        }
199
200        false
201    }
202}

204/// Places one configured tree feature in a live world.
205fn place_tree(world: &Arc<World>, pos: BlockPos, rng: &mut dyn Rng, key: &'static str) -> bool {
206    let identifier = Identifier::vanilla_static(key);
207    let Some(feature) = REGISTRY.configured_features.by_key(&identifier) else {
208        log::warn!("a tree grower names an unknown configured feature {identifier}");
209        return false;
210    };
211    let ConfiguredFeatureKind::Tree(config) = &feature.kind else {
212        log::warn!("a tree grower names {identifier}, which is not a tree feature");
213        return false;
214    };
215
216    let mut random = WorldgenRandom::from_seed(rng.random());
217    // A tree grown from a sapling has no nested ground feature to place: the
218    // only one is the pale moss patch, and placing it needs the worldgen
219    // dispatcher. A pale oak grown by hand comes up without its moss.
220    let mut ground_features = Vec::new();
221    FeatureDecorationRunner::place_tree_feature(
222        world,
223        &REGISTRY,
224        &mut random,
225        config,
226        pos,
227        &mut ground_features,
228    )
229}

231/// Generates a plugin-requested tree with a deterministic world/position seed.
232pub fn generate_tree(world: &Arc<World>, pos: BlockPos, key: &'static str) -> bool {
233    let seed = world.seed() as u64
234        ^ (pos.x() as u64).wrapping_mul(341_873_128_712)
235        ^ (pos.z() as u64).wrapping_mul(132_897_987_541);
236    let mut random = SmallRng::seed_from_u64(seed);
237    place_tree(world, pos, &mut random, key)
238}

240/// Vanilla parity: `TreeGrower.isTwoByTwoSapling`.
241fn is_two_by_two_sapling(
242    world: &Arc<World>,
243    state: BlockStateId,
244    pos: BlockPos,
245    dx: i32,
246    dz: i32,
247) -> bool {
248    let block = state.get_block();
249    [(0, 0), (1, 0), (0, 1), (1, 1)]
250        .into_iter()
251        .all(|(ox, oz)| {
252            world
253                .get_block_state(pos.offset(dx + ox, 0, dz + oz))
254                .get_block()
255                == block
256        })
257}

259/// Vanilla parity: `TreeGrower.hasFlowers`.
260fn has_flowers_near(world: &Arc<World>, pos: BlockPos) -> bool {
261    let from = pos.offset(-FLOWER_SEARCH_RADIUS, -1, -FLOWER_SEARCH_RADIUS);
262    let to = pos.offset(FLOWER_SEARCH_RADIUS, 1, FLOWER_SEARCH_RADIUS);
263    BlockPos::between_closed(from, to).any(|candidate| {
264        world
265            .get_block_state(candidate)
266            .get_block()
267            .has_tag(&BlockTag::FLOWERS)
268    })
269}

318    /// Every sapling in the extracted data names a grower this table knows.
319    ///
320    /// Without this, a sapling whose grower is missing would simply never grow
321    /// and nothing would say so.
322    #[test]
323    fn every_sapling_grower_name_resolves() {
324        for name in [
325            "oak", "spruce", "birch", "jungle", "acacia", "cherry", "dark_oak", "pale_oak",
326            "mangrove", "azalea",
327        ] {
328            assert!(
329                TreeGrower::by_name(name).is_some(),
330                "no grower named {name}"
331            );
332        }
333        assert!(TreeGrower::by_name("nonexistent").is_none());
334    }
335
336    /// Every feature the table names exists and really is a tree.
337    ///
338    /// A typo here would be a sapling that consumes itself and grows nothing.
339    #[test]
340    fn every_named_feature_is_a_tree_that_exists() {
341        init_vanilla_registry();
342
343        for grower in [
344            &OAK, &SPRUCE, &MANGROVE, &AZALEA, &BIRCH, &JUNGLE, &ACACIA, &CHERRY, &DARK_OAK,
345            &PALE_OAK,
346        ] {
347            for key in [
348                grower.mega_tree,
349                grower.secondary_mega_tree,
350                grower.tree,
351                grower.secondary_tree,
352                grower.flowers,
353                grower.secondary_flowers,
354            ]
355            .into_iter()
356            .flatten()
357            {
358                let identifier = Identifier::vanilla_static(key);
359                let feature = REGISTRY
360                    .configured_features
361                    .by_key(&identifier)
362                    .unwrap_or_else(|| panic!("unknown configured feature {identifier}"));
363                assert!(
364                    matches!(feature.kind, ConfiguredFeatureKind::Tree(_)),
365                    "{identifier} is not a tree feature"
366                );
367            }
368        }
369    }

371    /// A grower with no ordinary tree only ever makes a 2x2 one.
372    #[test]
373    fn dark_oak_has_no_single_sapling_tree() {

375        assert!(DARK_OAK.configured_feature(&mut rng, false).is_none());
376        assert_eq!(DARK_OAK.configured_mega_feature(&mut rng), Some("dark_oak"));

390        ));
