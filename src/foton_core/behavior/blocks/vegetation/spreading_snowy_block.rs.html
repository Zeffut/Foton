1//! Vanilla `SpreadingSnowyBlock`: the random tick grass and mycelium share.
2//!
3//! Two rules, both on the block above. Cover one and it goes back to dirt;
4//! leave it open and it creeps onto the dirt around it. Vanilla stopped asking
5//! about the light level here -- 26.2 asks whether the block above dampens all
6//! fifteen levels of light, which is why grass survives an unlit cave but not a
7//! slab laid on top of it.
8

24/// Vanilla parity: `LightEngine.LIGHT_BLOCKED`, the dampening a fully occluding
25/// face reports. Grass needs anything less than that above it.
26const FULLY_DAMPENED: u8 = 15;
27
28/// Light the block above needs for grass to spread onto a neighbour.
29///
30/// Vanilla parity: the `getMaxLocalRawBrightness(pos.above()) >= 9` of
31/// `SpreadingSnowyBlock.randomTick`.
32const MIN_SPREAD_LIGHT: u8 = 9;
33
34/// Vanilla parity: `SpreadingSnowyBlock.canStayAlive`.
35fn can_stay_alive(state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
36    let above = pos.above();
37    let above_state = world.get_block_state(above);
38
39    // A single layer of snow is the one cover grass lives under.
40    if above_state.get_block() == &vanilla_blocks::SNOW
41        && above_state.get_value(&BlockStateProperties::LAYERS) == 1
42    {
43        return true;
44    }
45
46    if above_state.get_fluid_state().is_full() {
47        return false;
48    }
49
50    get_light_block_into(
51        state,
52        above_state,
53        Direction::Up,
54        above_state.get_light_dampening(),
55    ) < FULLY_DAMPENED
56}
57
58/// Vanilla parity: `SpreadingSnowyBlock.canPropagate`. Grass will not climb into
59/// a block that has water sitting on it, even though it would survive there.
60fn can_propagate(state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
61    can_stay_alive(state, world, pos)
62        && !world
63            .get_block_state(pos.above())
64            .get_fluid_state()
65            .fluid_id
66            .has_tag(&FluidTag::WATER)
67}
68
69/// Vanilla parity: `SpreadingSnowyBlock.randomTick`.
70///
71/// `base_block` is what this block falls back to when it is smothered --
72/// `BlockItemIds.DIRT` for both grass and mycelium.
73pub(super) fn random_tick(
74    block: BlockRef,
75    base_block: BlockRef,
76    state: BlockStateId,
77    world: &Arc<World>,
78    pos: BlockPos,
79) {
80    if !can_stay_alive(state, world.as_ref(), pos) {
81        world.set_block(pos, base_block.default_state(), UpdateFlags::UPDATE_ALL);
82        return;
83    }
84
85    if world.max_local_raw_brightness(pos.above(), world.sky_darkening()) < MIN_SPREAD_LIGHT {
86        return;
87    }
88
89    let spread_state = block.default_state();
90    let mut rng = rand::rng();
91    // Four tries at a neighbour, biased downwards: vanilla's y offset spans
92    // -3..=1, so grass runs down a slope faster than it climbs one.
93    for _ in 0..4 {
94        let test_pos = pos.offset(
95            rng.random_range(0..3) - 1,
96            rng.random_range(0..5) - 3,
97            rng.random_range(0..3) - 1,
98        );
99        if world.get_block_state(test_pos).get_block() == base_block
100            && can_propagate(spread_state, world.as_ref(), test_pos)
101        {
102            let snowy = is_snowy_setting(world.get_block_state(test_pos.above()));
103            world.set_block(
104                test_pos,
105                spread_state.set_value(&BlockStateProperties::SNOWY, snowy),
106                UpdateFlags::UPDATE_ALL,
107            );
108        }
109    }
110}

112#[cfg(test)]
113mod tests {
114    use foton_registry::blocks::block_state_ext::BlockStateExt as _;
115    use foton_registry::blocks::properties::BlockStateProperties;
116    use foton_registry::{init_vanilla_registry, vanilla_blocks};
117    use foton_utils::BlockPos;
118
119    use super::{can_propagate, can_stay_alive};
120    use crate::test_support::TestLevel;
121
122    const POS: BlockPos = BlockPos::new(0, 64, 0);
123
124    fn covered_by(state: foton_utils::BlockStateId) -> TestLevel {
125        TestLevel::default()
126            .with_min_y(0)
127            .with_block(POS.above(), state)
128    }
129
130    /// The rule 26.2 replaced the old light check with: what matters is whether
131    /// the block on top blocks light outright, not how bright it is.
132    #[test]
133    fn a_solid_lid_smothers_grass_but_glass_does_not() {
134        init_vanilla_registry();
135
136        let grass = vanilla_blocks::GRASS_BLOCK.default_state();
137
138        assert!(can_stay_alive(
139            grass,
140            &TestLevel::default().with_min_y(0),
141            POS
142        ));
143        assert!(!can_stay_alive(
144            grass,
145            &covered_by(vanilla_blocks::STONE.default_state()),
146            POS
147        ));
148        assert!(can_stay_alive(
149            grass,
150            &covered_by(vanilla_blocks::GLASS.default_state()),
151            POS
152        ));
153    }
154
155    /// One layer of snow is a lid grass lives under; more is not.
156    #[test]
157    fn one_layer_of_snow_is_survivable_and_a_full_block_is_not() {
158        init_vanilla_registry();
159
160        let grass = vanilla_blocks::GRASS_BLOCK.default_state();
161
162        assert!(can_stay_alive(
163            grass,
164            &covered_by(vanilla_blocks::SNOW.default_state()),
165            POS
166        ));
167        assert!(!can_stay_alive(
168            grass,
169            &covered_by(vanilla_blocks::SNOW_BLOCK.default_state()),
170            POS
171        ));
172    }
173
174    /// Standing water smothers grass outright. Water that is only running over
175    /// it does not -- but grass still will not climb into it, which is the whole
176    /// reason `canPropagate` asks a second question `canStayAlive` does not.
177    #[test]
178    fn deep_water_kills_and_shallow_water_only_blocks_spreading() {
179        init_vanilla_registry();
180
181        let grass = vanilla_blocks::GRASS_BLOCK.default_state();
182
183        let under_a_source = covered_by(vanilla_blocks::WATER.default_state());
184        assert!(!can_stay_alive(grass, &under_a_source, POS));
185
186        let flowing = vanilla_blocks::WATER
187            .default_state()
188            .set_value(&BlockStateProperties::LEVEL, 1);
189        let under_a_trickle = covered_by(flowing);
190        assert!(can_stay_alive(grass, &under_a_trickle, POS));
191        assert!(!can_propagate(grass, &under_a_trickle, POS));
192    }
