1//! Creaking heart behavior.
2//!
3//! Vanilla parity: `CreakingHeartBlock`.
4//!
5//! The heart is the block buried in a pale oak that keeps a creaking alive. Everything the
6//! block itself owns is here: the three-state machine (`uprooted` when the pale oak logs
7//! that hold it are gone, `dormant` by day, `awake` by night), the axis it is placed on, the
8//! twenty-odd experience a naturally generated one drops when a player breaks it, and the
9//! comparator output.
10//!
11//! The spawn, the tether and the hurt-transfer all live on
12//! `CreakingHeartBlockEntity`; what the block adds is breaking the heart, which
13//! kills the creaking with the breaker credited for it.
14//!
15//! Not implemented: `onExplosionHit`, which does the same for a heart blown up
16//! rather than mined. Foton's explosions have no per-block callback, so a heart
17//! destroyed by a blast still tears its creaking down -- through
18//! `pre_remove_side_effects` -- but as an anonymous crumble rather than a death
19//! the blast is credited with.
20

47/// Vanilla `CreakingHeartBlock.AXIS`.
48pub(crate) const AXIS: &EnumProperty<Axis> = &BlockStateProperties::AXIS;
49/// Vanilla `CreakingHeartBlock.STATE`.
50pub(crate) const STATE: &EnumProperty<CreakingHeartState> =
51    &BlockStateProperties::CREAKING_HEART_STATE;
52/// Vanilla `CreakingHeartBlock.NATURAL`.
53const NATURAL: &BoolProperty = &BlockStateProperties::NATURAL;
54
55/// Vanilla `CreakingHeartBlock.tryAwardExperience`: `nextIntBetweenInclusive(20, 24)`.
56const NATURAL_BREAK_EXPERIENCE: RangeInclusive<i32> = 20..=24;
57
58/// Vanilla `CreakingHeartBlock`.
59#[block_behavior]
60pub struct CreakingHeartBlock {
61    block: BlockRef,
62}
63
64impl CreakingHeartBlock {
65    /// Creates the creaking heart behavior.
66    #[must_use]
67    pub const fn new(block: BlockRef) -> Self {
68        Self { block }
69    }
70
71    /// Vanilla `CreakingHeartBlock.hasRequiredLogs`: pale oak logs on both ends of the axis,
72    /// laid along the same axis as the heart.
73    #[must_use]
74    pub fn has_required_logs(state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> bool {
75        let axis = state.get_value(AXIS);
76        axis_directions(axis).into_iter().all(|direction| {
77            let neighbor = world.get_block_state(pos.relative(direction));
78            neighbor.get_block().has_tag(&BlockTag::PALE_OAK_LOGS)
79                && neighbor.try_get_value(AXIS) == Some(axis)
80        })
81    }
82
83    /// Vanilla `CreakingHeartBlock.updateState`.
84    ///
85    /// This only ever wakes an uprooted heart that has just found its logs; the walk back
86    /// to `uprooted`, and the day-night flip between `dormant` and `awake`, belong to
87    /// `CreakingHeartBlockEntity`, which is only ticked once the heart is rooted.
88    #[must_use]
89    pub fn updated_state(state: BlockStateId, world: &World, pos: BlockPos) -> BlockStateId {
90        let uprooted = state.get_value(STATE) == CreakingHeartState::Uprooted;
91        if !uprooted || !Self::has_required_logs(state, world, pos) {
92            return state;
93        }
94        state.set_value(STATE, awake_or_dormant(world))
95    }
96}
97
98/// Vanilla `Direction.Axis.getDirections`, which is `{positive, negative}`.
99const fn axis_directions(axis: Axis) -> [Direction; 2] {
100    match axis {
101        Axis::X => [Direction::East, Direction::West],
102        Axis::Y => [Direction::Up, Direction::Down],
103        Axis::Z => [Direction::South, Direction::North],
104    }
105}
106
107/// Vanilla's `EnvironmentAttributes.CREAKING_ACTIVE` branch, shared with the block entity.
108pub(crate) fn awake_or_dormant(world: &World) -> CreakingHeartState {
109    if world.creaking_active() {
110        CreakingHeartState::Awake
111    } else {
112        CreakingHeartState::Dormant
113    }
114}

116impl BlockBehavior for CreakingHeartBlock {
117    /// Vanilla `CreakingHeartBlock.getStateForPlacement`.
118    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
119        let placed = self
120            .block
121            .default_state()
122            .set_value(AXIS, context.clicked_face().get_axis());
123        Some(CreakingHeartBlock::updated_state(
124            placed,
125            context.world.as_ref(),
126            context.place_pos(),
127        ))
128    }
129
130    /// Vanilla `CreakingHeartBlock.updateShape`, which only schedules the state check.
131    fn update_shape(
132        &self,
133        state: BlockStateId,
134        world: &dyn ScheduledTickAccess,
135        pos: BlockPos,
136        _direction: Direction,
137        _neighbor_pos: BlockPos,
138        _neighbor_state: BlockStateId,
139    ) -> BlockStateId {
140        world.schedule_block_tick_default(pos, self.block, 1);
141        state
142    }
143
144    /// Vanilla `CreakingHeartBlock.tick`.
145    fn tick(&self, state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
146        let updated = CreakingHeartBlock::updated_state(state, world.as_ref(), pos);
147        if updated != state {
148            world.set_block(pos, updated, UpdateFlags::UPDATE_ALL);
149        }
150    }
151
152    fn new_block_entity(
153        &self,
154        level: Weak<World>,
155        pos: BlockPos,
156        state: BlockStateId,
157    ) -> BlockEntityCreation {
158        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
159            &vanilla_block_entity_types::CREAKING_HEART,
160            level,
161            pos,
162            state,
163        ))
164    }
165
166    /// Vanilla `CreakingHeartBlock.getTicker`, which refuses to tick an uprooted heart.
167    fn get_block_entity_ticker(
168        &self,
169        _world: &Arc<World>,
170        state: BlockStateId,
171        block_entity_type: BlockEntityTypeRef,
172    ) -> Option<BlockEntityTicker> {
173        if state.get_value(STATE) == CreakingHeartState::Uprooted {
174            return None;
175        }
176        BlockEntityTicker::for_matching_entity_tick(
177            block_entity_type,
178            &vanilla_block_entity_types::CREAKING_HEART,
179        )
180    }
181
182    /// Vanilla `CreakingHeartBlock.affectNeighborsAfterRemoval`, whose one line is the
183    /// `Containers` after-destroy neighbor update: a comparator reading the heart has to
184    /// notice that it is gone.
185    fn affect_neighbors_after_removal(
186        &self,
187        state: BlockStateId,
188        world: &Arc<World>,
189        pos: BlockPos,
190        _moved_by_piston: bool,
191    ) {
192        world.update_neighbor_for_output_signal(pos, state.get_block());
193    }
194
195    /// Vanilla `CreakingHeartBlock.playerWillDestroy`.
196    ///
197    /// Breaking the heart is how a player is rid of the creaking, and the blow
198    /// is credited to them: the creaking dies of a player attack rather than
199    /// simply crumbling, which is what makes the kill count.
200    fn player_will_destroy(
201        &self,
202        state: BlockStateId,
203        world: &Arc<World>,
204        pos: BlockPos,
205        player: &Player,
206    ) -> BlockStateId {
207        if let Some(block_entity) = world.get_block_entity(pos)
208            && let Some(heart) = block_entity.downcast_ref::<CreakingHeartBlockEntity>()
209        {
210            let source = DamageSource::environment(&vanilla_damage_types::PLAYER_ATTACK)
211                .with_causing_entity(player.id())
212                .with_direct_entity(player.id());
213            heart.remove_protector(Some(&source));
214        }
215
216        // Vanilla `Player.preventsBlockDrops` is the creative `instabuild` ability.
217        if !player.is_spectator() && !player.has_infinite_materials() && state.get_value(NATURAL) {
218            world.pop_experience(pos, rand::rng().random_range(NATURAL_BREAK_EXPERIENCE));
219        }
220        state
221    }
222
223    fn has_analog_output_signal(&self, _state: BlockStateId) -> bool {
224        true
225    }
226
227    /// Vanilla `CreakingHeartBlock.getAnalogOutputSignal`, which scales with how far the
228    /// creaking has wandered: fifteen at the heart, falling to zero at the edge of the
229    /// thirty-two block roam radius. A heart with no creaking reads zero.
230    fn get_analog_output_signal(
231        &self,
232        state: BlockStateId,
233        world: &dyn LevelReader,
234        pos: BlockPos,
235        _direction: Direction,
236    ) -> i32 {
237        if state.get_value(STATE) == CreakingHeartState::Uprooted {
238            return 0;
239        }
240        world.get_block_entity(pos).map_or(0, |block_entity| {
241            block_entity
242                .downcast_ref::<CreakingHeartBlockEntity>()
243                .map_or(0, CreakingHeartBlockEntity::analog_output_signal)
244        })
245    }
246}

249mod tests {
250    use foton_registry::{init_vanilla_registry, vanilla_blocks};
251    use foton_utils::ChunkPos;

253    use super::*;
254    use crate::behavior::init_behaviors;
255    use crate::block_entity::init_block_entities;
256    use crate::test_support::{fresh_test_world, insert_ready_full_chunk};
257
258    fn init() {
259        init_vanilla_registry();
260        init_behaviors();
261        init_block_entities();
262    }
263
264    fn heart(axis: Axis, state: CreakingHeartState) -> BlockStateId {
265        vanilla_blocks::CREAKING_HEART
266            .default_state()
267            .set_value(AXIS, axis)
268            .set_value(STATE, state)
269    }
270
271    /// A heart only counts logs that run along its own axis, which is what makes a heart
272    /// generated inside a trunk different from one a player wedged between two crossways
273    /// logs. Getting the axis check wrong would let any two logs root a heart.
274    #[test]
275    fn a_heart_is_rooted_only_by_logs_lying_along_its_own_axis() {
276        init();
277        let pos = BlockPos::new(8, 70, 8);
278        let world = fresh_test_world("creaking_heart_logs");
279        insert_ready_full_chunk(&world, ChunkPos::from_block_pos(pos));
280
281        let upright = heart(Axis::Y, CreakingHeartState::Uprooted);
282        assert!(!CreakingHeartBlock::has_required_logs(
283            upright,
284            world.as_ref(),
285            pos
286        ));
287
288        let upright_log = vanilla_blocks::PALE_OAK_LOG
289            .default_state()
290            .set_value(AXIS, Axis::Y);
291        world.set_block(pos.above(), upright_log, UpdateFlags::UPDATE_NONE);
292        assert!(
293            !CreakingHeartBlock::has_required_logs(upright, world.as_ref(), pos),
294            "one log is not enough; vanilla needs both ends"
295        );
296
297        world.set_block(pos.below(), upright_log, UpdateFlags::UPDATE_NONE);
298        assert!(CreakingHeartBlock::has_required_logs(
299            upright,
300            world.as_ref(),
301            pos
302        ));
303
304        let crossways_log = vanilla_blocks::PALE_OAK_LOG
305            .default_state()
306            .set_value(AXIS, Axis::X);
307        world.set_block(pos.below(), crossways_log, UpdateFlags::UPDATE_NONE);
308        assert!(
309            !CreakingHeartBlock::has_required_logs(upright, world.as_ref(), pos),
310            "a log turned across the heart's axis does not root it"
311        );
312    }
313
314    /// A heart that lost its logs must fall back to `uprooted` and stop ticking, and one
315    /// that finds them must wake. This is the whole observable state machine, so a wrong
316    /// transition here is the difference between a working heart and an inert block.
317    #[test]
318    fn a_heart_uproots_without_logs_and_roots_again_with_them() {
319        init();
320        let pos = BlockPos::new(8, 70, 8);
321        let world = fresh_test_world("creaking_heart_state");
322        insert_ready_full_chunk(&world, ChunkPos::from_block_pos(pos));
323        let behavior = CreakingHeartBlock::new(&vanilla_blocks::CREAKING_HEART);
324
325        let uprooted = heart(Axis::Y, CreakingHeartState::Uprooted);
326        world.set_block(pos, uprooted, UpdateFlags::UPDATE_NONE);
327
328        behavior.tick(uprooted, &world, pos);
329        assert_eq!(
330            world.get_block_state(pos).get_value(STATE),
331            CreakingHeartState::Uprooted,
332            "a heart with no logs stays uprooted"
333        );
334
335        let log = vanilla_blocks::PALE_OAK_LOG
336            .default_state()
337            .set_value(AXIS, Axis::Y);
338        world.set_block(pos.above(), log, UpdateFlags::UPDATE_NONE);
339        world.set_block(pos.below(), log, UpdateFlags::UPDATE_NONE);
340
341        behavior.tick(uprooted, &world, pos);
342        assert_ne!(
343            world.get_block_state(pos).get_value(STATE),
344            CreakingHeartState::Uprooted,
345            "a heart with logs on both ends roots itself"
346        );
347    }

349    /// An uprooted heart is inert, and vanilla expresses that by refusing it a ticker at
350    /// all. Ticking one anyway would spawn a creaking out of a block the player has just
351    /// dug free of its tree.
352    #[test]
353    fn only_a_rooted_heart_is_given_a_ticker() {
354        init();
355        let world = fresh_test_world("creaking_heart_ticker");
356        let behavior = CreakingHeartBlock::new(&vanilla_blocks::CREAKING_HEART);
357
358        assert!(
359            behavior
360                .get_block_entity_ticker(
361                    &world,
362                    heart(Axis::Y, CreakingHeartState::Uprooted),
363                    &vanilla_block_entity_types::CREAKING_HEART,
364                )
365                .is_none()
366        );
367        assert!(
368            behavior
369                .get_block_entity_ticker(
370                    &world,
371                    heart(Axis::Y, CreakingHeartState::Dormant),
372                    &vanilla_block_entity_types::CREAKING_HEART,
373                )
374                .is_some()
375        );
376    }
