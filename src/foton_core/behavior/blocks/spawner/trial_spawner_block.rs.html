1//! The trial spawner block.
2//!
3//! Vanilla parity: `net.minecraft.world.level.block.TrialSpawnerBlock`.
4
5use std::sync::{Arc, Weak};
6
7use foton_macros::block_behavior;
8use foton_registry::block_entity_type::BlockEntityTypeRef;
9use foton_registry::blocks::BlockRef;
10use foton_registry::vanilla_block_entity_types;
11use foton_utils::{BlockPos, BlockStateId};
12
13use crate::behavior::block::{BlockBehavior, BlockEntityCreation};
14use crate::behavior::context::BlockPlaceContext;
15use crate::block_entity::{BLOCK_ENTITIES, BlockEntityTicker};
16use crate::world::World;
17
18/// Vanilla `TrialSpawnerBlock`.
19///
20/// The block itself is nearly empty: it owns the two state properties, which
21/// the extracted block registry already carries, and hands the tick to the
22/// block entity.
23#[block_behavior]
24pub struct TrialSpawnerBlock {
25    _block: BlockRef,
26}
27
28impl TrialSpawnerBlock {
29    /// Creates the trial spawner block behavior.
30    #[must_use]
31    pub const fn new(block: BlockRef) -> Self {
32        Self { _block: block }
33    }
34}
35
36impl BlockBehavior for TrialSpawnerBlock {
37    fn get_state_for_placement(&self, _context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
38        None
39    }
40
41    fn new_block_entity(
42        &self,
43        level: Weak<World>,
44        pos: BlockPos,
45        state: BlockStateId,
46    ) -> BlockEntityCreation {
47        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
48            &vanilla_block_entity_types::TRIAL_SPAWNER,
49            level,
50            pos,
51            state,
52        ))
53    }
54
55    /// Vanilla parity: `TrialSpawnerBlock.getTicker`, whose server branch reads
56    /// the ominous property off the live state and hands it to `tickServer`.
57    /// Foton's block entity reads the same property itself, so the ticker is the
58    /// plain one.
59    fn get_block_entity_ticker(
60        &self,
61        _world: &Arc<World>,
62        _state: BlockStateId,
63        block_entity_type: BlockEntityTypeRef,
64    ) -> Option<BlockEntityTicker> {
65        BlockEntityTicker::for_matching_entity_tick(
66            block_entity_type,
67            &vanilla_block_entity_types::TRIAL_SPAWNER,
68        )
69    }
70}
