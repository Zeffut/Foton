1//! Sculk catalyst behavior.
2//!
3//! Vanilla parity: `SculkCatalystBlock`.
4//!
5//! The catalyst is the block that pays for sculk. Its block entity listens for a mob dying
6//! within eight blocks, takes the experience that death was about to drop, and spends it as
7//! sculk growing outward; the block itself only carries the `bloom` pulse that shows it
8//! happened and the experience it drops when mined. See `SculkCatalystBlockEntity`.
9

30/// Vanilla `SculkCatalystBlock.PULSE`, which is the `bloom` block state property.
31const PULSE: &BoolProperty = &BlockStateProperties::BLOOM;
32
33/// Vanilla `SculkCatalystBlock`.
34#[block_behavior]
35pub struct SculkCatalystBlock {
36    block: BlockRef,
37    #[json_arg(int_provider, json = "xp_range")]
38    experience: IntProvider,
39}
40
41impl SculkCatalystBlock {
42    /// Creates the sculk catalyst behavior with its extracted experience provider.
43    #[must_use]
44    pub const fn new(block: BlockRef, experience: IntProvider) -> Self {
45        Self { block, experience }
46    }
47}
48
49impl BlockBehavior for SculkCatalystBlock {
50    fn get_state_for_placement(&self, _context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
51        Some(self.block.default_state().set_value(PULSE, false))
52    }
53
54    /// Vanilla `SculkCatalystBlock.tick`: the bloom lasts eight ticks and then clears.
55    fn tick(&self, state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
56        if !state.get_value(PULSE) {
57            return;
58        }
59        world.set_block(pos, state.set_value(PULSE, false), UpdateFlags::UPDATE_ALL);
60    }
61
62    fn new_block_entity(
63        &self,
64        level: Weak<World>,
65        pos: BlockPos,
66        state: BlockStateId,
67    ) -> BlockEntityCreation {
68        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
69            &vanilla_block_entity_types::SCULK_CATALYST,
70            level,
71            pos,
72            state,
73        ))
74    }
75
76    fn get_block_entity_ticker(
77        &self,
78        _world: &Arc<World>,
79        _state: BlockStateId,
80        block_entity_type: BlockEntityTypeRef,
81    ) -> Option<BlockEntityTicker> {
82        BlockEntityTicker::for_matching_entity_tick(
83            block_entity_type,
84            &vanilla_block_entity_types::SCULK_CATALYST,
85        )
86    }
87
88    fn spawn_after_break(
89        &self,
90        _state: BlockStateId,
91        world: &Arc<World>,
92        pos: BlockPos,
93        tool: &ItemStack,
94        drop_experience: bool,
95    ) {
96        if drop_experience {
97            try_drop_experience(world, pos, tool, &self.experience);
98        }
99    }
100}

102#[cfg(test)]
103mod tests {
104    use foton_registry::{init_vanilla_registry, vanilla_blocks};
105    use foton_utils::ChunkPos;
106
107    use super::*;
108    use crate::behavior::init_behaviors;
109    use crate::block_entity::init_block_entities;
110    use crate::test_support::{fresh_test_world, insert_ready_full_chunk};
111
112    /// The bloom is a one-shot pulse driven by a scheduled tick. A catalyst that never
113    /// cleared it would sit lit forever after the first death near it.
114    #[test]
115    fn a_blooming_catalyst_goes_dark_again_on_its_scheduled_tick() {
116        init_vanilla_registry();
117        init_behaviors();
118        init_block_entities();
119
120        let pos = BlockPos::new(8, 70, 8);
121        let world = fresh_test_world("sculk_catalyst_bloom");
122        insert_ready_full_chunk(&world, ChunkPos::from_block_pos(pos));
123
124        let behavior =
125            SculkCatalystBlock::new(&vanilla_blocks::SCULK_CATALYST, IntProvider::Constant(5));
126        let blooming = vanilla_blocks::SCULK_CATALYST
127            .default_state()
128            .set_value(PULSE, true);
129        assert!(world.set_block(pos, blooming, UpdateFlags::UPDATE_ALL));
130
131        behavior.tick(blooming, &world, pos);
132
133        assert!(!world.get_block_state(pos).get_value(PULSE));
134    }
