1//! Sculk sensor behavior, plain and calibrated.
2//!
3//! Vanilla parity: `SculkSensorBlock` and `CalibratedSculkSensorBlock`.
4//!
5//! A vibration reaching the block entity's listener ends in [`activate_sculk_sensor`],
6//! which drives the phase machine, the redstone output, the comparator read and the
7//! resonance that amethyst beside the sensor re-emits.

46/// Vanilla `SculkSensorBlock.ACTIVE_TICKS`.
47const ACTIVE_TICKS: i32 = 30;
48/// Vanilla `CalibratedSculkSensorBlock.getActiveTicks`.
49const CALIBRATED_ACTIVE_TICKS: i32 = 10;
50/// Vanilla `SculkSensorBlock.COOLDOWN_TICKS`.
51const COOLDOWN_TICKS: i32 = 10;
52/// Vanilla `SculkSensorBlock.spawnAfterBreak` uses `ConstantInt.of(5)`.
53const SENSOR_EXPERIENCE: IntProvider = IntProvider::Constant(5);
54/// Vanilla `SculkSensorBlock.onPlace` clears the power with flags `18`.
55const PLACEMENT_POWER_RESET_FLAGS: UpdateFlags =
56    UpdateFlags::UPDATE_CLIENTS.union(UpdateFlags::UPDATE_KNOWN_SHAPE);
57
58const PHASE: &EnumProperty<SculkSensorPhase> = &BlockStateProperties::SCULK_SENSOR_PHASE;
59const POWER: &IntProperty = &BlockStateProperties::POWER;
60const WATERLOGGED: &BoolProperty = &BlockStateProperties::WATERLOGGED;
61const FACING: &EnumProperty<Direction> = &BlockStateProperties::HORIZONTAL_FACING;

63/// Vanilla `SculkSensorBlock.getPhase`.
64#[must_use]
65pub fn sculk_sensor_phase(state: BlockStateId) -> SculkSensorPhase {
66    state.get_value(PHASE)
67}

69/// Vanilla `SculkSensorBlock.canActivate`.
70#[must_use]
71pub fn can_activate_sculk_sensor(state: BlockStateId) -> bool {
72    sculk_sensor_phase(state) == SculkSensorPhase::Inactive
73}

75/// Vanilla `SculkSensorBlock.updateNeighbours`.
76///
77/// The sensor updates the block under it as well as its own neighbors, which is what lets
78/// a sensor sitting on a redstone component drive it directly.
79fn update_neighbors(world: &Arc<World>, pos: BlockPos, state: BlockStateId) {
80    let block = state.get_block();
81    world.update_neighbors_at(pos, block);
82    world.update_neighbors_at(pos.below(), block);
83}

85/// Vanilla `SculkSensorBlock.deactivate`.
86pub fn deactivate_sculk_sensor(world: &Arc<World>, pos: BlockPos, state: BlockStateId) {
87    world.set_block(
88        pos,
89        state
90            .set_value(PHASE, SculkSensorPhase::Cooldown)
91            .set_value(POWER, 0),
92        UpdateFlags::UPDATE_ALL,
93    );
94    world.schedule_block_tick_default(pos, state.get_block(), COOLDOWN_TICKS);
95    update_neighbors(world, pos, state);
96}

98/// Vanilla `SculkSensorBlock.RESONANCE_PITCH_BEND`, kept as the notes it bends to.
99///
100/// Vanilla derives the sixteen pitches from this tone map through
101/// `NoteBlock.getPitchFromNote`; keeping the notes and converting on demand avoids a
102/// lazily-initialized float table for sixteen values.
103const RESONANCE_TONES: [u8; 16] = [0, 0, 2, 4, 6, 7, 9, 10, 12, 14, 15, 18, 19, 21, 22, 24];
104
105/// Vanilla `SculkSensorBlock.tryResonateVibration`.
106///
107/// Amethyst touching an activating sensor re-emits the vibration at its own frequency,
108/// which is how a redstone build filters one sound out of many.
109pub fn try_resonate_vibration(
110    source_entity: Option<&dyn Entity>,
111    world: &Arc<World>,
112    pos: BlockPos,
113    vibration_frequency: i32,
114) {
115    let Ok(tone_index) = usize::try_from(vibration_frequency) else {
116        return;
117    };
118    let (Some(tone), Some(event)) = (
119        RESONANCE_TONES.get(tone_index),
120        resonance_event_by_frequency(vibration_frequency),
121    ) else {
122        return;
123    };
124
125    for direction in Direction::ALL {
126        let relative_pos = pos.relative(direction);
127        let block_state = world.get_block_state(relative_pos);
128        if !REGISTRY
129            .blocks
130            .is_in_tag(block_state.get_block(), &BlockTag::VIBRATION_RESONATORS)
131        {
132            continue;
133        }
134
135        world.game_event(
136            event,
137            relative_pos,
138            &GameEventContext::new(source_entity, Some(block_state)),
139        );
140        world.play_block_sound(
141            &sound_events::BLOCK_AMETHYST_BLOCK_RESONATE,
142            relative_pos,
143            1.0,
144            NoteBlock::pitch_from_note(*tone),
145            None,
146        );
147    }
148}

150/// Vanilla `SculkSensorBlock.activate`, reached from the block entity's vibration user.
151///
152/// Vanilla dispatches on the block instance the sensor's state names. The two sensors differ
153/// only in how long they stay active, so the state selects that here instead of a downcast
154/// back to the behavior.
155pub fn activate_sculk_sensor(
156    source_entity: Option<&dyn Entity>,
157    world: &Arc<World>,
158    pos: BlockPos,
159    state: BlockStateId,
160    calculated_power: u8,
161    vibration_frequency: i32,
162) {
163    let active_ticks = if state.get_block() == &vanilla_blocks::CALIBRATED_SCULK_SENSOR {
164        CALIBRATED_ACTIVE_TICKS
165    } else {
166        ACTIVE_TICKS
167    };
168    activate(
169        source_entity,
170        world,
171        pos,
172        state,
173        calculated_power,
174        vibration_frequency,
175        active_ticks,
176    );
177}

179/// Vanilla `SculkSensorBlock.activate`.
180fn activate(
181    source_entity: Option<&dyn Entity>,
182    world: &Arc<World>,
183    pos: BlockPos,
184    state: BlockStateId,
185    calculated_power: u8,
186    vibration_frequency: i32,
187    active_ticks: i32,
188) {
189    world.set_block(
190        pos,
191        state
192            .set_value(PHASE, SculkSensorPhase::Active)
193            .set_value(POWER, calculated_power),
194        UpdateFlags::UPDATE_ALL,
195    );
196    world.schedule_block_tick_default(pos, state.get_block(), active_ticks);
197    update_neighbors(world, pos, state);
198    try_resonate_vibration(source_entity, world, pos, vibration_frequency);
199    world.game_event(
200        &vanilla_game_events::SCULK_SENSOR_TENDRILS_CLICKING,
201        pos,
202        &GameEventContext::new(source_entity, None),
203    );
204    if !state.get_value(WATERLOGGED) {
205        world.play_block_sound(
206            &sound_events::BLOCK_SCULK_SENSOR_CLICKING,
207            pos,
208            1.0,
209            rand::random::<f32>().mul_add(0.2, 0.8),
210            None,
211        );
212    }
213}

215/// The half of the two sculk sensors that does not depend on the calibration face.
216///
217/// Rust has no `extends`, so vanilla's `CalibratedSculkSensorBlock extends SculkSensorBlock`
218/// becomes shared state both behaviors delegate to. `active_ticks` is the one value the
219/// subclass overrides.
220struct SculkSensorCore {
221    block: BlockRef,
222    active_ticks: i32,
223    calibrated: bool,
224}
225
226impl SculkSensorCore {
227    const fn new(block: BlockRef, active_ticks: i32, calibrated: bool) -> Self {
228        Self {
229            block,
230            active_ticks,
231            calibrated,
232        }
233    }
234
235    /// Vanilla `SculkSensorBlock.tick`.
236    ///
237    /// The phase machine is identical for both sensors, so this needs nothing off `self`.
238    fn tick(state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
239        match sculk_sensor_phase(state) {
240            SculkSensorPhase::Active => deactivate_sculk_sensor(world, pos, state),
241            SculkSensorPhase::Cooldown => {
242                world.set_block(
243                    pos,
244                    state.set_value(PHASE, SculkSensorPhase::Inactive),
245                    UpdateFlags::UPDATE_ALL,
246                );
247                if !state.get_value(WATERLOGGED) {
248                    world.play_block_sound(
249                        &sound_events::BLOCK_SCULK_SENSOR_CLICKING_STOP,
250                        pos,
251                        1.0,
252                        rand::random::<f32>().mul_add(0.2, 0.8),
253                        None,
254                    );
255                }
256            }
257            SculkSensorPhase::Inactive => {}
258        }
259    }
260
261    /// Vanilla `SculkSensorBlock.activate`.
262    fn activate(
263        &self,
264        source_entity: Option<&dyn Entity>,
265        world: &Arc<World>,
266        pos: BlockPos,
267        state: BlockStateId,
268        calculated_power: u8,
269        vibration_frequency: i32,
270    ) {
271        activate(
272            source_entity,
273            world,
274            pos,
275            state,
276            calculated_power,
277            vibration_frequency,
278            self.active_ticks,
279        );
280    }
281
282    /// Vanilla `SculkSensorBlock.onPlace`.
283    ///
284    /// A sensor pasted in powered -- by a structure, a command or a schematic -- has no
285    /// vibration behind that power, so vanilla clears it unless a tick is already pending.
286    fn on_place(
287        &self,
288        state: BlockStateId,
289        world: &Arc<World>,
290        pos: BlockPos,
291        old_state: BlockStateId,
292    ) {
293        if state.get_block() == old_state.get_block()
294            || state.get_value(POWER) == 0
295            || world.has_scheduled_block_tick(pos, self.block)
296        {
297            return;
298        }
299
300        world.set_block(pos, state.set_value(POWER, 0), PLACEMENT_POWER_RESET_FLAGS);
301    }
302
303    /// Vanilla `SculkSensorBlock.getStateForPlacement`.
304    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> BlockStateId {
305        let waterlogged = get_fluid_state(context.world, context.place_pos()).is_water();
306        let state = self
307            .block
308            .default_state()
309            .set_value(WATERLOGGED, waterlogged);
310        if self.calibrated {
311            // Vanilla `CalibratedSculkSensorBlock.getStateForPlacement` faces the player.
312            return state.set_value(FACING, context.horizontal_direction());
313        }
314        state
315    }
316
317    fn new_block_entity(
318        &self,
319        level: Weak<World>,
320        pos: BlockPos,
321        state: BlockStateId,
322    ) -> BlockEntityCreation {
323        let block_entity_type = if self.calibrated {
324            &vanilla_block_entity_types::CALIBRATED_SCULK_SENSOR
325        } else {
326            &vanilla_block_entity_types::SCULK_SENSOR
327        };
328        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
329            block_entity_type,
330            level,
331            pos,
332            state,
333        ))
334    }
335
336    /// Vanilla `SculkSensorBlock.getTicker`, which only ticks the vibration system.
337    fn block_entity_ticker(
338        &self,
339        block_entity_type: BlockEntityTypeRef,
340    ) -> Option<BlockEntityTicker> {
341        let expected = if self.calibrated {
342            &vanilla_block_entity_types::CALIBRATED_SCULK_SENSOR
343        } else {
344            &vanilla_block_entity_types::SCULK_SENSOR
345        };
346        BlockEntityTicker::for_matching_entity_tick(block_entity_type, expected)
347    }
348
349    /// Vanilla `SculkSensorBlock.getAnalogOutputSignal`.
350    fn analog_output_signal(state: BlockStateId, world: &dyn LevelReader, pos: BlockPos) -> i32 {
351        if sculk_sensor_phase(state) != SculkSensorPhase::Active {
352            return 0;
353        }
354
355        world
356            .get_block_entity(pos)
357            .and_then(|block_entity| {
358                block_entity
359                    .downcast_ref::<SculkSensorBlockEntity>()
360                    .map(SculkSensorBlockEntity::last_vibration_frequency)
361            })
362            .unwrap_or(0)
363    }
364}

753    /// Puts `state` in a real world without letting `onPlace` see a block change.
754    ///
755    /// The sensor clears stray power the moment it is placed, so a test that wants a
756    /// pre-powered sensor has to place the plain one first and then swap the state.
757    fn world_with_sensor(name: &'static str, pos: BlockPos, state: BlockStateId) -> Arc<World> {
758        let world = fresh_test_world(name);
759        insert_ready_full_chunk(&world, ChunkPos::from_block_pos(pos));
760        assert!(world.set_block(
761            pos,
762            state.get_block().default_state(),
763            UpdateFlags::UPDATE_ALL
764        ));
765        // The second write is a no-op when `state` already is the default, and
766        // `set_block` reports that as "nothing changed" rather than as a failure.
767        let _ = world.set_block(pos, state, UpdateFlags::UPDATE_ALL);
768        assert_eq!(world.get_block_state(pos), state);
769        world
770    }

779    /// The whole point of the cooldown phase is that a sensor cannot fire twice in a
780    /// row: it drops its power the moment the active window ends, and only ten ticks
781    /// later can it hear anything again.
782    #[test]
783    fn an_active_sensor_drops_its_power_before_it_can_listen_again() {
784        init();
785        let pos = BlockPos::new(8, 70, 8);
786        let active = sensor_state(SculkSensorPhase::Active, 9);
787        let world = world_with_sensor("sculk_sensor_cooldown", pos, active);
788        let behavior = SculkSensorBlock::new(&vanilla_blocks::SCULK_SENSOR);
789
790        behavior.tick(active, &world, pos);
791
792        let cooling = world.get_block_state(pos);
793        assert_eq!(sculk_sensor_phase(cooling), SculkSensorPhase::Cooldown);
794        assert_eq!(cooling.get_value(POWER), 0);
795        assert!(!can_activate_sculk_sensor(cooling));
796        assert!(world.has_scheduled_block_tick(pos, &vanilla_blocks::SCULK_SENSOR));
797
798        behavior.tick(cooling, &world, pos);
799
800        let idle = world.get_block_state(pos);
801        assert_eq!(sculk_sensor_phase(idle), SculkSensorPhase::Inactive);
802        assert!(can_activate_sculk_sensor(idle));
803    }
