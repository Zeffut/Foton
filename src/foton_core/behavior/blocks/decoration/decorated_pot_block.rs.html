1//! Decorated pot block behavior.
2//!
3//! Vanilla parity: `DecoratedPotBlock`. A pot's whole identity -- the four
4//! sherds pressed into its sides -- lives on a block entity and has to survive
5//! the round trip through an item, and it is one of the few blocks that drops
6//! something different depending on how it was broken: hit it with a pickaxe
7//! and it shatters back into the four pieces it was made of, take it with silk
8//! touch and the pot comes back whole.
9//!
10//! Not ported here: `getCloneItemStack`, because Foton's signature has neither
11//! world nor position and so cannot reach the block entity the sherds live on;
12//! the `Stats.ITEM_USED` award of `useItemOn`, because Foton has no statistics
13//! foundation; and `onProjectileHit`, which cracks a pot an arrow hits.
14

51/// Which way the decorated front faces.
52const HORIZONTAL_FACING: &EnumProperty<Direction> = &BlockStateProperties::HORIZONTAL_FACING;
53/// Whether the pot has been hit hard enough to come apart when it breaks.
54const CRACKED: &BoolProperty = &BlockStateProperties::CRACKED;
55/// Whether the pot is standing in water.
56const WATERLOGGED: &BoolProperty = &BlockStateProperties::WATERLOGGED;
57
58/// How many items one use puts into the pot.
59///
60/// Vanilla parity: the literal `1` of `DecoratedPotBlock.useItemOn`.
61const ITEMS_PER_INSERT: i32 = 1;
62/// Volume of the insert and refuse sounds.
63const SOUND_VOLUME: f32 = 1.0;
64/// Pitch of the refuse sound.
65const REFUSE_PITCH: f32 = 1.0;
66/// Base pitch of the insert sound, before the fullness bend.
67///
68/// Vanilla parity: the `0.7F + 0.5F * pitchBend` of `DecoratedPotBlock.useItemOn`.
69const INSERT_BASE_PITCH: f32 = 0.7;
70/// How far the insert sound's pitch rises as the pot fills.
71const INSERT_PITCH_BEND: f32 = 0.5;
72/// How many dust particles puff out of the pot's mouth.
73const DUST_PLUME_COUNT: i32 = 7;
74/// Height above the block origin the dust plume comes out at.
75const DUST_PLUME_HEIGHT: f64 = 1.2;
76/// Horizontal center of the pot's mouth.
77const BLOCK_CENTER: f64 = 0.5;

79/// Behavior for the decorated pot.
80#[block_behavior]
81pub struct DecoratedPotBlock {
82    block: BlockRef,
83}

85impl DecoratedPotBlock {
86    /// Creates a decorated pot block behavior.
87    #[must_use]
88    pub const fn new(block: BlockRef) -> Self {
89        Self { block }
90    }
91
92    /// Returns whether the tool takes the pot apart rather than picking it up.
93    ///
94    /// Vanilla parity: the condition of `DecoratedPotBlock.playerWillDestroy`
95    /// -- `ItemTags.BREAKS_DECORATED_POTS`, which is the swords, pickaxes and
96    /// the rest of the tools, minus anything carrying
97    /// `EnchantmentTags.PREVENTS_DECORATED_POT_SHATTERING`, which is silk
98    /// touch.
99    fn shatters_the_pot(tool: &ItemStack) -> bool {
100        tool.item().has_tag(&ItemTag::BREAKS_DECORATED_POTS) && !Self::prevents_shattering(tool)
101    }
102
103    /// Returns whether the tool's enchantments keep the pot in one piece.
104    ///
105    /// Vanilla parity: `EnchantmentHelper.hasTag(stack,
106    /// EnchantmentTags.PREVENTS_DECORATED_POT_SHATTERING)`.
107    fn prevents_shattering(tool: &ItemStack) -> bool {
108        let Some(enchantments) = tool.get_enchantments() else {
109            return false;
110        };
111        enchantments.iter().any(|(key, _)| {
112            REGISTRY
113                .enchantments
114                .by_key(key)
115                .is_some_and(|enchantment| {
116                    REGISTRY.enchantments.is_in_tag(
117                        enchantment,
118                        &EnchantmentTag::PREVENTS_DECORATED_POT_SHATTERING,
119                    )
120                })
121        })
122    }
123
124    /// Returns whether the pot accepts what is being offered to it.
125    ///
126    /// Vanilla parity: the condition of `DecoratedPotBlock.useItemOn`. A pot
127    /// takes anything while it is empty and only more of the same after that,
128    /// which is what makes it a one-item filter rather than a small chest.
129    fn accepts(stored: &ItemStack, offered: &ItemStack) -> bool {
130        if offered.is_empty() {
131            return false;
132        }
133        stored.is_empty()
134            || (ItemStack::is_same_item_same_components(stored, offered)
135                && stored.count() < stored.max_stack_size())
136    }
137}

140    /// Vanilla parity: `DecoratedPotBlock.getStateForPlacement`. The decorated
141    /// front turns towards the player, unlike most blocks, which face away.
142    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
143        Some(
144            self.block
145                .default_state()
146                .set_value(HORIZONTAL_FACING, context.horizontal_direction())

192    /// Presses the item's sherds into the pot that was just placed.
193    ///
194    /// Vanilla parity: the `applyImplicitComponents` of
195    /// `DecoratedPotBlockEntity`. Vanilla assigns unconditionally; an all-brick
196    /// pot is skipped here because that is already what a fresh block entity
197    /// holds, and writing it back would mark the chunk dirty and sweep the
198    /// comparators for nothing every time any pot is placed.
199    fn set_placed_by(
200        &self,
201        _state: BlockStateId,
202        world: &Arc<World>,
203        pos: BlockPos,
204        source: &PlacementSource<'_>,
205    ) {
206        let Some(decorations) = source.with_item(|item| item.get(POT_DECORATIONS).cloned()) else {
207            return;
208        };
209        if decorations == PotDecorations::EMPTY {
210            return;
211        }
212        let Some(block_entity) = world.get_block_entity(pos) else {
213            return;
214        };
215        let Some(pot) = block_entity.downcast_ref::<DecoratedPotBlockEntity>() else {
216            return;
217        };
218        pot.set_decorations(decorations);
219    }

221    /// Cracks the pot the player is about to break with the wrong tool.
222    ///
223    /// Vanilla parity: `DecoratedPotBlock.playerWillDestroy`. The state this
224    /// returns is what the loot is resolved from, which is how one block ends
225    /// up with two completely different drops.
226    fn player_will_destroy(
227        &self,
228        state: BlockStateId,
229        world: &Arc<World>,
230        pos: BlockPos,
231        player: &Player,
232    ) -> BlockStateId {
233        let shatters = {
234            let inventory = player.inventory.lock();
235            Self::shatters_the_pot(inventory.get_item_in_hand(InteractionHand::MainHand))
236        };
237        if !shatters {
238            return state;
239        }
240
241        let cracked = state.set_value(CRACKED, true);
242        // Vanilla's flag 260 is Foton's `UPDATE_NONE`: the pot is about to be
243        // removed anyway, so neither the neighbors nor the client need to be
244        // told about a crack that lasts one call.
245        world.set_block(pos, cracked, UpdateFlags::UPDATE_NONE);
246        cracked
247    }
248
249    /// Returns either the pot or the four pieces it was made of.
250    ///
251    /// Vanilla parity: `DecoratedPotBlock.getDrops` together with the
252    /// `decorated_pot` loot table it feeds, whose two alternatives are selected
253    /// by the `cracked` property -- the `sherds` dynamic drop when it is set,
254    /// and the pot with `copy_components` of `pot_decorations` when it is not.
255    ///
256    /// Foton has no loot tables, so both alternatives are resolved here. The
257    /// choice still comes from the block state exactly as in vanilla:
258    /// [`Self::player_will_destroy`] is the only thing that sets `cracked`, so
259    /// a pot destroyed by something other than a player -- an explosion, a
260    /// piston -- drops whole, which is what vanilla does too.
261    fn get_drops(
262        &self,
263        state: BlockStateId,
264        context: &BlockLootContext<'_>,
265    ) -> Option<Vec<ItemStack>> {
266        let decorations =
267            context
268                .world()
269                .get_block_entity(context.pos())
270                .and_then(|block_entity| {
271                    block_entity
272                        .downcast_ref::<DecoratedPotBlockEntity>()
273                        .map(DecoratedPotBlockEntity::decorations)
274                });
275
276        if state.get_value(CRACKED) {
277            // Vanilla parity: `PotDecorations.ordered`, which fills every
278            // undecorated side back in with the brick it was made from, so a
279            // shattered pot gives back exactly four items -- and nothing at all
280            // with no block entity to ask, because vanilla's `sherds` dynamic
281            // drop then has nobody to supply it.
282            return Some(decorations.map_or_else(Vec::new, |decorations| {
283                decorations
284                    .ordered()
285                    .into_iter()
286                    .map(ItemStack::new)
287                    .collect()
288            }));
289        }
290
291        let mut pot = ItemStack::new(REGISTRY.items.by_key(&self.block.key)?);
292        // Vanilla's `copy_components` copies nothing when the block entity is
293        // missing, which leaves the item's own empty decorations in place.
294        pot.set(
295            POT_DECORATIONS,
296            decorations.unwrap_or(PotDecorations::EMPTY),
297        );
298        Some(vec![pot])
299    }

301    /// Vanilla parity: `DecoratedPotBlock.affectNeighborsAfterRemoval`, whose
302    /// one line is the `Containers` after-destroy neighbor update -- a
303    /// comparator reading the pot has to notice that it is gone.
304    fn affect_neighbors_after_removal(
305        &self,
306        state: BlockStateId,
307        world: &Arc<World>,
308        pos: BlockPos,
309        _moved_by_piston: bool,
310    ) {
311        world.update_neighbor_for_output_signal(pos, state.get_block());
312    }

314    /// Puts one item into the pot.
315    ///
316    /// Vanilla parity: `DecoratedPotBlock.useItemOn`, minus its client branch,
317    /// which Foton has no counterpart for.
318    fn use_item_on(
319        &self,
320        _state: BlockStateId,
321        world: &Arc<World>,
322        pos: BlockPos,
323        player: &Player,
324        _hand: InteractionHand,
325        _hit_result: &BlockHitResult,
326        inv: &mut InventoryAccess,
327    ) -> InteractionResult {
328        let Some(block_entity) = world.get_block_entity(pos) else {
329            return InteractionResult::Pass;
330        };
331        let Some(pot) = block_entity.downcast_ref::<DecoratedPotBlockEntity>() else {
332            return InteractionResult::Pass;
333        };
334
335        let stored = pot.the_item();
336        if !inv.with_item(|offered| Self::accepts(&stored, offered)) {
337            return InteractionResult::TryEmptyHandInteraction;
338        }
339
340        pot.wobble(world, WobbleStyle::Positive);
341
342        // Vanilla parity: `ItemStack.consumeAndReturn(1, player)`, which hands
343        // back one item and only charges a player who is not in creative.
344        let keeps_the_stack = player.has_infinite_materials();
345        let taken = inv.with_item(|offered| {
346            let taken = offered.copy_with_count(ITEMS_PER_INSERT);
347            if !keeps_the_stack {
348                offered.shrink(ITEMS_PER_INSERT);
349            }
350            taken
351        });
352
353        let inserted = if stored.is_empty() {
354            taken
355        } else {
356            let mut grown = stored;
357            grown.grow(ITEMS_PER_INSERT);
358            grown
359        };
360        // Vanilla parity: the pitch rises with how full the pot is, which is
361        // how a player hears a pot filling without being able to look inside.
362        let fullness = inserted.count() as f32 / inserted.max_stack_size() as f32;
363        pot.set_the_item(inserted);
364
365        world.play_block_sound(
366            &sound_events::BLOCK_DECORATED_POT_INSERT,
367            pos,
368            SOUND_VOLUME,
369            INSERT_BASE_PITCH + INSERT_PITCH_BEND * fullness,
370            None,
371        );
372        world.send_particles(
373            ParticleData::simple(&vanilla_particle_types::DUST_PLUME),
374            DVec3::new(
375                f64::from(pos.x()) + BLOCK_CENTER,
376                f64::from(pos.y()) + DUST_PLUME_HEIGHT,
377                f64::from(pos.z()) + BLOCK_CENTER,
378            ),
379            DUST_PLUME_COUNT,
380            DVec3::ZERO,
381            0.0,
382        );
383        world.game_event(
384            &vanilla_game_events::BLOCK_CHANGE,
385            pos,
386            &GameEventContext::new(Some(player), None),
387        );
388        InteractionResult::Success
389    }

391    /// Vanilla parity: `DecoratedPotBlock.useWithoutItem`. An empty hand takes
392    /// nothing back out of a pot; it only rocks and thuds, which is the game
393    /// saying that the way back in is to break it.
394    fn use_without_item(
395        &self,
396        _state: BlockStateId,
397        world: &Arc<World>,
398        pos: BlockPos,
399        player: &Player,
400        _hit_result: &BlockHitResult,
401        _inv: &mut InventoryAccess,
402    ) -> InteractionResult {
403        let Some(block_entity) = world.get_block_entity(pos) else {
404            return InteractionResult::Pass;
405        };
406        let Some(pot) = block_entity.downcast_ref::<DecoratedPotBlockEntity>() else {
407            return InteractionResult::Pass;
408        };
409
410        world.play_block_sound(
411            &sound_events::BLOCK_DECORATED_POT_INSERT_FAIL,
412            pos,
413            SOUND_VOLUME,
414            REFUSE_PITCH,
415            None,
416        );
417        pot.wobble(world, WobbleStyle::Negative);
418        world.game_event(
419            &vanilla_game_events::BLOCK_CHANGE,
420            pos,
421            &GameEventContext::new(Some(player), None),
422        );
423        InteractionResult::Success
424    }

427    /// the block entity. Returning `true` is what sends the wobble to clients.

430        _state: BlockStateId,
431        world: &Arc<World>,
432        pos: BlockPos,

434        param_b: i32,

439    }

445    /// Vanilla parity: `AbstractContainerMenu.getRedstoneSignalFromBlockEntity`.
446    /// One slot means a single stack fills the pot, so a comparator behind one
447    /// is a counter for whatever a hopper above it is feeding in.
448    fn get_analog_output_signal(
449        &self,
450        _state: BlockStateId,
451        world: &dyn LevelReader,
452        pos: BlockPos,
453        _direction: Direction,
454    ) -> i32 {
455        let Some(container_ref) = world
456            .get_block_entity(pos)
457            .and_then(ContainerRef::from_block_entity)
458        else {
459            return 0;
460        };
461        let guard = ContainerLockGuard::lock_all(&[&container_ref]);
462        guard
463            .get(container_ref.container_id())
464            .map_or(0, calculate_redstone_signal_from_container)
465    }

537    /// The whole round trip: a pot crafted from sherds keeps them when it is
538    /// put down and hands them back when it is picked up again. Losing them on
539    /// the way down would make archaeology pointless; losing them on the way
540    /// out would delete work that cost a brush and a block of suspicious sand.
541    #[test]
542    fn the_sherds_survive_being_placed_and_broken() {
543        let world = placed_pot("decorated_pot_round_trip");
544        let behavior = DecoratedPotBlock::new(&vanilla_blocks::DECORATED_POT);
545        place_with(&behavior, &world, &mut pot_item_with_sherds());
546
547        let drops = behavior
548            .get_drops(
549                vanilla_blocks::DECORATED_POT.default_state(),
550                &BlockLootContext::new(&world, POT_POS),
551            )
552            .expect("a decorated pot overrides its own loot");
553
554        assert_eq!(drops.len(), 1, "one pot, not a pile of pieces");
555        let dropped = drops.into_iter().next().expect("the pot");
556        assert!(dropped.is(&vanilla_items::DECORATED_POT));
557        assert_eq!(
558            dropped.get(POT_DECORATIONS),
559            Some(&two_sherds()),
560            "the dropped pot lost its sherds"
561        );
562    }

613    /// Which of the two drops happens is decided by the tool, and silk touch is

617    fn a_pickaxe_cracks_the_pot_and_silk_touch_does_not() {
618        init_vanilla_registry();
619

642    /// The pot's one slot is what a comparator behind it measures, so a full
643    /// stack has to read fifteen and an empty pot nothing. Having exactly one
644    /// slot is what makes a decorated pot a usable item counter.
645    #[test]
646    fn a_comparator_reads_the_single_slot() {
647        let world = placed_pot("decorated_pot_comparator");
648        let behavior = DecoratedPotBlock::new(&vanilla_blocks::DECORATED_POT);
649        let state = vanilla_blocks::DECORATED_POT.default_state();
650        let read =
651            || behavior.get_analog_output_signal(state, world.as_ref(), POT_POS, Direction::North);
652
653        assert_eq!(read(), 0, "an empty pot powers nothing");
654
655        let block_entity = world
656            .get_block_entity(POT_POS)
657            .expect("a placed pot has a block entity");
658        let pot = block_entity
659            .downcast_ref::<DecoratedPotBlockEntity>()
660            .expect("it is a decorated pot");
661
662        pot.set_the_item(ItemStack::new(&vanilla_items::DIAMOND));
663        assert_eq!(read(), 1, "one item is still a signal");
664
665        pot.set_the_item(ItemStack::with_count(&vanilla_items::DIAMOND, 64));
666        assert_eq!(read(), 15, "a full stack fills the only slot");
667    }
