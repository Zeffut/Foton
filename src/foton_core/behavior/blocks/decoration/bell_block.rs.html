1//! Bell behavior.
2//!
3//! Vanilla parity: `BellBlock`. Hitting a bell rings it, but only from a side
4//! it can actually swing towards -- a floor bell hangs on an axis and will not
5//! ring if struck end-on. Redstone rings it too, once per rising edge.
6//!
7//! Ringing is only the start of it. The swing, the sixty-block sweep that tells
8//! every mob nearby the bell rang, and the resonance that gives away the
9//! raiders standing in it all live on [`BellBlockEntity`]; this block is the
10//! part that decides whether a hit counts and hands the ring over to it.
11

37/// How the bell is hung.
38const ATTACHMENT: &EnumProperty<BellAttachType> = &BlockStateProperties::BELL_ATTACHMENT;
39
40/// Which way it faces.
41const FACING: &EnumProperty<Direction> = &BlockStateProperties::HORIZONTAL_FACING;
42
43/// Whether redstone is currently holding it.
44const POWERED: &BoolProperty = &BlockStateProperties::POWERED;
45
46/// How loud a bell is.
47///
48/// Vanilla parity: the `2.0F` of `attemptToRing` -- twice as loud as most
49/// blocks, which is the point of a bell.
50const BELL_VOLUME: f32 = 2.0;
51
52/// Above this height on the block, a side hit counts as hitting the top.
53///
54/// Vanilla parity: the `0.8124F` of `isProperHit`. Striking the very top of a
55/// bell does not swing it.
56const TOP_HIT_HEIGHT: f64 = 0.812_4;
57
58/// Behavior for the bell.
59#[block_behavior]
60pub struct BellBlock {
61    block: BlockRef,
62}
63
64impl BellBlock {
65    /// Creates a bell behavior.
66    #[must_use]
67    pub const fn new(block: BlockRef) -> Self {
68        Self { block }
69    }
70
71    /// Returns whether a hit from `direction` at `hit_height` can swing it.
72    ///
73    /// Vanilla parity: `BellBlock.isProperHit`. A bell swings on one axis, and
74    /// which axis depends on how it is hung: a floor bell swings along its
75    /// facing, a wall bell across it.
76    fn is_proper_hit(state: BlockStateId, direction: Direction, hit_height: f64) -> bool {
77        if direction.axis() == Axis::Y || hit_height > TOP_HIT_HEIGHT {
78            return false;
79        }
80
81        let facing = state.get_value(FACING);
82        match state.get_value(ATTACHMENT) {
83            BellAttachType::Floor => facing.axis() == direction.axis(),
84            BellAttachType::SingleWall | BellAttachType::DoubleWall => {
85                facing.axis() != direction.axis()
86            }
87            BellAttachType::Ceiling => true,
88        }
89    }
90
91    /// Rings the bell at `pos`, if there is one there to ring.
92    ///
93    /// Vanilla parity: `BellBlock.attemptToRing`, both overloads -- `direction`
94    /// of `None` is the `@Nullable Direction` it falls back to the bell's own
95    /// facing for. It is an associated function because vanilla's is an
96    /// instance method that never touches the instance: everything it needs it
97    /// reads back out of the level.
98    ///
99    /// Returns whether there was a bell block entity to ring, which is what
100    /// tells a player whether the swing earned them the statistic.
101    pub fn attempt_to_ring(
102        world: &Arc<World>,
103        pos: BlockPos,
104        direction: Option<Direction>,
105        ringer: Option<&dyn Entity>,
106    ) -> bool {
107        let Some(block_entity) = world.get_block_entity(pos) else {
108            return false;
109        };
110        let Some(bell) = block_entity.downcast_ref::<BellBlockEntity>() else {
111            return false;
112        };
113        let direction = direction.unwrap_or_else(|| world.get_block_state(pos).get_value(FACING));
114        bell.on_hit(world, direction);
115        world.play_block_sound(&sound_events::BLOCK_BELL_USE, pos, BELL_VOLUME, 1.0, None);
116        world.game_event(
117            &vanilla_game_events::BLOCK_CHANGE,
118            pos,
119            &GameEventContext::new(ringer, None),
120        );
121        true
122    }
123}
124
125impl BlockBehavior for BellBlock {
126    /// Vanilla parity: `BellBlock.getStateForPlacement`.
127    ///
128    /// A bell hung between two walls reads as `DoubleWall` and swings the
129    /// other way from one hung on a single wall.
130    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
131        let clicked_face = context.clicked_face();
132        let pos = context.place_pos();
133        let world = context.world;
134
135        let (attachment, facing) = match clicked_face {
136            Direction::Up => (BellAttachType::Floor, context.horizontal_direction()),
137            Direction::Down => (BellAttachType::Ceiling, context.horizontal_direction()),
138            side => {
139                let opposite = side.opposite();
140                let has_other_wall = world
141                    .get_block_state(pos.relative(side))
142                    .is_face_sturdy_for_at(pos.relative(side), opposite, SupportType::Full);
143                let attachment = if has_other_wall {
144                    BellAttachType::DoubleWall
145                } else {
146                    BellAttachType::SingleWall
147                };
148                (attachment, opposite)
149            }
150        };
151
152        Some(
153            self.block
154                .default_state()
155                .set_value(ATTACHMENT, attachment)
156                .set_value(FACING, facing),
157        )
158    }
159
160    /// Vanilla parity: `BellBlock.useWithoutItem`.
161    fn use_without_item(
162        &self,
163        state: BlockStateId,
164        world: &Arc<World>,
165        pos: BlockPos,
166        player: &Player,
167        hit_result: &BlockHitResult,
168        _inv: &mut InventoryAccess,
169    ) -> InteractionResult {
170        let hit_height = hit_result.location.y - f64::from(pos.y());
171        if !Self::is_proper_hit(state, hit_result.direction, hit_height) {
172            return InteractionResult::Pass;
173        }
174
175        Self::attempt_to_ring(world, pos, Some(hit_result.direction), Some(player));
176        // TODO: Award stat BELL_RING; Foton has no statistics registry.
177        InteractionResult::Success
178    }
179
180    /// Vanilla parity: `BellBlock.neighborChanged`, which rings once on the
181    /// rising edge rather than continuously while powered.
182    fn handle_neighbor_changed(
183        &self,
184        state: BlockStateId,
185        world: &Arc<World>,
186        pos: BlockPos,
187        _source_block: BlockRef,
188        _moved_by_piston: bool,
189    ) {
190        let signal = world.has_neighbor_signal(pos);
191        if signal == state.get_value(POWERED) {
192            return;
193        }
194        if signal {
195            Self::attempt_to_ring(world, pos, None, None);
196        }
197        world.set_block(
198            pos,
199            state.set_value(POWERED, signal),
200            UpdateFlags::UPDATE_ALL,
201        );
202    }
203
204    /// Vanilla parity: `BellBlock.newBlockEntity`.
205    fn new_block_entity(
206        &self,
207        level: Weak<World>,
208        pos: BlockPos,
209        state: BlockStateId,
210    ) -> BlockEntityCreation {
211        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
212            &vanilla_block_entity_types::BELL,
213            level,
214            pos,
215            state,
216        ))
217    }
218
219    /// Vanilla parity: the server half of `BellBlock.getTicker`.
220    fn get_block_entity_ticker(
221        &self,
222        _world: &Arc<World>,
223        _state: BlockStateId,
224        block_entity_type: BlockEntityTypeRef,
225    ) -> Option<BlockEntityTicker> {
226        BlockEntityTicker::for_matching_entity_tick(
227            block_entity_type,
228            &vanilla_block_entity_types::BELL,
229        )
230    }
231
232    /// Vanilla parity: `BaseEntityBlock.triggerEvent`, which forwards the event
233    /// to the block entity -- the swing counter and the raider sweep both start
234    /// from there rather than from the block.
235    fn trigger_event(
236        &self,
237        _state: BlockStateId,
238        world: &Arc<World>,
239        pos: BlockPos,
240        param_a: i32,
241        param_b: i32,
242    ) -> bool {
243        world
244            .get_block_entity(pos)
245            .is_some_and(|block_entity| block_entity.trigger_event(param_a, param_b))
246    }
