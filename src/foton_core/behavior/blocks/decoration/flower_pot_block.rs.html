1//! Flower pot behavior.
2//!
3//! Vanilla parity: `FlowerPotBlock`. Thirty-nine blocks: one empty pot and
4//! thirty-eight already holding something. Putting a plant in swaps the empty
5//! pot for the matching full one, and taking it out swaps back -- there is no
6//! block entity anywhere, which is why the potted variants exist as separate
7//! blocks at all.
8//!
9//! Which plant each pot holds comes from the extracted `potted`, and the
10//! reverse -- plant to pot -- is built once from the same data through
11//! [`BlockBehavior::potted_content`], the way vanilla's constructors fill
12//! `POTTED_BY_CONTENT`.
13

32/// Which pot holds which plant.
33///
34/// Vanilla builds this in `FlowerPotBlock`'s constructor; Foton builds it once
35/// from the behaviors, which carry the same extracted data.
36static POTTED_BY_CONTENT: LazyLock<FxHashMap<Identifier, BlockRef>> = LazyLock::new(|| {
37    REGISTRY
38        .blocks
39        .iter()
40        .filter_map(|(_, block)| {
41            let potted = BLOCK_BEHAVIORS.get_behavior(block).potted_content()?;
42            Some((potted.key.clone(), block))
43        })
44        .collect()
45});
46
47/// Behavior for a flower pot, empty or full.
48#[block_behavior]
49pub struct FlowerPotBlock {
50    block: BlockRef,
51    /// What this pot holds, or air for the empty one.
52    #[json_arg(vanilla_blocks, json = "potted")]
53    potted: BlockRef,
54}
55
56impl FlowerPotBlock {
57    /// Creates a flower pot behavior.
58    #[must_use]
59    pub const fn new(block: BlockRef, potted: BlockRef) -> Self {
60        Self { block, potted }
61    }
62
63    /// Returns whether this is the empty pot.
64    #[must_use]
65    fn is_empty(&self) -> bool {
66        self.potted == &vanilla_blocks::AIR
67    }
68}
69
70impl BlockBehavior for FlowerPotBlock {
71    fn get_state_for_placement(&self, _context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
72        Some(self.block.default_state())
73    }
74
75    /// Vanilla parity: `FlowerPotBlock.getPotted`, which is what lets the
76    /// reverse map be built.
77    fn potted_content(&self) -> Option<BlockRef> {
78        (!self.is_empty()).then_some(self.potted)
79    }
80
81    /// Puts a plant in the pot.
82    ///
83    /// Vanilla parity: `FlowerPotBlock.useItemOn`. A pot that already holds
84    /// something consumes the click rather than swapping, which is why you
85    /// cannot replace a plant without taking the first one out.
86    fn use_item_on(
87        &self,
88        _state: BlockStateId,
89        world: &Arc<World>,
90        pos: BlockPos,
91        player: &Player,
92        _hand: InteractionHand,
93        _hit_result: &BlockHitResult,
94        inv: &mut InventoryAccess,
95    ) -> InteractionResult {
96        let held_key = inv.with_item(|item| item.item().key.clone());
97        let Some(full_pot) = POTTED_BY_CONTENT.get(&held_key) else {
98            return InteractionResult::TryEmptyHandInteraction;
99        };
100
101        if !self.is_empty() {
102            return InteractionResult::Consume;
103        }
104
105        let planted = full_pot.default_state();
106        world.set_block(pos, planted, UpdateFlags::UPDATE_ALL);
107        world.game_event(
108            &vanilla_game_events::BLOCK_CHANGE,
109            pos,
110            &GameEventContext::new(Some(player), Some(planted)),
111        );
112
113        if !player.has_infinite_materials() {
114            inv.with_item(|item| item.shrink(1));
115        }
116
117        InteractionResult::Success
118    }
119
120    /// Takes the plant back out.
121    ///
122    /// Vanilla parity: `FlowerPotBlock.useWithoutItem`.
123    fn use_without_item(
124        &self,
125        _state: BlockStateId,
126        world: &Arc<World>,
127        pos: BlockPos,
128        player: &Player,
129        _hit_result: &BlockHitResult,
130        _inv: &mut InventoryAccess,
131    ) -> InteractionResult {
132        if self.is_empty() {
133            return InteractionResult::Consume;
134        }
135
136        let Some(plant_item) = REGISTRY.items.by_key(&self.potted.key) else {
137            return InteractionResult::Consume;
138        };
139        player.add_item_or_drop(ItemStack::new(plant_item));
140
141        let empty = vanilla_blocks::FLOWER_POT.default_state();
142        world.set_block(pos, empty, UpdateFlags::UPDATE_ALL);
143        world.game_event(
144            &vanilla_game_events::BLOCK_CHANGE,
145            pos,
146            &GameEventContext::new(Some(player), Some(empty)),
147        );
148
149        InteractionResult::Success
150    }
151
152    /// Vanilla parity: `FlowerPotBlock.getCloneItemStack`, which picks the
153    /// plant rather than the pot when the pot has something in it.
154    fn get_clone_item_stack(
155        &self,
156        block: BlockRef,
157        _state: BlockStateId,
158        _include_data: bool,
159    ) -> Option<ItemStack> {
160        if self.is_empty() {
161            return REGISTRY.items.by_key(&block.key).map(ItemStack::new);
162        }
163        REGISTRY.items.by_key(&self.potted.key).map(ItemStack::new)
164    }
165}

167#[cfg(test)]
168mod tests {
169    use foton_registry::init_vanilla_registry;
170
171    use super::*;
172    use crate::behavior::init_behaviors;
173
174    /// Every full pot knows its plant, and the empty one knows it holds none.
175    #[test]
176    fn a_pot_knows_what_it_holds() {
177        init_vanilla_registry();
178
179        let empty = FlowerPotBlock::new(&vanilla_blocks::FLOWER_POT, &vanilla_blocks::AIR);
180        assert!(empty.is_empty());
181        assert!(empty.potted_content().is_none());
182
183        let cactus = FlowerPotBlock::new(&vanilla_blocks::POTTED_CACTUS, &vanilla_blocks::CACTUS);
184        assert!(!cactus.is_empty());
185        assert_eq!(
186            cactus.potted_content().map(|block| &block.key),
187            Some(&vanilla_blocks::CACTUS.key)
188        );
189    }
190
191    /// The reverse map really covers the potted blocks.
192    ///
193    /// It is built from the behaviors rather than written out, so this is the
194    /// check that the extracted `potted` is reaching them -- a pot that named
195    /// nothing would simply refuse every plant, silently.
196    #[test]
197    fn the_plant_to_pot_map_is_built_from_the_behaviors() {
198        init_vanilla_registry();
199        init_behaviors();
200
201        for (plant, pot) in [
202            (&vanilla_blocks::CACTUS, &vanilla_blocks::POTTED_CACTUS),
203            (
204                &vanilla_blocks::OAK_SAPLING,
205                &vanilla_blocks::POTTED_OAK_SAPLING,
206            ),
207            (
208                &vanilla_blocks::DANDELION,
209                &vanilla_blocks::POTTED_DANDELION,
210            ),
211        ] {
212            let found = POTTED_BY_CONTENT
213                .get(&plant.key)
214                .unwrap_or_else(|| panic!("no pot holds {}", plant.key));
215            assert_eq!(found.key, pot.key);
216        }
217
218        assert!(
219            POTTED_BY_CONTENT.len() > 30,
220            "only {} pots were found; the extracted `potted` is not reaching them",
221            POTTED_BY_CONTENT.len()
222        );
223        assert!(
224            !POTTED_BY_CONTENT.contains_key(&vanilla_blocks::AIR.key),
225            "the empty pot must not claim to hold air"
226        );
227    }
