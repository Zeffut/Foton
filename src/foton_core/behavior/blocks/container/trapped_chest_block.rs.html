1//! Trapped chest behavior.
2//!
3//! Vanilla parity: `TrappedChestBlock`, which extends `ChestBlock` and adds
4//! exactly one thing -- a redstone signal counting the players looking inside.
5//! Everything else a chest does it still does: it pairs into a double chest,
6//! it refuses to open under a solid block, and hoppers reach it either way.
7//! So the chest behavior is held rather than reimplemented, and only the four
8//! signal methods differ.
9
10use std::sync::{Arc, Weak};
11
12use foton_macros::block_behavior;
13use foton_registry::blocks::BlockRef;
14use foton_registry::blocks::properties::Direction;
15use foton_utils::{BlockPos, BlockStateId};

17use crate::behavior::InventoryAccess;
18use crate::behavior::block::{BlockBehavior, BlockEntityCreation};
19use crate::behavior::blocks::container::ChestBlock;
20use crate::behavior::context::{BlockHitResult, BlockPlaceContext, InteractionResult};

25/// Strongest signal a trapped chest can give.
26///
27/// Vanilla parity: the `Mth.clamp(getOpenCount(...), 0, 15)` of `ownSignal`.
28const MAX_SIGNAL: i32 = 15;
29
30/// Behavior for the trapped chest.
31#[block_behavior]
32pub struct TrappedChestBlock {
33    /// The chest this is, in every respect but the signal.
34    chest: ChestBlock,
35    block: BlockRef,
36}
37
38impl TrappedChestBlock {
39    /// Creates a trapped chest behavior.
40    #[must_use]
41    pub const fn new(block: BlockRef) -> Self {
42        Self {
43            chest: ChestBlock::trapped(block),
44            block,
45        }
46    }
47
48    /// Returns how many players are looking inside.
49    ///
50    /// Vanilla parity: `ChestBlockEntity.getOpenCount`. Only this half is
51    /// counted; a double trapped chest gives each half its own count, which is
52    /// why opening one side powers only that side.
53    fn viewers(world: &dyn LevelReader, pos: BlockPos) -> i32 {
54        world
55            .get_block_entity(pos)
56            .map_or(0, |entity| entity.base().opener_count())
57    }
58}
59
60impl BlockBehavior for TrappedChestBlock {
61    // The chest half, delegated.
62
63    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
64        self.chest.get_state_for_placement(context)
65    }
66
67    fn update_shape(
68        &self,
69        state: BlockStateId,
70        world: &dyn ScheduledTickAccess,
71        pos: BlockPos,
72        direction: Direction,
73        neighbor_pos: BlockPos,
74        neighbor_state: BlockStateId,
75    ) -> BlockStateId {
76        self.chest
77            .update_shape(state, world, pos, direction, neighbor_pos, neighbor_state)
78    }
79
80    fn use_without_item(
81        &self,
82        state: BlockStateId,
83        world: &Arc<World>,
84        pos: BlockPos,
85        player: &Player,
86        hit_result: &BlockHitResult,
87        inv: &mut InventoryAccess,
88    ) -> InteractionResult {
89        self.chest
90            .use_without_item(state, world, pos, player, hit_result, inv)
91    }
92
93    fn new_block_entity(
94        &self,
95        level: Weak<World>,
96        pos: BlockPos,
97        state: BlockStateId,
98    ) -> BlockEntityCreation {
99        self.chest.new_block_entity(level, pos, state)
100    }
101
102    fn get_attached_containers(
103        &self,
104        state: BlockStateId,
105        world: &dyn LevelReader,
106        pos: BlockPos,
107    ) -> AttachedContainers {
108        self.chest.get_attached_containers(state, world, pos)
109    }
110
111    fn has_analog_output_signal(&self, state: BlockStateId) -> bool {
112        self.chest.has_analog_output_signal(state)
113    }
114
115    fn get_analog_output_signal(
116        &self,
117        state: BlockStateId,
118        world: &dyn LevelReader,
119        pos: BlockPos,
120        direction: Direction,
121    ) -> i32 {
122        self.chest
123            .get_analog_output_signal(state, world, pos, direction)
124    }
125
126    // And the part that makes it a trapped chest.
127
128    /// Vanilla parity: `TrappedChestBlock.isSignalSource`.
129    fn is_signal_source(&self, _state: BlockStateId, _context: SignalQueryContext) -> bool {
130        true
131    }
132
133    /// Vanilla parity: `TrappedChestBlock.ownSignal`, one level of signal per
134    /// player looking inside, up to fifteen.
135    ///
136    /// This is `own_signal` rather than `get_signal` because that is the one
137    /// vanilla overrides. `get_signal` falls back to `own_signal`, so this
138    /// covers both, whereas overriding only `get_signal` would leave
139    /// `get_best_own_or_neighbour_signal` reading zero. Nothing asks that
140    /// question of a trapped chest today -- wire reaches this through
141    /// `get_signal` either way -- so the difference is not observable yet; it
142    /// is written the vanilla way so it stays right when something does.
143    fn get_own_signal(
144        &self,
145        _state: BlockStateId,
146        world: &dyn LevelReader,
147        pos: BlockPos,
148        _context: SignalQueryContext,
149    ) -> i32 {
150        Self::viewers(world, pos).clamp(0, MAX_SIGNAL)
151    }
152
153    /// Vanilla parity: `TrappedChestBlock.getDirectSignal`, which powers only
154    /// the block above -- the reason a trapped chest cannot strongly power a
155    /// block beside it.
156    fn get_direct_signal(
157        &self,
158        state: BlockStateId,
159        world: &dyn LevelReader,
160        pos: BlockPos,
161        direction: Direction,
162        context: SignalQueryContext,
163    ) -> i32 {
164        if direction == Direction::Up {
165            self.get_signal(state, world, pos, direction, context)
166        } else {
167            0
168        }
169    }
170
171    /// Tells the neighbors when somebody opens or closes it.
172    ///
173    /// Vanilla parity: `TrappedChestBlockEntity.signalOpenCount`. Without this
174    /// the signal would be correct and nothing would ever read it again, so the
175    /// chest would appear to work only when something else happened to poke the
176    /// block beside it.
177    ///
178    /// The block below is updated as well as the block itself -- the same pair
179    /// a pressure plate updates -- so the signal also reaches a wire buried
180    /// under the chest, not just one running alongside it.
181    fn on_opener_count_changed(
182        &self,
183        world: &Arc<World>,
184        pos: BlockPos,
185        _state: BlockStateId,
186        previous: i32,
187        current: i32,
188    ) {
189        if previous == current {
190            return;
191        }
192        world.update_neighbors_at(pos, self.block);
193        world.update_neighbors_at(pos.below(), self.block);
194    }
195}
