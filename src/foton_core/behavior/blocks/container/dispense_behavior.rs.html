39/// Where the dispensing happens.
40///
41/// Vanilla parity: `BlockSource`. The facing is unpacked from the state, which
42/// vanilla's behaviors read out of it themselves.
43pub struct DispenseSource<'a> {
44    /// The world the dispenser lives in.
45    pub world: &'a Arc<World>,
46    /// The dispenser's own position.
47    pub pos: BlockPos,
48    /// The face it points at.
49    pub facing: Direction,
50    /// The nine slots behind the face.
51    ///
52    /// Vanilla parity: `BlockSource.blockEntity`, which only
53    /// `consumeWithRemainder` reads -- it is how the empty bucket a water
54    /// bucket leaves gets back inside the dispenser instead of on the floor.
55    pub block_entity: &'a DispenserBlockEntity,
56}
57
58impl DispenseSource<'_> {
59    /// Returns the point in front of the block where things appear.
60    ///
61    /// Vanilla parity: `DispenserBlock.getDispensePosition`, whose `offset` is
62    /// what lifts a fired projectile slightly above the middle of the face.
63    #[must_use]
64    pub fn dispense_position(&self, scale: f64, offset: DVec3) -> DVec3 {
65        let center = DVec3::new(
66            f64::from(self.pos.x()) + 0.5,
67            f64::from(self.pos.y()) + 0.5,
68            f64::from(self.pos.z()) + 0.5,
69        );
70        center + self.normal() * scale + offset
71    }
72
73    /// Returns the unit vector the dispenser points along.
74    #[must_use]
75    pub fn normal(&self) -> DVec3 {
76        let (x, y, z) = self.facing.offset();
77        DVec3::new(f64::from(x), f64::from(y), f64::from(z))
78    }
79}

81/// What became of the item, and whether the block should celebrate.
82///
83/// Vanilla parity: the success flag of `OptionalDispenseItemBehavior`, which is
84/// why a dispenser that could not act makes the flat failure click instead of
85/// its usual clack and puff of smoke.
86pub enum DispenseOutcome {
87    /// The behavior acted; play the dispense sound and the smoke.
88    Acted {
89        /// What stays in the slot.
90        remainder: ItemStack,
91        /// A level event to play instead of the usual dispense sound.
92        sound_override: Option<i32>,
93    },
94    /// Nothing happened; play the failure click and leave the slot alone.
95    Failed(ItemStack),
96}
97
98impl DispenseOutcome {
99    /// Acts with the usual dispenser sound.
100    #[must_use]
101    pub const fn acted(remainder: ItemStack) -> Self {
102        Self::Acted {
103            remainder,
104            sound_override: None,
105        }
106    }
107}

109/// What a dispenser does with one item.
110///
111/// Vanilla parity: `DispenseItemBehavior`.
112pub trait DispenseItemBehavior: Send + Sync {
113    /// Acts on one item taken from the dispenser.
114    fn execute(&self, source: &DispenseSource<'_>, stack: ItemStack) -> DispenseOutcome;
115}

117/// Whether a block is currently alight.
118const LIT: &BoolProperty = &BlockStateProperties::LIT;

120/// Whether a block is standing in water.
121const WATERLOGGED: &BoolProperty = &BlockStateProperties::WATERLOGGED;

123/// Distance in front of the block that projectiles and items appear at.
124///
125/// Vanilla parity: the `0.7` shared by `getDispensePosition` and the default
126/// `DispenseConfig`.
127pub const DISPENSE_OFFSET: f64 = 0.7;

129/// Extra lift a fired projectile gets.
130///
131/// Vanilla parity: the `new Vec3(0.0, 0.1, 0.0)` of
132/// `ProjectileItem.DispenseConfig.Builder`.
133const PROJECTILE_LIFT: f64 = 0.1;
134
135/// Speed a dispensed projectile leaves at.
136///
137/// Vanilla parity: `DispenseConfig.Builder.power`.
138const PROJECTILE_POWER: f32 = 1.1;
139
140/// Spread of a dispensed projectile.
141///
142/// Vanilla parity: `DispenseConfig.Builder.uncertainty`.
143const PROJECTILE_UNCERTAINTY: f32 = 6.0;
144
145/// Shoots an arrow out of the dispenser.
146///
147/// Vanilla parity: the `ProjectileDispenseBehavior` registered for
148/// `Items.ARROW`.
149struct ArrowDispenseBehavior;
150
151impl DispenseItemBehavior for ArrowDispenseBehavior {
152    fn execute(&self, source: &DispenseSource<'_>, mut stack: ItemStack) -> DispenseOutcome {
153        let position =
154            source.dispense_position(DISPENSE_OFFSET, DVec3::new(0.0, PROJECTILE_LIFT, 0.0));
155        let arrow = Arc::new(ArrowEntity::new(
156            &vanilla_entities::ARROW,
157            next_entity_id(),
158            position,
159            Arc::downgrade(source.world),
160        ));
161        arrow.shoot(source.normal(), PROJECTILE_POWER, PROJECTILE_UNCERTAINTY);
162
163        if let Err(error) = source
164            .world
165            .try_add_entity(Arc::clone(&arrow) as Arc<dyn Entity>)
166        {
167            log::debug!("dispensed arrow rejected: {error}");
168            return DispenseOutcome::Failed(stack);
169        }
170
171        stack.shrink(1);
172        DispenseOutcome::Acted {
173            remainder: stack,
174            sound_override: Some(level_events::SOUND_DISPENSER_PROJECTILE_LAUNCH),
175        }
176    }
177}

179/// How far in front of the face a fire charge appears.
180///
181/// Vanilla parity: the `getDispensePosition(source, 1.0, Vec3.ZERO)` of
182/// `FireChargeItem.createDispenseConfig`, which is further out and without the
183/// lift the other projectiles get.
184const FIRE_CHARGE_OFFSET: f64 = 1.0;
185
186/// Speed a dispensed fire charge leaves at.
187///
188/// Vanilla parity: `FireChargeItem.createDispenseConfig().power`.
189const FIRE_CHARGE_POWER: f32 = 1.0;
190
191/// Spread of a dispensed fire charge.
192///
193/// Vanilla parity: `FireChargeItem.createDispenseConfig().uncertainty`.
194const FIRE_CHARGE_UNCERTAINTY: f32 = 6.666_666_5;

196/// Shoots a small fireball out of the dispenser.
197///
198/// Vanilla parity: the `registerProjectileBehavior(Items.FIRE_CHARGE)` of
199/// `DispenseItemBehavior.bootStrap`, which builds a `ProjectileDispenseBehavior`
200/// around `FireChargeItem`.
201///
202/// `FireChargeItem.asProjectile` jitters the direction with `random.triangle`
203/// before handing the fireball over, but `ProjectileDispenseBehavior.execute`
204/// then calls `Projectile.shoot`, which overwrites the velocity outright. The
205/// jitter is unobservable, so it is not reproduced here.
206struct FireChargeDispenseBehavior;
207
208impl DispenseItemBehavior for FireChargeDispenseBehavior {
209    fn execute(&self, source: &DispenseSource<'_>, mut stack: ItemStack) -> DispenseOutcome {
210        let position = source.dispense_position(FIRE_CHARGE_OFFSET, DVec3::ZERO);
211        let fireball = Arc::new(SmallFireballEntity::new(
212            &vanilla_entities::SMALL_FIREBALL,
213            next_entity_id(),
214            position,
215            Arc::downgrade(source.world),
216        ));
217        // Vanilla parity: the `fireball.setItem(itemStack)` of
218        // `FireChargeItem.asProjectile`, which is what the client draws.
219        fireball.set_item(stack.copy_with_count(1));
220        fireball.shoot(source.normal(), FIRE_CHARGE_POWER, FIRE_CHARGE_UNCERTAINTY);
221
222        if let Err(error) = source
223            .world
224            .try_add_entity(Arc::clone(&fireball) as Arc<dyn Entity>)
225        {
226            log::debug!("dispensed fire charge rejected: {error}");
227            return DispenseOutcome::Failed(stack);
228        }
229
230        stack.shrink(1);
231        DispenseOutcome::Acted {
232            remainder: stack,
233            // Vanilla parity: the `overrideDispenseEvent(1018)` of the same
234            // config -- a dispensed fire charge roars, it does not click.
235            sound_override: Some(level_events::SOUND_BLAZE_FIREBALL),
236        }
237    }
238}

240/// Places primed TNT in front of the dispenser.
241///
242/// Vanilla parity: the `OptionalDispenseItemBehavior` registered for
243/// `Blocks.TNT`.
244struct TntDispenseBehavior;
245
246impl DispenseItemBehavior for TntDispenseBehavior {
247    fn execute(&self, source: &DispenseSource<'_>, mut stack: ItemStack) -> DispenseOutcome {
248        if !source.world.get_game_rule(&TNT_EXPLODES) {
249            return DispenseOutcome::Failed(stack);
250        }
251
252        let target = source.pos.relative(source.facing);
253        if feed_sulfur_cube(source.world, target, &mut stack) {
254            return DispenseOutcome::acted(stack);
255        }
256
257        let tnt = PrimedTntEntity::prime(source.world, target, None);
258        source.world.play_sound_at(
259            &sound_events::ENTITY_TNT_PRIMED,
260            SoundSource::Blocks,
261            tnt.position(),
262            1.0,
263            1.0,
264            None,
265        );
266
267        // TODO: vanilla also fires the ENTITY_PLACE game event, which Foton
268        // does not have here yet.
269        stack.shrink(1);
270        DispenseOutcome::acted(stack)
271    }
272}

274/// Grows whatever is in front of the dispenser.
275///
276/// Vanilla parity: the `Items.BONE_MEAL` entry of
277/// `DispenseItemBehavior.bootStrap`.
278///
279/// TODO: vanilla also falls back to `BoneMealItem.growWaterPlant`, which spreads
280/// seagrass and coral across open water; Foton has no equivalent yet, so bone
281/// meal aimed at water does nothing rather than doing the wrong thing.
282struct BoneMealDispenseBehavior;
283
284impl DispenseItemBehavior for BoneMealDispenseBehavior {
285    fn execute(&self, source: &DispenseSource<'_>, mut stack: ItemStack) -> DispenseOutcome {
286        let target = source.pos.relative(source.facing);
287        let state = source.world.get_block_state(target);
288        let behavior = BLOCK_BEHAVIORS.get_behavior(state.get_block());
289        let Some(bonemealable) = behavior.as_bonemealable() else {
290            return DispenseOutcome::Failed(stack);
291        };
292
293        if !bonemealable.is_valid_bonemeal_target(state, source.world.as_ref(), target) {
294            return DispenseOutcome::Failed(stack);
295        }
296
297        let mut rng = rand::rng();
298        if !bonemealable.is_bonemeal_success(state, source.world, &mut rng, target) {
299            // Vanilla still counts the attempt as a success and consumes the
300            // item; only the growth is left to the next try.
301            stack.shrink(1);
302            return DispenseOutcome::acted(stack);
303        }
304
305        bonemealable.perform_bonemeal(state, source.world, &mut rng, target);
306        source.world.level_event(
307            level_events::PARTICLES_AND_SOUND_PLANT_GROWTH,
308            target,
309            15,
310            None,
311        );
312        stack.shrink(1);
313        DispenseOutcome::acted(stack)
314    }
315}

317/// Sets fire to whatever is in front of the dispenser.
318///
319/// Vanilla parity: `FlintAndSteelDispenseItemBehavior`.
320///
321///
322/// TODO: vanilla also lights a sulfur cube standing in front of the
323/// dispenser, through `SulfurCube.primeTime`. Foton's flint and foton only
324/// reaches blocks.
325struct FlintAndSteelDispenseBehavior;
326
327impl DispenseItemBehavior for FlintAndSteelDispenseBehavior {
328    fn execute(&self, source: &DispenseSource<'_>, mut stack: ItemStack) -> DispenseOutcome {
329        let target = source.pos.relative(source.facing);
330        let state = source.world.get_block_state(target);
331
332        let lit = if FireBlock::can_be_placed_at(source.world, target, source.facing) {
333            source.world.set_block(
334                target,
335                FireBlock::get_state(source.world.as_ref(), target),
336                UpdateFlags::UPDATE_ALL,
337            )
338        } else if can_be_lit(state) {
339            source
340                .world
341                .set_block(target, state.set_value(LIT, true), UpdateFlags::UPDATE_ALL)
342        } else if state.get_block() == &vanilla_blocks::TNT {
343            let _ = PrimedTntEntity::prime(source.world, target, None);
344            source.world.remove_block(target, false)
345        } else {
346            false
347        };
348
349        if !lit {
350            return DispenseOutcome::Failed(stack);
351        }
352
353        // Vanilla parity: the flint and foton wears out one point per use, and
354        // the dispenser keeps the broken remainder rather than a ghost item.
355        if stack.hurt_and_break(1, false) {
356            return DispenseOutcome::acted(ItemStack::empty());
357        }
358        DispenseOutcome::acted(stack)
359    }
360}

362/// Returns whether this block is something a flame can be set to.
363///
364/// Vanilla parity: the `CampfireBlock.canLight`, `CandleBlock.canLight` and
365/// `CandleCakeBlock.canLight` chain, which all come down to an unlit block that
366/// carries the lit property and is not standing in water.
367fn can_be_lit(state: BlockStateId) -> bool {
368    let unlit = state.try_get_value(LIT) == Some(false);
369    let dry = state.try_get_value(WATERLOGGED) != Some(true);
370    unlit && dry
371}

373/// Shears whatever is standing in front of the dispenser.
374///
375/// Vanilla parity: `ShearsDispenseItemBehavior`.
376///
377/// TODO: vanilla also shears a full beehive and carves a pumpkin; Foton has the
378/// beehive block but no honeycomb-dropping path to reuse here yet.
379struct ShearsDispenseBehavior;
380
381impl DispenseItemBehavior for ShearsDispenseBehavior {
382    fn execute(&self, source: &DispenseSource<'_>, mut stack: ItemStack) -> DispenseOutcome {
383        let target = source.pos.relative(source.facing);
384        let box_at_target = WorldAabb::new(
385            f64::from(target.x()),
386            f64::from(target.y()),
387            f64::from(target.z()),
388            f64::from(target.x()) + 1.0,
389            f64::from(target.y()) + 1.0,
390            f64::from(target.z()) + 1.0,
391        );
392
393        let sheared = source
394            .world
395            .get_entities_in_aabb(&box_at_target)
396            .into_iter()
397            .find_map(|entity| {
398                let sheep = entity.as_ref().downcast_ref::<SheepEntity>()?;
399                sheep.ready_for_shearing().then(|| {
400                    sheep.shear(source.world.as_ref(), &stack);
401                })
402            })
403            .is_some();
404
405        if !sheared {
406            return DispenseOutcome::Failed(stack);
407        }
408
409        if stack.hurt_and_break(1, false) {
410            return DispenseOutcome::acted(ItemStack::empty());
411        }
412        DispenseOutcome::acted(stack)
413    }
414}

417///
418/// Vanilla parity: `EquipmentDispenseItemBehavior`. This is how an armour
419/// dispenser works, and how a saddle goes onto a pig without a player -- it is

427            return DispenseOutcome::acted(stack);
428        }
429        DefaultDispenseBehavior.execute(source, stack)
430    }
431}

433/// Equips the first thing in front of the dispenser that will take `stack`.
434///
435/// Vanilla parity: `EquipmentDispenseItemBehavior.dispenseEquipment`, which
436/// several registered behaviors fall back to as well. Returns whether anything
437/// took it.
438pub(super) fn dispense_equipment(source: &DispenseSource<'_>, stack: &mut ItemStack) -> bool {
439    let target = source.pos.relative(source.facing);
440    let Some(equippable) = stack.get_equippable() else {
441        return false;
442    };
443    let slot = equippable.slot;
444
445    let candidates = source.world.get_entities_in_aabb(&block_aabb(target));
446    let Some(wearer) = candidates.iter().find_map(|entity| {
447        let living = entity.as_living_entity()?;
448        living.can_equip_with_dispenser(stack).then_some(living)
449    }) else {
450        return false;
451    };
452
453    let equipped = stack.split(1);
454    wearer.set_item_slot(slot, equipped);
455    if let Some(mob) = wearer.as_mob() {
456        mob.set_guaranteed_drop(slot);
457        mob.set_persistence_required();
458    }
459    true
460}

462/// Puts the mob a spawn egg makes in front of the dispenser.
463///
464/// Vanilla parity: `SpawnEggItemBehavior`, reached through
465/// `DispenserBlock.getDefaultDispenseMethod` rather than the registry.
466struct SpawnEggDispenseBehavior;
467
468impl DispenseItemBehavior for SpawnEggDispenseBehavior {
469    fn execute(&self, source: &DispenseSource<'_>, mut stack: ItemStack) -> DispenseOutcome {
470        let Some(entity_type) = SpawnEggItem::entity_type(&stack) else {
471            return DispenseOutcome::Failed(stack);
472        };
473        let target = source.pos.relative(source.facing);
474
475        // Vanilla's `type.spawn(..., tryMoveDown = direction != Direction.UP,
476        // movedUp = false)`; Foton's spawn helper has no downward search yet, so
477        // an egg fired at a ceiling lands in the block it was aimed at.
478        if SpawnEggItem::spawn_at(source.world, entity_type, target).is_none() {
479            return DispenseOutcome::Failed(stack);
480        }
481
482        source.world.game_event(
483            &vanilla_game_events::ENTITY_PLACE,
484            source.pos,
485            &GameEventContext::new(None, None),
486        );
487        stack.shrink(1);
488        DispenseOutcome::acted(stack)
489    }
490}

507/// Empties a bucket that carries something in front of the dispenser.
508///
509/// Vanilla parity: the anonymous `filledBucketBehavior` of
510/// `DispenseItemBehavior.bootStrap`, shared by every bucket with contents --
511/// water, lava, powder snow and all six mob buckets. This is the half of a
512/// redstone farm a dispenser could not do at all: `emptyContents` takes a
513/// nullable user precisely so that this call site can pass nothing.
514struct FilledBucketDispenseBehavior;
515
516impl DispenseItemBehavior for FilledBucketDispenseBehavior {
517    fn execute(&self, source: &DispenseSource<'_>, stack: ItemStack) -> DispenseOutcome {
518        let behavior = ITEM_BEHAVIORS.get_behavior(stack.item());
519        let Some(container) = behavior.as_dispensible_container() else {
520            return DefaultDispenseBehavior.execute(source, stack);
521        };
522
523        let target = source.pos.relative(source.facing);
524        if !container.empty_contents(None, source.world, target, None) {
525            return DefaultDispenseBehavior.execute(source, stack);
526        }
527
528        // Vanilla reads the stack here, before `consumeWithRemainder` shrinks
529        // it, which is how the fish keeps the name the bucket was carrying.
530        container.check_extra_content(None, source.world, &stack, target);
531        DispenseOutcome::acted(consume_with_remainder(
532            source,
533            stack,
534            ItemStack::new(&vanilla_items::BUCKET),
535        ))
536    }
537}

539/// Fills an empty bucket from whatever is in front of the dispenser.
540///
541/// Vanilla parity: the `Items.BUCKET` entry of
542/// `DispenseItemBehavior.bootStrap`. Nothing here plays the pickup sound: the
543/// hand-held path does, but the dispenser fires only the game event.
544struct EmptyBucketDispenseBehavior;
545
546impl DispenseItemBehavior for EmptyBucketDispenseBehavior {
547    fn execute(&self, source: &DispenseSource<'_>, stack: ItemStack) -> DispenseOutcome {
548        let target = source.pos.relative(source.facing);
549        let state = source.world.get_block_state(target);
550        let behavior = BLOCK_BEHAVIORS.get_behavior(state.get_block());
551
552        // The waterlogging fallback is the same one the hand-held empty bucket
553        // uses; without it a dispenser could not drain a block a player can.
554        let picked = behavior
555            .pickup_block(source.world, target, state, None)
556            .or_else(|| pickup_waterlogged_block(behavior, source.world, target, state, None));
557        let Some(result) = picked.filter(|result| !result.filled_bucket.is_empty()) else {
558            return DefaultDispenseBehavior.execute(source, stack);
559        };
560
561        source.world.game_event(
562            &vanilla_game_events::FLUID_PICKUP,
563            target,
564            &GameEventContext::new(None, None),
565        );
566        DispenseOutcome::acted(consume_with_remainder(source, stack, result.filled_bucket))
567    }
568}

570/// Spends one item and finds a home for what it leaves behind.
571///
572/// Vanilla parity: `DefaultDispenseItemBehavior.consumeWithRemainder`. The
573/// remainder only takes the dispensed slot when that slot is now empty;
574/// otherwise it goes into the block's own inventory, and out onto the floor
575/// when there is no room, with a second clack and puff of smoke.
576fn consume_with_remainder(
577    source: &DispenseSource<'_>,
578    mut dispensed: ItemStack,
579    remainder: ItemStack,
580) -> ItemStack {
581    dispensed.shrink(1);
582    if dispensed.is_empty() {
583        return remainder;
584    }
585
586    // Vanilla parity: `addToInventoryOrDispense`.
587    let leftover = source.block_entity.insert_item(remainder);
588    if !leftover.is_empty() {
589        spawn_dispensed_item(source.world, source.pos, source.facing, leftover);
590        play_dispense_effects(source.world, source.pos, source.facing);
591    }
592    dispensed
593}

628    LazyLock::new(|| {

632            vanilla_items::ARROW.key.clone(),

634        );
635        behaviors.insert(
636            vanilla_items::FIRE_CHARGE.key.clone(),
637            Box::new(FireChargeDispenseBehavior),
638        );
639        behaviors.insert(
640            vanilla_items::TNT.key.clone(),
641            Box::new(TntDispenseBehavior),
642        );
643        behaviors.insert(
644            vanilla_items::BONE_MEAL.key.clone(),
645            Box::new(BoneMealDispenseBehavior),
646        );
647        behaviors.insert(
648            vanilla_items::FLINT_AND_STEEL.key.clone(),
649            Box::new(FlintAndSteelDispenseBehavior),
650        );
651        behaviors.insert(
652            vanilla_items::SHEARS.key.clone(),
653            Box::new(ShearsDispenseBehavior),
654        );
655        for bucket in filled_buckets() {
656            behaviors.insert(bucket.key.clone(), Box::new(FilledBucketDispenseBehavior));
657        }
658        behaviors.insert(
659            vanilla_items::BUCKET.key.clone(),
660            Box::new(EmptyBucketDispenseBehavior),
661        );
662        behaviors
663    });

685/// Feeds a block to a sulfur cube standing in front of the dispenser.
686///
687/// Vanilla parity: `SulfurCubeBlockDispenseItemBehavior.dispenseBlock`, which
688/// is how a dispenser loads a cube -- it is checked before the ordinary
689/// behavior, so a dispenser aimed at a cube feeds it rather than throwing.
690/// Returns whether a cube took the block.
691pub(super) fn feed_sulfur_cube(world: &Arc<World>, pos: BlockPos, stack: &mut ItemStack) -> bool {
692    let bounds = WorldAabb::new(
693        f64::from(pos.x()),
694        f64::from(pos.y()),
695        f64::from(pos.z()),
696        f64::from(pos.x()) + 1.0,
697        f64::from(pos.y()) + 1.0,
698        f64::from(pos.z()) + 1.0,
699    );
700    for entity in world.get_entities_in_aabb(&bounds) {
701        let Some(cube) = entity.downcast_ref::<SulfurCubeEntity>() else {
702            continue;
703        };
704        if cube.equip_item(stack) {
705            stack.shrink(1);
706            return true;
707        }
708    }
709    false
710}

712/// Returns whether a sulfur cube would swallow this item.
713///
714/// Vanilla parity: the `itemStack.is(ItemTags.SULFUR_CUBE_SWALLOWABLE)` branch
715/// of `DispenserBlock.getDefaultDispenseMethod`.
716#[must_use]
717pub(super) fn is_sulfur_cube_swallowable(item: ItemRef) -> bool {
718    REGISTRY
719        .items
720        .is_in_tag(item, &ItemTag::SULFUR_CUBE_SWALLOWABLE)
721}

723/// Returns what the dispenser should do with `stack`.
724///
725/// Vanilla parity: `DispenserBlock.getDispenseMethod` -- the registry first,
726/// then `getDefaultDispenseMethod` for the three things a dispenser knows how
727/// to do without an entry of their own.
728#[must_use]
729pub fn dispense_behavior_for(stack: &ItemStack) -> &'static dyn DispenseItemBehavior {
730    if let Some(registered) = DISPENSE_BEHAVIORS.get(&stack.item().key) {
731        return registered.as_ref();
732    }
733    default_dispense_behavior_for(stack)
734}
735
736/// Returns the behavior an item with no registry entry gets.
737///
738/// Vanilla parity: `DispenserBlock.getDefaultDispenseMethod`, in its order:
739/// equippable first, then a block a sulfur cube would swallow, then a spawn
740/// egg, and a plain throw for everything else.
741#[must_use]
742fn default_dispense_behavior_for(stack: &ItemStack) -> &'static dyn DispenseItemBehavior {
743    if stack.get_equippable().is_some() {
744        return &EquipmentDispenseBehavior;
745    }
746    if is_sulfur_cube_swallowable(stack.item()) {
747        return &SulfurCubeDispenseBehavior;
748    }
749    if SpawnEggItem::entity_type(stack).is_some() {
750        return &SpawnEggDispenseBehavior;
751    }
752    &DefaultDispenseBehavior
753}
