30/// Behavior for barrel blocks.
31///
32/// Barrels are container block entities with 27 slots (3x9 grid).
33/// They use the same menu as chests but cannot form double containers.
34#[block_behavior]
35pub struct BarrelBlock {
36    block: BlockRef,
37}
38
39const FACING: &EnumProperty<Direction> = &BlockStateProperties::FACING;
40
41/// Whether the barrel is being looked into.
42const OPEN: &BoolProperty = &BlockStateProperties::OPEN;
43
44/// How loud the lid is.
45///
46/// Vanilla parity: the `0.5F` of `BarrelBlockEntity.playSound`.
47const LID_VOLUME: f32 = 0.5;
48
49/// Plays the lid half a block out from the face the barrel opens through.
50///
51/// Vanilla parity: `BarrelBlockEntity.playSound`. The offset is audible: a
52/// barrel in a wall should sound from the side you can reach, not from inside
53/// the wall.
54fn play_lid(world: &Arc<World>, pos: BlockPos, state: BlockStateId, sound: SoundEventRef) {
55    let (step_x, step_y, step_z) = state.get_value(FACING).offset();
56    let position = DVec3::new(
57        f64::from(pos.x()) + 0.5 + f64::from(step_x) / 2.0,
58        f64::from(pos.y()) + 0.5 + f64::from(step_y) / 2.0,
59        f64::from(pos.z()) + 0.5 + f64::from(step_z) / 2.0,
60    );
61    world.play_sound_at(
62        sound,
63        SoundSource::Blocks,
64        position,
65        LID_VOLUME,
66        rand::random::<f32>().mul_add(0.1, 0.9),
67        None,
68    );
69}

71impl BarrelBlock {
72    /// Creates a new barrel block behavior.
73    #[must_use]
74    pub const fn new(block: BlockRef) -> Self {
75        Self { block }

79impl BlockBehavior for BarrelBlock {
80    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
81        // Barrel faces opposite to the player's look direction (all 6 directions).
82        let facing = context.get_nearest_looking_direction().opposite();
83
84        Some(self.block.default_state().set_value(FACING, facing))
85    }
86
87    fn use_without_item(
88        &self,
89        _state: BlockStateId,
90        world: &Arc<World>,
91        pos: BlockPos,
92        player: &Player,
93        _hit_result: &BlockHitResult,
94        _inv: &mut InventoryAccess,
95    ) -> InteractionResult {
96        // Get the block entity
97        let Some(block_entity) = world.get_block_entity(pos) else {
98            return InteractionResult::Pass;
99        };
100
101        // Create a container reference from the block entity
102        let Some(container_ref) = ContainerRef::from_block_entity(block_entity.clone()) else {
103            return InteractionResult::Pass;
104        };
105
106        // Vanilla parity: `RandomizableContainerBlockEntity.createMenu`
107        // unpacks with the opening player, whose luck the roll uses.
108        container_ref.unpack_loot_table(Some(player));
109
110        // Open the chest menu (3 rows for barrel)
111        let inventory = player.inventory.clone();
112        player.open_menu(
113            block_entity.display_name(TextComponent::translated(
114                translations::CONTAINER_BARREL.msg(),
115            )),
116            move |context| chest(inventory, context.container_id, container_ref, 3),
117        );
118
119        // TODO: Award stat OPEN_BARREL, and anger nearby piglins; Foton has no
120        // statistics registry and no piglins.
121        InteractionResult::Success
122    }
123
124    /// Vanilla parity: the `onOpen` of `BarrelBlockEntity.openersCounter`.
125    ///
126    /// The `open` property is the only thing that makes a barrel look open, so
127    /// without it a player has no idea anyone is in theirs.
128    fn on_container_open(&self, world: &Arc<World>, pos: BlockPos, state: BlockStateId) {
129        play_lid(world, pos, state, &sound_events::BLOCK_BARREL_OPEN);
130        world.set_block(pos, state.set_value(OPEN, true), UpdateFlags::UPDATE_ALL);
131    }
132
133    /// Vanilla parity: the `onClose` of `BarrelBlockEntity.openersCounter`.
134    fn on_container_close(&self, world: &Arc<World>, pos: BlockPos, state: BlockStateId) {
135        play_lid(world, pos, state, &sound_events::BLOCK_BARREL_CLOSE);
136        world.set_block(pos, state.set_value(OPEN, false), UpdateFlags::UPDATE_ALL);
137    }
138
139    fn new_block_entity(
140        &self,
141        level: Weak<World>,
142        pos: BlockPos,
143        state: BlockStateId,
144    ) -> BlockEntityCreation {
145        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
146            &vanilla_block_entity_types::BARREL,
147            level,
148            pos,
149            state,
150        ))
151    }
152
153    fn has_analog_output_signal(&self, _state: BlockStateId) -> bool {
154        true
155    }
156
157    fn get_analog_output_signal(
158        &self,
159        _state: BlockStateId,
160        world: &dyn LevelReader,
161        pos: BlockPos,
162        _direction: Direction,
163    ) -> i32 {
164        // Get the block entity and calculate signal from container contents
165        let Some(container_ref) = world
166            .get_block_entity(pos)
167            .and_then(ContainerRef::from_block_entity)
168        else {
169            return 0;
170        };
171        let guard = ContainerLockGuard::lock_all(&[&container_ref]);
172        guard
173            .get(container_ref.container_id())
174            .map_or(0, |container| {
175                calculate_redstone_signal_from_container(container)
176            })
177    }
178}
