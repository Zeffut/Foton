11    pub(super) fn load_gzip_nbt(
12        registry: &Registry,
13        bytes: &[u8],
14        context: &str,
15    ) -> Result<Self, String> {
16        let mut decoder = GzDecoder::new(bytes);
17        let mut data = Vec::new();
18        decoder
19            .read_to_end(&mut data)
20            .map_err(|err| format!("failed to decompress structure template {context}: {err}"))?;
21
22        let nbt = read_nbt(&mut Cursor::new(&data))
23            .map_err(|err| format!("failed to parse structure template {context}: {err}"))?;
24        let root = match nbt {
25            BorrowedNbt::Some(root) => root,
26            BorrowedNbt::None => {
27                return Err(format!("structure template {context} is empty"));
28            }
29        };
30        let compound = root.as_compound();
31
32        let size = Self::read_vec3(compound.list("size"), context, "size")?;
33        let palettes = Self::read_palettes(registry, &compound, context)?;
34        let blocks = compound
35            .list("blocks")
36            .and_then(|list| list.compounds())
37            .ok_or_else(|| format!("structure template {context} has non-compound blocks list"))?;
38
39        let mut loaded_palettes = Vec::with_capacity(palettes.len());
40        for palette in &palettes {
41            loaded_palettes.push(StructureTemplatePalette {
42                blocks: Self::read_blocks(registry, &blocks, palette, context)?,
43            });
44        }
45
46        let entities = Self::read_entities(&compound, context)?;
47        let author = compound
48            .string("author")
49            .map(|author| author.to_str().into_owned())
50            .unwrap_or_default();
51
52        Ok(Self {
53            author,
54            size,
55            palettes: loaded_palettes,
56            entities,
57        })
58    }

60    pub(super) fn read_vec3(
61        list: Option<BorrowedNbtList<'_, '_>>,
62        context: &str,
63        field: &str,
64    ) -> Result<IVec3, String> {
65        let ints = list
66            .and_then(|list| list.ints())
67            .ok_or_else(|| format!("structure template {context} has non-int {field} list"))?;
68        if ints.len() < 3 {
69            return Err(format!(
70                "structure template {context} {field} list has fewer than 3 entries"
71            ));
72        }
73        Ok(IVec3::new(ints[0], ints[1], ints[2]))
74    }

92    pub(super) fn read_palettes(
93        registry: &Registry,
94        compound: &BorrowedNbtCompound<'_, '_>,
95        context: &str,
96    ) -> Result<Vec<Vec<BlockStateId>>, String> {
97        if let Some(palette) = compound.list("palette").and_then(|list| list.compounds()) {
98            return Ok(vec![Self::read_palette(registry, &palette, context)?]);
99        }
100
101        let palettes = compound
102            .list("palettes")
103            .and_then(|list| list.lists())
104            .ok_or_else(|| {
105                format!("structure template {context} is missing palette or palettes")
106            })?;
107        if palettes.is_empty() {
108            return Err(format!(
109                "structure template {context} has empty palettes list"
110            ));
111        }
112
113        let mut result = Vec::with_capacity(palettes.len());
114        for palette in palettes {
115            let entries = palette.compounds().ok_or_else(|| {
116                format!("structure template {context} has non-compound palette entry")
117            })?;
118            result.push(Self::read_palette(registry, &entries, context)?);
119        }
120        Ok(result)
121    }

123    pub(super) fn read_palette(
124        registry: &Registry,
125        entries: &BorrowedNbtCompoundList<'_, '_>,
126        context: &str,
127    ) -> Result<Vec<BlockStateId>, String> {
128        let mut states = Vec::with_capacity(entries.len());
129        for entry in entries.clone() {
130            let Some(name) = entry.string("Name") else {
131                return Err(format!(
132                    "structure template {context} has palette entry without Name"
133                ));
134            };
135            let name = Identifier::from_str(name.to_str().as_ref()).map_err(|err| {
136                format!("structure template {context} has invalid block identifier: {err}")
137            })?;
138            let mut properties = BTreeMap::new();
139            if let Some(props) = entry.compound("Properties") {
140                for (key, value) in props.iter() {
141                    let Some(value) = value.string() else {
142                        return Err(format!(
143                            "structure template {context} has non-string property {} on {name}",
144                            key.to_str()
145                        ));
146                    };
147                    properties.insert(key.to_str().into_owned(), value.to_str().into_owned());
148                }
149            }
150            states.push(WorldgenStateResolver::block_state_from_data(
151                registry,
152                &BlockStateData { name, properties },
153                "structure template palette",
154            ));
155        }
156        Ok(states)
157    }

159    pub(super) fn read_blocks(
160        registry: &Registry,
161        blocks: &BorrowedNbtCompoundList<'_, '_>,
162        palette: &[BlockStateId],
163        context: &str,
164    ) -> Result<Vec<StructureBlockInfo>, String> {
165        let mut full_blocks = Vec::new();
166        let mut other_blocks = Vec::new();
167        let mut block_entities = Vec::new();
168
169        for block in blocks.clone() {
170            let pos = Self::read_vec3(block.list("pos"), context, "block pos")?;
171            let state_index = block
172                .int("state")
173                .ok_or_else(|| format!("structure template {context} block is missing state"))?;
174            if state_index < 0 {
175                return Err(format!(
176                    "structure template {context} has negative palette state {state_index}"
177                ));
178            }
179            let state_index = usize::try_from(state_index).map_err(|_| {
180                format!("structure template {context} state index does not fit usize")
181            })?;
182            let Some(&state) = palette.get(state_index) else {
183                return Err(format!(
184                    "structure template {context} state index {state_index} exceeds palette length {}",
185                    palette.len()
186                ));
187            };
188            let nbt = block.compound("nbt").map(|nbt| nbt.to_owned());
189            let info = StructureBlockInfo {
190                pos: BlockPos::new(pos[0], pos[1], pos[2]),
191                state,
192                nbt,
193            };
194
195            if info.nbt.is_some() {
196                block_entities.push(info);
197            } else if Self::is_static_full_block(registry, state) {
198                full_blocks.push(info);
199            } else {
200                other_blocks.push(info);
201            }
202        }
203
204        Self::sort_block_infos(&mut full_blocks);
205        Self::sort_block_infos(&mut other_blocks);
206        Self::sort_block_infos(&mut block_entities);
207
208        full_blocks.extend(other_blocks);
209        full_blocks.extend(block_entities);
210        Ok(full_blocks)
211    }

213    pub(super) fn read_entities(
214        compound: &BorrowedNbtCompound<'_, '_>,
215        context: &str,
216    ) -> Result<Vec<StructureEntityInfo>, String> {
217        let Some(entities) = compound.list("entities").and_then(|list| list.compounds()) else {
218            return Ok(Vec::new());
219        };
220
221        let mut result = Vec::with_capacity(entities.len());
222        for entity in entities.clone() {
223            let pos = Self::read_vec3d(entity.list("pos"), context, "entity pos")?;
224            let block_pos = Self::read_vec3(entity.list("blockPos"), context, "entity blockPos")?;
225            let entity_nbt = entity.compound("nbt").ok_or_else(|| {
226                format!("structure template {context} has entity entry without nbt")
227            })?;
228            // A template's entities are ordinary vanilla save compounds, so the
229            // same reader `/summon` uses decodes them. `Pos` and `UUID` are the
230            // two fields a template does not take from the compound: the
231            // position is the template-relative `pos` above, and each placement
232            // mints a fresh UUID so two copies of one structure are two
233            // entities.
234            let loaded = read_entity_nbt(&entity_nbt).ok_or_else(|| {
235                let id = entity_nbt
236                    .string("id")
237                    .map_or_else(|| "<missing>".to_owned(), |id| id.to_str().into_owned());
238                format!("structure template {context} references unknown entity type {id}")
239            })?;
240            result.push(StructureEntityInfo {
241                pos,
242                block_pos: BlockPos::new(block_pos[0], block_pos[1], block_pos[2]),
243                entity_type: loaded.entity_type,
244                rotation: loaded.rotation,
245                velocity: loaded.velocity,
246                fall_distance: loaded.fall_distance,
247                fire_freeze: loaded.fire_freeze,
248                on_ground: loaded.on_ground,
249                save_data: loaded.save_data,
250                nbt: loaded.remainder,
251            });
252        }
253
254        Ok(result)
255    }

257    pub(super) fn is_static_full_block(registry: &Registry, state: BlockStateId) -> bool {
258        let Some(block) = registry.blocks.by_state_id(state) else {
259            return false;
260        };
261        !block.config.dynamic_shape
262            && blocks::shapes::is_shape_full_block(
263                registry.blocks.get_static_collision_shape(state),
264            )
265    }

267    pub(super) fn sort_block_infos(blocks: &mut [StructureBlockInfo]) {
268        blocks.sort_by(|left, right| {
269            left.pos
270                .y()
271                .cmp(&right.pos.y())
272                .then(left.pos.x().cmp(&right.pos.x()))
273                .then(left.pos.z().cmp(&right.pos.z()))
274        });
275    }

288    pub(crate) const fn zero_position_with_transform(
289        &self,
290        zero_pos: BlockPos,
291        rotation: Rotation,
292    ) -> BlockPos {
293        let x = self.size.x - 1;
294        let z = self.size.z - 1;
295        match rotation {
296            Rotation::None => zero_pos,
297            Rotation::Clockwise90 => zero_pos.offset(z, 0, 0),
298            Rotation::Clockwise180 => zero_pos.offset(x, 0, z),
299            Rotation::CounterClockwise90 => zero_pos.offset(0, 0, x),
300        }
301    }
