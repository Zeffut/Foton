1//! Feature-stage structure piece placement boundary.
2//!
3//! Structure starts are generated before noise, but vanilla emits the piece
4//! blocks during biome decoration. This module is the single dispatch point for
5//! that pass; individual family placers must fill in exact vanilla behavior
6//! before any payload variant starts writing blocks.
7
8/// Builds the mob a structure places, the way vanilla's
9/// `EntityType.create(level, EntitySpawnReason.STRUCTURE)` does, and snaps it
10/// into position.
11///
12/// Every structure that places a mob used to raise a `RawEntity` instead. That
13/// type exists so a save file can round-trip an entity Foton cannot model yet:
14/// it carries a type, a position and its original NBT, and nothing else -- no
15/// goals, no brain, no `tick`. So every mob a structure placed stood still
16/// forever, the woodland mansion's evokers and vindicators among them.
17pub(super) fn create_structure_mob(
18    world: &Weak<World>,
19    entity_pos: DVec3,
20    entity_type: EntityTypeRef,
21) -> Option<SharedEntity> {
22    let Some(entity) = ENTITIES.create(entity_type, next_entity_id(), entity_pos, world.clone())
23    else {
24        log::warn!(
25            "a structure wanted a {} and the entity registry has no factory for it",
26            entity_type.key
27        );
28        return None;
29    };
30    // Vanilla parity: the `snapTo(position, 0.0F, 0.0F)` every caller runs.
31    let _ = entity.try_set_position(entity_pos);
32    entity.set_rotation((0.0, 0.0));
33    entity.set_old_position_to_current();
34    Some(entity)
35}
36
37/// Runs the `finalizeSpawn` a vanilla structure mob gets.
38///
39/// The end city's shulker is the one that does not: vanilla places it with a
40/// bare `setPos` and `addFreshEntity`.
41pub(super) fn finalize_structure_mob(world: &Weak<World>, entity: &SharedEntity) {
42    let Some(mob) = entity.as_mob() else {
43        return;
44    };
45    let Some(world) = world.upgrade() else {
46        return;
47    };
48    let _ = mob.finalize_spawn(&world, EntitySpawnReason::Structure, None);
49}

97    ///
98    /// Returns whether the vanilla placement call succeeded. Later milestones
99    /// must implement each remaining payload variant completely before it can

105    pub(crate) fn place_piece(
106        region: &impl WorldGenLevel,
107        registry: &Registry,
108        piece: &mut StructurePiece,
109        reference_pos: BlockPos,
110        clip: BoundingBox,
111        random: &mut WorldgenRandom,
112        biome_zoom_seed: i64,
113    ) -> bool {
114        let mut piece_bounding_box = piece.bounding_box;
115        let piece_orientation = piece.orientation;
116        let placed = match &mut piece.payload {
117            StructurePiecePayload::Jigsaw(data) => Self::place_pool_element(
118                region,
119                registry,
120                &data.pool_element,
121                BlockPos::new(data.position.x, data.position.y, data.position.z),
122                reference_pos,
123                data.rotation,
124                clip,
125                random,
126                data.liquid_settings,
127                biome_zoom_seed,
128                // Chunk generation always replaces jigsaws; only a jigsaw
129                // block's generate button can keep them.
130                false,
131            ),
132            StructurePiecePayload::Template(data) => Self::place_template_piece(
133                region,
134                registry,
135                data,
136                &mut piece_bounding_box,
137                reference_pos,
138                clip,
139                random,
140            ),
141            StructurePiecePayload::Procedural(ProceduralPieceData::Mineshaft(data)) => {
142                Self::place_mineshaft_piece(
143                    region,
144                    registry,
145                    piece_bounding_box,
146                    piece_orientation,
147                    data,
148                    clip,
149                    random,
150                    biome_zoom_seed,
151                )
152            }
153            StructurePiecePayload::Procedural(ProceduralPieceData::NetherFortress(data)) => {
154                Self::place_nether_fortress_piece(
155                    region,
156                    registry,
157                    piece_bounding_box,
158                    piece_orientation,
159                    data,
160                    clip,
161                    random,
162                )
163            }
164            StructurePiecePayload::Procedural(ProceduralPieceData::OceanMonument(data)) => {
165                Self::place_ocean_monument_piece(
166                    region,
167                    registry,
168                    piece_bounding_box,
169                    piece_orientation,
170                    data,
171                    clip,
172                    random,
173                )
174            }
175            StructurePiecePayload::Procedural(ProceduralPieceData::Stronghold(data)) => {
176                Self::place_stronghold_piece(
177                    region,
178                    registry,
179                    piece_bounding_box,
180                    piece_orientation,
181                    data,
182                    clip,
183                    random,
184                )
185            }
186            StructurePiecePayload::Procedural(ProceduralPieceData::BuriedTreasure) => {
187                Self::place_buried_treasure_piece(region, &mut piece_bounding_box, clip, random)
188            }
189            StructurePiecePayload::Procedural(ProceduralPieceData::DesertPyramid(data)) => {
190                Self::place_desert_pyramid_piece(
191                    region,
192                    registry,
193                    &mut piece_bounding_box,
194                    piece_orientation,
195                    data,
196                    clip,
197                    random,
198                )
199            }
200            StructurePiecePayload::Procedural(ProceduralPieceData::JungleTemple(data)) => {
201                Self::place_jungle_temple_piece(
202                    region,
203                    registry,
204                    &mut piece_bounding_box,
205                    piece_orientation,
206                    data,
207                    clip,
208                    random,
209                )
210            }
211            StructurePiecePayload::Procedural(ProceduralPieceData::SwampHut(data)) => {
212                Self::place_swamp_hut_piece(
213                    region,
214                    registry,
215                    &mut piece_bounding_box,
216                    piece_orientation,
217                    data,
218                    clip,
219                    random,
220                )
221            }
222            StructurePiecePayload::Procedural(ProceduralPieceData::Unimplemented) => false,
223        };
224        piece.bounding_box = piece_bounding_box;
225        placed
226    }

239    const VANILLA_HORIZONTAL_DIRECTIONS: [Direction; 4] = [
240        Direction::North,
241        Direction::East,
242        Direction::South,
243        Direction::West,
244    ];

246    pub(super) fn reorient_chest(
247        region: &impl WorldGenLevel,
248        pos: BlockPos,
249        state: BlockStateId,
250    ) -> BlockStateId {
251        let mut solid_neighbor = None;
252
253        for direction in Self::VANILLA_HORIZONTAL_DIRECTIONS {
254            let relative_pos = pos.relative(direction);
255            let neighbor = region.get_block_state(relative_pos);
256            if neighbor.get_block() == &vanilla_blocks::CHEST {
257                return state;
258            }
259
260            if neighbor.is_solid_render() {
261                if solid_neighbor.is_some() {
262                    solid_neighbor = None;
263                    break;
264                }
265                solid_neighbor = Some(direction);
266            }
267        }
268
269        if let Some(direction) = solid_neighbor {
270            return state.set_value(
271                &BlockStateProperties::HORIZONTAL_FACING,
272                direction.opposite(),
273            );
274        }
275
276        let mut lock_dir = state.get_value(&BlockStateProperties::HORIZONTAL_FACING);
277        let mut relative_pos = pos.relative(lock_dir);
278        if region.get_block_state(relative_pos).is_solid_render() {
279            lock_dir = lock_dir.opposite();
280            relative_pos = pos.relative(lock_dir);
281        }
282        if region.get_block_state(relative_pos).is_solid_render() {
283            lock_dir = lock_dir.rotate_y_clockwise();
284            relative_pos = pos.relative(lock_dir);
285        }
286        if region.get_block_state(relative_pos).is_solid_render() {
287            lock_dir = lock_dir.opposite();
288        }
289        state.set_value(&BlockStateProperties::HORIZONTAL_FACING, lock_dir)
290    }

292    pub(super) fn create_loot_chest(
293        region: &impl WorldGenLevel,
294        clip: BoundingBox,
295        random: &mut WorldgenRandom,
296        pos: BlockPos,
297        loot_table: &'static str,
298    ) -> bool {
299        if !clip.contains_blockpos(pos)
300            || region.get_block_state(pos).get_block() == &vanilla_blocks::CHEST
301        {
302            return false;
303        }
304
305        let state = Self::reorient_chest(region, pos, vanilla_blocks::CHEST.default_state());
306        if !region.set_block_state(pos, state, UpdateFlags::UPDATE_CLIENTS) {
307            return false;
308        }
309
310        Self::set_loot_table_block_entity(
311            region,
312            pos,
313            &vanilla_block_entity_types::CHEST,
314            state,
315            loot_table,
316            random.next_i64(),
317        )
318    }

320    pub(super) fn set_loot_table_block_entity(
321        region: &impl WorldGenLevel,
322        pos: BlockPos,
323        block_entity_type: BlockEntityTypeRef,
324        state: BlockStateId,
325        loot_table: &'static str,
326        seed: i64,
327    ) -> bool {
328        let mut nbt = NbtCompound::new();
329        nbt.insert("LootTable", loot_table);
330        if seed != 0 {
331            nbt.insert("LootTableSeed", seed);
332        }
333        region.set_block_entity_data(pos, block_entity_type, state, nbt)
334    }

336    pub(super) fn set_spawner_entity(
337        region: &impl WorldGenLevel,
338        pos: BlockPos,
339        state: BlockStateId,
340        entity_id: &'static str,
341    ) -> bool {
342        let mut entity = NbtCompound::new();
343        entity.insert("id", entity_id);
344
345        let mut spawn_data = NbtCompound::new();
346        spawn_data.insert("entity", NbtTag::Compound(entity));
347
348        let mut nbt = NbtCompound::new();
349        nbt.insert("Delay", 20_i16);
350        nbt.insert("MinSpawnDelay", 200_i16);
351        nbt.insert("MaxSpawnDelay", 800_i16);
352        nbt.insert("SpawnCount", 4_i16);
353        nbt.insert("MaxNearbyEntities", 6_i16);
354        nbt.insert("RequiredPlayerRange", 16_i16);
355        nbt.insert("SpawnRange", 4_i16);
356        nbt.insert("SpawnData", NbtTag::Compound(spawn_data));
357        nbt.insert(
358            "SpawnPotentials",
359            NbtTag::List(NbtList::Compound(Vec::new())),
360        );
361
362        region.set_block_entity_data(pos, &vanilla_block_entity_types::MOB_SPAWNER, state, nbt)
363    }

381    pub(super) const fn orientation_transform(
382        orientation: Option<Direction>,
383    ) -> (StructureMirror, Rotation) {
384        match orientation {
385            None | Some(Direction::North | Direction::Up | Direction::Down) => {
386                (StructureMirror::None, Rotation::None)
387            }
388            Some(Direction::South) => (StructureMirror::LeftRight, Rotation::None),
389            Some(Direction::West) => (StructureMirror::LeftRight, Rotation::Clockwise90),
390            Some(Direction::East) => (StructureMirror::None, Rotation::Clockwise90),
391        }
392    }

399            || block == &vanilla_blocks::OAK_FENCE
400            || block == &vanilla_blocks::SPRUCE_FENCE
401            || block == &vanilla_blocks::DARK_OAK_FENCE
402            || block == &vanilla_blocks::PALE_OAK_FENCE
403            || block == &vanilla_blocks::ACACIA_FENCE
404            || block == &vanilla_blocks::BIRCH_FENCE
405            || block == &vanilla_blocks::JUNGLE_FENCE
406            || block == &vanilla_blocks::LADDER
