22    pub(crate) fn place_pool_element(
23        region: &impl WorldGenLevel,
24        registry: &Registry,
25        element: &PoolElement,
26        position: BlockPos,
27        reference_pos: BlockPos,
28        rotation: Rotation,
29        clip: BoundingBox,
30        random: &mut WorldgenRandom,
31        liquid_settings: LiquidSettingsData,
32        biome_zoom_seed: i64,
33        keep_jigsaws: bool,
34    ) -> bool {
35        match element {
36            PoolElement::Single {
37                location,
38                processors,
39                projection,
40            } => Self::place_single_pool_element(
41                region,
42                registry,
43                location,
44                processors,
45                *projection,
46                StructureBlockIgnore::StructureBlock,
47                StructureBlockIgnore::None,
48                position,
49                reference_pos,
50                rotation,
51                clip,
52                random,
53                liquid_settings,
54                keep_jigsaws,
55            ),
56            PoolElement::LegacySingle {
57                location,
58                processors,
59                projection,
60            } => Self::place_single_pool_element(
61                region,
62                registry,
63                location,
64                processors,
65                *projection,
66                StructureBlockIgnore::None,
67                StructureBlockIgnore::StructureAndAir,
68                position,
69                reference_pos,
70                rotation,
71                clip,
72                random,
73                liquid_settings,
74                keep_jigsaws,
75            ),
76            PoolElement::Empty => true,
77            PoolElement::Feature { feature, .. } => {
78                FeatureDecorationRunner::place_structure_pool_feature(
79                    region,
80                    registry,
81                    random,
82                    position,
83                    feature,
84                    biome_zoom_seed,
85                )
86            }
87            PoolElement::List { elements, .. } => {
88                for element in elements {
89                    if !Self::place_pool_element(
90                        region,
91                        registry,
92                        element,
93                        position,
94                        reference_pos,
95                        rotation,
96                        clip,
97                        random,
98                        liquid_settings,
99                        biome_zoom_seed,
100                        keep_jigsaws,
101                    ) {
102                        return false;
103                    }
104                }
105                true
106            }
107        }
108    }

110    #[expect(
111        clippy::too_many_arguments,
112        reason = "mirrors SinglePoolElement.place and StructureTemplate.placeInWorld"
113    )]
114    fn place_single_pool_element(
115        region: &impl WorldGenLevel,
116        registry: &Registry,
117        location: &Identifier,
118        processors: &ProcessorList,
119        projection: Projection,
120        block_ignore: StructureBlockIgnore,
121        late_block_ignore: StructureBlockIgnore,
122        position: BlockPos,
123        reference_pos: BlockPos,
124        rotation: Rotation,
125        clip: BoundingBox,
126        random: &mut WorldgenRandom,
127        liquid_settings: LiquidSettingsData,
128        keep_jigsaws: bool,
129    ) -> bool {
130        let template = match StructureTemplate::load_vanilla(registry, location) {
131            Ok(template) => template,
132            Err(err) => panic!("{err}"),
133        };
134        let processor_list = Self::pool_processors(registry, processors);
135        let settings = StructurePlaceSettings {
136            mirror: StructureMirror::None,
137            rotation,
138            rotation_pivot: BlockPos::ZERO,
139            bounding_box: clip,
140            processors: processor_list,
141            block_ignore,
142            late_block_ignore,
143            // Vanilla parity: `SinglePoolElement.getSettings` adds
144            // `JigsawReplacementProcessor` unless the caller asked to keep the
145            // jigsaws standing, which only a jigsaw block's generate button does.
146            replace_jigsaws: !keep_jigsaws,
147            projection: Some(projection),
148            processor_random: StructureProcessorRandom::Positional,
149            liquid_settings,
150            ignore_entities: false,
151        };
152
153        template.place_in_world(
154            region,
155            registry,
156            position,
157            reference_pos,
158            &settings,
159            random,
160            Self::JIGSAW_UPDATE_FLAGS,
161        )
162    }

164    fn pool_processors<'a>(
165        registry: &'a Registry,
166        processors: &'a ProcessorList,
167    ) -> &'a [StructureProcessorKind] {
168        match processors {
169            ProcessorList::Empty => &[],
170            ProcessorList::Registry(key) => {
171                let Some(processor_list) = registry.structure_processors.by_key(key) else {
172                    panic!("template pool references unknown processor list {key}");
173                };
174                &processor_list.data.processors
175            }
176        }
177    }
