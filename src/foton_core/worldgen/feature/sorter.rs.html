10/// One placed feature a biome's generation settings list for a step.
11///
12/// Vanilla parity: a `Holder<PlacedFeature>`, which can be a registry reference
13/// or a direct value. The flat generator's `FILL_LAYER` layers are the only
14/// direct ones Foton builds, and they exist only for the world that built them,
15/// so they are carried here rather than registered.
16#[derive(Clone, Debug)]
17pub(crate) enum FeatureEntry {
18    /// A registered placed feature.
19    Registered(PlacedFeatureEntryRef),
20    /// A placed feature a generator built for itself, with the identity the
21    /// generator gave it.
22    Inline(usize, Arc<PlacedFeatureData>),
23}
24
25impl FeatureEntry {
26    fn identity(&self) -> FeatureIdentity {
27        match self {
28            Self::Registered(feature) => {
29                let Some(id) = feature.try_id() else {
30                    panic!("placed feature {} is not registered", feature.key);
31                };
32                FeatureIdentity::Registered(id)
33            }
34            Self::Inline(identity, _) => FeatureIdentity::Inline(*identity),
35        }
36    }
37}

39/// The placed features one biome contributes, step by step.
40///
41/// Vanilla parity: `BiomeGenerationSettings.features()` for that biome, after
42/// whatever the generator does to it -- the flat generator rewrites its own
43/// biome's list in `adjustGenerationSettings`.
44pub(crate) struct BiomeFeatures {
45    /// The biome these features belong to.
46    pub(crate) biome: BiomeRef,
47    /// Features per decoration step.
48    pub(crate) steps: Vec<Vec<FeatureEntry>>,
49}

51/// Identity of one placed feature, which is what vanilla's index lookup is
52/// built on.
53#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
54enum FeatureIdentity {
55    Registered(usize),
56    Inline(usize),
57}

72#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
73struct FeatureVertex {
74    step: usize,
75    order: usize,
76    identity: FeatureIdentity,
77}
78
79impl Ord for FeatureVertex {
80    fn cmp(&self, other: &Self) -> Ordering {
81        (self.step, self.order, self.identity).cmp(&(other.step, other.order, other.identity))
82    }
83}
84
85impl PartialOrd for FeatureVertex {
86    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
87        Some(self.cmp(other))
88    }
89}

91impl FeatureSorter {
92    #[must_use]
93    pub(super) fn build(sources: &[BiomeFeatures]) -> Self {
94        let mut feature_order_by_identity = FxHashMap::default();
95        let mut next_feature_order = 0usize;
96        let mut edges = BTreeMap::<FeatureVertex, BTreeSet<FeatureVertex>>::new();
97
98        for source in sources {
99            let mut biome_features = Vec::new();
100
101            for (step, feature_stage) in source.steps.iter().enumerate() {
102                for entry in feature_stage {
103                    let identity = entry.identity();
104                    let feature_order =
105                        *feature_order_by_identity
106                            .entry(identity)
107                            .or_insert_with(|| {
108                                let order = next_feature_order;
109                                next_feature_order += 1;
110                                order
111                            });
112
113                    let vertex = FeatureVertex {
114                        step,
115                        order: feature_order,
116                        identity,
117                    };
118                    edges.entry(vertex).or_default();
119                    biome_features.push(vertex);
120                }
121            }
122
123            for feature_pair in biome_features.windows(2) {
124                edges
125                    .entry(feature_pair[0])
126                    .or_default()
127                    .insert(feature_pair[1]);
128            }
129        }
130
131        let sorted_features = Self::topological_sort(&edges);
132        Self::from_sorted_features(&sorted_features, sources)
133    }
134
135    #[must_use]
136    pub(super) fn step_count(&self) -> usize {
137        self.steps.len()
138    }
139
140    pub(super) fn step(&self, step: usize) -> Option<&FeatureStepData> {
141        self.steps.get(step)
142    }
143
144    fn topological_sort(
145        edges: &BTreeMap<FeatureVertex, BTreeSet<FeatureVertex>>,
146    ) -> Vec<FeatureVertex> {
147        let mut sorted = Vec::with_capacity(edges.len());
148        let mut discovered = BTreeSet::new();
149        let mut visiting = BTreeSet::new();
150        let vertices = edges.keys().copied().collect::<Vec<_>>();
151
152        for vertex in vertices {
153            assert!(
154                !Self::visit(vertex, edges, &mut discovered, &mut visiting, &mut sorted),
155                "biome decoration placed-feature order contains a cycle"
156            );
157        }
158
159        sorted.reverse();
160        sorted
161    }
162
163    fn visit(
164        vertex: FeatureVertex,
165        edges: &BTreeMap<FeatureVertex, BTreeSet<FeatureVertex>>,
166        discovered: &mut BTreeSet<FeatureVertex>,
167        visiting: &mut BTreeSet<FeatureVertex>,
168        sorted: &mut Vec<FeatureVertex>,
169    ) -> bool {
170        if discovered.contains(&vertex) {
171            return false;
172        }
173        if !visiting.insert(vertex) {
174            return true;
175        }
176
177        if let Some(neighbors) = edges.get(&vertex) {
178            for &neighbor in neighbors {
179                if Self::visit(neighbor, edges, discovered, visiting, sorted) {
180                    return true;
181                }
182            }
183        }
184
185        visiting.remove(&vertex);
186        discovered.insert(vertex);
187        sorted.push(vertex);
188        false
189    }
190
191    #[must_use]
192    fn from_sorted_features(sorted_features: &[FeatureVertex], sources: &[BiomeFeatures]) -> Self {
193        let Some(max_step) = sorted_features.iter().map(|feature| feature.step).max() else {
194            return Self {
195                steps: Box::new([]),
196            };
197        };
198
199        let entry_by_identity: FxHashMap<FeatureIdentity, FeatureEntry> = sources
200            .iter()
201            .flat_map(|source| source.steps.iter().flatten())
202            .map(|entry| (entry.identity(), entry.clone()))
203            .collect();
204
205        let mut steps = Vec::with_capacity(max_step + 1);
206        for step in 0..=max_step {
207            let mut features = Vec::new();
208            let mut index_by_identity = FxHashMap::default();
209
210            for feature in sorted_features
211                .iter()
212                .filter(|feature| feature.step == step)
213            {
214                let Some(entry) = entry_by_identity.get(&feature.identity) else {
215                    panic!("feature sorter references a placed feature no biome listed");
216                };
217                let index = features.len();
218                features.push(entry.clone());
219                index_by_identity.insert(feature.identity, index);
220            }
221
222            steps.push(FeatureStepData {
223                features: features.into_boxed_slice(),
224                index_by_identity,
225                feature_indices_by_biome_id: FxHashMap::default(),
226            });
227        }
228
229        for source in sources {
230            let Some(biome_id) = source.biome.try_id() else {
231                panic!("possible biome {} is not registered", source.biome.key);
232            };
233
234            for (step, feature_stage) in source.steps.iter().enumerate() {
235                let Some(step_data) = steps.get_mut(step) else {
236                    continue;
237                };
238
239                let mut indices = Vec::with_capacity(feature_stage.len());
240                for entry in feature_stage {
241                    let Some(feature_index) = step_data.feature_index(entry.identity()) else {
242                        panic!(
243                            "a placed feature from biome {} was not included in decoration step {step}",
244                            source.biome.key
245                        );
246                    };
247                    indices.push(feature_index);
248                }
249
250                if indices.is_empty() {
251                    continue;
252                }
253
254                indices.sort_unstable();
255                indices.dedup();
256                step_data
257                    .feature_indices_by_biome_id
258                    .insert(biome_id, indices.into_boxed_slice());
259            }
260        }
261
262        Self {
263            steps: steps.into_boxed_slice(),
264        }
265    }
266}

268impl FeatureStepData {
269    fn feature_index(&self, identity: FeatureIdentity) -> Option<usize> {
270        self.index_by_identity.get(&identity).copied()
271    }
272
273    pub(super) fn feature(&self, index: usize) -> Option<&FeatureEntry> {
274        self.features.get(index)
275    }
276
277    pub(super) fn feature_indices_for_biome(&self, biome_id: usize) -> Option<&[usize]> {
278        self.feature_indices_by_biome_id
279            .get(&biome_id)
280            .map(Box::as_ref)
281    }
