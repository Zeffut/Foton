1//! The exit portal at the center of the End.
2//!
3//! Vanilla parity: `EndPodiumFeature`. The bedrock pillar with the four torches
4//! that the dragon dies over, and the portal home that opens inside it.
5//!
6//! This is not a worldgen feature in the ordinary sense. Nothing places it
7//! during chunk generation: `EnderDragonFight` builds it directly into a live
8//! level, inactive when the world is first entered and active once the dragon
9//! is dead. So it takes a [`World`] rather than a `WorldGenRegion`, and it is
10//! not registered in the configured-feature tables.
11//!
12//! [`location`] is the other half of its job, and the more widely used one: it
13//! is what the dragon, its phases and the fight all aim at.
14

25/// Half-width of the podium's base.
26///
27/// Vanilla parity: `EndPodiumFeature.PODIUM_RADIUS`.
28pub const PODIUM_RADIUS: i32 = 4;
29
30/// Height of the bedrock pillar the portal sits on.
31///
32/// Vanilla parity: `EndPodiumFeature.PODIUM_PILLAR_HEIGHT`.
33pub const PODIUM_PILLAR_HEIGHT: i32 = 4;
34
35/// How far above the origin the podium clears the air out.
36///
37/// Vanilla parity: the `origin.getY() + 32` of the clearing loop.
38const PODIUM_CLEARANCE: i32 = 32;
39
40/// Radius inside which the podium is bedrock and portal.
41///
42/// Vanilla parity: the `closerThan(origin, 2.5)` of `place`.
43const INNER_RIM: f64 = 2.5;
44
45/// Radius inside which the podium is end stone and rim.
46///
47/// Vanilla parity: the `closerThan(origin, 3.5)` of `place`.
48const OUTER_RIM: f64 = 3.5;
49
50/// Where the exit portal stands for a fight based at `origin`.
51///
52/// Vanilla parity: `EndPodiumFeature.getLocation`. `END_PODIUM_LOCATION` is
53/// `BlockPos.ZERO`, so this is the fight origin unchanged -- but every caller
54/// in vanilla goes through this rather than using the origin directly, and the
55/// dragon's phases follow suit.
56#[must_use]
57pub const fn location(origin: BlockPos) -> BlockPos {
58    origin
59}
60
61/// Builds the podium and its portal at `origin`.
62///
63/// `active` is vanilla's constructor flag: an active podium opens the portal
64/// and breaks whatever is in the way, dropping it; an inactive one leaves the
65/// portal socket as air and only overwrites.
66///
67/// Vanilla parity: `EndPodiumFeature.place`.
68///
69/// [`EnderDragonFight`](crate::dimension::end::EnderDragonFight) is the only
70/// caller, here as in vanilla: it places an inactive podium when the End is
71/// first entered and an active one when the dragon dies.
72pub fn place(world: &Arc<World>, origin: BlockPos, active: bool) {
73    let bedrock = vanilla_blocks::BEDROCK.default_state();
74    let end_stone = vanilla_blocks::END_STONE.default_state();
75    let air = vanilla_blocks::AIR.default_state();
76    let end_portal = vanilla_blocks::END_PORTAL.default_state();
77
78    for x in origin.x() - PODIUM_RADIUS..=origin.x() + PODIUM_RADIUS {
79        for y in origin.y() - 1..=origin.y() + PODIUM_CLEARANCE {
80            for z in origin.z() - PODIUM_RADIUS..=origin.z() + PODIUM_RADIUS {
81                let pos = BlockPos::new(x, y, z);
82                let inside_rim = closer_than(pos, origin, INNER_RIM);
83                if !inside_rim && !closer_than(pos, origin, OUTER_RIM) {
84                    continue;
85                }
86
87                if pos.y() < origin.y() {
88                    if inside_rim {
89                        world.set_block_state(pos, bedrock, UpdateFlags::UPDATE_ALL);
90                    } else if active {
91                        drop_previous_and_set_block(world, pos, end_stone);
92                    } else {
93                        world.set_block_state(pos, end_stone, UpdateFlags::UPDATE_ALL);
94                    }
95                } else if pos.y() > origin.y() {
96                    if active {
97                        drop_previous_and_set_block(world, pos, air);
98                    } else {
99                        world.set_block_state(pos, air, UpdateFlags::UPDATE_ALL);
100                    }
101                } else if !inside_rim {
102                    world.set_block_state(pos, bedrock, UpdateFlags::UPDATE_ALL);
103                } else if active {
104                    drop_previous_and_set_block(world, pos, end_portal);
105                } else {
106                    world.set_block_state(pos, air, UpdateFlags::UPDATE_ALL);
107                }
108            }
109        }
110    }
111
112    for y in 0..PODIUM_PILLAR_HEIGHT {
113        world.set_block_state(origin.above_n(y), bedrock, UpdateFlags::UPDATE_ALL);
114    }
115
116    let center_of_pillar = origin.above_n(2);
117    for face in Direction::HORIZONTAL {
118        let torch = vanilla_blocks::WALL_TORCH
119            .default_state()
120            .set_value(&BlockStateProperties::HORIZONTAL_FACING, face);
121        world.set_block_state(
122            center_of_pillar.relative(face),
123            torch,
124            UpdateFlags::UPDATE_ALL,
125        );
126    }
127}

129/// Vanilla parity: `EndPodiumFeature.dropPreviousAndSetBlock`. An active podium
130/// breaks what it replaces rather than deleting it, so the obsidian a player
131/// piled on the portal comes back to them.
132fn drop_previous_and_set_block(
133    world: &Arc<World>,
134    pos: BlockPos,
135    state: foton_utils::BlockStateId,
136) {
137    if world.get_block_state(pos).get_block() == state.get_block() {
138        return;
139    }
140    world.destroy_block(pos, true);
141    world.set_block_state(pos, state, UpdateFlags::UPDATE_ALL);
142}

144/// Vanilla `Vec3i.closerThan(Vec3i, double)` -- center-to-center.
145fn closer_than(pos: BlockPos, other: BlockPos, distance: f64) -> bool {
146    let dx = f64::from(pos.x() - other.x());
147    let dy = f64::from(pos.y() - other.y());
148    let dz = f64::from(pos.z() - other.z());
149    dx.mul_add(dx, dy.mul_add(dy, dz * dz)) < distance * distance
150}

152#[cfg(test)]
153mod tests {
154    use super::*;
155
156    #[test]
157    fn the_portal_stands_where_the_fight_is_centered() {
158        assert_eq!(location(BlockPos::new(0, 0, 0)), BlockPos::new(0, 0, 0));
159        assert_eq!(
160            location(BlockPos::new(96, 0, -32)),
161            BlockPos::new(96, 0, -32)
162        );
163    }
164
165    #[test]
166    fn the_rim_is_a_ring_one_block_wide_around_the_portal_socket() {
167        let origin = BlockPos::new(0, 64, 0);
168        // Vanilla's two radii differ by exactly one block, which is what makes
169        // the bedrock rim a single ring rather than a slab.
170        let inner = BlockPos::new(2, 64, 0);
171        let rim = BlockPos::new(3, 64, 0);
172        let outside = BlockPos::new(4, 64, 0);
173
174        assert!(closer_than(inner, origin, INNER_RIM));
175        assert!(!closer_than(rim, origin, INNER_RIM));
176        assert!(closer_than(rim, origin, OUTER_RIM));
177        assert!(!closer_than(outside, origin, OUTER_RIM));
178    }
