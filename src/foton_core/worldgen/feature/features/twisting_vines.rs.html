4impl FeatureDecorationRunner {
5    /// Takes any `LevelAccessor` rather than only a worldgen region: bone meal
6    /// on warped nylium places this feature in a live world.
7    pub(crate) fn place_twisting_vines_feature(
8        level: &impl LevelAccessor,
9        random: &mut WorldgenRandom,
10        config: &TwistingVinesConfiguration,
11        origin: BlockPos,
12    ) -> bool {
13        if Self::twisting_vines_invalid_placement_location(level, origin) {
14            return false;
15        }
16
17        for _ in 0..config.spread_width * config.spread_width {
18            let mut place_pos = origin.offset(
19                random.next_i32_between(-config.spread_width, config.spread_width),
20                random.next_i32_between(-config.spread_height, config.spread_height),
21                random.next_i32_between(-config.spread_width, config.spread_width),
22            );
23
24            if Self::find_first_air_block_above_ground(level, &mut place_pos)
25                && !Self::twisting_vines_invalid_placement_location(level, place_pos)
26            {
27                let mut vine_height = random.next_i32_between(1, config.max_height);
28                if random.next_i32_bounded(6) == 0 {
29                    vine_height *= 2;
30                }
31
32                if random.next_i32_bounded(5) == 0 {
33                    vine_height = 1;
34                }
35
36                Self::place_twisting_vines_column(level, random, place_pos, vine_height, 17, 25);
37            }
38        }
39
40        true
41    }
42
43    fn find_first_air_block_above_ground(
44        level: &impl LevelReader,
45        place_pos: &mut BlockPos,
46    ) -> bool {
47        loop {
48            *place_pos = place_pos.below();
49            if level.is_outside_build_height(place_pos.y()) {
50                return false;
51            }
52
53            if !level.get_block_state(*place_pos).is_air() {
54                *place_pos = place_pos.above();
55                return true;
56            }
57        }
58    }
59
60    fn place_twisting_vines_column(
61        level: &impl LevelAccessor,
62        random: &mut WorldgenRandom,
63        mut place_pos: BlockPos,
64        total_height: i32,
65        min_age: u8,
66        max_age: u8,
67    ) {
68        for height in 1..=total_height {
69            if level.get_block_state(place_pos).is_air() {
70                if height == total_height || !level.get_block_state(place_pos.above()).is_air() {
71                    let state = vanilla_blocks::TWISTING_VINES.default_state().set_value(
72                        &BlockStateProperties::AGE_25,
73                        random.next_i32_between(i32::from(min_age), i32::from(max_age)) as u8,
74                    );
75                    let _ = level.set_block_state(place_pos, state, UpdateFlags::UPDATE_CLIENTS);
76                    break;
77                }
78
79                let _ = level.set_block_state(
80                    place_pos,
81                    vanilla_blocks::TWISTING_VINES_PLANT.default_state(),
82                    UpdateFlags::UPDATE_CLIENTS,
83                );
84            }
85
86            place_pos = place_pos.above();
87        }
88    }
89
90    fn twisting_vines_invalid_placement_location(level: &impl LevelReader, pos: BlockPos) -> bool {
91        if !level.get_block_state(pos).is_air() {
92            return true;
93        }
94
95        let below_block = level.get_block_state(pos.below()).get_block();
96        below_block != &vanilla_blocks::NETHERRACK
97            && below_block != &vanilla_blocks::WARPED_NYLIUM
98            && below_block != &vanilla_blocks::WARPED_WART_BLOCK
99    }
