1//! Creaking heart block entity.
2//!
3//! Vanilla parity: `CreakingHeartBlockEntity`. The heart is the whole of the
4//! creaking's life cycle: it wakes one at night while a player is near, keeps
5//! it on a thirty-four block tether, takes every blow the creaking is dealt --
6//! spreading resin and creaking audibly as it does -- and tears the creaking
7//! down when it is broken, when the night ends, or when the creaking has wedged
8//! itself inside a player.
9//!
10//! Vanilla holds the protector as `Either<Creaking, UUID>`: the left branch is
11//! only a resolved cache of the right, and every read falls back to looking the
12//! UUID up in the level. Foton keeps the UUID alone and does that lookup, which
13//! is the same behavior with one field instead of two -- including the thirty
14//! tick grace period after a load, during which an unresolvable UUID is kept
15//! rather than dropped, because the creaking's chunk may not be back yet.
16

155    /// Vanilla `CreakingHeartBlockEntity.getCreakingProtector`.
156    ///
157    /// A UUID that will not resolve is kept for the first thirty ticks of the
158    /// block entity's life and dropped after that, so a heart loaded before its
159    /// creaking's chunk does not orphan itself.
160    #[must_use]
161    fn creaking_protector(&self, world: &Arc<World>) -> Option<SharedEntity> {
162        let creaking = self.state.lock().creaking?;
163        let resolved = world.get_entity_by_uuid(&creaking).filter(|entity| {
164            entity.downcast_ref::<CreakingEntity>().is_some() && !entity.is_removed()
165        });
166        if resolved.is_some() {
167            return resolved;
168        }
169
170        if self.state.lock().ticks_existed >= TICKS_GRACE_PERIOD {
171            self.clear_creaking_info();
172        }
173        None
174    }

195    /// Vanilla `CreakingHeartBlockEntity.computeAnalogOutputSignal`, which
196    /// falls from fifteen at the heart to zero at the edge of the roam radius.
197    #[must_use]
198    fn compute_analog_output_signal(&self, world: &Arc<World>) -> i32 {
199        if self.creaking_protector(world).is_none() {
200            return 0;
201        }
202        let scaled = self
203            .distance_to_creaking(world)
204            .clamp(0.0, CREAKING_ROAMING_RADIUS)
205            / CREAKING_ROAMING_RADIUS;
206        #[expect(
207            clippy::cast_possible_truncation,
208            reason = "a comparator level, between zero and fifteen"
209        )]
210        let steps = (scaled * MAX_SIGNAL).floor() as i32;
211        15 - steps
212    }

251    /// Vanilla `CreakingHeartBlockEntity.creakingHurt`.
252    ///
253    /// The creaking itself takes no damage; this is where the blow actually
254    /// lands. Two or three resin clumps grow out of the tree while the heart is
255    /// awake, and the hundred-tick emitter that follows is the trail of
256    /// particles and the repeated hurt noise travelling from the heart out to
257    /// the creaking.
258    pub fn creaking_hurt(&self) {
259        let Some(world) = self.get_level() else {
260            return;
261        };
262        let Some(creaking) = self.creaking_protector(&world) else {
263            return;
264        };
265        if self.state.lock().emitter > 0 {
266            return;
267        }
268
269        self.emit_particles(&world, HURT_PARTICLE_COUNT, false);
270        if self.get_block_state().get_value(CREAKING_HEART_STATE) == CreakingHeartState::Awake {
271            let clumps = rand::rng().random_range(2..=3);
272            for _ in 0..clumps {
273                let Some(placed) = self.spread_resin(&world) else {
274                    continue;
275                };
276                world.play_sound(
277                    &sound_events::BLOCK_RESIN_PLACE,
278                    SoundSource::Blocks,
279                    placed,
280                    1.0,
281                    1.0,
282                    None,
283                );
284                world.game_event(
285                    &vanilla_game_events::BLOCK_PLACE,
286                    placed,
287                    &GameEventContext::new(None, Some(self.get_block_state())),
288                );
289            }
290        }
291
292        let mut state = self.state.lock();
293        state.emitter = HURT_CALL_TOTAL_TICKS;
294        state.emitter_target = Some(bounding_box_center(&creaking));
295    }

297    /// Vanilla `CreakingHeartBlockEntity.spreadResin`.
298    ///
299    /// A breadth-first walk out along the pale oak logs the heart is buried in,
300    /// looking for the first face with room for a resin clump. Depth two and
301    /// sixty-four nodes are vanilla's limits, and they are what keeps the resin
302    /// on the tree rather than crawling across a forest.
303    fn spread_resin(&self, world: &Arc<World>) -> Option<BlockPos> {
304        let start = self.get_block_pos();
305        let mut visited = vec![start];
306        let mut queue = vec![(start, 0_u32)];
307        let mut head = 0;
308
309        while head < queue.len() && visited.len() <= RESIN_MAX_COUNT {
310            let (pos, depth) = queue[head];
311            head += 1;
312
313            if world
314                .get_block_state(pos)
315                .get_block()
316                .has_tag(&BlockTag::PALE_OAK_LOGS)
317                && let Some(placed) = try_place_resin_on(world, pos)
318            {
319                return Some(placed);
320            }
321
322            if depth >= RESIN_MAX_DEPTH {
323                continue;
324            }
325            for direction in shuffled_directions() {
326                let neighbor = pos.relative(direction);
327                if visited.contains(&neighbor) {
328                    continue;
329                }
330                if !world
331                    .get_block_state(neighbor)
332                    .get_block()
333                    .has_tag(&BlockTag::PALE_OAK_LOGS)
334                {
335                    continue;
336                }
337                visited.push(neighbor);
338                queue.push((neighbor, depth + 1));
339            }
340        }
341
342        None
343    }

345    /// Vanilla `CreakingHeartBlockEntity.emitParticles`.
346    fn emit_particles(&self, world: &Arc<World>, count: i32, towards_creaking: bool) {
347        let Some(creaking) = self.creaking_protector(world) else {
348            return;
349        };
350        let color = if towards_creaking {
351            CREAKING_ORANGE
352        } else {
353            CREAKING_GRAY
354        };
355        let bounds = creaking.bounding_box();
356        let corner = self.get_block_pos();
357
358        for _ in 0..count {
359            let on_creaking = DVec3::new(
360                bounds
361                    .width()
362                    .mul_add(rand::random::<f64>(), bounds.min_x()),
363                bounds
364                    .height()
365                    .mul_add(rand::random::<f64>(), bounds.min_y()),
366                bounds
367                    .depth()
368                    .mul_add(rand::random::<f64>(), bounds.min_z()),
369            );
370            let on_heart = DVec3::new(
371                f64::from(corner.x()) + rand::random::<f64>(),
372                f64::from(corner.y()) + rand::random::<f64>(),
373                f64::from(corner.z()) + rand::random::<f64>(),
374            );
375            let (source, destination) = if towards_creaking {
376                (on_heart, on_creaking)
377            } else {
378                (on_creaking, on_heart)
379            };
380
381            world.send_particles_with_options(
382                ParticleData::new(
383                    &vanilla_particle_types::TRAIL,
384                    TrailParticleOption::new(
385                        destination,
386                        RgbColor::new(color),
387                        rand::random_range(0..40) + 10,
388                    ),
389                ),
390                true,
391                true,
392                source,
393                1,
394                DVec3::ZERO,
395                0.0,
396            );
397        }
398    }

400    /// Vanilla `CreakingHeartBlockEntity.removeProtector`.
401    ///
402    /// With no damage source the creaking simply crumbles; with one it dies
403    /// properly first, so whoever broke the heart is credited with the kill.
404    pub fn remove_protector(&self, source: Option<&DamageSource>) {
405        let Some(world) = self.get_level() else {
406            return;
407        };
408        let Some(creaking) = self.creaking_protector(&world) else {
409            return;
410        };
411
412        let Some(creaking) = as_creaking(&creaking) else {
413            return;
414        };
415        match source {
416            None => creaking.tear_down(),
417            Some(source) => {
418                creaking.creaking_death_effects(source);
419                creaking.set_tearing_down();
420                creaking.set_health(0.0);
421            }
422        }
423        self.clear_creaking_info();
424    }

426    /// Vanilla `CreakingHeartBlockEntity.updateCreakingState`.
427    ///
428    /// A heart that has lost its logs only uproots once it has no creaking left
429    /// to keep alive -- breaking the tree out from under a woken heart does not
430    /// free you of what it has already sent.
431    fn updated_state(
432        &self,
433        state: BlockStateId,
434        world: &Arc<World>,
435        pos: BlockPos,
436    ) -> BlockStateId {
437        if !CreakingHeartBlock::has_required_logs(state, world.as_ref(), pos)
438            && self.state.lock().creaking.is_none()
439        {
440            return state.set_value(CREAKING_HEART_STATE, CreakingHeartState::Uprooted);
441        }
442
443        state.set_value(
444            CREAKING_HEART_STATE,
445            creaking_heart_awake_or_dormant(world.as_ref()),
446        )
447    }

449    /// Vanilla parity: the `emitter` block of `CreakingHeartBlockEntity.serverTick`.
450    fn tick_emitter(&self, world: &Arc<World>) {
451        let (emitter, emitter_target) = {
452            let state = self.state.lock();
453            (state.emitter, state.emitter_target)
454        };
455        if emitter <= 0 {
456            return;
457        }
458
459        if emitter > HURT_CALL_PARTICLE_TICKS {
460            self.emit_particles(world, 1, true);
461            self.emit_particles(world, 1, false);
462        }
463
464        if emitter % HURT_CALL_INTERVAL == 0
465            && let Some(target) = emitter_target
466        {
467            // Vanilla re-reads the creaking's center so the noise follows a
468            // creaking that has moved since the blow landed.
469            let target = self
470                .creaking_protector(world)
471                .map_or(target, |creaking| bounding_box_center(&creaking));
472            self.state.lock().emitter_target = Some(target);
473
474            let (x, y, z) = self.get_block_pos().get_center();
475            let heart = DVec3::new(x, y, z);
476            #[expect(
477                clippy::cast_precision_loss,
478                reason = "a countdown below a hundred, used as an interpolation factor"
479            )]
480            let progress = 0.8f32.mul_add((HURT_CALL_TOTAL_TICKS - emitter) as f32 / 100.0, 0.2);
481            let sound_position = (heart - target) * f64::from(progress) + target;
482            let sound_pos =
483                BlockPos::containing(sound_position.x, sound_position.y, sound_position.z);
484            #[expect(
485                clippy::cast_precision_loss,
486                reason = "a countdown below a hundred, used as a volume"
487            )]
488            let volume = emitter as f32 / 2.0 / 100.0 + 0.5;
489            world.play_sound(
490                &sound_events::BLOCK_CREAKING_HEART_HURT,
491                SoundSource::Blocks,
492                sound_pos,
493                volume,
494                1.0,
495                None,
496            );
497        }
498
499        self.state.lock().emitter -= 1;
500    }

502    /// Vanilla parity: the twenty-tick block of `serverTick` that wakes, keeps
503    /// or drops the protector.
504    fn tick_protector(&self, world: &Arc<World>, pos: BlockPos) {
505        let state = self.get_block_state();
506        let updated = self.updated_state(state, world, pos);
507        if updated != state {
508            world.set_block(pos, updated, UpdateFlags::UPDATE_ALL);
509            if updated.get_value(CREAKING_HEART_STATE) == CreakingHeartState::Uprooted {
510                return;
511            }
512        }
513
514        if self.state.lock().creaking.is_none() {
515            if updated.get_value(CREAKING_HEART_STATE) != CreakingHeartState::Awake {
516                return;
517            }
518            if !world.get_game_rule(&SPAWN_MONSTERS) || world.difficulty() == Difficulty::Peaceful {
519                return;
520            }
521            let (x, y, z) = pos.get_center();
522            // Vanilla's `getNearestPlayer(x, y, z, range, false)` takes any
523            // player that is not a spectator; the `false` is `isCreative`
524            // being allowed through.
525            if world
526                .nearest_player(DVec3::new(x, y, z), PLAYER_DETECTION_RANGE, |player| {
527                    !player.is_spectator()
528                })
529                .is_none()
530            {
531                return;
532            }
533            let Some(creaking) = self.spawn_protector(world) else {
534                return;
535            };
536            self.set_creaking_info(creaking.uuid());
537            if let Some(creaking) = as_creaking(&creaking) {
538                creaking.make_sound(Some(&sound_events::ENTITY_CREAKING_SPAWN));
539            }
540            world.play_sound(
541                &sound_events::BLOCK_CREAKING_HEART_SPAWN,
542                SoundSource::Blocks,
543                pos,
544                1.0,
545                1.0,
546                None,
547            );
548            return;
549        }
550
551        let Some(creaking) = self.creaking_protector(world) else {
552            return;
553        };
554        let Some(creaking) = as_creaking(&creaking) else {
555            return;
556        };
557        let night_is_over = !world.creaking_active() && !creaking.is_persistence_required();
558        if night_is_over
559            || self.distance_to_creaking(world) > DISTANCE_CREAKING_TOO_FAR
560            || creaking.player_is_stuck_in_you()
561        {
562            self.remove_protector(None);
563        }
564    }

577/// Vanilla parity: `Util.shuffledCopy(Direction.values(), random)`.
578fn shuffled_directions() -> [Direction; 6] {
579    let mut directions = [
580        Direction::Down,
581        Direction::Up,
582        Direction::North,
583        Direction::South,
584        Direction::West,
585        Direction::East,
586    ];
587    for index in (1..directions.len()).rev() {
588        directions.swap(index, rand::random_range(0..=index));
589    }
590    directions
591}

593/// Vanilla parity: the inner loop of `spreadResin`, which grows a clump on the
594/// first free face of the log at `pos`.
595fn try_place_resin_on(world: &Arc<World>, pos: BlockPos) -> Option<BlockPos> {
596    for direction in shuffled_directions() {
597        let neighbor_pos = pos.relative(direction);
598        let neighbor_state = world.get_block_state(neighbor_pos);
599        let opposite = direction.opposite();
600
601        let base = if neighbor_state.is_air() {
602            vanilla_blocks::RESIN_CLUMP.default_state()
603        } else if neighbor_state.get_block() == &vanilla_blocks::WATER
604            && neighbor_state.get_fluid_state().is_source()
605        {
606            vanilla_blocks::RESIN_CLUMP
607                .default_state()
608                .set_value(&BlockStateProperties::WATERLOGGED, true)
609        } else {
610            neighbor_state
611        };
612
613        if base.get_block() != &vanilla_blocks::RESIN_CLUMP {
614            continue;
615        }
616        let face = multiface_face_property(opposite);
617        if base.get_value(face) {
618            continue;
619        }
620
621        world.set_block(
622            neighbor_pos,
623            base.set_value(face, true),
624            UpdateFlags::UPDATE_ALL,
625        );
626        return Some(neighbor_pos);
627    }
628    None
629}
630
631impl BlockEntity for CreakingHeartBlockEntity {
632    fn base(&self) -> &BlockEntityBase {
633        &self.base
634    }
635
636    /// Vanilla `CreakingHeartBlockEntity.serverTick`.
637    fn tick(&self, world: &Arc<World>) {
638        self.state.lock().ticks_existed += 1;
639
640        let pos = self.get_block_pos();
641        let computed = self.compute_analog_output_signal(world);
642        let changed = {
643            let mut state = self.state.lock();
644            let changed = state.output_signal != computed;
645            state.output_signal = computed;
646            changed
647        };
648        if changed {
649            world.update_neighbor_for_output_signal(pos, &vanilla_blocks::CREAKING_HEART);
650        }
651
652        self.tick_emitter(world);
653
654        {
655            let mut state = self.state.lock();
656            state.ticker -= 1;
657            if state.ticker >= 0 {
658                return;
659            }
660            state.ticker = rand::rng().random_range(0..UPDATE_TICKS_VARIANCE) + UPDATE_TICKS;
661        }
662
663        self.tick_protector(world, pos);
664    }
665
666    /// Vanilla `CreakingHeartBlockEntity.preRemoveSideEffects`, which is what
667    /// makes breaking the heart the way to be rid of the creaking.
668    fn pre_remove_side_effects(&self, _pos: BlockPos, _state: BlockStateId) {
669        self.remove_protector(None);
670    }
671
672    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
673        let view: NbtCompoundView<'_, '_> = nbt.into();
674        let creaking = view
675            .int_array("creaking")
676            .and_then(|array| Uuid::from_int_array(&array));
677        match creaking {
678            Some(creaking) => self.set_creaking_info(creaking),
679            None => self.clear_creaking_info(),
680        }
681    }
682
683    fn save_additional(&self, nbt: &mut NbtCompound) {
684        if let Some(creaking) = self.state.lock().creaking {
685            nbt.insert(
686                "creaking",
687                NbtTag::IntArray(creaking.to_int_array().to_vec()),
688            );
689        }
690    }
691
692    fn get_update_tag(&self) -> Option<NbtCompound> {
693        Some(self.save_custom_only())
694    }
695}
