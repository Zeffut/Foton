1//! The jigsaw block block entity.
2//!
3//! Vanilla parity: `JigsawBlockEntity`. Seven fields and nothing else: the name
4//! other jigsaws aim at, the name this one aims at, the template pool it draws
5//! from, the state it leaves behind, whether a connected piece may roll, and the
6//! two priorities that order a pool's candidates.
7//!
8//! Foton's worldgen already reads all of this out of the *template* copies of
9//! jigsaw blocks when it builds a village; this is the placed block, a jigsaw a
10//! map-maker puts down by hand and configures through its editor.
11//!
12//! `generate` runs the assembly in
13//! [`crate::worldgen::structure::generate_jigsaw`], which is
14//! `JigsawPlacement.generateJigsaw`.
15

31/// The identifier an unconfigured jigsaw uses for its name, target and pool.
32///
33/// Vanilla parity: `JigsawBlockEntity.EMPTY_ID` and `Pools.EMPTY`.
34const EMPTY_ID: &str = "empty";
35
36/// The state an unconfigured jigsaw leaves behind.
37///
38/// Vanilla parity: `JigsawBlockEntity.DEFAULT_FINAL_STATE`.
39const DEFAULT_FINAL_STATE: &str = "minecraft:air";
40
41/// Whether a piece connected here may roll around the joint axis.
42///
43/// Vanilla parity: `JigsawBlockEntity.JointType`.
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub enum JigsawJointType {
46    /// The connected piece may be rotated freely about the joint.
47    Rollable,
48    /// The connected piece keeps its up direction.
49    Aligned,
50}
51
52impl JigsawJointType {
53    /// Returns the joint a serialized name describes.
54    #[must_use]
55    pub fn from_name(name: &str) -> Option<Self> {
56        match name {
57            "rollable" => Some(Self::Rollable),
58            "aligned" => Some(Self::Aligned),
59            _ => None,
60        }
61    }
62
63    /// Returns the serialized name of this joint.
64    #[must_use]
65    pub const fn name(self) -> &'static str {
66        match self {
67            Self::Rollable => "rollable",
68            Self::Aligned => "aligned",
69        }
70    }
71}

73/// The default joint for a jigsaw in `state`.
74///
75/// Vanilla parity: `StructureTemplate.getDefaultJointType`, which a jigsaw
76/// falls back to when its NBT names no joint: a jigsaw pointing up or down is
77/// rollable, because there is no meaningful up for the piece it connects.
78#[must_use]
79pub fn default_joint_type(state: BlockStateId) -> JigsawJointType {
80    let Some(orientation) = state.try_get_value(&BlockStateProperties::ORIENTATION) else {
81        return JigsawJointType::Aligned;
82    };
83    if orientation.front().axis() == Axis::Y {
84        JigsawJointType::Rollable
85    } else {
86        JigsawJointType::Aligned
87    }
88}

90/// The seven fields a jigsaw block remembers.
91#[derive(Debug, Clone)]
92struct JigsawState {
93    name: Identifier,
94    target: Identifier,
95    pool: Identifier,
96    final_state: String,
97    joint: JigsawJointType,
98    placement_priority: i32,
99    selection_priority: i32,
100}

102/// Everything a jigsaw editor sends in one go.
103///
104/// Vanilla passes these as seven separate setter calls; one struct keeps the
105/// two same-typed priorities from being swapped by position.
106pub struct JigsawSettings {
107    /// The name other jigsaws aim at to connect here.
108    pub name: Identifier,
109    /// The name this jigsaw aims at.
110    pub target: Identifier,
111    /// The template pool this jigsaw draws its next piece from.
112    pub pool: Identifier,
113    /// The block state left behind once the jigsaw has been used.
114    pub final_state: String,
115    /// Whether a connected piece may roll around the joint axis.
116    pub joint: JigsawJointType,
117    /// How early this jigsaw is chosen among its pool's candidates.
118    pub selection_priority: i32,
119    /// How early the piece it places is expanded.
120    pub placement_priority: i32,
121}

123/// A jigsaw block.
124pub struct JigsawBlockEntity {
125    base: BlockEntityBase,
126    state: SyncMutex<JigsawState>,
127}

134impl JigsawBlockEntity {
135    /// Creates a jigsaw block entity.
136    #[must_use]
137    pub fn new(level: Weak<World>, pos: BlockPos, block_state: BlockStateId) -> Self {
138        Self {
139            base: BlockEntityBase::new(
140                &vanilla_block_entity_types::JIGSAW,
141                level,
142                pos,
143                block_state,
144            ),
145            state: SyncMutex::new(JigsawState {
146                name: Identifier::vanilla_static(EMPTY_ID),
147                target: Identifier::vanilla_static(EMPTY_ID),
148                pool: Identifier::vanilla_static(EMPTY_ID),
149                final_state: DEFAULT_FINAL_STATE.to_owned(),
150                joint: JigsawJointType::Rollable,
151                placement_priority: 0,
152                selection_priority: 0,
153            }),
154        }
155    }
156
157    /// Returns the name other jigsaws aim at to connect here.
158    #[must_use]
159    pub fn name(&self) -> Identifier {
160        self.state.lock().name.clone()
161    }
162
163    /// Returns the name this jigsaw aims at.
164    #[must_use]
165    pub fn target(&self) -> Identifier {
166        self.state.lock().target.clone()
167    }
168
169    /// Returns the template pool this jigsaw draws its next piece from.
170    #[must_use]
171    pub fn pool(&self) -> Identifier {
172        self.state.lock().pool.clone()
173    }
174
175    /// Returns the block state left behind once the jigsaw has been used.
176    #[must_use]
177    pub fn final_state(&self) -> String {
178        self.state.lock().final_state.clone()
179    }
180
181    /// Returns whether a connected piece may roll around the joint axis.
182    #[must_use]
183    pub fn joint(&self) -> JigsawJointType {
184        self.state.lock().joint
185    }
186
187    /// Returns how early the piece this jigsaw places is expanded.
188    #[must_use]
189    pub fn placement_priority(&self) -> i32 {
190        self.state.lock().placement_priority
191    }
192
193    /// Returns how early this jigsaw is chosen among its pool's candidates.
194    #[must_use]
195    pub fn selection_priority(&self) -> i32 {
196        self.state.lock().selection_priority
197    }
198
199    /// Assembles this jigsaw's pool into the world and places the pieces.
200    ///
201    /// Vanilla parity: `JigsawBlockEntity.generate`. The assembly starts at the
202    /// block this jigsaw faces, at that block's own Y -- not at a chunk corner
203    /// and not at a sampled start height. Returns whether an assembly was found.
204    pub fn generate(&self, world: &Arc<World>, levels: i32, keep_jigsaws: bool) -> bool {
205        let Some(orientation) = self
206            .base
207            .block_state()
208            .try_get_value(&BlockStateProperties::ORIENTATION)
209        else {
210            return false;
211        };
212        let position = self.base.pos().relative(orientation.front());
213        // The lock is released before the placement runs: placing pieces writes
214        // block entities back into the same world.
215        let (pool, target) = {
216            let state = self.state.lock();
217            (state.pool.clone(), state.target.clone())
218        };
219
220        generate_jigsaw(world, &pool, &target, levels, position, keep_jigsaws)
221    }
222
223    /// Stores everything a jigsaw editor sends at once.
224    ///
225    /// Vanilla parity: the seven setters `handleSetJigsawBlock` calls in a row.
226    pub fn configure(&self, settings: JigsawSettings) {
227        {
228            let mut state = self.state.lock();
229            state.name = settings.name;
230            state.target = settings.target;
231            state.pool = settings.pool;
232            state.final_state = settings.final_state;
233            state.joint = settings.joint;
234            state.selection_priority = settings.selection_priority;
235            state.placement_priority = settings.placement_priority;
236        }
237        self.base.set_changed();
238    }
239}

241impl BlockEntity for JigsawBlockEntity {
242    fn base(&self) -> &BlockEntityBase {
243        &self.base
244    }
245
246    fn save_additional(&self, nbt: &mut NbtCompound) {
247        let state = self.state.lock();
248        nbt.insert("name", state.name.to_string());
249        nbt.insert("target", state.target.to_string());
250        nbt.insert("pool", state.pool.to_string());
251        nbt.insert("final_state", state.final_state.clone());
252        nbt.insert("joint", state.joint.name());
253        nbt.insert("placement_priority", state.placement_priority);
254        nbt.insert("selection_priority", state.selection_priority);
255    }
256
257    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
258        let view: NbtCompoundView<'_, '_> = nbt.into();
259        let mut state = self.state.lock();
260        state.name = identifier_or_empty(&view, "name");
261        state.target = identifier_or_empty(&view, "target");
262        state.pool = identifier_or_empty(&view, "pool");
263        state.final_state = view
264            .string("final_state")
265            .map_or_else(|| DEFAULT_FINAL_STATE.to_owned(), ToString::to_string);
266        // Vanilla parity: a jigsaw with no stored joint takes the one its
267        // orientation implies, not a flat default.
268        state.joint = view
269            .string("joint")
270            .and_then(|value| JigsawJointType::from_name(&value.to_str()))
271            .unwrap_or_else(|| default_joint_type(self.base.block_state()));
272        state.placement_priority = view.int("placement_priority").unwrap_or(0);
273        state.selection_priority = view.int("selection_priority").unwrap_or(0);
274    }
275
276    /// Vanilla parity: `JigsawBlockEntity.getUpdateTag`, which is
277    /// `saveCustomOnly` -- the editor is filled from the same shape the chunk
278    /// packet carries.
279    fn get_update_tag(&self) -> Option<NbtCompound> {
280        let mut nbt = NbtCompound::new();
281        self.save_additional(&mut nbt);
282        Some(nbt)
283    }
284}

286/// Reads an identifier field, falling back to `minecraft:empty`.
287fn identifier_or_empty(nbt: &NbtCompoundView<'_, '_>, key: &str) -> Identifier {
288    nbt.string(key)
289        .and_then(|value| value.to_str().parse::<Identifier>().ok())
290        .unwrap_or_else(|| Identifier::vanilla_static(EMPTY_ID))
291}

303    fn jigsaw(state: BlockStateId) -> JigsawBlockEntity {
304        JigsawBlockEntity::new(Weak::new(), BlockPos::new(8, 64, 8), state)

307    fn reload(state: BlockStateId, nbt: &NbtCompound) -> JigsawBlockEntity {
308        let mut bytes = Vec::new();
309        nbt.write(&mut bytes);
310        let borrowed = read_borrowed_compound(&mut Cursor::new(&bytes))
311            .unwrap_or_else(|error| panic!("test nbt should reborrow: {error}"));
312        let entity = jigsaw(state);
313        entity.load_additional(&borrowed);
314        entity
315    }

317    fn oriented(front_and_top: FrontAndTop) -> BlockStateId {
318        vanilla_blocks::JIGSAW
319            .default_state()
320            .set_value(&BlockStateProperties::ORIENTATION, front_and_top)
321    }

323    /// Everything a map-maker types into the editor has to survive a save, and
324    /// the two priorities in particular are easy to swap.
325    #[test]
326    fn every_configured_field_round_trips() {
327        init_vanilla_registry();
328        let entity = jigsaw(oriented(FrontAndTop::NorthUp));
329        entity.configure(JigsawSettings {
330            name: Identifier::vanilla_static("street"),
331            target: Identifier::vanilla_static("house"),
332            pool: Identifier::vanilla_static("village/plains/houses"),
333            final_state: "minecraft:stone".to_owned(),
334            joint: JigsawJointType::Aligned,
335            selection_priority: 7,
336            placement_priority: 3,
337        });
338
339        let mut nbt = NbtCompound::new();
340        entity.save_additional(&mut nbt);
341        assert_eq!(nbt.int("selection_priority"), Some(7));
342        assert_eq!(nbt.int("placement_priority"), Some(3));
343
344        let reloaded = reload(oriented(FrontAndTop::NorthUp), &nbt);
345        assert_eq!(reloaded.name(), Identifier::vanilla_static("street"));
346        assert_eq!(reloaded.target(), Identifier::vanilla_static("house"));
347        assert_eq!(
348            reloaded.pool(),
349            Identifier::vanilla_static("village/plains/houses")
350        );
351        assert_eq!(reloaded.final_state(), "minecraft:stone");
352        assert_eq!(reloaded.joint(), JigsawJointType::Aligned);
353        assert_eq!(reloaded.selection_priority(), 7);
354        assert_eq!(reloaded.placement_priority(), 3);
355    }

357    /// A jigsaw with no stored joint takes the one its orientation implies: a
358    /// vertical jigsaw is rollable because a piece hanging off it has no
359    /// meaningful up, and a horizontal one is aligned.
360    #[test]
361    fn a_missing_joint_comes_from_the_orientation() {
362        init_vanilla_registry();
363        assert_eq!(
364            reload(oriented(FrontAndTop::UpNorth), &NbtCompound::new()).joint(),
365            JigsawJointType::Rollable
366        );
367        assert_eq!(
368            reload(oriented(FrontAndTop::DownNorth), &NbtCompound::new()).joint(),
369            JigsawJointType::Rollable
370        );
371        assert_eq!(
372            reload(oriented(FrontAndTop::NorthUp), &NbtCompound::new()).joint(),
373            JigsawJointType::Aligned
374        );
375    }
