1//! The lectern block entity.
2//!
3//! Vanilla parity: `LecternBlockEntity`. It holds one book and remembers which
4//! page is open, and that page is the whole redstone story: turning it pulses
5//! the block and a comparator reads how far through the book the reader is.
6

24/// The strongest signal a lectern's comparator gives.
25///
26/// Vanilla parity: the `pageProgress * 14 + 1` of
27/// `LecternBlockEntity.getRedstoneSignal` -- a book open at its first page
28/// already reads 1, and only the last page reads 15.
29const SIGNAL_RANGE: f32 = 14.0;
30
31/// The lectern.
32pub struct LecternBlockEntity {
33    base: BlockEntityBase,
34    state: SyncMutex<LecternState>,
35}
36
37// SAFETY: This key is owned by Foton and uniquely identifies the block entity.
38unsafe impl DowncastType for LecternBlockEntity {
39    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:block_entity/lectern");
40}
41
42struct LecternState {
43    book: ItemStack,
44    page: i32,
45    page_count: i32,
46}
47
48impl LecternBlockEntity {
49    /// Creates a lectern block entity.
50    #[must_use]
51    pub fn new(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
52        Self {
53            base: BlockEntityBase::new(&vanilla_block_entity_types::LECTERN, level, pos, state),
54            state: SyncMutex::new(LecternState {
55                book: ItemStack::empty(),
56                page: 0,
57                page_count: 0,
58            }),
59        }
60    }
61
62    /// Returns a copy of the book on the stand.
63    #[must_use]
64    pub fn book(&self) -> ItemStack {
65        self.state.lock().book.clone()
66    }
67
68    /// Returns whether there is a readable book on the stand.
69    ///
70    /// Vanilla parity: `LecternBlockEntity.hasBook`, which asks whether the
71    /// item has book *content* rather than whether the slot is occupied -- an
72    /// empty writable book still counts.
73    #[must_use]
74    pub fn has_book(&self) -> bool {
75        let state = self.state.lock();
76        state.book.has(WRITABLE_BOOK_CONTENT) || state.book.has(WRITTEN_BOOK_CONTENT)
77    }
78
79    /// Puts a book on the stand, open at the first page.
80    pub fn set_book(&self, book: ItemStack) {
81        let page_count = page_count_of(&book);
82        let mut state = self.state.lock();
83        state.book = book;
84        state.page = 0;
85        state.page_count = page_count;
86        drop(state);
87        self.base.set_changed();
88    }
89
90    /// Takes the book off the stand and returns it.
91    #[must_use]
92    pub fn take_book(&self) -> ItemStack {
93        let mut state = self.state.lock();
94        let book = mem::replace(&mut state.book, ItemStack::empty());
95        state.page = 0;
96        state.page_count = 0;
97        drop(state);
98        self.base.set_changed();
99        book
100    }
101
102    /// Returns the open page.
103    #[must_use]
104    pub fn page(&self) -> i32 {
105        self.state.lock().page
106    }
107
108    /// Turns to `page`, and reports whether it actually moved.
109    ///
110    /// Vanilla parity: `LecternBlockEntity.setPage`, which clamps to the book
111    /// and only signals when the page really changed -- clicking "next" on the
112    /// last page must not pulse the redstone.
113    pub fn set_page(&self, page: i32) -> bool {
114        let mut state = self.state.lock();
115        let clamped = page.clamp(0, (state.page_count - 1).max(0));
116        if clamped == state.page {
117            return false;
118        }
119        state.page = clamped;
120        drop(state);
121        self.base.set_changed();
122        true
123    }
124
125    /// Returns what a comparator reads off the lectern.
126    ///
127    /// Vanilla parity: `LecternBlockEntity.getRedstoneSignal`.
128    #[must_use]
129    pub fn redstone_signal(&self) -> i32 {
130        let has_book = self.has_book();
131        let state = self.state.lock();
132        let progress = if state.page_count > 1 {
133            state.page as f32 / (state.page_count - 1) as f32
134        } else {
135            1.0
136        };
137        (progress * SIGNAL_RANGE).floor() as i32 + i32::from(has_book)
138    }
139}
140
141/// Returns how many pages a book has.
142///
143/// Vanilla parity: `LecternBlockEntity.getPageCount`.
144fn page_count_of(book: &ItemStack) -> i32 {
145    if let Some(written) = book.get(WRITTEN_BOOK_CONTENT) {
146        return written.pages().len() as i32;
147    }
148    if let Some(writable) = book.get(WRITABLE_BOOK_CONTENT) {
149        return writable.pages().len() as i32;
150    }
151    0
152}

154impl BlockEntity for LecternBlockEntity {
155    fn base(&self) -> &BlockEntityBase {
156        &self.base
157    }
158
159    fn save_additional(&self, nbt: &mut NbtCompound) {
160        let state = self.state.lock();
161        if !state.book.is_empty()
162            && let NbtTag::Compound(book_nbt) = state.book.clone().to_nbt_tag()
163        {
164            nbt.insert("Book", book_nbt);
165            nbt.insert("Page", NbtTag::Int(state.page));
166        }
167    }
168
169    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
170        let view: NbtCompoundView<'_, '_> = nbt.into();
171        let book = view
172            .compound("Book")
173            .and_then(|compound| ItemStack::from_borrowed_compound(&compound))
174            .unwrap_or_else(ItemStack::empty);
175        let page_count = page_count_of(&book);
176        let page = view.int("Page").unwrap_or(0);
177
178        let mut state = self.state.lock();
179        state.page = page.clamp(0, (page_count - 1).max(0));
180        state.page_count = page_count;
181        state.book = book;
182    }
183
184    fn get_update_tag(&self) -> Option<NbtCompound> {
185        None
186    }
187}

189#[cfg(test)]
190mod tests {
191    use std::sync::Weak;
192
193    use foton_registry::data_components::components::{Filterable, WritableBookContent};
194    use foton_registry::{init_vanilla_registry, vanilla_blocks, vanilla_items};
195
196    use super::*;
197
198    /// Builds a lectern with no world behind it.
199    fn detached_lectern() -> LecternBlockEntity {
200        init_vanilla_registry();
201        LecternBlockEntity::new(
202            Weak::new(),
203            BlockPos::new(0, 0, 0),
204            vanilla_blocks::LECTERN.default_state(),
205        )
206    }
207
208    /// Builds a writable book with `pages` blank pages.
209    fn book_with_pages(pages: usize) -> ItemStack {
210        let mut book = ItemStack::new(&vanilla_items::WRITABLE_BOOK);
211        let content = WritableBookContent::new(
212            (0..pages)
213                .map(|index| Filterable::new(format!("page {index}"), None))
214                .collect(),
215        )
216        .expect("a handful of short pages is a valid book");
217        book.set(WRITABLE_BOOK_CONTENT, content);
218        book
219    }
220
221    /// An empty lectern has no book on it.
222    ///
223    /// Its raw `redstone_signal` is *not* zero -- vanilla's formula treats a
224    /// bookless lectern as being at the end of a one-page book and returns 14.
225    /// Nothing ever sees that: `LecternBlock.getAnalogOutputSignal` checks
226    /// `has_book` first and answers zero. The guard belongs to the block, so
227    /// this asserts what the block entity is actually asked.
228    #[test]
229    fn an_empty_lectern_has_no_book() {
230        let lectern = detached_lectern();
231        assert!(!lectern.has_book());
232    }
233
234    /// A book open at its first page already reads 1.
235    ///
236    /// Vanilla parity: the `+ (hasBook ? 1 : 0)` of `getRedstoneSignal`. It is
237    /// what lets a comparator tell an occupied lectern from an empty one even
238    /// before a page is turned.
239    #[test]
240    fn the_first_page_of_a_book_reads_one() {
241        let lectern = detached_lectern();
242        lectern.set_book(book_with_pages(8));
243
244        assert!(lectern.has_book());
245        assert_eq!(lectern.page(), 0);
246        assert_eq!(lectern.redstone_signal(), 1);
247    }
248
249    /// The last page reads full.
250    #[test]
251    fn the_last_page_reads_fifteen() {
252        let lectern = detached_lectern();
253        lectern.set_book(book_with_pages(8));
254
255        assert!(lectern.set_page(7));
256        assert_eq!(lectern.redstone_signal(), 15);
257    }
258
259    /// Turning past the end does not move, and so must not report a change.
260    ///
261    /// Vanilla parity: the guard in `setPage`. Without it, holding "next" on
262    /// the last page would pulse the redstone forever -- a lectern would be a
263    /// clock.
264    #[test]
265    fn turning_past_the_last_page_changes_nothing() {
266        let lectern = detached_lectern();
267        lectern.set_book(book_with_pages(3));
268        assert!(lectern.set_page(2));
269
270        assert!(
271            !lectern.set_page(3),
272            "there is no fourth page, so nothing moved"
273        );
274        assert_eq!(lectern.page(), 2);
275    }
276
277    /// Taking the book leaves the stand empty and forgets the page.
278    #[test]
279    fn taking_the_book_resets_the_stand() {
280        let lectern = detached_lectern();
281        lectern.set_book(book_with_pages(5));
282        lectern.set_page(3);
283
284        let taken = lectern.take_book();
285
286        assert!(taken.is(&vanilla_items::WRITABLE_BOOK));
287        assert!(!lectern.has_book());
288        assert_eq!(lectern.page(), 0);
289    }
290
291    /// A one-page book reads full on its only page.
292    ///
293    /// Vanilla parity: the `pageCount > 1` guard, which avoids dividing by
294    /// zero and treats a single page as being at the end.
295    #[test]
296    fn a_one_page_book_reads_fifteen() {
297        let lectern = detached_lectern();
298        lectern.set_book(book_with_pages(1));
299        assert_eq!(lectern.redstone_signal(), 15);
300    }
