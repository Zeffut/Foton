27/// Number of slots in a barrel (3 rows of 9).
28pub const BARREL_SLOTS: usize = 27;
29
30/// Barrel block entity.
31///
32/// A simple container with 27 slots, using the same menu as chests.
33pub struct BarrelBlockEntity {
34    base: Arc<BlockEntityBase>,
35    container: Arc<SyncMutex<BarrelContainer>>,
36    container_ref: ContainerRef,
37    /// Vanilla parity: the `RandomizableContainer` half of a barrel, which is
38    /// how a trial chamber or village barrel arrives stocked.
39    loot: Arc<ContainerLoot>,
40    /// Vanilla parity: the `name` of `BaseContainerBlockEntity`, the anvil
41    /// name this block was placed with.
42    name: BlockEntityName,
43}
44
45struct BarrelContainer {
46    items: Vec<ItemStack>,
47}
48
49// SAFETY: This key is owned by Foton and uniquely identifies `BarrelBlockEntity`.
50unsafe impl DowncastType for BarrelBlockEntity {
51    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:block_entity/barrel");
52}
53
54// SAFETY: This key is owned by Foton and uniquely identifies the independently
55// lockable inventory data used by a barrel block entity.
56unsafe impl DowncastType for BarrelContainer {
57    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:container/barrel");
58}

60impl BarrelBlockEntity {
61    /// Creates a new barrel block entity.
62    #[must_use]
63    pub fn new(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
64        let base = Arc::new(BlockEntityBase::new(
65            &vanilla_block_entity_types::BARREL,
66            level,
67            pos,
68            state,
69        ));
70        let container = Arc::new(SyncMutex::new(BarrelContainer {
71            items: vec![ItemStack::empty(); BARREL_SLOTS],
72        }));
73        let shared_container: SharedContainer = container.clone();
74        let loot = Arc::new(ContainerLoot::new());
75        Self {
76            container_ref: ContainerRef::owned_by_randomizable_block_entity(
77                shared_container,
78                Arc::clone(&base),
79                Arc::clone(&loot),
80            ),
81            base,
82            container,
83            loot,
84            name: BlockEntityName::new(),
85        }
86    }
87
88    /// Returns the name an anvil gave this barrel, if any.
89    ///
90    /// Vanilla parity: `Nameable.getCustomName`.
91    #[must_use]
92    pub fn custom_name(&self) -> Option<TextComponent> {
93        self.name.custom_name()
94    }
95}

97impl BlockEntity for BarrelBlockEntity {
98    fn base(&self) -> &BlockEntityBase {
99        &self.base
100    }
101
102    fn pre_remove_side_effects(&self, pos: BlockPos, _state: BlockStateId) {
103        // Vanilla reads the contents out through `Container.getItem`, which
104        // rolls a packed table first.
105        self.container_ref.unpack_loot_table(None);
106        let items = {
107            let mut container = self.container.lock();
108            mem::replace(&mut container.items, vec![ItemStack::empty(); BARREL_SLOTS])
109        };
110        let Some(world) = self.get_level() else {
111            return;
112        };
113        for item in items {
114            world.drop_item_stack(pos, item);
115        }
116    }
117
118    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
119        // Convert to NbtCompound view for accessing methods
120        let nbt_view: NbtCompoundView<'_, '_> = nbt.into();
121        self.name.load(&nbt_view);
122        // Vanilla parity: a barrel stores either a loot table or its items.
123        let packed = self.loot.try_load_loot_table(&nbt_view);
124        let mut container = self.container.lock();
125        container.items.fill(ItemStack::empty());
126        if packed {
127            return;
128        }
129
130        // Load items from NBT using borrowed NBT for proper ItemStack parsing
131        if let Some(items_list) = nbt_view.list("Items")
132            && let Some(compounds) = items_list.compounds()
133        {
134            for compound in compounds {
135                // Each item has a "Slot" byte and item data
136                if let Some(slot) = compound.byte("Slot") {
137                    let slot = slot as usize;
138                    if slot < BARREL_SLOTS {
139                        // Parse item directly from the borrowed compound
140                        if let Some(item) = ItemStack::from_borrowed_compound(&compound) {
141                            container.items[slot] = item;
142                        }
143                    }
144                }
145            }
146        }
147    }
148
149    fn save_additional(&self, nbt: &mut NbtCompound) {
150        self.name.save(nbt);
151        if self.loot.try_save_loot_table(nbt) {
152            return;
153        }
154        // Save items to NBT (only non-empty slots)
155        let container = self.container.lock();
156        let mut items: Vec<NbtCompound> = Vec::new();
157        for (slot, item) in container.items.iter().enumerate() {
158            if !item.is_empty() {
159                // Use ItemStack's ToNbtTag implementation for proper component serialization
160                if let NbtTag::Compound(mut item_nbt) = item.clone().to_nbt_tag() {
161                    item_nbt.insert("Slot", slot as i8);
162                    items.push(item_nbt);
163                }
164            }
165        }
166        nbt.insert("Items", NbtList::Compound(items));
167    }
168
169    fn get_update_tag(&self) -> Option<NbtCompound> {
170        // Barrels don't need to send inventory to clients on chunk load
171        // (unlike signs which display text)
172        None
173    }
174
175    fn container_ref(&self) -> Option<ContainerRef> {
176        Some(self.container_ref.clone())
177    }
178
179    /// Vanilla parity: `BaseContainerBlockEntity.getName`, which falls back to
180    /// the block's own name.
181    fn display_name(&self, default_name: TextComponent) -> TextComponent {
182        self.name.display_name(default_name)
183    }
184
185    /// Vanilla parity: the `CUSTOM_NAME` half of
186    /// `BaseContainerBlockEntity.collectImplicitComponents`. `CONTAINER` and
187    /// `LOCK` are not collected: no vanilla loot table asks this block for
188    /// either, and Foton has no lock on a container yet.
189    fn collect_implicit_components(&self, components: &mut DataComponentMap) {
190        self.name.collect_implicit_components(components);
191    }
192
193    /// Vanilla parity: the `CUSTOM_NAME` half of
194    /// `BaseContainerBlockEntity.applyImplicitComponents`.
195    fn apply_implicit_components(&self, input: &ImplicitComponentInput<'_>) {
196        self.name.apply_implicit_components(input);
197    }
198}

200impl Container for BarrelContainer {
201    fn items(&self) -> &[ItemStack] {
202        &self.items
203    }
204
205    fn items_mut(&mut self) -> &mut [ItemStack] {
206        &mut self.items
207    }
208
209    fn get_container_size(&self) -> usize {
210        BARREL_SLOTS
211    }
212
213    fn set_item(&mut self, slot: usize, mut stack: ItemStack) {
214        if slot < BARREL_SLOTS {
215            let max_stack_size = self.get_max_stack_size_for_item(&stack);
216            if !stack.is_empty() && stack.count() > max_stack_size {
217                stack.set_count(max_stack_size);
218            }
219            self.items[slot] = stack;
220        }
221    }
222
223    fn get_max_stack_size(&self) -> i32 {
224        64
225    }
226
227    fn set_changed(&mut self) {}
228}

230#[cfg(test)]
231mod tests {
232    use foton_registry::{init_vanilla_registry, vanilla_blocks, vanilla_items};
233
234    use super::*;
235
236    fn test_barrel() -> BarrelBlockEntity {
237        init_vanilla_registry();
238        BarrelBlockEntity::new(
239            Weak::new(),
240            BlockPos::new(1, 2, 3),
241            vanilla_blocks::BARREL.default_state(),
242        )
243    }
244
245    #[test]
246    fn set_item_limits_stack_to_vanilla_container_maximum() {
247        let barrel = test_barrel();
248        barrel
249            .container
250            .lock()
251            .set_item(0, ItemStack::with_count(&vanilla_items::STONE, 100));
252
253        assert_eq!(barrel.container.lock().get_item(0).count(), 64);
254    }
255
256    #[test]
257    fn pre_remove_preserves_slots_for_existing_menu_references() {
258        let barrel = test_barrel();
259        barrel
260            .container
261            .lock()
262            .set_item(0, ItemStack::new(&vanilla_items::STONE));
263
264        barrel.pre_remove_side_effects(
265            BlockPos::new(1, 2, 3),
266            vanilla_blocks::BARREL.default_state(),
267        );
268
269        let container = barrel.container.lock();
270        assert_eq!(container.items.len(), BARREL_SLOTS);
271        assert!(container.items.iter().all(ItemStack::is_empty));
272    }
