1//! Brewing stand block entity.
2//!
3//! Vanilla parity: `BrewingStandBlockEntity`. Three bottles brew at once from
4//! one ingredient, which is what makes brewing worth the walk to the nether:
5//! the ingredient is spent once and every bottle it applies to is converted.
6//! Bottles it does not apply to are left alone rather than blocking the brew.

37/// First of the three bottle slots.
38pub const SLOT_FIRST_BOTTLE: usize = 0;
39/// Slot holding the ingredient being brewed in.
40pub const SLOT_INGREDIENT: usize = 3;
41/// Slot holding the blaze powder.
42pub const SLOT_FUEL: usize = 4;
43/// Total slots in a brewing stand.
44pub const BREWING_STAND_SLOTS: usize = 5;
45/// Bottles a stand brews at once.
46pub const BOTTLE_SLOTS: usize = 3;
47
48/// The three bottle-occupied flags on the block.
49///
50/// Vanilla parity: `BrewingStandBlock.HAS_BOTTLE`, which is what draws the
51/// bottles on the model.
52static HAS_BOTTLE: [&BoolProperty; BOTTLE_SLOTS] = [
53    &BlockStateProperties::HAS_BOTTLE_0,
54    &BlockStateProperties::HAS_BOTTLE_1,
55    &BlockStateProperties::HAS_BOTTLE_2,
56];

58/// Brewing progress mirrored to every open menu.
59///
60/// Vanilla parity: the two-entry `ContainerData` of `BrewingStandBlockEntity`.
61/// As with the furnace, vanilla hands the menu the block entity itself; Foton
62/// republishes the two values so a menu never takes the block entity's lock.
63#[derive(Debug, Default)]
64pub struct BrewingStandDataSlots {
65    /// Ticks left on the current brew, counting down.
66    pub brew_time: AtomicI32,
67    /// Brews left in the blaze powder already consumed.
68    pub fuel: AtomicI32,
69}
70
71impl BrewingStandDataSlots {
72    /// Reads the two values in the order the vanilla protocol expects.
73    #[must_use]
74    pub fn snapshot(&self) -> [i16; 2] {
75        [
76            clamp_to_i16(self.brew_time.load(Ordering::Relaxed)),
77            clamp_to_i16(self.fuel.load(Ordering::Relaxed)),
78        ]
79    }
80}
81
82/// Narrows a counter to the `i16` the protocol carries.
83fn clamp_to_i16(value: i32) -> i16 {
84    i16::try_from(value).unwrap_or(i16::MAX)
85}

98/// Items and brewing state, held under one lock because a tick mutates both.
99struct BrewingStandContainer {
100    items: Vec<ItemStack>,
101    /// Ticks left on the current brew.
102    brew_time: i32,
103    /// Brews left in the blaze powder already consumed.
104    fuel: i32,
105    /// The ingredient the current brew started with.
106    ///
107    /// Vanilla keeps this to notice a player swapping the ingredient mid-brew,
108    /// which cancels the brew rather than quietly producing the wrong potion.
109    brewing_ingredient: Option<ItemStack>,
110    /// Which bottle slots were filled when the block state was last written.
111    last_bottle_flags: [bool; BOTTLE_SLOTS],
112}

117}

125impl BrewingStandContainer {
126    /// Returns whether the ingredient converts at least one bottle.
127    ///
128    /// Vanilla parity: `BrewingStandBlockEntity.isBrewable`.
129    fn is_brewable(&self) -> bool {
130        let ingredient = &self.items[SLOT_INGREDIENT];
131        if ingredient.is_empty() || !potion_brewing::is_ingredient(ingredient) {
132            return false;
133        }
134        self.items[..BOTTLE_SLOTS]
135            .iter()
136            .any(|bottle| !bottle.is_empty() && potion_brewing::has_mix(bottle, ingredient))
137    }
138
139    /// Converts every bottle the ingredient applies to and spends it.
140    ///
141    /// Vanilla parity: `BrewingStandBlockEntity.doBrew`. Returns the remainder
142    /// the caller has to drop when the ingredient slot could not hold it, which
143    /// is how a dragon's breath leaves its empty bottle behind.
144    fn brew(&mut self) -> Option<ItemStack> {
145        let ingredient = self.items[SLOT_INGREDIENT].clone();
146        for slot in 0..BOTTLE_SLOTS {
147            self.items[slot] = potion_brewing::mix_with(&ingredient, &self.items[slot]);
148        }
149
150        let remainder = self.items[SLOT_INGREDIENT].item().get_crafting_remainder();
151        let count = self.items[SLOT_INGREDIENT].count() - 1;
152        self.items[SLOT_INGREDIENT].set_count(count);
153
154        if remainder.is_empty() {
155            return None;
156        }
157        if self.items[SLOT_INGREDIENT].is_empty() {
158            self.items[SLOT_INGREDIENT] = remainder;
159            return None;
160        }
161        Some(remainder)
162    }
163
164    /// Returns which bottle slots are occupied.
165    ///
166    /// Vanilla parity: `BrewingStandBlockEntity.getPotionBits`.
167    fn bottle_flags(&self) -> [bool; BOTTLE_SLOTS] {
168        array::from_fn(|slot| !self.items[slot].is_empty())
169    }
170}

214impl BlockEntity for BrewingStandBlockEntity {
215    fn base(&self) -> &BlockEntityBase {
216        &self.base
217    }
218
219    fn tick(&self, world: &Arc<World>) {
220        let pos = self.get_block_pos();
221        let mut container = self.container.lock();
222
223        // Vanilla parity: fuel is taken the moment the stand runs dry, whether
224        // or not there is anything to brew, which is why a stand refills itself
225        // from a hopper before a player ever opens it.
226        if container.fuel <= 0 && potion_brewing::is_brewing_fuel(&container.items[SLOT_FUEL]) {
227            container.fuel = potion_brewing::FUEL_USES;
228            let remaining = container.items[SLOT_FUEL].count() - 1;
229            container.items[SLOT_FUEL].set_count(remaining);
230        }
231
232        let brewable = container.is_brewable();
233        let mut remainder = None;
234
235        if container.brew_time > 0 {
236            container.brew_time -= 1;
237            let ingredient_swapped = container
238                .brewing_ingredient
239                .as_ref()
240                .is_none_or(|started| !container.items[SLOT_INGREDIENT].is(started.item()));
241
242            if container.brew_time == 0 && brewable {
243                remainder = container.brew();
244                world.level_event(SOUND_BREWING_STAND_BREW, pos, 0, None);
245            } else if !brewable || ingredient_swapped {
246                container.brew_time = 0;
247            }
248        } else if brewable && container.fuel > 0 {
249            container.fuel -= 1;
250            container.brew_time = potion_brewing::BREWING_TIME_TICKS;
251            container.brewing_ingredient = Some(container.items[SLOT_INGREDIENT].clone());
252        }
253
254        self.data
255            .brew_time
256            .store(container.brew_time, Ordering::Relaxed);
257        self.data.fuel.store(container.fuel, Ordering::Relaxed);
258
259        let flags = container.bottle_flags();
260        let flags_changed = flags != container.last_bottle_flags;
261        if flags_changed {
262            container.last_bottle_flags = flags;
263        }
264        drop(container);
265
266        if let Some(remainder) = remainder {
267            world.drop_item_stack(pos, remainder);
268        }
269
270        if flags_changed {
271            let mut state = self.get_block_state();
272            for (property, filled) in HAS_BOTTLE.iter().zip(flags) {
273                state = state.set_value(*property, filled);
274            }
275            world.set_block(pos, state, UpdateFlags::UPDATE_CLIENTS);
276        }
277    }
278
279    fn pre_remove_side_effects(&self, pos: BlockPos, _state: BlockStateId) {
280        let items = {
281            let mut container = self.container.lock();
282            mem::replace(
283                &mut container.items,
284                vec![ItemStack::empty(); BREWING_STAND_SLOTS],
285            )
286        };
287        let Some(world) = self.get_level() else {
288            return;
289        };
290        for item in items {
291            world.drop_item_stack(pos, item);
292        }
293    }
294
295    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
296        let nbt_view: NbtCompoundView<'_, '_> = nbt.into();
297        self.name.load(&nbt_view);
298        let mut container = self.container.lock();
299        container.items.fill(ItemStack::empty());
300
301        if let Some(items_list) = nbt_view.list("Items")
302            && let Some(compounds) = items_list.compounds()
303        {
304            for compound in compounds {
305                if let Some(slot) = compound.byte("Slot") {
306                    let slot = slot as usize;
307                    if slot < BREWING_STAND_SLOTS
308                        && let Some(item) = ItemStack::from_borrowed_compound(&compound)
309                    {
310                        container.items[slot] = item;
311                    }
312                }
313            }
314        }
315
316        container.brew_time = nbt_view.short("BrewTime").unwrap_or(0).into();
317        container.fuel = nbt_view.byte("Fuel").unwrap_or(0).into();
318        // Vanilla reloads the in-progress ingredient from the slot, so a brew
319        // interrupted by a save resumes instead of cancelling on the next tick.
320        container.brewing_ingredient =
321            (container.brew_time > 0).then(|| container.items[SLOT_INGREDIENT].clone());
322        container.last_bottle_flags = container.bottle_flags();
323    }
324
325    fn save_additional(&self, nbt: &mut NbtCompound) {
326        self.name.save(nbt);
327        let container = self.container.lock();
328        let mut items: Vec<NbtCompound> = Vec::new();
329        for (slot, item) in container.items.iter().enumerate() {
330            if !item.is_empty()
331                && let NbtTag::Compound(mut item_nbt) = item.clone().to_nbt_tag()
332            {
333                item_nbt.insert("Slot", slot as i8);
334                items.push(item_nbt);
335            }
336        }
337        nbt.insert("Items", NbtList::Compound(items));
338        nbt.insert("BrewTime", container.brew_time as i16);
339        nbt.insert("Fuel", container.fuel as i8);
340    }
341
342    fn get_update_tag(&self) -> Option<NbtCompound> {
343        None
344    }
345
346    fn container_ref(&self) -> Option<ContainerRef> {
347        Some(self.container_ref.clone())
348    }

371/// Slots a hopper above a brewing stand may fill.
372///
373/// Vanilla parity: `BrewingStandBlockEntity.SLOTS_FOR_UP`.
374static SLOTS_FOR_UP: [usize; 1] = [SLOT_INGREDIENT];
375
376/// Slots a hopper below a brewing stand may drain.
377///
378/// Vanilla parity: `BrewingStandBlockEntity.SLOTS_FOR_DOWN`.
379static SLOTS_FOR_DOWN: [usize; 4] = [0, 1, 2, SLOT_INGREDIENT];
380
381/// Slots a hopper at the side of a brewing stand may reach.
382///
383/// Vanilla parity: `BrewingStandBlockEntity.SLOTS_FOR_SIDES`. Fuel goes in from
384/// the side, which is why an automatic stand feeds blaze powder sideways and
385/// ingredients from above.
386static SLOTS_FOR_SIDES: [usize; 4] = [0, 1, 2, SLOT_FUEL];
387
388impl Container for BrewingStandContainer {
389    fn items(&self) -> &[ItemStack] {
390        &self.items
391    }
392
393    fn items_mut(&mut self) -> &mut [ItemStack] {
394        &mut self.items
395    }
396
397    fn get_container_size(&self) -> usize {
398        BREWING_STAND_SLOTS
399    }
400
401    /// Vanilla parity: `BrewingStandBlockEntity.canPlaceItem`.
402    fn can_place_item(&self, slot: usize, stack: &ItemStack) -> bool {
403        match slot {
404            SLOT_INGREDIENT => potion_brewing::is_ingredient(stack),
405            SLOT_FUEL => potion_brewing::is_brewing_fuel(stack),
406            // A bottle slot takes one bottle and only while it is empty, which
407            // is what stops a hopper stacking three potions into one slot.
408            _ => {
409                (stack.is(&vanilla_items::POTION)
410                    || stack.is(&vanilla_items::SPLASH_POTION)
411                    || stack.is(&vanilla_items::LINGERING_POTION)
412                    || stack.is(&vanilla_items::GLASS_BOTTLE))
413                    && self.items[slot].is_empty()
414            }
415        }
416    }
417
418    /// Vanilla parity: `BrewingStandBlockEntity.getSlotsForFace`.
419    fn slots_for_face(&self, direction: Direction) -> SlotsForFace {
420        match direction {
421            Direction::Up => SlotsForFace::Explicit(&SLOTS_FOR_UP),
422            Direction::Down => SlotsForFace::Explicit(&SLOTS_FOR_DOWN),
423            _ => SlotsForFace::Explicit(&SLOTS_FOR_SIDES),
424        }
425    }
426
427    /// Vanilla parity: `BrewingStandBlockEntity.canTakeItemThroughFace`. Only an
428    /// emptied bottle leaves the ingredient slot, so a hopper cannot steal the
429    /// nether wart a stand is about to brew with.
430    fn can_take_item_through_face(
431        &self,
432        slot: usize,
433        stack: &ItemStack,
434        _direction: Direction,
435    ) -> bool {
436        if slot == SLOT_INGREDIENT {
437            return stack.is(&vanilla_items::GLASS_BOTTLE);
438        }
439        true
440    }
441
442    fn get_max_stack_size(&self) -> i32 {
443        64
444    }
445
446    fn set_changed(&mut self) {}
447}
