1//! The command block block entity.
2//!
3//! Vanilla parity: `CommandBlockEntity`. It owns a [`BaseCommandBlock`] plus the
4//! three flags a command block needs on top of it: whether redstone is holding
5//! it powered, whether it is set to "always active", and whether its condition
6//! was met on the last attempt.
7//!
8//! The mode is not stored -- it is read back from which of the three blocks is
9//! standing there, exactly as `getMode` does.
10

27/// How a command block decides when to run.
28///
29/// Vanilla parity: `CommandBlockEntity.Mode`, and the wire order of
30/// `ServerboundSetCommandBlockPacket`'s enum.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub enum CommandBlockMode {
33    /// The chain command block: runs when the block behind it succeeds.
34    Sequence,
35    /// The repeating command block: runs every tick while active.
36    Auto,
37    /// The plain command block: runs on a rising redstone edge.
38    Redstone,
39}
40
41impl CommandBlockMode {
42    /// Returns the mode a wire value names.
43    ///
44    /// Vanilla parity: `FriendlyByteBuf.readEnum` over `CommandBlockEntity.Mode`,
45    /// which is ordinal order.
46    #[must_use]
47    pub const fn from_wire(value: i32) -> Option<Self> {
48        match value {
49            0 => Some(Self::Sequence),
50            1 => Some(Self::Auto),
51            2 => Some(Self::Redstone),
52            _ => None,
53        }
54    }
55
56    /// Returns the block this mode is stored as.
57    ///
58    /// Vanilla parity: the `switch (packet.getMode())` of
59    /// `handleSetCommandBlock`.
60    #[must_use]
61    pub const fn block(self) -> BlockRef {
62        match self {
63            Self::Sequence => &vanilla_blocks::CHAIN_COMMAND_BLOCK,
64            Self::Auto => &vanilla_blocks::REPEATING_COMMAND_BLOCK,
65            Self::Redstone => &vanilla_blocks::COMMAND_BLOCK,
66        }
67    }
68}

70/// The three flags a command block keeps beyond its command.
71#[derive(Debug)]
72struct CommandBlockFlags {
73    powered: bool,
74    auto: bool,
75    condition_met: bool,
76}

78impl CommandBlockFlags {
79    const fn new() -> Self {
80        Self {
81            powered: false,
82            auto: false,
83            condition_met: false,
84        }
85    }
86}

100impl CommandBlockEntity {
101    /// Creates a command block entity.
102    #[must_use]
103    pub fn new(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
104        Self {
105            base: BlockEntityBase::new(
106                &vanilla_block_entity_types::COMMAND_BLOCK,
107                level,
108                pos,
109                state,
110            ),
111            command_block: Arc::new(BaseCommandBlock::new()),
112            flags: SyncMutex::new(CommandBlockFlags::new()),
113        }
114    }
115
116    /// Returns the command store.
117    ///
118    /// Vanilla parity: `CommandBlockEntity.getCommandBlock`.
119    #[must_use]
120    pub const fn command_block(&self) -> &Arc<BaseCommandBlock> {
121        &self.command_block
122    }
123
124    /// Returns whether redstone is holding this block powered.
125    #[must_use]
126    pub fn is_powered(&self) -> bool {
127        self.flags.lock().powered
128    }
129
130    /// Sets whether redstone is holding this block powered.
131    pub fn set_powered(&self, powered: bool) {
132        self.flags.lock().powered = powered;
133    }
134
135    /// Returns whether this block is set to "always active".
136    ///
137    /// Vanilla parity: `CommandBlockEntity.isAutomatic`.
138    #[must_use]
139    pub fn is_automatic(&self) -> bool {
140        self.flags.lock().auto
141    }
142
143    /// Sets "always active", scheduling a tick when it turns on.
144    ///
145    /// Vanilla parity: `CommandBlockEntity.setAutomatic`. A chain block is
146    /// excluded because it never schedules itself -- the block in front of it
147    /// drives it.
148    pub fn set_automatic(&self, auto: bool) {
149        let previous = {
150            let mut flags = self.flags.lock();
151            let previous = flags.auto;
152            flags.auto = auto;
153            previous
154        };
155
156        if !previous && auto && !self.is_powered() && self.mode() != CommandBlockMode::Sequence {
157            self.schedule_tick();
158        }
159    }
160
161    /// Reschedules after the editor switched this block to another mode.
162    ///
163    /// Vanilla parity: `CommandBlockEntity.onModeSwitch`.
164    pub fn on_mode_switch(&self) {
165        if self.mode() != CommandBlockMode::Auto {
166            return;
167        }
168        if self.is_powered() || self.is_automatic() {
169            self.schedule_tick();
170        }
171    }
172
173    fn schedule_tick(&self) {
174        let Some(world) = self.get_level() else {
175            return;
176        };
177        let block = self.get_block_state().get_block();
178        if !is_command_block(block) {
179            return;
180        }
181        self.mark_condition_met();
182        world.schedule_block_tick_default(self.get_block_pos(), block, 1);
183    }
184
185    /// Returns whether the last attempt found its condition satisfied.
186    ///
187    /// Vanilla parity: `CommandBlockEntity.wasConditionMet`.
188    #[must_use]
189    pub fn was_condition_met(&self) -> bool {
190        self.flags.lock().condition_met
191    }
192
193    /// Re-evaluates the condition and returns it.
194    ///
195    /// Vanilla parity: `CommandBlockEntity.markConditionMet`. An unconditional
196    /// block is always met; a conditional one is met only when the command
197    /// block *behind* it -- opposite the way it faces -- last succeeded.
198    pub fn mark_condition_met(&self) -> bool {
199        let met = self.evaluate_condition();
200        self.flags.lock().condition_met = met;
201        met
202    }
203
204    fn evaluate_condition(&self) -> bool {
205        if !self.is_conditional() {
206            return true;
207        }
208        let Some(world) = self.get_level() else {
209            return false;
210        };
211        let pos = self.get_block_pos();
212        let Some(facing) = self
213            .get_block_state()
214            .try_get_value(&BlockStateProperties::FACING)
215        else {
216            return false;
217        };
218        let behind = pos.relative(facing.opposite());
219        if !is_command_block(world.get_block_state(behind).get_block()) {
220            return false;
221        }
222        world
223            .get_block_entity(behind)
224            .and_then(|entity| {
225                foton_utils::Downcast::downcast_ref::<Self>(entity.as_ref())
226                    .map(|block| block.command_block.success_count() > 0)
227            })
228            .unwrap_or(false)
229    }
230
231    /// Returns when this block runs.
232    ///
233    /// Vanilla parity: `CommandBlockEntity.getMode`, which reads the block
234    /// rather than any stored field.
235    #[must_use]
236    pub fn mode(&self) -> CommandBlockMode {
237        match self.get_block_state().get_block() {
238            block if block == &vanilla_blocks::REPEATING_COMMAND_BLOCK => CommandBlockMode::Auto,
239            block if block == &vanilla_blocks::CHAIN_COMMAND_BLOCK => CommandBlockMode::Sequence,
240            _ => CommandBlockMode::Redstone,
241        }
242    }
243
244    /// Returns whether this block only runs when the one behind it succeeded.
245    ///
246    /// Vanilla parity: `CommandBlockEntity.isConditional`.
247    ///
248    /// Vanilla re-reads the world here. Foton reads the state this block entity
249    /// caches, which `on_block_state_changed` keeps current -- the same answer,
250    /// and one that still works when the entity has no world, as it does in a
251    /// unit test and briefly during a load.
252    #[must_use]
253    pub fn is_conditional(&self) -> bool {
254        let state = self.get_block_state();
255        if !is_command_block(state.get_block()) {
256            return false;
257        }
258        state
259            .try_get_value(&BlockStateProperties::CONDITIONAL)
260            .unwrap_or(false)
261    }
262
263    /// Sends this block's editor data to everyone tracking it.
264    ///
265    /// Vanilla parity: the `sendBlockUpdated` of `CommandBlockEntity`'s
266    /// `onUpdated`, which is how an open editor learns the new output.
267    pub fn broadcast_update(&self, world: &Arc<World>) {
268        let mut nbt = NbtCompound::new();
269        self.save_additional(&mut nbt);
270        world.broadcast_block_entity_update(self.get_block_pos(), self.get_type(), nbt);
271    }
272}

274impl CommandBlockEntity {
275    /// Returns where the command runs from.
276    ///
277    /// Vanilla parity: the `Vec3.atCenterOf(worldPosition)` of
278    /// `createCommandSourceStack`.
279    #[must_use]
280    pub fn command_source_position(&self) -> DVec3 {
281        let pos = self.get_block_pos();
282        DVec3::new(
283            f64::from(pos.x()) + 0.5,
284            f64::from(pos.y()) + 0.5,
285            f64::from(pos.z()) + 0.5,
286        )
287    }
288
289    /// Returns the rotation the command runs with.
290    ///
291    /// Vanilla parity: the `new Vec2(0.0F, facing.toYRot())` of
292    /// `createCommandSourceStack` -- a command block looks along its face and
293    /// never up or down, which is what `^ ^ ^` coordinates resolve against.
294    #[must_use]
295    pub fn command_source_rotation(&self) -> (f32, f32) {
296        let facing = self
297            .get_block_state()
298            .try_get_value(&BlockStateProperties::FACING)
299            .unwrap_or(Direction::North);
300        (facing_to_y_rotation(facing), 0.0)
301    }
302}
303
304impl BlockEntity for CommandBlockEntity {
305    fn base(&self) -> &BlockEntityBase {
306        &self.base
307    }
308
309    fn save_additional(&self, nbt: &mut NbtCompound) {
310        self.command_block.save(nbt);
311        let flags = self.flags.lock();
312        nbt.insert("powered", flags.powered);
313        nbt.insert("conditionMet", flags.condition_met);
314        nbt.insert("auto", flags.auto);
315    }
316
317    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
318        let view: NbtCompoundView<'_, '_> = nbt.into();
319        self.command_block.load(view);
320        {
321            let mut flags = self.flags.lock();
322            flags.powered = view.byte("powered").is_some_and(|value| value != 0);
323            flags.condition_met = view.byte("conditionMet").is_some_and(|value| value != 0);
324        }
325        self.set_automatic(view.byte("auto").is_some_and(|value| value != 0));
326    }
327
328    /// Vanilla parity: `openCommandBlock` sends `saveCustomOnly`, and the chunk
329    /// packet carries the same shape, so the editor can be filled from either.
330    fn get_update_tag(&self) -> Option<NbtCompound> {
331        let mut nbt = NbtCompound::new();
332        self.save_additional(&mut nbt);
333        Some(nbt)
334    }
335}

337/// Returns whether `block` is one of the three command blocks.
338///
339/// Vanilla parity: the `instanceof CommandBlock` checks of `CommandBlockEntity`.
340#[must_use]
341pub fn is_command_block(block: BlockRef) -> bool {
342    block == &vanilla_blocks::COMMAND_BLOCK
343        || block == &vanilla_blocks::REPEATING_COMMAND_BLOCK
344        || block == &vanilla_blocks::CHAIN_COMMAND_BLOCK
345}
346
347/// Returns the yaw a block face points along.
348///
349/// Vanilla parity: `Direction.toYRot`, which reports zero for the vertical
350/// faces because a yaw cannot express them.
351const fn facing_to_y_rotation(facing: Direction) -> f32 {
352    match facing {
353        Direction::West => 90.0,
354        Direction::North => 180.0,
355        Direction::East => 270.0,
356        // Vanilla parity: a vertical face has no yaw, and `toYRot` reports
357        // zero for it -- the same answer as south.
358        Direction::South | Direction::Down | Direction::Up => 0.0,
359    }
360}

371    fn command_block(state: BlockStateId) -> CommandBlockEntity {
372        CommandBlockEntity::new(Weak::new(), BlockPos::new(8, 64, 8), state)

375    fn reload(state: BlockStateId, nbt: &NbtCompound) -> CommandBlockEntity {
376        let mut bytes = Vec::new();
377        nbt.write(&mut bytes);
378        let borrowed = read_borrowed_compound(&mut Cursor::new(&bytes))
379            .unwrap_or_else(|error| panic!("test nbt should reborrow: {error}"));
380        let entity = command_block(state);
381        entity.load_additional(&borrowed);
382        entity
383    }

385    /// The mode is read off the block, not stored, so the three command blocks
386    /// have to answer differently for the same block entity type.
387    #[test]
388    fn the_mode_comes_from_which_command_block_is_standing_there() {
389        init_vanilla_registry();
390        assert_eq!(
391            command_block(vanilla_blocks::COMMAND_BLOCK.default_state()).mode(),
392            CommandBlockMode::Redstone
393        );
394        assert_eq!(
395            command_block(vanilla_blocks::REPEATING_COMMAND_BLOCK.default_state()).mode(),
396            CommandBlockMode::Auto
397        );
398        assert_eq!(
399            command_block(vanilla_blocks::CHAIN_COMMAND_BLOCK.default_state()).mode(),
400            CommandBlockMode::Sequence
401        );
402    }

404    /// Choosing a mode in the editor swaps the block, and reading the block
405    /// back gives the mode again. The two directions have to be inverses: if
406    /// `block()` were wrong, picking "repeating" would place a chain block and
407    /// the editor would reopen showing the wrong mode.
408    #[test]
409    fn the_mode_and_the_block_it_is_stored_as_are_inverses() {
410        init_vanilla_registry();
411        for mode in [
412            CommandBlockMode::Sequence,
413            CommandBlockMode::Auto,
414            CommandBlockMode::Redstone,
415        ] {
416            let placed = command_block(mode.block().default_state());
417            assert_eq!(placed.mode(), mode, "round trip through {mode:?}");
418        }
419    }

421    /// `readEnum` is ordinal order, and getting it wrong would silently turn
422    /// every repeating block a client saves into a chain block.
423    #[test]
424    fn the_wire_mode_order_is_sequence_auto_redstone() {
425        assert_eq!(
426            CommandBlockMode::from_wire(0),
427            Some(CommandBlockMode::Sequence)
428        );
429        assert_eq!(CommandBlockMode::from_wire(1), Some(CommandBlockMode::Auto));
430        assert_eq!(
431            CommandBlockMode::from_wire(2),
432            Some(CommandBlockMode::Redstone)
433        );
434        assert_eq!(CommandBlockMode::from_wire(3), None);
435        assert_eq!(CommandBlockMode::from_wire(-1), None);
436    }

438    /// All three flags persist. A repeating block that forgot `auto` would stop
439    /// running after a chunk reload, which is the classic broken-map symptom.
440    #[test]
441    fn the_three_flags_survive_a_save() {
442        init_vanilla_registry();
443        let entity = command_block(vanilla_blocks::COMMAND_BLOCK.default_state());
444        entity.set_powered(true);
445        entity.set_automatic(true);
446        entity.mark_condition_met();
447        entity.command_block().set_command("say hi".to_owned());
448
449        let mut nbt = NbtCompound::new();
450        entity.save_additional(&mut nbt);
451        assert_eq!(nbt.byte("powered"), Some(1));
452        assert_eq!(nbt.byte("auto"), Some(1));
453        assert_eq!(nbt.byte("conditionMet"), Some(1));
454
455        let reloaded = reload(vanilla_blocks::COMMAND_BLOCK.default_state(), &nbt);
456        assert!(reloaded.is_powered());
457        assert!(reloaded.is_automatic());
458        assert!(reloaded.was_condition_met());
459        assert_eq!(reloaded.command_block().command(), "say hi");
460    }

462    /// An unconditional block is always met without consulting the world, which
463    /// is why `markConditionMet` works off-level at all.
464    #[test]
465    fn an_unconditional_block_is_always_met() {
466        init_vanilla_registry();
467        let entity = command_block(vanilla_blocks::COMMAND_BLOCK.default_state());
468        assert!(entity.mark_condition_met());
469        assert!(entity.was_condition_met());
470    }

472    /// A conditional block with nothing behind it is not met. Off-level there
473    /// is no block behind it at all, so this is also the no-world case.
