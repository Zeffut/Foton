22/// Vanilla `TrialSpawnerBlockEntity`.
23pub struct TrialSpawnerBlockEntity {
24    base: BlockEntityBase,
25    trial_spawner: TrialSpawner,
26}
27
28// SAFETY: This key is owned by Foton and uniquely identifies `TrialSpawnerBlockEntity`.
29unsafe impl DowncastType for TrialSpawnerBlockEntity {
30    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:block_entity/trial_spawner");
31}
32
33impl TrialSpawnerBlockEntity {
34    /// Creates the storage behind one trial spawner block.
35    #[must_use]
36    pub fn new(world: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
37        Self {
38            base: BlockEntityBase::new(
39                &vanilla_block_entity_types::TRIAL_SPAWNER,
40                world,
41                pos,
42                state,
43            ),
44            trial_spawner: TrialSpawner::new(),
45        }
46    }
47
48    /// Returns the spawner behind this block.
49    ///
50    /// Vanilla parity: `TrialSpawnerBlockEntity.getTrialSpawner`.
51    #[must_use]
52    pub const fn trial_spawner(&self) -> &TrialSpawner {
53        &self.trial_spawner
54    }
55}
56
57impl TrialSpawnerStateAccessor for TrialSpawnerBlockEntity {
58    /// Vanilla parity: `TrialSpawnerBlockEntity.getState`, which falls back to
59    /// `INACTIVE` for a block that has no such property -- which happens while
60    /// a block entity outlives the block it belonged to.
61    fn trial_spawner_state(&self) -> TrialSpawnerState {
62        self.get_block_state()
63            .try_get_value(&BlockStateProperties::TRIAL_SPAWNER_STATE)
64            .unwrap_or(TrialSpawnerState::Inactive)
65    }
66
67    /// Vanilla parity: `TrialSpawnerBlockEntity.setState`.
68    fn set_trial_spawner_state(&self, world: &Arc<World>, state: TrialSpawnerState) {
69        self.set_changed();
70        let pos = self.get_block_pos();
71        let updated = world
72            .get_block_state(pos)
73            .set_value(&BlockStateProperties::TRIAL_SPAWNER_STATE, state);
74        world.set_block(pos, updated, UpdateFlags::UPDATE_ALL);
75    }
76
77    /// Vanilla parity: `TrialSpawnerBlockEntity.markUpdated`.
78    fn mark_trial_spawner_updated(&self) {
79        self.set_changed();
80        if let Some(world) = self.get_level() {
81            world.send_block_updated(self.get_block_pos());
82        }
83    }
84}
85
86impl Spawner for TrialSpawnerBlockEntity {
87    /// Vanilla parity: `TrialSpawnerBlockEntity.setEntityId`.
88    fn set_spawner_entity_id(&self, entity_type: EntityTypeRef) {
89        let Some(world) = self.get_level() else {
90            log::warn!("trial spawner retargeted with no level");
91            return;
92        };
93        self.trial_spawner
94            .override_entity_to_spawn(self, &world, &entity_type.key);
95        self.set_changed();
96    }
97}
98
99impl BlockEntity for TrialSpawnerBlockEntity {
100    fn base(&self) -> &BlockEntityBase {
101        &self.base
102    }
103
104    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
105        self.trial_spawner.load(&nbt.into());
106        if self.get_level().is_some() {
107            self.mark_trial_spawner_updated();
108        }
109    }
110
111    fn save_additional(&self, nbt: &mut NbtCompound) {
112        self.trial_spawner.save(nbt);
113    }
114
115    /// Vanilla parity: `TrialSpawnerBlockEntity.getUpdateTag`, which sends only
116    /// the live state -- never the configuration.
117    fn get_update_tag(&self) -> Option<NbtCompound> {
118        Some(
119            self.trial_spawner
120                .with_data(|data| data.update_tag(self.trial_spawner_state())),
121        )
122    }
123
124    fn tick(&self, world: &Arc<World>) {
125        let is_ominous = self
126            .get_block_state()
127            .try_get_value(&BlockStateProperties::OMINOUS)
128            .unwrap_or(false);
129        self.trial_spawner
130            .tick_server(self, world, self.get_block_pos(), is_ominous);
131    }
132}

134#[cfg(test)]
135mod tests {
136    use std::sync::Arc;
137
138    use foton_registry::{init_vanilla_registry, vanilla_blocks};
139
140    use super::*;
141
142    /// A spawn egg reaches a spawner through the `Spawner` capability, and the
143    /// only thing that connects the two is the keyed downcast. If it stops
144    /// answering, clicking a trial spawner silently places a mob beside it
145    /// instead of retargeting it -- which looks like the egg working.
146    #[test]
147    fn a_trial_spawner_answers_as_a_spawner() {
148        init_vanilla_registry();
149        let entity: Arc<dyn BlockEntity> = Arc::new(TrialSpawnerBlockEntity::new(
150            Weak::new(),
151            BlockPos::new(8, 64, 8),
152            vanilla_blocks::TRIAL_SPAWNER.default_state(),
153        ));
154
155        assert!(
156            entity.as_spawner().is_some(),
157            "a trial spawner must be reachable as a Spawner"
158        );
159    }
