1//! Skull and head block entity.
2//!
3//! Vanilla parity: `SkullBlockEntity`. Which creature a skull came from is in
4//! the block, but whose head it is, is not -- the profile lives here, and it
5//! is the only thing that tells one player head from another.
6//!
7//! Without this every player head placed in the world would be a blank one,
8//! and the head that comes back when it is broken would have forgotten who it
9//! belonged to. The same entity also carries the sound identifier a player
10//! head lends to the note block underneath it.
11
12use std::str::FromStr as _;
13use std::sync::{Arc, Weak};
14
15use foton_registry::data_components::vanilla_components::ResolvableProfile;
16use foton_registry::vanilla_block_entity_types;
17use foton_utils::locks::SyncMutex;
18use foton_utils::{BlockPos, BlockStateId, DowncastType, DowncastTypeKey, Identifier};
19use simdnbt::borrow::{BaseNbtCompound as BorrowedNbtCompound, NbtCompound as NbtCompoundView};
20use simdnbt::owned::NbtCompound;
21use simdnbt::{FromNbtTag as _, ToNbtTag as _};
22use text_components::TextComponent;
23
24use crate::block_entity::{BlockEntity, BlockEntityBase};
25use crate::world::World;
26
27/// Skull block entity, shared by every skull and head, standing or on a wall.
28pub struct SkullBlockEntity {
29    base: Arc<BlockEntityBase>,
30    state: SyncMutex<SkullState>,
31}
32
33/// What a skull remembers beyond its block state.
34#[derive(Default)]
35struct SkullState {
36    /// The profile a player head wears.
37    owner: Option<ResolvableProfile>,
38    /// The sound a note block under this head plays instead of an instrument.
39    note_block_sound: Option<Identifier>,
40    /// The name an anvil gave the head before it was placed.
41    name: Option<TextComponent>,
42}
43
44// SAFETY: This key is owned by Foton and uniquely identifies `SkullBlockEntity`.
45unsafe impl DowncastType for SkullBlockEntity {
46    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:block_entity/skull");
47}
48
49impl SkullBlockEntity {
50    /// Creates a skull block entity.
51    #[must_use]
52    pub fn new(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
53        Self {
54            base: Arc::new(BlockEntityBase::new(
55                &vanilla_block_entity_types::SKULL,
56                level,
57                pos,
58                state,
59            )),
60            state: SyncMutex::new(SkullState::default()),
61        }
62    }
63
64    /// Returns the profile this head wears.
65    ///
66    /// Vanilla parity: `SkullBlockEntity.getOwnerProfile`.
67    #[must_use]
68    pub fn owner_profile(&self) -> Option<ResolvableProfile> {
69        self.state.lock().owner.clone()
70    }
71
72    /// Returns the sound a note block under this head should play.
73    ///
74    /// Vanilla parity: `SkullBlockEntity.getNoteBlockSound`, read by
75    /// `NoteBlock.getCustomSoundId`. Foton's note block cannot use it yet: the
76    /// sound packet only carries registered sound events, while vanilla wraps
77    /// this identifier in a direct `Holder<SoundEvent>`. Inline sound events in
78    /// `CSound` are the missing system.
79    #[must_use]
80    pub fn note_block_sound(&self) -> Option<Identifier> {
81        self.state.lock().note_block_sound.clone()
82    }
83
84    /// Returns the skull's custom name, if it was renamed.
85    #[must_use]
86    pub fn custom_name(&self) -> Option<TextComponent> {
87        self.state.lock().name.clone()
88    }
89
90    /// Replaces what the skull remembers.
91    ///
92    /// Vanilla parity: `SkullBlockEntity.applyImplicitComponents`, which is
93    /// what carries a head item's profile onto the block it becomes.
94    pub fn set_from_item(
95        &self,
96        owner: Option<ResolvableProfile>,
97        note_block_sound: Option<Identifier>,
98        name: Option<TextComponent>,
99    ) {
100        {
101            let mut state = self.state.lock();
102            state.owner = owner;
103            state.note_block_sound = note_block_sound;
104            state.name = name;
105        }
106        self.set_changed();
107    }
108}
109
110impl BlockEntity for SkullBlockEntity {
111    fn base(&self) -> &BlockEntityBase {
112        &self.base
113    }
114
115    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
116        let nbt_view: NbtCompoundView<'_, '_> = nbt.into();
117        let mut state = self.state.lock();
118        state.owner = nbt_view
119            .get("profile")
120            .and_then(ResolvableProfile::from_nbt_tag);
121        state.note_block_sound = nbt_view
122            .string("note_block_sound")
123            .and_then(|value| Identifier::from_str(&value.to_str()).ok());
124        state.name = nbt_view
125            .get("custom_name")
126            .map(|tag| tag.to_owned())
127            .as_ref()
128            .and_then(TextComponent::from_nbt);
129    }
130
131    fn save_additional(&self, nbt: &mut NbtCompound) {
132        let state = self.state.lock();
133        if let Some(owner) = &state.owner {
134            nbt.insert("profile", owner.clone().to_nbt_tag());
135        }
136        if let Some(sound) = &state.note_block_sound {
137            nbt.insert("note_block_sound", sound.to_string());
138        }
139        if let Some(name) = &state.name {
140            nbt.insert("custom_name", name.to_codec_nbt());
141        }
142    }
143
144    /// Vanilla parity: `SkullBlockEntity.getUpdateTag`, which sends the whole
145    /// saved compound. The client draws the face from the profile, so it needs
146    /// it the moment the chunk arrives rather than after an interaction.
147    fn get_update_tag(&self) -> Option<NbtCompound> {
148        let mut nbt = NbtCompound::new();
149        self.save_additional(&mut nbt);
150        Some(nbt)
151    }
152}
