1//! Conduit block entity implementation.
2//!
3//! A conduit is inert until it is sealed inside a 3x3x3 pocket of water and
4//! ringed by prismarine. It then hands Conduit Power to swimmers nearby, and
5//! once the frame is complete it picks a hostile mob out of the water every two
6//! seconds and hits it.
7//!
8//! Everything the player sees spinning and sparking is drawn client-side:
9//! vanilla's `animationTick` and `getActiveRotation` never run on a server, and
10//! the client re-derives whether a conduit is active by running `updateShape`
11//! itself. The one thing the server has to tell it is which mob is being
12//! attacked, so the beam has somewhere to point -- which is why `Target` is the
13//! whole of this entity's saved and synced state.
14

38/// Frame blocks needed before a conduit switches on.
39///
40/// Vanilla parity: `ConduitBlockEntity.MIN_ACTIVE_SIZE`.
41const MIN_ACTIVE_SIZE: usize = 16;
42
43/// Frame blocks needed before a conduit starts hitting mobs.
44///
45/// Vanilla parity: `ConduitBlockEntity.MIN_KILL_SIZE`.
46const MIN_KILL_SIZE: usize = 42;
47
48/// How far a conduit looks for something to hit.
49///
50/// Vanilla parity: `ConduitBlockEntity.KILL_RANGE`.
51const KILL_RANGE: f64 = 8.0;
52
53/// Vanilla parity: the `260` of `ConduitBlockEntity.applyEffects`, which is
54/// longer than the two seconds between applications so the effect never lapses
55/// while a player stays in range.
56const EFFECT_DURATION_TICKS: i32 = 260;
57
58/// Vanilla parity: the `4.0F` of `updateAndAttackTarget`.
59const ATTACK_DAMAGE: f32 = 4.0;
60
61/// How often the whole conduit beat runs.
62///
63/// Vanilla parity: the `gameTime % 40` of `ConduitBlockEntity.serverTick`.
64const BEAT_INTERVAL_TICKS: i64 = 40;
65
66/// Vanilla parity: the `gameTime % 80` ambient sound.
67const AMBIENT_INTERVAL_TICKS: i64 = 80;

69const TARGET_NBT_KEY: &str = "Target";

71/// Vanilla parity: `ConduitBlockEntity.VALID_BLOCKS`, a hardcoded array rather
72/// than a block tag.
73fn is_frame_block(block: BlockRef) -> bool {
74    block == &vanilla_blocks::PRISMARINE
75        || block == &vanilla_blocks::PRISMARINE_BRICKS
76        || block == &vanilla_blocks::SEA_LANTERN
77        || block == &vanilla_blocks::DARK_PRISMARINE
78}

80/// Vanilla parity: `Vec3i.closerThan`, which compares raw block coordinates.
81fn closer_than(pos: BlockPos, other: BlockPos, distance: f64) -> bool {
82    let dx = f64::from(pos.x() - other.x());
83    let dy = f64::from(pos.y() - other.y());
84    let dz = f64::from(pos.z() - other.z());
85    dz.mul_add(dz, dx.mul_add(dx, dy * dy)) < distance * distance
86}

88/// Everything the conduit works out again on each beat, plus the one thing it
89/// remembers between them.
90struct ConduitState {
91    /// Vanilla `isActive`. Kept only to notice the transition that plays the
92    /// activate and deactivate sounds.
93    active: bool,
94    /// How many frame blocks the last beat found. Vanilla keeps the positions
95    /// themselves, but only the client's particles ever read them back.
96    frame_blocks: usize,
97    /// Vanilla `destroyTarget`, an `EntityReference<LivingEntity>`. Foton keeps
98    /// the UUID alone and resolves it against the world, the way
99    /// `CreakingHeartBlockEntity` does.
100    destroy_target: Option<Uuid>,
101    /// Vanilla `nextAmbientSoundActivation`, which is deliberately not saved.
102    next_ambient_sound: i64,
103}

105/// A conduit's activation, effect and attack state.
106pub struct ConduitBlockEntity {
107    base: BlockEntityBase,
108    state: SyncMutex<ConduitState>,
109}

116impl ConduitBlockEntity {
117    /// Creates a conduit block entity.
118    #[must_use]
119    pub fn new(level: Weak<World>, pos: BlockPos, state: BlockStateId) -> Self {
120        Self {
121            base: BlockEntityBase::new(&vanilla_block_entity_types::CONDUIT, level, pos, state),
122            state: SyncMutex::new(ConduitState {
123                active: false,
124                frame_blocks: 0,
125                destroy_target: None,
126                next_ambient_sound: 0,
127            }),
128        }
129    }
130
131    /// Returns whether the last beat found a complete pocket and enough frame.
132    #[must_use]
133    pub fn is_active(&self) -> bool {
134        self.state.lock().active
135    }
136
137    /// Returns how many frame blocks the last beat counted.
138    #[must_use]
139    pub fn frame_block_count(&self) -> usize {
140        self.state.lock().frame_blocks
141    }
142
143    /// Returns the mob this conduit is currently hitting.
144    #[must_use]
145    pub fn destroy_target(&self) -> Option<Uuid> {
146        self.state.lock().destroy_target
147    }
148
149    fn play(&self, world: &Arc<World>, sound: SoundEventRef) {
150        world.play_sound(sound, SoundSource::Blocks, self.base.pos(), 1.0, 1.0, None);
151    }
152
153    /// Runs one two-second beat.
154    fn beat(&self, world: &Arc<World>, pos: BlockPos) {
155        let frame_blocks = count_frame_blocks(world, pos);
156        let active = frame_blocks.len() >= MIN_ACTIVE_SIZE;
157
158        let was_active = {
159            let mut state = self.state.lock();
160            let was_active = state.active;
161            state.active = active;
162            state.frame_blocks = frame_blocks.len();
163            was_active
164        };
165
166        if active != was_active {
167            self.play(
168                world,
169                if active {
170                    &sound_events::BLOCK_CONDUIT_ACTIVATE
171                } else {
172                    &sound_events::BLOCK_CONDUIT_DEACTIVATE
173                },
174            );
175        }
176
177        if !active {
178            return;
179        }
180        apply_effects(world, pos, frame_blocks.len());
181        self.update_and_attack_target(world, pos, frame_blocks.len() >= MIN_KILL_SIZE);
182    }
183
184    /// Vanilla parity: `ConduitBlockEntity.updateAndAttackTarget`.
185    fn update_and_attack_target(&self, world: &Arc<World>, pos: BlockPos, hunting: bool) {
186        let previous = self.state.lock().destroy_target;
187        let (target, resolved) = update_destroy_target(world, pos, previous, hunting);
188
189        if let Some(entity) = resolved
190            && let Some(living) = entity.as_living_entity()
191        {
192            world.play_sound_at(
193                &sound_events::BLOCK_CONDUIT_ATTACK_TARGET,
194                SoundSource::Blocks,
195                entity.position(),
196                1.0,
197                1.0,
198                None,
199            );
200            living.hurt_server(
201                world,
202                &DamageSource::environment(&vanilla_damage_types::MAGIC),
203                ATTACK_DAMAGE,
204            );
205        }
206
207        if target == previous {
208            return;
209        }
210        self.state.lock().destroy_target = target;
211        // Vanilla's `sendBlockUpdated(pos, state, state, 2)`: clients only, no
212        // neighbour update. The client needs the new target to know where to
213        // draw the beam, and nothing else about a conduit crosses the wire.
214        //
215        // Deliberately no `set_changed`: vanilla does not mark the chunk dirty
216        // here either. The target is worked out again on the next beat, so a
217        // save that misses it costs nothing, and dirtying a chunk every two
218        // seconds for a cosmetic pointer would be a real cost.
219        world.broadcast_block_entity_if_needed(pos);
220    }
221}

223/// Vanilla parity: `ConduitBlockEntity.updateShape`.
224///
225/// Returns the frame blocks found, which is empty when the 3x3x3 pocket around
226/// the conduit is not all water -- vanilla returns early there and leaves its
227/// list cleared, which is what keeps a drained conduit from counting as hunting.
228fn count_frame_blocks(world: &Arc<World>, pos: BlockPos) -> Vec<BlockPos> {
229    for ox in -1..=1 {
230        for oy in -1..=1 {
231            for oz in -1..=1 {
232                if !is_water_at(world, pos.offset(ox, oy, oz)) {
233                    return Vec::new();
234                }
235            }
236        }
237    }
238
239    let mut frame = Vec::new();
240    for ox in -2..=2_i32 {
241        for oy in -2..=2_i32 {
242            for oz in -2..=2_i32 {
243                if !is_on_frame(ox, oy, oz) {
244                    continue;
245                }
246                let test = pos.offset(ox, oy, oz);
247                if is_frame_block(world.get_block_state(test).get_block()) {
248                    frame.push(test);
249                }
250            }
251        }
252    }
253    frame
254}

256/// Whether an offset lies on one of the three axis-aligned rings of the frame.
257///
258/// Vanilla parity: the offset test inside `updateShape`'s second loop.
259const fn is_on_frame(ox: i32, oy: i32, oz: i32) -> bool {
260    let (ax, ay, az) = (ox.abs(), oy.abs(), oz.abs());
261    (ax > 1 || ay > 1 || az > 1)
262        && (ox == 0 && (ay == 2 || az == 2)
263            || oy == 0 && (ax == 2 || az == 2)
264            || oz == 0 && (ax == 2 || ay == 2))
265}

267/// Vanilla parity: `Level.isWaterAt`, which asks only whether the fluid is
268/// water -- flowing water counts, and a source is not required.
269fn is_water_at(world: &Arc<World>, pos: BlockPos) -> bool {
270    get_fluid_state(world, pos).is_water()
271}

273/// Vanilla parity: `ConduitBlockEntity.applyEffects`.
274///
275/// Deviation: vanilla collects players from an inflated box and then tests each
276/// one with `closerThan`. Foton walks the world's player list instead, the way
277/// `BeaconBlockEntity` does, and applies only the `closerThan` test. The box is
278/// a superset of that sphere, so the two agree on every player.
279fn apply_effects(world: &Arc<World>, pos: BlockPos, frame_blocks: usize) {
280    let range = (frame_blocks / 7 * 16) as f64;
281
282    let mut in_range = Vec::new();
283    world.players.iter_players(|_, player| {
284        if closer_than(pos, player.block_position(), range) && player.is_in_water_or_rain() {
285            in_range.push(Arc::clone(player));
286        }
287        true
288    });
289
290    for player in &in_range {
291        player.add_mob_effect(
292            MobEffectInstance::with_duration(
293                vanilla_mob_effects::CONDUIT_POWER,
294                EFFECT_DURATION_TICKS,
295                0,
296            )
297            .with_ambient(true),
298        );
299    }
300}

302/// Vanilla parity: `ConduitBlockEntity.updateDestroyTarget`.
303///
304/// Returns the target to keep and the entity it resolves to, if any. Vanilla
305/// drops a target that has died or swum away and does not pick a replacement
306/// until the next beat, which this preserves.
307fn update_destroy_target(
308    world: &Arc<World>,
309    pos: BlockPos,
310    current: Option<Uuid>,
311    hunting: bool,
312) -> (Option<Uuid>, Option<SharedEntity>) {
313    if !hunting {
314        return (None, None);
315    }
316
317    let Some(current) = current else {
318        let selected = select_new_target(world, pos);
319        let uuid = selected.as_ref().map(|entity| entity.uuid());
320        return (uuid, selected);
321    };
322
323    let resolved = world
324        .get_entity_by_uuid(&current)
325        .filter(|entity| !entity.is_removed());
326    let keep = resolved.as_ref().is_some_and(|entity| {
327        entity
328            .as_living_entity()
329            .is_some_and(LivingEntity::is_alive)
330            && closer_than(pos, entity.block_position(), KILL_RANGE)
331    });
332    if keep {
333        (Some(current), resolved)
334    } else {
335        (None, None)
336    }
337}

339/// Vanilla parity: `ConduitBlockEntity.selectNewTarget`, which picks uniformly
340/// at random rather than taking the nearest.
341fn select_new_target(world: &Arc<World>, pos: BlockPos) -> Option<SharedEntity> {
342    let (x, y, z) = (f64::from(pos.x()), f64::from(pos.y()), f64::from(pos.z()));
343    let aabb = WorldAabb::new(x, y, z, x + 1.0, y + 1.0, z + 1.0).inflate(KILL_RANGE);
344    let candidates = world.get_entities_in_aabb_matching(&aabb, |entity| {
345        entity.is_enemy() && entity.is_in_water_or_rain() && entity.as_living_entity().is_some()
346    });
347    if candidates.is_empty() {
348        return None;
349    }
350    let index = rand::rng().random_range(0..candidates.len());
351    candidates.get(index).cloned()
352}

354impl BlockEntity for ConduitBlockEntity {
355    fn base(&self) -> &BlockEntityBase {
356        &self.base
357    }
358
359    /// Vanilla parity: `ConduitBlockEntity.serverTick`.
360    fn tick(&self, world: &Arc<World>) {
361        let game_time = world.game_time();
362        let pos = self.get_block_pos();
363
364        if game_time.rem_euclid(BEAT_INTERVAL_TICKS) == 0 {
365            self.beat(world, pos);
366        }
367
368        if !self.is_active() {
369            return;
370        }
371
372        if game_time.rem_euclid(AMBIENT_INTERVAL_TICKS) == 0 {
373            self.play(world, &sound_events::BLOCK_CONDUIT_AMBIENT);
374        }
375
376        let due = {
377            let mut state = self.state.lock();
378            let due = game_time > state.next_ambient_sound;
379            if due {
380                state.next_ambient_sound =
381                    game_time + 60 + i64::from(rand::rng().random_range(0..40u32));
382            }
383            due
384        };
385        if due {
386            self.play(world, &sound_events::BLOCK_CONDUIT_AMBIENT_SHORT);
387        }
388    }
389
390    fn load_additional(&self, nbt: &BorrowedNbtCompound<'_>) {
391        let view: NbtCompoundView<'_, '_> = nbt.into();
392        self.state.lock().destroy_target = view
393            .int_array(TARGET_NBT_KEY)
394            .and_then(|array| Uuid::from_int_array(&array));
395    }
396
397    fn save_additional(&self, nbt: &mut NbtCompound) {
398        let Some(target) = self.state.lock().destroy_target else {
399            return;
400        };
401        nbt.insert(
402            TARGET_NBT_KEY,
403            NbtTag::IntArray(target.to_int_array().to_vec()),
404        );
405    }
406
407    fn get_update_tag(&self) -> Option<NbtCompound> {
408        Some(self.save_custom_only())
409    }
410}

412#[cfg(test)]
413mod tests {
414    use super::*;

416    /// The frame is three flat 5x5 rings, one per axis plane, overlapping in
417    /// pairs. Sixteen blocks each, six shared between two rings, so 42 -- which
418    /// is exactly `MIN_KILL_SIZE`, and the reason that number is what it is.
419    /// If the offset test ever drifted, this count would move off 42.
420    #[test]
421    fn a_full_frame_is_forty_two_blocks_on_three_rings() {
422        let mut count = 0;
423        let mut planes = [0; 3];
424        for ox in -2..=2 {
425            for oy in -2..=2 {
426                for oz in -2..=2 {
427                    if !is_on_frame(ox, oy, oz) {
428                        continue;
429                    }
430                    count += 1;
431                    // Every accepted offset lies in at least one axis plane.
432                    assert!(
433                        ox == 0 || oy == 0 || oz == 0,
434                        "({ox},{oy},{oz}) is on no axis plane"
435                    );
436                    planes[0] += i32::from(ox == 0);
437                    planes[1] += i32::from(oy == 0);
438                    planes[2] += i32::from(oz == 0);
439                }
440            }
441        }
442        assert_eq!(
443            count, MIN_KILL_SIZE,
444            "a full frame is exactly the kill threshold"
445        );
446        assert_eq!(planes, [16, 16, 16], "each ring is a 5x5 square minus 3x3");
447
448        // A corner of the 5x5x5 cube lies in no axis plane.
449        assert!(!is_on_frame(2, 2, 2));
450        // Nor does anything else off all three planes.
451        assert!(!is_on_frame(2, 2, 1));
452        assert!(!is_on_frame(2, 1, 1));
453        // Two rings share the middle of each cube edge that touches a plane.
454        assert!(is_on_frame(2, 0, 0));
455        assert!(is_on_frame(2, 2, 0));
456        assert!(is_on_frame(0, 2, 1));
457        // Nothing inside the 3x3x3 water pocket counts.
458        assert!(!is_on_frame(1, 1, 1));
459        assert!(!is_on_frame(0, 0, 0));
460        assert!(!is_on_frame(0, 1, 1));
461    }
462
463    /// The effect range is a step function of the frame, not a smooth one:
464    /// vanilla divides by seven with integer division before multiplying.
465    #[test]
466    fn the_effect_range_steps_with_every_seventh_block() {
467        let range = |blocks: usize| blocks / 7 * 16;
468        assert_eq!(range(MIN_ACTIVE_SIZE), 32);
469        assert_eq!(range(20), 32, "four spare blocks buy nothing");
470        assert_eq!(range(21), 48);
471        assert_eq!(range(MIN_KILL_SIZE), 96);
472    }
473
474    /// `closerThan` compares raw block coordinates and is strict, so a mob
475    /// exactly `KILL_RANGE` away in one axis is out of reach.
476    #[test]
477    fn the_kill_range_is_a_strict_sphere_on_block_coordinates() {
478        let conduit = BlockPos::new(0, 64, 0);
479        assert!(closer_than(conduit, BlockPos::new(7, 64, 0), KILL_RANGE));
480        assert!(!closer_than(conduit, BlockPos::new(8, 64, 0), KILL_RANGE));
481        // Diagonals are measured, not clamped to a box.
482        assert!(!closer_than(conduit, BlockPos::new(6, 64, 6), KILL_RANGE));
483        assert!(closer_than(conduit, BlockPos::new(0, 71, 0), KILL_RANGE));
484    }
