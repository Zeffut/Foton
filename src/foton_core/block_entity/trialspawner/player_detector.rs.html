1//! Who a trial spawner or a vault counts as being watched by.
2//!
3//! Vanilla parity:
4//! `net.minecraft.world.level.block.entity.trialspawner.PlayerDetector`.
5//!
6//! Vanilla threads a second interface, `PlayerDetector.EntitySelector`, so a
7//! game test can hand the detector a fixed list of players instead of the
8//! level. Foton has no such test harness, so the level is the only source and
9//! the indirection is left out. `PlayerDetector.SHEEP` is likewise absent: it
10//! exists only behind `SharedConstants.DEBUG_TRIAL_SPAWNER_DETECTS_SHEEP_AS_PLAYERS`.
11

21/// Which players a spawner is willing to see.
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub enum PlayerDetector {
24    /// Vanilla parity: `PlayerDetector.NO_CREATIVE_PLAYERS`, the trial spawner's.
25    NoCreativePlayers,
26    /// Vanilla parity: `PlayerDetector.INCLUDING_CREATIVE_PLAYERS`, the vault's.
27    IncludingCreativePlayers,
28}
29
30impl PlayerDetector {
31    /// Returns the players inside `required_player_range` of `spawner_pos`.
32    ///
33    /// Vanilla parity: `PlayerDetector.detect`. The range test is vanilla's
34    /// `BlockPos.closerThan`, which measures from block position to block
35    /// position, not from the player's feet.
36    #[must_use]
37    pub fn detect(
38        self,
39        world: &Arc<World>,
40        spawner_pos: BlockPos,
41        required_player_range: f64,
42        require_line_of_sight: bool,
43    ) -> Vec<Uuid> {
44        let center = center_of(spawner_pos);
45        let range_sqr = required_player_range * required_player_range;
46        let mut detected = Vec::new();
47
48        world.players.iter_players(|_, player| {
49            if player.is_spectator() {
50                return true;
51            }
52            if self == Self::NoCreativePlayers && player.has_infinite_materials() {
53                return true;
54            }
55            if !closer_than(player.block_position(), spawner_pos, range_sqr) {
56                return true;
57            }
58            let eye = DVec3::new(player.position().x, player.get_eye_y(), player.position().z);
59            if require_line_of_sight && !in_line_of_sight(world, center, eye) {
60                return true;
61            }
62            detected.push(player.uuid());
63            true
64        });
65
66        detected
67    }
68}

70/// Vanilla parity: `Vec3.atCenterOf`.
71pub(crate) fn center_of(pos: BlockPos) -> DVec3 {
72    DVec3::new(
73        f64::from(pos.x()) + 0.5,
74        f64::from(pos.y()) + 0.5,
75        f64::from(pos.z()) + 0.5,
76    )
77}
78
79/// Vanilla parity: `BlockPos.closerThan(Vec3i, double)`, which compares squared
80/// distances between block positions.
81fn closer_than(left: BlockPos, right: BlockPos, range_sqr: f64) -> bool {
82    let dx = f64::from(left.x() - right.x());
83    let dy = f64::from(left.y() - right.y());
84    let dz = f64::from(left.z() - right.z());
85    dz.mul_add(dz, dx.mul_add(dx, dy * dy)) < range_sqr
86}

88/// Returns whether nothing solid stands between the two points.
89///
90/// Vanilla parity: the private `inLineOfSight` of `PlayerDetector` and
91/// `TrialSpawner`, which clip from the destination back to the origin and
92/// accept a hit on the origin block itself.
93pub(crate) fn in_line_of_sight(world: &Arc<World>, origin: DVec3, destination: DVec3) -> bool {
94    let hit = world.clip(destination, origin, ClipBlockShape::Visual, ClipFluid::None);
95    hit.miss
96        || hit.block_pos
97            == BlockPos::new(
98                origin.x.floor() as i32,
99                origin.y.floor() as i32,
100                origin.z.floor() as i32,
101            )
102}
