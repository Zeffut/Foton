1//! The data components a block entity carries between block and item.
2//!
3//! Vanilla parity: the `components` field of `BlockEntity` together with
4//! `applyImplicitComponents`/`collectImplicitComponents`.
5//!
6//! A block entity holds two kinds of component. The *implicit* ones are the
7//! ones it already stores as fields -- a banner's patterns, a hive's bees, a
8//! container's name -- and it rebuilds them on demand. Everything else an item
9//! put on it is kept verbatim in the stored map so the block can hand it back
10//! when it becomes an item again.
11
12use std::cell::RefCell;
13
14use foton_registry::data_components::{
15    Component, DataComponentMap, DataComponentType,
16    vanilla_components::{BLOCK_ENTITY_DATA, BLOCK_STATE},
17};
18use foton_registry::item_stack::ItemStack;
19use foton_utils::{DowncastType, Identifier};
20
21/// The merged item view an `apply_implicit_components` override reads from.
22///
23/// Vanilla parity: the anonymous `DataComponentGetter` built inside
24/// `BlockEntity.applyComponents`. It records every component type an override
25/// asks for, because those are exactly the ones the block entity has taken
26/// responsibility for; the rest of the item's patch is stored verbatim.
27pub struct ImplicitComponentInput<'a> {
28    stack: &'a ItemStack,
29    /// The types an override has asked for. Only ever touched by the single
30    /// thread running the override, which is why a `RefCell` is enough.
31    read: RefCell<Vec<Identifier>>,
32}
33
34impl<'a> ImplicitComponentInput<'a> {
35    /// Wraps the item whose components are being applied.
36    #[must_use]
37    pub const fn new(stack: &'a ItemStack) -> Self {
38        Self {
39            stack,
40            read: RefCell::new(Vec::new()),
41        }
42    }
43
44    /// Vanilla parity: `DataComponentGetter.get`.
45    #[must_use]
46    pub fn get<T: Component + DowncastType + Clone>(
47        &self,
48        component: DataComponentType<T>,
49    ) -> Option<T> {
50        self.read.borrow_mut().push(component.key().clone());
51        self.stack.get(component).cloned()
52    }
53
54    /// Vanilla parity: `DataComponentGetter.getOrDefault`.
55    #[must_use]
56    pub fn get_or_default<T: Component + DowncastType + Clone>(
57        &self,
58        component: DataComponentType<T>,
59        default: T,
60    ) -> T {
61        self.get(component).unwrap_or(default)
62    }
63
64    /// The stored half of `BlockEntity.applyComponents`: the item's patch with
65    /// every type the overrides claimed forgotten, keeping only what it adds.
66    ///
67    /// `BLOCK_ENTITY_DATA` and `BLOCK_STATE` are claimed whether an override
68    /// asked for them or not, exactly as vanilla seeds its implicit set.
69    pub(super) fn leftover_components(self) -> DataComponentMap {
70        let mut patch = self.stack.patch().clone();
71        patch.clear_key(BLOCK_ENTITY_DATA.key());
72        patch.clear_key(BLOCK_STATE.key());
73        for key in self.read.into_inner() {
74            patch.clear_key(&key);
75        }
76        patch.added()
77    }
78}
