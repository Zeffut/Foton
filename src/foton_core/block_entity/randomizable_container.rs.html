1//! The loot table a container carries until something first looks inside it.
2//!
3//! Vanilla parity: `net.minecraft.world.RandomizableContainer`, whose
4//! `lootTable`/`lootTableSeed` pair lives on `RandomizableContainerBlockEntity`
5//! for blocks and is duplicated by `ContainerEntity` for the chest minecart,
6//! the hopper minecart and the chest boats. Rust has no inheritance, so every
7//! randomizable container owns a [`ContainerLoot`] and forwards to it.
8//!
9//! Worldgen writes the table; nothing rolls it until the container is read or
10//! written. That is deliberate in vanilla: a chest generated a thousand blocks
11//! away costs nothing until a player -- or a hopper -- reaches it, and the
12//! stored seed makes the eventual roll independent of when that happens.
13

35/// Vanilla parity: `RandomizableContainer.LOOT_TABLE_TAG`.
36const LOOT_TABLE_TAG: &str = "LootTable";
37
38/// Vanilla parity: `RandomizableContainer.LOOT_TABLE_SEED_TAG`.
39const LOOT_TABLE_SEED_TAG: &str = "LootTableSeed";
40
41/// Lets [`foton_registry::loot_table::LootTable::fill`] place rolled loot into
42/// any Foton container.
43///
44/// `LootTable` lives in `foton-registry`, which cannot see [`Container`], so
45/// the adapter has to be written on this side of the crate boundary.
46impl LootFillContainer for dyn Container {
47    fn get_container_size(&self) -> usize {
48        Container::get_container_size(self)
49    }
50
51    fn get_item(&self, slot: usize) -> &ItemStack {
52        Container::get_item(self, slot)
53    }
54
55    fn set_item(&mut self, slot: usize, stack: ItemStack) {
56        Container::set_item(self, slot, stack);
57    }
58}

60/// A loot table that has not been rolled into its container yet.
61#[derive(Debug, Clone, PartialEq, Eq)]
62struct PackedLootTable {
63    key: Identifier,
64    seed: i64,
65}
66
67/// The `lootTable`/`lootTableSeed` pair of a randomizable container.
68///
69/// Vanilla parity: the fields and default methods of `RandomizableContainer`.
70///
71/// `packed` mirrors "a table is still waiting" without the mutex, because every
72/// read of and write to the container has to ask, and almost every container in
73/// the world answers no.
74#[derive(Debug, Default)]
75pub struct ContainerLoot {
76    packed: AtomicBool,
77    table: SyncMutex<Option<PackedLootTable>>,
78}
79
80impl ContainerLoot {
81    /// Creates a container loot slot with nothing packed in it.
82    #[must_use]
83    pub fn new() -> Self {
84        Self::default()
85    }
86
87    /// Arms this container with a table to roll on first access.
88    ///
89    /// Vanilla parity: `RandomizableContainer.setLootTable(lootTable, seed)`.
90    pub fn set_loot_table(&self, key: Identifier, seed: i64) {
91        *self.table.lock() = Some(PackedLootTable { key, seed });
92        self.packed.store(true, Ordering::Relaxed);
93    }
94
95    /// Returns the table still waiting to be rolled.
96    ///
97    /// Vanilla parity: `RandomizableContainer.getLootTable`.
98    #[must_use]
99    pub fn loot_table(&self) -> Option<Identifier> {
100        self.table.lock().as_ref().map(|packed| packed.key.clone())
101    }
102
103    /// Returns the seed the packed table will be rolled with.
104    ///
105    /// Vanilla parity: `RandomizableContainer.getLootTableSeed`.
106    #[must_use]
107    pub fn loot_table_seed(&self) -> i64 {
108        self.table.lock().as_ref().map_or(0, |packed| packed.seed)
109    }
110
111    /// Returns whether a table is still waiting to be rolled.
112    #[must_use]
113    pub fn is_packed(&self) -> bool {
114        self.packed.load(Ordering::Relaxed)
115    }
116
117    /// Reads the pair off disk, answering whether a table was found.
118    ///
119    /// Vanilla parity: `RandomizableContainer.tryLoadLootTable`. A `false`
120    /// answer is the caller's signal to load the `Items` list instead: a
121    /// container never stores both.
122    pub fn try_load_loot_table(&self, nbt: &NbtCompoundView<'_, '_>) -> bool {
123        let key = nbt
124            .string(LOOT_TABLE_TAG)
125            .and_then(|value| Identifier::from_str(&value.to_string()).ok());
126        let seed = nbt.long(LOOT_TABLE_SEED_TAG).unwrap_or(0);
127
128        let found = key.is_some();
129        *self.table.lock() = key.map(|key| PackedLootTable { key, seed });
130        self.packed.store(found, Ordering::Relaxed);
131        found
132    }
133
134    /// Writes the pair to disk, answering whether there was one to write.
135    ///
136    /// Vanilla parity: `RandomizableContainer.trySaveLootTable`, which omits a
137    /// zero seed because zero means "roll me freshly" rather than "roll me with
138    /// seed zero".
139    pub fn try_save_loot_table(&self, nbt: &mut NbtCompound) -> bool {
140        let Some(packed) = self.table.lock().clone() else {
141            return false;
142        };
143        nbt.insert(LOOT_TABLE_TAG, packed.key.to_string());
144        if packed.seed != 0 {
145            nbt.insert(LOOT_TABLE_SEED_TAG, packed.seed);
146        }
147        true
148    }
149
150    /// Rolls a still-packed table into the container owned by `base`.
151    ///
152    /// Vanilla parity: `RandomizableContainer.unpackLootTable`, whose `ORIGIN`
153    /// is the center of the block.
154    pub(crate) fn unpack_for_block_entity(
155        &self,
156        base: &BlockEntityBase,
157        container: &SharedContainer,
158        player: Option<&Player>,
159    ) {
160        if !self.is_packed() {
161            return;
162        }
163        let Some(world) = base.level() else {
164            return;
165        };
166        let pos = base.pos();
167        let origin = DVec3::new(
168            f64::from(pos.x()) + 0.5,
169            f64::from(pos.y()) + 0.5,
170            f64::from(pos.z()) + 0.5,
171        );
172        if self.unpack_at(&world, origin, container, player) {
173            // Vanilla parity: `fill` reaches the container through `setItem`,
174            // which is what marks a block entity changed.
175            base.set_changed();
176        }
177    }
178
179    /// Rolls a still-packed table into `container`, answering whether it did.
180    ///
181    /// Vanilla parity: the body shared by `RandomizableContainer.unpackLootTable`
182    /// and `ContainerEntity.unpackChestVehicleLootTable`; they differ only in
183    /// where `ORIGIN` comes from, so a moving container passes its own position.
184    pub fn unpack_at(
185        &self,
186        world: &Arc<World>,
187        origin: DVec3,
188        container: &SharedContainer,
189        player: Option<&Player>,
190    ) -> bool {
191        // Vanilla clears the field before filling, so the `setItem` calls that
192        // follow cannot roll the same table a second time.
193        let Some(packed) = self.take() else {
194            return false;
195        };
196        let Some(table) = REGISTRY.loot_tables.by_key(&packed.key) else {
197            log::warn!(
198                "Container at {origin} referenced unknown loot table {}",
199                packed.key
200            );
201            return false;
202        };
203
204        // Vanilla fires this before it fills, on the table it just took.
205        if let Some(player) = player {
206            triggers::world::player_generates_container_loot(player, &packed.key);
207        }
208
209        if packed.seed == 0 {
210            // Vanilla parity: `LootContext.Builder.withOptionalRandomSeed`
211            // leaves the level's own random source in place for seed zero.
212            Self::fill(table, container, &mut rand::rng(), world, origin, player);
213        } else {
214            Self::fill(
215                table,
216                container,
217                &mut StdRng::seed_from_u64(packed.seed as u64),
218                world,
219                origin,
220                player,
221            );
222        }
223        true
224    }
225
226    /// Clears the packed table and returns it, if there was one.
227    fn take(&self) -> Option<PackedLootTable> {
228        if !self.is_packed() {
229            return None;
230        }
231        let packed = self.table.lock().take();
232        self.packed.store(false, Ordering::Relaxed);
233        packed
234    }
235
236    fn fill<R: rand::Rng>(
237        table: LootTableRef,
238        container: &SharedContainer,
239        rng: &mut R,
240        world: &Arc<World>,
241        origin: DVec3,
242        player: Option<&Player>,
243    ) {
244        let mut ctx = LootContext::new(rng)
245            .with_world(&**world)
246            .with_origin(origin.x, origin.y, origin.z)
247            .with_game_time(world.game_time());
248        if let Some(player) = player {
249            ctx = ctx
250                .with_luck(player.get_luck())
251                .with_this_entity(entity_loot_ref(player));
252        }
253
254        let mut locked = container.lock();
255        table.fill(&mut *locked, &mut ctx);
256    }
257}

261    use std::io::Cursor;

263    use simdnbt::borrow::read_compound as read_borrowed_compound;

269    fn view_of(nbt: &NbtCompound) -> Vec<u8> {
270        let mut bytes = Vec::new();
271        nbt.write(&mut bytes);
272        bytes
273    }
274
275    fn load(loot: &ContainerLoot, nbt: &NbtCompound) -> bool {
276        let bytes = view_of(nbt);
277        let base = read_borrowed_compound(&mut Cursor::new(bytes.as_slice()))
278            .expect("test nbt should reborrow");
279        let view: NbtCompoundView<'_, '_> = (&base).into();
280        loot.try_load_loot_table(&view)
281    }
282
283    #[test]
284    fn a_saved_pair_survives_a_round_trip() {
285        let loot = ContainerLoot::new();
286        loot.set_loot_table(
287            Identifier::new_static("minecraft", "chests/simple_dungeon"),
288            42,
289        );
290
291        let mut saved = NbtCompound::new();
292        assert!(loot.try_save_loot_table(&mut saved));
293
294        let reloaded = ContainerLoot::new();
295        assert!(load(&reloaded, &saved));
296        assert_eq!(
297            reloaded.loot_table().map(|key| key.to_string()),
298            Some("minecraft:chests/simple_dungeon".to_owned())
299        );
300        assert_eq!(reloaded.loot_table_seed(), 42);
301    }
302
303    /// Zero is vanilla's "pick a seed when you roll me", so it is not written.
304    #[test]
305    fn a_zero_seed_is_left_out_of_the_save() {
306        let loot = ContainerLoot::new();
307        loot.set_loot_table(Identifier::new_static("minecraft", "chests/igloo_chest"), 0);
308
309        let mut saved = NbtCompound::new();
310        assert!(loot.try_save_loot_table(&mut saved));
311
312        assert!(saved.string("LootTable").is_some());
313        assert!(saved.long("LootTableSeed").is_none());
314    }
315
316    #[test]
317    fn a_container_with_no_table_saves_nothing_and_tells_the_caller() {
318        let loot = ContainerLoot::new();
319        let mut saved = NbtCompound::new();
320
321        assert!(!loot.try_save_loot_table(&mut saved));
322        assert!(saved.string("LootTable").is_none());
323        assert!(!loot.is_packed());
324    }
325
326    /// Loading a chest that has already been opened must clear an older table,
327    /// otherwise a block entity reused for a second position would roll again.
328    #[test]
329    fn loading_without_a_table_clears_a_previous_one() {
330        let loot = ContainerLoot::new();
331        loot.set_loot_table(
332            Identifier::new_static("minecraft", "chests/simple_dungeon"),
333            7,
334        );
335
336        let mut empty = NbtCompound::new();
337        empty.insert("Items", NbtList::Compound(Vec::new()));
338        assert!(!load(&loot, &empty));
339
340        assert!(!loot.is_packed());
341        assert_eq!(loot.loot_table(), None);
342    }
