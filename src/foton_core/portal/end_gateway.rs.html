19const GATEWAY_HEIGHT_ABOVE_SURFACE: i32 = 10;
20const EXIT_PORTAL_SEARCH_DISTANCE: f64 = 1024.0;
21const EXIT_PORTAL_SEARCH_STEP: f64 = 16.0;
22const EXIT_PORTAL_SEARCH_LIMIT: i32 = 16;
23const EXIT_POSITION_SEARCH_RADIUS: i32 = 5;
24const VALID_TELEPORT_SEARCH_RADIUS: i32 = 16;
25const GENERATED_ISLAND_Y: i32 = 75;

27/// Initial chunk preparation needed before resolving an End gateway transition.
28pub(crate) enum EndGatewayChunkPreparation {
29    /// Requested chunks are enough to calculate the transition after they load.
30    Ready(Vec<ChunkPos>),
31    /// Requested chunks only cover vanilla's tentative outer-island search.
32    SearchPath(Vec<ChunkPos>),
33}
34
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36enum GatewayExitState {
37    Stored { exit: BlockPos, exact: bool },
38    Missing { exact: bool },
39}
40
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42enum GatewayTeleportAnchor {
43    Existing(BlockPos),
44    NeedsIsland(BlockPos),
45}
46
47impl GatewayTeleportAnchor {
48    const fn pos(self) -> BlockPos {
49        match self {
50            Self::Existing(pos) | Self::NeedsIsland(pos) => pos,
51        }
52    }
53}

55/// Returns the first chunks that must be ready before an End gateway transition can be resolved.
56#[must_use]
57pub(crate) fn initial_chunks(
58    world: &World,
59    portal_pos: BlockPos,
60    source_is_end: bool,
61) -> Option<EndGatewayChunkPreparation> {
62    let Some(state) = gateway_exit_state(world, portal_pos) else {
63        // The block is there and the player walked into it, so the block
64        // entity that holds the exit should be there too. If it is not, the
65        // gateway is inert and nothing else in the chain can say so.
66        log::warn!(
67            "end gateway at {portal_pos:?} has no gateway block entity, so it leads nowhere"
68        );
69        return None;
70    };
71    match state {
72        GatewayExitState::Stored { exit, exact: true } => Some(EndGatewayChunkPreparation::Ready(
73            chunks_for_block_square(exit, 0),
74        )),
75        GatewayExitState::Stored { exit, exact: false } => Some(EndGatewayChunkPreparation::Ready(
76            chunks_for_block_square(exit.offset(0, 2, 0), EXIT_POSITION_SEARCH_RADIUS),
77        )),
78        GatewayExitState::Missing { .. } if source_is_end => Some(
79            EndGatewayChunkPreparation::SearchPath(exit_search_candidate_chunks(portal_pos)),
80        ),
81        // Vanilla only searches for an outer island from inside the End. A
82        // gateway anywhere else with no stored exit has nowhere to send anyone.
83        GatewayExitState::Missing { .. } => {
84            log::warn!(
85                "end gateway at {portal_pos:?} stores no exit and is not in the End, \
86                 so there is nowhere to search"
87            );
88            None
89        }
90    }
91}

93/// Returns final chunks needed after the tentative outer-island search chunks are ready.
94#[must_use]
95pub(crate) fn final_chunks_after_search(
96    world: &World,
97    portal_pos: BlockPos,
98    source_is_end: bool,
99) -> Option<Vec<ChunkPos>> {
100    let Some(state) = gateway_exit_state(world, portal_pos) else {
101        log::warn!("end gateway at {portal_pos:?} lost its block entity mid-search");
102        return None;
103    };
104    match state {
105        GatewayExitState::Stored { exit, exact: true } => Some(chunks_for_block_square(exit, 0)),
106        GatewayExitState::Stored { exit, exact: false } => Some(chunks_for_block_square(
107            exit.offset(0, 2, 0),
108            EXIT_POSITION_SEARCH_RADIUS,
109        )),
110        GatewayExitState::Missing { .. } if source_is_end => {
111            // The search walks chunk by chunk toward the outer islands and
112            // gives up the moment one of them is not loaded, which is the
113            // likeliest way a gateway the dragon opened ends up inert.
114            let Some(anchor) = find_teleport_anchor(world, portal_pos) else {
115                log::warn!(
116                    "end gateway at {portal_pos:?} could not read its way out to the \
117                     outer islands; a chunk along the search line is missing"
118                );
119                return None;
120            };
121            Some(chunks_for_block_square(
122                anchor.pos(),
123                VALID_TELEPORT_SEARCH_RADIUS,
124            ))
125        }
126        GatewayExitState::Missing { .. } => {
127            log::warn!(
128                "end gateway at {portal_pos:?} stores no exit and is not in the End, \
129                 so the search has nowhere to go"
130            );
131            None
132        }
133    }
134}

136/// Calculates vanilla's End gateway transition after the required chunks are available.
137#[must_use]
138pub(crate) fn calculate_transition(
139    world: &Arc<World>,
140    entity: &dyn Entity,
141    portal_pos: BlockPos,
142    source_is_end: bool,
143) -> Option<TeleportTransition> {
144    let Some(state) = gateway_exit_state(world, portal_pos) else {
145        log::warn!("end gateway at {portal_pos:?} has no block entity to read an exit from");
146        return None;
147    };
148    let (exit, exact) = match state {
149        GatewayExitState::Stored { exit, exact } => (exit, exact),
150        // The gateways the dragon's death leaves behind take this path every
151        // time: they store no exit, so the outer island has to be found first.
152        GatewayExitState::Missing { exact } if source_is_end => {
153            let Some(found) = find_or_create_valid_teleport_pos(world, portal_pos) else {
154                log::warn!(
155                    "end gateway at {portal_pos:?} found no outer-island landing spot to aim at"
156                );
157                return None;
158            };
159            let exit = found.above_n(GATEWAY_HEIGHT_ABOVE_SURFACE);
160            if !world.create_end_gateway_portal(exit, portal_pos, false) {
161                log::error!("Unable to create End gateway portal at {}", world.key);
162                return None;
163            }
164            if !set_gateway_exit_position(world, portal_pos, exit, exact) {
165                log::warn!("end gateway at {portal_pos:?} could not record its exit at {exit:?}");
166                return None;
167            }
168            (exit, exact)
169        }
170        GatewayExitState::Missing { .. } => {
171            log::warn!(
172                "end gateway at {portal_pos:?} stores no exit and is not in the End, \
173                 so there is nowhere to search"
174            );
175            return None;
176        }
177    };
178
179    let destination = if exact {
180        exit
181    } else {
182        find_exit_position(world, exit)
183    };
184    Some(gateway_transition(world, entity, destination))
185}

187fn gateway_exit_state(world: &World, portal_pos: BlockPos) -> Option<GatewayExitState> {
188    let block_entity = world.get_block_entity(portal_pos)?;
189    let gateway = block_entity.downcast_ref::<EndGatewayBlockEntity>()?;
190    Some(match gateway.exit_portal() {
191        Some(exit) => GatewayExitState::Stored {
192            exit,
193            exact: gateway.exact_teleport(),
194        },
195        None => GatewayExitState::Missing {
196            exact: gateway.exact_teleport(),
197        },
198    })
199}

201fn set_gateway_exit_position(
202    world: &World,
203    portal_pos: BlockPos,
204    exit: BlockPos,
205    exact: bool,
206) -> bool {
207    let Some(block_entity) = world.get_block_entity(portal_pos) else {
208        return false;
209    };
210    let Some(gateway) = block_entity.downcast_ref::<EndGatewayBlockEntity>() else {
211        return false;
212    };
213    gateway.set_exit_position(exit, exact);
214    true
215}

217fn find_exit_position(world: &World, exit_portal: BlockPos) -> BlockPos {
218    world
219        .find_end_gateway_tallest_block(
220            exit_portal.offset(0, 2, 0),
221            EXIT_POSITION_SEARCH_RADIUS,
222            false,
223        )
224        .above()
225}

227fn find_or_create_valid_teleport_pos(
228    world: &Arc<World>,
229    gateway_pos: BlockPos,
230) -> Option<BlockPos> {
231    let anchor = find_teleport_anchor(world, gateway_pos)?;
232    if let GatewayTeleportAnchor::NeedsIsland(pos) = anchor
233        && !world.create_end_island(pos)
234    {
235        log::error!("Unable to create End island at {}", world.key);
236        return None;
237    }
238
239    Some(world.find_end_gateway_tallest_block(anchor.pos(), VALID_TELEPORT_SEARCH_RADIUS, true))
240}

242fn find_teleport_anchor(world: &World, gateway_pos: BlockPos) -> Option<GatewayTeleportAnchor> {
243    let tentative = find_exit_portal_xz_pos_tentative(world, gateway_pos)?;
244    let chunk = chunk_for_xz_vec(tentative);
245    if let Some(pos) = world.find_end_gateway_valid_spawn_in_chunk(chunk) {
246        return Some(GatewayTeleportAnchor::Existing(pos));
247    }
248
249    Some(GatewayTeleportAnchor::NeedsIsland(BlockPos::new(
250        (tentative.x + 0.5).floor() as i32,
251        GENERATED_ISLAND_Y,
252        (tentative.z + 0.5).floor() as i32,
253    )))
254}
255
256fn find_exit_portal_xz_pos_tentative(world: &World, gateway_pos: BlockPos) -> Option<DVec3> {
257    let direction = xz_direction(gateway_pos);
258    let mut tentative = direction * EXIT_PORTAL_SEARCH_DISTANCE;
259
260    let mut remaining = EXIT_PORTAL_SEARCH_LIMIT;
261    while !is_chunk_empty(world, tentative)? && remaining > 0 {
262        remaining -= 1;
263        tentative -= direction * EXIT_PORTAL_SEARCH_STEP;
264    }
265
266    let mut remaining = EXIT_PORTAL_SEARCH_LIMIT;
267    while is_chunk_empty(world, tentative)? && remaining > 0 {
268        remaining -= 1;
269        tentative += direction * EXIT_PORTAL_SEARCH_STEP;
270    }
271
272    Some(tentative)
273}

275fn is_chunk_empty(world: &World, xz_pos: DVec3) -> Option<bool> {
276    world.is_end_gateway_chunk_empty(chunk_for_xz_vec(xz_pos))
277}

279fn gateway_transition(
280    world: &Arc<World>,
281    entity: &dyn Entity,
282    destination: BlockPos,
283) -> TeleportTransition {
284    let is_ender_pearl = entity.entity_type() == &vanilla_entities::ENDER_PEARL;
285    TeleportTransition {
286        target_world: world.clone(),
287        cause: TeleportTransitionCause::EndGateway,
288        position: block_bottom_center(destination),
289        rotation: (0.0, 0.0),
290        velocity: DVec3::ZERO,
291        relatives: if is_ender_pearl {
292            RelativeMovement::NONE
293        } else {
294            RelativeMovement::DELTA.union(RelativeMovement::ROTATION)
295        },
296        portal_cooldown: entity.dimension_changing_delay(),
297        as_passenger: false,
298        post_transition: TeleportPostTransition::place_portal_ticket(
299            PortalTicketTarget::Destination,
300        ),
301    }
302}

304fn exit_search_candidate_chunks(gateway_pos: BlockPos) -> Vec<ChunkPos> {
305    let direction = xz_direction(gateway_pos);
306    let start = direction * EXIT_PORTAL_SEARCH_DISTANCE;
307    let mut chunks = Vec::with_capacity((EXIT_PORTAL_SEARCH_LIMIT * 2 + 1) as usize);
308    for step in -EXIT_PORTAL_SEARCH_LIMIT..=EXIT_PORTAL_SEARCH_LIMIT {
309        chunks.push(chunk_for_xz_vec(
310            start + direction * (f64::from(step) * EXIT_PORTAL_SEARCH_STEP),
311        ));
312    }
313    chunks
314}

316fn chunks_for_block_square(center: BlockPos, block_radius: i32) -> Vec<ChunkPos> {
317    let min_chunk_x = SectionPos::block_to_section_coord(center.x() - block_radius);
318    let max_chunk_x = SectionPos::block_to_section_coord(center.x() + block_radius);
319    let min_chunk_z = SectionPos::block_to_section_coord(center.z() - block_radius);
320    let max_chunk_z = SectionPos::block_to_section_coord(center.z() + block_radius);
321    let mut chunks = Vec::with_capacity(
322        ((max_chunk_x - min_chunk_x + 1) * (max_chunk_z - min_chunk_z + 1)) as usize,
323    );
324
325    for chunk_z in min_chunk_z..=max_chunk_z {
326        for chunk_x in min_chunk_x..=max_chunk_x {
327            chunks.push(ChunkPos::new(chunk_x, chunk_z));
328        }
329    }
330    chunks
331}
332
333fn chunk_for_xz_vec(pos: DVec3) -> ChunkPos {
334    ChunkPos::new((pos.x / 16.0).floor() as i32, (pos.z / 16.0).floor() as i32)
335}
336
337fn xz_direction(pos: BlockPos) -> DVec3 {
338    let vector = DVec3::new(f64::from(pos.x()), 0.0, f64::from(pos.z()));
339    let length = vector.length();
340    if length < 1.0E-4 {
341        DVec3::ZERO
342    } else {
343        vector / length
344    }
345}
346
347fn block_bottom_center(pos: BlockPos) -> DVec3 {
348    let (x, y, z) = pos.get_bottom_center();
349    DVec3::new(x, y, z)
350}

371    /// Puts a gateway with no stored exit where the dragon fight puts them.
372    ///
373    /// This is the case that matters. `EnderDragonFight.spawnNewGateway` places
374    /// `END_GATEWAY_DELAYED`, whose configuration carries no exit at all, so
375    /// every gateway the fight opens has to find the outer islands from
376    /// nothing. A gateway with a stored exit skips the entire path below.
377    fn dragon_gateway(world: &Arc<World>, pos: BlockPos) {
378        insert_ready_full_chunk(world, ChunkPos::from_block_pos(pos));
379        let state = vanilla_blocks::END_GATEWAY.default_state();
380        world.set_block(pos, state, UpdateFlags::UPDATE_ALL);
381        world.set_block_entity(Arc::new(EndGatewayBlockEntity::new(
382            Arc::downgrade(world),
383            pos,
384            state,
385        )));
386    }

388    /// Serves exactly the chunks the teleport job would have loaded.
389    ///
390    /// The job is two chunk requests wrapped around `calculate_transition` and
391    /// nothing else, so a world holding those chunks exercises everything the
392    /// job would have reached -- without needing a `Server` to schedule them.
393    fn load_what_the_job_loads(world: &Arc<World>, gateway: BlockPos) {
394        let Some(EndGatewayChunkPreparation::SearchPath(search)) =
395            initial_chunks(world, gateway, true)
396        else {
397            panic!("a gateway with no exit, inside the End, has to ask for a search path");
398        };
399        // The job's two requests overlap, and the chunk map refuses the same
400        // chunk twice, so this keeps track of what it has already served.
401        let mut loaded = vec![ChunkPos::from_block_pos(gateway)];
402        let serve = |chunks: Vec<ChunkPos>, loaded: &mut Vec<ChunkPos>| {
403            for chunk in chunks {
404                if !loaded.contains(&chunk) {
405                    loaded.push(chunk);
406                    insert_ready_full_chunk(world, chunk);
407                }
408            }
409        };
410        serve(search, &mut loaded);
411        let final_chunks = final_chunks_after_search(world, gateway, true)
412            .expect("the search path is loaded, so the anchor has to resolve");
413        serve(final_chunks, &mut loaded);
414    }

416    /// A gateway the dragon opened resolves somewhere to send you.
417    ///
418    /// Everything before this is covered elsewhere: the block triggers, the
419    /// portal processor fires, the world change is queued. All the server adds
420    /// is the two chunk loads this test performs by hand -- so this is the
421    /// last stretch, and it was broken.
422    ///
423    /// `create_end_gateway_portal` writes a three-by-three-by-five box, most of
424    /// it air, into a spot on the outer island that is already air. `set_block`
425    /// answers `false` for an unchanged write, the same as vanilla's
426    /// `LevelChunk.setBlockState` returning null -- but vanilla's
427    /// `Feature.setBlock` returns `void` and never reads it. Foton read it, and
428    /// gave up on the second block of the box. Every gateway the dragon opened
429    /// was inert, every time, and nothing was logged.
430    #[test]
431    fn a_gateway_with_no_stored_exit_still_resolves_a_destination() {
432        init_vanilla_registry();
433        init_behaviors();
434        let world = fresh_end_test_world("gateway_destination_from_scratch");
435
436        // Ninety-six blocks out on the ring, which is where the fight puts them.
437        let gateway = BlockPos::new(96, 75, 0);
438        dragon_gateway(&world, gateway);
439        load_what_the_job_loads(&world, gateway);
440
441        let traveler = PigEntity::new(
442            &vanilla_entities::PIG,
443            next_entity_id(),
444            DVec3::new(96.5, 75.0, 0.5),
445            Arc::downgrade(&world),
446        );
447
448        let transition = calculate_transition(&world, &traveler, gateway, true).expect(
449            "a gateway the dragon opened resolved nowhere to send anyone, which is \
450             exactly what an inert portal looks like from inside the game",
451        );
452        assert!(
453            transition.position.x > 512.0,
454            "the outer islands are a thousand blocks out along the gateway's bearing, \
455             so a destination near the arena means the search never left home"
456        );
457
458        // Vanilla writes the exit down, so the next traveler takes the stored
459        // path instead of searching the whole way again.
460        let stored = world.get_block_entity(gateway).and_then(|entity| {
461            entity
462                .downcast_ref::<EndGatewayBlockEntity>()
463                .and_then(EndGatewayBlockEntity::exit_portal)
464        });
465        assert!(
466            stored.is_some(),
467            "the gateway found its exit and did not record it, so every later \
468             traveler pays for the whole search again"
469        );
470    }
