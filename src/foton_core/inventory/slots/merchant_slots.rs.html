1//! The trading screen's two payment slots and its result slot.
2//!
3//! Vanilla parity: `MerchantContainer` plus `MerchantResultSlot`. Vanilla makes
4//! one three-slot container and gives slot 2 special behavior; Foton splits it
5//! the way every other result-bearing menu here is split -- a plain container
6//! for the payment and a [`ResultContainer`] for what comes out -- so nothing
7//! can write into the result by clicking it.
8
9use std::sync::Arc;
10use std::sync::atomic::{AtomicI32, Ordering};

12use foton_registry::item_stack::ItemStack;
13use foton_registry::trading::MerchantOffer;
14use foton_utils::locks::Shared;
15
16use crate::inventory::container::{Container as _, ResultContainer, SimpleContainer};
17use crate::inventory::lock::{ContainerId, ContainerLockGuard, ContainerRef};
18use crate::inventory::slots::ResultHandler;
19use crate::player::Player;
20use crate::trading::Merchant;

22/// The selection hint a freshly opened trading screen carries.
23///
24/// Vanilla parity: the `0` a new `MerchantContainer.selectionHint` holds. It is
25/// not a sentinel: `getRecipeFor` only honors a hint above zero, so trade 0 is
26/// found by the scan rather than by the hint.
27pub const NO_TRADE_SELECTED: i32 = 0;
28
29/// Keeps the trading screen's result in step with what the player has paid.
30#[derive(Clone)]
31pub struct MerchantHandler {
32    payment_container: Shared<SimpleContainer>,
33    result_container: Shared<ResultContainer>,
34    merchant: Arc<dyn Merchant>,
35    /// The trade the player clicked, shared with the menu because the click and
36    /// the take arrive as separate packets.
37    selection_hint: Arc<AtomicI32>,
38}
39
40impl MerchantHandler {
41    /// Creates a handler over the trading screen's two containers.
42    #[must_use]
43    pub const fn new(
44        payment_container: Shared<SimpleContainer>,
45        result_container: Shared<ResultContainer>,
46        merchant: Arc<dyn Merchant>,
47        selection_hint: Arc<AtomicI32>,
48    ) -> Self {
49        Self {
50            payment_container,
51            result_container,
52            merchant,
53            selection_hint,
54        }
55    }
56
57    /// The merchant this screen trades with.
58    #[must_use]
59    pub const fn merchant(&self) -> &Arc<dyn Merchant> {
60        &self.merchant
61    }
62
63    /// The payment container's id.
64    #[must_use]
65    pub fn payment_id(&self) -> ContainerId {
66        ContainerId::from_arc(&self.payment_container)
67    }
68
69    /// The result container's id.
70    #[must_use]
71    pub fn result_id(&self) -> ContainerId {
72        ContainerId::from_arc(&self.result_container)
73    }
74
75    /// The two stacks the player has put up, in the order the trade reads them.
76    ///
77    /// Vanilla parity: the opening of `MerchantContainer.updateSellItem`, which
78    /// slides a lone second payment into first position so that paying into
79    /// either slot works.
80    fn payment(&self, guard: &ContainerLockGuard) -> (ItemStack, ItemStack) {
81        let Some(container) = guard.get(self.payment_id()) else {
82            return (ItemStack::empty(), ItemStack::empty());
83        };
84        let first = container.get_item(0).clone();
85        let second = container.get_item(1).clone();
86        if first.is_empty() {
87            (second, ItemStack::empty())
88        } else {
89            (first, second)
90        }
91    }
92
93    /// Finds the trade the current payment buys, as `(index, snapshot)`.
94    ///
95    /// Vanilla parity: the body of `MerchantContainer.updateSellItem`. It tries
96    /// the payment both ways round, so a two-cost trade can be paid into either
97    /// slot, and it refuses a trade that is out of stock.
98    ///
99    /// The index is what makes this usable from Rust: vanilla keeps a reference
100    /// into the merchant's live list, which an `Arc<dyn Merchant>` cannot hand
101    /// out, so the caller re-locks and indexes when it needs to change the trade.
102    fn active_trade(&self, guard: &ContainerLockGuard) -> Option<(usize, MerchantOffer)> {
103        let (buy_a, buy_b) = self.payment(guard);
104        if buy_a.is_empty() {
105            return None;
106        }
107
108        let hint = self.selection_hint.load(Ordering::Relaxed);
109        let offers = self.merchant.offers().lock();
110
111        let index = offers
112            .recipe_index_for(&buy_a, &buy_b, hint)
113            .or_else(|| offers.recipe_index_for(&buy_b, &buy_a, hint))
114            .filter(|&index| !offers[index].is_out_of_stock())?;
115
116        Some((index, offers[index].clone()))
117    }
118
119    /// The experience the merchant would gain from the trade now set up.
120    ///
121    /// Vanilla parity: `MerchantContainer.getFutureXp`.
122    #[must_use]
123    pub fn future_xp(&self, guard: &ContainerLockGuard) -> i32 {
124        self.active_trade(guard).map_or(0, |(_, offer)| offer.xp())
125    }
126}
127
128impl ResultHandler for MerchantHandler {
129    fn result_container(&self) -> ContainerRef {
130        ContainerRef::from(self.result_container.clone())
131    }
132
133    fn dependencies(&self) -> Vec<ContainerRef> {
134        vec![ContainerRef::from(self.payment_container.clone())]
135    }
136
137    fn update_result(&self, guard: &mut ContainerLockGuard) {
138        let result = self
139            .active_trade(guard)
140            .map_or_else(ItemStack::empty, |(_, offer)| offer.assemble());
141
142        let result_id = self.result_id();
143        if let Some(container) = guard.get_typed_mut::<ResultContainer>(result_id) {
144            container.set_item(0, result.clone());
145            container.set_changed();
146        }
147
148        // Vanilla parity: `updateSellItem` reaches `notifyTradeUpdated` only
149        // down the branch where a payment exists. Emptying the slots clears the
150        // result silently -- a villager grunts at what you offer it, not at you
151        // taking your things back.
152        let (buy_a, _) = self.payment(guard);
153        if !buy_a.is_empty() {
154            self.merchant.notify_trade_updated(&result);
155        }
156    }
157
158    /// Vanilla parity: `MerchantResultSlot.onTake`.
159    ///
160    /// The order is vanilla's and it matters: the payment is spent through the
161    /// offer itself, so demand and reputation decide how much is taken; the
162    /// merchant is told only if that succeeded; and `overrideXp` runs either
163    /// way, which on a real mob is the no-op vanilla makes it.
164    fn on_result_taken(
165        &self,
166        guard: &mut ContainerLockGuard,
167        _player: &Player,
168    ) -> Option<ItemStack> {
169        let (index, offer) = self.active_trade(guard)?;
170
171        let payment_id = self.payment_id();
172        let (mut buy_a, mut buy_b) = {
173            let container = guard.get(payment_id)?;
174            (container.get_item(0).clone(), container.get_item(1).clone())
175        };
176
177        if offer.take(&mut buy_a, &mut buy_b) || offer.take(&mut buy_b, &mut buy_a) {
178            self.merchant.notify_trade(index);
179            if let Some(container) = guard.get_mut(payment_id) {
180                container.set_item(0, buy_a);
181                container.set_item(1, buy_b);
182                container.set_changed();
183            }
184        }
185
186        self.merchant
187            .override_xp(self.merchant.villager_xp() + offer.xp());
188
189        self.update_result(guard);
190        None
191    }
192
193    fn is_result_valid(&self, guard: &ContainerLockGuard, _player: &Player) -> bool {
194        self.active_trade(guard).is_some()
195    }
196}
