1//! The grindstone's two inputs and its result.
2//!
3//! Vanilla parity: the anonymous slots of `GrindstoneMenu`. A grindstone does
4//! two unrelated things through one pair of slots -- it strips enchantments
5//! off a single item, and it welds two damaged copies of the same item into
6//! one -- and in both cases it keeps the curses, which is the whole reason a
7//! cursed item cannot be laundered.
8

22/// The first input slot.
23pub const GRINDSTONE_INPUT: usize = 0;
24/// The second input slot.
25pub const GRINDSTONE_ADDITIONAL: usize = 1;
26
27/// How much of an item's durability welding two of them gives back.
28///
29/// Vanilla parity: the `durability * 5 / 100` bonus of
30/// `GrindstoneMenu.mergeItems`.
31const WELD_BONUS_PERCENT: i32 = 5;
32
33/// Keeps a grindstone's result in step with its two inputs.
34#[derive(Clone)]
35pub struct GrindstoneHandler {
36    input_container: Shared<SimpleContainer>,
37    result_container: Shared<ResultContainer>,
38}
39
40impl GrindstoneHandler {
41    /// Creates a handler over the grindstone's containers.
42    #[must_use]
43    pub const fn new(
44        input_container: Shared<SimpleContainer>,
45        result_container: Shared<ResultContainer>,
46    ) -> Self {
47        Self {
48            input_container,
49            result_container,
50        }
51    }
52
53    fn input_id(&self) -> ContainerId {
54        ContainerId::from_arc(&self.input_container)
55    }
56
57    fn result_id(&self) -> ContainerId {
58        ContainerId::from_arc(&self.result_container)
59    }
60
61    /// Returns both inputs.
62    #[must_use]
63    pub fn input_snapshot(&self, guard: &ContainerLockGuard) -> Option<(ItemStack, ItemStack)> {
64        let container = guard.get(self.input_id())?;
65        Some((
66            container.get_item(GRINDSTONE_INPUT).clone(),
67            container.get_item(GRINDSTONE_ADDITIONAL).clone(),
68        ))
69    }
70
71    /// The currently previewed result, empty when the inputs make nothing.
72    pub fn result_snapshot(&self, guard: &ContainerLockGuard) -> ItemStack {
73        guard
74            .get(self.result_id())
75            .map_or_else(ItemStack::empty, |c| c.get_item(0).clone())
76    }
77
78    /// Writes a plugin's edited inputs and result back under the existing menu
79    /// lock.
80    ///
81    /// Takes the guard the caller already holds, so the whole preview-edit-write
82    /// cycle happens without the containers being unlocked in between.
83    pub fn apply_snapshot(
84        &self,
85        guard: &mut ContainerLockGuard,
86        upper: ItemStack,
87        lower: ItemStack,
88        result: ItemStack,
89    ) {
90        if let Some(container) = guard.get_mut(self.input_id()) {
91            container.set_item(GRINDSTONE_INPUT, upper);
92            container.set_item(GRINDSTONE_ADDITIONAL, lower);
93            container.set_changed();
94        }
95        if let Some(container) = guard.get_typed_mut::<ResultContainer>(self.result_id()) {
96            container.set_item(0, result);
97            container.set_changed();
98        }
99    }
100
101    /// Returns the experience the two inputs are worth.
102    ///
103    /// Vanilla parity: `getExperienceAmount`. Half the enchantments' minimum
104    /// cost, rounded up, plus a random amount up to that again -- so grinding
105    /// the same item twice does not give the same experience.
106    #[must_use]
107    pub fn experience(&self, guard: &ContainerLockGuard) -> i32 {
108        let Some((first, second)) = self.input_snapshot(guard) else {
109            return 0;
110        };
111        let total = experience_from(&first) + experience_from(&second);
112        if total <= 0 {
113            return 0;
114        }
115        // `div_ceil` is unstable for i32 here, so the round-up is written out.
116        let half = (total + 1) / 2;
117        half + (rand::random::<u32>() % half.max(1) as u32) as i32
118    }
119
120    /// Returns whether there is nothing to take.
121    #[must_use]
122    pub fn result_is_empty(&self, guard: &ContainerLockGuard) -> bool {
123        let result_id = self.result_id();
124        guard
125            .get(result_id)
126            .is_none_or(|container| container.get_item(0).is_empty())
127    }
128
129    /// Empties both inputs, which is what taking the result costs.
130    pub fn clear_inputs(&self, guard: &mut ContainerLockGuard) {
131        let input_id = self.input_id();
132        let Some(container) = guard.get_mut(input_id) else {
133            return;
134        };
135        container.set_item(GRINDSTONE_INPUT, ItemStack::empty());
136        container.set_item(GRINDSTONE_ADDITIONAL, ItemStack::empty());
137        container.set_changed();
138    }
139}

141/// Returns the experience one item's non-curse enchantments are worth.
142fn experience_from(item: &ItemStack) -> i32 {
143    let Some(enchantments) = item.get_enchantments_for_crafting() else {
144        return 0;
145    };
146    enchantments
147        .iter()
148        .filter(|(id, _)| !is_curse(id))
149        .filter_map(|(id, level)| {
150            let enchantment = REGISTRY.enchantments.by_key(id)?;
151            let level = *level as i32;
152            // Vanilla parity: `Enchantment.getMinCost(level)`.
153            Some(
154                enchantment.min_cost.base
155                    + (level - 1) * enchantment.min_cost.per_level_above_first,
156            )
157        })
158        .sum()
159}

161/// Returns whether an enchantment is a curse, and so survives the grindstone.
162fn is_curse(id: &Identifier) -> bool {
163    REGISTRY.enchantments.by_key(id).is_some_and(|enchantment| {
164        REGISTRY
165            .enchantments
166            .is_in_tag(enchantment, &EnchantmentTag::CURSE)
167    })
168}

170/// Strips everything but the curses, and prices what is left.
171///
172/// Vanilla parity: `GrindstoneMenu.removeNonCursesFrom`. An enchanted book
173/// with nothing cursed left on it becomes a plain book, because an enchanted
174/// book with no enchantments is not a thing.
175fn remove_non_curses(mut item: ItemStack) -> ItemStack {
176    let kept: Vec<(Identifier, u32)> = item
177        .get_enchantments_for_crafting()
178        .map(|enchantments| {
179            enchantments
180                .iter()
181                .filter(|(id, _)| is_curse(id))
182                .map(|(id, level)| (id.clone(), *level))
183                .collect()
184        })
185        .unwrap_or_default();
186
187    item.replace_enchantments(&kept);
188
189    if item.is(&vanilla_items::ENCHANTED_BOOK) && kept.is_empty() {
190        let mut book = ItemStack::with_count(&vanilla_items::BOOK, item.count());
191        book.set(REPAIR_COST, 0);
192        return book;
193    }
194
195    // Vanilla parity: one doubling of the repair cost per surviving
196    // enchantment, so a cursed item stays expensive on an anvil afterwards.
197    let mut repair_cost = 0i32;
198    for _ in 0..kept.len() {
199        repair_cost = repair_cost.saturating_mul(2).saturating_add(1);
200    }
201    item.set(REPAIR_COST, repair_cost);
202    item
203}

205/// Welds two of the same item into one.
206///
207/// Vanilla parity: `GrindstoneMenu.mergeItems`.
208fn merge_items(first: &ItemStack, second: &ItemStack) -> ItemStack {
209    if !ItemStack::is_same_item(first, second) {
210        return ItemStack::empty();
211    }
212
213    let durability = first.get_max_damage().max(second.get_max_damage());
214    let remaining = (first.get_max_damage() - first.get_damage_value())
215        + (second.get_max_damage() - second.get_damage_value())
216        + durability * WELD_BONUS_PERCENT / 100;
217
218    let mut count = 1;
219    if !first.is_damageable_item() {
220        // Two undamageable items only stack; anything that cannot stack to two
221        // has nothing to weld.
222        if first.max_stack_size() < 2 || !ItemStack::is_same_item_same_components(first, second) {
223            return ItemStack::empty();
224        }
225        count = 2;
226    }
227
228    let mut welded = first.copy_with_count(count);
229    if welded.is_damageable_item() {
230        welded.set(MAX_DAMAGE, durability);
231        welded.set_damage_value((durability - remaining).max(0));
232    }
233
234    merge_enchantments_from(&mut welded, second);
235    remove_non_curses(welded)
236}

238/// Carries `source`'s enchantments onto `target` before the strip.
239///
240/// Vanilla parity: `mergeEnchantsFrom`. Curses are only carried when the
241/// target does not already have them, which stops two cursed items stacking
242/// the same curse to a higher level.
243fn merge_enchantments_from(target: &mut ItemStack, source: &ItemStack) {
244    let Some(source_enchantments) = source.get_enchantments_for_crafting() else {
245        return;
246    };
247    let mut merged = target
248        .get_enchantments_for_crafting()
249        .cloned()
250        .unwrap_or_else(ItemEnchantments::empty);
251
252    for (id, level) in source_enchantments.iter() {
253        if !is_curse(id) || merged.get_level(id) == 0 {
254            merged.upgrade(id.clone(), *level);
255        }
256    }
257
258    let levels: Vec<(Identifier, u32)> = merged
259        .iter()
260        .map(|(id, level)| (id.clone(), *level))
261        .collect();
262    target.replace_enchantments(&levels);
263}

265/// Returns what the grindstone would make of these two inputs.
266///
267/// Vanilla parity: `GrindstoneMenu.computeResult`.
268#[must_use]
269pub fn grindstone_result(first: &ItemStack, second: &ItemStack) -> ItemStack {
270    if first.is_empty() && second.is_empty() {
271        return ItemStack::empty();
272    }
273    // A grindstone works on single items; a stack in either slot does nothing.
274    if first.count() > 1 || second.count() > 1 {
275        return ItemStack::empty();
276    }
277
278    if first.is_empty() || second.is_empty() {
279        let lone = if first.is_empty() { second } else { first };
280        let has_enchantments = lone
281            .get_enchantments_for_crafting()
282            .is_some_and(|enchantments| !enchantments.is_empty());
283        if !has_enchantments {
284            return ItemStack::empty();
285        }
286        return remove_non_curses(lone.clone());
287    }
288
289    merge_items(first, second)
290}

292impl ResultHandler for GrindstoneHandler {
293    fn result_container(&self) -> ContainerRef {
294        ContainerRef::from(self.result_container.clone())
295    }
296
297    fn dependencies(&self) -> Vec<ContainerRef> {
298        vec![ContainerRef::from(self.input_container.clone())]
299    }
300
301    fn update_result(&self, guard: &mut ContainerLockGuard) {
302        let result = self
303            .input_snapshot(guard)
304            .map_or_else(ItemStack::empty, |(first, second)| {
305                grindstone_result(&first, &second)
306            });
307
308        let result_id = self.result_id();
309        let Some(container) = guard.get_typed_mut::<ResultContainer>(result_id) else {
310            return;
311        };
312        container.set_item(0, result);
313        container.set_changed();
314    }
315
316    /// Vanilla parity: the `onTake` of the result slot, which empties both
317    /// inputs -- a grindstone consumes what it grinds whole.
318    fn on_result_taken(
319        &self,
320        guard: &mut ContainerLockGuard,
321        _player: &Player,
322    ) -> Option<ItemStack> {
323        self.clear_inputs(guard);
324        self.update_result(guard);
325        None
326    }
327
328    fn is_result_valid(&self, guard: &ContainerLockGuard, _player: &Player) -> bool {
329        self.input_snapshot(guard)
330            .is_some_and(|(first, second)| !grindstone_result(&first, &second).is_empty())
331    }
332}

334/// A grindstone's inputs only take things it can work on.
335///
336/// Vanilla parity: the `mayPlace` of both input slots -- a slot that refuses
337/// what it cannot grind is what stops a player parking a stack of cobblestone
338/// in one.
339#[must_use]
340pub fn grindstone_accepts(item: &ItemStack) -> bool {
341    item.is_damageable_item()
342        || item
343            .get_enchantments_for_crafting()
344            .is_some_and(|enchantments| !enchantments.is_empty())
345}

359    /// Builds an enchanted item.
360    fn enchanted(item: &'static Item, enchantment: &'static str, level: u32) -> ItemStack {
361        let mut stack = ItemStack::new(item);
362        stack.set_enchantments(&[(Identifier::vanilla_static(enchantment), level)], false);
363        stack
364    }
