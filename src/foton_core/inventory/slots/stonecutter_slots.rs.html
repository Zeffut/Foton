1//! The stonecutter's input and result slots.
2//!
3//! Vanilla parity: the anonymous result slot of `StonecutterMenu`. What makes
4//! it different from a crafting result is that the recipe is not deduced from
5//! the input -- one block usually cuts into a dozen things, so the player picks
6//! which, and the pick has to survive between the button press and the take.
7
8use std::sync::Arc;
9use std::sync::atomic::{AtomicI32, Ordering};
10
11use foton_registry::REGISTRY;
12use foton_registry::item_stack::ItemStack;
13use foton_utils::locks::Shared;
14
15use crate::inventory::container::{Container as _, ResultContainer, SimpleContainer};
16use crate::inventory::lock::{ContainerId, ContainerLockGuard, ContainerRef};
17use crate::inventory::slots::ResultHandler;
18use crate::player::Player;
19
20/// The value the selection holds when the player has not chosen.
21///
22/// Vanilla parity: the `-1` `StonecutterMenu.setupRecipeList` resets to.
23pub const NO_SELECTION: i32 = -1;
24
25/// Keeps a stonecutter's result in step with its input and the chosen recipe.
26#[derive(Clone)]
27pub struct StonecutterHandler {
28    input_container: Shared<SimpleContainer>,
29    result_container: Shared<ResultContainer>,
30    /// Which of the input's recipes the player picked.
31    ///
32    /// Shared with the menu rather than owned by it: the button press that
33    /// changes it and the take that reads it arrive as separate packets.
34    selected: Arc<AtomicI32>,
35}
36
37impl StonecutterHandler {
38    /// Creates a handler over the stonecutter's two containers.
39    #[must_use]
40    pub const fn new(
41        input_container: Shared<SimpleContainer>,
42        result_container: Shared<ResultContainer>,
43        selected: Arc<AtomicI32>,
44    ) -> Self {
45        Self {
46            input_container,
47            result_container,
48            selected,
49        }
50    }
51
52    /// Returns how many recipes the current input offers.
53    #[must_use]
54    pub fn recipe_count(&self, guard: &ContainerLockGuard) -> usize {
55        let Some(input) = self.input(guard) else {
56            return 0;
57        };
58        if input.is_empty() {
59            return 0;
60        }
61        REGISTRY.recipes.stonecutting_recipes_for(&input).len()
62    }
63
64    /// Returns a copy of the input stack, for the menu to compare against.
65    #[must_use]
66    pub fn input_snapshot(&self, guard: &ContainerLockGuard) -> Option<ItemStack> {
67        self.input(guard)
68    }
69
70    /// Returns a copy of the input stack.
71    fn input(&self, guard: &ContainerLockGuard) -> Option<ItemStack> {
72        let container = guard.get(self.input_id())?;
73        Some(container.get_item(0).clone())
74    }
75
76    /// The input container's id, for tests that read it directly.
77    #[cfg(test)]
78    pub(crate) fn input_id_for_tests(&self) -> ContainerId {
79        self.input_id()
80    }
81
82    /// The result container's id, for tests that read it directly.
83    #[cfg(test)]
84    pub(crate) fn result_id_for_tests(&self) -> ContainerId {
85        self.result_id()
86    }
87
88    fn input_id(&self) -> ContainerId {
89        ContainerId::from_arc(&self.input_container)
90    }
91
92    fn result_id(&self) -> ContainerId {
93        ContainerId::from_arc(&self.result_container)
94    }
95
96    /// Returns what the chosen recipe makes from the current input.
97    fn chosen_result(&self, guard: &ContainerLockGuard) -> ItemStack {
98        let Some(input) = self.input(guard) else {
99            return ItemStack::empty();
100        };
101        if input.is_empty() {
102            return ItemStack::empty();
103        }
104
105        let index = self.selected.load(Ordering::Relaxed);
106        if index < 0 {
107            return ItemStack::empty();
108        }
109
110        let recipes = REGISTRY.recipes.stonecutting_recipes_for(&input);
111        recipes
112            .get(index as usize)
113            .map_or_else(ItemStack::empty, |recipe| recipe.result.to_item_stack())
114    }
115}
116
117impl ResultHandler for StonecutterHandler {
118    fn result_container(&self) -> ContainerRef {
119        ContainerRef::from(self.result_container.clone())
120    }
121
122    fn dependencies(&self) -> Vec<ContainerRef> {
123        vec![ContainerRef::from(self.input_container.clone())]
124    }
125
126    fn update_result(&self, guard: &mut ContainerLockGuard) {
127        let result = self.chosen_result(guard);
128        let result_id = self.result_id();
129        let Some(container) = guard.get_typed_mut::<ResultContainer>(result_id) else {
130            return;
131        };
132        container.set_item(0, result);
133        container.set_changed();
134    }
135
136    /// Vanilla parity: the `onTake` of the result slot -- one input is spent
137    /// per take, and the result is rebuilt so the player can keep taking.
138    fn on_result_taken(
139        &self,
140        guard: &mut ContainerLockGuard,
141        _player: &Player,
142    ) -> Option<ItemStack> {
143        let input_id = self.input_id();
144        if let Some(container) = guard.get_mut(input_id) {
145            container.get_item_mut(0).shrink(1);
146            container.set_changed();
147        }
148        self.update_result(guard);
149        None
150    }
151
152    /// Vanilla parity: a stonecutter result is only real while the input still
153    /// matches the chosen recipe, which stops a stale result being taken after
154    /// the input was swapped underneath it.
155    fn is_result_valid(&self, guard: &ContainerLockGuard, _player: &Player) -> bool {
156        !self.chosen_result(guard).is_empty()
157    }
158}
