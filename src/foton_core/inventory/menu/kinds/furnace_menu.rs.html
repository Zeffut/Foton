19/// Builds a furnace-style menu for `menu_type`.
20///
21/// Vanilla parity: `AbstractFurnaceMenu`. The same layout backs the furnace, the
22/// smoker and the blast furnace; only the menu type and the recipe family differ.
23#[must_use]
24pub fn furnace(
25    inventory: Shared<PlayerInventory>,
26    container_id: u8,
27    container: impl Into<ContainerRef>,
28    menu_type: MenuTypeRef,
29    data: Arc<FurnaceDataSlots>,
30    block_entity: SharedBlockEntity,
31) -> Menu {
32    let container = container.into();
33    let mut builder = MenuBuilder::new(menu_type, container_id);
34
35    let input = builder.section(&container, 1);
36    // Vanilla also refuses a second empty bucket while one already sits here; the
37    // rule cannot see the slot's current contents from this predicate, so that
38    // corner case is left to Container::can_place_item.
39    let fuel = builder.section_with(
40        &container,
41        1,
42        SectionKind::restricted(|_slot, stack| {
43            fuel::is_fuel(stack) || stack.is(&vanilla_items::BUCKET)
44        }),
45    );
46    let result = builder.section_with(
47        &container,
48        1,
49        SectionKind::custom(move |container, index| {
50            Box::new(FurnaceResultSlot::new(
51                container.clone(),
52                index,
53                Arc::clone(&block_entity),
54            ))
55        }),
56    );
57    let player = builder.player_inventory(&inventory);
58
59    let data_slots = [
60        builder.data_slot(0),
61        builder.data_slot(0),
62        builder.data_slot(0),
63        builder.data_slot(0),
64    ];
65
66    builder.route([input, fuel, result], player.all(), FillDirection::Backward);
67    // TODO: mirror AbstractFurnaceMenu.quickMoveStack, which sends a smeltable item
68    // to the input slot and a fuel item to the fuel slot instead of filling in order.
69    builder.route(player.all(), [input, fuel], FillDirection::Forward);
70
71    builder.build(FurnaceKind {
72        container,
73        data,
74        data_slots,
75    })
76}

78/// Per-menu furnace state.
79pub struct FurnaceKind {
80    /// The backing container.
81    container: ContainerRef,
82    /// Progress published by the block entity each tick.
83    data: Arc<FurnaceDataSlots>,
84    /// Handles to the four synced values.
85    data_slots: [DataSlot; 4],
86}
87
88// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
89// within the process.
90unsafe impl foton_utils::DowncastType for FurnaceKind {
91    const TYPE_KEY: foton_utils::DowncastTypeKey =
92        foton_utils::DowncastTypeKey::new("foton:menu/furnace");
93}
94
95impl MenuKind for FurnaceKind {
96    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
97        self.container.still_valid(player)
98    }
99
100    /// Pushes the block entity's progress into the synced data slots.
101    ///
102    /// Vanilla hands the menu the block entity's own `ContainerData`, so the two
103    /// are the same object. Foton keeps the cooking state behind the block
104    /// entity's lock and republishes it here, which is why the menu polls.
105    fn on_tick(
106        &mut self,
107        behavior: &mut MenuBehavior,
108        _guard: &mut ContainerLockGuard,
109        _player: &Player,
110    ) {
111        let values = self.data.snapshot();
112        for (slot, value) in self.data_slots.iter().zip(values) {
113            slot.set(behavior, value);
114        }
115    }
116}
