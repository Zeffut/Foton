1//! The mount inventory screen.
2//!
3//! Vanilla parity: `AbstractMountInventoryMenu` and the two subclasses that
4//! only differ in their slot sprites and in which mount they cast to --
5//! `HorseInventoryMenu` and `NautilusInventoryMenu`. Foton builds one menu for
6//! both and takes the cast as a parameter.
7//!
8//! Slot layout:
9//! - Slot 0: the saddle, an equipment slot of the mount itself
10//! - Slot 1: the body armor, likewise
11//! - Slots 2..2+`columns`*3: the cargo grid, empty for a mount with no chest
12//! - the player's 36 slots after that
13//!
14//! Two things make this screen unlike every other menu. It is opened by
15//! [`CMountScreenOpen`] rather than by the open-screen packet, because the
16//! client builds the menu from the entity it already tracks; and its first two
17//! slots are not storage of their own but the mount's worn equipment, which is
18//! why they sit on [`LivingEntityBase::equipment_slot_container`].
19

29/// Reads the inventory a mount owns right now.
30///
31/// Vanilla parity: the `hasInventoryChanged` override each mount menu carries.
32/// It is nothing but a cast to the concrete mount, so Foton passes the cast in
33/// rather than splitting the menu in two for it. A mount that grew or lost its
34/// chest has replaced its inventory, and the screen has to close.
35pub type MountInventoryProbe = fn(&dyn Entity) -> Option<ContainerId>;
36
37/// Reads the current inventory of a horse-shaped mount.
38///
39/// Vanilla parity: `HorseInventoryMenu.hasInventoryChanged`.
40#[must_use]
41pub fn horse_inventory(entity: &dyn Entity) -> Option<ContainerId> {
42    let horse = entity.as_abstract_horse()?;
43    Some(ContainerId::from_arc(
44        &horse.abstract_horse_base().inventory(),
45    ))
46}

59/// Opens `mount`'s own inventory screen for `player`.
60///
61/// Vanilla parity: `ServerPlayer.openHorseInventory` and
62/// `ServerPlayer.openNautilusInventory`, which are `openMenu` with the mount
63/// screen packet in place of the open-screen one. The tame and rider gates
64/// belong to `openCustomInventoryScreen` and have already run by here.
65pub fn open_mount_screen(
66    mount: &SharedEntity,
67    inventory: Shared<SimpleContainer>,
68    inventory_columns: usize,
69    current_inventory: MountInventoryProbe,
70    player: &Player,
71) {
72    let Some(living) = mount.as_living_entity() else {
73        return;
74    };
75    let (equipment, saddle_index) = living
76        .living_base()
77        .equipment_slot_container(EquipmentSlot::Saddle);
78    let body_index = living
79        .living_base()
80        .equipment_slot_container(EquipmentSlot::Body)
81        .1;
82    let entity_id = Entity::id(mount.as_ref());
83    let mount = Arc::clone(mount);
84
85    // The mount screen carries no title: the client names it after the entity.
86    player.open_menu("", move |context| {
87        mount_menu(MountMenuParts {
88            player_inventory: context.player.inventory.clone(),
89            container_id: context.container_id,
90            mount,
91            entity_id,
92            equipment: ContainerRef::from(equipment),
93            saddle_index,
94            body_index,
95            inventory,
96            inventory_columns,
97            current_inventory,
98        })
99    });
100}

102/// Everything [`mount_menu`] needs, resolved before the menu is built.
103struct MountMenuParts {
104    player_inventory: Shared<PlayerInventory>,
105    container_id: u8,
106    mount: SharedEntity,
107    entity_id: i32,
108    /// The mount's own equipment, which backs the saddle and armor slots.
109    equipment: ContainerRef,
110    saddle_index: usize,
111    body_index: usize,
112    inventory: Shared<SimpleContainer>,
113    inventory_columns: usize,
114    current_inventory: MountInventoryProbe,
115}
116
117/// Builds the mount screen.
118fn mount_menu(parts: MountMenuParts) -> Menu {
119    let MountMenuParts {
120        player_inventory,
121        container_id,
122        mount,
123        entity_id,
124        equipment,
125        saddle_index,
126        body_index,
127        inventory,
128        inventory_columns,
129        current_inventory,
130    } = parts;
131
132    let weak_mount = Arc::downgrade(&mount);
133    let inventory = ContainerRef::from(inventory);
134    let mut builder = MenuBuilder::new(None, container_id);
135    builder.mount_screen(inventory_columns, entity_id);
136
137    let saddle = builder.section_at(
138        &equipment,
139        [saddle_index],
140        equipment_slots(&weak_mount, EquipmentSlot::Saddle),
141    );
142    let body = builder.section_at(
143        &equipment,
144        [body_index],
145        equipment_slots(&weak_mount, EquipmentSlot::Body),
146    );
147    let cargo = builder.section_all(&inventory);
148    let player = builder.player_inventory(&player_inventory);
149
150    builder.build(MountKind {
151        mount: weak_mount,
152        inventory,
153        current_inventory,
154        saddle,
155        body,
156        cargo,
157        main: player.main(),
158        hotbar: player.hotbar(),
159        player: player.all(),
160    })
161}

163/// A section kind that builds [`MountEquipmentSlot`]s for `slot`.
164fn equipment_slots(mount: &WeakEntity, slot: EquipmentSlot) -> SectionKind {
165    let mount = mount.clone();
166    SectionKind::custom(move |container, index| {
167        Box::new(MountEquipmentSlot {
168            base: ArmorSlot::new(container.clone(), index, slot, mount.clone()),
169            mount: mount.clone(),
170        })
171    })
172}

174/// The saddle or body-armor slot of a mount screen.
175///
176/// Vanilla parity: the `ArmorSlot` that `HorseInventoryMenu` builds over
177/// `Mob.createEquipmentSlotContainer`. It differs from the player's own
178/// [`ArmorSlot`] by having an owner: what may be placed depends on the mount,
179/// and equipping something has to reach it.
180///
181/// Vanilla also switches these two slots off through `Slot.isActive` -- an
182/// unsaddleable mount shows the saddle slot greyed out. That is a client-side
183/// decision: nothing on the server reads `isActive`, and the client rebuilds
184/// this menu itself, so Foton has nothing to send for it.
185struct MountEquipmentSlot {
186    base: ArmorSlot,
187    mount: WeakEntity,
188}
189
190// SAFETY: This Foton-owned key uniquely identifies `MountEquipmentSlot`.
191unsafe impl DowncastType for MountEquipmentSlot {
192    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:slot/mount_equipment");
193}
194
195impl Slot for MountEquipmentSlot {
196    fn storage(&self) -> &SlotStorage {
197        self.base.storage()
198    }
199
200    fn get_item<'a>(&self, guard: &'a ContainerLockGuard) -> &'a ItemStack {
201        self.base.get_item(guard)
202    }
203
204    fn get_item_mut<'a>(&self, guard: &'a mut ContainerLockGuard) -> &'a mut ItemStack {
205        self.base.get_item_mut(guard)
206    }
207
208    fn set_item(&self, guard: &mut ContainerLockGuard, stack: ItemStack) {
209        self.base.set_item(guard, stack);
210    }
211
212    /// Vanilla parity: `ArmorSlot.setByPlayer` followed by the `setTheItem` of
213    /// `Mob.createEquipmentSlotContainer`, which is what the write reaches
214    /// through.
215    ///
216    /// The mount's own locks are taken with the containers released: the item
217    /// being written lives in the mount's equipment, so playing the equip
218    /// sound under the menu's lock would take that same mutex twice.
219    fn set_by_player(
220        &self,
221        guard: &mut ContainerLockGuard,
222        stack: ItemStack,
223        previous: &ItemStack,
224    ) {
225        let slot = self.base.equipment_slot();
226        if let Some(mount) = self.mount.upgrade() {
227            let equipped = stack.clone();
228            guard.run_unlocked(|| {
229                if let Some(living) = mount.as_living_entity() {
230                    living.on_equip_item(slot, previous, &equipped);
231                }
232                if !equipped.is_empty()
233                    && let Some(mob) = mount.as_mob()
234                {
235                    mob.set_guaranteed_drop(slot);
236                    mob.set_persistence_required();
237                }
238            });
239        }
240        self.base.set_item(guard, stack);
241    }
242
243    /// Vanilla parity: `ArmorSlot.mayPlace`, whose `isEquippableInSlot` is the
244    /// mount's -- horse armor does not fit a llama and llama armor does not fit
245    /// a horse.
246    fn may_place(&self, stack: &ItemStack) -> bool {
247        let Some(mount) = self.mount.upgrade() else {
248            return false;
249        };
250        let Some(living) = mount.as_living_entity() else {
251            return false;
252        };
253        living.is_equippable_in_slot(stack, self.base.equipment_slot())
254    }
255
256    fn may_pickup(&self, guard: &ContainerLockGuard, player: &Player) -> bool {
257        self.base.may_pickup(guard, player)
258    }
259
260    fn get_max_stack_size(&self, guard: &ContainerLockGuard) -> i32 {
261        self.base.get_max_stack_size(guard)
262    }
263
264    fn set_changed(&self, guard: &mut ContainerLockGuard) {
265        self.base.set_changed(guard);
266    }
267
268    fn get_container_slot(&self) -> usize {
269        self.base.get_container_slot()
270    }
271}

273/// Per-menu mount state: who the screen belongs to and which inventory it was
274/// built over.
275pub struct MountKind {
276    mount: WeakEntity,
277    /// The mount inventory this menu was built over.
278    inventory: ContainerRef,
279    current_inventory: MountInventoryProbe,
280    saddle: Section,
281    body: Section,
282    cargo: Section,
283    main: Section,
284    hotbar: Section,
285    player: Section,
286}
287
288// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
289// within the process.
290unsafe impl DowncastType for MountKind {
291    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:menu/mount");
292}
293
294impl MenuKind for MountKind {
295    /// Vanilla parity: `AbstractMountInventoryMenu.stillValid`.
296    ///
297    /// The inventory check is the interesting one: strapping a chest onto a
298    /// donkey replaces its inventory with a wider one, and the screen showing
299    /// the old one has to close rather than keep writing into it.
300    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
301        let Some(mount) = self.mount.upgrade() else {
302            return false;
303        };
304        (self.current_inventory)(mount.as_ref()) == Some(self.inventory.container_id())
305            && self.inventory.still_valid(player)
306            && Entity::is_alive(mount.as_ref())
307            && player.is_within_entity_interaction_range(mount.bounding_box(), 4.0)
308    }
309
310    /// Vanilla parity: `AbstractMountInventoryMenu.quickMoveStack`.
311    ///
312    /// The order is the mount's own: body armor first, then the saddle, then
313    /// the cargo grid, so that shift-clicking a saddle onto an already-armored
314    /// horse still saddles it.
315    fn quick_move(
316        &mut self,
317        behavior: &mut MenuBehavior,
318        guard: &mut ContainerLockGuard,
319        slot_index: usize,
320        player: &Player,
321    ) -> Option<ItemStack> {
322        let clicked = behavior.slots()[slot_index].get_item(guard).clone();
323        if clicked.is_empty() {
324            return Some(ItemStack::empty());
325        }
326
327        let mut stack = clicked.clone();
328        let player_start = self.player.start();
329        let slot_count = behavior.slots().len();
330        let body = self.body.start();
331        let saddle = self.saddle.start();
332
333        let moved = if slot_index < player_start {
334            behavior.move_item_stack_to(
335                guard,
336                slot_index,
337                &mut stack,
338                player_start,
339                slot_count,
340                FillDirection::Backward,
341            )
342        } else if accepts(behavior, guard, body, &stack) {
343            behavior.move_item_stack_to(
344                guard,
345                slot_index,
346                &mut stack,
347                body,
348                body + 1,
349                FillDirection::Forward,
350            )
351        } else if accepts(behavior, guard, saddle, &stack) {
352            behavior.move_item_stack_to(
353                guard,
354                slot_index,
355                &mut stack,
356                saddle,
357                saddle + 1,
358                FillDirection::Forward,
359            )
360        } else if self.cargo.is_empty()
361            || !behavior.move_item_stack_to(
362                guard,
363                slot_index,
364                &mut stack,
365                self.cargo.start(),
366                self.cargo.end(),
367                FillDirection::Forward,
368            )
369        {
370            // Nothing on the mount wanted it, so it only moves between the
371            // player's own two halves. Vanilla returns empty from here without
372            // marking the source slot; Foton's move works on a copy, so the
373            // copy still has to be written back or the items would double.
374            let shuffled = self.shuffle_player_halves(behavior, guard, slot_index, &mut stack);
375            if shuffled {
376                behavior.update_quick_move_source(guard, slot_index, &stack, &clicked);
377            }
378            return Some(ItemStack::empty());
379        } else {
380            true
381        };
382
383        if !moved {
384            return Some(ItemStack::empty());
385        }
386
387        behavior.update_quick_move_source(guard, slot_index, &stack, &clicked);
388        if stack.count == clicked.count {
389            return Some(ItemStack::empty());
390        }
391        if let Some(remainder) = behavior.slots()[slot_index].on_take(guard, &stack, player) {
392            player.add_item_or_drop_with_guard(guard, remainder);
393        }
394        Some(clicked)
395    }
396}

398/// Whether an empty equipment slot would take `stack`.
399///
400/// Vanilla parity: the `getSlot(n).mayPlace(stack) && !getSlot(n).hasItem()`
401/// pair the mount menu tests before each armor slot.
402fn accepts(
403    behavior: &MenuBehavior,
404    guard: &ContainerLockGuard,
405    slot_index: usize,
406    stack: &ItemStack,
407) -> bool {
408    let slot = &behavior.slots()[slot_index];
409    slot.may_place(stack) && !slot.has_item(guard)
410}

412impl MountKind {
413    /// Moves a stack between the player's main inventory and hotbar.
414    ///
415    /// Vanilla parity: the tail of `AbstractMountInventoryMenu.quickMoveStack`.
416    fn shuffle_player_halves(
417        &self,
418        behavior: &MenuBehavior,
419        guard: &mut ContainerLockGuard,
420        slot_index: usize,
421        stack: &mut ItemStack,
422    ) -> bool {
423        let (start, end) = if self.hotbar.contains(slot_index) {
424            (self.main.start(), self.main.end())
425        } else if self.main.contains(slot_index) {
426            (self.hotbar.start(), self.hotbar.end())
427        } else {
428            (self.hotbar.start(), self.main.end())
429        };
430        behavior.move_item_stack_to(guard, slot_index, stack, start, end, FillDirection::Forward)
431    }
432}
