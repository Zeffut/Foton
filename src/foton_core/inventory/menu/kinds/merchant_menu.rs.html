1//! The villager trading screen.
2//!
3//! Vanilla parity: `MerchantMenu`. Three slots and a list of trades: two the
4//! player pays into, one the trade comes out of, and a selection the player
5//! makes by clicking a trade rather than by filling a slot.

20/// Builds the trading menu for `merchant`.
21#[must_use]
22pub fn merchant_menu(
23    inventory: Shared<PlayerInventory>,
24    container_id: u8,
25    merchant: Arc<dyn Merchant>,
26) -> Menu {
27    // Vanilla's MerchantContainer is one three-slot container. The payment half
28    // is the only half a click may write into, so it is the only half that is a
29    // plain container here.
30    let payment_container = SimpleContainer::new(2).into_shared();
31    let result_container = ResultContainer::new().into_shared();
32    let selection_hint = Arc::new(AtomicI32::new(NO_TRADE_SELECTED));
33
34    let handler = MerchantHandler::new(
35        payment_container.clone(),
36        result_container.clone(),
37        Arc::clone(&merchant),
38        Arc::clone(&selection_hint),
39    );
40
41    let mut builder = MenuBuilder::new(&vanilla_menu_types::MERCHANT, container_id);
42    let payment = builder.section_all(&payment_container);
43    let result = builder.result_slot(handler.clone());
44    let player = builder.player_inventory(&inventory);
45
46    // Vanilla parity: `MerchantMenu.quickMoveStack`. Taking the result fills the
47    // inventory back to front, paying fills it front to back.
48    builder.route(result, player.all(), FillDirection::Backward);
49    builder.route(payment, player.all(), FillDirection::Forward);
50    builder.route(player.all(), payment, FillDirection::Forward);
51    // Vanilla parity: the half of `MerchantMenu.removed` that hands the payment
52    // back to the player rather than eating it.
53    builder.drain(payment);
54
55    builder.build(MerchantKind {
56        handler,
57        merchant,
58        selection_hint,
59    })
60}

62/// Per-menu trading state.
63pub struct MerchantKind {
64    handler: MerchantHandler,
65    merchant: Arc<dyn Merchant>,
66    /// The trade the player clicked, shared with the handler.
67    selection_hint: Arc<AtomicI32>,
68}
69
70impl MerchantKind {
71    /// The container holding the two payment slots.
72    #[cfg(test)]
73    pub(crate) fn payment_id_for_tests(&self) -> ContainerId {
74        self.handler.payment_id()
75    }
76
77    /// The container holding what the trade produces.
78    #[cfg(test)]
79    pub(crate) fn result_id_for_tests(&self) -> ContainerId {
80        self.handler.result_id()
81    }
82}

91impl MenuKind for MerchantKind {
92    /// Vanilla parity: `Merchant.openTradingScreen`, which sends the offers
93    /// right after the menu packet. An empty list is not sent at all, which is
94    /// what leaves an unemployed villager's screen blank rather than
95    /// desynchronized.
96    fn on_open(
97        &mut self,
98        behavior: &mut MenuBehavior,
99        guard: &mut ContainerLockGuard,
100        player: &Player,
101    ) {
102        self.merchant.set_trading_player(Some(player.uuid()));
103
104        let offers = self.merchant.offers().lock().clone();
105        if !offers.is_empty() {
106            player.send_packet(CMerchantOffers {
107                container_id: i32::from(behavior.container_id()),
108                offers,
109                villager_level: self.merchant.merchant_level(),
110                villager_xp: self.merchant.villager_xp(),
111                show_progress: self.merchant.show_progress_bar(),
112                can_restock: self.merchant.can_restock(),
113            });
114        }
115
116        self.handler.update_result(guard);
117    }
118
119    /// Vanilla parity: `MerchantMenu.slotsChanged`, which rebuilds the result
120    /// from whatever now sits in the payment slots.
121    fn slots_changed(
122        &mut self,
123        _behavior: &mut MenuBehavior,
124        guard: &mut ContainerLockGuard,
125        _player: &Player,
126    ) {
127        self.handler.update_result(guard);
128    }
129
130    fn on_select_trade(
131        &mut self,
132        _behavior: &mut MenuBehavior,
133        guard: &mut ContainerLockGuard,
134        _player: &Player,
135        selected_trade: i32,
136    ) {
137        self.selection_hint.store(selected_trade, Ordering::Relaxed);
138        self.handler.update_result(guard);
139    }
140
141    /// Vanilla parity: `MerchantMenu.canTakeItemForPickAll`, which answers false
142    /// for every slot -- double-clicking in a trading screen gathers nothing.
143    fn can_take_item_for_pick_all(&self, _carried: &ItemStack, _slot_index: usize) -> bool {
144        false
145    }
146
147    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
148        self.merchant.still_valid(player)
149    }
150
151    /// Vanilla parity: the rest of `MerchantMenu.removed`. Handing the payment
152    /// back is the builder's `drain`; what is left is releasing the merchant so
153    /// the next customer can open the screen.
154    fn removed(&mut self, _behavior: &mut MenuBehavior, _player: &Player) {
155        self.merchant.set_trading_player(None);
156    }
157}
