1//! The villager trading screen.
2//!
3//! Vanilla parity: `MerchantMenu`. Three slots and a list of trades: two the
4//! player pays into, one the trade comes out of, and a selection the player
5//! makes by clicking a trade rather than by filling a slot.

21/// The result slot's index in the trading screen.
22const RESULT_SLOT: usize = 2;
23
24/// Builds the trading menu for `merchant`.
25#[must_use]
26pub fn merchant_menu(
27    inventory: Shared<PlayerInventory>,
28    container_id: u8,
29    merchant: Arc<dyn Merchant>,
30) -> Menu {
31    // Vanilla's MerchantContainer is one three-slot container. The payment half
32    // is the only half a click may write into, so it is the only half that is a
33    // plain container here.
34    let payment_container = SimpleContainer::new(2).into_shared();
35    let result_container = ResultContainer::new().into_shared();
36    let selection_hint = Arc::new(AtomicI32::new(NO_TRADE_SELECTED));
37
38    let handler = MerchantHandler::new(
39        payment_container.clone(),
40        result_container.clone(),
41        Arc::clone(&merchant),
42        Arc::clone(&selection_hint),
43    );
44
45    let mut builder = MenuBuilder::new(&vanilla_menu_types::MERCHANT, container_id);
46    let payment = builder.section_all(&payment_container);
47    let result = builder.result_slot(handler.clone());
48    let player = builder.player_inventory(&inventory);
49
50    // Vanilla parity: `MerchantMenu.quickMoveStack`. Taking the result fills the
51    // inventory back to front, paying fills it front to back.
52    builder.route(result, player.all(), FillDirection::Backward);
53    builder.route(payment, player.all(), FillDirection::Forward);
54    builder.route(player.all(), payment, FillDirection::Forward);
55    // Vanilla parity: the half of `MerchantMenu.removed` that hands the payment
56    // back to the player rather than eating it.
57    builder.drain(payment);
58
59    builder.build(MerchantKind {
60        handler,
61        merchant,
62        selection_hint,
63    })
64}

66/// Per-menu trading state.
67pub struct MerchantKind {
68    handler: MerchantHandler,
69    merchant: Arc<dyn Merchant>,
70    /// The trade the player clicked, shared with the handler.
71    selection_hint: Arc<AtomicI32>,
72}
73
74impl MerchantKind {
75    /// Sends the merchant's offers to the screen.
76    ///
77    /// Vanilla parity: the offers half of `Merchant.openTradingScreen`.
78    fn send_offers(&self, behavior: &MenuBehavior, player: &Player) {
79        let offers = self.merchant.offers().lock().clone();
80        if offers.is_empty() {
81            return;
82        }
83        player.send_packet(CMerchantOffers {
84            container_id: i32::from(behavior.container_id()),
85            offers,
86            villager_level: self.merchant.merchant_level(),
87            villager_xp: self.merchant.villager_xp(),
88            show_progress: self.merchant.show_progress_bar(),
89            can_restock: self.merchant.can_restock(),
90        });
91    }
92
93    /// Asks plugins whether the trade now set up may be taken. True when
94    /// there is none to take.
95    ///
96    /// Paper parity: `PlayerPurchaseEvent` from `MerchantResultSlot.onTake`,
97    /// asked before the take here so that a refusal leaves nothing to undo.
98    /// A refusal sends the offers again, as Paper does, so the screen shows
99    /// the stock as it really is.
100    fn purchase_allowed(
101        &self,
102        behavior: &MenuBehavior,
103        guard: &ContainerLockGuard,
104        player: &Player,
105    ) -> bool {
106        let Some(offer) = self.handler.current_trade(guard) else {
107            return true;
108        };
109        let offers = self.merchant.offers().lock().to_vec();
110        let mut event =
111            PlayerPurchaseEvent::new(player.uuid(), self.merchant.trader(), offer, offers);
112        player.fire_event(&mut event);
113        if !event.is_cancelled() {
114            return true;
115        }
116        self.send_offers(behavior, player);
117        false
118    }
119
120    /// Whether `click` takes from the result slot, and so buys.
121    fn takes_result(behavior: &MenuBehavior, guard: &ContainerLockGuard, click: Click) -> bool {
122        match click {
123            Click::Pickup { slot, .. } => {
124                if slot != RESULT_SLOT {
125                    return false;
126                }
127                // Vanilla takes the result into an empty hand, or onto a
128                // matching stack with room for all of it.
129                let carried = behavior.carried();
130                let Some(result) = behavior
131                    .slots()
132                    .get(RESULT_SLOT)
133                    .map(|view| view.get_item(guard).clone())
134                else {
135                    return false;
136                };
137                carried.is_empty()
138                    || (ItemStack::is_same_item_same_components(carried, &result)
139                        && carried.count() + result.count() <= carried.max_stack_size())
140            }
141            Click::Swap { slot, .. } | Click::Throw { slot, .. } => slot == RESULT_SLOT,
142            _ => false,
143        }
144    }
145
146    /// The container holding the two payment slots.
147    #[cfg(test)]
148    pub(crate) fn payment_id_for_tests(&self) -> ContainerId {
149        self.handler.payment_id()
150    }
151
152    /// The container holding what the trade produces.
153    #[cfg(test)]
154    pub(crate) fn result_id_for_tests(&self) -> ContainerId {
155        self.handler.result_id()
156    }
157}

166impl MenuKind for MerchantKind {
167    /// Vanilla parity: `Merchant.openTradingScreen`, which sends the offers
168    /// right after the menu packet. An empty list is not sent at all, which is
169    /// what leaves an unemployed villager's screen blank rather than
170    /// desynchronized.
171    fn on_open(
172        &mut self,
173        behavior: &mut MenuBehavior,
174        guard: &mut ContainerLockGuard,
175        player: &Player,
176    ) {
177        self.merchant.set_trading_player(Some(player.uuid()));
178        self.send_offers(behavior, player);
179        self.handler.update_result(guard);
180    }
181
182    /// Vanilla parity: `MerchantMenu.slotsChanged`, which rebuilds the result
183    /// from whatever now sits in the payment slots.
184    fn slots_changed(
185        &mut self,
186        _behavior: &mut MenuBehavior,
187        guard: &mut ContainerLockGuard,
188        _player: &Player,
189    ) {
190        self.handler.update_result(guard);
191    }
192
193    fn on_select_trade(
194        &mut self,
195        _behavior: &mut MenuBehavior,
196        guard: &mut ContainerLockGuard,
197        _player: &Player,
198        selected_trade: i32,
199    ) {
200        self.selection_hint.store(selected_trade, Ordering::Relaxed);
201        self.handler.update_result(guard);
202    }
203
204    fn on_slot_clicked(
205        &mut self,
206        behavior: &mut MenuBehavior,
207        guard: &mut ContainerLockGuard,
208        click: Click,
209        player: &Player,
210    ) -> ClickOutcome {
211        if Self::takes_result(behavior, guard, click)
212            && !self.purchase_allowed(behavior, guard, player)
213        {
214            return ClickOutcome::Consume;
215        }
216        ClickOutcome::Fallthrough
217    }
218
219    /// A shift-click on the result buys once per pass of vanilla's loop, so
220    /// each pass asks plugins; a refusal ends the loop with nothing moved.
221    fn quick_move(
222        &mut self,
223        behavior: &mut MenuBehavior,
224        guard: &mut ContainerLockGuard,
225        slot_index: usize,
226        player: &Player,
227    ) -> Option<ItemStack> {
228        if slot_index != RESULT_SLOT || self.purchase_allowed(behavior, guard, player) {
229            return None;
230        }
231        Some(ItemStack::empty())
232    }
233
234    /// Vanilla parity: `MerchantMenu.canTakeItemForPickAll`, which answers false
235    /// for every slot -- double-clicking in a trading screen gathers nothing.
236    fn can_take_item_for_pick_all(&self, _carried: &ItemStack, _slot_index: usize) -> bool {
237        false
238    }
239
240    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
241        self.merchant.still_valid(player)
242    }
243
244    /// Vanilla parity: the rest of `MerchantMenu.removed`. Handing the payment
245    /// back is the builder's `drain`; what is left is releasing the merchant so
246    /// the next customer can open the screen.
247    fn removed(&mut self, _behavior: &mut MenuBehavior, _player: &Player) {
248        self.merchant.set_trading_player(None);
249    }
250}
