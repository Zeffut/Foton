1//! Hopper menu.
2//!
3//! Vanilla parity: `HopperMenu`. Five slots in a row, then the player
4//! inventory. Layout:
5//! - Slots 0 to 4: the hopper
6//! - Slots 5 to 31: main inventory (27)
7//! - Slots 32 to 40: hotbar (9)
8
9use foton_registry::vanilla_menu_types;
10
11use crate::inventory::prelude::*;
12use crate::player::player_inventory::PlayerInventory;
13
14/// Slots in a hopper.
15///
16/// Vanilla parity: `HopperMenu.CONTAINER_SIZE`.
17const HOPPER_SLOTS: usize = 5;
18
19/// Builds the hopper menu.
20#[must_use]
21pub fn hopper(
22    inventory: Shared<PlayerInventory>,
23    container_id: u8,
24    container: impl Into<ContainerRef>,
25) -> Menu {
26    let container = container.into();
27
28    let mut builder = MenuBuilder::new(&vanilla_menu_types::HOPPER, container_id);
29    let hopper = builder.section(&container, HOPPER_SLOTS);
30    let player = builder.player_inventory(&inventory);
31
32    builder.route(hopper, player.all(), FillDirection::Backward);
33    builder.route(player.all(), hopper, FillDirection::Forward);
34
35    builder.build(HopperKind { container })
36}
37
38/// Per-menu hopper state: the backing container for the validity check.
39pub struct HopperKind {
40    container: ContainerRef,
41}
42
43// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
44// within the process.
45unsafe impl foton_utils::DowncastType for HopperKind {
46    const TYPE_KEY: foton_utils::DowncastTypeKey =
47        foton_utils::DowncastTypeKey::new("foton:menu/hopper");
48}
49
50impl MenuKind for HopperKind {
51    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
52        self.container.still_valid(player)
53    }
54}

56#[cfg(test)]
57mod tests {
58    use foton_utils::locks::IntoShared as _;
59
60    use super::*;
61    use crate::inventory::container::SimpleContainer;
62
63    #[test]
64    fn hopper_menu_exposes_five_slots_plus_the_player_inventory() {
65        let inventory = PlayerInventory::new().into_shared();
66        let container = SimpleContainer::new(HOPPER_SLOTS).into_shared();
67
68        let menu = hopper(inventory, 1, container);
69
70        assert_eq!(menu.behavior().slot_count(), HOPPER_SLOTS + 36);
71    }
