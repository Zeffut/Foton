1//! Stonecutter menu.
2//!
3//! Vanilla parity: `StonecutterMenu`. Two slots and one number: what went in,
4//! what comes out, and which of the input's many cuts the player picked. The
5//! number is the whole difference from a crafting table -- a block of stone
6//! offers a dozen results and nothing but the player's choice separates them.
7

25/// Builds the stonecutter menu.
26#[must_use]
27pub fn stonecutter(
28    inventory: Shared<PlayerInventory>,
29    container_id: u8,
30    block_pos: BlockPos,
31) -> Menu {
32    let input_container = SimpleContainer::new(1).into_shared();
33    let result_container = ResultContainer::new().into_shared();
34    let selected = Arc::new(AtomicI32::new(NO_SELECTION));
35
36    let handler = StonecutterHandler::new(
37        input_container.clone(),
38        result_container.clone(),
39        Arc::clone(&selected),
40    );
41
42    let mut builder = MenuBuilder::new(&vanilla_menu_types::STONECUTTER, container_id);
43    let input = builder.section_all(&input_container);
44    let result = builder.result_slot(handler.clone());
45    let player = builder.player_inventory(&inventory);
46
47    // Vanilla parity: the single `selectedRecipeIndex` data slot, which is how
48    // the client knows which button to draw pressed.
49    let selected_slot = builder.data_slot(i16::try_from(NO_SELECTION).unwrap_or(-1));
50
51    builder.route(result, player.all(), FillDirection::Backward);
52    builder.route(input, player.all(), FillDirection::Forward);
53    builder.route(player.all(), input, FillDirection::Forward);
54    builder.drain(input);
55
56    builder.build(StonecutterKind {
57        handler,
58        selected,
59        selected_slot,
60        result,
61        block_pos,
62        last_input: ItemStack::empty(),
63    })
64}

66/// Per-menu stonecutter state.
67pub struct StonecutterKind {
68    handler: StonecutterHandler,
69    /// The chosen recipe, shared with the handler.
70    selected: Arc<AtomicI32>,
71    /// The same number, mirrored to the client.
72    selected_slot: DataSlot,
73    /// The result section, so pickup-all can be kept out of it.
74    result: Section,
75    block_pos: BlockPos,
76    /// What was in the input last time it was looked at.
77    ///
78    /// Vanilla parity: the `input` field of `StonecutterMenu`, which exists so
79    /// that stacking one more block into the slot does not throw away the
80    /// choice the player already made.
81    last_input: ItemStack,
82}
83
84// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
85// within the process.
86unsafe impl foton_utils::DowncastType for StonecutterKind {
87    const TYPE_KEY: foton_utils::DowncastTypeKey =
88        foton_utils::DowncastTypeKey::new("foton:menu/stonecutter");
89}

91impl StonecutterKind {
92    /// The container the input sits in.
93    #[cfg(test)]
94    pub(crate) fn input_id(&self) -> ContainerId {
95        self.handler.input_id_for_tests()
96    }
97
98    /// The container the cut appears in.
99    #[cfg(test)]
100    pub(crate) fn result_id(&self) -> ContainerId {
101        self.handler.result_id_for_tests()
102    }
103
104    /// Writes the selection to the client and rebuilds the result.
105    fn apply_selection(
106        &mut self,
107        behavior: &mut MenuBehavior,
108        guard: &mut ContainerLockGuard,
109        index: i32,
110    ) {
111        self.selected.store(index, Ordering::Relaxed);
112        self.selected_slot
113            .set(behavior, i16::try_from(index).unwrap_or(-1));
114        self.handler.update_result(guard);
115    }
116}
117
118impl MenuKind for StonecutterKind {
119    /// Vanilla parity: `StonecutterMenu.slotsChanged`, which only resets the
120    /// choice when the *kind* of item in the slot changes. Adding a second
121    /// block of the same stone keeps the cut the player picked.
122    fn slots_changed(
123        &mut self,
124        behavior: &mut MenuBehavior,
125        guard: &mut ContainerLockGuard,
126        _player: &Player,
127    ) {
128        let input = self
129            .handler
130            .input_snapshot(guard)
131            .unwrap_or_else(ItemStack::empty);
132
133        if ItemStack::is_same_item(&input, &self.last_input) {
134            return;
135        }
136        self.last_input = input;
137        self.apply_selection(behavior, guard, NO_SELECTION);
138    }
139
140    /// Vanilla parity: `StonecutterMenu.clickMenuButton`. An out-of-range
141    /// button is ignored rather than clearing the choice, so a click that races
142    /// a slot change cannot wipe what the player picked.
143    fn on_button_click(
144        &mut self,
145        behavior: &mut MenuBehavior,
146        guard: &mut ContainerLockGuard,
147        _player: &Player,
148        button: i32,
149    ) -> bool {
150        if self.selected.load(Ordering::Relaxed) == button {
151            return false;
152        }
153        let count = self.handler.recipe_count(guard);
154        if button >= 0 && (button as usize) < count {
155            self.apply_selection(behavior, guard, button);
156        }
157        true
158    }
159
160    fn can_take_item_for_pick_all(&self, _carried: &ItemStack, slot_index: usize) -> bool {
161        !self.result.contains(slot_index)
162    }
163
164    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
165        let world = player.get_world();
166        world.get_block_state(self.block_pos).get_block() == &vanilla_blocks::STONECUTTER
167            && player.is_within_block_interaction_range_with_buffer(self.block_pos, 4.0)
168    }
169}
