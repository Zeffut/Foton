1//! Beacon menu.
2//!
3//! Vanilla parity: `BeaconMenu`. One slot that takes a single ingot or gem,
4//! three numbers the client reads to draw the effect buttons, and the player
5//! inventory. Picking the effects is not a click: it arrives as its own
6//! packet, which is why nothing here handles buttons.

27/// Builds the beacon menu.
28#[must_use]
29pub fn beacon(
30    inventory: Shared<PlayerInventory>,
31    container_id: u8,
32    data: Arc<BeaconDataSlots>,
33    block_entity: SharedBlockEntity,
34) -> Menu {
35    let payment: ContainerRef = SimpleContainer::new(1).into_shared().into();
36
37    let mut builder = MenuBuilder::new(&vanilla_menu_types::BEACON, container_id);
38    // Vanilla parity: `BeaconMenu.PaymentSlot`, which takes one item and only
39    // from the payment tag.
40    let payment_slot = builder.section_with(
41        payment.clone(),
42        1,
43        SectionKind::restricted(|_slot, stack| {
44            REGISTRY
45                .items
46                .is_in_tag(stack.item(), &ItemTag::BEACON_PAYMENT_ITEMS)
47        }),
48    );
49    let data_slots: [DataSlot; BEACON_DATA_SLOTS] = array::from_fn(|_| builder.data_slot(0));
50    let player = builder.player_inventory(&inventory);
51
52    builder.route(payment_slot, player.all(), FillDirection::Backward);
53    builder.route(player.all(), payment_slot, FillDirection::Forward);
54    // Vanilla parity: `BeaconMenu.removed`, which hands the payment back
55    // rather than keeping it -- a beacon has no storage.
56    builder.drain(payment_slot);
57
58    builder.build(BeaconKind {
59        payment,
60        data,
61        data_slots,
62        block_entity,
63    })
64}

66/// Per-menu beacon state.
67pub struct BeaconKind {
68    /// The one-slot payment container.
69    payment: ContainerRef,
70    /// Levels and effects, shared with the block entity.
71    data: Arc<BeaconDataSlots>,
72    /// Handles to the three synced values.
73    data_slots: [DataSlot; BEACON_DATA_SLOTS],
74    /// The beacon itself, which owns the chosen effects.
75    block_entity: SharedBlockEntity,
76}
77
78// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
79// within the process.
80unsafe impl foton_utils::DowncastType for BeaconKind {
81    const TYPE_KEY: foton_utils::DowncastTypeKey =
82        foton_utils::DowncastTypeKey::new("foton:menu/beacon");
83}
84
85impl MenuKind for BeaconKind {
86    /// Vanilla parity: `BeaconMenu.stillValid`, which checks the block is
87    /// still a beacon and the player still in reach.
88    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
89        let pos = self.block_entity.get_block_pos();
90        let world = player.get_world();
91        world.get_block_state(pos).get_block() == &vanilla_blocks::BEACON
92            && player.is_within_block_interaction_range_with_buffer(pos, 4.0)
93    }
94
95    /// Vanilla parity: `BeaconMenu.updateEffects`.
96    ///
97    /// The payment is only taken once the effects are accepted, so a request
98    /// the beacon refuses -- too small a pyramid, an effect it does not offer
99    /// -- costs the player nothing.
100    fn on_set_beacon_effects(
101        &mut self,
102        _behavior: &mut MenuBehavior,
103        guard: &mut ContainerLockGuard,
104        _player: &Player,
105        primary: Option<i32>,
106        secondary: Option<i32>,
107    ) {
108        let paid = guard
109            .get(self.payment.container_id())
110            .is_some_and(|container| !container.get_item(0).is_empty());
111        if !paid {
112            return;
113        }
114
115        let Some(beacon) = self.block_entity.downcast_ref::<BeaconBlockEntity>() else {
116            return;
117        };
118        if !beacon.set_effects(
119            primary.and_then(effect_from_holder_id),
120            secondary.and_then(effect_from_holder_id),
121        ) {
122            return;
123        }
124
125        if let Some(container) = guard.get_mut(self.payment.container_id()) {
126            container.remove_item(0, 1);
127            container.set_changed();
128        }
129    }
130
131    /// Pushes the pyramid size and the chosen effects into the synced data
132    /// slots, which is what draws the buttons the client offers.
133    fn on_tick(
134        &mut self,
135        behavior: &mut MenuBehavior,
136        _guard: &mut ContainerLockGuard,
137        _player: &Player,
138    ) {
139        let values = self.data.snapshot();
140        for (slot, value) in self.data_slots.iter().zip(values) {
141            slot.set(behavior, value);
142        }
143    }
144}
