1//! Lectern menu.
2//!
3//! Vanilla parity: `LecternMenu`. One number the player can change and one
4//! book they cannot pick up: the buttons turn pages or take the book away.
5//! Everything arrives as a button rather than a slot click, which is why this
6//! menu has almost no slot handling at all.

26/// Vanilla parity: `LecternMenu.BUTTON_PREV_PAGE`.
27const BUTTON_PREVIOUS_PAGE: i32 = 1;
28/// Vanilla parity: `LecternMenu.BUTTON_NEXT_PAGE`.
29const BUTTON_NEXT_PAGE: i32 = 2;
30/// Vanilla parity: `LecternMenu.BUTTON_TAKE_BOOK`.
31const BUTTON_TAKE_BOOK: i32 = 3;
32/// Vanilla parity: `LecternMenu.BUTTON_PAGE_JUMP_RANGE_START`, above which a
33/// button id is a page number rather than a command.
34const BUTTON_PAGE_JUMP_RANGE_START: i32 = 100;
35
36/// Builds the lectern menu.
37#[must_use]
38pub fn lectern(
39    inventory: Shared<PlayerInventory>,
40    container_id: u8,
41    block_pos: BlockPos,
42    world: &Arc<World>,
43) -> Menu {
44    // Vanilla parity: `LecternMenu` shows the book in one slot and carries no
45    // player inventory -- the book is not something the player can reach
46    // without the Take Book button. The slot exists so the client has
47    // something to draw; the book itself lives on the block entity.
48    let _ = inventory;
49
50    let book_display = SimpleContainer::new(1).into_shared();
51    if let Some(book) = book_on(world, block_pos) {
52        book_display.lock().set_item(0, book);
53    }
54
55    let mut builder = MenuBuilder::new(&vanilla_menu_types::LECTERN, container_id);
56    let _book_slot = builder.section_all(&book_display);
57    let page = builder.data_slot(0);
58
59    builder.build(LecternKind {
60        page,
61        block_pos,
62        world: Arc::clone(world),
63        book_display,
64    })
65}

67/// Per-menu lectern state.
68pub struct LecternKind {
69    /// The open page, mirrored to the client.
70    page: DataSlot,
71    block_pos: BlockPos,
72    world: Arc<World>,
73    /// A copy of the book, purely so the client has something to draw.
74    book_display: Shared<SimpleContainer>,
75}

77// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
78// within the process.
79unsafe impl foton_utils::DowncastType for LecternKind {
80    const TYPE_KEY: foton_utils::DowncastTypeKey =
81        foton_utils::DowncastTypeKey::new("foton:menu/lectern");
82}
83
84impl MenuKind for LecternKind {
85    /// Vanilla parity: `LecternMenu.clickMenuButton`.
86    ///
87    /// The page buttons only pulse the block when the page really moved --
88    /// clicking "next" on the last page of a book must not tick the redstone,
89    /// which is what stops a lectern being used as a clock.
90    fn on_button_click(
91        &mut self,
92        behavior: &mut MenuBehavior,
93        _guard: &mut ContainerLockGuard,
94        player: &Player,
95        button: i32,
96    ) -> bool {
97        let current = i32::from(self.page.get(behavior));
98
99        let target = match button {
100            BUTTON_PREVIOUS_PAGE => current - 1,
101            BUTTON_NEXT_PAGE => current + 1,
102            BUTTON_TAKE_BOOK => {
103                let mut event = PlayerTakeLecternBookEvent::new(
104                    player.uuid(),
105                    self.world.key.to_string(),
106                    self.block_pos,
107                );
108                self.world.fire_event(&mut event);
109                if event.is_cancelled() {
110                    return true;
111                }
112                let book = take_book_from(&self.world, self.block_pos);
113                if !book.is_empty() {
114                    // Vanilla parity: the book goes to the player's inventory,
115                    // or on the floor if there is no room for it.
116                    player.add_item_or_drop(book);
117                }
118                self.book_display.lock().set_item(0, ItemStack::empty());
119                return true;
120            }
121            jump if jump >= BUTTON_PAGE_JUMP_RANGE_START => jump - BUTTON_PAGE_JUMP_RANGE_START,
122            _ => return false,
123        };
124
125        if !turn_page(&self.world, self.block_pos, target) {
126            return false;
127        }
128        signal_lectern_page_change(&self.world, self.block_pos);
129        self.page
130            .set(behavior, i16::try_from(target).unwrap_or(i16::MAX));
131        true
132    }
133
134    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
135        let world = player.get_world();
136        world.get_block_state(self.block_pos).get_block() == &vanilla_blocks::LECTERN
137            && player.is_within_block_interaction_range_with_buffer(self.block_pos, 4.0)
138    }
139}

141/// Returns the book on the lectern at `pos`, if there is one.
142fn book_on(world: &Arc<World>, pos: BlockPos) -> Option<ItemStack> {
143    use foton_utils::Downcast as _;
144
145    let entity = world.get_block_entity(pos)?;
146    let lectern = entity.downcast_ref::<LecternBlockEntity>()?;
147    Some(lectern.book())
148}
149
150/// Turns the lectern at `pos` to `page`, reporting whether it moved.
151fn turn_page(world: &Arc<World>, pos: BlockPos, page: i32) -> bool {
152    use foton_utils::Downcast as _;
153
154    world
155        .get_block_entity(pos)
156        .and_then(|entity| {
157            entity
158                .downcast_ref::<LecternBlockEntity>()
159                .map(|lectern| lectern.set_page(page))
160        })
161        .unwrap_or(false)
162}
