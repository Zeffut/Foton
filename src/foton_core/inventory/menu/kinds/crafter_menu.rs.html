1//! Crafter menu.
2//!
3//! Vanilla parity: `CrafterMenu`. A 3x3 grid, the player inventory, and one
4//! slot on the right showing what the recipe would make -- a preview only: it
5//! takes nothing and gives nothing, because the crafter builds its result on a
6//! redstone pulse rather than when a player clicks.
7//!
8//! The nine grid slots can each be switched off from the client, which is the
9//! one thing this menu does that no other does.
10

29/// Builds the crafter menu.
30#[must_use]
31pub fn crafter(
32    inventory: Shared<PlayerInventory>,
33    container_id: u8,
34    container: impl Into<ContainerRef>,
35    data: Arc<CrafterDataSlots>,
36) -> Menu {
37    let container = container.into();
38    let preview: ContainerRef = SimpleContainer::new(1).into_shared().into();
39
40    let mut builder = MenuBuilder::new(&vanilla_menu_types::CRAFTER_3X3, container_id);
41    let grid = builder.section(&container, CRAFTER_SLOTS);
42    let player = builder.player_inventory(&inventory);
43    // Vanilla parity: `NonInteractiveResultSlot` -- neither half of a click
44    // does anything to it.
45    let result = builder.section_with(
46        preview.clone(),
47        1,
48        SectionKind::guarded(
49            |_slot, _stack| false,
50            |_slot, _stack, _guard, _player| false,
51        ),
52    );
53
54    let data_slots: [DataSlot; CRAFTER_DATA_SLOTS] = array::from_fn(|_| builder.data_slot(0));
55
56    builder.route(grid, player.all(), FillDirection::Backward);
57    builder.route(player.all(), grid, FillDirection::Forward);
58    // The preview holds nothing a shift-click could move, but leaving it out of
59    // the routes keeps it from ever being picked as a target.
60    let _ = result;
61
62    builder.build(CrafterKind {
63        container,
64        preview,
65        data,
66        data_slots,
67    })
68}

70/// Per-menu crafter state.
71pub struct CrafterKind {
72    /// The nine grid slots, owned by the block entity.
73    container: ContainerRef,
74    /// The preview slot's own one-slot container.
75    preview: ContainerRef,
76    /// Slot states and the redstone flag, shared with the block entity.
77    data: Arc<CrafterDataSlots>,
78    /// Handles to the ten synced values.
79    data_slots: [DataSlot; CRAFTER_DATA_SLOTS],
80}
81
82// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
83// within the process.
84unsafe impl foton_utils::DowncastType for CrafterKind {
85    const TYPE_KEY: foton_utils::DowncastTypeKey =
86        foton_utils::DowncastTypeKey::new("foton:menu/crafter");
87}
88
89impl CrafterKind {
90    /// Recomputes what the grid would make.
91    ///
92    /// Vanilla parity: `CrafterMenu.refreshRecipeResult`.
93    ///
94    /// Both containers are reached through the guard rather than locked
95    /// directly: the guard already holds every container this menu covers, and
96    /// the preview is one of them.
97    fn refresh_preview(&self, guard: &mut ContainerLockGuard) {
98        let Some(grid) = guard.get_typed::<CrafterContainer>(self.container.container_id()) else {
99            return;
100        };
101        let input = CraftingInput::new(CRAFTER_WIDTH, CRAFTER_HEIGHT, grid.items().to_vec());
102        let result = REGISTRY
103            .recipes
104            .find_crafting_recipe(&input)
105            .map_or_else(ItemStack::empty, |recipe| recipe.assemble());
106
107        if let Some(preview) = guard.get_mut(self.preview.container_id()) {
108            preview.set_item(0, result);
109            preview.set_changed();
110        }
111    }
112}
113
114impl MenuKind for CrafterKind {
115    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
116        self.container.still_valid(player)
117    }
118
119    fn on_open(
120        &mut self,
121        _behavior: &mut MenuBehavior,
122        guard: &mut ContainerLockGuard,
123        _player: &Player,
124    ) {
125        self.refresh_preview(guard);
126    }
127
128    fn slots_changed(
129        &mut self,
130        _behavior: &mut MenuBehavior,
131        guard: &mut ContainerLockGuard,
132        _player: &Player,
133    ) {
134        self.refresh_preview(guard);
135    }
136
137    /// Switches one grid slot on or off.
138    ///
139    /// Vanilla parity: `CrafterMenu.setSlotState`. A slot that holds something
140    /// cannot be switched off, which the block entity enforces; the preview is
141    /// recomputed either way because a disabled slot is not an empty one as far
142    /// as the recipe is concerned.
143    fn on_slot_state_changed(
144        &mut self,
145        _behavior: &mut MenuBehavior,
146        guard: &mut ContainerLockGuard,
147        _player: &Player,
148        slot: usize,
149        enabled: bool,
150    ) {
151        if slot >= CRAFTER_SLOTS {
152            return;
153        }
154        let holds_something = guard
155            .get_typed::<CrafterContainer>(self.container.container_id())
156            .is_some_and(|grid| !grid.get_item(slot).is_empty());
157        if holds_something {
158            return;
159        }
160
161        self.data.set_slot_disabled(slot, !enabled);
162        if let Some(owner) = self.container.owner_block_entity() {
163            owner.set_changed();
164        }
165        self.refresh_preview(guard);
166    }
167
168    /// Pushes the slot states and the redstone flag into the synced data slots.
169    ///
170    /// Vanilla hands the menu the block entity's own `ContainerData`; Foton
171    /// republishes it here for the same reason the furnace menu does.
172    fn on_tick(
173        &mut self,
174        behavior: &mut MenuBehavior,
175        _guard: &mut ContainerLockGuard,
176        _player: &Player,
177    ) {
178        let values = self.data.snapshot();
179        for (slot, value) in self.data_slots.iter().zip(values) {
180            slot.set(behavior, value);
181        }
182    }
183}
