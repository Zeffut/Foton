1//! Chest menu for chest-like containers (chests, barrels, ender chests, shulker boxes).
2//!
3//! 1-6 rows of 9 slots. Layout:
4//! - Slots 0 to `rows * 9 - 1`: Container
5//! - Slots `rows * 9` to `rows * 9 + 26`: Main inventory (27)
6//! - Slots `rows * 9 + 27` to `rows * 9 + 35`: Hotbar (9)
7

18/// Builds a chest-like menu with `rows` rows of 9 slots plus the player inventory.
19///
20/// # Panics
21/// Panics if `rows` is 0 or greater than 6.
22#[must_use]
23pub fn chest(
24    inventory: Shared<PlayerInventory>,
25    container_id: u8,
26    container: impl Into<ContainerRef>,
27    rows: usize,
28) -> Menu {
29    let container = container.into();
30    assert!(
31        (1..=6).contains(&rows),
32        "Chest rows must be between 1 and 6"
33    );
34
35    let mut builder = MenuBuilder::new(menu_type_for_rows(rows), container_id);
36    let chest = builder.section(&container, rows * 9);
37    let player = builder.player_inventory(&inventory);
38
39    builder.route(chest, player.all(), FillDirection::Backward);
40    builder.route(player.all(), chest, FillDirection::Forward);
41
42    builder.build(ChestKind {
43        container,
44        second_container: None,
45    })
46}

48/// Builds a double chest menu: two 27-slot halves presented as one 54-slot
49/// container, plus the player inventory.
50///
51/// Vanilla parity: `ChestBlock` combines two `ChestBlockEntity` halves through
52/// a `CompoundContainer`. Foton keeps the halves as two independently lockable
53/// containers and joins them at the menu layer instead, which preserves the
54/// per-container locking contract while exposing the same 54 logical slots.
55#[must_use]
56pub fn double_chest(
57    inventory: Shared<PlayerInventory>,
58    container_id: u8,
59    first: impl Into<ContainerRef>,
60    second: impl Into<ContainerRef>,
61) -> Menu {
62    let first = first.into();
63    let second = second.into();
64
65    let mut builder = MenuBuilder::new(&vanilla_menu_types::GENERIC_9X6, container_id);
66    let upper = builder.section(&first, CHEST_HALF_SLOTS);
67    let lower = builder.section(&second, CHEST_HALF_SLOTS);
68    let player = builder.player_inventory(&inventory);
69
70    builder.route([upper, lower], player.all(), FillDirection::Backward);
71    builder.route(player.all(), [upper, lower], FillDirection::Forward);
72
73    builder.build(ChestKind {
74        container: first,
75        second_container: Some(second),
76    })
77}
78
79/// Slots in a single chest half.
80const CHEST_HALF_SLOTS: usize = 27;
81
82/// Menu type for a chest of `rows` rows.
83///
84/// # Panics
85/// Panics if `rows` is 0 or greater than 6.
86#[must_use]
87pub fn menu_type_for_rows(rows: usize) -> MenuTypeRef {
88    match rows {
89        1 => &vanilla_menu_types::GENERIC_9X1,
90        2 => &vanilla_menu_types::GENERIC_9X2,
91        3 => &vanilla_menu_types::GENERIC_9X3,
92        4 => &vanilla_menu_types::GENERIC_9X4,
93        5 => &vanilla_menu_types::GENERIC_9X5,
94        6 => &vanilla_menu_types::GENERIC_9X6,
95        _ => panic!("Invalid row count: {rows}"),
96    }
97}

99/// Per-menu chest state: the backing container(s) for the validity check.
100pub struct ChestKind {
101    /// The backing container. For a double chest, this is the upper half.
102    container: ContainerRef,
103    /// The lower half of a double chest, if any.
104    second_container: Option<ContainerRef>,
105}
106
107// SAFETY: This Foton-owned key uniquely identifies the concrete menu kind
108// within the process.
109unsafe impl foton_utils::DowncastType for ChestKind {
110    const TYPE_KEY: foton_utils::DowncastTypeKey =
111        foton_utils::DowncastTypeKey::new("foton:menu/chest");
112}
113
114impl ChestKind {
115    /// Runs `f` on the block entity behind each half, if there is one.
116    ///
117    /// A double chest counts as one opener on each half, which is what makes
118    /// both lids rise together.
119    fn for_each_owner(&self, f: impl Fn(&BlockEntityBase)) {
120        for container in iter::once(&self.container).chain(self.second_container.as_ref()) {
121            if let Some(owner) = container.owner_block_entity() {
122                f(&owner);
123            }
124        }
125    }
126}
127
128impl MenuKind for ChestKind {
129    /// Vanilla parity: `ChestBlockEntity.startOpen`, which ignores spectators
130    /// -- a spectator looking into a chest must not raise its lid or power a
131    /// trapped chest, because nobody can see them do it.
132    fn on_open(
133        &mut self,
134        _behavior: &mut MenuBehavior,
135        _guard: &mut ContainerLockGuard,
136        player: &Player,
137    ) {
138        if player.is_spectator() {
139            return;
140        }
141        self.for_each_owner(BlockEntityBase::increment_openers);
142    }
143
144    /// Vanilla parity: `ChestBlockEntity.stopOpen`.
145    fn removed(&mut self, _behavior: &mut MenuBehavior, player: &Player) {
146        if player.is_spectator() {
147            return;
148        }
149        self.for_each_owner(BlockEntityBase::decrement_openers);
150    }
151
152    /// Returns true if every backing container is still valid for the player.
153    ///
154    /// Vanilla parity: a double chest closes as soon as either half becomes
155    /// invalid.
156    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
157        if !self.container.still_valid(player) {
158            return false;
159        }
160        self.second_container
161            .as_ref()
162            .is_none_or(|second| second.still_valid(player))
163    }
164}
