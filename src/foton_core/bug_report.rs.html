1//! Bug reports filed by players from inside the game.
2//!
3//! A test session's whole value is what the testers noticed, and the gap
4//! between noticing and reporting is where most of it is lost. So the report
5//! is filed where the player is standing, and everything that can be captured
6//! without asking -- who, where, which world, which build -- is captured
7//! rather than typed.
8//!
9//! Reports are appended to `reports/bugs.jsonl`, one JSON object per line.
10//! Append-only and line-delimited on purpose: concurrent writers cannot
11//! corrupt each other's records the way a rewritten array would, a truncated
12//! last line costs one report rather than the file, and every ordinary tool
13//! reads it.
14

32/// The longest description a single report may carry.

34/// Long enough for real reproduction steps, short enough that one player

38/// What part of the game a report is about.
39///
40/// A free-text category would be unsortable within a day, so the list is
41/// closed. `Other` is deliberate: a tester who cannot place their bug should
42/// still be able to file it.
43#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
44#[serde(rename_all = "snake_case")]
45pub enum BugCategory {
46    /// Placing, breaking, or a block's own behavior.
47    Blocks,
48    /// What an item does in hand, or its components.
49    Items,
50    /// Spawning, AI, pathing, or a mob's own behavior.
51    Mobs,
52    /// Damage, knockback, shields, enchantment effects.
53    Combat,
54    /// Signals, timing, and the blocks that carry them.
55    Redstone,
56    /// Terrain, biomes, structures, and their contents.
57    Worldgen,
58    /// A command's parsing, permissions, or result.
59    Commands,
60    /// Containers, menus, crafting, and item movement.
61    Inventory,
62    /// A sound that is missing, wrong, or misplaced.
63    Sounds,
64    /// Lag spikes, stutter, and anything that got slow.
65    Performance,
66    /// Anything that fits nowhere else; never reject a report for want of a category.
67    Other,
68}

70impl BugCategory {
71    /// Every category, in the order a form should offer them.
72    pub const ALL: [Self; 11] = [
73        Self::Blocks,
74        Self::Items,
75        Self::Mobs,
76        Self::Combat,
77        Self::Redstone,
78        Self::Worldgen,
79        Self::Commands,
80        Self::Inventory,
81        Self::Sounds,
82        Self::Performance,
83        Self::Other,
84    ];
85
86    /// The name this category is written and parsed as.
87    #[must_use]
88    pub const fn name(self) -> &'static str {
89        match self {
90            Self::Blocks => "blocks",
91            Self::Items => "items",
92            Self::Mobs => "mobs",
93            Self::Combat => "combat",
94            Self::Redstone => "redstone",
95            Self::Worldgen => "worldgen",
96            Self::Commands => "commands",
97            Self::Inventory => "inventory",
98            Self::Sounds => "sounds",
99            Self::Performance => "performance",
100            Self::Other => "other",
101        }
102    }
103
104    /// The name a player reads in the report form.
105    ///
106    /// Separate from [`Self::name`] on purpose: the written name is a stable
107    /// key the report file is sorted by, and must not move when the wording
108    /// shown to a tester is improved.
109    #[must_use]
110    pub const fn label(self) -> &'static str {
111        match self {
112            Self::Blocks => "Blocks",
113            Self::Items => "Items",
114            Self::Mobs => "Mobs and AI",
115            Self::Combat => "Combat and damage",
116            Self::Redstone => "Redstone",
117            Self::Worldgen => "World generation",
118            Self::Commands => "Commands",
119            Self::Inventory => "Inventory and menus",
120            Self::Sounds => "Sounds",
121            Self::Performance => "Lag and performance",
122            Self::Other => "Something else",
123        }
124    }
125
126    /// Parses a category name, case-insensitively.
127    #[must_use]
128    pub fn parse(raw: &str) -> Option<Self> {
129        let lowered = raw.to_ascii_lowercase();
130        Self::ALL.into_iter().find(|c| c.name() == lowered)
131    }
132}

134/// One report, as it is written to disk.
135#[derive(Debug, Clone, Serialize, Deserialize)]
136pub struct BugReport {
137    /// Seconds since the Unix epoch, so the file sorts by time on its own.
138    pub at: u64,
139    /// The reporter's name at the time of filing.
140    pub player: String,
141    /// The reporter's UUID, which their name is not a substitute for.
142    pub uuid: String,
143    /// The world the report was filed in.
144    pub world: String,
145    /// Where the reporter was standing, rounded to a sensible precision.
146    pub position: [f64; 3],
147    /// What part of the game the report is about.
148    pub category: BugCategory,
149    /// What the reporter typed: what happened and how to see it again.
150    pub description: String,
151    /// The build the report was filed against.
152    ///
153    /// Without it a fixed bug and a live one look identical in the file.
154    pub version: String,
155}
156
157impl BugReport {
158    /// Appends this report and returns its one-based number in the file.
159    ///
160    /// # Errors
161    ///
162    /// Returns the underlying I/O error if the directory cannot be created or
163    /// the file cannot be appended to.
164    pub fn append_in(&self, run_dir: &Path) -> IoResult<usize> {
165        let dir = run_dir.join(REPORTS_DIR);
166        fs::create_dir_all(&dir)?;
167        let path = dir.join(REPORTS_FILE);
168
169        let mut line = serde_json::to_string(self).map_err(IoError::other)?;
170        line.push('\n');
171
172        let mut file = OpenOptions::new().create(true).append(true).open(&path)?;
173        file.write_all(line.as_bytes())?;
174        file.flush()?;
175
176        Ok(count_lines(&path).unwrap_or(0))
177    }
178
179    /// Reads every report on file, oldest first.
180    ///
181    /// A line that does not parse is skipped rather than fatal: one bad record
182    /// must not hide the rest.
183    ///
184    /// # Errors
185    ///
186    /// Returns the underlying I/O error if the file exists but cannot be read.
187    pub fn read_all(run_dir: &Path) -> IoResult<Vec<Self>> {
188        let path = reports_path(run_dir);
189        if !path.exists() {
190            return Ok(Vec::new());
191        }
192        let file = fs::File::open(&path)?;
193        Ok(BufReader::new(file)
194            .lines()
195            .map_while(Result::ok)
196            .filter(|line| !line.trim().is_empty())
197            .filter_map(|line| serde_json::from_str(&line).ok())
198            .collect())
199    }
200
201    /// Builds a report stamped with the current time.
202    #[must_use]
203    pub fn now(
204        player: String,
205        uuid: String,
206        world: String,
207        position: [f64; 3],
208        category: BugCategory,
209        description: String,
210    ) -> Self {
211        Self {
212            at: SystemTime::now()
213                .duration_since(UNIX_EPOCH)
214                .map_or(0, |since| since.as_secs()),
215            player,
216            uuid,
217            world,
218            position: position.map(|value| (value * 100.0).round() / 100.0),
219            category,
220            description,
221            version: env!("CARGO_PKG_VERSION").to_owned(),
222        }
223    }
224}

226/// The path reports live at for a given server directory.
227#[must_use]
228pub fn reports_path(run_dir: &Path) -> PathBuf {
229    run_dir.join(REPORTS_DIR).join(REPORTS_FILE)
230}
231
232fn count_lines(path: &Path) -> IoResult<usize> {
233    let file = fs::File::open(path)?;
234    Ok(BufReader::new(file)
235        .lines()
236        .map_while(Result::ok)
237        .filter(|line| !line.trim().is_empty())
238        .count())
239}

241/// How many times one report is offered to the webhook before it is left behind.
242const FORWARD_ATTEMPTS: usize = 3;
243
244/// How long a single delivery may take.
245const FORWARD_TIMEOUT: Duration = Duration::from_secs(5);
246
247/// How long to wait before offering a report again.
248const FORWARD_RETRY_DELAY: Duration = Duration::from_secs(2);
249
250/// One report as the endpoint receives it: the record plus its file number.
251///
252/// The number is what makes a delivery identifiable. Without it the receiver
253/// cannot tell a retry from a second report with the same text, and a
254/// backfill cannot know where it left off.
255#[derive(Serialize)]
256struct ForwardedReport<'report> {
257    number: usize,
258    #[serde(flatten)]
259    report: &'report BugReport,
260}
261
262/// Sends a freshly filed report to the configured endpoint, in the background.
263///
264/// Deliberately fire-and-forget: the report is already on disk before this is
265/// called, so a receiver that is down or slow costs the site some freshness
266/// and costs the reporter nothing. The final failure names the report's number
267/// so a backfill knows exactly what to replay out of `bugs.jsonl`.
268pub fn forward(webhook: &BugReportWebhook, report: &BugReport, number: usize) {
269    let Ok(body) = serde_json::to_vec(&ForwardedReport { number, report }) else {
270        log::error!("bug report #{number} could not be encoded for forwarding");
271        return;
272    };
273    let url = webhook.url.clone();
274    let token = webhook.token.clone();
275
276    drop(tokio::spawn(async move {
277        let client = reqwest::Client::new();
278        for attempt in 1..=FORWARD_ATTEMPTS {
279            let mut request = client
280                .post(url.clone())
281                .header(CONTENT_TYPE, "application/json")
282                .timeout(FORWARD_TIMEOUT)
283                .body(body.clone());
284            if let Some(token) = &token {
285                request = request.bearer_auth(token);
286            }
287
288            match request.send().await {
289                Ok(response) if response.status().is_success() => return,
290                Ok(response) => {
291                    let status = response.status();
292                    // A refusal the receiver meant is not worth two more
293                    // rounds; only a server-side fault is likely to pass later.
294                    if !status.is_server_error() {
295                        log::warn!(
296                            "bug report #{number} was refused with {status}; it stays in \
297                             bugs.jsonl and can be backfilled"
298                        );
299                        return;
300                    }
301                    log::debug!("bug report #{number} attempt {attempt} got {status}");
302                }
303                Err(error) => {
304                    log::debug!("bug report #{number} attempt {attempt} failed: {error}");
305                }
306            }
307
308            if attempt < FORWARD_ATTEMPTS {
309                sleep(FORWARD_RETRY_DELAY).await;
310            }
311        }
312
313        log::warn!(
314            "bug report #{number} was not delivered after {FORWARD_ATTEMPTS} attempts; it is \
315             on disk and the site is behind by at least this one"
316        );
317    }));
318}

328    /// The payload is the record itself with its file number beside it.
329    ///
330    /// This is the contract whatever receives these reads, so it is pinned
331    /// rather than left to whatever `serde` happens to do. The number is what
332    /// makes a delivery identifiable: without it a receiver cannot tell a
333    /// retry from a second report with the same text, and a backfill has no
334    /// way to know where it left off.
335    #[test]
336    fn a_forwarded_report_carries_its_number_beside_the_record() {
337        let report = sample("the pig went through a wall");
338
339        let payload = serde_json::to_value(ForwardedReport {
340            number: 7,
341            report: &report,
342        })
343        .expect("a report encodes");
344        let object = payload.as_object().expect("the payload is one flat object");
345
346        assert_eq!(
347            object.get("number").and_then(serde_json::Value::as_u64),
348            Some(7),
349            "the number identifies the delivery"
350        );
351        for field in [
352            "at",
353            "player",
354            "uuid",
355            "world",
356            "position",
357            "category",
358            "description",
359            "version",
360        ] {
361            assert!(
362                object.contains_key(field),
363                "the record's `{field}` has to survive forwarding; the receiver has \
364                 nothing else to work from"
365            );
366        }
367        assert_eq!(
368            object.get("player").and_then(serde_json::Value::as_str),
369            Some("Tester"),
370            "the record is flattened beside the number, not nested under a key"
371        );
372    }

410    /// Two reports filed in a row are both kept, and numbered as they land.
411    ///
412    /// The obvious wrong implementation -- serialize the whole list and
413    /// rewrite the file -- passes a single-report test and silently drops
414    /// every report but the last as soon as two testers file at once.
415    #[test]
416    fn reports_accumulate_rather_than_replace_each_other() {
417        let dir = temp_dir().join(format!("foton-bugs-{}", process_id()));
418        let _ = fs::remove_dir_all(&dir);
419        fs::create_dir_all(&dir).expect("temp dir");
420
421        assert_eq!(sample("first").append_in(&dir).expect("append"), 1);
422        assert_eq!(sample("second").append_in(&dir).expect("append"), 2);
423
424        let all = BugReport::read_all(&dir).expect("read");
425        assert_eq!(all.len(), 2, "both reports have to survive");
426        assert_eq!(all[0].description, "first");
427        assert_eq!(all[1].description, "second");
428        assert!(
429            (all[0].position[0] - 1.23).abs() < 1.0e-9,
430            "coordinates keep two decimals: enough to walk back to the spot,              short enough to read in a listing"
431        );
432
433        let _ = fs::remove_dir_all(&dir);
434    }
