1//! Item modifiers: the loot functions `/item modify` runs on a stack.
2//!
3//! Vanilla parity: `ResourceOrIdArgument.lootModifier` and
4//! `LootItemFunctions.ROOT_CODEC`. A modifier is a datapack `item_modifier/*.json`
5//! file, or the same value written inline as SNBT; either way it is one
6//! function, or a list of them run in order.
7//!
8//! The loot engine in `foton_registry::loot_table` is built from `&'static`
9//! generated data, so a modifier read at run time cannot be expressed as
10//! `LootFunction` for every function. Each function the engine can already
11//! apply from owned data is decoded into it and run through
12//! [`LootFunction::apply`]; the rest are applied to the stack directly, with
13//! the same `ItemStack` calls the engine makes. A function outside the
14//! supported set, or any function carrying `conditions`, fails to decode and
15//! says so rather than being skipped.
16

32/// A decoded item modifier.
33#[derive(Debug, Clone)]
34pub struct ItemModifier {
35    functions: Vec<ModifierFunction>,
36}
37
38/// What a modifier needs from the command that runs it.
39///
40/// Vanilla parity: the `LootContextParamSets.COMMAND` params,
41/// `ORIGIN` and an optional `THIS_ENTITY`.
42pub struct ModifierContext<'a> {
43    /// Where the command runs, which `Score`-free providers never read but
44    /// functions such as `exploration_map` aim from.
45    pub origin: (f64, f64, f64),
46    /// The world the command runs in.
47    pub world: &'a dyn LootWorldView,
48}
49
50#[derive(Debug, Clone)]
51enum ModifierFunction {
52    /// A function the loot engine applies from owned data.
53    Loot(LootFunction),
54    /// `SetNameFunction`, whose component cannot live in the engine's `fn()`.
55    SetName {
56        name: Box<TextComponent>,
57        target: NameTarget,
58    },
59    /// `SetEnchantmentsFunction`, whose map the engine holds as `&'static`.
60    SetEnchantments {
61        enchantments: Vec<(Identifier, NumberProvider)>,
62        add: bool,
63    },
64    /// `SetCustomDataFunction`.
65    SetCustomData(NbtCompound),
66}
67
68impl ItemModifier {
69    /// Runs every function in order on `stack`.
70    ///
71    /// Vanilla parity: `ItemCommands.applyModifier`, which ends by limiting the
72    /// stack to its own maximum.
73    #[must_use]
74    pub fn apply(&self, mut stack: ItemStack, context: &ModifierContext<'_>) -> ItemStack {
75        let mut rng = rand::rng();
76        let mut loot = LootContext::new(&mut rng)
77            .with_world(context.world)
78            .with_origin(context.origin.0, context.origin.1, context.origin.2);
79        for function in &self.functions {
80            match function {
81                ModifierFunction::Loot(function) => function.apply(&mut stack, &mut loot),
82                ModifierFunction::SetName { name, target } => {
83                    stack.set_name(TextComponent::clone(name), *target);
84                }
85                ModifierFunction::SetEnchantments { enchantments, add } => {
86                    let resolved = enchantments
87                        .iter()
88                        .map(|(key, provider)| {
89                            let level = provider.get_int(loot.rng).max(0);
90                            (key.clone(), u32::try_from(level).unwrap_or(0))
91                        })
92                        .collect::<Vec<_>>();
93                    stack.set_enchantments(&resolved, *add);
94                }
95                ModifierFunction::SetCustomData(tag) => {
96                    if let Some(data) = CustomData::try_from_compound(tag.clone()) {
97                        stack.set_custom_data(&data);
98                    }
99                }
100            }
101        }
102        stack.count = stack.count.min(stack.max_stack_size());
103        stack
104    }
105}

107/// Reads a value's `NbtTag` as the JSON the codec decoder works on.
108///
109/// SNBT keeps the number types JSON loses; the decoder accepts any numeric
110/// type where the codec accepts a number, and `1b`/`0b` where it accepts a
111/// boolean, so nothing is lost by flattening here.
112fn nbt_to_json(tag: &NbtTag) -> serde_json::Value {
113    use serde_json::Value;
114    match tag {
115        NbtTag::Byte(value) => Value::from(*value),
116        NbtTag::Short(value) => Value::from(*value),
117        NbtTag::Int(value) => Value::from(*value),
118        NbtTag::Long(value) => Value::from(*value),
119        NbtTag::Float(value) => Value::from(f64::from(*value)),
120        NbtTag::Double(value) => Value::from(*value),
121        NbtTag::String(value) => Value::from(value.to_str().into_owned()),
122        NbtTag::Compound(compound) => Value::Object(
123            compound
124                .iter()
125                .map(|(key, value)| (key.to_str().into_owned(), nbt_to_json(value)))
126                .collect(),
127        ),
128        other => nbt_collection_values(other).map_or(Value::Null, |values| {
129            Value::Array(values.iter().map(nbt_to_json).collect())
130        }),
131    }
