1//! Choosing which enchantments an enchanting table offers, and rolling them.
2//!
3//! Vanilla parity: the selection half of `EnchantmentHelper` --
4//! `getEnchantmentCost`, `getAvailableEnchantmentResults`, `selectEnchantment`
5//! and `enchantItem`. Foton's [`crate::enchantment_helper`] covers the other
6//! half, applying enchantments once an item has them; nothing here had an
7//! implementation, which is why an enchanting table had nothing to offer.
8

20/// Most bookshelves an enchanting table counts.
21///
22/// Vanilla parity: the `if (bookcases > 15)` clamp of `getEnchantmentCost`.
23/// Fifteen is the ring of shelves around a table; a sixteenth adds nothing.
24pub const MAX_BOOKSHELVES: i32 = 15;
25
26/// Enchantment offers a table shows at once.
27pub const OFFER_COUNT: usize = 3;
28
29/// One enchantment at one level, with the weight it is drawn by.
30///
31/// Vanilla parity: `EnchantmentInstance`.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub struct EnchantmentInstance {
34    /// The enchantment itself.
35    pub enchantment: EnchantmentRef,
36    /// The level rolled for it.
37    pub level: u32,
38}
39
40/// Returns the experience level the given offer slot asks for.
41///
42/// Vanilla parity: `EnchantmentHelper.getEnchantmentCost`. The three slots read
43/// the same roll differently, which is why the top offer is always cheap and
44/// the bottom one only reaches thirty with a full ring of shelves.
45#[must_use]
46pub fn enchantment_cost(
47    random: &mut impl Random,
48    slot: usize,
49    bookshelves: i32,
50    item: &ItemStack,
51) -> i32 {
52    if item.get(ENCHANTABLE).is_none() {
53        return 0;
54    }
55
56    let bookshelves = bookshelves.min(MAX_BOOKSHELVES);
57    let selected = random.next_i32_bounded(8)
58        + 1
59        + (bookshelves >> 1)
60        + random.next_i32_bounded(bookshelves + 1);
61
62    match slot {
63        0 => (selected / 3).max(1),
64        1 => selected * 2 / 3 + 1,
65        _ => selected.max(bookshelves * 2),
66    }
67}

69/// The enchantments an enchanting table is allowed to roll.
70///
71/// Vanilla parity: `EnchantmentTags.IN_ENCHANTING_TABLE`, which the menu passes
72/// to `selectEnchantment` instead of the whole registry. It is why Mending and
73/// the curses never appear at a table however many shelves surround it.
74#[must_use]
75pub fn enchanting_table_candidates() -> &'static [EnchantmentRef] {
76    static CANDIDATES: LazyLock<Vec<EnchantmentRef>> = LazyLock::new(|| {
77        REGISTRY
78            .enchantments
79            .iter()
80            .map(|(_id, enchantment)| enchantment)
81            .filter(|enchantment| {
82                REGISTRY
83                    .enchantments
84                    .is_in_tag(*enchantment, &EnchantmentTag::IN_ENCHANTING_TABLE)
85            })
86            .collect()
87    });
88    &CANDIDATES
89}

91/// Returns every enchantment in `source` that could be rolled onto this item at
92/// `value`.
93///
94/// Vanilla parity: `EnchantmentHelper.getAvailableEnchantmentResults`. Only the
95/// highest level whose cost window contains `value` is offered per enchantment,
96/// which is what makes a high-level table offer Sharpness V rather than a
97/// choice between all five. The candidate set is the caller's, because a table
98/// draws from a narrower list than a loot table does.
99#[must_use]
100pub fn available_enchantment_results(
101    value: i32,
102    item: &ItemStack,
103    source: &[EnchantmentRef],
104) -> Vec<EnchantmentInstance> {
105    let is_book = item.is(&vanilla_items::BOOK);
106    let mut results = Vec::new();
107
108    for &enchantment in source {
109        if !is_book && !is_primary_item(enchantment, item) {
110            continue;
111        }
112
113        // Walk down from the top level and take the first that fits, exactly as
114        // vanilla does; the windows overlap, so walking up would offer level one
115        // of everything.
116        for level in (1..=enchantment.max_level).rev() {
117            if value >= min_cost(enchantment, level) && value <= max_cost(enchantment, level) {
118                results.push(EnchantmentInstance { enchantment, level });
119                break;
120            }
121        }
122    }
123
124    results
125}

127/// Rolls the enchantments one offer actually grants.
128///
129/// Vanilla parity: `EnchantmentHelper.selectEnchantment`. The cost is nudged by
130/// the item's enchantability and a small random span before anything is looked
131/// up, then extra enchantments are drawn with halving odds -- which is why a
132/// thirty-level offer on a golden item often lands three enchantments and the
133/// same offer on iron lands one.
134#[must_use]
135pub fn select_enchantment(
136    random: &mut impl Random,
137    item: &ItemStack,
138    cost: i32,
139    source: &[EnchantmentRef],
140) -> Vec<EnchantmentInstance> {
141    let mut results: Vec<EnchantmentInstance> = Vec::new();
142    let Some(enchantable) = item.get(ENCHANTABLE) else {
143        return results;
144    };
145
146    let quarter = enchantable.value() / 4 + 1;
147    let mut cost = cost + 1 + random.next_i32_bounded(quarter) + random.next_i32_bounded(quarter);
148
149    let span = (random.next_f32() + random.next_f32() - 1.0) * 0.15;
150    #[expect(
151        clippy::cast_possible_truncation,
152        reason = "vanilla rounds this to an int and the value is a small cost"
153    )]
154    let nudged = (cost as f32).mul_add(span, cost as f32).round() as i32;
155    cost = nudged.max(1);
156
157    let mut candidates = available_enchantment_results(cost, item, source);
158    if candidates.is_empty() {
159        return results;
160    }
161
162    if let Some(first) = draw_weighted(random, &candidates) {
163        results.push(first);
164    }
165
166    while random.next_i32_bounded(50) <= cost {
167        if let Some(last) = results.last().copied() {
168            retain_compatible(&mut candidates, last);
169        }
170        if candidates.is_empty() {
171            break;
172        }
173        if let Some(next) = draw_weighted(random, &candidates) {
174            results.push(next);
175        }
176        cost /= 2;
177    }
178
179    results
180}

182/// Writes an already-rolled offer onto an item.
183///
184/// Vanilla parity: the second half of `EnchantmentHelper.enchantItem`. A plain
185/// book becomes an enchanted book, which is the only case where the item itself
186/// changes -- and the only reason this is worth naming, because the enchanting
187/// table rolls its offer in two steps (once for the clue it shows before the
188/// click, once for the click) and must not re-implement the swap.
189#[must_use]
190pub fn apply_enchantments(item: &ItemStack, rolled: &[EnchantmentInstance]) -> ItemStack {
191    let mut result = if item.is(&vanilla_items::BOOK) {
192        // Vanilla parity: `itemStack.transmuteCopy(Items.ENCHANTED_BOOK)`, which
193        // carries the count and the component patch across. Building a bare
194        // stack instead returned a single blank book, so a named book lost its
195        // name and anything past the first book of a stack ceased to exist.
196        ItemStack::with_count_and_patch(
197            &vanilla_items::ENCHANTED_BOOK,
198            item.count(),
199            item.components_patch().clone(),
200        )
201    } else {
202        item.clone()
203    };
204
205    let applied: Vec<_> = rolled
206        .iter()
207        .map(|instance| (instance.enchantment.key.clone(), instance.level))
208        .collect();
209    result.set_enchantments(&applied, true);
210    result
211}

213/// Drops every candidate that clashes with one already chosen.
214///
215/// Vanilla parity: `EnchantmentHelper.filterCompatibleEnchantments`.
216fn retain_compatible(candidates: &mut Vec<EnchantmentInstance>, chosen: EnchantmentInstance) {
217    candidates
218        .retain(|candidate| Enchantment::are_compatible(chosen.enchantment, candidate.enchantment));
219}

221/// Picks one candidate, weighted by how common vanilla makes it.
222///
223/// Vanilla parity: `WeightedRandom.getRandomItem`.
224fn draw_weighted(
225    random: &mut impl Random,
226    candidates: &[EnchantmentInstance],
227) -> Option<EnchantmentInstance> {
228    let total: i32 = candidates
229        .iter()
230        .map(|candidate| i32::try_from(candidate.enchantment.weight).unwrap_or(i32::MAX))
231        .sum();
232    if total <= 0 {
233        return None;
234    }
235
236    let mut roll = random.next_i32_bounded(total);
237    for candidate in candidates {
238        roll -= i32::try_from(candidate.enchantment.weight).unwrap_or(i32::MAX);
239        if roll < 0 {
240            return Some(*candidate);
241        }
242    }
243    None
244}

246/// Returns whether an enchanting table may roll this enchantment onto this item.
247///
248/// Vanilla parity: `Enchantment.isPrimaryItem`. An enchantment with no primary
249/// set may go on anything it supports; one with a primary set is only rolled
250/// onto that narrower list, which is how Mending stays off the table entirely
251/// while still being applicable from a book.
252#[must_use]
253pub fn is_primary_item(enchantment: EnchantmentRef, item: &ItemStack) -> bool {
254    enchantment.can_enchant(item.item()) && enchantment.is_primary_item(item.item())
255}

257/// Vanilla parity: `Enchantment.getMinCost`.
258fn min_cost(enchantment: EnchantmentRef, level: u32) -> i32 {
259    let level = i32::try_from(level).unwrap_or(i32::MAX);
260    enchantment.min_cost.base + enchantment.min_cost.per_level_above_first * (level - 1)
261}
262
263/// Vanilla parity: `Enchantment.getMaxCost`.
264fn max_cost(enchantment: EnchantmentRef, level: u32) -> i32 {
265    let level = i32::try_from(level).unwrap_or(i32::MAX);
266    enchantment.max_cost.base + enchantment.max_cost.per_level_above_first * (level - 1)
267}

291    #[test]
292    fn the_bottom_offer_needs_a_full_ring_of_shelves() {
293        init_vanilla_registry();
294        let sword = diamond_sword();
295
296        // With no shelves the third offer cannot reach the thirty levels a
297        // player expects; with fifteen it always does, because the slot takes
298        // the greater of the roll and twice the shelf count.
299        let mut bare = LegacyRandom::from_seed(7);
300        let mut ringed = LegacyRandom::from_seed(7);
301        let without = enchantment_cost(&mut bare, 2, 0, &sword);
302        let with = enchantment_cost(&mut ringed, 2, MAX_BOOKSHELVES, &sword);
303
304        assert!(without < 30);
305        assert_eq!(with, 30);
306    }
