1//! A game-event listener that follows the entity carrying it.
2//!
3//! Vanilla parity: `net.minecraft.world.level.gameevent.DynamicGameEventListener`.
4//! A block's listener sits in one chunk section forever, so the chunk registry
5//! can own it; a mob's listener has to move between sections as the mob does,
6//! and this is the wrapper that re-registers it when it crosses a boundary.
7
8use std::sync::Arc;
9
10use foton_utils::SectionPos;
11use foton_utils::locks::SyncMutex;
12
13use super::listener::SharedGameEventListener;
14use crate::world::World;
15
16/// Which half of the listener's life this call is.
17///
18/// Vanilla parity: the three method references `ServerLevel` passes as its
19/// `BiConsumer<DynamicGameEventListener<?>, ServerLevel>` -- `add`, `remove`
20/// and `move`.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum DynamicListenerAction {
23    /// The entity joined the world.
24    Add,
25    /// The entity left it.
26    Remove,
27    /// The entity changed chunk section.
28    Move,
29}
30
31/// Keeps one listener registered in whichever section its source is in.
32pub struct DynamicGameEventListener {
33    listener: SharedGameEventListener,
34    last_section: SyncMutex<Option<SectionPos>>,
35}
36
37impl DynamicGameEventListener {
38    /// Wraps `listener` so a moving entity can carry it.
39    #[must_use]
40    pub fn new(listener: SharedGameEventListener) -> Self {
41        Self {
42            listener,
43            last_section: SyncMutex::new(None),
44        }
45    }
46
47    /// Applies one of the three lifecycle actions.
48    ///
49    /// Vanilla parity: `DynamicGameEventListener.add`, which is just `move`,
50    /// plus `remove` and `move` themselves.
51    pub fn apply(&self, action: DynamicListenerAction, world: &Arc<World>) {
52        match action {
53            DynamicListenerAction::Add | DynamicListenerAction::Move => self.move_to_current(world),
54            DynamicListenerAction::Remove => self.remove(world),
55        }
56    }
57
58    /// Vanilla parity: `DynamicGameEventListener.move`.
59    fn move_to_current(&self, world: &Arc<World>) {
60        let Some(position) = self.listener.listener_pos() else {
61            return;
62        };
63        let current_section = SectionPos::from_entity_pos(position);
64
65        let mut last_section = self.last_section.lock();
66        if *last_section == Some(current_section) {
67            return;
68        }
69        if let Some(previous) = *last_section {
70            world.unregister_game_event_listener(previous, &self.listener);
71        }
72        *last_section = Some(current_section);
73        world.register_game_event_listener(current_section, Arc::clone(&self.listener));
74    }
75
76    /// Vanilla parity: `DynamicGameEventListener.remove`.
77    fn remove(&self, world: &Arc<World>) {
78        let mut last_section = self.last_section.lock();
79        if let Some(previous) = last_section.take() {
80            world.unregister_game_event_listener(previous, &self.listener);
81        }
82    }
83}
