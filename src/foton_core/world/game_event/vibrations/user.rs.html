20/// Where a vibration listener sits, and where its particles fly to.
21///
22/// Vanilla parity: `PositionSource`, the level-side interface with `getPosition`. Foton's
23/// [`foton_registry::position_source::PositionSource`] is only that interface's network
24/// payload, so the two halves are separate types here.
25#[derive(Clone)]
26pub enum VibrationPositionSource {
27    /// Vanilla `BlockPositionSource`.
28    Block(BlockPos),
29    /// Vanilla `EntityPositionSource`.
30    Entity {
31        /// The world the entity is tracked in.
32        world: Weak<World>,
33        /// The entity's network id.
34        entity_id: i32,
35        /// How far above the entity's feet the listener sits.
36        y_offset: f32,
37    },
38}
39
40impl VibrationPositionSource {
41    /// Vanilla `PositionSource.getPosition`.
42    #[must_use]
43    pub fn resolve(&self) -> Option<DVec3> {
44        match self {
45            Self::Block(pos) => Some(DVec3::new(
46                f64::from(pos.x()) + 0.5,
47                f64::from(pos.y()) + 0.5,
48                f64::from(pos.z()) + 0.5,
49            )),
50            Self::Entity {
51                world,
52                entity_id,
53                y_offset,
54            } => {
55                let entity = world.upgrade()?.get_entity_by_id(*entity_id)?;
56                Some(entity.position() + DVec3::new(0.0, f64::from(*y_offset), 0.0))
57            }
58        }
59    }
60
61    /// Returns the payload the vibration particle carries to the client.
62    #[must_use]
63    pub fn to_particle_source(&self) -> ParticlePositionSource {
64        match self {
65            Self::Block(pos) => ParticlePositionSource::new(
66                &vanilla_position_source_types::BLOCK,
67                BlockPositionSource::new(*pos),
68            ),
69            Self::Entity {
70                entity_id,
71                y_offset,
72                ..
73            } => ParticlePositionSource::new(
74                &vanilla_position_source_types::ENTITY,
75                EntityPositionSource::new(*entity_id, *y_offset),
76            ),
77        }
78    }
79}

81/// Vanilla `VibrationSystem.User`.
82///
83/// One implementation per thing that hears vibrations: the two sculk sensors, the shrieker,
84/// the warden and the allay.
85pub trait VibrationUser: Send + Sync {
86    /// Vanilla `VibrationSystem.User.getListenerRadius`.
87    fn listener_radius(&self) -> i32;
88
89    /// Vanilla `VibrationSystem.User.getPositionSource`.
90    fn position_source(&self) -> VibrationPositionSource;
91
92    /// Vanilla `VibrationSystem.User.canReceiveVibration`.
93    fn can_receive_vibration(
94        &self,
95        world: &Arc<World>,
96        pos: BlockPos,
97        event: GameEventRef,
98        context: &GameEventContext<'_>,
99    ) -> bool;
100
101    /// Vanilla `VibrationSystem.User.onReceiveVibration`.
102    fn on_receive_vibration(
103        &self,
104        world: &Arc<World>,
105        pos: BlockPos,
106        event: GameEventRef,
107        source_entity: Option<&dyn Entity>,
108        projectile_owner: Option<&dyn Entity>,
109        receiving_distance: f32,
110    );
111
112    /// Vanilla `VibrationSystem.User.getListenableEvents`.
113    fn listenable_events(&self) -> Identifier {
114        GameEventTag::VIBRATIONS
115    }
116
117    /// Vanilla `VibrationSystem.User.canTriggerAvoidVibration`.
118    fn can_trigger_avoid_vibration(&self) -> bool {
119        false
120    }
121
122    /// Vanilla `VibrationSystem.User.requiresAdjacentChunksToBeTicking`.
123    fn requires_adjacent_chunks_to_be_ticking(&self) -> bool {
124        false
125    }
126
127    /// Vanilla `VibrationSystem.User.calculateTravelTimeInTicks`.
128    ///
129    /// This is the delay that makes a sensor feel like it is hearing something rather than
130    /// touching it: one tick per block between the source and the listener.
131    fn calculate_travel_time_in_ticks(&self, distance_to_destination: f32) -> i32 {
132        distance_to_destination.floor() as i32
133    }
134
135    /// Vanilla `VibrationSystem.User.isValidVibration`.
136    ///
137    /// Not implemented: the `AVOID_VIBRATION` advancement trigger a sneaking player earns
138    /// here, because Foton has no advancement criteria system.
139    fn is_valid_vibration(&self, event: GameEventRef, context: &GameEventContext<'_>) -> bool {
140        if !REGISTRY
141            .game_events
142            .is_in_tag(event, &self.listenable_events())
143        {
144            return false;
145        }
146
147        if let Some(source_entity) = context.source_entity() {
148            if source_entity.is_spectator() {
149                return false;
150            }
151            if source_entity.is_stepping_carefully()
152                && REGISTRY
153                    .game_events
154                    .is_in_tag(event, &GameEventTag::IGNORE_VIBRATIONS_SNEAKING)
155            {
156                return false;
157            }
158            if source_entity.dampens_vibrations() {
159                return false;
160            }
161        }
162
163        context.affected_state().is_none_or(|state| {
164            !REGISTRY
165                .blocks
166                .is_in_tag(state.get_block(), &BlockTag::DAMPENS_VIBRATIONS)
167        })
168    }
169
170    /// Vanilla `VibrationSystem.User.onDataChanged`.
171    fn on_data_changed(&self) {}
172}
