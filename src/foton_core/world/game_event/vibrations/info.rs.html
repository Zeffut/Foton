1//! One vibration, from the moment it is scheduled to the moment it arrives.

3use std::sync::Arc;
4
5use foton_registry::game_events::GameEventRef;
6use foton_registry::{REGISTRY, RegistryExt as _};
7use foton_utils::{Identifier, UuidExt as _};
8use glam::DVec3;
9use simdnbt::borrow::NbtCompound as NbtCompoundView;
10use simdnbt::owned::{NbtCompound, NbtList, NbtTag};
11use uuid::Uuid;
12
13use crate::entity::{Entity, SharedEntity};
14use crate::world::World;
15
16/// Vanilla `VibrationInfo`.
17///
18/// Deviation: vanilla's record keeps a nullable hard reference to the source entity beside
19/// its UUID so a vibration still names its source after the entity left the level. Foton
20/// keeps the same pair -- a [`SharedEntity`] and the UUID -- because the entity handle is
21/// reference counted and the UUID is what survives the save.
22#[derive(Clone)]
23pub struct VibrationInfo {
24    game_event: GameEventRef,
25    distance: f32,
26    pos: DVec3,
27    uuid: Option<Uuid>,
28    projectile_owner_uuid: Option<Uuid>,
29    entity: Option<SharedEntity>,
30}
31
32impl VibrationInfo {
33    /// Vanilla `VibrationInfo(Holder<GameEvent>, float, Vec3, Entity)`.
34    ///
35    /// The world is only used to promote the borrowed source entity to a handle the
36    /// vibration can carry until it arrives, which is what vanilla's direct field does.
37    #[must_use]
38    pub fn new(
39        game_event: GameEventRef,
40        distance: f32,
41        pos: DVec3,
42        source_entity: Option<&dyn Entity>,
43        world: &Arc<World>,
44    ) -> Self {
45        Self {
46            game_event,
47            distance,
48            pos,
49            uuid: source_entity.map(Entity::uuid),
50            projectile_owner_uuid: source_entity
51                .and_then(|entity| entity.as_projectile()?.get_owner())
52                .map(|owner| owner.uuid()),
53            entity: source_entity.and_then(|entity| world.get_entity_by_id(entity.id())),
54        }
55    }
56
57    /// Returns the game event that produced this vibration.
58    #[must_use]
59    pub const fn game_event(&self) -> GameEventRef {
60        self.game_event
61    }
62
63    /// Returns how far the vibration has to travel, in blocks.
64    #[must_use]
65    pub const fn distance(&self) -> f32 {
66        self.distance
67    }
68
69    /// Returns where the vibration started.
70    #[must_use]
71    pub const fn pos(&self) -> DVec3 {
72        self.pos
73    }
74
75    /// Vanilla `VibrationInfo.getEntity`.
76    #[must_use]
77    pub fn get_entity(&self, world: &Arc<World>) -> Option<SharedEntity> {
78        self.entity
79            .clone()
80            .or_else(|| world.get_entity_by_uuid(&self.uuid?))
81    }
82
83    /// Vanilla `VibrationInfo.getProjectileOwner`.
84    #[must_use]
85    pub fn get_projectile_owner(&self, world: &Arc<World>) -> Option<SharedEntity> {
86        self.get_entity(world)
87            .and_then(|entity| entity.as_projectile()?.get_owner())
88            .or_else(|| world.get_entity_by_uuid(&self.projectile_owner_uuid?))
89    }
90
91    /// Writes vanilla's `VibrationInfo.CODEC` shape.
92    pub fn save(&self, nbt: &mut NbtCompound) {
93        nbt.insert("game_event", self.game_event.key.to_string());
94        nbt.insert("distance", self.distance);
95        nbt.insert(
96            "pos",
97            NbtTag::List(NbtList::Double(vec![self.pos.x, self.pos.y, self.pos.z])),
98        );
99        if let Some(uuid) = self.uuid {
100            nbt.insert("source", NbtTag::IntArray(uuid.to_int_array().to_vec()));
101        }
102        if let Some(uuid) = self.projectile_owner_uuid {
103            nbt.insert(
104                "projectile_owner",
105                NbtTag::IntArray(uuid.to_int_array().to_vec()),
106            );
107        }
108    }
109
110    /// Reads vanilla's `VibrationInfo.CODEC` shape.
111    ///
112    /// The entity handle is deliberately absent: a reloaded vibration only has the UUID,
113    /// exactly as in vanilla, and resolves its entity through the level.
114    #[must_use]
115    pub fn load(nbt: &NbtCompoundView<'_, '_>) -> Option<Self> {
116        let game_event = nbt
117            .string("game_event")
118            .and_then(|key| key.to_str().parse::<Identifier>().ok())
119            .and_then(|key| REGISTRY.game_events.by_key(&key))?;
120        let distance = nbt.float("distance")?;
121        if distance < 0.0 {
122            return None;
123        }
124        let pos = nbt.list("pos").and_then(|list| list.doubles())?;
125        let [x, y, z] = pos[..] else { return None };
126
127        Some(Self {
128            game_event,
129            distance,
130            pos: DVec3::new(x, y, z),
131            uuid: nbt
132                .int_array("source")
133                .and_then(|values| Uuid::from_int_array(&values)),
134            projectile_owner_uuid: nbt
135                .int_array("projectile_owner")
136                .and_then(|values| Uuid::from_int_array(&values)),
137            entity: None,
138        })
139    }
140}
