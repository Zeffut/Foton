1//! Where a mob is allowed to appear.
2//!
3//! Vanilla parity: `SpawnPlacementTypes`, plus the per-entity-type table that
4//! `SpawnPlacements` builds in code rather than in data. Nearly every mob wants
5//! solid ground, so that is the default; the exceptions are listed here exactly
6//! as vanilla registers them, because they cannot be derived from the mob
7//! category. A drowned is a monster and spawns in water, a strider is a monster
8//! and spawns in lava, and a phantom is a monster that spawns in mid-air.
9

27/// Somewhere the spawn rules can read blocks from.
28///
29/// Vanilla passes a `LevelReader`, which a chunk still being generated
30/// satisfies just as well as a running world. Foton's `World` and its
31/// `WorldGenRegion` share no such trait, so the placement tests take this
32/// instead and both spawners get the same answers from the same code.
33pub trait SpawnBlockSource {
34    /// Returns the block at `pos`.
35    fn spawn_block_state(&self, pos: BlockPos) -> BlockStateId;
36}
37
38impl SpawnBlockSource for Arc<World> {
39    fn spawn_block_state(&self, pos: BlockPos) -> BlockStateId {
40        self.get_block_state(pos)
41    }
42}

44/// Brightest a block may glow and still be spawned on.

46/// Vanilla parity: the `getLightEmission() < 14` half of the default
47/// `BlockBehaviour.Properties.isValidSpawn`.
48const MAX_SPAWNABLE_LIGHT_EMISSION: u8 = 14;

67    /// Standing on a block, with room to stand.

70    OnGround,

131/// Returns where this kind of mob needs to be for a natural spawn.
132///
133/// Vanilla parity: `SpawnPlacements.getPlacementType`. Vanilla returns
134/// `ON_GROUND` for anything it was never told about, and so does this.
135#[must_use]
136pub fn spawn_placement_for(entity_type: EntityTypeRef) -> SpawnPlacementType {
137    PLACEMENT_EXCEPTIONS
138        .iter()
139        .find(|(candidate, _)| ptr::eq(*candidate, entity_type))
140        .map_or(SpawnPlacementType::OnGround, |(_, placement)| *placement)
141}

143impl SpawnPlacementType {
144    /// Returns whether a mob of `entity_type` fits at `pos`.
145    ///
146    /// Vanilla parity: `SpawnPlacementType.isSpawnPositionOk`. The world-border
147    /// test vanilla runs first is left to the caller, which already has to
148    /// reject positions for being too near or too far from a player.
149    #[must_use]
150    pub fn is_spawn_position_ok(
151        self,
152        level: &impl SpawnBlockSource,
153        pos: BlockPos,
154        entity_type: EntityTypeRef,
155    ) -> bool {
156        match self {
157            Self::NoRestrictions => true,
158            Self::InWater => {
159                let above = pos.above();
160                is_water_fluid(level.spawn_block_state(pos).get_fluid_state().fluid_id)
161                    && !is_redstone_conductor(level, above)
162            }
163            Self::InLava => is_lava_fluid(level.spawn_block_state(pos).get_fluid_state().fluid_id),
164            Self::OnGround => {
165                let below = pos.below();
166                is_valid_spawn_block(level, below)
167                    && is_valid_empty_spawn_block(level, pos, entity_type)
168                    && is_valid_empty_spawn_block(level, pos.above(), entity_type)
169            }
170        }
171    }
172
173    /// Nudges a candidate position down onto the block a mob would stand on.
174    ///
175    /// Vanilla parity: `SpawnPlacementType.adjustSpawnPosition`. Only ground
176    /// spawns move: the heightmap points at the first free block, and vanilla
177    /// steps back down into it when a mob could walk there.
178    #[must_use]
179    pub fn adjust_spawn_position(
180        self,
181        level: &impl SpawnBlockSource,
182        candidate: BlockPos,
183    ) -> BlockPos {
184        if self != Self::OnGround {
185            return candidate;
186        }
187        let below = candidate.below();
188        if level
189            .spawn_block_state(below)
190            .is_pathfindable(PathComputationType::Land)
191        {
192            below
193        } else {
194            candidate
195        }
196    }
197}

199/// Returns whether a block can be stood on by a spawning mob.
200///
201/// Vanilla parity: the default `BlockBehaviour.Properties.isValidSpawn`.
202///
203/// TODO: the handful of blocks that override `isValidSpawn` -- magma, nether
204/// wart blocks, the ones that only take fire-immune mobs -- still answer with
205/// the default here, because Foton's block behaviors have no such hook yet.
206pub(crate) fn is_valid_spawn_block(level: &impl SpawnBlockSource, pos: BlockPos) -> bool {
207    let state = level.spawn_block_state(pos);
208    state.is_face_sturdy_at(pos, Direction::Up)
209        && state.get_light_emission() < MAX_SPAWNABLE_LIGHT_EMISSION
210}

212/// Returns whether a mob may occupy this block.
213///
214/// Vanilla parity: `NaturalSpawner.isValidEmptySpawnBlock`.
215fn is_valid_empty_spawn_block(
216    level: &impl SpawnBlockSource,
217    pos: BlockPos,
218    entity_type: EntityTypeRef,
219) -> bool {
220    let state = level.spawn_block_state(pos);
221    if is_collision_shape_full_block(state) {
222        return false;
223    }
224    if is_signal_source(state) {
225        return false;
226    }
227    if !state.get_fluid_state().fluid_id.is_empty {
228        return false;
229    }
230    if state
231        .get_block()
232        .has_tag(&BlockTag::PREVENT_MOB_SPAWNING_INSIDE)
233    {
234        return false;
235    }
236    !is_block_dangerous_for(entity_type.fire_immune, state)
237}

239/// Returns whether the block at `pos` blocks redstone the way a full block does.
240///
241/// Vanilla parity: the default `BlockBehaviour.Properties.isRedstoneConductor`,
242/// which is `isCollisionShapeFullBlock`.
243fn is_redstone_conductor(level: &impl SpawnBlockSource, pos: BlockPos) -> bool {
244    is_collision_shape_full_block(level.spawn_block_state(pos))
245}
246
247/// Returns whether a block fills its cube for collision.
248///
249/// Vanilla parity: `BlockStateBase.isCollisionShapeFullBlock`, which vanilla
250/// caches on the state; the static shape is that cached answer, so spawning
251/// asks the state rather than the level.
252fn is_collision_shape_full_block(state: BlockStateId) -> bool {
253    is_shape_full_block(state.get_static_collision_shape())
254}
255
256/// Returns whether a block emits redstone on its own.
257///
258/// Vanilla parity: `BlockState.isSignalSource`, which natural spawning uses to
259/// keep mobs out of a running circuit. Foton keeps that answer on the block
260/// behavior rather than on the state, so it is asked here with the same default
261/// context the rest of the redstone code uses.
262fn is_signal_source(state: BlockStateId) -> bool {
263    BLOCK_BEHAVIORS
264        .get_behavior(state.get_block())
265        .is_signal_source(state, SignalQueryContext::DEFAULT)
266}

268/// Returns whether a block would hurt a mob that spawned in it.
269///
270/// Vanilla parity: `EntityType.isBlockDangerous`, keyed by the entity type
271/// rather than by a live entity because natural spawning tests the position
272/// before anything is created.
273pub(crate) fn is_block_dangerous_for(fire_immune: bool, state: BlockStateId) -> bool {
274    // TODO: mirror vanilla EntityType.immuneTo once entity types carry their
275    // entity-specific immune block tag.
276    if !fire_immune && WalkPathEvaluator::is_burning_block(state) {
277        return true;
278    }
279
280    let block = state.get_block();
281    block == &vanilla_blocks::WITHER_ROSE
282        || block == &vanilla_blocks::SWEET_BERRY_BUSH
283        || block == &vanilla_blocks::CACTUS
284        || block == &vanilla_blocks::POWDER_SNOW
285}

287#[cfg(test)]
288mod tests {
289    use super::*;

291    #[test]
292    fn unlisted_mobs_spawn_on_the_ground() {
293        foton_registry::init_vanilla_registry();
294        assert_eq!(
295            spawn_placement_for(&vanilla_entities::COW),
296            SpawnPlacementType::OnGround
297        );
298        assert_eq!(
299            spawn_placement_for(&vanilla_entities::ZOMBIE),
300            SpawnPlacementType::OnGround
301        );
302    }
303
304    #[test]
305    fn water_mobs_are_not_derived_from_their_category() {
306        foton_registry::init_vanilla_registry();
307
308        // A drowned is a monster, and a squid is a water creature. Both spawn in
309        // water, which is why the table is explicit rather than category-driven.
310        assert_eq!(
311            spawn_placement_for(&vanilla_entities::DROWNED),
312            SpawnPlacementType::InWater
313        );
314        assert_eq!(
315            spawn_placement_for(&vanilla_entities::SQUID),
316            SpawnPlacementType::InWater
317        );
318    }
319
320    #[test]
321    fn striders_spawn_in_lava_and_phantoms_anywhere() {
322        foton_registry::init_vanilla_registry();
323        assert_eq!(
324            spawn_placement_for(&vanilla_entities::STRIDER),
325            SpawnPlacementType::InLava
326        );
327        assert_eq!(
328            spawn_placement_for(&vanilla_entities::PHANTOM),
329            SpawnPlacementType::NoRestrictions
330        );
331    }
332
333    #[test]
334    fn fire_burns_anything_not_immune_to_it() {
335        foton_registry::init_vanilla_registry();
336        let fire = vanilla_blocks::FIRE.default_state();
337        assert!(is_block_dangerous_for(false, fire));
338        assert!(!is_block_dangerous_for(true, fire));
339    }
340
341    #[test]
342    fn cactus_hurts_even_a_fire_immune_mob() {
343        foton_registry::init_vanilla_registry();
344        let cactus = vanilla_blocks::CACTUS.default_state();
345        assert!(is_block_dangerous_for(true, cactus));
346    }
