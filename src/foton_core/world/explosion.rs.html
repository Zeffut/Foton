29/// How far a player can be from a blast and still be told about it.
30///
31/// Vanilla parity: the `distanceToSqr(center) < 4096.0` of `ServerLevel.explode`.
32const EXPLOSION_PACKET_RANGE_SQ: f64 = 4096.0;
33
34/// What an explosion does to the blocks it reaches.
35///
36/// Vanilla parity: `Explosion.BlockInteraction`.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub enum ExplosionBlockInteraction {
39    /// Leaves every block standing.
40    Keep,
41    /// Breaks blocks and drops all of their loot.
42    Destroy,
43    /// Breaks blocks and thins their loot by the blast radius.
44    ///
45    /// The only thing this changes is that the block loot roll receives the
46    /// explosion radius, which is what `survives_explosion` and
47    /// `explosion_decay` read. Which of the two destroying modes a blast uses
48    /// is decided by the matching `*_explosion_drop_decay` game rule -- see
49    /// [`World::explosion_destroy_type`].
50    DestroyWithDecay,
51}
52
53impl ExplosionBlockInteraction {
54    /// Whether this interaction breaks the blocks the blast reaches.
55    #[must_use]
56    pub const fn destroys_blocks(self) -> bool {
57        matches!(self, Self::Destroy | Self::DestroyWithDecay)
58    }
59
60    /// The blast radius handed to the block loot roll, if drops decay.
61    #[must_use]
62    const fn loot_explosion_radius(self, radius: f32) -> Option<f32> {
63        match self {
64            Self::DestroyWithDecay => Some(radius),
65            Self::Keep | Self::Destroy => None,
66        }
67    }
68}

70/// Rays are cast from a 16x16x16 grid, using only its outer shell.
71const RAY_GRID_SIZE: i32 = 16;
72/// Distance each ray advances per step.
73const RAY_STEP: f32 = 0.3;
74/// Power every ray loses per step regardless of what it crosses.
75const RAY_STEP_COST: f32 = 0.225_000_01;

77/// An explosion, apart from where it lands.
78///
79/// Both [`World::explode`] and [`World::explode_sparing`] take one. It used to
80/// be the sparing variant's alone, with the plain one keeping its arguments
81/// loose; splitting the source entity into vanilla's direct/causing pair
82/// pushed that list past what one call can carry legibly.
83pub struct ExplosionSpec {
84    /// The entity the blast came out of, if any: the TNT, the fireball, the
85    /// creeper.
86    ///
87    /// Vanilla parity: `Explosion.getDirectSourceEntity`.
88    pub direct_entity_id: Option<i32>,
89    /// Who is ultimately to blame, if anyone: the player who lit the TNT, the
90    /// ghast that spat the fireball, or the creeper itself.
91    ///
92    /// Vanilla parity: `Explosion.getIndirectSourceEntity`, which is a switch
93    /// over the source's class. Foton has no class hierarchy to switch on, so
94    /// each caller passes its own answer -- a primed TNT its owner, a
95    /// projectile its living shooter, a living entity itself.
96    pub causing_entity_id: Option<i32>,
97    /// What the blast hurts entities with.
98    pub damage_source: Option<DamageSource>,
99    /// How far it reaches.
100    pub radius: f32,
101    /// Whether it leaves fires behind.
102    pub fire: bool,
103    /// What it does to blocks.
104    pub interaction: ExplosionBlockInteraction,
105    /// Whether the blast hurts what it reaches.
106    ///
107    /// Vanilla parity: the `damagesEntities` of
108    /// `SimpleExplosionDamageCalculator`. A wind charge is the reason this
109    /// exists: it shoves everything around it and hurts none of it.
110    pub damages_entities: bool,
111    /// How hard the blast shoves, against an ordinary explosion.
112    ///
113    /// Vanilla parity: the `knockbackMultiplier` of the same calculator.
114    pub knockback_multiplier: f64,
115    /// The emitter drawn at the center of a blast the client calls small.
116    ///
117    /// Vanilla parity: the `smallExplosionParticles` argument of
118    /// `Level.explode`.
119    pub small_particle: ParticleData,
120    /// The emitter drawn at the center of every other blast.
121    ///
122    /// Vanilla parity: the `largeExplosionParticles` argument.
123    pub large_particle: ParticleData,
124    /// The debris the broken blocks throw off.
125    ///
126    /// Vanilla parity: the `blockParticles` argument.
127    pub block_particles: WeightedList<ExplosionParticleInfo>,
128    /// The sound the blast makes.
129    ///
130    /// Vanilla parity: the `explosionSound` argument. It travels inside the
131    /// explosion packet rather than an ordinary sound packet, so it is the
132    /// packet or nothing.
133    pub sound: SoundEventRef,
134}

256    /// Detonates an explosion that leaves some blocks standing.
257    ///
258    /// Vanilla parity: the `Explosion.shouldBlockExplode` hook, which the
259    /// source entity answers. Only the TNT minecart uses it, and only to spare
260    /// the rails it was running on -- otherwise one cart would take the track
261    /// with it and no cart cannon would work twice.
262    pub fn explode_sparing(
263        self: &Arc<Self>,
264        spec: ExplosionSpec,
265        center: DVec3,
266        should_explode: &dyn Fn(BlockPos) -> bool,
267    ) -> Vec<BlockPos> {
268        let mut to_blow = self.calculate_exploded_positions(center, spec.radius);
269        to_blow.retain(|pos| should_explode(*pos));
270        if spec.direct_entity_id.is_some() {
271            let entity = spec
272                .direct_entity_id
273                .and_then(|id| self.get_entity_by_id(id))
274                .map(|entity| entity.uuid());
275            let mut event = EntityExplodeEvent::new(
276                entity,
277                self.key.to_string(),
278                to_blow.clone(),
279                1.0,
280                match spec.interaction {
281                    ExplosionBlockInteraction::Keep => "KEEP",
282                    ExplosionBlockInteraction::Destroy => "BLOCK",
283                    ExplosionBlockInteraction::DestroyWithDecay => "DESTROY",
284                },
285            );
286            self.fire_event(&mut event);
287            if event.is_cancelled() {
288                return Vec::new();
289            }
290            to_blow = event.blocks().to_vec();
291        } else {
292            let source = BlockPos::new(
293                center.x.floor() as i32,
294                center.y.floor() as i32,
295                center.z.floor() as i32,
296            );
297            let mut event = BlockExplodeEvent::new(self.key.to_string(), source, to_blow.clone());
298            self.fire_event(&mut event);
299            if event.is_cancelled() {
300                return Vec::new();
301            }
302            to_blow = event.blocks().to_vec();
303        }
304        let hit_players = self.hurt_entities_from_explosion(&spec, center);
305
306        if spec.interaction.destroys_blocks() {
307            let loot_radius = spec.interaction.loot_explosion_radius(spec.radius);
308            // `ServerExplosion.source` is the direct entity, not the one to
309            // blame: a fireball's drops are the fireball's context, even
310            // though the ghast is what answers for the damage.
311            let source_entity = spec
312                .direct_entity_id
313                .and_then(|entity_id| self.get_entity_by_id(entity_id));
314            for pos in &to_blow {
315                self.destroy_block_from_explosion(
316                    *pos,
317                    source_entity.as_deref(),
318                    loot_radius,
319                    spec.causing_entity_id,
320                );
321            }
322        }
323
324        if spec.fire {
325            self.create_explosion_fire(&to_blow);
326        }
327
328        self.send_explosion_packets(&spec, center, to_blow.len(), &hit_players);
329
330        to_blow
331    }

333    /// Tells every player near the blast that it happened, and each pushed one
334    /// how hard it pushed them.
335    ///
336    /// Vanilla parity: the loop at the end of `ServerLevel.explode`. This is the
337    /// only channel by which an explosion reaches a client at all -- its sound,
338    /// its emitter and its debris all ride here -- and the only one by which it
339    /// moves a player. A player is authoritative over their own position, so
340    /// the server's `set_velocity` above is a value the player never feels;
341    /// `ClientPacketListener.handleExplosion` applying `playerKnockback` to the
342    /// local player is what actually launches them.
343    fn send_explosion_packets(
344        &self,
345        spec: &ExplosionSpec,
346        center: DVec3,
347        block_count: usize,
348        hit_players: &[(i32, DVec3)],
349    ) {
350        let block_count = i32::try_from(block_count).unwrap_or(i32::MAX);
351        let particle = spec.explosion_particle();
352        self.players.iter_players(|_, player| {
353            if player.position().distance_squared(center) >= EXPLOSION_PACKET_RANGE_SQ {
354                return true;
355            }
356            let knockback = hit_players
357                .iter()
358                .find(|(entity_id, _)| *entity_id == player.id())
359                .map(|(_, knockback)| *knockback);
360            player.send_packet(CExplode::new(
361                center,
362                spec.radius,
363                block_count,
364                knockback,
365                particle.clone(),
366                spec.sound,
367                spec.block_particles.clone(),
368            ));
369            true
370        });
371    }

373    /// Casts the rays that decide which blocks the blast breaks.
374    ///
375    /// Vanilla parity: `ServerExplosion.calculateExplodedPositions`.
376    fn calculate_exploded_positions(self: &Arc<Self>, center: DVec3, radius: f32) -> Vec<BlockPos> {
377        let mut to_blow = Vec::new();
378        let last = RAY_GRID_SIZE - 1;
379
380        for x in 0..RAY_GRID_SIZE {
381            for y in 0..RAY_GRID_SIZE {
382                for z in 0..RAY_GRID_SIZE {
383                    // Only the shell of the grid produces rays; the interior would
384                    // duplicate directions already covered.
385                    if x != 0 && x != last && y != 0 && y != last && z != 0 && z != last {
386                        continue;
387                    }
388
389                    let mut dx = f64::from(x) / f64::from(last) * 2.0 - 1.0;
390                    let mut dy = f64::from(y) / f64::from(last) * 2.0 - 1.0;
391                    let mut dz = f64::from(z) / f64::from(last) * 2.0 - 1.0;
392                    let length = dx.mul_add(dx, dy.mul_add(dy, dz * dz)).sqrt();
393                    if length == 0.0 {
394                        continue;
395                    }
396                    dx /= length;
397                    dy /= length;
398                    dz /= length;
399
400                    // Vanilla varies each ray's power between 70% and 130%.
401                    let mut power = radius * 0.6f32.mul_add(rand::random::<f32>(), 0.7);
402                    let mut position = center;
403
404                    while power > 0.0 {
405                        let pos = BlockPos::new(
406                            position.x.floor() as i32,
407                            position.y.floor() as i32,
408                            position.z.floor() as i32,
409                        );
410                        if self.is_outside_build_height(pos.y()) {
411                            break;
412                        }
413
414                        let state = self.get_block_state(pos);
415                        let resistance = self.explosion_resistance_at(pos, state);
416                        if let Some(resistance) = resistance {
417                            power -= (resistance + 0.3) * RAY_STEP;
418                        }
419
420                        if power > 0.0 && !to_blow.contains(&pos) {
421                            to_blow.push(pos);
422                        }
423
424                        position.x += dx * f64::from(RAY_STEP);
425                        position.y += dy * f64::from(RAY_STEP);
426                        position.z += dz * f64::from(RAY_STEP);
427                        power -= RAY_STEP_COST;
428                    }
429                }
430            }
431        }
432
433        to_blow
434    }

436    /// Returns the resistance a block and its fluid oppose to a blast.
437    ///
438    /// Vanilla parity: `ExplosionDamageCalculator.getBlockExplosionResistance`.
439    /// Air with no fluid offers nothing and costs the ray only its step price.
440    fn explosion_resistance_at(
441        self: &Arc<Self>,
442        pos: BlockPos,
443        state: BlockStateId,
444    ) -> Option<f32> {
445        let fluid = get_fluid_state(self, pos);
446        if state.is_air() && fluid.is_empty() {
447            return None;
448        }
449        let block_resistance = state.get_block().config.explosion_resistance;
450        let fluid_resistance = fluid.fluid_id.explosion_resistance;
451        Some(block_resistance.max(fluid_resistance))
452    }

454    /// Damages and pushes every entity the blast reaches, and returns the push
455    /// each player took.
456    ///
457    /// Vanilla parity: `ServerExplosion.hurtEntities`, whose `hitPlayers` map
458    /// exists for exactly one reason: a player's own client has to be handed
459    /// the push, because the server's own copy of their velocity is not
460    /// something they can feel.
461    fn hurt_entities_from_explosion(
462        self: &Arc<Self>,
463        spec: &ExplosionSpec,
464        center: DVec3,
465    ) -> Vec<(i32, DVec3)> {
466        let mut hit_players = Vec::new();
467        let (direct_entity_id, causing_entity_id) = (spec.direct_entity_id, spec.causing_entity_id);
468        let radius = spec.radius;
469        if radius < 1.0e-5 {
470            return hit_players;
471        }
472        let reach = f64::from(radius) * 2.0;
473        let aabb = WorldAabb::new(
474            center.x - reach - 1.0,
475            center.y - reach - 1.0,
476            center.z - reach - 1.0,
477            center.x + reach + 1.0,
478            center.y + reach + 1.0,
479            center.z + reach + 1.0,
480        );
481
482        let mut damage_source = spec
483            .damage_source
484            .clone()
485            .unwrap_or_else(|| default_explosion_damage_source(direct_entity_id, causing_entity_id))
486            .with_source_position(center);
487        if let Some(id) = direct_entity_id {
488            damage_source = damage_source.with_direct_entity(id);
489        }
490        if let Some(id) = causing_entity_id {
491            damage_source = damage_source.with_causing_entity(id);
492        }
493
494        // Vanilla hands `onExplosionHit` the whole source entity, because
495        // `ServerPlayer` needs its type to tell a wind charge from anything
496        // else. Resolved once rather than per affected entity.
497        let source_entity = direct_entity_id.and_then(|id| self.get_entity_by_id(id));
498
499        for entity in self.get_entities_in_aabb(&aabb) {
500            // Vanilla parity: `hurtEntities` asks the level for everything
501            // *except* the source, so a blast never hits what it came out of.
502            if Some(entity.id()) == direct_entity_id {
503                continue;
504            }
505            let distance = entity.position().distance(center) / reach;
506            if distance > 1.0 {
507                continue;
508            }
509
510            let exposure = f64::from(self.seen_percent(center, entity.as_ref()));
511
512            // Vanilla parity: `ExplosionDamageCalculator.getEntityDamageAmount`.
513            let impact = (1.0 - distance) * exposure;
514            if spec.damages_entities {
515                let damage = (impact.mul_add(impact, impact) / 2.0).mul_add(7.0 * reach, 1.0);
516                entity.hurt(self, &damage_source, damage as f32);
517            }
518
519            // Vanilla pushes from the blast toward the entity's eyes.
520            let origin = entity.position().with_y(entity.get_eye_y());
521            let direction = origin - center;
522            if direction.length_squared() > 0.0 {
523                let resistance = entity.as_living_entity().map_or(0.0, |living| {
524                    living
525                        .attributes()
526                        .lock()
527                        .required_value(vanilla_attributes::EXPLOSION_KNOCKBACK_RESISTANCE)
528                });
529                let power = impact * (1.0 - resistance) * spec.knockback_multiplier;
530                let knockback = direction.normalize() * power;
531                entity.set_velocity(entity.velocity() + knockback);
532                entity.mark_velocity_sync();
533
534                // Vanilla parity: the `hitPlayers.put` branch. A spectator is
535                // not pushed at all, and neither is a player flying in
536                // creative -- their client would fight the shove.
537                if let Some(player) = entity.as_player()
538                    && !player.is_spectator()
539                    && !(player.has_infinite_materials() && player.is_flying())
540                {
541                    hit_players.push((player.id(), knockback));
542                }
543            }
544
545            // Vanilla parity: the `entity.onExplosionHit(this.source)` that
546            // closes the per-entity body of `hurtEntities`. Vanilla runs it
547            // for every entity the blast reached, whether or not it moved them.
548            entity.on_explosion_hit(source_entity.as_ref());
549        }
550
551        hit_players
552    }

554    /// Returns the fraction of an entity the blast can see, from 0 to 1.
555    ///
556    /// Vanilla parity: `ServerExplosion.getSeenPercent`. Vanilla samples a grid of
557    /// points across the entity's bounding box and counts how many reach the center
558    /// without crossing a block.
559    fn seen_percent(&self, center: DVec3, entity: &dyn Entity) -> f32 {
560        let bb = entity.bounding_box();
561        let step_x = 1.0 / ((bb.max_x() - bb.min_x()) * 2.0 + 1.0);
562        let step_y = 1.0 / ((bb.max_y() - bb.min_y()) * 2.0 + 1.0);
563        let step_z = 1.0 / ((bb.max_z() - bb.min_z()) * 2.0 + 1.0);
564        if step_x < 0.0 || step_y < 0.0 || step_z < 0.0 {
565            return 0.0;
566        }
567
568        let offset_x = (1.0 - (1.0 / step_x).floor() * step_x) / 2.0;
569        let offset_z = (1.0 - (1.0 / step_z).floor() * step_z) / 2.0;
570
571        let mut hits = 0u32;
572        let mut total = 0u32;
573
574        let mut fx = 0.0;
575        while fx <= 1.0 {
576            let mut fy = 0.0;
577            while fy <= 1.0 {
578                let mut fz = 0.0;
579                while fz <= 1.0 {
580                    let from = DVec3::new(
581                        bb.min_x() + (bb.max_x() - bb.min_x()) * fx + offset_x,
582                        bb.min_y() + (bb.max_y() - bb.min_y()) * fy,
583                        bb.min_z() + (bb.max_z() - bb.min_z()) * fz + offset_z,
584                    );
585                    let hit = self.clip(from, center, ClipBlockShape::Collider, ClipFluid::None);
586                    if hit.is_miss() {
587                        hits += 1;
588                    }
589                    total += 1;
590                    fz += step_z;
591                }
592                fy += step_y;
593            }
594            fx += step_x;
595        }
596
597        if total == 0 {
598            return 0.0;
599        }
600        hits as f32 / total as f32
601    }

603    /// Sets fire in a third of the exposed positions above solid ground.
604    ///
605    /// Vanilla parity: `ServerExplosion.createFire`.
606    fn create_explosion_fire(self: &Arc<Self>, positions: &[BlockPos]) {
607        for pos in positions {
608            if rand::random_range(0..3) != 0 {
609                continue;
610            }
611            if !self.get_block_state(*pos).is_air() {
612                continue;
613            }
614            if !self.get_block_state(pos.below()).is_solid() {
615                continue;
616            }
617            self.set_block(
618                *pos,

626#[cfg(test)]
627mod tests {
628    use super::default_explosion_damage_source;
629
630    /// A blast somebody is answerable for is a different damage type from one
631    /// nobody is, and the difference is load-bearing: `player_explosion` is in
632    /// `can_break_armor_stand` and `explosion` is not.
633    #[test]
634    fn an_attributed_blast_and_an_unattributed_one_are_different_damage_types() {
635        let attributed = default_explosion_damage_source(Some(7), Some(3));
636        let unattributed = default_explosion_damage_source(Some(7), None);
637        assert_eq!(attributed.damage_type.key.path, "player_explosion");
638        assert_eq!(unattributed.damage_type.key.path, "explosion");
639    }
640
641    /// A blast with no entity at all -- a respawn anchor, a bed -- is the
642    /// ordinary kind, never the attributed one.
643    #[test]
644    fn a_blast_with_no_entity_behind_it_is_the_ordinary_kind() {
645        let source = default_explosion_damage_source(None, None);
646        assert_eq!(source.damage_type.key.path, "explosion");
647    }
648
649    /// The shooter is answerable for a fireball, but the fireball is what
650    /// arrived. Vanilla keeps the two apart, and `DamageSource::is_direct`
651    /// reads exactly that difference -- before this pair existed every blast
652    /// claimed to be direct.
653    #[test]
654    fn a_shot_blast_credits_the_shooter_without_claiming_to_be_direct() {
655        let source = default_explosion_damage_source(Some(41), Some(9))
656            .with_direct_entity(41)
657            .with_causing_entity(9);
658        assert_eq!(source.direct_entity_id, Some(41));
659        assert_eq!(source.causing_entity_id, Some(9));
660        assert!(
661            !source.is_direct(),
662            "a shot blast reported itself as direct"
663        );
664    }
665
666    /// A creeper is its own cause, so its blast is attributed and direct at
667    /// once. This is the case that makes the `player_explosion` name
668    /// misleading, and it is worth pinning so nobody "fixes" it back.
669    #[test]
670    fn a_creeper_is_its_own_cause_and_its_blast_is_both_attributed_and_direct() {
671        let source = default_explosion_damage_source(Some(5), Some(5))
672            .with_direct_entity(5)
673            .with_causing_entity(5);
674        assert_eq!(source.damage_type.key.path, "player_explosion");
675        assert!(source.is_direct(), "a creeper's own blast was not direct");
676    }
