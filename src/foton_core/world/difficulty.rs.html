1//! Local difficulty.
2//!
3//! Vanilla parity: `DifficultyInstance`. Mobs that scale with difficulty read
4//! their strength from here rather than from the raw level setting, because
5//! vanilla also folds in how long the world has run and how long players have
6//! lingered in the chunk.
7
8use foton_registry::vanilla_world_clocks;
9use foton_utils::BlockPos;
10use foton_utils::types::Difficulty;
11
12use crate::world::World;
13
14/// Ticks of world age that do not count toward difficulty yet.
15///
16/// Vanilla parity: `DifficultyInstance.DIFFICULTY_TIME_GLOBAL_OFFSET`.
17const TIME_GLOBAL_OFFSET: f32 = -72_000.0;
18
19/// World age at which the global term stops growing.
20///
21/// Vanilla parity: `DifficultyInstance.MAX_DIFFICULTY_TIME_GLOBAL`.
22const MAX_TIME_GLOBAL: f32 = 1_440_000.0;
23
24/// Chunk inhabited time at which the local term stops growing.
25///
26/// Vanilla parity: `DifficultyInstance.MAX_DIFFICULTY_TIME_LOCAL`.
27const MAX_TIME_LOCAL: f32 = 3_600_000.0;
28
29/// Share of the scale the world age can add.
30const GLOBAL_TERM_WEIGHT: f32 = 0.25;
31
32/// Scale a world starts at before any of the growing terms apply.
33const BASE_SCALE: f32 = 0.75;
34
35/// Weight of the inhabited-time term outside hard difficulty.
36const LOCAL_TERM_WEIGHT: f32 = 0.75;
37
38/// Factor easy difficulty applies to the whole local term.
39const EASY_LOCAL_FACTOR: f32 = 0.5;
40
41/// Effective difficulty below which the special multiplier stays at zero.
42const SPECIAL_MULTIPLIER_FLOOR: f32 = 2.0;
43
44/// Effective difficulty at which the special multiplier reaches one.
45const SPECIAL_MULTIPLIER_CEILING: f32 = 4.0;
46
47/// The difficulty in effect at one position.
48///
49/// Vanilla parity: `DifficultyInstance`.
50#[derive(Debug, Clone, Copy, PartialEq)]
51pub struct DifficultyInstance {
52    base: Difficulty,
53    effective_difficulty: f32,
54}
55
56impl DifficultyInstance {
57    /// Computes the local difficulty from its four vanilla inputs.
58    #[must_use]
59    pub fn new(
60        base: Difficulty,
61        total_game_time: i64,
62        local_game_time: i64,
63        moon_brightness: f32,
64    ) -> Self {
65        Self {
66            base,
67            effective_difficulty: calculate_difficulty(
68                base,
69                total_game_time,
70                local_game_time,
71                moon_brightness,
72            ),
73        }
74    }
75
76    /// Returns the level difficulty setting this was scaled from.
77    ///
78    /// Vanilla parity: `DifficultyInstance.getDifficulty`.
79    #[must_use]
80    pub const fn difficulty(self) -> Difficulty {
81        self.base
82    }
83
84    /// Returns the scaled difficulty mobs read.
85    ///
86    /// Vanilla parity: `DifficultyInstance.getEffectiveDifficulty`.
87    #[must_use]
88    pub const fn effective_difficulty(self) -> f32 {
89        self.effective_difficulty
90    }
91
92    /// Returns whether the local difficulty has reached hard.
93    ///
94    /// Vanilla parity: `DifficultyInstance.isHard`.
95    #[must_use]
96    pub fn is_hard(self) -> bool {
97        self.effective_difficulty >= f32::from(Difficulty::Hard as u8)
98    }
99
100    /// Returns whether the local difficulty is above `required_difficulty`.
101    ///
102    /// Vanilla parity: `DifficultyInstance.isHarderThan`.
103    #[must_use]
104    pub fn is_harder_than(self, required_difficulty: f32) -> bool {
105        self.effective_difficulty > required_difficulty
106    }
107
108    /// Returns the 0-to-1 ramp mobs use for their rarer perks.
109    ///
110    /// Vanilla parity: `DifficultyInstance.getSpecialMultiplier`.
111    #[must_use]
112    pub fn special_multiplier(self) -> f32 {
113        if self.effective_difficulty < SPECIAL_MULTIPLIER_FLOOR {
114            0.0
115        } else if self.effective_difficulty > SPECIAL_MULTIPLIER_CEILING {
116            1.0
117        } else {
118            (self.effective_difficulty - SPECIAL_MULTIPLIER_FLOOR)
119                / (SPECIAL_MULTIPLIER_CEILING - SPECIAL_MULTIPLIER_FLOOR)
120        }
121    }
122}
123
124/// Vanilla parity: `DifficultyInstance.calculateDifficulty`.
125fn calculate_difficulty(
126    base: Difficulty,
127    total_game_time: i64,
128    local_game_time: i64,
129    moon_brightness: f32,
130) -> f32 {
131    if base == Difficulty::Peaceful {
132        return 0.0;
133    }
134
135    let is_hard = base == Difficulty::Hard;
136    let global_scale = ((total_game_time as f32 + TIME_GLOBAL_OFFSET) / MAX_TIME_GLOBAL)
137        .clamp(0.0, 1.0)
138        * GLOBAL_TERM_WEIGHT;
139
140    let inhabited_weight = if is_hard { 1.0 } else { LOCAL_TERM_WEIGHT };
141    let mut local_scale =
142        (local_game_time as f32 / MAX_TIME_LOCAL).clamp(0.0, 1.0) * inhabited_weight;
143    // A bright moon only counts for as much as the age of the world allows, so a
144    // full moon on the first night changes nothing.
145    local_scale += (moon_brightness * GLOBAL_TERM_WEIGHT).clamp(0.0, global_scale);
146    if base == Difficulty::Easy {
147        local_scale *= EASY_LOCAL_FACTOR;
148    }
149
150    f32::from(base as u8) * (BASE_SCALE + global_scale + local_scale)
151}

153impl World {
154    /// Returns the difficulty in effect at `pos`.
155    ///
156    /// Vanilla parity: `ServerLevel.getCurrentDifficultyAt`, minus two inputs
157    /// Foton does not track yet. Chunks carry no inhabited time, and the moon
158    /// phase is not exposed as an environment attribute, so both are passed as
159    /// zero. Neither can lower the result, so Foton is at worst slightly gentler
160    /// than vanilla in a world players have lived in for a long time.
161    #[must_use]
162    pub fn get_current_difficulty_at(&self, _pos: BlockPos) -> DifficultyInstance {
163        let clock_time = self
164            .clock_total_ticks(&vanilla_world_clocks::OVERWORLD)
165            .unwrap_or_default();
166        DifficultyInstance::new(self.difficulty(), clock_time, 0, 0.0)
167    }
168}

170#[cfg(test)]
171mod tests {
172    use foton_utils::types::Difficulty;
173
174    use super::{DifficultyInstance, calculate_difficulty};
175
176    /// Vanilla treats peaceful as no difficulty at all, whatever the world age.
177    #[test]
178    fn peaceful_is_always_zero() {
179        let peaceful = calculate_difficulty(Difficulty::Peaceful, 10_000_000, 3_600_000, 1.0);
180        assert!(peaceful.abs() < f32::EPSILON);
181    }
182
183    /// A fresh world sits at the base scale, so the `140 * (int) difficulty` of
184    /// `Husk.doHurtTarget` yields one step rather than two.
185    #[test]
186    fn fresh_world_uses_the_base_scale() {
187        let normal = calculate_difficulty(Difficulty::Normal, 0, 0, 0.0);
188        assert!((normal - 1.5).abs() < 1e-5, "normal was {normal}");
189        assert_eq!(normal as i32, 1);
190
191        let hard = calculate_difficulty(Difficulty::Hard, 0, 0, 0.0);
192        assert!((hard - 2.25).abs() < 1e-5, "hard was {hard}");
193        assert_eq!(hard as i32, 2);
194    }
195
196    /// The world-age term only starts after the first hour, and it is capped.
197    #[test]
198    fn world_age_term_is_offset_and_capped() {
199        let before_offset = calculate_difficulty(Difficulty::Normal, 72_000, 0, 0.0);
200        assert!((before_offset - 1.5).abs() < 1e-5, "was {before_offset}");
201
202        // 2 * (0.75 + 0.25), with no inhabited time.
203        let capped = calculate_difficulty(Difficulty::Normal, i64::from(i32::MAX), 0, 0.0);
204        assert!((capped - 2.0).abs() < 1e-5, "capped was {capped}");
205    }
206
207    /// Hard difficulty weighs inhabited time more heavily than the rest.
208    #[test]
209    fn inhabited_time_counts_more_on_hard() {
210        // 2 * (0.75 + 0.75).
211        let normal = calculate_difficulty(Difficulty::Normal, 0, 3_600_000, 0.0);
212        assert!((normal - 3.0).abs() < 1e-5, "normal was {normal}");
213
214        // 3 * (0.75 + 1.0).
215        let hard = calculate_difficulty(Difficulty::Hard, 0, 3_600_000, 0.0);
216        assert!((hard - 5.25).abs() < 1e-5, "hard was {hard}");
217    }
218
219    /// Easy halves the whole local term.
220    #[test]
221    fn easy_halves_the_local_term() {
222        // 1 * (0.75 + 0 + 0.75 * 0.5).
223        let easy = calculate_difficulty(Difficulty::Easy, 0, 3_600_000, 0.0);
224        assert!((easy - 1.125).abs() < 1e-5, "easy was {easy}");
225    }
226
227    /// A full moon cannot outgrow the world-age term that bounds it.
228    #[test]
229    fn moon_brightness_is_bounded_by_the_world_age_term() {
230        let young = calculate_difficulty(Difficulty::Normal, 0, 0, 1.0);
231        assert!((young - 1.5).abs() < 1e-5, "young was {young}");
232
233        // The moon adds min(0.25, 0.25) on top of the fully grown global term.
234        let old = calculate_difficulty(Difficulty::Normal, i64::from(i32::MAX), 0, 1.0);
235        assert!((old - 2.5).abs() < 1e-5, "old was {old}");
236    }
237
238    /// Vanilla parity: `DifficultyInstance.getSpecialMultiplier`.
239    #[test]
240    fn special_multiplier_ramps_between_two_and_four() {
241        let instance = |effective_difficulty: f32| DifficultyInstance {
242            base: Difficulty::Normal,
243            effective_difficulty,
244        };
245        assert!(instance(1.9).special_multiplier().abs() < f32::EPSILON);
246        assert!((instance(3.0).special_multiplier() - 0.5).abs() < 1e-5);
247        assert!((instance(5.0).special_multiplier() - 1.0).abs() < f32::EPSILON);
248    }
249
250    /// Vanilla parity: `DifficultyInstance.isHard`, which reads the scaled value
251    /// rather than the setting, so a brand new hard world is not yet hard.
252    #[test]
253    fn is_hard_tracks_the_effective_value_not_the_setting() {
254        let fresh = DifficultyInstance::new(Difficulty::Hard, 0, 0, 0.0);
255        assert!(!fresh.is_hard(), "2.25 is below the hard threshold of 3");
256        assert!(fresh.is_harder_than(2.0));
257
258        let lived_in =
259            DifficultyInstance::new(Difficulty::Hard, i64::from(i32::MAX), 3_600_000, 1.0);
260        assert!(lived_in.is_hard());
261    }
