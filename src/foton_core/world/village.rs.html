1//! What counts as a village, and which raid is besieging it.
2//!
3//! Vanilla parity: the village and raid accessors of `ServerLevel` --
4//! `isVillage`, `isCloseToVillage`, `sectionsToVillage`, `getRaids`,
5//! `getRaidAt` and `isRaided`. They sit together because they are one question
6//! asked two ways: the point-of-interest index answers where a village is, and
7//! the raid manager answers what is happening to it.
8
9use std::sync::Arc;
10
11use foton_registry::blocks::block_state_ext::BlockStateExt as _;
12use foton_registry::vanilla_blocks;
13use foton_utils::{BlockPos, SectionPos};
14
15use super::World;
16use crate::poi::MAX_VILLAGE_DISTANCE;
17use crate::raid::{Raid, Raids, VALID_RAID_RADIUS_SQR};
18
19impl World {
20    /// Returns the raids running in this loaded world.
21    ///
22    /// Vanilla parity: `ServerLevel.getRaids`.
23    #[must_use]
24    pub const fn raids(&self) -> &Raids {
25        &self.raids
26    }
27
28    /// Returns whether `pos` is inside a village.
29    ///
30    /// Vanilla parity: `ServerLevel.isVillage(BlockPos)`.
31    #[must_use]
32    pub fn is_village(&self, pos: BlockPos) -> bool {
33        self.is_close_to_village(pos, 1)
34    }
35
36    /// Returns whether the center of `section` is inside a village.
37    ///
38    /// Vanilla parity: `ServerLevel.isVillage(SectionPos)`.
39    #[must_use]
40    pub fn is_village_section(&self, section: SectionPos) -> bool {
41        self.is_village(BlockPos::new(
42            (section.x() << 4) + 8,
43            (section.y() << 4) + 8,
44            (section.z() << 4) + 8,
45        ))
46    }
47
48    /// Returns whether a village center is within `section_distance` sections.
49    ///
50    /// Vanilla parity: `ServerLevel.isCloseToVillage`.
51    #[must_use]
52    pub fn is_close_to_village(&self, pos: BlockPos, section_distance: i32) -> bool {
53        if section_distance > MAX_VILLAGE_DISTANCE {
54            return false;
55        }
56        self.sections_to_village(SectionPos::from_block_pos(pos)) <= section_distance
57    }
58
59    /// Returns how many sections away the nearest village center is.
60    ///
61    /// Vanilla parity: `ServerLevel.sectionsToVillage`.
62    #[must_use]
63    pub fn sections_to_village(&self, section: SectionPos) -> i32 {
64        self.poi_storage.lock().sections_to_village(section)
65    }
66
67    /// Returns the raid `pos` is inside, if any.
68    ///
69    /// Vanilla parity: `ServerLevel.getRaidAt`.
70    #[must_use]
71    pub fn get_raid_at(&self, pos: BlockPos) -> Option<Arc<Raid>> {
72        self.raids.nearby_raid(pos, VALID_RAID_RADIUS_SQR)
73    }
74
75    /// Returns the id of the raid `pos` is inside, if any.
76    ///
77    /// The id rather than the raid, for callers comparing against a raid they
78    /// already hold -- vanilla compares by object identity, which an `Arc`
79    /// pulled out of the manager cannot answer without the manager's lock.
80    #[must_use]
81    pub fn raid_id_at(&self, pos: BlockPos) -> Option<i32> {
82        self.get_raid_at(pos).map(|raid| raid.id())
83    }
84
85    /// Returns whether a raid is running over `pos`.
86    ///
87    /// Vanilla parity: `ServerLevel.isRaided`.
88    #[must_use]
89    pub fn is_raided(&self, pos: BlockPos) -> bool {
90        self.get_raid_at(pos).is_some()
91    }
92
93    /// Returns whether `pos` is open air over a snow layer.
94    ///
95    /// Vanilla parity: the `getBlockState(pos.below()).is(Blocks.SNOW) &&
96    /// getBlockState(pos).isAir()` half of `Raid.findRandomSpawnPos`, which is
97    /// what lets a raid spawn on a snowy plain where the ravager's own spawn
98    /// placement refuses the surface.
99    #[must_use]
100    pub fn is_snow_over_air(&self, pos: BlockPos) -> bool {
101        self.get_block_state(pos.below()).get_block() == &vanilla_blocks::SNOW
102            && self.get_block_state(pos).is_air()
103    }
104}
