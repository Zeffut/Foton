1//! Natural mob spawning.
2//!
3//! Vanilla parity: `NaturalSpawner`, with one documented structural difference.
4//! Vanilla walks every ticking chunk and applies a per-chunk cap; Foton samples
5//! candidate positions in the ring around each player instead, because chunk
6//! ticking does not expose the per-chunk mob accounting vanilla relies on. The
7//! observable rules that matter are reproduced: the 24-block exclusion around
8//! players, the 128-block outer limit, the per-category budget, where each kind
9//! of mob is allowed to stand, and the spawn rules the mob itself imposes.
10

30/// Closest a mob may spawn to a player.
31///
32/// Vanilla parity: `NaturalSpawner.MIN_SPAWN_DISTANCE`.
33const MIN_SPAWN_DISTANCE: f64 = 24.0;
34
35/// Furthest a mob may spawn from a player.
36///
37/// Vanilla parity: `NaturalSpawner.SPAWN_DISTANCE_BLOCK`.
38const MAX_SPAWN_DISTANCE: f64 = 128.0;
39
40/// Candidate positions tried per category, per player, each spawn cycle.
41const ATTEMPTS_PER_PLAYER: u32 = 16;
42
43/// Groups attempted at one sampled position.
44///
45/// Vanilla parity: the `for (groupCount = 0; groupCount < 3; groupCount++)` of
46/// `NaturalSpawner.spawnCategoryForPosition`.
47const GROUPS_PER_POSITION: u32 = 3;
48
49/// Ticks between spawn cycles.
50///
51/// Vanilla runs the spawner every tick over many chunks; Foton samples fewer
52/// positions less often for the same rough rate.
53const SPAWN_INTERVAL_TICKS: u64 = 20;
54
55/// Ticks between attempts at the categories that stay loaded.
56///
57/// Vanilla parity: the `getGameTime() % 400L == 0L` of
58/// `ServerChunkCache.tickChunks`. Creatures are persistent, so vanilla offers
59/// them a turn only every four hundred ticks; that slow cadence is why the
60/// countryside does not silt up with cows, and why most animals a player meets
61/// were placed when the chunk was made rather than spawned later.
62const PERSISTENT_SPAWN_INTERVAL_TICKS: u64 = 400;
63
64/// Chunks covered by the box [`World::mobs_near`] counts mobs in.
65///
66/// The box is `MAX_SPAWN_DISTANCE` in every direction, so 256 blocks on a side,
67/// so sixteen chunks on a side.
68const SAMPLED_CHUNK_COUNT: i32 = 16 * 16;
69
70/// Vanilla's reference area for a category budget.
71///
72/// Vanilla parity: `NaturalSpawner.MAGIC_NUMBER`, seventeen chunks squared.
73/// `maxInstancesPerChunk` is a count per that area, not per chunk, which is why
74/// dividing by it is what turns the category figure into a density.
75const SPAWN_CAP_REFERENCE_CHUNKS: i32 = 17 * 17;
76
77/// Categories the spawner populates, in vanilla's declaration order.
78///
79/// `Misc` is absent: it is the category of boats and item frames, and vanilla
80/// gives it a negative per-chunk maximum precisely so that nothing spawns it.
81const SPAWNABLE_CATEGORIES: [MobCategory; 7] = [
82    MobCategory::Monster,
83    MobCategory::Creature,
84    MobCategory::Ambient,
85    MobCategory::Axolotls,
86    MobCategory::UndergroundWaterCreature,
87    MobCategory::WaterCreature,
88    MobCategory::WaterAmbient,
89];
90
91/// Returns how many mobs of `category` may live in one player's spawn ring.
92///
93/// Vanilla parity: `SpawnState.canSpawnForCategoryGlobal` allows
94/// `maxInstancesPerChunk * spawnableChunkCount / MAGIC_NUMBER` mobs across
95/// every chunk that can spawn. That is a density, and Foton applies the same
96/// density to the box it actually counts in.
97///
98/// The figure this replaced was a flat twenty-four monsters, which is 0.12 per
99/// chunk against vanilla's 0.24 -- half the ceiling. In the End that is plainly
100/// visible, because the enderman is the whole monster table there and nothing
101/// else takes up the slack.
102fn default_category_cap(category: MobCategory) -> usize {
103    let max = category.max_instances_per_chunk();
104    if max <= 0 {
105        return 0;
106    }
107    let scaled = max * SAMPLED_CHUNK_COUNT / SPAWN_CAP_REFERENCE_CHUNKS;
108    scaled.max(1) as usize
109}
110
111/// Rolls how many mobs one group asks for, from the biome entry's own counts.
112///
113/// Vanilla parity: the `max = spawnerData.minCount() + random.nextInt(1 +
114/// spawnerData.maxCount() - spawnerData.minCount())` of
115/// `NaturalSpawner.spawnCategoryForPosition`.
116///
117/// These two counts sit in every biome file and were parsed into
118/// [`SpawnerData`] all along; the spawner simply never read them and spawned
119/// one mob per success. The End states `minCount: 4, maxCount: 4` for the
120/// enderman, so a quarter of the endermen vanilla places were appearing.
121fn roll_pack_size(min_count: i32, max_count: i32) -> i32 {
122    let span = (max_count - min_count).max(0) + 1;
123    min_count + rand::random_range(0..span)
124}

127    /// Runs one natural spawning cycle.
128    ///
129    /// Vanilla parity: `NaturalSpawner.spawnForChunk`, applied per player.
130    pub fn tick_natural_spawn(self: &Arc<Self>, tick_count: u64) {
131        if !tick_count.is_multiple_of(SPAWN_INTERVAL_TICKS) {
132            return;
133        }
134        if !self.get_game_rule(&SPAWN_MOBS) {
135            return;
136        }
137        let spawn_monsters = self.get_game_rule(&SPAWN_MONSTERS);
138        let spawn_persistent = tick_count.is_multiple_of(PERSISTENT_SPAWN_INTERVAL_TICKS);
139
140        // One snapshot per cycle, not one walk of the player index per candidate.
141        // `nearest_player_distance_sqr` goes through `iter_players`, which locks
142        // the order list, collects it into a fresh `Vec<Uuid>` and then looks each
143        // player up by uuid -- and the pack spawner asks for the nearest player
144        // once per mob it tries, so that walk went from sixteen times a cycle to
145        // several hundred. Vanilla calls `getNearestPlayer` just as often; what it
146        // does not do is rebuild the player list to answer.
147        //
148        // Spectators are excluded here for the same reason
149        // `nearest_player_distance_sqr` excludes them.
150        let mut player_positions = Vec::new();
151        self.players.iter_players(|_, player| {
152            if !player.is_spectator() {
153                player_positions.push(player.position());
154            }
155            true
156        });
157
158        for &origin in &player_positions {
159            for category in SPAWNABLE_CATEGORIES {
160                // Vanilla parity: `NaturalSpawner.getFilteredSpawningCategories`
161                // drops the monsters when `doMobSpawning` is off, and drops the
162                // persistent categories on every tick but the four-hundredth.
163                if (!category.is_friendly() && (!spawn_monsters || !self.allow_monsters()))
164                    || (category.is_friendly() && !self.allow_animals())
165                {
166                    continue;
167                }
168                if category.is_persistent() && !spawn_persistent {
169                    continue;
170                }
171                let interval = self
172                    .spawn_ticks(category)
173                    .unwrap_or(if category.is_persistent() { 400 } else { 20 });
174                if interval == 0 || !tick_count.is_multiple_of(interval as u64) {
175                    continue;
176                }
177                let cap = self
178                    .spawn_limit(category)
179                    .unwrap_or_else(|| default_category_cap(category) as i32)
180                    .max(0) as usize;
181                // Deliberately stricter than vanilla, which reads the category
182                // ceiling once per tick in `getFilteredSpawningCategories` and
183                // then lets that tick overshoot it freely -- its `state::canSpawn`
184                // is the spawn-cost predicate, not the ceiling. Vanilla can afford
185                // that because it walks chunks; Foton samples a ring per player,
186                // so an unchecked cycle would place a pack at every one of its
187                // sixteen positions and blow past the ceiling in one go.
188                let mut budget = cap.saturating_sub(self.mobs_near(origin, category));
189                if budget == 0 {
190                    continue;
191                }
192                for _ in 0..ATTEMPTS_PER_PLAYER {
193                    if budget == 0 {
194                        break;
195                    }
196                    if let Some(pos) = self.pick_spawn_position(origin) {
197                        self.try_spawn_at(pos, category, &player_positions, &mut budget);
198                    }
199                }
200            }
201        }
202    }

204    /// Counts mobs of `category` already alive within the sampling radius.
205    fn mobs_near(self: &Arc<Self>, origin: DVec3, category: MobCategory) -> usize {
206        let aabb = WorldAabb::new(
207            origin.x - MAX_SPAWN_DISTANCE,
208            origin.y - MAX_SPAWN_DISTANCE,
209            origin.z - MAX_SPAWN_DISTANCE,
210            origin.x + MAX_SPAWN_DISTANCE,
211            origin.y + MAX_SPAWN_DISTANCE,
212            origin.z + MAX_SPAWN_DISTANCE,
213        );
214        self.get_entities_in_aabb(&aabb)
215            .iter()
216            .filter(|entity| entity.entity_type().mob_category == category)
217            .count()
218    }

220    /// Picks a candidate position in the spawn ring around `origin`.
221    ///
222    /// Vanilla parity: `NaturalSpawner.getRandomPosWithin`, which picks a column
223    /// and a height without judging either. Whether the spot suits the mob is
224    /// decided later, by that mob's placement type, because the answer differs
225    /// for a cow and for a cod.
226    fn pick_spawn_position(self: &Arc<Self>, origin: DVec3) -> Option<BlockPos> {
227        let angle = rand::random::<f64>() * TAU;
228        let distance = (MAX_SPAWN_DISTANCE - MIN_SPAWN_DISTANCE)
229            .mul_add(rand::random::<f64>(), MIN_SPAWN_DISTANCE);
230        let x = distance.mul_add(angle.cos(), origin.x).floor() as i32;
231        let z = distance.mul_add(angle.sin(), origin.z).floor() as i32;
232
233        // Search a vertical band around the player rather than the whole column.
234        let center_y = origin.y.floor() as i32;
235        let y = center_y + rand::random_range(-16..=16);
236        if self.is_outside_build_height(y) {
237            return None;
238        }
239        Some(BlockPos::new(x, y, z))
240    }

242    /// Fills one sampled position with a pack, the way vanilla fills a chosen one.
243    ///
244    /// Vanilla parity: `NaturalSpawner.spawnCategoryForPosition`. Three groups
245    /// are attempted at the position. Each draws one biome entry, takes its pack
246    /// size from that entry, and walks a short random offset per mob so the pack
247    /// lands as a cluster. `group_data` is threaded from each mob to the next,
248    /// which is what lets a pack share one variant or one leader.
249    ///
250    /// `budget` is the category's remaining headroom and is spent as mobs join,
251    /// standing in for the running count vanilla keeps in `SpawnState`.
252    fn try_spawn_at(
253        self: &Arc<Self>,
254        pos: BlockPos,
255        category: MobCategory,
256        player_positions: &[DVec3],
257        budget: &mut usize,
258    ) {
259        let mut cluster_size = 0;
260
261        for _ in 0..GROUPS_PER_POSITION {
262            let mut x = pos.x();
263            let mut z = pos.z();
264            let mut entry: Option<(EntityTypeRef, i32, i32)> = None;
265            let mut group_data: Option<SpawnGroupData> = None;
266            let mut group_size = 0;
267            // Vanilla opens with a one-to-four budget and replaces it with the
268            // biome entry's own count the moment an entry is drawn.
269            let mut remaining = (rand::random::<f32>() * 4.0).ceil() as i32;
270            let mut attempt = 0;
271
272            while attempt < remaining {
273                attempt += 1;
274                x += rand::random_range(0..6) - rand::random_range(0..6);
275                z += rand::random_range(0..6) - rand::random_range(0..6);
276                let candidate = BlockPos::new(x, pos.y(), z);
277                let (center_x, center_y, center_z) = candidate.get_bottom_center();
278                let center = DVec3::new(center_x, center_y, center_z);
279
280                // Vanilla refuses any position closer than 24 blocks to a player.
281                let nearest_player_distance_sqr = player_positions
282                    .iter()
283                    .map(|player| player.distance_squared(center))
284                    .min_by(f64::total_cmp);
285                if nearest_player_distance_sqr.is_some_and(|distance_sqr| {
286                    distance_sqr < MIN_SPAWN_DISTANCE * MIN_SPAWN_DISTANCE
287                }) {
288                    continue;
289                }
290
291                if entry.is_none() {
292                    let Some(drawn) = self.pick_spawner_entry(candidate, category) else {
293                        break;
294                    };
295                    let (_, min_count, max_count) = drawn;
296                    remaining = roll_pack_size(min_count, max_count);
297                    entry = Some(drawn);
298                }
299                let Some((entity_type, _, _)) = entry else {
300                    break;
301                };
302
303                if !self.is_valid_spawn_position_for_type(
304                    entity_type,
305                    category,
306                    candidate,
307                    nearest_player_distance_sqr,
308                ) {
309                    continue;
310                }
311
312                let mut pre_spawn = PreCreatureSpawnEvent::new(
313                    self.key.to_string(),
314                    center.x,
315                    center.y,
316                    center.z,
317                    entity_type.key.to_string(),
318                    "Natural".to_owned(),
319                );
320                self.fire_event(&mut pre_spawn);
321                if pre_spawn.is_cancelled() {
322                    continue;
323                }
324
325                // Vanilla parity: a `getMobForSpawn` that comes back null ends the
326                // whole attempt, not just this group -- a type that cannot be built
327                // will not build on the next try either.
328                let Some(entity) =
329                    ENTITIES.create(entity_type, next_entity_id(), center, Arc::downgrade(self))
330                else {
331                    return;
332                };
333
334                // Vanilla asks the entity type's registered predicate before creating
335                // anything. Foton has no path from a type to its behavior, so the mob is
336                // created and asked; one that answers no is dropped here, unspawned.
337                let Some(mob) = entity.as_mob() else {
338                    return;
339                };
340                if !mob.check_spawn_rules(self, EntitySpawnReason::Natural, candidate) {
341                    continue;
342                }
343
344                group_data = mob.finalize_spawn(self, EntitySpawnReason::Natural, group_data);
345
346                let mut spawn_event = CreatureSpawnEvent::new(
347                    entity.uuid(),
348                    self.key.to_string(),
349                    entity.position().x,
350                    entity.position().y,
351                    entity.position().z,
352                    "Natural".to_owned(),
353                );
354                self.begin_pending_spawn(Arc::clone(&entity));
355                self.fire_event(&mut spawn_event);
356                self.end_pending_spawn(&entity.uuid());
357                if spawn_event.is_cancelled() {
358                    continue;
359                }
360                if let Err(error) = self.try_add_entity(Arc::clone(&entity)) {
361                    log::debug!("natural spawn rejected: {error}");
362                    continue;
363                }
364
365                cluster_size += 1;
366                group_size += 1;
367                *budget -= 1;
368                if *budget == 0 || cluster_size >= mob.max_spawn_cluster_size() {
369                    return;
370                }
371                if mob.is_max_group_size_reached(group_size) {
372                    break;
373                }
374            }
375        }
376    }

378    /// Picks a mob of `category` from the biome's weighted spawn list.
379    ///
380    /// Vanilla parity: `NaturalSpawner.getRandomSpawnMobAt`.
381    /// Applies the per-mob gate vanilla runs before it builds anything.
382    ///
383    /// Vanilla parity: `NaturalSpawner.isValidSpawnPostitionForType`, in its
384    /// order. Foton previously ran only the placement type's own test, which was
385    /// survivable while one mob was placed per position. A pack walks up to
386    /// twenty blocks from where its entry was drawn, so the entry now has to be
387    /// re-checked against the biome it actually landed in, and the clearance has
388    /// to be checked at all -- four mobs at the same `y`, a few blocks apart,
389    /// were being stacked into whatever happened to be there.
390    ///
391    /// One gap remains, and it is a data one: vanilla scales the spawn box by
392    /// `EntityType.spawnDimensionsScale`, which `FotonExtractor` does not
393    /// extract.
394    /// It is `1.0` for every type but the three slimes -- `minecraft:slime` and
395    /// `minecraft:magma_cube` at `4.0`, `minecraft:sulfur_cube` at `2.0` --
396    /// which reserve room for the largest size they may roll. Until the
397    /// extractor carries it, those three are checked against their base size and
398    /// so are held to a smaller clearance than vanilla asks. Writing the three
399    /// numbers here by hand is what rule 2 forbids.
400    fn is_valid_spawn_position_for_type(
401        self: &Arc<Self>,
402        entity_type: EntityTypeRef,
403        category: MobCategory,
404        pos: BlockPos,
405        nearest_player_distance_sqr: Option<f64>,
406    ) -> bool {
407        if entity_type.mob_category == MobCategory::Misc {
408            return false;
409        }
410
411        // Vanilla parity: a mob that despawns far from players is not spawned
412        // out there either.
413        if !entity_type.can_spawn_far_from_player
414            && let Some(distance_sqr) = nearest_player_distance_sqr
415        {
416            let despawn = f64::from(category.despawn_distance());
417            if distance_sqr > despawn * despawn {
418                return false;
419            }
420        }
421
422        if !entity_type.summonable {
423            return false;
424        }
425
426        // Vanilla parity: `canSpawnMobAt`, which re-reads the biome's list at
427        // the position actually reached rather than trusting the draw.
428        if !self.biome_lists_mob(pos, category, entity_type) {
429            return false;
430        }
431
432        if !spawn_placement_for(entity_type).is_spawn_position_ok(self, pos, entity_type) {
433            return false;
434        }
435
436        self.spawn_box_is_clear(entity_type, pos)
437    }

439    /// Whether the biome at `pos` still offers `entity_type` for `category`.
440    fn biome_lists_mob(
441        self: &Arc<Self>,
442        pos: BlockPos,
443        category: MobCategory,
444        entity_type: EntityTypeRef,
445    ) -> bool {
446        let Some(biome) = self.biome_at(pos) else {
447            return false;
448        };
449        let Some(candidates) = biome.spawners.get(category.name()) else {
450            return false;
451        };
452        candidates
453            .iter()
454            .any(|entry| entry.entity_type == entity_type.key)
455    }

457    /// Whether the mob's spawn box is free of blocks and other entities.
458    ///
459    /// Vanilla parity: the `level.noCollision(type.getSpawnAABB(...))` that
460    /// closes `isValidSpawnPostitionForType`.
461    fn spawn_box_is_clear(self: &Arc<Self>, entity_type: EntityTypeRef, pos: BlockPos) -> bool {
462        let (x, y, z) = pos.get_bottom_center();
463        let dimensions = entity_type.dimensions;
464        let aabb = WorldAabb::entity_box(
465            x,
466            y,
467            z,
468            f64::from(dimensions.half_width()),
469            f64::from(dimensions.height),
470        );
471        let collision_world = WorldCollisionProvider::new(self);
472        !collision_world.has_entity_collision(&aabb)
473            && !collision_world
474                .has_block_collision_with_context(&aabb, BlockCollisionContext::empty())
475    }

477    /// Chooses one biome spawn entry for `pos`, weighted as vanilla weights it.
478    ///
479    /// Vanilla parity: `NaturalSpawner.getRandomSpawnMobAt`. The pack size
480    /// travels with the entry: vanilla reads `minCount`/`maxCount` off the very
481    /// entry it just drew, and spawning one mob where the biome asked for four
482    /// is what made the End look empty.
483    fn pick_spawner_entry(
484        self: &Arc<Self>,
485        pos: BlockPos,
486        category: MobCategory,
487    ) -> Option<(EntityTypeRef, i32, i32)> {
488        let biome = self.biome_at(pos)?;
489        let candidates: &Vec<SpawnerData> = biome.spawners.get(category.name())?;
490
491        let total: i32 = candidates.iter().map(|entry| entry.weight).sum();
492        if total <= 0 {
493            return None;
494        }
495
496        let mut roll = rand::random_range(0..total);
497        for entry in candidates {
498            roll -= entry.weight;
499            if roll < 0 {
500                // Only entity types Foton actually implements can spawn; the rest
501                // are skipped until their entity exists.
502                let entity_type = REGISTRY.entity_types.by_key(&entry.entity_type)?;
503                return Some((entity_type, entry.min_count, entry.max_count));
504            }
505        }
506        None
507    }

554    #[test]
555    fn pack_size_honors_the_biome_entry() {
556        // The End asks for `minCount: 4, maxCount: 4` endermen. A spawner that
557        // ignores the entry and places one mob per success -- which is what
558        // Foton did -- puts a quarter of vanilla's endermen in the world, and
559        // the enderman is the whole monster table there.

565        let mut seen_low = false;
566        let mut seen_high = false;
567        for _ in 0..512 {

569            assert!(

574            seen_high |= rolled == 4;
575        }
576        assert!(seen_low && seen_high, "the range's ends are unreachable");
577    }

579    #[test]
580    fn budgets_match_vanillas_density() {
581        // Vanilla allows `maxInstancesPerChunk` mobs per 289 chunks. Foton
582        // counts mobs in a fixed box instead of walking chunks, so the budget
583        // has to carry that same density across -- the flat twenty-four
584        // monsters this replaced was half of it, and half a ceiling is what a
585        // player reads as an empty world.
586        for category in SPAWNABLE_CATEGORIES {
587            let vanilla = f64::from(category.max_instances_per_chunk())
588                / f64::from(SPAWN_CAP_REFERENCE_CHUNKS);
589            let foton = default_category_cap(category) as f64 / f64::from(SAMPLED_CHUNK_COUNT);
590            assert!(
591                (foton - vanilla).abs() < 0.01,
592                "{category:?} holds {foton:.3} mobs per chunk against vanilla's {vanilla:.3}"
593            );
594        }
595    }
