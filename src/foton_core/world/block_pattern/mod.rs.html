1//! Matching multi-block shapes against the world.
2//!
3//! Vanilla parity: `net.minecraft.world.level.block.state.pattern.BlockPattern`,
4//! `BlockPatternBuilder` and `BlockInWorld`. Vanilla builds patterns for the
5//! carved pumpkin's three golems, for the wither, and for the end crystal
6//! respawn ritual, so this lives next to the world rather than inside any one
7//! block behavior.

22/// Vanilla `Direction.values()`, in declaration order.
23///
24/// `BlockPattern.find` returns the first orientation that matches, so the
25/// iteration order is observable.
26const ALL_DIRECTIONS: [Direction; 6] = [
27    Direction::Down,
28    Direction::Up,
29    Direction::North,
30    Direction::South,
31    Direction::West,
32    Direction::East,
33];

35/// One position of a pattern, resolved against the world.
36///
37/// Vanilla parity: `BlockInWorld`. Vanilla resolves the state lazily; Foton
38/// resolves it through the per-search cache instead, which serves the same
39/// purpose of reading each position once.
40pub struct BlockInWorld<'a> {
41    level: &'a dyn LevelReader,
42    pos: BlockPos,
43    state: BlockStateId,
44}
45
46impl<'a> BlockInWorld<'a> {
47    /// Reads the block at `pos`.
48    #[must_use]
49    pub fn new(level: &'a dyn LevelReader, pos: BlockPos) -> Self {
50        Self {
51            level,
52            pos,
53            state: level.get_block_state(pos),
54        }
55    }
56
57    /// Returns the state at this position.
58    #[must_use]
59    pub const fn state(&self) -> BlockStateId {
60        self.state
61    }
62
63    /// Returns this position.
64    #[must_use]
65    pub const fn pos(&self) -> BlockPos {
66        self.pos
67    }
68
69    /// Returns the block entity at this position, when the level has one.
70    #[must_use]
71    pub fn block_entity(&self) -> Option<SharedBlockEntity> {
72        self.level.get_block_entity(self.pos)
73    }
74}

76/// A test one pattern position has to pass.
77pub type BlockPatternPredicate = Arc<dyn Fn(&BlockInWorld<'_>) -> bool + Send + Sync>;
78
79/// Lifts a block-state test into a pattern predicate.
80///
81/// Vanilla parity: `BlockInWorld.hasState`.
82#[must_use]
83pub fn has_state(
84    predicate: impl Fn(BlockStateId) -> bool + Send + Sync + 'static,
85) -> BlockPatternPredicate {
86    Arc::new(move |block: &BlockInWorld<'_>| predicate(block.state()))
87}

89/// Reads each position of a search at most once.
90///
91/// Vanilla parity: the Guava `LoadingCache` of `BlockPattern.createLevelCache`.
92/// A single `find` walks the same positions under 24 orientations, so without
93/// this every candidate origin would re-read the whole neighborhood.
94struct LevelCache<'a> {
95    level: &'a dyn LevelReader,
96    states: RefCell<FxHashMap<BlockPos, BlockStateId>>,
97}
98
99impl<'a> LevelCache<'a> {
100    fn new(level: &'a dyn LevelReader) -> Self {
101        Self {
102            level,
103            states: RefCell::new(FxHashMap::default()),
104        }
105    }
106
107    fn get(&self, pos: BlockPos) -> BlockInWorld<'a> {
108        let state = *self
109            .states
110            .borrow_mut()
111            .entry(pos)
112            .or_insert_with(|| self.level.get_block_state(pos));
113        BlockInWorld {
114            level: self.level,
115            pos,
116            state,
117        }
118    }
119}

121/// A shape of block predicates that can be searched for in the world.
122///
123/// Vanilla parity: `BlockPattern`.
124pub struct BlockPattern {
125    /// Indexed `[depth][height][width]`, like vanilla's `Predicate[][][]`.
126    pattern: Vec<Vec<Vec<BlockPatternPredicate>>>,
127    depth: usize,
128    height: usize,
129    width: usize,
130}
131
132impl BlockPattern {
133    /// Returns how many aisles deep this pattern is.
134    #[must_use]
135    pub const fn depth(&self) -> usize {
136        self.depth
137    }
138
139    /// Returns how many rows tall this pattern is.
140    #[must_use]
141    pub const fn height(&self) -> usize {
142        self.height
143    }
144
145    /// Returns how many columns wide this pattern is.
146    #[must_use]
147    pub const fn width(&self) -> usize {
148        self.width
149    }
150
151    /// Tests this pattern at one origin and orientation.
152    ///
153    /// Vanilla parity: `BlockPattern.matches(LevelReader, BlockPos, Direction,
154    /// Direction)`. Returns `None` when `up` is parallel to `forwards`, which
155    /// vanilla rejects by throwing.
156    #[must_use]
157    pub fn matches<'a>(
158        &self,
159        level: &'a dyn LevelReader,
160        origin: BlockPos,
161        forwards: Direction,
162        up: Direction,
163    ) -> Option<BlockPatternMatch<'a>> {
164        if up == forwards || up == forwards.opposite() {
165            return None;
166        }
167
168        let cache = LevelCache::new(level);
169        if !self.matches_in_cache(origin, forwards, up, &cache) {
170            return None;
171        }
172
173        Some(self.build_match(origin, forwards, up, cache))
174    }
175
176    /// Searches for this pattern near `origin`.
177    ///
178    /// Vanilla parity: `BlockPattern.find`.
179    #[must_use]
180    pub fn find<'a>(
181        &self,
182        level: &'a dyn LevelReader,
183        origin: BlockPos,
184    ) -> Option<BlockPatternMatch<'a>> {
185        let cache = LevelCache::new(level);
186        let dist = self.width.max(self.height).max(self.depth) as i32;
187
188        for test_pos in
189            BlockPos::between_closed(origin, origin.offset(dist - 1, dist - 1, dist - 1))
190        {
191            for forwards in ALL_DIRECTIONS {
192                for up in ALL_DIRECTIONS {
193                    if up == forwards || up == forwards.opposite() {
194                        continue;
195                    }
196                    if self.matches_in_cache(test_pos, forwards, up, &cache) {
197                        return Some(self.build_match(test_pos, forwards, up, cache));
198                    }
199                }
200            }
201        }
202
203        None
204    }
205
206    const fn build_match<'a>(
207        &self,
208        front_top_left: BlockPos,
209        forwards: Direction,
210        up: Direction,
211        cache: LevelCache<'a>,
212    ) -> BlockPatternMatch<'a> {
213        BlockPatternMatch {
214            front_top_left,
215            forwards,
216            up,
217            cache,
218            width: self.width,
219            height: self.height,
220            depth: self.depth,
221        }
222    }
223
224    fn matches_in_cache(
225        &self,
226        origin: BlockPos,
227        forwards: Direction,
228        up: Direction,
229        cache: &LevelCache<'_>,
230    ) -> bool {
231        for x in 0..self.width {
232            for y in 0..self.height {
233                for z in 0..self.depth {
234                    let pos =
235                        translate_and_rotate(origin, forwards, up, x as i32, y as i32, z as i32);
236                    if !self.pattern[z][y][x](&cache.get(pos)) {
237                        return false;
238                    }
239                }
240            }
241        }
242
243        true
244    }
245}

247/// Maps a pattern cell onto a world position.
248///
249/// Vanilla parity: `BlockPattern.translateAndRotate`. `right`, `down` and
250/// `forward` are the cell's column, row and aisle.
251#[must_use]
252fn translate_and_rotate(
253    origin: BlockPos,
254    forwards: Direction,
255    up: Direction,
256    right: i32,
257    down: i32,
258    forward: i32,
259) -> BlockPos {
260    let forwards_vector = forwards.offset_vec();
261    let up_vector = up.offset_vec();
262    let right_vector = forwards_vector.cross(up_vector);
263    origin.offset(
264        up_vector.x * -down + right_vector.x * right + forwards_vector.x * forward,
265        up_vector.y * -down + right_vector.y * right + forwards_vector.y * forward,
266        up_vector.z * -down + right_vector.z * right + forwards_vector.z * forward,
267    )
268}

270/// A pattern that was found in the world, with the frame to read it back.
271///
272/// Vanilla parity: `BlockPattern.BlockPatternMatch`.
273pub struct BlockPatternMatch<'a> {
274    front_top_left: BlockPos,
275    forwards: Direction,
276    up: Direction,
277    cache: LevelCache<'a>,
278    width: usize,
279    height: usize,
280    depth: usize,
281}
282
283impl<'a> BlockPatternMatch<'a> {
284    /// Returns the position the pattern's first cell landed on.
285    #[must_use]
286    pub const fn front_top_left(&self) -> BlockPos {
287        self.front_top_left
288    }
289
290    /// Returns the direction the pattern's aisles run in.
291    #[must_use]
292    pub const fn forwards(&self) -> Direction {
293        self.forwards
294    }
295
296    /// Returns the direction the pattern's rows count down from.
297    #[must_use]
298    pub const fn up(&self) -> Direction {
299        self.up
300    }
301
302    /// Returns how many columns wide the matched pattern is.
303    #[must_use]
304    pub const fn width(&self) -> usize {
305        self.width
306    }
307
308    /// Returns how many rows tall the matched pattern is.
309    #[must_use]
310    pub const fn height(&self) -> usize {
311        self.height
312    }
313
314    /// Returns how many aisles deep the matched pattern is.
315    #[must_use]
316    pub const fn depth(&self) -> usize {
317        self.depth
318    }
319
320    /// Returns the block one pattern cell landed on.
321    ///
322    /// Vanilla parity: `BlockPatternMatch.getBlock`.
323    #[must_use]
324    pub fn block(&self, right: i32, down: i32, forward: i32) -> BlockInWorld<'a> {
325        self.cache.get(translate_and_rotate(
326            self.front_top_left,
327            self.forwards,
328            self.up,
329            right,
330            down,
331            forward,
332        ))
333    }
334}

336/// Builds a [`BlockPattern`] from character aisles.
337///
338/// Vanilla parity: `BlockPatternBuilder`.
339pub struct BlockPatternBuilder {
340    /// One entry per aisle, each a grid of `[height][width]` characters.
341    pattern: Vec<Vec<Vec<char>>>,
342    lookup: FxHashMap<char, BlockPatternPredicate>,
343    height: usize,
344    width: usize,
345    unknown_characters: BTreeSet<char>,
346}
347
348impl BlockPatternBuilder {
349    /// Starts an empty pattern.
350    ///
351    /// Vanilla parity: `BlockPatternBuilder.start`. A space always matches.
352    #[must_use]
353    pub fn start() -> Self {
354        let mut lookup: FxHashMap<char, BlockPatternPredicate> = FxHashMap::default();
355        lookup.insert(' ', Arc::new(|_: &BlockInWorld<'_>| true));
356        Self {
357            pattern: Vec::new(),
358            lookup,
359            height: 0,
360            width: 0,
361            unknown_characters: BTreeSet::new(),
362        }
363    }
364
365    /// Appends one aisle, given as its rows from top to bottom.
366    ///
367    /// Vanilla parity: `BlockPatternBuilder.aisle`.
368    ///
369    /// # Panics
370    ///
371    /// Panics when the aisle is empty or does not have the same dimensions as
372    /// the aisles already added. Patterns are written as literals in Foton's
373    /// own code, so a mismatch is a programming error rather than bad input.
374    #[must_use]
375    pub fn aisle(mut self, aisle: &[&str]) -> Self {
376        assert!(
377            !aisle.is_empty() && !aisle[0].is_empty(),
378            "empty pattern for aisle"
379        );
380
381        if self.pattern.is_empty() {
382            self.height = aisle.len();
383            self.width = aisle[0].chars().count();
384        }
385
386        assert_eq!(
387            aisle.len(),
388            self.height,
389            "aisle heights must all match the first aisle"
390        );
391
392        let mut rows = Vec::with_capacity(aisle.len());
393        for row in aisle {
394            let chars: Vec<char> = row.chars().collect();
395            assert_eq!(
396                chars.len(),
397                self.width,
398                "aisle row widths must all match the first aisle"
399            );
400
401            for &character in &chars {
402                if !self.lookup.contains_key(&character) {
403                    self.unknown_characters.insert(character);
404                }
405            }
406
407            rows.push(chars);
408        }
409
410        self.pattern.push(rows);
411        self
412    }
413
414    /// Binds `character` to a predicate.
415    ///
416    /// Vanilla parity: `BlockPatternBuilder.where`.
417    #[must_use]
418    pub fn where_char(mut self, character: char, predicate: BlockPatternPredicate) -> Self {
419        self.lookup.insert(character, predicate);
420        self.unknown_characters.remove(&character);
421        self
422    }
423
424    /// Finishes the pattern.
425    ///
426    /// Vanilla parity: `BlockPatternBuilder.build`.
427    ///
428    /// # Panics
429    ///
430    /// Panics when a character used in an aisle has no predicate.
431    #[must_use]
432    pub fn build(self) -> BlockPattern {
433        assert!(
434            self.unknown_characters.is_empty(),
435            "predicates for some pattern characters are missing: {:?}",
436            self.unknown_characters
437        );
438
439        let never: BlockPatternPredicate = Arc::new(|_: &BlockInWorld<'_>| false);
440        let pattern = self
441            .pattern
442            .iter()
443            .map(|aisle| {
444                aisle
445                    .iter()
446                    .map(|row| {
447                        row.iter()
448                            .map(|character| {
449                                // `aisle` rejects characters with no predicate,
450                                // so the fallback is unreachable.
451                                self.lookup
452                                    .get(character)
453                                    .map_or_else(|| Arc::clone(&never), Arc::clone)
454                            })
455                            .collect()
456                    })
457                    .collect()
458            })
459            .collect();
460
461        BlockPattern {
462            depth: self.pattern.len(),
463            height: self.height,
464            width: self.width,
465            pattern,
466        }
467    }
