1//! The `/bug` form, as a server-built dialog.
2//!
3//! Minecraft has carried a server-driven dialog system since 1.21.6, so a bug
4//! form needs no client mod: the server sends the whole screen and the client
5//! sends back what was typed. That matters for a public test session, where
6//! asking testers to install anything loses most of them.
7//!
8//! The dialog is built here rather than registered as a datapack entry because
9//! it belongs to one command at one moment. `ClientboundShowDialogPacket`
10//! carries a `Holder<Dialog>`, and the inline half of that holder is exactly
11//! this case.
12
13use foton_protocol::packets::common::CShowDialog;
14use foton_utils::Identifier;
15use simdnbt::owned::{NbtCompound, NbtList, NbtTag};
16
17use crate::bug_report::{BugCategory, MAX_DESCRIPTION};

19/// The action id the submit button carries back.
20///
21/// Vanilla parity: the `id` of a `minecraft:dynamic/custom` action, which the
22/// client returns in `ServerboundCustomClickActionPacket`.
23pub const BUG_REPORT_ACTION: &str = "bug_report";
24
25/// The dialog input holding what the reporter typed.
26pub const DESCRIPTION_KEY: &str = "description";
27
28/// The dialog input holding the chosen category.
29pub const CATEGORY_KEY: &str = "category";
30
31/// A plain text component, in the shape the dialog codec reads.
32fn text(value: &str) -> NbtTag {
33    let mut compound = NbtCompound::new();
34    compound.insert("text", value);
35    NbtTag::Compound(compound)
36}
37
38/// The category dropdown, one entry per [`BugCategory`].
39///
40/// The list is closed on purpose: a free-text category is unsortable within a
41/// day of a real test session.
42fn category_input() -> NbtCompound {
43    let mut options = Vec::new();
44    for (index, category) in BugCategory::ALL.into_iter().enumerate() {
45        let mut option = NbtCompound::new();
46        option.insert("id", category.name());
47        option.insert("display", text(category.label()));
48        // Vanilla rejects a dialog with more than one initial option, so only
49        // the first may claim it.
50        option.insert("initial", i8::from(index == 0));
51        options.push(option);
52    }
53
54    let mut input = NbtCompound::new();
55    input.insert("key", CATEGORY_KEY);
56    input.insert("type", "minecraft:single_option");
57    input.insert("label", text("What does it affect?"));
58    input.insert("width", 300);
59    input.insert("options", NbtList::Compound(options));
60    input
61}
62
63/// The multi-line box the reproduction steps go in.
64fn description_input() -> NbtCompound {
65    let mut multiline = NbtCompound::new();
66    multiline.insert("max_lines", 12);
67    multiline.insert("height", 128);
68
69    let mut input = NbtCompound::new();
70    input.insert("key", DESCRIPTION_KEY);
71    input.insert("type", "minecraft:text");
72    input.insert("label", text("What happened, and how do we see it again?"));
73    input.insert("width", 300);
74    input.insert(
75        "max_length",
76        i32::try_from(MAX_DESCRIPTION).unwrap_or(i32::MAX),
77    );
78    input.insert("multiline", NbtTag::Compound(multiline));
79    input
80}

82/// The submit button, carrying the action that returns the form.
83fn submit_button() -> NbtCompound {
84    let mut action = NbtCompound::new();
85    action.insert("type", "minecraft:dynamic/custom");
86    action.insert("id", Identifier::from_foton(BUG_REPORT_ACTION).to_string());
87
88    let mut button = NbtCompound::new();
89    button.insert("label", text("Send report"));
90    button.insert("width", 150);
91    button.insert("action", NbtTag::Compound(action));
92    button
93}

95/// Builds the `/bug` dialog.
96#[must_use]
97pub fn bug_dialog() -> NbtCompound {
98    let mut body = NbtCompound::new();
99    body.insert("type", "minecraft:plain_message");
100    body.insert(
101        "contents",
102        text("Where you are, which world, and which build are recorded for you."),
103    );
104
105    let mut dialog = NbtCompound::new();
106    dialog.insert("type", "minecraft:multi_action");
107    dialog.insert("title", text("Report a bug"));
108    dialog.insert("can_close_with_escape", 1_i8);
109    // A paused dialog freezes the client's own screen but not the server, and
110    // on a shared test server that only makes the reporter miss what happens
111    // while they type.
112    dialog.insert("pause", 0_i8);
113    dialog.insert("body", NbtList::Compound(vec![body]));
114    dialog.insert(
115        "inputs",
116        NbtList::Compound(vec![category_input(), description_input()]),
117    );
118    dialog.insert("actions", NbtList::Compound(vec![submit_button()]));
119    dialog.insert("columns", 1);
120    dialog
121}
122
123/// The packet that puts the form on a player's screen.
124#[must_use]
125pub fn show_bug_dialog() -> CShowDialog {
126    CShowDialog::inline(bug_dialog())
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132
133    /// The dialog names every field the handler later reads.
134    ///
135    /// A dialog whose input keys drift from the handler's still opens, still
136    /// accepts typing, and silently files empty reports -- the failure has no
137    /// symptom at the point it happens.
138    #[test]
139    fn the_form_carries_the_keys_the_handler_reads() {
140        let dialog = bug_dialog();
141        let inputs = dialog.list("inputs").expect("the form has inputs");
142        let compounds = inputs.compounds().expect("inputs are compounds");
143        let keys: Vec<_> = compounds
144            .iter()
145            .filter_map(|input| input.string("key").map(ToString::to_string))
146            .collect();
147        assert!(keys.contains(&CATEGORY_KEY.to_owned()));
148        assert!(keys.contains(&DESCRIPTION_KEY.to_owned()));
149    }
150
151    /// Exactly one option may be initial; vanilla rejects the dialog otherwise.
152    #[test]
153    fn exactly_one_category_starts_selected() {
154        let input = category_input();
155        let options = input.list("options").expect("options");
156        let initial = options
157            .compounds()
158            .expect("compounds")
159            .iter()
160            .filter(|option| option.byte("initial").is_some_and(|value| value != 0))
161            .count();
162        assert_eq!(initial, 1, "vanilla refuses a dialog with several initials");
163    }
164
165    /// Every category the storage knows is offered by the form.
166    #[test]
167    fn the_form_offers_every_category() {
168        let input = category_input();
169        let options = input.list("options").expect("options");
170        assert_eq!(
171            options.compounds().expect("compounds").len(),
172            BugCategory::ALL.len(),
173            "a category the form cannot reach may as well not exist"
174        );
175    }
176}
