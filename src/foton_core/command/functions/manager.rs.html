17/// The tag whose functions run once after every load.
18pub(crate) const LOAD_FUNCTION_TAG: Identifier = Identifier::vanilla_static("load");
19/// The tag whose functions run at the start of every normally-running tick.
20pub(crate) const TICK_FUNCTION_TAG: Identifier = Identifier::vanilla_static("tick");

22/// What one datapack load produced, for the log line and for `/reload`.
23#[derive(Debug, Default, PartialEq, Eq)]
24pub(crate) struct FunctionReloadReport {
25    pub(crate) functions: usize,
26    pub(crate) tags: usize,
27    pub(crate) errors: Vec<String>,
28}

30/// The server's command functions and the tags that drive them.
31///
32/// Vanilla parity: `ServerFunctionManager` over a `ServerFunctionLibrary`. A
33/// reload builds a whole new library and swaps it in, so a call that is already
34/// running keeps the entries it started with.
35pub(crate) struct FunctionManager {
36    root: PathBuf,
37    state: SyncRwLock<FunctionManagerState>,
38}
39
40struct FunctionManagerState {
41    library: Arc<FunctionLibrary>,
42    ticking: Vec<Arc<CommandFunction>>,
43    packs: Vec<DatapackInfo>,
44    post_reload: bool,
45}
46
47impl FunctionManager {
48    pub(crate) fn new(root: PathBuf) -> Self {
49        Self {
50            root,
51            state: SyncRwLock::new(FunctionManagerState {
52                library: Arc::new(FunctionLibrary::default()),
53                ticking: Vec::new(),
54                packs: Vec::new(),
55                post_reload: false,
56            }),
57        }
58    }
59
60    pub(crate) fn root(&self) -> &Path {
61        &self.root
62    }
63
64    /// Reads every datapack and replaces the loaded library with the result.
65    pub(crate) fn reload(
66        &self,
67        dispatcher: &CommandDispatcher,
68        compilation_source: &CommandSource,
69    ) -> FunctionReloadReport {
70        let contents = loader::collect(&self.root);
71        let mut errors = contents.errors;
72
73        let mut functions: FxHashMap<Identifier, Arc<CommandFunction>> = FxHashMap::default();
74        for (id, source) in contents.functions {
75            match parse_function(id.clone(), &source.text, dispatcher, compilation_source) {
76                Ok(function) => {
77                    functions.insert(id, Arc::new(function));
78                }
79                Err(error) => errors.push(format!(
80                    "failed to load function {id} from datapack {}: {error}",
81                    source.source_pack
82                )),
83            }
84        }
85
86        let tags = build_tags(&contents.tags, &functions, &mut errors);
87        let library = Arc::new(FunctionLibrary::new(functions, tags));
88        let report = FunctionReloadReport {
89            functions: library.function_count(),
90            tags: library.tag_count(),
91            errors,
92        };
93
94        let ticking = library.tag(&TICK_FUNCTION_TAG).to_vec();
95        let mut state = self.state.write();
96        state.library = library;
97        state.ticking = ticking;
98        state.packs = contents.packs;
99        state.post_reload = true;
100        report
101    }
102
103    /// Drops every loaded function.
104    ///
105    /// A parsed function holds the command source it was compiled with, and that
106    /// source holds the server, so the loaded library keeps the server alive.
107    /// Shutdown breaks that cycle here rather than leaking the whole server.
108    pub(crate) fn unload(&self) {
109        let mut state = self.state.write();
110        state.library = Arc::new(FunctionLibrary::default());
111        state.ticking.clear();
112        state.packs.clear();
113        state.post_reload = false;
114    }
115
116    pub(crate) fn library(&self) -> Arc<FunctionLibrary> {
117        Arc::clone(&self.state.read().library)
118    }
119
120    /// Returns the `#minecraft:load` functions once per reload.
121    pub(crate) fn take_load_functions(&self) -> Vec<Arc<CommandFunction>> {
122        let mut state = self.state.write();
123        if !state.post_reload {
124            return Vec::new();
125        }
126        state.post_reload = false;
127        state.library.tag(&LOAD_FUNCTION_TAG).to_vec()
128    }
129
130    /// Returns the `#minecraft:tick` functions cached by the last reload.
131    pub(crate) fn ticking_functions(&self) -> Vec<Arc<CommandFunction>> {
132        self.state.read().ticking.clone()
133    }
134
135    pub(crate) fn datapack_records(&self, enabled_only: bool) -> Vec<String> {
136        self.state
137            .read()
138            .packs
139            .iter()
140            .filter(|pack| !enabled_only || pack.enabled)
141            .map(|pack| format!("{}\t{}\t{}", pack.name, pack.compatibility, pack.enabled))
142            .collect()
143    }
144}

146/// Resolves raw tag entries into function lists, in dependency order.
147///
148/// Vanilla parity: `TagLoader.build`. A tag with an unresolved required entry is
149/// dropped with an error instead of becoming an empty tag, so a typo in a tag
150/// file cannot silently turn `#minecraft:tick` into "run nothing".
151pub(super) fn build_tags(
152    raw: &FxHashMap<Identifier, Vec<TagEntry>>,
153    functions: &FxHashMap<Identifier, Arc<CommandFunction>>,
154    errors: &mut Vec<String>,
155) -> FxHashMap<Identifier, Arc<[Arc<CommandFunction>]>> {
156    let mut built: FxHashMap<Identifier, Arc<[Arc<CommandFunction>]>> = FxHashMap::default();
157    let mut failed: FxHashSet<Identifier> = FxHashSet::default();
158    let mut visiting: Vec<Identifier> = Vec::new();
159    for id in raw.keys() {
160        resolve_tag(
161            id,
162            raw,
163            functions,
164            &mut built,
165            &mut failed,
166            &mut visiting,
167            errors,
168        );
169    }
170    built
171}

173fn resolve_tag(
174    id: &Identifier,
175    raw: &FxHashMap<Identifier, Vec<TagEntry>>,
176    functions: &FxHashMap<Identifier, Arc<CommandFunction>>,
177    built: &mut FxHashMap<Identifier, Arc<[Arc<CommandFunction>]>>,
178    failed: &mut FxHashSet<Identifier>,
179    visiting: &mut Vec<Identifier>,
180    errors: &mut Vec<String>,
181) -> Option<Arc<[Arc<CommandFunction>]>> {
182    if let Some(tag) = built.get(id) {
183        return Some(Arc::clone(tag));
184    }
185    if failed.contains(id) {
186        return None;
187    }
188    let entries = raw.get(id)?;
189    if visiting.contains(id) {
190        errors.push(format!(
191            "couldn't load function tag {id}: it is part of a reference cycle"
192        ));
193        failed.insert(id.clone());
194        return None;
195    }
196
197    visiting.push(id.clone());
198    let mut values: Vec<Arc<CommandFunction>> = Vec::new();
199    let mut seen: FxHashSet<Identifier> = FxHashSet::default();
200    let mut missing: Vec<String> = Vec::new();
201
202    for entry in entries {
203        if entry.references_tag {
204            let referenced =
205                resolve_tag(&entry.id, raw, functions, built, failed, visiting, errors);
206            match referenced {
207                Some(referenced) => {
208                    for function in referenced.iter() {
209                        if seen.insert(function.id().clone()) {
210                            values.push(Arc::clone(function));
211                        }
212                    }
213                }
214                None if entry.required => {
215                    missing.push(format!("#{} (from {})", entry.id, entry.source_pack));
216                }
217                None => {}
218            }
219            continue;
220        }
221        match functions.get(&entry.id) {
222            Some(function) => {
223                if seen.insert(function.id().clone()) {
224                    values.push(Arc::clone(function));
225                }
226            }
227            None if entry.required => {
228                missing.push(format!("{} (from {})", entry.id, entry.source_pack));
229            }
230            None => {}
231        }
232    }
233    visiting.pop();
234
235    if !missing.is_empty() {
236        errors.push(format!(
237            "couldn't load function tag {id} as it is missing the following references: {}",
238            missing.join(", ")
239        ));
240        failed.insert(id.clone());
241        return None;
242    }
243
244    let tag: Arc<[Arc<CommandFunction>]> = values.into();
245    built.insert(id.clone(), Arc::clone(&tag));
246    Some(tag)
247}
