15/// One loaded `.mcfunction`.
16///
17/// Vanilla parity: `CommandFunction`. A file with no macro lines is already
18/// runnable; one with macro lines only becomes runnable once a call supplies
19/// its arguments, which is what [`Self::instantiate`] is for.
20pub(crate) struct CommandFunction {
21    id: Identifier,
22    body: FunctionBody<CommandSource>,
23}
24
25impl CommandFunction {
26    pub(crate) const fn new(id: Identifier, body: FunctionBody<CommandSource>) -> Self {
27        Self { id, body }
28    }
29
30    pub(crate) const fn id(&self) -> &Identifier {
31        &self.id
32    }
33
34    /// Returns the command actions this call should run, in file order.
35    ///
36    /// Vanilla parity: `CommandFunction.instantiate`. The error is the reason a
37    /// macro could not be filled in, which the caller wraps in the command
38    /// error its own message uses.
39    pub(crate) fn instantiate(
40        &self,
41        arguments: Option<&NbtCompound>,
42        dispatcher: &CommandDispatcher<CommandSource, FotonCommandRuntime>,
43    ) -> Result<FunctionEntries<CommandSource>, Box<TextComponent>> {
44        match &self.body {
45            FunctionBody::Plain(entries) => Ok(Arc::clone(entries)),
46            FunctionBody::Macro(function) => function.instantiate(&self.id, arguments, dispatcher),
47        }
48    }
49}
50
51/// Every function and function tag one datapack load produced.
52///
53/// Vanilla parity: `ServerFunctionLibrary`. A reload builds a whole new library
54/// and swaps it in, so a running function keeps the entries it started with.
55#[derive(Default)]
56pub(crate) struct FunctionLibrary {
57    functions: FxHashMap<Identifier, Arc<CommandFunction>>,
58    tags: FxHashMap<Identifier, Arc<[Arc<CommandFunction>]>>,
59    item_modifiers: FxHashMap<Identifier, Arc<ItemModifier>>,
60}
61
62impl FunctionLibrary {
63    pub(crate) fn new(
64        functions: FxHashMap<Identifier, Arc<CommandFunction>>,
65        tags: FxHashMap<Identifier, Arc<[Arc<CommandFunction>]>>,
66    ) -> Self {
67        Self {
68            functions,
69            tags,
70            item_modifiers: FxHashMap::default(),
71        }
72    }
73
74    /// Adds the item modifiers the same datapack load produced.
75    pub(crate) fn with_item_modifiers(
76        mut self,
77        item_modifiers: FxHashMap<Identifier, Arc<ItemModifier>>,
78    ) -> Self {
79        self.item_modifiers = item_modifiers;
80        self
81    }
82
83    pub(crate) fn item_modifier(&self, id: &Identifier) -> Option<&Arc<ItemModifier>> {
84        self.item_modifiers.get(id)
85    }
86
87    pub(crate) fn item_modifier_names(&self) -> impl Iterator<Item = &Identifier> {
88        self.item_modifiers.keys()
89    }
90
91    pub(crate) fn function(&self, id: &Identifier) -> Option<&Arc<CommandFunction>> {
92        self.functions.get(id)
93    }
94
95    /// Returns a tag's functions, or an empty slice when the tag does not exist.
96    ///
97    /// A tag whose entries could not all be resolved is absent rather than
98    /// empty, matching `TagLoader.build`: an unresolvable tag is dropped with an
99    /// error instead of silently becoming a tag that runs nothing.
100    pub(crate) fn tag(&self, id: &Identifier) -> &[Arc<CommandFunction>] {
101        self.tags.get(id).map_or(&[], |tag| &**tag)
102    }
103
104    pub(crate) fn function_names(&self) -> impl Iterator<Item = &Identifier> {
105        self.functions.keys()
106    }
107
108    pub(crate) fn tag_names(&self) -> impl Iterator<Item = &Identifier> {
109        self.tags.keys()
110    }
111
112    pub(crate) fn function_count(&self) -> usize {
113        self.functions.len()
114    }
115
116    pub(crate) fn tag_count(&self) -> usize {
117        self.tags.len()
118    }
