14/// One loaded `.mcfunction`.
15///
16/// Vanilla parity: `CommandFunction`. A file with no macro lines is already
17/// runnable; one with macro lines only becomes runnable once a call supplies
18/// its arguments, which is what [`Self::instantiate`] is for.
19pub(crate) struct CommandFunction {
20    id: Identifier,
21    body: FunctionBody<CommandSource>,
22}
23
24impl CommandFunction {
25    pub(crate) const fn new(id: Identifier, body: FunctionBody<CommandSource>) -> Self {
26        Self { id, body }
27    }
28
29    pub(crate) const fn id(&self) -> &Identifier {
30        &self.id
31    }
32
33    /// Returns the command actions this call should run, in file order.
34    ///
35    /// Vanilla parity: `CommandFunction.instantiate`. The error is the reason a
36    /// macro could not be filled in, which the caller wraps in the command
37    /// error its own message uses.
38    pub(crate) fn instantiate(
39        &self,
40        arguments: Option<&NbtCompound>,
41        dispatcher: &CommandDispatcher<CommandSource, FotonCommandRuntime>,
42    ) -> Result<FunctionEntries<CommandSource>, Box<TextComponent>> {
43        match &self.body {
44            FunctionBody::Plain(entries) => Ok(Arc::clone(entries)),
45            FunctionBody::Macro(function) => function.instantiate(&self.id, arguments, dispatcher),
46        }
47    }
48}

50/// Every function and function tag one datapack load produced.
51///
52/// Vanilla parity: `ServerFunctionLibrary`. A reload builds a whole new library
53/// and swaps it in, so a running function keeps the entries it started with.
54#[derive(Default)]
55pub(crate) struct FunctionLibrary {
56    functions: FxHashMap<Identifier, Arc<CommandFunction>>,
57    tags: FxHashMap<Identifier, Arc<[Arc<CommandFunction>]>>,
58}
59
60impl FunctionLibrary {
61    pub(crate) const fn new(
62        functions: FxHashMap<Identifier, Arc<CommandFunction>>,
63        tags: FxHashMap<Identifier, Arc<[Arc<CommandFunction>]>>,
64    ) -> Self {
65        Self { functions, tags }
66    }
67
68    pub(crate) fn function(&self, id: &Identifier) -> Option<&Arc<CommandFunction>> {
69        self.functions.get(id)
70    }
71
72    /// Returns a tag's functions, or an empty slice when the tag does not exist.
73    ///
74    /// A tag whose entries could not all be resolved is absent rather than
75    /// empty, matching `TagLoader.build`: an unresolvable tag is dropped with an
76    /// error instead of silently becoming a tag that runs nothing.
77    pub(crate) fn tag(&self, id: &Identifier) -> &[Arc<CommandFunction>] {
78        self.tags.get(id).map_or(&[], |tag| &**tag)
79    }
80
81    pub(crate) fn function_names(&self) -> impl Iterator<Item = &Identifier> {
82        self.functions.keys()
83    }
84
85    pub(crate) fn tag_names(&self) -> impl Iterator<Item = &Identifier> {
86        self.tags.keys()
87    }
88
89    pub(crate) fn function_count(&self) -> usize {
90        self.functions.len()
91    }
92
93    pub(crate) fn tag_count(&self) -> usize {
94        self.tags.len()
95    }
96}
