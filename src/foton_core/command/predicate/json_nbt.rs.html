1//! JSON to NBT, the way vanilla's `Dynamic.convert(NbtOps.INSTANCE)` does it.
2
3use serde_json::Value;
4use simdnbt::owned::{NbtCompound, NbtList, NbtTag};
5
6/// Converts datapack JSON to NBT.
7///
8/// Vanilla parity: `JsonOps.convertTo`. A JSON integer becomes the *smallest*
9/// of byte, short, int and long that holds it, and a number with a fractional
10/// part becomes a float if that is exact and a double otherwise. Component
11/// values that are compared whole -- `custom_data` above all -- are compared
12/// with those types, so `{"nuke": 1}` in a predicate means `nuke: 1b`, and only
13/// matches an item that stores a byte.
14pub(super) fn json_to_nbt(path: &str, value: &Value) -> Result<NbtTag, String> {
15    match value {
16        Value::Null => Err(format!("{path}: null has no NBT form")),
17        Value::Bool(flag) => Ok(NbtTag::Byte(i8::from(*flag))),
18        Value::Number(number) => Ok(number_tag(number)),
19        Value::String(text) => Ok(NbtTag::String(text.as_str().into())),
20        Value::Array(items) => list_tag(path, items).map(NbtTag::List),
21        Value::Object(entries) => {
22            let mut compound = NbtCompound::new();
23            for (key, entry) in entries {
24                compound.insert(key.as_str(), json_to_nbt(&format!("{path}.{key}"), entry)?);
25            }
26            Ok(NbtTag::Compound(compound))
27        }
28    }
29}

31#[expect(
32    clippy::cast_possible_truncation,
33    clippy::float_cmp,
34    reason = "each narrowing is guarded by an exact round-trip, as in JsonOps.convertTo"
35)]
36fn number_tag(number: &serde_json::Number) -> NbtTag {
37    if let Some(integer) = number
38        .as_i64()
39        .or_else(|| number.as_u64().and_then(|big| i64::try_from(big).ok()))
40    {
41        return if i64::from(integer as i8) == integer {
42            NbtTag::Byte(integer as i8)
43        } else if i64::from(integer as i16) == integer {
44            NbtTag::Short(integer as i16)
45        } else if i64::from(integer as i32) == integer {
46            NbtTag::Int(integer as i32)
47        } else {
48            NbtTag::Long(integer)
49        };
50    }
51    let double = number.as_f64().unwrap_or(f64::NAN);
52    if f64::from(double as f32) == double {
53        NbtTag::Float(double as f32)
54    } else {
55        NbtTag::Double(double)
56    }
57}

59fn list_tag(path: &str, items: &[Value]) -> Result<NbtList, String> {
60    let tags = items
61        .iter()
62        .enumerate()
63        .map(|(index, item)| json_to_nbt(&format!("{path}[{index}]"), item))
64        .collect::<Result<Vec<_>, _>>()?;
65    let mixed = || format!("{path}: a list mixing different value types is not supported");
66    macro_rules! homogeneous {
67        ($variant:ident, $list:ident) => {
68            tags.iter()
69                .map(|tag| match tag {
70                    NbtTag::$variant(value) => Ok(value.clone()),
71                    _ => Err(mixed()),
72                })
73                .collect::<Result<Vec<_>, _>>()
74                .map(NbtList::$list)
75        };
76    }
77    match tags.first() {
78        None => Ok(NbtList::Empty),
79        Some(NbtTag::Byte(_)) => homogeneous!(Byte, Byte),
80        Some(NbtTag::Short(_)) => homogeneous!(Short, Short),
81        Some(NbtTag::Int(_)) => homogeneous!(Int, Int),
82        Some(NbtTag::Long(_)) => homogeneous!(Long, Long),
83        Some(NbtTag::Float(_)) => homogeneous!(Float, Float),
84        Some(NbtTag::Double(_)) => homogeneous!(Double, Double),
85        Some(NbtTag::String(_)) => homogeneous!(String, String),
86        Some(NbtTag::List(_)) => homogeneous!(List, List),
87        Some(NbtTag::Compound(_)) => homogeneous!(Compound, Compound),
88        Some(_) => Err(mixed()),
89    }
90}

92#[cfg(test)]
93mod tests {
94    use super::*;
95
96    #[test]
97    fn integers_take_the_smallest_type_that_holds_them() {
98        let tag = json_to_nbt(
99            "t",
100            &serde_json::json!({"a": 1, "b": 300, "c": 70000, "d": 5_000_000_000_i64, "e": 0.5}),
101        )
102        .expect("converts");
103        let NbtTag::Compound(compound) = tag else {
104            panic!("an object should become a compound");
105        };
106        assert_eq!(compound.get("a"), Some(&NbtTag::Byte(1)));
107        assert_eq!(compound.get("b"), Some(&NbtTag::Short(300)));
108        assert_eq!(compound.get("c"), Some(&NbtTag::Int(70_000)));
109        assert_eq!(compound.get("d"), Some(&NbtTag::Long(5_000_000_000)));
110        assert_eq!(compound.get("e"), Some(&NbtTag::Float(0.5)));
111    }
