1//! Lowers a datapack `predicate/*.json` file into the typed predicates of
2//! [`foton_registry::advancement::predicate`].
3//!
4//! Vanilla parity: `LootItemCondition.DIRECT_CODEC`, which reads either one
5//! typed condition or a list of them (an implicit `all_of`), with the entity
6//! sub-predicate map of `EntityPredicate.CODEC` inside.
7//!
8//! Every object is read key by key through [`Object`], and anything left over
9//! is an error. A predicate Foton cannot model is therefore refused at load,
10//! with the path of the key, instead of loading as one that asks less than the
11//! datapack wrote and answering `true` for the wrong things.
12
13use foton_registry::advancement::predicate::{
14    BlockPredicate, ConditionTerm, ContextAwarePredicate, DistancePredicate, DoubleBounds,
15    EntityComponentMatch, EntityEquipmentPredicate, EntityFlagsPredicate, EntityPredicate,
16    IntBounds, ItemPredicate, LocationPredicate, RegistrySet, StatePropertyMatch,
17};
18use foton_registry::item_predicate::ItemPredicate as FullItemPredicate;
19use foton_utils::Identifier;
20use serde_json::{Map, Value};

24type Parsed<T> = Result<T, String>;

26/// Hands a parsed value to the lifetime the predicate types ask for.
27///
28/// The advancement predicate model is built from `&'static` data because its
29/// vanilla instances are generated at build time. [`super::PredicateCache`]
30/// interns by source text, so a `/reload` of unchanged files leaks nothing new.
31fn leak<T>(value: T) -> &'static T {
32    Box::leak(Box::new(value))
33}
34
35fn leak_slice<T>(values: Vec<T>) -> &'static [T] {
36    Vec::leak(values)
37}
38
39fn leak_str(text: &str) -> &'static str {
40    Box::leak(text.to_owned().into_boxed_str())
41}

43/// Parses a whole predicate file.
44pub(super) fn parse_predicate(value: &Value) -> Parsed<ContextAwarePredicate> {
45    match value {
46        Value::Array(list) => condition_list("predicate", list),
47        other => Ok(leak_slice(vec![condition("predicate", other)?])),
48    }
49}

51fn condition_list(path: &str, list: &[Value]) -> Parsed<&'static [ConditionTerm]> {
52    let terms = list
53        .iter()
54        .enumerate()
55        .map(|(index, term)| condition(&format!("{path}[{index}]"), term))
56        .collect::<Parsed<Vec<_>>>()?;
57    Ok(leak_slice(terms))
58}
59
60/// A condition that never passes: `AnyOf` with no alternatives.
61const NEVER: ConditionTerm = ConditionTerm::AnyOf(&[]);
62/// A condition that always passes: `AllOf` with no requirements.
63const ALWAYS: ConditionTerm = ConditionTerm::AllOf(&[]);
64
65fn condition(path: &str, value: &Value) -> Parsed<ConditionTerm> {
66    let mut object = Object::new(path, value)?;
67    let kind = object.string("condition")?;
68    let term = match strip_minecraft(&kind) {
69        "entity_properties" => {
70            let target = object.optional_string("entity")?;
71            let predicate = object.take("predicate");
72            // Vanilla `LootItemEntityPropertyCondition.test` passes outright
73            // without a predicate; `/execute if predicate` only supplies
74            // `this`, so any other target is a null entity that fails.
75            match (predicate, target.as_deref().map(strip_minecraft)) {
76                (None, _) => ALWAYS,
77                (Some(_), Some(other)) if other != "this" => NEVER,
78                (Some(predicate), _) => ConditionTerm::EntityProperties(leak(entity_predicate(
79                    &object.child("predicate"),
80                    &predicate,
81                )?)),
82            }
83        }
84        "location_check" => {
85            let offset_x = object.optional_i32("offsetX")?.unwrap_or(0);
86            let offset_y = object.optional_i32("offsetY")?.unwrap_or(0);
87            let offset_z = object.optional_i32("offsetZ")?.unwrap_or(0);
88            let predicate = object.take("predicate");
89            ConditionTerm::LocationCheck {
90                offset_x,
91                offset_y,
92                offset_z,
93                predicate: leak(match predicate {
94                    Some(predicate) => location_predicate(&object.child("predicate"), &predicate)?,
95                    None => LocationPredicate::ANY,
96                }),
97            }
98        }
99        // `/execute if predicate` has no TOOL and no BLOCK_STATE, and vanilla's
100        // `MatchTool` and `BlockStateProperty` both fail without theirs. They
101        // are still parsed so a malformed file is refused like any other.
102        "match_tool" => {
103            let predicate = object.take("predicate");
104            if let Some(predicate) = predicate {
105                item_predicate(&object.child("predicate"), &predicate)?;
106            }
107            NEVER
108        }
109        "block_state_property" => {
110            object.string("block")?;
111            if let Some(properties) = object.take("properties") {
112                state_properties(&object.child("properties"), &properties)?;
113            }
114            NEVER
115        }
116        "any_of" | "all_of" => {
117            let terms = object.take_required("terms")?;
118            let Value::Array(list) = &terms else {
119                return Err(format!(
120                    "{}: expected a list of conditions",
121                    object.child("terms")
122                ));
123            };
124            let terms = condition_list(&object.child("terms"), list)?;
125            if strip_minecraft(&kind) == "any_of" {
126                ConditionTerm::AnyOf(terms)
127            } else {
128                ConditionTerm::AllOf(terms)
129            }
130        }
131        "inverted" => {
132            let term = object.take_required("term")?;
133            ConditionTerm::Inverted(leak(condition(&object.child("term"), &term)?))
134        }
135        "random_chance" => {
136            let chance = object.take_required("chance")?;
137            ConditionTerm::RandomChance(constant_number(&object.child("chance"), &chance)?)
138        }
139        "weather_check" => ConditionTerm::WeatherCheck {
140            raining: object.optional_bool("raining")?,
141            thundering: object.optional_bool("thundering")?,
142        },
143        // Conditions that need a context parameter the command source does not
144        // carry, or a Foton system that does not exist yet, are refused by
145        // name instead of being guessed at.
146        other => {
147            return Err(format!(
148                "{path}: the condition `{other}` is not supported by Foton's datapack predicates"
149            ));
150        }
151    };
152    object.finish()?;
153    Ok(term)
154}

156/// Vanilla parity: `NumberProviders.CODEC`, narrowed to a plain number or the
157/// `minecraft:constant` provider, which is all a chance needs to be fixed.
158fn constant_number(path: &str, value: &Value) -> Parsed<f32> {
159    let number = match value {
160        Value::Number(number) => number.as_f64(),
161        Value::Object(_) => {
162            let mut object = Object::new(path, value)?;
163            let kind = object.optional_string("type")?;
164            if kind.as_deref().map_or("constant", strip_minecraft) != "constant" {
165                return Err(format!(
166                    "{path}: only constant number providers are supported"
167                ));
168            }
169            let value = object.take_required("value")?;
170            object.finish()?;
171            value.as_f64()
172        }
173        _ => None,
174    };
175    number
176        .map(|number| number as f32)
177        .ok_or_else(|| format!("{path}: expected a number"))
178}

180fn entity_predicate(path: &str, value: &Value) -> Parsed<EntityPredicate> {
181    let mut object = Object::new(path, value)?;
182    let mut predicate = EntityPredicate::ANY;
183
184    if let Some(types) = object.take_sub("entity_type") {
185        predicate.entity_type = Some(registry_set(&object.child("entity_type"), &types)?);
186    }
187    if let Some(location) = object.take_sub("location") {
188        let location = location_predicate(&object.child("location"), &location)?;
189        predicate.location = Some(leak(location));
190    }
191    if let Some(location) = object.take_sub("stepping_on") {
192        let location = location_predicate(&object.child("stepping_on"), &location)?;
193        predicate.stepping_on = Some(leak(location));
194    }
195    if let Some(distance) = object.take_sub("distance") {
196        predicate.distance = Some(distance_predicate(&object.child("distance"), &distance)?);
197    }
198    if let Some(flags) = object.take_sub("flags") {
199        predicate.flags = Some(flags_predicate(&object.child("flags"), &flags)?);
200    }
201    if let Some(equipment) = object.take_sub("equipment") {
202        let equipment = equipment_predicate(&object.child("equipment"), &equipment)?;
203        predicate.equipment = Some(leak(equipment));
204    }
205    if let Some(components) = object.take_sub("components") {
206        predicate.components = entity_components(&object.child("components"), &components)?;
207    }
208    if let Some(vehicle) = object.take_sub("vehicle") {
209        let vehicle = entity_predicate(&object.child("vehicle"), &vehicle)?;
210        predicate.vehicle = Some(leak(vehicle));
211    }
212    if let Some(passenger) = object.take_sub("passenger") {
213        let passenger = entity_predicate(&object.child("passenger"), &passenger)?;
214        predicate.passenger = Some(leak(passenger));
215    }
216    if let Some(player) = object.take_sub("type_specific/player") {
217        let path = object.child("type_specific/player");
218        let mut player = Object::new(&path, &player)?;
219        if let Some(target) = player.take("looking_at") {
220            let target = entity_predicate(&player.child("looking_at"), &target)?;
221            predicate.looking_at = Some(leak(target));
222        }
223        player.finish()?;
224    }
225    if let Some(lightning) = object.take_sub("type_specific/lightning") {
226        let path = object.child("type_specific/lightning");
227        let mut lightning = Object::new(&path, &lightning)?;
228        if let Some(blocks) = lightning.take("blocks_set_on_fire") {
229            predicate.lightning_blocks_set_on_fire =
230                Some(int_bounds(&lightning.child("blocks_set_on_fire"), &blocks)?);
231        }
232        lightning.finish()?;
233    }
234
235    object.finish()?;
236    Ok(predicate)
237}

253fn equipment_predicate(path: &str, value: &Value) -> Parsed<EntityEquipmentPredicate> {
254    let mut object = Object::new(path, value)?;
255    let mut slot = |key: &str| -> Parsed<Option<ItemPredicate>> {
256        object
257            .take(key)
258            .map(|stack| item_predicate(&format!("{path}.{key}"), &stack))
259            .transpose()
260    };
261    let equipment = EntityEquipmentPredicate {
262        head: slot("head")?,
263        chest: slot("chest")?,
264        legs: slot("legs")?,
265        feet: slot("feet")?,
266        mainhand: slot("mainhand")?,
267        offhand: slot("offhand")?,
268        body: slot("body")?,
269    };
270    object.finish()?;
271    Ok(equipment)
272}

274/// Vanilla parity: `ItemPredicate.CODEC`, decoded whole by the same code that
275/// reads item predicates out of lock components, so every `components` and
276/// `predicates` entry vanilla knows is honored rather than the few that the
277/// advancement model has fields for.
278fn item_predicate(path: &str, value: &Value) -> Parsed<ItemPredicate> {
279    if !value.is_object() {
280        return Err(format!("{path}: expected an object"));
281    }
282    let tag = json_to_nbt(path, value)?;
283    let full = FullItemPredicate::from_nbt(&tag)
284        .ok_or_else(|| format!("{path}: not a valid item predicate"))?;
285    Ok(ItemPredicate {
286        datapack: Some(leak(full)),
287        ..ItemPredicate::ANY
288    })
289}

291fn entity_components(path: &str, value: &Value) -> Parsed<&'static [EntityComponentMatch]> {
292    let object = Object::new(path, value)?;
293    let components = object
294        .entries
295        .iter()
296        .map(|(key, expected)| {
297            let Value::String(expected) = expected else {
298                return Err(format!(
299                    "{path}.{key}: only registry-keyed component values are supported"
300                ));
301            };
302            Ok(EntityComponentMatch {
303                component: leak_str(key),
304                value: identifier(&format!("{path}.{key}"), expected)?,
305            })
306        })
307        .collect::<Parsed<Vec<_>>>()?;
308    Ok(leak_slice(components))
309}

311fn location_predicate(path: &str, value: &Value) -> Parsed<LocationPredicate> {
312    let mut object = Object::new(path, value)?;
313    let mut predicate = LocationPredicate::ANY;
314    if let Some(position) = object.take("position") {
315        let mut position = Object::new(&object.child("position"), &position)?;
316        for (axis, target) in [
317            ("x", &mut predicate.x),
318            ("y", &mut predicate.y),
319            ("z", &mut predicate.z),
320        ] {
321            if let Some(bounds) = position.take(axis) {
322                *target = double_bounds(&position.child(axis), &bounds)?;
323            }
324        }
325        position.finish()?;
326    }
327    if let Some(biomes) = object.take("biomes") {
328        predicate.biomes = Some(registry_set(&object.child("biomes"), &biomes)?);
329    }
330    if let Some(structures) = object.take("structures") {
331        predicate.structures = Some(registry_set(&object.child("structures"), &structures)?);
332    }
333    if let Some(dimension) = object.optional_string("dimension")? {
334        predicate.dimension = Some(identifier(&object.child("dimension"), &dimension)?);
335    }
336    if let Some(block) = object.take("block") {
337        predicate.block = Some(block_predicate(&object.child("block"), &block)?);
338    }
339    predicate.smokey = object.optional_bool("smokey")?;
340    object.finish()?;
341    Ok(predicate)
342}

344fn block_predicate(path: &str, value: &Value) -> Parsed<BlockPredicate> {
345    let mut object = Object::new(path, value)?;
346    let blocks = object
347        .take("blocks")
348        .map(|blocks| registry_set(&object.child("blocks"), &blocks))
349        .transpose()?;
350    let state = match object.take("state") {
351        Some(state) => state_properties(&object.child("state"), &state)?,
352        None => &[],
353    };
354    object.finish()?;
355    Ok(BlockPredicate { blocks, state })
356}

358/// Vanilla parity: `StatePropertiesPredicate`, exact values only.
359fn state_properties(path: &str, value: &Value) -> Parsed<&'static [StatePropertyMatch]> {
360    let object = Object::new(path, value)?;
361    let properties = object
362        .entries
363        .iter()
364        .map(|(name, expected)| {
365            let expected = match expected {
366                Value::String(text) => text.clone(),
367                Value::Bool(flag) => flag.to_string(),
368                Value::Number(number) => number.to_string(),
369                _ => {
370                    return Err(format!(
371                        "{path}.{name}: ranged state values are not supported"
372                    ));
373                }
374            };
375            Ok(StatePropertyMatch {
376                name: leak_str(name),
377                value: leak_str(&expected),
378            })
379        })
380        .collect::<Parsed<Vec<_>>>()?;
381    Ok(leak_slice(properties))
382}

384fn distance_predicate(path: &str, value: &Value) -> Parsed<DistancePredicate> {
385    let mut object = Object::new(path, value)?;
386    let mut axis = |key: &str| -> Parsed<DoubleBounds> {
387        object.take(key).map_or(Ok(DoubleBounds::ANY), |value| {
388            double_bounds(&format!("{path}.{key}"), &value)
389        })
390    };
391    let distance = DistancePredicate {
392        x: axis("x")?,
393        y: axis("y")?,
394        z: axis("z")?,
395        horizontal: axis("horizontal")?,
396        absolute: axis("absolute")?,
397    };
398    object.finish()?;
399    Ok(distance)
400}

402/// Vanilla parity: the `HolderSet` codec -- `"#tag"`, one id, or a list of ids.
403fn registry_set(path: &str, value: &Value) -> Parsed<RegistrySet> {
404    match value {
405        Value::String(text) => match text.strip_prefix('#') {
406            Some(tag) => Ok(RegistrySet::Tag(identifier(path, tag)?)),
407            None => Ok(RegistrySet::Entries(leak_slice(vec![identifier(
408                path, text,
409            )?]))),
410        },
411        Value::Array(list) => {
412            let entries = list
413                .iter()
414                .map(|entry| match entry {
415                    Value::String(text) if !text.starts_with('#') => identifier(path, text),
416                    _ => Err(format!(
417                        "{path}: a list holds ids, not tags or other values"
418                    )),
419                })
420                .collect::<Parsed<Vec<_>>>()?;
421            Ok(RegistrySet::Entries(leak_slice(entries)))
422        }
423        _ => Err(format!("{path}: expected an id, a #tag or a list of ids")),
424    }
425}

427/// Vanilla parity: `MinMaxBounds`, a bare number or a `{min, max}` object.
428fn bounds<T: Copy>(
429    path: &str,
430    value: &Value,
431    read: impl Fn(&Value) -> Option<T>,
432) -> Parsed<(Option<T>, Option<T>)> {
433    if let Some(exact) = read(value) {
434        return Ok((Some(exact), Some(exact)));
435    }
436    let mut object = Object::new(path, value)?;
437    let mut bound = |key: &str| -> Parsed<Option<T>> {
438        object.take(key).map_or(Ok(None), |bound| {
439            read(&bound)
440                .map(Some)
441                .ok_or_else(|| format!("{path}.{key}: expected a number"))
442        })
443    };
444    let range = (bound("min")?, bound("max")?);
445    object.finish()?;
446    Ok(range)
447}

449fn double_bounds(path: &str, value: &Value) -> Parsed<DoubleBounds> {
450    let (min, max) = bounds(path, value, Value::as_f64)?;
451    Ok(DoubleBounds { min, max })
452}

454fn int_bounds(path: &str, value: &Value) -> Parsed<IntBounds> {
455    let read = |value: &Value| value.as_i64().and_then(|number| i32::try_from(number).ok());
456    let (min, max) = bounds(path, value, read)?;
457    Ok(IntBounds { min, max })
458}

469/// A JSON object read key by key, which refuses to leave any key behind.
470struct Object {
471    path: String,
472    entries: Map<String, Value>,
473}
474
475impl Object {
476    fn new(path: &str, value: &Value) -> Parsed<Self> {
477        match value {
478            Value::Object(entries) => Ok(Self {
479                path: path.to_owned(),
480                entries: entries.clone(),
481            }),
482            _ => Err(format!("{path}: expected an object")),
483        }
484    }
485
486    fn child(&self, key: &str) -> String {
487        format!("{}.{key}", self.path)
488    }
489
490    fn take(&mut self, key: &str) -> Option<Value> {
491        self.entries.remove(key)
492    }
493
494    /// Takes an entity sub-predicate, which may be spelled with or without the
495    /// `minecraft:` namespace because the map is keyed by registry id.
496    fn take_sub(&mut self, key: &str) -> Option<Value> {
497        self.entries
498            .remove(key)
499            .or_else(|| self.entries.remove(&format!("minecraft:{key}")))
500    }
501
502    fn take_required(&mut self, key: &str) -> Parsed<Value> {
503        self.take(key)
504            .ok_or_else(|| format!("{}: required key `{key}` is missing", self.path))
505    }
506
507    fn string(&mut self, key: &str) -> Parsed<String> {
508        self.optional_string(key)?
509            .ok_or_else(|| format!("{}: required key `{key}` is missing", self.path))
510    }
511
512    fn optional_string(&mut self, key: &str) -> Parsed<Option<String>> {
513        match self.take(key) {
514            None => Ok(None),
515            Some(Value::String(text)) => Ok(Some(text)),
516            Some(_) => Err(format!("{}: expected a string", self.child(key))),
517        }
518    }
519
520    fn optional_bool(&mut self, key: &str) -> Parsed<Option<bool>> {
521        match self.take(key) {
522            None => Ok(None),
523            Some(Value::Bool(flag)) => Ok(Some(flag)),
524            Some(_) => Err(format!("{}: expected a boolean", self.child(key))),
525        }
526    }
527
528    fn optional_i32(&mut self, key: &str) -> Parsed<Option<i32>> {
529        match self.take(key) {
530            None => Ok(None),
531            Some(value) => value
532                .as_i64()
533                .and_then(|number| i32::try_from(number).ok())
534                .map(Some)
535                .ok_or_else(|| format!("{}: expected an integer", self.child(key))),
536        }
537    }
538
539    fn finish(self) -> Parsed<()> {
540        if self.entries.is_empty() {
541            return Ok(());
542        }
543        Err(format!(
544            "{}: unsupported keys {:?}; Foton refuses the predicate rather than load one that \
545             asks for less than the datapack wrote",
546            self.path,
547            self.entries.keys().collect::<Vec<_>>()
548        ))
549    }
550}
