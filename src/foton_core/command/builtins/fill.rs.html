31/// Vanilla `FillCommand.Mode`: what happens to the positions of the region.
32#[derive(Clone, Copy, PartialEq, Eq)]
33enum FillMode {
34    Replace,
35    Outline,
36    Hollow,
37    Destroy,
38}
39
40/// Which positions of the region are touched.
41#[derive(Clone, Copy)]
42enum FillFilter<'a> {
43    Everything,
44    /// `fill ... keep`: vanilla's `level.isEmptyBlock(pos)`.
45    Air,
46    Predicate(&'a BlockPredicate),
47}

49/// Vanilla `BlockInput`: a state, the properties the command spelled out, and
50/// optional block-entity data.
51struct BlockInput {
52    state: BlockStateId,
53    properties: Vec<(Box<str>, Box<str>)>,
54    nbt: Option<NbtCompound>,
55}

57pub(super) fn registration() -> CommandRegistration<CommandSource> {
58    CommandRegistration::new(Identifier::vanilla_static("fill"), |_| command())

61fn command() -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
62    literal("fill").then(
63        argument("from", FotonArgumentType::block_pos()).then(
64            argument("to", FotonArgumentType::block_pos()).then(
65                wrap_with_mode(
66                    argument("block", FotonArgumentType::block_state()),
67                    FillFilterSource::Everything,
68                )
69                .then(
70                    literal("replace")
71                        .executes(|c| {
72                            fill(c, FillMode::Replace, FillFilterSource::Everything, false)
73                        })
74                        .then(wrap_with_mode(
75                            argument("filter", FotonArgumentType::block_predicate()),
76                            FillFilterSource::Argument,
77                        )),
78                )
79                .then(
80                    literal("keep")
81                        .executes(|c| fill(c, FillMode::Replace, FillFilterSource::Air, false)),
82                ),
83            ),
84        ),
85    )
86}

88/// Where the filter of a fill comes from, resolved once the context exists.
89#[derive(Clone, Copy)]
90enum FillFilterSource {
91    Everything,
92    Air,
93    Argument,
94}

96/// Vanilla `FillCommand.wrapWithMode`: the plain executor plus the four mode
97/// literals and `strict`.
98fn wrap_with_mode(
99    builder: CommandNodeBuilder<CommandSource, FotonCommandRuntime>,
100    filter: FillFilterSource,
101) -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
102    builder
103        .executes(move |c| fill(c, FillMode::Replace, filter, false))
104        .then(literal("outline").executes(move |c| fill(c, FillMode::Outline, filter, false)))
105        .then(literal("hollow").executes(move |c| fill(c, FillMode::Hollow, filter, false)))
106        .then(literal("destroy").executes(move |c| fill(c, FillMode::Destroy, filter, false)))
107        .then(literal("strict").executes(move |c| fill(c, FillMode::Replace, filter, true)))
108}

110/// Vanilla parity: `FillCommand.fillBlocks`.
111fn fill(
112    context: &FotonCommandContext<CommandSource>,
113    mode: FillMode,
114    filter: FillFilterSource,
115    strict: bool,
116) -> Result<i32, CommandSyntaxError> {
117    let from = loaded_block_pos(context, "from")?;
118    let to = loaded_block_pos(context, "to")?;
119    let target = block_input(context)?;
120    let filter = match filter {
121        FillFilterSource::Everything => FillFilter::Everything,
122        FillFilterSource::Air => FillFilter::Air,
123        FillFilterSource::Argument => FillFilter::Predicate(context.block_predicate("filter")?),
124    };
125    let world = context.source().world();
126
127    let (min, max) = (
128        BlockPos::new(
129            from.x().min(to.x()),
130            from.y().min(to.y()),
131            from.z().min(to.z()),
132        ),
133        BlockPos::new(
134            from.x().max(to.x()),
135            from.y().max(to.y()),
136            from.z().max(to.z()),
137        ),
138    );
139    let area = i64::from(max.x() - min.x() + 1)
140        * i64::from(max.y() - min.y() + 1)
141        * i64::from(max.z() - min.z() + 1);
142    let limit = world.get_game_rule(&MAX_BLOCK_MODIFICATIONS);
143    if area > i64::from(limit) {
144        return Err(CommandSyntaxError::dynamic(
145            translations::COMMANDS_FILL_TOOBIG
146                .message([limit.to_string(), area.to_string()])
147                .component(),
148        ));
149    }
150
151    let hollow_core = BlockInput {
152        state: vanilla_blocks::AIR.default_state(),
153        properties: Vec::new(),
154        nbt: None,
155    };
156    // Vanilla: `2 | (strict ? 816 : 256)`.
157    let flags = if strict {
158        UpdateFlags::UPDATE_CLIENTS
159            | UpdateFlags::UPDATE_SKIP_BLOCK_ENTITY_SIDEEFFECTS
160            | UpdateFlags::UPDATE_SKIP_ON_PLACE
161            | UpdateFlags::UPDATE_SUPPRESS_DROPS
162            | UpdateFlags::UPDATE_KNOWN_SHAPE
163    } else {
164        UpdateFlags::UPDATE_CLIENTS | UpdateFlags::UPDATE_SKIP_BLOCK_ENTITY_SIDEEFFECTS
165    };
166
167    let mut count = 0_i32;
168    let mut updates = Vec::new();
169    for pos in BlockPos::between_closed(min, max) {
170        if !filter_matches(world, filter, pos) {
171            continue;
172        }
173        let old_state = world.get_block_state(pos);
174        let affected = mode == FillMode::Destroy && world.destroy_block(pos, true);
175
176        let on_shell = pos.x() == min.x()
177            || pos.x() == max.x()
178            || pos.y() == min.y()
179            || pos.y() == max.y()
180            || pos.z() == min.z()
181            || pos.z() == max.z();
182        let block = match mode {
183            FillMode::Replace | FillMode::Destroy => Some(&target),
184            FillMode::Outline => on_shell.then_some(&target),
185            FillMode::Hollow => Some(if on_shell { &target } else { &hollow_core }),
186        };
187
188        let placed = block.is_some_and(|block| block.place(world, pos, flags));
189        if placed && !strict {
190            updates.push((pos, old_state));
191        }
192        if placed || affected {
193            count += 1;
194        }
195    }
196
197    for (pos, old_state) in updates {
198        world.update_neighbour_on_block_set(pos, old_state);
199        let state = world.get_block_state(pos);
200        if state.has_fluid() {
201            world.schedule_fluid_tick(
202                pos,
203                state.get_fluid_state().fluid_id,
204                0,
205                TickPriority::ExtremelyHigh,
206            );
207        }
208    }
209
210    if count == 0 {
211        return Err(CommandSyntaxError::dynamic(TextComponent::from(
212            &translations::COMMANDS_FILL_FAILED,
213        )));
214    }
215    context.source().send_success(
216        &translations::COMMANDS_FILL_SUCCESS
217            .message([count.to_string()])
218            .component(),
219        true,
220    );
221    Ok(count)
222}

224/// Vanilla parity: `BlockPosArgument.getLoadedBlockPos`.
225pub(super) fn loaded_block_pos(
226    context: &FotonCommandContext<CommandSource>,
227    name: &str,
228) -> Result<BlockPos, CommandSyntaxError> {
229    let pos = context.coordinates(name)?.block_pos(context.source());
230    let world = context.source().world();
231    if !world.is_full_chunk_loaded_at(pos) {
232        return Err(CommandSyntaxError::dynamic(TextComponent::from(
233            &translations::ARGUMENT_POS_UNLOADED,
234        )));
235    }
236    if !world.is_in_valid_bounds(pos) {
237        return Err(CommandSyntaxError::dynamic(TextComponent::from(
238            &translations::ARGUMENT_POS_OUTOFWORLD,
239        )));
240    }
241    Ok(pos)
242}

244fn block_input(
245    context: &FotonCommandContext<CommandSource>,
246) -> Result<BlockInput, CommandSyntaxError> {
247    let BlockPredicate::Block {
248        block,
249        properties,
250        nbt,
251    } = context.block_predicate("block")?
252    else {
253        // The block-state parser refuses tags, so this arm is unreachable in
254        // practice; an error is cheaper than a dead server if that changes.
255        return Err(CommandSyntaxError::dynamic(
256            "A block tag cannot be placed; name a single block.",
257        ));
258    };
259    let state = REGISTRY
260        .blocks
261        .state_id_from_block_properties(
262            block,
263            properties
264                .iter()
265                .map(|(name, value)| (name.as_ref(), value.as_ref())),
266        )
267        .ok_or_else(|| {
268            CommandSyntaxError::dynamic(
269                "This Block is not registered or a property name/value is invalid.",
270            )
271        })?;
272    Ok(BlockInput {
273        state,
274        properties: properties.clone(),
275        nbt: nbt.clone(),
276    })
277}

279/// Vanilla's `Predicate<BlockInWorld>` filter: state match, then block-entity
280/// data when the predicate carries any.
281fn filter_matches(world: &Arc<World>, filter: FillFilter<'_>, pos: BlockPos) -> bool {
282    let predicate = match filter {
283        FillFilter::Everything => return true,
284        FillFilter::Air => return world.get_block_state(pos).is_air(),
285        FillFilter::Predicate(predicate) => predicate,
286    };
287    if !predicate.matches_state(world.get_block_state(pos)) {
288        return false;
289    }
290    let Some(expected) = predicate.nbt() else {
291        return true;
292    };
293    world.get_block_entity(pos).is_some_and(|entity| {
294        compare_nbt_compounds(expected, &entity.save_with_full_metadata(), true)
295    })
296}

298impl BlockInput {
299    /// Vanilla parity: `BlockInput.place`. Returns whether anything changed.
300    fn place(&self, world: &Arc<World>, pos: BlockPos, flags: UpdateFlags) -> bool {
301        let mut state = if flags.contains(UpdateFlags::UPDATE_KNOWN_SHAPE) {
302            self.state
303        } else {
304            update_from_neighbour_shapes(world, self.state, pos)
305        };
306        if state.is_air() {
307            state = self.state;
308        }
309        state = self.overwrite_with_defined_properties(state);
310
311        let mut affected = world.set_block(pos, state, flags);
312        let Some(nbt) = &self.nbt else {
313            return affected;
314        };
315        let Some(entity) = world.get_block_entity(pos) else {
316            return affected;
317        };
318
319        let before = entity.save_without_metadata();
320        let mut bytes = Vec::new();
321        nbt.write(&mut bytes);
322        let Ok(borrowed) = read_compound(&mut Cursor::new(bytes.as_slice())) else {
323            return affected;
324        };
325        entity.load_with_components(&borrowed);
326        if !nbt_compounds_equal(&before, &entity.save_without_metadata()) {
327            affected = true;
328            entity.set_changed();
329            world.send_block_updated(pos);
330        }
331        affected
332    }
333
334    /// Vanilla parity: `BlockInput.overwriteWithDefinedProperties`. Neighbour
335    /// shape updates may have changed properties the command named; the command
336    /// wins, but only for properties the resulting block actually has.
337    fn overwrite_with_defined_properties(&self, state: BlockStateId) -> BlockStateId {
338        if state == self.state || self.properties.is_empty() {
339            return state;
340        }
341        let block = state.get_block();
342        let mut merged: Vec<(String, String)> = REGISTRY
343            .blocks
344            .get_properties(state)
345            .into_iter()
346            .map(|(name, value)| (name.to_owned(), value.to_owned()))
347            .collect();
348        for (name, value) in &self.properties {
349            if let Some(slot) = merged
350                .iter_mut()
351                .find(|(existing, _)| existing == name.as_ref())
352            {
353                slot.1 = value.to_string();
354            }
355        }
356        REGISTRY
357            .blocks
358            .state_id_from_block_properties(
359                block,
360                merged
361                    .iter()
362                    .map(|(name, value)| (name.as_str(), value.as_str())),
363            )
364            .unwrap_or(state)
365    }
