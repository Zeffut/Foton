1//! Vanilla stopsound command.
2
3use foton_protocol::packets::game::{CStopSound, SoundSource};
4use foton_utils::{Identifier, translations};
5use text_components::TextComponent;
6
7use super::super::{
8    brigadier::{CommandNodeBuilder, CommandSyntaxError},
9    execution::{
10        CommandSource, FotonArgumentType, FotonCommandContext, FotonCommandRuntime, argument,
11        literal,
12    },
13    registration::CommandRegistration,
14};
15
16pub(super) fn registration() -> CommandRegistration<CommandSource> {
17    CommandRegistration::new(Identifier::vanilla_static("stopsound"), |_| command())
18}
19
20/// Vanilla parity: `StopSoundCommand.register`. Permission level 2 is applied by the registry.
21fn command() -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
22    let mut targets = argument("targets", FotonArgumentType::players())
23        .executes(|c| stop(c, None, None))
24        .then(
25            literal("*").then(
26                argument("sound", FotonArgumentType::identifier())
27                    .executes(|c| stop(c, None, Some(c.identifier("sound")?.clone()))),
28            ),
29        );
30    for source in SoundSource::ALL {
31        targets = targets.then(
32            literal(source.name())
33                .executes(move |c| stop(c, Some(source), None))
34                .then(
35                    argument("sound", FotonArgumentType::identifier()).executes(move |c| {
36                        stop(c, Some(source), Some(c.identifier("sound")?.clone()))
37                    }),
38                ),
39        );
40    }
41    literal("stopsound").then(targets)
42}
43
44/// Vanilla parity: `StopSoundCommand.stopSound`.
45fn stop(
46    context: &FotonCommandContext<CommandSource>,
47    source: Option<SoundSource>,
48    sound: Option<Identifier>,
49) -> Result<i32, CommandSyntaxError> {
50    let targets = context.players("targets")?;
51    let packet = CStopSound {
52        sound: sound.clone(),
53        source,
54    };
55    for player in &targets {
56        player.send_packet(packet.clone());
57    }
58
59    let sound_name = sound.map(|sound| TextComponent::plain(sound.to_string()));
60    let message = match (source, sound_name) {
61        (Some(source), Some(sound)) => translations::COMMANDS_STOPSOUND_SUCCESS_SOURCE_SOUND
62            .message([sound, TextComponent::plain(source.name())])
63            .component(),
64        (Some(source), None) => translations::COMMANDS_STOPSOUND_SUCCESS_SOURCE_ANY
65            .message([TextComponent::plain(source.name())])
66            .component(),
67        (None, Some(sound)) => translations::COMMANDS_STOPSOUND_SUCCESS_SOURCELESS_SOUND
68            .message([sound])
69            .component(),
70        (None, None) => {
71            TextComponent::from(&translations::COMMANDS_STOPSOUND_SUCCESS_SOURCELESS_ANY)
72        }
73    };
74    context.source().send_success(&message, true);
75    Ok(i32::try_from(targets.len()).unwrap_or(i32::MAX))
76}
