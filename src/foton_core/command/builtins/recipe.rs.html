1//! Vanilla recipe book command.
2//!
3//! Vanilla parity: `RecipeCommand`.

5use foton_registry::REGISTRY;
6use foton_utils::{Identifier, translations};
7use text_components::TextComponent;

9use super::super::{
10    brigadier::{CommandNodeBuilder, CommandSyntaxError},
11    execution::{
12        CommandSource, FotonArgumentType, FotonCommandContext, FotonCommandRuntime, argument,
13        literal,
14    },
15    registration::CommandRegistration,
16};
17use crate::{entity::Entity as _, player::Player};
18
19pub(super) fn registration() -> CommandRegistration<CommandSource> {
20    CommandRegistration::new(Identifier::vanilla_static("recipe"), |_| command())
21}
22
23fn command() -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
24    literal("recipe")
25        .then(literal("give").then(targets_then_recipes(give_one, give_all)))
26        .then(literal("take").then(targets_then_recipes(take_one, take_all)))
27}
28
29fn targets_then_recipes(
30    one: fn(&FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError>,
31    all: fn(&FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError>,
32) -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
33    argument("targets", FotonArgumentType::players())
34        .then(argument("recipe", FotonArgumentType::recipe()).executes(one))
35        .then(literal("*").executes(all))
36}
37
38fn give_one(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
39    give(context, &[named_recipe(context)?])
40}
41
42fn give_all(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
43    give(context, &REGISTRY.recipes.recipe_keys())
44}
45
46fn take_one(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
47    take(context, &[named_recipe(context)?])
48}
49
50fn take_all(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
51    take(context, &REGISTRY.recipes.recipe_keys())
52}

54/// Vanilla parity: `ResourceKeyArgument.getRecipe`, which looks the key up when
55/// the command runs rather than when it parses, so a recipe a later reload
56/// adds is not a syntax error.
57fn named_recipe(
58    context: &FotonCommandContext<CommandSource>,
59) -> Result<Identifier, CommandSyntaxError> {
60    let key = context.recipe_key("recipe")?;
61    if !REGISTRY.recipes.contains_key(key) {
62        let message = translations::RECIPE_NOT_FOUND
63            .message([key.to_string()])
64            .component();
65        return Err(CommandSyntaxError::dynamic(message));
66    }
67    Ok(key.clone())
68}

70/// Vanilla parity: `RecipeCommand.giveRecipes`. The success message counts the
71/// recipes asked for, while the command's result counts the displays actually
72/// added.
73fn give(
74    context: &FotonCommandContext<CommandSource>,
75    recipes: &[Identifier],
76) -> Result<i32, CommandSyntaxError> {
77    let players = context.players("targets")?;
78    let success = players
79        .iter()
80        .map(|player| player.award_recipes(recipes))
81        .sum::<usize>();
82    if success == 0 {
83        return Err(CommandSyntaxError::dynamic(TextComponent::from(
84            &translations::COMMANDS_RECIPE_GIVE_FAILED,
85        )));
86    }
87    let message = if let [player] = players.as_slice() {
88        translations::COMMANDS_RECIPE_GIVE_SUCCESS_SINGLE
89            .message([
90                TextComponent::from(recipes.len().to_string()),
91                display_name(player),
92            ])
93            .component()
94    } else {
95        translations::COMMANDS_RECIPE_GIVE_SUCCESS_MULTIPLE
96            .message([
97                TextComponent::from(recipes.len().to_string()),
98                TextComponent::from(players.len().to_string()),
99            ])
100            .component()
101    };
102    context.source().send_success(&message, true);
103    result(success)
104}

106/// Vanilla parity: `RecipeCommand.takeRecipes`.
107fn take(
108    context: &FotonCommandContext<CommandSource>,
109    recipes: &[Identifier],
110) -> Result<i32, CommandSyntaxError> {
111    let players = context.players("targets")?;
112    let success = players
113        .iter()
114        .map(|player| player.reset_recipes(recipes))
115        .sum::<usize>();
116    if success == 0 {
117        return Err(CommandSyntaxError::dynamic(TextComponent::from(
118            &translations::COMMANDS_RECIPE_TAKE_FAILED,
119        )));
120    }
121    let message = if let [player] = players.as_slice() {
122        translations::COMMANDS_RECIPE_TAKE_SUCCESS_SINGLE
123            .message([
124                TextComponent::from(recipes.len().to_string()),
125                display_name(player),
126            ])
127            .component()
128    } else {
129        translations::COMMANDS_RECIPE_TAKE_SUCCESS_MULTIPLE
130            .message([
131                TextComponent::from(recipes.len().to_string()),
132                TextComponent::from(players.len().to_string()),
133            ])
134            .component()
135    };
136    context.source().send_success(&message, true);
137    result(success)
138}
139
140fn display_name(player: &Player) -> TextComponent {
141    TextComponent::plain(player.plain_text_name())
142}
143
144fn result(success: usize) -> Result<i32, CommandSyntaxError> {
145    i32::try_from(success)
146        .map_err(|_| CommandSyntaxError::dynamic("Recipe count exceeds the command result range"))
147}
