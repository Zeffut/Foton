34fn command() -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
35    literal("summon").then(
36        argument("entity", FotonArgumentType::summonable_entity())
37            .executes(|context| {
38                summon_entity(
39                    context,
40                    context.source().position(),
41                    &NbtCompound::new(),
42                    true,
43                )
44            })
45            .then(
46                argument("pos", FotonArgumentType::vec3(true))
47                    .executes(|context| {
48                        let position = context.coordinates("pos")?;
49                        summon_entity(
50                            context,
51                            position.position(context.source()),
52                            &NbtCompound::new(),
53                            true,
54                        )
55                    })
56                    // Vanilla hangs the compound off `pos`, not off `entity`,
57                    // so `/summon pig {...}` without a position is not a thing.
58                    .then(
59                        argument("nbt", FotonArgumentType::nbt_compound_tag()).executes(
60                            |context| {
61                                let position = context.coordinates("pos")?;
62                                let nbt = context.nbt_compound("nbt")?.clone();
63                                summon_entity(
64                                    context,
65                                    position.position(context.source()),
66                                    &nbt,
67                                    false,
68                                )
69                            },
70                        ),
71                    ),
72            ),
73    )
74}

76fn summon_entity(
77    context: &FotonCommandContext<CommandSource>,
78    position: DVec3,
79    nbt: &NbtCompound,
80    finalize: bool,
81) -> Result<i32, CommandSyntaxError> {
82    let entity_type = context.entity_type("entity")?;
83    let entity = create_entity(context, entity_type, position, nbt, finalize)?;
84    let message = translations::COMMANDS_SUMMON_SUCCESS
85        .message([entity.display_name()])
86        .component();
87    context.source().send_success(&message, true);
88    Ok(1)
89}

91/// Builds one entity and everything riding it, then puts the tree in the world.
92///
93/// Vanilla parity: `SummonCommand.createEntity`. `finalize` is vanilla's own
94/// flag and is false for the NBT branch even when the compound is empty:
95/// `/summon zombie ~ ~ ~ {}` gets no equipment, no random armour and no baby
96/// roll, because which node executed is what decides, not what the compound
97/// happens to contain.
98pub(super) fn create_entity(
99    context: &FotonCommandContext<CommandSource>,
100    entity_type: EntityTypeRef,
101    position: DVec3,
102    nbt: &NbtCompound,
103    finalize: bool,
104) -> Result<SharedEntity, CommandSyntaxError> {
105    if !World::is_in_spawnable_bounds(BlockPos::from(position)) {
106        return Err(command_failed(
107            &translations::COMMANDS_SUMMON_INVALID_POSITION,
108        ));
109    }
110
111    let world = context.source().world();
112    if world.difficulty() == Difficulty::Peaceful && !entity_type.allowed_in_peaceful {
113        return Err(command_failed(
114            &translations::COMMANDS_SUMMON_FAILED_PEACEFUL,
115        ));
116    }
117
118    // Vanilla parity: `entityTag.putString("id", ...)` on a *copy*. The
119    // argument-supplied type wins over any `id` the caller wrote, and the
120    // caller's compound is not modified.
121    //
122    // The removal is not decoration. `NbtCompound::insert` appends instead of
123    // replacing and the readers return the first match, so writing `id` over a
124    // compound that already has one leaves the caller's in front:
125    // `/summon minecraft:pig ~ ~ ~ {id:"minecraft:cow"}` would produce a cow.
126    // Vanilla's `CompoundTag` is a map and cannot express the duplicate at all.
127    let mut tag = nbt.clone();
128    while tag.remove("id").is_some() {}
129    tag.insert("id", entity_type.key.to_string());
130
131    let mut tree = Vec::new();
132    let entity = load_entity_tree(&tag, position, world, &mut tree)
133        .ok_or_else(|| command_failed(&translations::COMMANDS_SUMMON_FAILED))?;
134
135    if finalize && let Some(mob) = entity.as_mob() {
136        let _ = mob.finalize_spawn(world, EntitySpawnReason::Command, None);
137    }
138
139    match world.try_add_entity_with_passengers(&tree) {
140        Ok(()) => Ok(entity),
141        Err(AddEntityError::DuplicateUuid { .. }) => {
142            Err(command_failed(&translations::COMMANDS_SUMMON_FAILED_UUID))
143        }
144        Err(_) => Err(command_failed(&translations::COMMANDS_SUMMON_FAILED)),
145    }
146}

148/// Builds one entity and its passengers, collecting the tree in place order.
149///
150/// Vanilla parity: `EntityType.loadEntityRecursive` with `SummonCommand`'s
151/// post-load processor. That processor is `snapTo(pos.x, pos.y, pos.z,
152/// e.getYRot(), e.getXRot())` and it runs on the vehicle *and* on every
153/// passenger, so the command's position beats any `Pos` in the compound while
154/// the rotation the compound asked for survives.
155fn load_entity_tree(
156    nbt: &NbtCompound,
157    position: DVec3,
158    world: &Arc<World>,
159    tree: &mut Vec<SharedEntity>,
160) -> Option<SharedEntity> {
161    let mut bytes = Vec::new();
162    nbt.write(&mut bytes);
163    let borrowed = read_borrowed_compound(&mut Cursor::new(bytes.as_slice())).ok()?;
164    let loaded = read_entity_nbt(&(&borrowed).into())?;
165    // The `summonable_entity` argument already refuses a type Foton has no
166    // implementation for, but a `Passengers` entry names its own type and
167    // nothing has checked it. Without this the raw fallback would seat an
168    // inert entity and report success.
169    if !ENTITIES.has_load_factory(loaded.entity_type) {
170        return None;
171    }
172
173    // The type's own reader only ever sees what is left after the base fields,
174    // which is the same split structure placement makes.
175    let mut remainder_bytes = Vec::new();
176    loaded.remainder.write(&mut remainder_bytes);
177    let remainder = read_borrowed_compound(&mut Cursor::new(remainder_bytes.as_slice())).ok()?;
178
179    let entity = ENTITIES.create_and_load_or_raw(
180        EntityLoadRequest {
181            entity_type: loaded.entity_type,
182            position,
183            uuid: loaded.uuid.unwrap_or_else(Uuid::new_v4),
184            velocity: loaded.velocity,
185            rotation: loaded.rotation,
186            fall_distance: loaded.fall_distance,
187            fire_freeze: loaded.fire_freeze,
188            on_ground: loaded.on_ground,
189            save_data: loaded.save_data,
190            world: Arc::downgrade(world),
191        },
192        &remainder,
193    );
194    tree.push(Arc::clone(&entity));
195
196    for passenger_nbt in &loaded.passengers {
197        let Some(passenger) = load_entity_tree(passenger_nbt, position, world, tree) else {
198            // Vanilla's `loadPassengersRecursive` skips a passenger it cannot
199            // build and keeps the vehicle. A rider the server does not know
200            // about is not a reason to lose the horse.
201            continue;
202        };
203        EntityBase::start_riding_relationship(&entity, &passenger);
204    }
205
206    Some(entity)
207}

209fn command_failed(translation: &'static Translation<0>) -> CommandSyntaxError {
210    CommandSyntaxError::dynamic(TextComponent::from(translation))

214mod tests {
215    use super::super::create_dispatcher;
216    use crate::bootstrap::init_globals_once;
217    use crate::command::{
218        brigadier::{CommandDispatcher, NodeId},
219        execution::{CommandSource, FotonArgumentType, FotonCommandRuntime},

221
222    type Dispatcher = CommandDispatcher<CommandSource, FotonCommandRuntime>;
223
224    fn child(dispatcher: &Dispatcher, parent: NodeId, name: &str) -> NodeId {
225        let Some(children) = dispatcher.children(parent) else {
226            panic!("parent node should exist");
227        };
228        let Some(child) = children.iter().copied().find(|child| {
229            dispatcher
230                .node(*child)
231                .is_some_and(|node| node.name() == name)
232        }) else {
233            panic!("child {name} should exist");
234        };
235        child
236    }
237
238    #[test]
239    fn summon_graph_uses_typed_entity_and_deferred_position_arguments() {
240        init_globals_once();
241        let Ok(dispatcher) = create_dispatcher() else {
242            panic!("built-in commands should register");
243        };
244        let root = child(&dispatcher, dispatcher.root(), "summon");
245        let entity = child(&dispatcher, root, "entity");
246        assert_eq!(
247            dispatcher
248                .node(entity)
249                .and_then(|node| node.argument_type()),
250            Some(&FotonArgumentType::summonable_entity())
251        );
252        assert!(matches!(
253            dispatcher.node(entity),
254            Some(node) if node.is_executable()
255        ));
256
257        let position = child(&dispatcher, entity, "pos");
258        assert_eq!(
259            dispatcher
260                .node(position)
261                .and_then(|node| node.argument_type()),
262            Some(&FotonArgumentType::vec3(true))
263        );
264        assert!(matches!(
265            dispatcher.node(position),
266            Some(node) if node.is_executable()
267        ));
268
269        // Vanilla hangs the compound off `pos`. Putting it under `entity`
270        // would be a syntax Foton invented and a client would not suggest.
271        let nbt = child(&dispatcher, position, "nbt");
272        assert_eq!(
273            dispatcher.node(nbt).and_then(|node| node.argument_type()),
274            Some(&FotonArgumentType::nbt_compound_tag())
275        );
276        assert!(matches!(
277            dispatcher.node(nbt),
278            Some(node) if node.is_executable()
279        ));
280        assert!(dispatcher.children(nbt).is_some_and(<[_]>::is_empty));
281    }
