23pub(super) fn registration() -> CommandRegistration<CommandSource> {
24    CommandRegistration::new(Identifier::vanilla_static("function"), |_| command())
25}

27fn command() -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
28    // TODO: `function <name> with block|entity|storage <...>` needs the data
29    // accessors `/data` has not been ported yet, so only the literal compound
30    // form of macro arguments is available.
31    literal("function").then(
32        argument("name", FotonArgumentType::function())
33            .executes_custom(RunFunction)
34            .then(
35                argument("arguments", FotonArgumentType::nbt_compound_tag())
36                    .executes_custom(RunFunction),
37            ),
38    )
39}

41struct RunFunction;
42
43impl CustomCommandExecutor<CommandSource> for RunFunction {
44    fn run(
45        &self,
46        source: Arc<CommandSource>,
47        chain: &FotonContextChain<CommandSource>,
48        modifiers: ChainModifiers,
49        control: &mut ExecutionControl<'_, CommandSource>,
50    ) {
51        // Vanilla parity: `CustomCommandExecutor.WithErrorHandling`, which
52        // reports the error to the sender and fails the command's callback.
53        if let Err(error) = run_function(&source, chain, modifiers, control) {
54            source.handle_error(&error, modifiers.is_forked());
55            source.callback().on_result(false, 0);
56        }
57    }
58}
59
60fn run_function(
61    source: &Arc<CommandSource>,
62    chain: &FotonContextChain<CommandSource>,
63    modifiers: ChainModifiers,
64    control: &mut ExecutionControl<'_, CommandSource>,
65) -> Result<(), CommandSyntaxError> {
66    let context = chain.top_context().copy_for(Arc::clone(source));
67    let reference = context.function_or_tag("name")?.clone();
68    let functions = resolve_functions(source, &reference)?;
69    let arguments = context.nbt_compound("arguments").ok().cloned();
70
71    let mut instantiated = Vec::with_capacity(functions.len());
72    for function in &functions {
73        let entries = instantiate(source, function, arguments.as_ref()).map_err(|reason| {
74            CommandSyntaxError::dynamic(
75                translations::COMMANDS_FUNCTION_INSTANTIATION_FAILURE
76                    .message([TextComponent::from(function.id().to_string()), *reason])
77                    .component(),
78            )
79        })?;
80        instantiated.push(entries);
81    }
82
83    source.send_success(&scheduled_message(&functions), true);
84    queue_functions(&functions, &instantiated, source, control, modifiers);
85    Ok(())
86}

88/// Compiles one function for this call's arguments.
89pub(crate) fn instantiate(
90    source: &CommandSource,
91    function: &CommandFunction,
92    arguments: Option<&NbtCompound>,
93) -> Result<FunctionEntries<CommandSource>, Box<TextComponent>> {
94    source
95        .server()
96        .with_command_dispatcher(|dispatcher| function.instantiate(arguments, dispatcher))
97}

99/// Resolves a parsed name into the functions it stands for.
100///
101/// Vanilla parity: `FunctionArgument.getFunction` for a plain name and
102/// `FunctionCommand.ERROR_NO_FUNCTIONS` for a tag that resolves to nothing. An
103/// unknown name never yields an empty list that would silently run nothing.
104pub(crate) fn resolve_functions(
105    source: &CommandSource,
106    reference: &FunctionOrTag,
107) -> Result<Vec<Arc<CommandFunction>>, CommandSyntaxError> {
108    let library = source.server().functions.library();
109    match reference {
110        FunctionOrTag::Function(id) => library
111            .function(id)
112            .map(|function| vec![Arc::clone(function)])
113            .ok_or_else(|| {
114                CommandSyntaxError::dynamic(
115                    translations::ARGUMENTS_FUNCTION_UNKNOWN
116                        .message([TextComponent::from(id.to_string())])
117                        .component(),
118                )
119            }),
120        FunctionOrTag::Tag(id) => {
121            let functions = library.tag(id);
122            if functions.is_empty() {
123                return Err(CommandSyntaxError::dynamic(
124                    translations::COMMANDS_FUNCTION_SCHEDULED_NO_FUNCTIONS
125                        .message([TextComponent::from(format!("#{id}"))])
126                        .component(),
127                ));
128            }
129            Ok(functions.to_vec())
130        }
131    }
132}

134fn scheduled_message(functions: &[Arc<CommandFunction>]) -> TextComponent {
135    if let [only] = functions {
136        return translations::COMMANDS_FUNCTION_SCHEDULED_SINGLE
137            .message([TextComponent::from(only.id().to_string())])
138            .component();
139    }
140    let names = functions
141        .iter()
142        .map(|function| function.id().to_string())
143        .collect::<Vec<_>>()
144        .join(", ");
145    translations::COMMANDS_FUNCTION_SCHEDULED_MULTIPLE
146        .message([TextComponent::from(names)])
147        .component()
148}

150/// Queues every selected function, wiring up the caller's result reporting.
151///
152/// Vanilla parity: `FunctionCommand.queueFunctions`.
153fn queue_functions(
154    functions: &[Arc<CommandFunction>],
155    instantiated: &[FunctionEntries<CommandSource>],
156    original_source: &Arc<CommandSource>,
157    control: &mut ExecutionControl<'_, CommandSource>,
158    modifiers: ChainModifiers,
159) {
160    // Vanilla parity: `FunctionCommand.modifySenderForExecution`. Vanilla also
161    // caps the permission level at gamemaster; Foton has no level ladder to cap,
162    // and a function keeps the caller's permission set, which cannot grant more
163    // than the caller already had.
164    let function_source = Arc::new(
165        original_source
166            .with_suppressed_output()
167            .with_callback(CommandResultCallback::empty()),
168    );
169
170    if modifiers.is_return() {
171        let shared = CommandResultCallback::chain(
172            original_source.callback(),
173            control.current_frame().return_value_consumer(),
174        );
175        for (function, entries) in functions.iter().zip(instantiated) {
176            let callback = decorate_output(original_source, function.id(), shared.clone());
177            control.queue_function_call(
178                Arc::clone(entries),
179                Arc::clone(&function_source),
180                callback,
181                true,
182            );
183        }
184        control.queue_fallthrough();
185        return;
186    }
187
188    let original_callback = original_source.callback();
189    match (functions, instantiated) {
190        ([], _) | (_, []) => {}
191        ([only], [entries]) => {
192            let callback = decorate_output(original_source, only.id(), original_callback);
193            control.queue_function_call(Arc::clone(entries), function_source, callback, false);
194        }
195        _ => {
196            // Vanilla parity: several functions report one summed result to the
197            // caller, and only when at least one of them produced a result.
198            let any_result = Arc::new(AtomicBool::new(false));
199            let sum = Arc::new(AtomicI32::new(0));
200            for (function, entries) in functions.iter().zip(instantiated) {
201                let any_result = Arc::clone(&any_result);
202                let sum = Arc::clone(&sum);
203                let partial = CommandResultCallback::new(move |_success, result| {
204                    any_result.store(true, Ordering::Relaxed);
205                    sum.fetch_add(result, Ordering::Relaxed);
206                });
207                let callback = decorate_output(original_source, function.id(), partial);
208                control.queue_function_call(
209                    Arc::clone(entries),
210                    Arc::clone(&function_source),
211                    callback,
212                    false,
213                );
214            }
215            control.queue_callback(move || {
216                if any_result.load(Ordering::Relaxed) {
217                    original_callback.on_result(true, sum.load(Ordering::Relaxed));
218                }
219            });
220        }
221    }
222}

224/// Adds vanilla's per-function `commands.function.result` feedback.
225fn decorate_output(
226    original_source: &Arc<CommandSource>,
227    id: &Identifier,
228    callback: CommandResultCallback,
229) -> CommandResultCallback {
230    if original_source.is_silent() {
231        return callback;
232    }
233    let source = Arc::clone(original_source);
234    let id = id.clone();
235    CommandResultCallback::new(move |success, result| {
236        let message = translations::COMMANDS_FUNCTION_RESULT
237            .message([
238                TextComponent::from(id.to_string()),
239                TextComponent::from(result.to_string()),
240            ])
241            .component();
242        source.send_success(&message, true);
243        callback.on_result(success, result);
244    })
245}

247#[cfg(test)]
248mod tests {
249    use foton_registry::init_vanilla_registry;
250
251    use super::super::create_dispatcher;
252    use crate::command::execution::FotonArgumentType;
253
254    #[test]
255    fn function_graph_takes_a_name_and_optional_macro_arguments() {
256        init_vanilla_registry();
257        let Ok(dispatcher) = create_dispatcher() else {
258            panic!("built-in commands should register");
259        };
260        let Some(root) = dispatcher.children(dispatcher.root()).and_then(|children| {
261            children.iter().copied().find(|child| {
262                dispatcher
263                    .node(*child)
264                    .is_some_and(|node| node.name() == "function")
265            })
266        }) else {
267            panic!("function root should exist");
268        };
269        let Some(children) = dispatcher.children(root) else {
270            panic!("function root should have children");
271        };
272        assert_eq!(children.len(), 1);
273        let Some(name) = dispatcher.node(children[0]) else {
274            panic!("function name argument should exist");
275        };
276        assert_eq!(name.name(), "name");
277        assert!(name.is_executable());
278        assert_eq!(name.argument_type(), Some(&FotonArgumentType::function()));
279
280        let Some(name_children) = dispatcher.children(children[0]) else {
281            panic!("function name should accept macro arguments");
282        };
283        assert_eq!(name_children.len(), 1);
284        let Some(arguments) = dispatcher.node(name_children[0]) else {
285            panic!("function arguments should exist");
286        };
287        assert_eq!(arguments.name(), "arguments");
288        assert!(arguments.is_executable());
289        assert_eq!(
290            arguments.argument_type(),
291            Some(&FotonArgumentType::nbt_compound_tag())
292        );
293    }
