1//! The `/team` command.
2//!
3//! Vanilla parity: `net.minecraft.server.commands.TeamCommand`.
4//!
5//! Only the subcommands backed by state that something reads are here. Vanilla
6//! also carries `displayName`, `color`, `prefix`, `suffix`,
7//! `nametagVisibility`, `deathMessageVisibility` and `collisionRule`; every one
8//! of those exists to reach a client through `CSetPlayerTeam`, which Foton does
9//! not send yet. Storing them would be writing data nothing can observe, so
10//! they are left out until the packet exists rather than accepted and dropped.
11//!
12//! What is here changes behavior: membership, which `@e[team=]` already reads,
13//! and the two options `Player.canHarmPlayer` and `Entity.isAlliedTo` consult.
14
15use foton_utils::{Identifier, translations};
16use text_components::TextComponent;

18use super::super::{
19    brigadier::{ArgumentType, CommandNodeBuilder, CommandSyntaxError},
20    execution::{
21        CommandSource, FotonArgumentType, FotonCommandContext, FotonCommandRuntime,
22        ScoreHolderWildcard, argument, literal,
23    },
24    registration::CommandRegistration,
25};
26use crate::scoreboard::{ScoreHolder, Scoreboard, ScoreboardTeam};
27
28pub(super) fn registration() -> CommandRegistration<CommandSource> {
29    CommandRegistration::new(Identifier::vanilla_static("team"), |_| command())
30}
31
32fn command() -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
33    literal("team")
34        .then(
35            literal("list")
36                .executes(list_teams)
37                .then(argument("team", ArgumentType::word()).executes(list_members)),
38        )
39        .then(literal("add").then(argument("team", ArgumentType::word()).executes(add_team)))
40        .then(literal("remove").then(argument("team", ArgumentType::word()).executes(remove_team)))
41        .then(literal("empty").then(argument("team", ArgumentType::word()).executes(empty_team)))
42        .then(
43            literal("join").then(
44                argument("team", ArgumentType::word()).then(
45                    argument("members", FotonArgumentType::score_holders()).executes(join_team),
46                ),
47            ),
48        )
49        .then(
50            literal("leave")
51                .then(argument("members", FotonArgumentType::score_holders()).executes(leave_team)),
52        )
53        .then(literal("modify").then(modify_subcommands()))
54}
55
56fn modify_subcommands() -> CommandNodeBuilder<CommandSource, FotonCommandRuntime> {
57    argument("team", ArgumentType::word())
58        .then(
59            literal("friendlyFire")
60                .then(argument("allowed", ArgumentType::bool()).executes(set_friendly_fire)),
61        )
62        .then(
63            literal("seeFriendlyInvisibles").then(
64                argument("allowed", ArgumentType::bool()).executes(set_see_friendly_invisibles),
65            ),
66        )
67}

69/// The scoreboard of the domain the command was run in.
70///
71/// Vanilla reaches one server-wide scoreboard; Foton keeps one per domain,
72/// beside the boss bars and the command storage.
73fn source_scoreboard(
74    context: &FotonCommandContext<CommandSource>,
75) -> Result<&Scoreboard, CommandSyntaxError> {
76    let source = context.source();
77    source
78        .server()
79        .scoreboards
80        .get(source.world().domain())
81        .ok_or_else(|| {
82            CommandSyntaxError::dynamic(format!(
83                "Domain '{}' has no scoreboard",
84                source.world().domain()
85            ))
86        })
87}

89/// Vanilla parity: `TeamArgument.getTeam`, which fails on an unknown name.
90fn team(
91    context: &FotonCommandContext<CommandSource>,
92) -> Result<ScoreboardTeam, CommandSyntaxError> {
93    let name = context.string("team")?.to_owned();
94    source_scoreboard(context)?
95        .team(&name)
96        .ok_or_else(|| CommandSyntaxError::dynamic(format!("Unknown team '{name}'")))
97}

99/// Vanilla parity: `TeamCommand.listTeams`.
100fn list_teams(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
101    let names = source_scoreboard(context)?.team_names();
102    let message = if names.is_empty() {
103        TextComponent::from(&translations::COMMANDS_TEAM_LIST_TEAMS_EMPTY)
104    } else {
105        translations::COMMANDS_TEAM_LIST_TEAMS_SUCCESS
106            .message([names.len().to_string(), names.join(", ")])
107            .component()
108    };
109    context.source().send_success(&message, false);
110    Ok(i32::try_from(names.len()).unwrap_or(i32::MAX))
111}

113/// Vanilla parity: `TeamCommand.listMembers`.
114fn list_members(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
115    let team = team(context)?;
116    let entries = source_scoreboard(context)?.team_entries(&team);
117    let message = if entries.is_empty() {
118        translations::COMMANDS_TEAM_LIST_MEMBERS_EMPTY
119            .message([team.name().to_owned()])
120            .component()
121    } else {
122        translations::COMMANDS_TEAM_LIST_MEMBERS_SUCCESS
123            .message([
124                team.name().to_owned(),
125                entries.len().to_string(),
126                entries.join(", "),
127            ])
128            .component()
129    };
130    context.source().send_success(&message, false);
131    Ok(i32::try_from(entries.len()).unwrap_or(i32::MAX))
132}

134/// Vanilla parity: `TeamCommand.createTeam`.
135fn add_team(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
136    let name = context.string("team")?.to_owned();
137    let scoreboard = source_scoreboard(context)?;
138    let Ok(team) = scoreboard.add_team(name) else {
139        return Err(CommandSyntaxError::dynamic(TextComponent::from(
140            &translations::COMMANDS_TEAM_ADD_DUPLICATE,
141        )));
142    };
143    let message = translations::COMMANDS_TEAM_ADD_SUCCESS
144        .message([team.name().to_owned()])
145        .component();
146    context.source().send_success(&message, true);
147    Ok(1)
148}

161/// Vanilla parity: `TeamCommand.emptyTeam`, which refuses an empty team.
162fn empty_team(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
163    let team = team(context)?;
164    let scoreboard = source_scoreboard(context)?;
165    let entries = scoreboard.team_entries(&team);
166    if entries.is_empty() {
167        return Err(CommandSyntaxError::dynamic(TextComponent::from(
168            &translations::COMMANDS_TEAM_EMPTY_UNCHANGED,
169        )));
170    }
171    for entry in &entries {
172        scoreboard.remove_holder_from_team(&ScoreHolder::new(entry.clone()));
173    }
174    let message = translations::COMMANDS_TEAM_EMPTY_SUCCESS
175        .message([entries.len().to_string(), team.name().to_owned()])
176        .component();
177    context.source().send_success(&message, true);
178    Ok(i32::try_from(entries.len()).unwrap_or(i32::MAX))
179}

181/// Vanilla parity: `TeamCommand.joinTeam`.
182fn join_team(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
183    let team = team(context)?;
184    let members = context.score_holders("members", ScoreHolderWildcard::Empty)?;
185    let scoreboard = source_scoreboard(context)?;
186    for member in &members {
187        scoreboard
188            .add_holder_to_team(member, &team)
189            .map_err(|error| CommandSyntaxError::dynamic(error.to_string()))?;
190    }
191
192    let message = if let [only] = members.as_slice() {
193        translations::COMMANDS_TEAM_JOIN_SUCCESS_SINGLE
194            .message([only.name().to_owned(), team.name().to_owned()])
195            .component()
196    } else {
197        translations::COMMANDS_TEAM_JOIN_SUCCESS_MULTIPLE
198            .message([members.len().to_string(), team.name().to_owned()])
199            .component()
200    };
201    context.source().send_success(&message, true);
202    Ok(i32::try_from(members.len()).unwrap_or(i32::MAX))
203}

205/// Vanilla parity: `TeamCommand.leaveTeam`.
206fn leave_team(context: &FotonCommandContext<CommandSource>) -> Result<i32, CommandSyntaxError> {
207    let members = context.score_holders("members", ScoreHolderWildcard::Empty)?;
208    let scoreboard = source_scoreboard(context)?;
209    for member in &members {
210        scoreboard.remove_holder_from_team(member);
211    }
212
213    let message = if let [only] = members.as_slice() {
214        translations::COMMANDS_TEAM_LEAVE_SUCCESS_SINGLE
215            .message([only.name().to_owned()])
216            .component()
217    } else {
218        translations::COMMANDS_TEAM_LEAVE_SUCCESS_MULTIPLE
219            .message([members.len().to_string()])
220            .component()
221    };
222    context.source().send_success(&message, true);
223    Ok(i32::try_from(members.len()).unwrap_or(i32::MAX))
224}

226/// Vanilla parity: `TeamCommand.setFriendlyFire`, which refuses a no-op.
227fn set_friendly_fire(
228    context: &FotonCommandContext<CommandSource>,
229) -> Result<i32, CommandSyntaxError> {
230    let team = team(context)?;
231    let allowed = context.boolean("allowed")?;
232    let scoreboard = source_scoreboard(context)?;
233    let mut options = scoreboard.team_options(&team);
234    if options.allow_friendly_fire == allowed {
235        return Err(CommandSyntaxError::dynamic(TextComponent::from(
236            if allowed {
237                &translations::COMMANDS_TEAM_OPTION_FRIENDLYFIRE_ALREADY_ENABLED
238            } else {
239                &translations::COMMANDS_TEAM_OPTION_FRIENDLYFIRE_ALREADY_DISABLED
240            },
241        )));
242    }
243
244    options.allow_friendly_fire = allowed;
245    scoreboard.set_team_options(&team, options);
246    let message = if allowed {
247        translations::COMMANDS_TEAM_OPTION_FRIENDLYFIRE_ENABLED
248            .message([team.name().to_owned()])
249            .component()
250    } else {
251        translations::COMMANDS_TEAM_OPTION_FRIENDLYFIRE_DISABLED
252            .message([team.name().to_owned()])
253            .component()
254    };
255    context.source().send_success(&message, true);
256    Ok(0)
257}
