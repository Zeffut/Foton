1//! World-level data: `level.dat` and the saved-data files beside it.
2//!
3//! Vanilla 26.x split the old `level.dat` `Data` compound: the seed, game rules,
4//! weather, clocks and per-dimension world border now live in
5//! `data/minecraft/*.dat`. Foton keeps one `level.toml` per world, so a source
6//! world's values are gathered here and written once per imported dimension.

50fn read_gzip_nbt(path: &Path) -> Result<Option<NbtCompound>, ImportError> {
51    let compressed = match fs::read(path) {
52        Ok(bytes) => bytes,
53        Err(error) if error.kind() == ErrorKind::NotFound => return Ok(None),
54        Err(error) => return Err(ImportError::io(path, error)),
55    };
56    let mut bytes = Vec::new();
57    GzDecoder::new(compressed.as_slice())
58        .read_to_end(&mut bytes)
59        .map_err(|error| ImportError::io(path, error))?;
60    match read(&mut Cursor::new(bytes.as_slice())) {
61        Ok(Nbt::Some(root)) => Ok(Some(root.as_compound())),
62        Ok(Nbt::None) => Ok(None),
63        Err(error) => Err(ImportError::Level(format!(
64            "{} is not valid NBT: {error}",
65            path.display()
66        ))),
67    }
68}

70/// The `data` compound of a `data/minecraft/<name>.dat` file, if it exists.
71fn read_saved_data(root: &Path, name: &str) -> Result<Option<NbtCompound>, ImportError> {
72    let path = root
73        .join("data")
74        .join("minecraft")
75        .join(format!("{name}.dat"));
76    Ok(
77        read_gzip_nbt(&path)?.and_then(|mut file| match file.take("data") {
78            Some(NbtTag::Compound(data)) => Some(data),
79            _ => None,
80        }),
81    )
82}

84fn rule_value(tag: &NbtTag) -> Option<serde_json::Value> {
85    match tag {
86        NbtTag::Byte(value) => Some(serde_json::Value::Bool(*value != 0)),
87        NbtTag::Short(value) => Some((*value).into()),
88        NbtTag::Int(value) => Some((*value).into()),
89        NbtTag::Long(value) => Some((*value).into()),
90        NbtTag::Float(value) => serde_json::Number::from_f64(f64::from(*value)).map(Into::into),
91        NbtTag::Double(value) => serde_json::Number::from_f64(*value).map(Into::into),
92        _ => None,
93    }
94}

96fn parse_difficulty(name: &str) -> Option<Difficulty> {
97    match name {
98        "peaceful" => Some(Difficulty::Peaceful),
99        "easy" => Some(Difficulty::Easy),
100        "normal" => Some(Difficulty::Normal),
101        "hard" => Some(Difficulty::Hard),
102        _ => None,
103    }
104}

106impl SourceLevel {
107    /// Reads the world's data, or `None` if the folder has no `level.dat`.
108    ///
109    /// # Errors
110    /// [`ImportError::DataVersion`] if `level.dat` was written by another
111    /// Minecraft version.
112    pub(super) fn read(root: &Path) -> Result<Option<Self>, ImportError> {
113        let Some(level) = read_gzip_nbt(&root.join("level.dat"))? else {
114            return Ok(None);
115        };
116        let data = level
117            .compound("Data")
118            .ok_or_else(|| ImportError::Level("level.dat has no Data compound".to_owned()))?;
119        let version = data.int("DataVersion");
120        if version != Some(WORLD_VERSION) {
121            return Err(ImportError::DataVersion {
122                what: "level.dat".to_owned(),
123                found: version,
124                skippable: false,
125            });
126        }
127
128        let mut source = Self {
129            game_time: data.long("Time").unwrap_or(0),
130            ..Self::default()
131        };
132        source.difficulty = data.compound("difficulty_settings").and_then(|settings| {
133            let difficulty = parse_difficulty(&settings.string("difficulty")?.to_str())?;
134            Some((
135                difficulty,
136                settings.byte("locked").is_some_and(|flag| flag != 0),
137            ))
138        });
139        source.spawn = data.compound("spawn").and_then(|spawn| {
140            let pos = spawn.int_array("pos").filter(|pos| pos.len() == 3)?;
141            Some(SourceSpawn {
142                dimension: spawn.string("dimension")?.to_str().parse().ok()?,
143                pos: [pos[0], pos[1], pos[2]],
144                yaw: spawn.float("yaw").unwrap_or(0.0),
145                pitch: spawn.float("pitch").unwrap_or(0.0),
146            })
147        });
148
149        source.seed =
150            read_saved_data(root, "world_gen_settings")?.and_then(|settings| settings.long("seed"));
151        source.weather = read_saved_data(root, "weather")?.map(|weather| WeatherState {
152            raining: weather.byte("raining").is_some_and(|flag| flag != 0),
153            rain_time: weather.int("rain_time").unwrap_or(0),
154            thundering: weather.byte("thundering").is_some_and(|flag| flag != 0),
155            thunder_time: weather.int("thunder_time").unwrap_or(0),
156            clear_weather_time: weather.int("clear_weather_time").unwrap_or(0),
157        });
158        if let Some(rules) = read_saved_data(root, "game_rules")? {
159            source.game_rules = rules
160                .iter()
161                .filter_map(|(name, tag)| {
162                    let name = name.to_str();
163                    let name = name.strip_prefix("minecraft:").unwrap_or(&name);
164                    Some((name.to_owned(), rule_value(tag)?))
165                })
166                .collect();
167        }
168        if let Some(clocks) = read_saved_data(root, "world_clocks")? {
169            source.clocks = clocks
170                .iter()
171                .filter_map(|(name, tag)| {
172                    let clock = tag.compound()?;
173                    Some(SourceClock {
174                        key: name.to_str().parse().ok()?,
175                        total_ticks: clock.long("total_ticks").unwrap_or(0),
176                        paused: clock.byte("paused").map(|flag| flag != 0),
177                        rate: clock.float("rate"),
178                    })
179                })
180                .collect();
181        }
182        Ok(Some(source))
183    }
184
185    /// Builds the `level.toml` content for one imported dimension.
186    ///
187    /// `generation` stays unset: the first start adopts the configured
188    /// generator settings, which is how every other pre-existing world is
189    /// treated.
190    pub(super) fn level_data(
191        &self,
192        seed: i64,
193        dimension: &Identifier,
194        border: Option<WorldBorderData>,
195        is_spawn_world: bool,
196    ) -> LevelData {
197        let (difficulty, locked) = self.difficulty.unwrap_or_default();
198        let mut data = LevelData::new_with_seed_and_difficulty(seed, difficulty);
199        data.difficulty_locked = locked;
200        data.game_time = self.game_time;
201        data.initialized = true;
202        if let Some(weather) = &self.weather {
203            data.weather = weather.clone();
204        }
205        if let Some(border) = border {
206            data.world_border = border;
207        }
208        data.game_rules = self.game_rules.iter().cloned().collect();
209
210        for clock in &self.clocks {
211            let Some(registered) = REGISTRY.world_clocks.by_key(&clock.key) else {
212                continue;
213            };
214            let _ = data
215                .world_clocks
216                .set_total_ticks(registered, clock.total_ticks);
217            if let Some(paused) = clock.paused {
218                let _ = data.world_clocks.set_paused(registered, paused);
219            }
220            if let Some(rate) = clock.rate {
221                let _ = data.world_clocks.set_rate(registered, rate);
222            }
223        }
224
225        if let Some(spawn) = self
226            .spawn
227            .as_ref()
228            .filter(|spawn| &spawn.dimension == dimension)
229        {
230            data.spawn = SpawnPoint {
231                x: spawn.pos[0],
232                y: spawn.pos[1],
233                z: spawn.pos[2],
234                angle: spawn.yaw,
235            };
236            if is_spawn_world {
237                data.respawn = Some(RespawnData::of(
238                    spawn.dimension.clone(),
239                    BlockPos::new(spawn.pos[0], spawn.pos[1], spawn.pos[2]),
240                    spawn.yaw,
241                    spawn.pitch,
242                ));
243            }
244        }
245        data
246    }
247}

249/// Reads a dimension's world border, which 26.x stores per dimension.
250pub(super) fn read_world_border(
251    dimension_dir: &Path,
252) -> Result<Option<WorldBorderData>, ImportError> {
253    let Some(border) = read_saved_data(dimension_dir, "world_border")? else {
254        return Ok(None);
255    };
256    let defaults = WorldBorderData::default();
257    Ok(Some(WorldBorderData {
258        center_x: border.double("center_x").unwrap_or(defaults.center_x),
259        center_z: border.double("center_z").unwrap_or(defaults.center_z),
260        damage_per_block: border
261            .double("damage_per_block")
262            .unwrap_or(defaults.damage_per_block),
263        safe_zone: border.double("safe_zone").unwrap_or(defaults.safe_zone),
264        warning_blocks: border
265            .int("warning_blocks")
266            .unwrap_or(defaults.warning_blocks),
267        warning_time: border.int("warning_time").unwrap_or(defaults.warning_time),
268        size: border.double("size").unwrap_or(defaults.size),
269        lerp_time: border.long("lerp_time").unwrap_or(defaults.lerp_time),
270        lerp_target: border.double("lerp_target").unwrap_or(defaults.lerp_target),
271    }))
