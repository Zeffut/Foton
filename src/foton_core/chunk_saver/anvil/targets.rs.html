16/// Resolves the Foton world each dimension is imported into.
17///
18/// With no `world_override`, a dimension goes to the default domain's world of
19/// the same name (`overworld`, `the_nether`, `the_end` in a stock
20/// `worlds.toml`), falling back to the world generated by the dimension's own
21/// vanilla generator. `world_override` is `world` in the default domain or
22/// `domain:world`, and is only meaningful for a single dimension.
23///
24/// # Errors
25/// Returns an error if `worlds.toml` is invalid, no world matches, two
26/// dimensions would land in one world, or the world is not stored on disk.
27pub fn resolve_targets(
28    worlds: &WorldsConfig,
29    dimensions: Vec<SourceDimension>,
30    world_override: Option<&str>,
31) -> Result<Vec<ImportTarget>, ImportError> {
32    let generators = WorldGeneratorRegistry::new_with_builtins()
33        .map_err(|error| layout_error(format!("world generator registry: {error}")))?;
34    let storage = WorldStorageRegistry::new_with_builtins()
35        .map_err(|error| layout_error(format!("world storage registry: {error}")))?;
36    let resolved = worlds
37        .validate_and_resolve(&generators, &storage)
38        .map_err(|error| layout_error(format!("worlds.toml: {error}")))?;
39    if world_override.is_some() && dimensions.len() != 1 {
40        return Err(layout_error(
41            "--world names one Foton world, so exactly one --dimension must be given".to_owned(),
42        ));
43    }
44
45    let mut targets: Vec<ImportTarget> = Vec::with_capacity(dimensions.len());
46    for dimension in dimensions {
47        let world = find_world(
48            &resolved.worlds,
49            &resolved.default_domain,
50            &dimension,
51            world_override,
52        )?;
53        if targets.iter().any(|target| target.world_key == world.key) {
54            return Err(layout_error(format!(
55                "two dimensions would both be imported into {}",
56                world.key
57            )));
58        }
59        let default_path = resolved
60            .save_path
61            .join(&world.domain)
62            .join("worlds")
63            .join(&world.name);
64        let output = storage
65            .create(&world.storage, &resolved.save_path, &default_path)
66            .map_err(|error| layout_error(format!("storage of {}: {error}", world.key)))?;
67        let (WorldStorageConfig::Disk { path }, Some(level_dir)) =
68            (output.storage, output.level_data_path)
69        else {
70            return Err(layout_error(format!(
71                "{} does not use disk storage, so there is nowhere to import into",
72                world.key
73            )));
74        };
75        let is_spawn_world = resolved
76            .domains
77            .iter()
78            .find(|domain| domain.name == resolved.default_domain)
79            .is_some_and(|domain| domain.default_world == world.key);
80        targets.push(ImportTarget {
81            dimension: dimension.key,
82            source_dir: dimension.dir,
83            world_key: world.key.clone(),
84            level_dir,
85            region_dir: PathBuf::from(path),
86            is_spawn_world,
87        });
88    }
89    Ok(targets)
90}

92fn find_world<'a>(
93    worlds: &'a [ResolvedWorldConfig],
94    default_domain: &str,
95    dimension: &SourceDimension,
96    world_override: Option<&str>,
97) -> Result<&'a ResolvedWorldConfig, ImportError> {
98    if let Some(wanted) = world_override {
99        let (domain, name) = wanted.split_once(':').unwrap_or((default_domain, wanted));
100        return worlds
101            .iter()
102            .find(|world| world.domain == domain && world.name == name)
103            .ok_or_else(|| {
104                layout_error(format!(
105                    "worlds.toml has no world {domain}:{name}; worlds are: {}",
106                    world_names(worlds)
107                ))
108            });
109    }
110    let by_name = worlds
111        .iter()
112        .find(|world| world.domain == default_domain && world.name == dimension.key.path);
113    let by_generator = || {
114        worlds
115            .iter()
116            .find(|world| world.domain == default_domain && world.generator == dimension.key)
117    };
118    by_name.or_else(by_generator).ok_or_else(|| {
119        layout_error(format!(
120            "no world in the default domain `{default_domain}` receives {}; worlds are: {}. \
121             Name one with --world",
122            dimension.key,
123            world_names(worlds)
124        ))
125    })
126}

128fn world_names(worlds: &[ResolvedWorldConfig]) -> String {
129    worlds
130        .iter()
131        .map(|world| world.key.to_string())
132        .collect::<Vec<_>>()
133        .join(", ")
134}

136/// Resolves a command-line dimension name: a bare vanilla path such as `nether`
137/// or `the_nether`, or a full id.
138#[must_use]
139pub fn parse_dimension_name(name: &str) -> Option<Identifier> {
140    let path = match name {
141        "nether" => "the_nether",
142        "end" => "the_end",
143        other => other,
144    };
145    path.parse().ok()
146}
