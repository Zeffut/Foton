8/// Writes `label: name x count, ...` for a non-empty tally.
9fn write_tally(
10    formatter: &mut Formatter<'_>,
11    label: &str,
12    tally: &BTreeMap<String, u64>,
13) -> fmt::Result {
14    if tally.is_empty() {
15        return Ok(());
16    }
17    let entries: Vec<String> = tally
18        .iter()
19        .map(|(name, count)| format!("{name} x{count}"))
20        .collect();
21    writeln!(formatter, "    {label}: {}", entries.join(", "))
22}

24impl Display for DimensionReport {
25    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
26        writeln!(formatter, "{} -> {}", self.dimension, self.world_key)?;
27        writeln!(
28            formatter,
29            "    {} regions, {} chunks found, {} imported",
30            self.regions, self.chunks_found, self.imported
31        )?;
32        writeln!(
33            formatter,
34            "    {} block entities, {} entities",
35            self.block_entities, self.entities
36        )?;
37        let not_full: u64 = self.not_full.values().sum();
38        if not_full > 0 {
39            writeln!(
40                formatter,
41                "    {not_full} chunks vanilla never finished generating were left out, so Foton generates them"
42            )?;
43        }
44        if self.unreadable > 0 {
45            writeln!(
46                formatter,
47                "    {} chunks could not be read and were left out:",
48                self.unreadable
49            )?;
50        }
51        for example in &self.unreadable_examples {
52            writeln!(formatter, "      {example}")?;
53        }
54        if self.outdated_chunks > 0 {
55            writeln!(
56                formatter,
57                "    {} chunks still at an older DataVersion were left out (--skip-outdated), so Foton generates them",
58                self.outdated_chunks
59            )?;
60        }
61        if self.outdated_entity_chunks > 0 {
62            writeln!(
63                formatter,
64                "    {} entity chunks still at an older DataVersion were left out; those chunks were imported without entities",
65                self.outdated_entity_chunks
66            )?;
67        }
68        if self.unreadable_entity_chunks > 0 {
69            writeln!(
70                formatter,
71                "    {} entity chunks could not be read; those chunks were imported without entities",
72                self.unreadable_entity_chunks
73            )?;
74        }
75        let issues = &self.issues;
76        write_tally(
77            formatter,
78            "unknown blocks, replaced by air",
79            &issues.unknown_blocks,
80        )?;
81        write_tally(
82            formatter,
83            "blocks with properties Foton rejects (property ignored)",
84            &issues.rejected_block_properties,
85        )?;
86        write_tally(
87            formatter,
88            "unknown biomes, replaced by plains",
89            &issues.unknown_biomes,
90        )?;
91        write_tally(
92            formatter,
93            "unknown block entities, dropped",
94            &issues.unknown_block_entities,
95        )?;
96        write_tally(
97            formatter,
98            "block entities on a block that does not take them, dropped",
99            &issues.misplaced_block_entities,
100        )?;
101        write_tally(
102            formatter,
103            "unknown entities, dropped",
104            &issues.unknown_entities,
105        )?;
106        if issues.malformed_entities > 0 {
107            writeln!(
108                formatter,
109                "    {} entities without a position or id, dropped",
110                issues.malformed_entities
111            )?;
112        }
113        if issues.chunks_without_light > 0 {
114            writeln!(
115                formatter,
116                "    {} chunks had no light data in the source; they will look dark until lit",
117                issues.chunks_without_light
118            )?;
119        }
120        Ok(())
121    }
122}

124impl Display for ImportReport {
125    fn fmt(&self, formatter: &mut Formatter<'_>) -> fmt::Result {
126        for dimension in &self.dimensions {
127            write!(formatter, "{dimension}")?;
128        }
129        match (self.level_written, self.seed) {
130            (true, Some(seed)) => writeln!(formatter, "level data written, seed {seed}")?,
131            _ => writeln!(
132                formatter,
133                "no level.dat found and no --seed given: level.toml was not written, so Foton \
134                 will pick its own seed and generate new terrain that does not match the old world"
135            )?,
136        }
137        writeln!(formatter, "done in {:.1?}", self.elapsed)
138    }
139}
