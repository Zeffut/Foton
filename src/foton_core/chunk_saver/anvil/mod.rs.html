1//! Import of vanilla Anvil (`.mca`) worlds into Foton's region format.
2//!
3//! Foton does not read Anvil at runtime. This module is the offline bridge for
4//! `foton import-anvil`: it reads a vanilla or Paper world and writes the chunks
5//! the way Foton's own chunk saver persists a `Full` chunk, so the server loads
6//! them as already generated and never regenerates them.
7//!
8//! ## No data fixing
9//!
10//! Foton has no `DataFixer`. The source must already carry the `DataVersion` of
11//! the targeted Minecraft version, in `level.dat` and in every chunk, and
12//! anything else is refused with instructions to upgrade it once with the
13//! vanilla (or Paper) server's `--forceUpgrade`.
14//!
15//! ## What is carried over
16//!
17//! Every block state, biomes, block entities (with their items and Paper's
18//! `PublicBukkitValues`), entities with scoreboard tags, custom names, custom
19//! data and `BukkitValues`, passengers, scheduled block and fluid ticks, pending
20//! post-processing, vanilla's light arrays, and the chunk's `ChunkBukkitValues`.
21//! From the level data: seed, game time, clocks, weather, difficulty, spawn,
22//! game rules and each dimension's world border.
23//!
24//! ## What is not
25//!
26//! Structure starts and references (the blocks of a structure are kept, but
27//! Foton will not know the structure is there), POI occupancy (villagers
28//! reclaim beds and workstations on their own), forced-chunk tickets,
29//! scoreboards, command storage, maps and player data. Chunks vanilla had not
30//! finished generating are left out so Foton generates them.

58/// Why an import was refused or failed.
59#[derive(Debug, Error)]
60pub enum ImportError {
61    /// A file could not be read or written.
62    #[error("{}: {source}", path.display())]
63    Io {
64        /// The file or folder involved.
65        path: PathBuf,
66        /// The underlying error.
67        source: io::Error,
68    },
69    /// A region file could not be opened.
70    #[error("{}: {source}", path.display())]
71    Region {
72        /// The region file.
73        path: PathBuf,
74        /// What was wrong with it.
75        source: RegionError,
76    },
77    /// The world was written by another Minecraft version.
78    #[error("{}", data_version_message(what, *found, *skippable))]
79    DataVersion {
80        /// What carried the version: `level.dat`, or a chunk.
81        what: String,
82        /// The version it carried, if it had one.
83        found: Option<i32>,
84        /// Whether `--skip-outdated` would get past it: true for a chunk,
85        /// false for `level.dat`.
86        skippable: bool,
87    },
88    /// The level data is missing something an import needs.
89    #[error("{0}")]
90    Level(String),
91    /// Nothing to import, or a dimension Foton cannot hold.
92    #[error("{0}")]
93    Layout(String),
94    /// The target Foton world already holds data.
95    #[error(
96        "{} already holds a Foton world; importing would mix two worlds. \
97         Pass --replace to move it aside first",
98        .0.display()
99    )]
100    TargetOccupied(PathBuf),
101}
102
103impl ImportError {
104    pub(crate) fn io(path: &Path, source: io::Error) -> Self {
105        Self::Io {
106            path: path.to_path_buf(),
107            source,
108        }
109    }
110
111    pub(crate) fn region(path: &Path, source: RegionError) -> Self {
112        Self::Region {
113            path: path.to_path_buf(),
114            source,
115        }
116    }
117}

119fn data_version_message(what: &str, found: Option<i32>, skippable: bool) -> String {
120    let found = found.map_or_else(
121        || "no DataVersion".to_owned(),
122        |version| format!("DataVersion {version}"),
123    );
124    let hint = if skippable {
125        " Vanilla's --forceUpgrade can leave a few chunks behind; if only a few are affected, \
126         pass --skip-outdated to leave them out and let Foton generate them."
127    } else {
128        ""
129    };
130    format!(
131        "{what} has {found}, but this Foton targets Minecraft {MC_VERSION} (DataVersion \
132         {WORLD_VERSION}). Foton does not convert worlds between versions. Open the world once \
133         with the vanilla {MC_VERSION} server using --forceUpgrade, which upgrades it in place \
134         (back it up first), then import it again.{hint}"
135    )
136}
