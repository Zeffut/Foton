7    pub(super) fn entities_to_persistent(entities: &[SharedEntity]) -> Vec<PersistentEntity> {
8        let mut visited = FxHashSet::default();
9        entities
10            .iter()
11            .filter(|entity| !entity.is_passenger())
12            .filter_map(|entity| {
13                Self::entity_to_persistent(
14                    entity,
15                    &mut visited,
16                    EntityPersistenceMode::ChunkSave,
17                    0,
18                )
19            })
20            .collect()
21    }

40    pub(crate) fn custom_name_to_persistent(custom_name: Option<&TextComponent>) -> Vec<u8> {
41        let Some(custom_name) = custom_name else {
42            return Vec::new();
43        };
44
45        let mut root = NbtCompound::new();
46        root.insert("CustomName", custom_name.to_codec_nbt());
47        let mut bytes = Vec::new();
48        root.write(&mut bytes);
49        bytes

52    pub(super) fn custom_name_from_persistent(
53        bytes: &[u8],
54        uuid: uuid::Uuid,
55    ) -> Option<TextComponent> {
56        if bytes.is_empty() {
57            return None;
58        }
59
60        let Ok(root) = read_borrowed_compound(&mut Cursor::new(bytes)) else {
61            tracing::warn!(
62                ?uuid,
63                "Failed to parse entity custom name NBT, defaulting to no custom name"
64            );
65            return None;
66        };
67        let root = simdnbt::borrow::NbtCompound::from(&root);
68        let tag = root.get("CustomName")?;
69        let custom_name = TextComponent::from_nbt(&tag.to_owned());
70        if custom_name.is_none() {
71            tracing::warn!(
72                ?uuid,
73                "Failed to decode entity custom name, defaulting to no custom name"
74            );
75            return None;
76        }
77        custom_name
78    }

90    pub(crate) fn compound_from_persistent(bytes: &[u8], owner: impl Debug) -> NbtCompound {
91        if bytes.is_empty() {
92            return NbtCompound::new();
93        }
94
95        let Ok(compound) = read_borrowed_compound(&mut Cursor::new(bytes)) else {
96            tracing::warn!(
97                ?owner,
98                "Failed to parse persisted compound NBT, defaulting to an empty compound"
99            );
100            return NbtCompound::new();
101        };
102        simdnbt::borrow::NbtCompound::from(&compound).to_owned()
103    }

147    pub(super) fn entity_to_persistent(
148        entity: &SharedEntity,
149        visited: &mut FxHashSet<i32>,
150        mode: EntityPersistenceMode,
151        depth: u32,
152    ) -> Option<PersistentEntity> {
153        if !Self::entity_should_persist(entity.as_ref(), mode) {
154            return None;
155        }
156
157        // The reader refuses a chain deeper than this, so writing one would
158        // save a chunk that can never be loaded again -- and building it
159        // recurses the same way the reader does, which is its own way to die.
160        //
161        // `>`, not `>=`, and the difference from `pool_element_to_persistent`
162        // is real: refusing here drops the passenger outright, so no `Nested`
163        // wrapper is spent on it, while an over-deep pool element still comes
164        // back as an `Empty` its parent wraps.
165        if depth > MAX_NESTING_DEPTH {
166            tracing::warn!(
167                uuid = ?entity.uuid(),
168                "Passenger chain is deeper than {MAX_NESTING_DEPTH}, saving it without its remaining riders"
169            );
170            return None;
171        }
172
173        if !visited.insert(entity.id()) {
174            tracing::warn!(
175                uuid = ?entity.uuid(),
176                "Entity passenger tree contains duplicate entity id {}, skipping duplicate save",
177                entity.id()
178            );
179            return None;
180        }
181
182        let pos = entity.position();
183        let stored_pos = if let Some(vehicle) = entity.vehicle() {
184            let vehicle_pos = vehicle.position();
185            DVec3::new(vehicle_pos.x, pos.y, vehicle_pos.z)
186        } else {
187            pos
188        };
189        let vel = entity.velocity();
190        let (yaw, pitch) = entity.rotation();
191        let fire_freeze = entity.fire_freeze_state();
192        let save_data = entity.base().save_data();
193
194        if !stored_pos.x.is_finite() || !stored_pos.y.is_finite() || !stored_pos.z.is_finite() {
195            tracing::warn!(
196                uuid = ?entity.uuid(),
197                "Entity has non-finite position {:?}, skipping save",
198                stored_pos
199            );
200            return None;
201        }
202
203        // The living half first, then the type's own, the way vanilla's
204        // `addAdditionalSaveData` chain runs `super` before the subclass.
205        // Without the first call a chunk file keeps no health, no potion
206        // effects and no attribute modifiers for anything alive.
207        let mut nbt = NbtCompound::new();
208        if let Some(living) = entity.as_living_entity() {
209            living.save_living(&mut nbt);
210        }
211        entity.save_additional(&mut nbt);
212        if let Some(plugin_spawn_reason) = entity.base().plugin_spawn_reason() {
213            nbt.insert("Paper.SpawnReason", plugin_spawn_reason.as_str());
214        }
215        let mut nbt_bytes = Vec::new();
216        nbt.write(&mut nbt_bytes);
217
218        let passengers = entity
219            .passengers()
220            .iter()
221            .filter_map(|passenger| {
222                Self::entity_to_persistent(passenger, visited, mode, depth + 1).map(Nested)
223            })
224            .collect();
225
226        Some(PersistentEntity {
227            entity_type: entity.entity_type().key.clone(),
228            uuid: *entity.uuid().as_bytes(),
229            pos: [stored_pos.x, stored_pos.y, stored_pos.z],
230            motion: [vel.x, vel.y, vel.z],
231            rotation: [yaw, pitch],
232            fall_distance: entity.fall_distance(),
233            remaining_fire_ticks: fire_freeze.remaining_fire_ticks(),
234            ticks_frozen: fire_freeze.ticks_frozen(),
235            is_in_powder_snow: fire_freeze.is_in_powder_snow(),
236            was_in_powder_snow: fire_freeze.was_in_powder_snow(),
237            has_visual_fire: fire_freeze.has_visual_fire(),
238            on_ground: entity.on_ground(),
239            no_gravity: save_data.no_gravity,
240            invulnerable: save_data.invulnerable,
241            air_supply: save_data.air_supply,
242            portal_cooldown: save_data.portal_cooldown,
243            custom_name_nbt: Self::custom_name_to_persistent(save_data.custom_name.as_ref()),
244            custom_name_visible: save_data.custom_name_visible,
245            silent: save_data.silent,
246            glowing: save_data.glowing,
247            tags: save_data.tags.iter().cloned().collect(),
248            custom_data_nbt: Self::compound_to_persistent(&save_data.custom_data),
249            bukkit_values_nbt: Self::compound_to_persistent(&save_data.bukkit_values),
250            nbt_data: nbt_bytes,
251            passengers,
252        })
253    }

255    pub(super) fn entity_should_save(entity: &dyn Entity) -> bool {
256        (!entity.is_removed()
257            || entity
258                .removal_reason()
259                .is_some_and(RemovalReason::should_save))
260            && entity.entity_type().can_serialize
261    }

291    pub(super) fn persistent_to_block_entity_at(
292        persistent: &PersistentBlockEntity,
293        pos: BlockPos,
294        level: Weak<World>,
295        state: BlockStateId,
296    ) -> Option<SharedBlockEntity> {
297        // Look up the block entity type
298        let block_entity_type_key = persistent.entity_type.as_ref()?;
299        let block_entity_type = REGISTRY.block_entity_types.by_key(block_entity_type_key)?;
300        if !block_entity_type.is_valid(state.get_block()) {
301            log::warn!(
302                "Skipping block entity {} at {pos:?}: block {} does not accept that type",
303                block_entity_type.key,
304                state.get_block().key,
305            );
306            return None;
307        }
308
309        // Parse and load NBT data
310        if persistent.nbt_data.is_empty() {
311            // No NBT data, just create the entity without loading
312            Some(BLOCK_ENTITIES.create_or_raw(block_entity_type, level, pos, state))
313        } else {
314            // Parse NBT from bytes as borrowed
315            let Ok(nbt) = read_borrowed_compound(&mut Cursor::new(&persistent.nbt_data)) else {
316                log::warn!(
317                    "Skipping block entity {} at {pos:?}: malformed NBT",
318                    block_entity_type.key,
319                );
320                return None;
321            };
322
323            // Create the block entity and load NBT
324            Some(BLOCK_ENTITIES.create_and_load_or_raw(block_entity_type, level, pos, state, &nbt))
325        }
326    }

329    pub(crate) fn persistent_to_entity_tree_at_level(
330        persistent: &PersistentEntity,
331        chunk_pos: ChunkPos,
332        level: &Weak<World>,
333    ) -> Vec<SharedEntity> {
334        let mut entities = Vec::new();
335        let Some(entity) = Self::persistent_to_entity_at_level(persistent, chunk_pos, level) else {
336            return entities;
337        };
338
339        entities.push(Arc::clone(&entity));
340        for persistent_passenger in &persistent.passengers {
341            Self::load_persistent_passenger_tree(
342                &persistent_passenger.0,
343                chunk_pos,
344                level,
345                &entity,
346                &mut entities,
347            );
348        }
349        entities
350    }

352    pub(super) fn load_persistent_passenger_tree(
353        persistent: &PersistentEntity,
354        chunk_pos: ChunkPos,
355        level: &Weak<World>,
356        vehicle: &SharedEntity,
357        entities: &mut Vec<SharedEntity>,
358    ) {
359        let Some(passenger) = Self::persistent_to_entity_at_level(persistent, chunk_pos, level)
360        else {
361            return;
362        };
363
364        EntityBase::restore_passenger_relationship(vehicle, &passenger);
365        entities.push(Arc::clone(&passenger));
366        for persistent_passenger in &persistent.passengers {
367            Self::load_persistent_passenger_tree(
368                &persistent_passenger.0,
369                chunk_pos,
370                level,
371                &passenger,
372                entities,
373            );
374        }
375    }

377    /// Converts one persistent entity to runtime format without loading passengers.
378    pub(super) fn persistent_to_entity_at_level(
379        persistent: &PersistentEntity,
380        chunk_pos: ChunkPos,
381        level: &Weak<World>,
382    ) -> Option<SharedEntity> {
383        use uuid::Uuid;
384
385        // Reconstruct base fields
386        let stored_pos = DVec3::new(persistent.pos[0], persistent.pos[1], persistent.pos[2]);
387        let mut velocity = DVec3::new(
388            persistent.motion[0],
389            persistent.motion[1],
390            persistent.motion[2],
391        );
392        let rotation = (persistent.rotation[0], persistent.rotation[1]);
393        let uuid = Uuid::from_bytes(persistent.uuid);
394
395        // Validate position is finite
396        if !stored_pos.x.is_finite() || !stored_pos.y.is_finite() || !stored_pos.z.is_finite() {
397            tracing::warn!(
398                ?uuid,
399                "Entity has non-finite position {:?}, skipping load",
400                stored_pos
401            );
402            return None;
403        }
404
405        if !rotation.0.is_finite() || !rotation.1.is_finite() {
406            tracing::warn!(
407                ?uuid,
408                "Entity has non-finite rotation {rotation:?}, skipping load"
409            );
410            return None;
411        }
412
413        let pos = Self::clamp_loaded_entity_position(stored_pos);
414
415        // Validate position is within expected chunk (sanity check)
416        let expected_chunk = ChunkPos::from_entity_pos(pos);
417        if chunk_pos != expected_chunk {
418            tracing::warn!(
419                ?uuid,
420                "Entity position {:?} doesn't match chunk {:?}, loading anyway",
421                pos,
422                chunk_pos
423            );
424        }
425
426        // Clamp motion values > 10.0 to 0 (vanilla behavior to prevent corruption)
427        if velocity.x.abs() > 10.0 {
428            velocity.x = 0.0;
429        }
430        if velocity.y.abs() > 10.0 {
431            velocity.y = 0.0;
432        }
433        if velocity.z.abs() > 10.0 {
434            velocity.z = 0.0;
435        }
436
437        // Look up entity type
438        let entity_type = REGISTRY.entity_types.by_key(&persistent.entity_type)?;
439        let save_data = Self::save_data_from_persistent(persistent, uuid);
440
441        // Parse NBT from bytes (or use empty compound data)
442        let nbt_bytes = if persistent.nbt_data.is_empty() {
443            // Empty compound body for `simdnbt::borrow::read_compound`.
444            &[0x00][..]
445        } else {
446            &persistent.nbt_data[..]
447        };
448
449        let Ok(nbt) = read_borrowed_compound(&mut Cursor::new(nbt_bytes)) else {
450            tracing::warn!(?uuid, "Failed to parse entity NBT, skipping");
451            return None;
452        };
453
454        let nbt_view = simdnbt::borrow::NbtCompound::from(&nbt);
455        let plugin_spawn_reason = nbt_view
456            .string("Paper.SpawnReason")
457            .and_then(|name| PluginSpawnReason::from_paper_name(name.to_str().as_ref()))
458            .unwrap_or_default();
459
460        let entity = ENTITIES.create_and_load_or_raw(
461            EntityLoadRequest {
462                entity_type,
463                position: pos,
464                uuid,
465                velocity,
466                rotation,
467                fall_distance: persistent.fall_distance,
468                fire_freeze: EntityFireFreezeState::from_parts(
469                    persistent.remaining_fire_ticks,
470                    persistent.ticks_frozen,
471                    persistent.is_in_powder_snow,
472                    persistent.was_in_powder_snow,
473                    persistent.has_visual_fire,
474                ),
475                on_ground: persistent.on_ground,
476                save_data,
477                world: Weak::clone(level),
478            },
479            &nbt,
480        );
481        entity.base().set_plugin_spawn_reason(plugin_spawn_reason);
482        Some(entity)
483    }
