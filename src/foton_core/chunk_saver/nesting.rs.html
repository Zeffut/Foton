1//! A depth ceiling for the self-nesting shapes in the persistence format.
2//!
3//! Two persisted types contain themselves: [`PersistentEntity`] holds its
4//! passengers, and `PersistentPoolElement::List` holds sub-elements. Both are
5//! read by a derived `SchemaRead`, which recurses once per nesting level, and
6//! `wincode` has no depth limit of its own. A corrupt or crafted region file
7//! nests tens of thousands of levels in a few hundred compressed bytes and
8//! overflows the stack.
9//!
10//! That is worse than an ordinary panic. A stack overflow is a SIGSEGV, so
11//! `clear_corrupt_chunk_if_unchanged` never runs, the slot is never
12//! quarantined, and the server dies again on the next start as soon as
13//! something asks for that chunk. The byte quota does not help: it bounds
14//! bytes, and a measured thirteen of them buy a level.
15//!
16//! Vanilla solves the same problem one layer down, in `NbtAccounter`: every
17//! compound and list calls `pushDepth`, which throws once past
18//! `MAX_STACK_DEPTH`. The exception is recoverable, so the chunk is rejected
19//! rather than the process killed. [`Nested`] is that mechanism, applied to the
20//! two edges that can actually recurse -- but with a ceiling of its own, for
21//! the reason [`MAX_NESTING_DEPTH`] gives.
22//!
23//! [`PersistentEntity`]: super::format::PersistentEntity
24

29use wincode::io::{Reader, Writer};
30use wincode::{ReadError, ReadResult, SchemaRead, SchemaWrite, WriteResult};

32/// Deepest nesting a persisted value may reach.
33///
34/// Vanilla's own ceiling is 512 (`NbtAccounter.MAX_STACK_DEPTH`), and that
35/// number does not transfer: vanilla's NBT frames are a hundred bytes or so,
36/// while a derived `SchemaRead` frame here is measured in kilobytes. On the
37/// 2 MiB stack tokio gives a worker, a debug build decodes 96 levels and
38/// overflows at 128, so 512 would be a stack overflow dressed as a limit.
39///
40/// 32 is chosen from the data instead of from vanilla's constant. The deepest
41/// `list_pool_element` nesting in the whole of vanilla's built-in datapacks is
42/// **1**, and a passenger chain past a handful is already something only
43/// `/ride` can build. 32 is thirty-two times the deepest real value and still
44/// a third of the measured debug ceiling, which leaves the frames already on
45/// the stack when the recursion starts their own room.
46///
47/// The trade-off is named rather than hidden, and it is bigger than it looks:
48/// a refusal is a decode failure, and a decode failure clears the slot so
49/// worldgen refills the column -- so a world that really did stack more than 32
50/// riders loses that column, not just its riders. That is why
51/// `quarantine_chunk_bytes` exists: the bytes are copied aside first, so the
52/// cost is a column to restore by hand rather than one destroyed. The price is
53/// still worth paying against a SIGSEGV that no handler can catch.
54pub(super) const MAX_NESTING_DEPTH: u32 = 32;
55
56thread_local! {
57    /// Levels of [`Nested`] currently open on this thread.
58    static DEPTH: Cell<u32> = const { Cell::new(0) };
59}
60
61/// Holds one level of nesting open, and gives it back on drop.
62struct DepthGuard;
63
64impl DepthGuard {
65    /// Opens a level, or refuses if the ceiling is already reached.
66    fn enter() -> ReadResult<Self> {
67        DEPTH.with(|depth| {
68            let current = depth.get();
69            if current >= MAX_NESTING_DEPTH {
70                return Err(ReadError::Custom(
71                    "persisted data nests deeper than the format allows",
72                ));
73            }
74            depth.set(current + 1);
75            Ok(Self)
76        })
77    }
78}
79
80impl Drop for DepthGuard {
81    fn drop(&mut self) {
82        DEPTH.with(|depth| depth.set(depth.get().saturating_sub(1)));
83    }
84}

86/// A value on a recursive edge of the persistence format.
87///
88/// Serializes exactly as the value it wraps -- the wrapper is transparent on
89/// the wire, so no saved world changes meaning -- but reading one costs a level
90/// of the depth budget.
91#[derive(Debug, Clone, PartialEq, Eq)]
92pub struct Nested<T>(pub T);
93
94impl<T> Nested<T> {
95    /// Unwraps to the value inside.
96    pub fn into_inner(self) -> T {
97        self.0
98    }
99}

101// SAFETY: `read` initializes `dst` only on the `Ok` path, and only with what
102// the inner schema's own `get` returned, which is initialized because `get`
103// returned `Ok`. The refusal path returns before `dst` is touched. `TYPE_META`
104// is left at the default `Dynamic`, which the trait documents as always safe.
105unsafe impl<'de, C, T> SchemaRead<'de, C> for Nested<T>
106where
107    C: ConfigCore,
108    T: SchemaRead<'de, C>,
109{
110    type Dst = Nested<T::Dst>;
111
112    fn read(reader: impl Reader<'de>, dst: &mut MaybeUninit<Self::Dst>) -> ReadResult<()> {
113        let _guard = DepthGuard::enter()?;
114        let inner = <T as SchemaRead<'de, C>>::get(reader)?;
115        dst.write(Nested(inner));
116        Ok(())
117    }
118}
119
120// SAFETY: both methods forward to the inner schema without writing a byte of
121// their own, so `size_of` reports exactly what `write` writes whenever the
122// inner schema does. `TYPE_META` is left at the default `Dynamic`.
123unsafe impl<C, T> SchemaWrite<C> for Nested<T>
124where
125    C: ConfigCore,
126    T: SchemaWrite<C>,
127    T::Src: Sized,
128{
129    type Src = Nested<T::Src>;
130
131    fn size_of(src: &Self::Src) -> WriteResult<usize> {
132        <T as SchemaWrite<C>>::size_of(&src.0)
133    }
134
135    fn write(writer: impl Writer, src: &Self::Src) -> WriteResult<()> {
136        <T as SchemaWrite<C>>::write(writer, &src.0)
137    }
138}

140#[cfg(test)]
141mod tests {
142    use std::thread::Builder;
143
144    use foton_utils::Identifier;
145
146    use super::*;
147    use crate::chunk_saver::{PersistentEntity, PersistentPoolElement};
148
149    /// Tokio's default worker and blocking thread stack. The decode runs on one
150    /// of those, so the ceiling has to fit inside it.
151    const PRODUCTION_STACK_BYTES: usize = 2 * 1024 * 1024;
152
153    /// A pool element tree that is `levels` deep below its root.
154    fn pool_element_nested(levels: u32) -> PersistentPoolElement {
155        let mut element = PersistentPoolElement::Empty;
156        for _ in 0..levels {
157            element = PersistentPoolElement::List {
158                elements: vec![Nested(element)],
159                projection: 0,
160            };
161        }
162        element
163    }
164
165    /// A passenger chain that is `levels` deep below its driver.
166    fn entity_nested(levels: u32) -> PersistentEntity {
167        let leaf = || PersistentEntity {
168            entity_type: Identifier::vanilla_static("pig"),
169            uuid: [7; 16],
170            pos: [0.0; 3],
171            motion: [0.0; 3],
172            rotation: [0.0; 2],
173            fall_distance: 0.0,
174            remaining_fire_ticks: 0,
175            ticks_frozen: 0,
176            is_in_powder_snow: false,
177            was_in_powder_snow: false,
178            has_visual_fire: false,
179            on_ground: false,
180            no_gravity: false,
181            invulnerable: false,
182            air_supply: 0,
183            portal_cooldown: 0,
184            custom_name_nbt: Vec::new(),
185            custom_name_visible: false,
186            silent: false,
187            glowing: false,
188            tags: Vec::new(),
189            custom_data_nbt: Vec::new(),
190            nbt_data: Vec::new(),
191            passengers: Vec::new(),
192        };
193
194        let mut entity = leaf();
195        for _ in 0..levels {
196            let mut vehicle = leaf();
197            vehicle.passengers = vec![Nested(entity)];
198            entity = vehicle;
199        }
200        entity
201    }
202
203    /// Runs `body` on a thread with exactly the stack production gives it.
204    fn on_a_production_sized_stack<T: Send + 'static>(
205        body: impl FnOnce() -> T + Send + 'static,
206    ) -> T {
207        Builder::new()
208            .stack_size(PRODUCTION_STACK_BYTES)
209            .spawn(body)
210            .expect("test thread should spawn")
211            .join()
212            .expect("decode should not kill its thread")
213    }
214
215    #[test]
216    fn nesting_past_the_ceiling_is_refused_instead_of_overflowing_the_stack() {
217        let bomb = pool_element_nested(MAX_NESTING_DEPTH + 1);
218        let encoded = wincode::serialize(&bomb).expect("the bomb should encode");
219
220        let decoded = on_a_production_sized_stack(move || {
221            wincode::deserialize_exact::<PersistentPoolElement>(&encoded).is_err()
222        });
223
224        assert!(
225            decoded,
226            "a tree one level past the ceiling must be refused, not decoded"
227        );
228    }
229
230    #[test]
231    fn the_ceiling_itself_still_decodes_on_a_production_sized_stack() {
232        // The outermost value is read directly, so `MAX_NESTING_DEPTH` wrapped
233        // levels sit exactly on the ceiling and must still be accepted. This is
234        // the test that earns the constant: vanilla's 512 is only usable here if
235        // 512 of *our* frames fit in the stack production actually gives them.
236        let deepest_allowed = entity_nested(MAX_NESTING_DEPTH);
237        let encoded = wincode::serialize(&deepest_allowed).expect("the chain should encode");
238
239        let depth = on_a_production_sized_stack(move || {
240            let mut entity = wincode::deserialize_exact::<PersistentEntity>(&encoded)
241                .expect("a chain on the ceiling should decode");
242            let mut depth = 0_u32;
243            while let Some(passenger) = entity.passengers.pop() {
244                entity = passenger.0;
245                depth += 1;
246            }
247            depth
248        });
249
250        assert_eq!(depth, MAX_NESTING_DEPTH);
251    }
252
253    #[test]
254    fn a_refused_read_gives_its_depth_budget_back() {
255        // The counter is thread-local and shared by both nesting shapes, so a
256        // refusal that leaked levels would poison every later read on the same
257        // worker thread -- one bad chunk would reject good ones.
258        on_a_production_sized_stack(|| {
259            let bomb = pool_element_nested(MAX_NESTING_DEPTH + 1);
260            let encoded = wincode::serialize(&bomb).expect("the bomb should encode");
261            assert!(wincode::deserialize_exact::<PersistentPoolElement>(&encoded).is_err());
262
263            let ordinary = pool_element_nested(3);
264            let encoded = wincode::serialize(&ordinary).expect("the tree should encode");
265            assert!(
266                wincode::deserialize_exact::<PersistentPoolElement>(&encoded).is_ok(),
267                "an ordinary tree must still decode after a refusal"
268            );
269        });
270    }
