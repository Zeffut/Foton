1//! Events: what happened, and who gets to hear about it before it takes effect.
2//!
3//! Nothing in vanilla corresponds to this. It exists because the server has no
4//! way to let anything outside `foton-core` observe or veto a state change, and
5//! every extension story — plugins, scripting, an audit log, an anti-cheat hook
6//! — starts by needing one.
7//!
8//! The shape is taken from `org.bukkit.event` on purpose, and only that far.
9//! Priorities, cancellation and the monitor contract are borrowed because
10//! `design/plugin-compatibility.md` measured which events a corpus of real
11//! plugins needs, and those plugins were written against those semantics: a bus
12//! that ordered listeners differently would run them in an order their authors
13//! never intended. Everything else about Bukkit's bus — its reflection, its
14//! annotations, its handler lists — is not copied, because Foton is not obliged
15//! to inherit twelve years of another project's compromises to be compatible
16//! with them.
17//!
18//! Unlike the block and item registries, this is not frozen after startup.
19//! Those hold game data that has to be stable once the world is running; this
20//! holds subscriptions, and a plugin being enabled or disabled while the server
21//! runs is ordinary.
22

77/// Something that happened, which a listener may observe and possibly stop.
78///
79/// Events are identified by their [`DowncastTypeKey`] rather than by `TypeId`.
80/// `AGENTS.md` requires it for anything a plugin can extend: `TypeId` is sound
81/// within one linked build and meaningless across two, and the whole point of
82/// this bus is that something outside this build will use it.
83pub trait Event: DowncastType + Send + Sync {
84    /// Whether this event has already been stopped.
85    ///
86    /// An event that cannot be stopped always answers `false`. That is why this
87    /// lives here with a default rather than in a separate trait: dispatch has
88    /// to ask every event the same question, and a bus that could only ask
89    /// *some* of them would need two dispatch paths for no gain.
90    fn is_cancelled(&self) -> bool {
91        false
92    }
93}
94
95/// When a listener runs, relative to the others listening for the same event.
96///
97/// The order is `org.bukkit.event.EventPriority`'s, including its one
98/// counter-intuitive part: `Lowest` runs *first* and `Highest` runs last, so
99/// that a listener at `Highest` gets the final word on the outcome.
100#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord)]
101pub enum EventPriority {
102    /// Runs first, and has the least say in the result.
103    Lowest,
104    /// Runs early.
105    Low,
106    /// The default.
107    #[default]
108    Normal,
109    /// Runs late.
110    High,
111    /// Runs last among listeners that may still change the outcome.
112    Highest,
113    /// Runs after everything, to observe the decision that was reached.
114    ///
115    /// A listener here must not change the outcome. Nothing enforces that —
116    /// nothing in Bukkit does either — but a listener that ignores it is
117    /// invisible to every other listener, which is its own punishment.
118    Monitor,
119}
120
121/// A listener's erased body: it downcasts back to its own event and runs.
122type Handler = Arc<dyn Fn(&mut dyn ErasedType) + Send + Sync>;
123
124struct Registration {
125    priority: EventPriority,
126    /// Whether this listener still runs once something has stopped the event.
127    ignores_cancelled: bool,
128    /// Who registered it, so that unloading them takes their listeners with it.
129    owner: Identifier,
130    handler: Handler,
131}
132
133/// Who is listening for what.
134///
135/// Listener lists are held behind an [`Arc`] and replaced rather than mutated,
136/// so dispatch clones one pointer and releases the lock before calling anyone.
137/// A listener that registers another listener — which plugins do — would
138/// otherwise deadlock on a lock it is already inside.
139#[derive(Default)]
140pub struct EventBus {
141    listeners: SyncRwLock<FxHashMap<DowncastTypeKey, Arc<Vec<Registration>>>>,
142}
143
144impl EventBus {
145    /// Creates an empty bus.
146    #[must_use]
147    pub fn new() -> Self {
148        Self::default()
149    }
150
151    /// Registers a listener for one event type.
152    ///
153    /// `ignores_cancelled` is Bukkit's flag inverted into the affirmative: a
154    /// listener that passes `true` still runs after something has stopped the
155    /// event, which is what a logger or a cleanup step wants and what an
156    /// ordinary listener does not.
157    pub fn listen<E, F>(
158        &self,
159        owner: Identifier,
160        priority: EventPriority,
161        ignores_cancelled: bool,
162        handler: F,
163    ) where
164        E: Event,
165        F: Fn(&mut E) + Send + Sync + 'static,
166    {
167        let erased: Handler = Arc::new(move |event: &mut dyn ErasedType| {
168            // The key was matched before dispatch, so this only fails if two
169            // types share a key -- which `DowncastType` makes an unsafe promise
170            // not to do. Silently skipping beats panicking in a listener loop.
171            if let Some(typed) = event.downcast_mut::<E>() {
172                handler(typed);
173            }
174        });
175
176        let mut listeners = self.listeners.write();
177        let existing = listeners
178            .get(&E::TYPE_KEY)
179            .map_or(&[][..], |list| list.as_slice());
180        let mut next = Vec::with_capacity(existing.len() + 1);
181        next.extend(existing.iter().map(Registration::clone_shallow));
182        next.push(Registration {
183            priority,
184            ignores_cancelled,
185            owner,
186            handler: erased,
187        });
188        // A stable sort keeps registration order within one priority, which is
189        // the only tie-break a listener can reason about.
190        next.sort_by_key(|registration| registration.priority);
191        listeners.insert(E::TYPE_KEY, Arc::new(next));
192    }
193
194    /// Registers a listener that stops running once the event is cancelled.
195    ///
196    /// The common case, spelled shorter so the common case reads shorter.
197    pub fn on<E, F>(&self, owner: Identifier, handler: F)
198    where
199        E: Event,
200        F: Fn(&mut E) + Send + Sync + 'static,
201    {
202        self.listen::<E, F>(owner, EventPriority::Normal, false, handler);
203    }
204
205    /// Drops every listener one owner registered.
206    ///
207    /// A plugin that is disabled must stop being called, and it cannot be
208    /// trusted to have kept a handle to each of its own registrations.
209    pub fn forget(&self, owner: &Identifier) {
210        let mut listeners = self.listeners.write();
211        listeners.retain(|_, list| {
212            if !list.iter().any(|registration| registration.owner == *owner) {
213                return true;
214            }
215            let kept: Vec<Registration> = list
216                .iter()
217                .filter(|registration| registration.owner != *owner)
218                .map(Registration::clone_shallow)
219                .collect();
220            if kept.is_empty() {
221                return false;
222            }
223            *list = Arc::new(kept);
224            true
225        });
226    }
227
228    /// Runs every listener for this event, in priority order.
229    ///
230    /// The event is handed out mutably: a listener changes the outcome by
231    /// changing the event, which is the only channel it gets.
232    pub fn fire<E: Event>(&self, event: &mut E) {
233        let Some(registrations) = self.listeners.read().get(&E::TYPE_KEY).map(Arc::clone) else {
234            return;
235        };
236
237        for registration in registrations.iter() {
238            if event.is_cancelled() && !registration.ignores_cancelled {
239                continue;
240            }
241            (registration.handler)(event);
242        }
243    }
244
245    /// How many listeners one event type has. For diagnostics and tests.
246    #[must_use]
247    pub fn listener_count<E: Event>(&self) -> usize {
248        self.listeners
249            .read()
250            .get(&E::TYPE_KEY)
251            .map_or(0, |list| list.len())
252    }
253}
254
255impl Registration {
256    /// Copies the registration, sharing the handler rather than the closure.
257    fn clone_shallow(&self) -> Self {
258        Self {
259            priority: self.priority,
260            ignores_cancelled: self.ignores_cancelled,
261            owner: self.owner.clone(),
262            handler: Arc::clone(&self.handler),
263        }
264    }
265}
