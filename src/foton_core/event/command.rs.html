1//! Events about commands.
2
3use std::sync::Arc;
4
5use foton_utils::downcast::{DowncastType, DowncastTypeKey};
6
7use super::Event;
8use crate::player::Player;
9
10/// Somebody typed a command, before the server tried to run it.
11///
12/// A listener that claims the command stops the server from parsing it at all.
13/// That is what lets a plugin own a name the server has never heard of, which
14/// is the only way a plugin command can work: Foton's dispatcher is a
15/// Brigadier tree built at startup and a plugin's command is not in it.
16///
17/// Claiming a name the server *does* know shadows the built-in, so a listener
18/// is expected to claim only what it was actually asked to own.
19pub struct CommandEvent {
20    player: Option<Arc<Player>>,
21    command: String,
22    handled: bool,
23}
24
25// SAFETY: This Foton-owned key uniquely identifies the concrete Rust type
26// within the process.
27unsafe impl DowncastType for CommandEvent {
28    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:event/command");
29}
30
31impl Event for CommandEvent {}
32
33impl CommandEvent {
34    /// Creates the event for one typed command, without its leading slash.
35    #[must_use]
36    pub fn new(player: Option<Arc<Player>>, command: impl Into<String>) -> Self {
37        Self {
38            player,
39            command: command.into(),
40            handled: false,
41        }
42    }
43
44    /// Who typed it, when that was a player rather than the console.
45    #[must_use]
46    pub const fn player(&self) -> Option<&Arc<Player>> {
47        self.player.as_ref()
48    }
49
50    /// The command line, without its leading slash.
51    #[must_use]
52    pub fn command(&self) -> &str {
53        &self.command
54    }
55
56    /// Whether somebody has taken responsibility for running this.
57    #[must_use]
58    pub const fn is_handled(&self) -> bool {
59        self.handled
60    }
61
62    /// Claims the command, so the server will not try to parse it.
63    pub const fn set_handled(&mut self, handled: bool) {
64        self.handled = handled;
65    }
66}
