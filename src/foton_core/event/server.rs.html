10/// A client asked for the server-list entry.
11///
12/// Bukkit parity: `ServerListPingEvent`. What a listener leaves as the MOTD
13/// and the player limit is what the client is shown.
14pub struct ServerListPingEvent {
15    address: IpAddr,
16    motd: TextComponent,
17    online: i32,
18    max_players: i32,
19}
20
21// SAFETY: This Foton-owned key uniquely identifies the concrete Rust type.
22unsafe impl DowncastType for ServerListPingEvent {
23    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:event/server_list_ping");
24}
25
26impl Event for ServerListPingEvent {}
27
28impl ServerListPingEvent {
29    /// Creates the event with the configured entry.
30    #[must_use]
31    pub const fn new(address: IpAddr, motd: TextComponent, online: i32, max_players: i32) -> Self {
32        Self {
33            address,
34            motd,
35            online,
36            max_players,
37        }
38    }
39
40    /// Who is asking.
41    #[must_use]
42    pub const fn address(&self) -> IpAddr {
43        self.address
44    }
45
46    /// The message of the day.
47    #[must_use]
48    pub const fn motd(&self) -> &TextComponent {
49        &self.motd
50    }
51
52    /// Changes the message of the day.
53    pub fn set_motd(&mut self, motd: TextComponent) {
54        self.motd = motd;
55    }
56
57    /// How many players are online.
58    #[must_use]
59    pub const fn online(&self) -> i32 {
60        self.online
61    }
62
63    /// The player limit shown.
64    #[must_use]
65    pub const fn max_players(&self) -> i32 {
66        self.max_players
67    }
68
69    /// Changes the player limit shown.
70    pub const fn set_max_players(&mut self, max_players: i32) {
71        self.max_players = max_players;
72    }
73}

75/// One game tick happened.
76///
77/// Not vanilla, and not gameplay: it exists so that something outside
78/// `foton-core` can do work on the tick thread without `foton-core` knowing
79/// what that work is. A plugin scheduler is the first user, and it is the
80/// reason `runTask` can promise what Bukkit promises -- that a task body runs
81/// where it is safe to touch the world.
82///
83/// Fired once per tick, after the tick's own work. A listener here runs inside
84/// the tick and delays it, which is the price of being allowed to touch
85/// anything at all.
86pub struct ServerTickEvent {
87    tick: u64,
88}
89
90// SAFETY: This Foton-owned key uniquely identifies the concrete Rust type
91// within the process.
92unsafe impl DowncastType for ServerTickEvent {
93    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:event/server_tick");
94}
95
96impl Event for ServerTickEvent {}
97
98impl ServerTickEvent {
99    /// Creates the event for one tick.
100    #[must_use]
101    pub const fn new(tick: u64) -> Self {
102        Self { tick }
103    }
104
105    /// Which tick this was, counted from the server starting.
106    #[must_use]
107    pub const fn tick(&self) -> u64 {
108        self.tick
109    }
110}
