7/// One game tick happened.
8///
9/// Not vanilla, and not gameplay: it exists so that something outside
10/// `foton-core` can do work on the tick thread without `foton-core` knowing
11/// what that work is. A plugin scheduler is the first user, and it is the
12/// reason `runTask` can promise what Bukkit promises -- that a task body runs
13/// where it is safe to touch the world.
14///
15/// Fired once per tick, after the tick's own work. A listener here runs inside
16/// the tick and delays it, which is the price of being allowed to touch
17/// anything at all.
18pub struct ServerTickEvent {
19    tick: u64,
20}
21
22// SAFETY: This Foton-owned key uniquely identifies the concrete Rust type
23// within the process.
24unsafe impl DowncastType for ServerTickEvent {
25    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:event/server_tick");
26}
27
28impl Event for ServerTickEvent {}
29
30impl ServerTickEvent {
31    /// Creates the event for one tick.
32    #[must_use]
33    pub const fn new(tick: u64) -> Self {
34        Self { tick }
35    }
36
37    /// Which tick this was, counted from the server starting.
38    #[must_use]
39    pub const fn tick(&self) -> u64 {
40        self.tick
41    }
42}
