5/// A player command before command dispatch.
6pub struct PlayerCommandPreprocessEvent {
7    player_id: Uuid,
8    message: String,
9    cancelled: bool,
10}
11// SAFETY: This Foton-owned key uniquely identifies the concrete Rust type.
12unsafe impl DowncastType for PlayerCommandPreprocessEvent {
13    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("foton:event/player_command_preprocess");
14}
15impl Event for PlayerCommandPreprocessEvent {
16    fn is_cancelled(&self) -> bool {
17        self.cancelled
18    }
19}
20impl PlayerCommandPreprocessEvent {
21    /// Called by Foton when it fires the event. A plugin receives one of these; it never builds one.
22    #[must_use]
23    pub const fn new(player_id: Uuid, message: String) -> Self {
24        Self {
25            player_id,
26            message,
27            cancelled: false,
28        }
29    }
30    /// Who did it.
31    #[must_use]
32    pub const fn player_id(&self) -> Uuid {
33        self.player_id
34    }
35    /// The command line as typed, leading slash included.
36    #[must_use]
37    pub fn message(&self) -> &str {
38        &self.message
39    }
40    /// Rewrites the line before the server parses it.
41    pub fn set_message(&mut self, message: String) {
42        self.message = message;
43    }
44    /// Whether a listener has stopped this from happening.
45    #[must_use]
46    pub const fn is_cancelled(&self) -> bool {
47        self.cancelled
48    }
49    /// Stops this from happening, or lets it happen again.
50    pub const fn set_cancelled(&mut self, cancelled: bool) {
51        self.cancelled = cancelled;
52    }
