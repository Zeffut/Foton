1//! Boss bars a command owns rather than a boss.
2//!
3//! Vanilla parity: `CustomBossEvent` and `CustomBossEvents`. A boss's bar
4//! lives and dies with the boss, and its progress is whatever fraction of its
5//! health is left. A named bar has neither: it is created by `/bossbar add`,
6//! it survives a restart, and its fill is a pair of whole numbers a datapack
7//! sets -- `value` out of `max` -- because a datapack counting objectives has
8//! nothing to divide.
9//!
10//! Two player sets, and the difference matters. [`ServerBossEvent`] holds the
11//! clients the bar is currently on. [`CustomBossEvent`] holds the UUIDs it is
12//! *assigned* to, which is what `/bossbar set <id> players` writes and what
13//! goes in the save file. A viewer who logs out leaves the first and stays in
14//! the second, so the bar is waiting for them when they come back.
15//!
16//! Vanilla keeps the whole collection in the overworld's data storage and
17//! reaches it through the server. Foton's scoreboards and command storage are
18//! per domain, and `execute store` addresses all three the same way, so these
19//! follow them rather than vanilla's single global.
20

43/// The fill a bar starts at.

45/// Vanilla parity: `CustomBossEvent.DEFAULT_MAX`.
46pub const DEFAULT_MAX: i32 = 100;

48/// A boss bar addressed by name instead of by the boss behind it.
49///
50/// Vanilla parity: `CustomBossEvent`.
51#[derive(Debug)]
52pub struct CustomBossEvent {
53    custom_id: Identifier,
54    event: ServerBossEvent,
55    /// Everyone the bar is assigned to, online or not.
56    assigned: SyncMutex<BTreeSet<Uuid>>,
57    value: AtomicI32,
58    max: AtomicI32,
59}
60
61impl CustomBossEvent {
62    /// Creates an empty bar under `custom_id`.
63    ///
64    /// Vanilla parity: the `CustomBossEvent` constructor, which starts at zero
65    /// progress rather than the full bar a boss's own starts at.
66    #[must_use]
67    pub fn new(id: Uuid, custom_id: Identifier, name: TextComponent) -> Self {
68        let event = ServerBossEvent::new(id, name, BossBarColor::White, BossBarOverlay::Progress);
69        event.set_progress(0.0);
70        Self {
71            custom_id,
72            event,
73            assigned: SyncMutex::new(BTreeSet::new()),
74            value: AtomicI32::new(0),
75            max: AtomicI32::new(DEFAULT_MAX),
76        }
77    }
78
79    /// The name a command addresses this bar by.
80    #[must_use]
81    pub const fn custom_id(&self) -> &Identifier {
82        &self.custom_id
83    }
84
85    /// The bar itself, for the title, color, overlay and visibility a
86    /// [`ServerBossEvent`] already owns.
87    #[must_use]
88    pub const fn event(&self) -> &ServerBossEvent {
89        &self.event
90    }
91
92    /// Vanilla parity: `CustomBossEvent.value`.
93    #[must_use]
94    pub fn value(&self) -> i32 {
95        self.value.load(Ordering::Relaxed)
96    }
97
98    /// Vanilla parity: `CustomBossEvent.max`.
99    #[must_use]
100    pub fn max(&self) -> i32 {
101        self.max.load(Ordering::Relaxed)
102    }
103
104    /// Vanilla parity: `CustomBossEvent.setValue`.
105    pub fn set_value(&self, value: i32) {
106        self.value.store(value, Ordering::Relaxed);
107        self.update_progress();
108        // Unconditionally, the way vanilla's `setDirty()` is: the fill may
109        // round to the same float against a large maximum and the value still
110        // has to reach the save file.
111        self.event.mark_dirty();
112    }
113
114    /// Vanilla parity: `CustomBossEvent.setMax`.
115    pub fn set_max(&self, max: i32) {
116        self.max.store(max, Ordering::Relaxed);
117        self.update_progress();
118        self.event.mark_dirty();
119    }
120
121    fn update_progress(&self) {
122        let progress = self.value() as f32 / self.max() as f32;
123        self.event.set_progress(progress.clamp(0.0, 1.0));
124    }
125
126    /// Assigns the bar to one more player and shows it to them.
127    ///
128    /// Vanilla parity: `CustomBossEvent.addPlayer`.
129    pub fn add_player(&self, player: &Arc<Player>) {
130        self.event.add_player(player);
131        if self.assigned.lock().insert(player.uuid()) {
132            self.event.mark_dirty();
133        }
134    }
135
136    /// Takes the bar off one player and unassigns them from it.
137    ///
138    /// Vanilla parity: `CustomBossEvent.removePlayer`.
139    pub fn remove_player(&self, player: &Player) {
140        self.event.remove_player(player);
141        if self.assigned.lock().remove(&player.uuid()) {
142            self.event.mark_dirty();
143        }
144    }
145
146    /// Vanilla parity: `CustomBossEvent.removeAllPlayers`.
147    pub fn remove_all_players(&self) {
148        self.event.remove_all_players();
149        let emptied = {
150            let mut assigned = self.assigned.lock();
151            let had_any = !assigned.is_empty();
152            assigned.clear();
153            had_any
154        };
155        if emptied {
156            self.event.mark_dirty();
157        }
158    }
159
160    /// The players this bar is assigned to, whether or not they are online.
161    #[must_use]
162    pub fn assigned_players(&self) -> Vec<Uuid> {
163        self.assigned.lock().iter().copied().collect()
164    }
165
166    /// Replaces the assignment with exactly `targets`.
167    ///
168    /// Returns whether anything changed, which is what `/bossbar set players`
169    /// reports as a failure when it did not.
170    ///
171    /// Vanilla parity: `CustomBossEvent.setPlayers`.
172    pub fn set_players(&self, targets: &[Arc<Player>]) -> bool {
173        let wanted = targets
174            .iter()
175            .map(|player| player.uuid())
176            .collect::<BTreeSet<_>>();
177        let current = self.assigned.lock().clone();
178
179        let dropped = current.difference(&wanted).copied().collect::<Vec<_>>();
180        for uuid in &dropped {
181            // Only a player who is online is on a screen to take the bar off.
182            if let Some(player) = self
183                .event
184                .players()
185                .into_iter()
186                .find(|player| player.uuid() == *uuid)
187            {
188                self.event.remove_player(&player);
189            }
190            self.assigned.lock().remove(uuid);
191        }
192
193        let mut added = 0;
194        for player in targets {
195            if current.contains(&player.uuid()) {
196                continue;
197            }
198            self.add_player(player);
199            added += 1;
200        }
201
202        let changed = !dropped.is_empty() || added > 0;
203        if changed {
204            self.event.mark_dirty();
205        }
206        changed
207    }
208
209    /// Puts the bar back on a player who was assigned to it before they left.
210    ///
211    /// Vanilla parity: `CustomBossEvent.onPlayerConnect`.
212    pub fn on_player_connect(&self, player: &Arc<Player>) {
213        if self.assigned.lock().contains(&player.uuid()) {
214            self.event.add_player(player);
215        }
216    }
217
218    /// Drops a leaving player from the viewers without unassigning them.
219    ///
220    /// Vanilla parity: `CustomBossEvent.onPlayerDisconnect`, which calls
221    /// `super.removePlayer` on purpose: logging out is not the same as being
222    /// taken off the bar, and the assignment has to survive it.
223    pub fn on_player_disconnect(&self, player: &Player) {
224        self.event.remove_player(player);
225    }

227    /// The bar's name as a command prints it.
228    ///
229    /// Vanilla parity: `CustomBossEvent.getDisplayName` -- the title in square
230    /// brackets, tinted by the bar's color, carrying its id as both a hover
231    /// and a shift-click insertion.
232    #[must_use]
233    pub fn display_name(&self) -> TextComponent {
234        let id = self.custom_id.to_string();
235        translations::CHAT_SQUARE_BRACKETS
236            .message([self.event.name()])
237            .component()
238            .color(self.event.color().chat_color())
239            .hover_event(HoverEvent::show_text(TextComponent::plain(id.clone())))
240            .insertion(id)
241    }
242}

245///
246/// Vanilla parity: `CustomBossEvent.Packed`. The name travels as component NBT
247/// for the same reason an entity's custom name does: a text component is a

252    visible: bool,

254    max: i32,

256    overlay: String,
257    darken_screen: bool,
258    play_boss_music: bool,
259    create_world_fog: bool,

274/// The named boss bars of one Foton domain.
275///
276/// Vanilla parity: `CustomBossEvents`.
277pub struct CustomBossEvents {
278    bars: SyncMutex<FxHashMap<Identifier, Arc<CustomBossEvent>>>,
279    revision: AtomicU64,
280    saved_revision: AtomicU64,
281}
282
283impl CustomBossEvents {
284    /// Creates an empty, clean collection.
285    #[must_use]
286    pub fn new() -> Self {
287        Self {
288            bars: SyncMutex::new(FxHashMap::default()),
289            revision: AtomicU64::new(0),
290            saved_revision: AtomicU64::new(0),
291        }
292    }
293
294    /// Returns the bar `id` names.
295    ///
296    /// Vanilla parity: `CustomBossEvents.get`.
297    #[must_use]
298    pub fn get(&self, id: &Identifier) -> Option<Arc<CustomBossEvent>> {
299        self.bars.lock().get(id).map(Arc::clone)
300    }
301
302    /// Creates a bar under `id`.
303    ///
304    /// Vanilla parity: `CustomBossEvents.create`. The caller is responsible
305    /// for refusing an id that already exists; this replaces it, the way
306    /// vanilla's `Map.put` does.
307    pub fn create(&self, id: Identifier, name: TextComponent) -> Arc<CustomBossEvent> {
308        let bar = Arc::new(CustomBossEvent::new(Uuid::new_v4(), id.clone(), name));
309        self.bars.lock().insert(id, Arc::clone(&bar));
310        self.revision.fetch_add(1, Ordering::Release);
311        bar
312    }
313
314    /// Vanilla parity: `CustomBossEvents.remove`.
315    pub fn remove(&self, id: &Identifier) -> bool {
316        let removed = self.bars.lock().remove(id).is_some();
317        if removed {
318            self.revision.fetch_add(1, Ordering::Release);
319        }
320        removed
321    }
322
323    /// Every bar id, in stable resource-location order.
324    ///
325    /// Vanilla iterates a hash map and takes whatever order the JVM gives it.
326    /// A command that lists bars should not reshuffle them between runs, so
327    /// these are sorted.
328    ///
329    /// Vanilla parity: `CustomBossEvents.getIds`.
330    #[must_use]
331    pub fn ids(&self) -> Vec<Identifier> {
332        let mut ids = self.bars.lock().keys().cloned().collect::<Vec<_>>();
333        ids.sort_by_cached_key(ToString::to_string);
334        ids
335    }
336
337    /// Every bar, in the same order as [`Self::ids`].
338    ///
339    /// Vanilla parity: `CustomBossEvents.getEvents`.
340    #[must_use]
341    pub fn events(&self) -> Vec<Arc<CustomBossEvent>> {
342        let mut events = self
343            .bars
344            .lock()
345            .values()
346            .map(Arc::clone)
347            .collect::<Vec<_>>();
348        events.sort_by_cached_key(|bar| bar.custom_id().to_string());
349        events
350    }
351
352    /// Vanilla parity: `CustomBossEvents.onPlayerConnect`.
353    pub fn on_player_connect(&self, player: &Arc<Player>) {
354        for bar in self.events() {
355            bar.on_player_connect(player);
356        }
357    }
358
359    /// Vanilla parity: `CustomBossEvents.onPlayerDisconnect`.
360    pub fn on_player_disconnect(&self, player: &Player) {
361        for bar in self.events() {
362            bar.on_player_disconnect(player);
363        }
364    }
365
366    fn from_persistent(persistent: PersistentCustomBossEvents) -> io::Result<Self> {
367        let mut bars = FxHashMap::default();
368        for (raw_id, packed) in persistent.bars {
369            let id = raw_id.parse::<Identifier>().map_err(|error| {
370                io::Error::new(
371                    io::ErrorKind::InvalidData,
372                    format!("invalid boss bar id '{raw_id}': {error}"),
373                )
374            })?;
375            bars.insert(id.clone(), Arc::new(load_bar(&id, packed)?));
376        }
377        Ok(Self {
378            bars: SyncMutex::new(bars),
379            revision: AtomicU64::new(0),
380            saved_revision: AtomicU64::new(0),
381        })
382    }
383
384    /// Folds every bar's own change flag into the collection's revision.
385    ///
386    /// A bar reports its own changes through [`ServerBossEvent::take_dirty`],
387    /// which is Foton's stand-in for the `setDirty` callback vanilla's
388    /// `CustomBossEvent` holds into its saved data. Draining them here is what
389    /// turns "a color changed" into "this domain needs writing".
390    fn absorb_bar_changes(&self) {
391        let dirty = self
392            .bars
393            .lock()
394            .values()
395            .filter(|bar| bar.event.take_dirty())
396            .count();
397        if dirty > 0 {
398            self.revision.fetch_add(1, Ordering::Release);
399        }
400    }
401
402    fn pending_save(&self) -> Option<CustomBossEventsSaveSnapshot> {
403        self.absorb_bar_changes();
404        let revision = self.revision.load(Ordering::Acquire);
405        if revision == self.saved_revision.load(Ordering::Acquire) {
406            return None;
407        }
408
409        let bars = self
410            .bars
411            .lock()
412            .iter()
413            .map(|(id, bar)| (id.to_string(), pack_bar(bar)))
414            .collect::<BTreeMap<_, _>>();
415        Some(CustomBossEventsSaveSnapshot {
416            revision,
417            state: PersistentCustomBossEvents { bars },
418        })
419    }
420
421    fn mark_saved(&self, revision: u64) {
422        self.saved_revision.fetch_max(revision, Ordering::Release);
423    }
424}

432/// Vanilla parity: `CustomBossEvent.pack`.
433fn pack_bar(bar: &CustomBossEvent) -> PersistentBossBar {
434    let event = bar.event();
435    let properties = event.properties();
436    let mut root = NbtCompound::new();
437    root.insert("Name", event.name().to_codec_nbt());
438    let mut name = Vec::new();
439    root.write(&mut name);
440
441    PersistentBossBar {
442        name,
443        visible: event.is_visible(),
444        value: bar.value(),
445        max: bar.max(),
446        color: event.color().serialized_name().to_owned(),
447        overlay: event.overlay().serialized_name().to_owned(),
448        darken_screen: properties.darken_screen,
449        play_boss_music: properties.play_boss_music,
450        create_world_fog: properties.create_world_fog,
451        players: bar
452            .assigned_players()
453            .into_iter()
454            .map(Uuid::into_bytes)
455            .collect(),
456    }
457}

459/// Vanilla parity: `CustomBossEvent.load`.
460fn load_bar(id: &Identifier, packed: PersistentBossBar) -> io::Result<CustomBossEvent> {
461    let name = read_bar_name(id, &packed.name)?;
462    let bar = CustomBossEvent::new(Uuid::new_v4(), id.clone(), name);
463    let event = bar.event();
464
465    event.set_visible(packed.visible);
466    bar.set_value(packed.value);
467    bar.set_max(packed.max);
468    event.set_color(bar_enum(
469        id,
470        "color",
471        &packed.color,
472        BossBarColor::from_serialized_name(&packed.color),
473    )?);
474    event.set_overlay(bar_enum(
475        id,
476        "overlay",
477        &packed.overlay,
478        BossBarOverlay::from_serialized_name(&packed.overlay),
479    )?);
480    event.set_darken_screen(packed.darken_screen);
481    event.set_play_boss_music(packed.play_boss_music);
482    event.set_create_world_fog(packed.create_world_fog);
483    {
484        let mut assigned = bar.assigned.lock();
485        for bytes in packed.players {
486            assigned.insert(Uuid::from_bytes(bytes));
487        }
488    }
489
490    // Everything above went through a setter that marks the bar changed. A bar
491    // that has just been read off disk has not changed, and leaving the flag
492    // set would rewrite every file on the first autosave.
493    let _ = event.take_dirty();
494    Ok(bar)
495}

497fn read_bar_name(id: &Identifier, bytes: &[u8]) -> io::Result<TextComponent> {
498    let root = read_borrowed_compound(&mut Cursor::new(bytes)).map_err(|error| {
499        io::Error::new(
500            io::ErrorKind::InvalidData,
501            format!("invalid name NBT for boss bar '{id}': {error:?}"),
502        )
503    })?;
504    let root = simdnbt::borrow::NbtCompound::from(&root);
505    root.get("Name")
506        .and_then(|tag| TextComponent::from_nbt(&tag.to_owned()))
507        .ok_or_else(|| {
508            io::Error::new(
509                io::ErrorKind::InvalidData,
510                format!("boss bar '{id}' has no readable name"),
511            )
512        })
513}
514
515fn bar_enum<T>(id: &Identifier, field: &str, raw: &str, parsed: Option<T>) -> io::Result<T> {
516    parsed.ok_or_else(|| {
517        io::Error::new(
518            io::ErrorKind::InvalidData,
519            format!("boss bar '{id}' has an unknown {field} '{raw}'"),
520        )
521    })
522}

524/// Loaded boss bars keyed by Foton domain.
525pub struct DomainCustomBossEvents {
526    domains: BTreeMap<String, CustomBossEvents>,
527    save_lock: AsyncMutex<()>,
528}
529
530impl DomainCustomBossEvents {
531    /// Loads one set of boss bars through each domain's default world.
532    pub async fn load(worlds: &WorldMap) -> io::Result<Self> {
533        let mut names = worlds.domain_names().collect::<Vec<_>>();
534        names.sort_unstable();
535        let mut domains = BTreeMap::new();
536        for domain in names {
537            let world = domain_default_world(worlds, domain)?;
538            let persistent: PersistentCustomBossEvents = world
539                .saved_data
540                .load_or_default(saved_data_names::CUSTOM_BOSS_EVENTS)
541                .await
542                .map_err(|error| boss_bar_io_error(domain, error))?;
543            let events = CustomBossEvents::from_persistent(persistent)
544                .map_err(|error| boss_bar_io_error(domain, error))?;
545            domains.insert(domain.to_owned(), events);
546        }
547        Ok(Self {
548            domains,
549            save_lock: AsyncMutex::new(()),
550        })
551    }
552
553    /// Returns the boss bars of a domain.
554    #[must_use]
555    pub fn get(&self, domain: &str) -> Option<&CustomBossEvents> {
556        self.domains.get(domain)
557    }
558
559    /// Runs `visitor` over every domain's bars.
560    ///
561    /// Player join and leave are server-wide events, and a player is only ever
562    /// in one domain at a time, but a bar can be assigned to somebody who is
563    /// not in its own; vanilla has one global collection and no such question
564    /// to answer.
565    pub fn for_each(&self, visitor: impl Fn(&CustomBossEvents)) {
566        for events in self.domains.values() {
567            visitor(events);
568        }
569    }
570
571    /// Saves every dirty domain's bars and returns the number written.
572    pub async fn save(&self, worlds: &WorldMap) -> io::Result<usize> {
573        let _save_guard = self.save_lock.lock().await;
574        let mut saved = 0;
575        for (domain, events) in &self.domains {
576            let Some(snapshot) = events.pending_save() else {
577                continue;
578            };
579            let world = domain_default_world(worlds, domain)?;
580            world
581                .saved_data
582                .save(saved_data_names::CUSTOM_BOSS_EVENTS, &snapshot.state)
583                .await
584                .map_err(|error| boss_bar_io_error(domain, error))?;
585            events.mark_saved(snapshot.revision);
586            saved += 1;
587        }
588        Ok(saved)
589    }
590}
591
592fn domain_default_world(worlds: &WorldMap, domain: &str) -> io::Result<Arc<World>> {
593    worlds.default_world(domain).ok_or_else(|| {
594        io::Error::new(
595            io::ErrorKind::NotFound,
596            format!("domain '{domain}' has no loaded default world"),
597        )
598    })
599}
600
601fn boss_bar_io_error(domain: &str, error: io::Error) -> io::Error {
602    io::Error::new(
603        error.kind(),
604        format!("boss bar I/O failed for domain '{domain}': {error}"),
605    )
606}
