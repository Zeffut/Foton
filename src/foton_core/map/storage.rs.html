1//! Where a domain's filled maps live and how they reach disk.
2//!
3//! Vanilla keeps every map in the server's single `SavedDataStorage`, one
4//! `data/maps/<id>.dat` per map plus a `data/maps/last_id.dat` counter, so a
5//! map made in one dimension is still readable from another. Foton's saved
6//! data is per world and its names are `&'static str`, so all of a domain's
7//! maps share one `data/maps.bin` written through the domain's default world --
8//! the same boundary the scoreboard and the command storage already use.
9

31/// A map behind its own lock, shared by every holder being ticked.
32pub type SharedMapData = Arc<SyncMutex<MapItemSavedData>>;
33
34/// Every filled map of one Foton domain.
35#[derive(Debug, Default)]
36pub struct MapStorage {
37    maps: SyncRwLock<FxHashMap<i32, SharedMapData>>,
38    /// Vanilla parity: `MapIndex.lastMapId`, which starts one below the first
39    /// id it will hand out.
40    last_id: AtomicI32,
41    /// Set when the id counter moves or a map is added; individual maps carry
42    /// their own dirty flag.
43    layout_dirty: SyncRwLock<bool>,
44}
45
46impl MapStorage {
47    /// Creates empty map storage for a domain that has never made one.
48    #[must_use]
49    pub fn new() -> Self {
50        Self {
51            maps: SyncRwLock::new(FxHashMap::default()),
52            last_id: AtomicI32::new(-1),
53            layout_dirty: SyncRwLock::new(false),
54        }
55    }
56
57    /// Vanilla parity: `ServerLevel.getMapData`.
58    #[must_use]
59    pub fn get(&self, id: MapId) -> Option<SharedMapData> {
60        self.maps.read().get(&id.id()).map(Arc::clone)
61    }
62
63    /// Vanilla parity: `ServerLevel.setMapData`.
64    pub fn set(&self, id: MapId, data: MapItemSavedData) -> SharedMapData {
65        let shared = Arc::new(SyncMutex::new(data));
66        self.maps.write().insert(id.id(), Arc::clone(&shared));
67        *self.layout_dirty.write() = true;
68        shared
69    }
70
71    /// Vanilla parity: `ServerLevel.getFreeMapId` via `MapIndex.getNextMapId`.
72    pub fn next_id(&self) -> MapId {
73        let id = self.last_id.fetch_add(1, Ordering::AcqRel) + 1;
74        *self.layout_dirty.write() = true;
75        MapId::new(id)
76    }
77
78    fn pending_save(&self) -> bool {
79        if *self.layout_dirty.read() {
80            return true;
81        }
82        self.maps.read().values().any(|map| map.lock().is_dirty())
83    }
84
85    fn snapshot(&self) -> PersistentMaps {
86        let maps = self.maps.read();
87        let mut entries: Vec<PersistentMap> = maps
88            .iter()
89            .map(|(id, map)| persist_map(*id, &map.lock()))
90            .collect();
91        entries.sort_unstable_by_key(|entry| entry.id);
92        PersistentMaps {
93            last_id: self.last_id.load(Ordering::Acquire),
94            maps: entries,
95        }
96    }
97
98    fn mark_saved(&self) {
99        *self.layout_dirty.write() = false;
100        for map in self.maps.read().values() {
101            map.lock().mark_saved();
102        }
103    }
104
105    /// Encodes and decodes this storage the way a restart would.
106    ///
107    /// # Panics
108    /// Panics if the snapshot does not survive its own encoder, which is the
109    /// point of the test that calls this.
110    #[cfg(test)]
111    #[must_use]
112    pub fn round_trip_for_tests(&self) -> Self {
113        let bytes = wincode::serialize(&self.snapshot()).expect("map snapshot should encode");
114        let persistent = wincode::deserialize_exact(&bytes).expect("map snapshot should decode");
115        Self::from_persistent(persistent)
116    }
117
118    fn from_persistent(persistent: PersistentMaps) -> Self {
119        let mut maps = FxHashMap::default();
120        for entry in persistent.maps {
121            let id = entry.id;
122            maps.insert(id, Arc::new(SyncMutex::new(restore_map(entry))));
123        }
124        Self {
125            maps: SyncRwLock::new(maps),
126            last_id: AtomicI32::new(persistent.last_id),
127            layout_dirty: SyncRwLock::new(false),
128        }
129    }
130}

132/// One `MapStorage` per Foton domain, mirroring `DomainScoreboards`.
133#[derive(Debug)]
134pub struct DomainMapData {
135    domains: BTreeMap<String, Arc<MapStorage>>,
136    save_lock: AsyncMutex<()>,
137}
138
139impl DomainMapData {
140    /// Reads every domain's maps through its default world.
141    pub fn load(worlds: &WorldMap) -> io::Result<Self> {
142        let mut domain_names = worlds.domain_names().collect::<Vec<_>>();
143        domain_names.sort_unstable();
144        let mut domains = BTreeMap::new();
145        for domain in domain_names {
146            let world = domain_default_world(worlds, domain)?;
147            let persistent: Option<PersistentMaps> = world
148                .saved_data
149                .sync_load_wincode(saved_data_names::MAPS)
150                .map_err(|error| map_io_error(domain, error))?;
151            let storage = persistent.map_or_else(MapStorage::new, MapStorage::from_persistent);
152            domains.insert(domain.to_owned(), Arc::new(storage));
153        }
154        Ok(Self {
155            domains,
156            save_lock: AsyncMutex::new(()),
157        })
158    }
159
160    /// Returns the map storage a world's domain shares.
161    #[must_use]
162    pub fn for_world(&self, world: &World) -> Option<&Arc<MapStorage>> {
163        self.domains.get(world.domain())
164    }
165
166    /// Returns the map storage of a named domain.
167    #[must_use]
168    pub fn get(&self, domain: &str) -> Option<&Arc<MapStorage>> {
169        self.domains.get(domain)
170    }
171
172    /// Writes every domain whose maps changed, returning how many were written.
173    pub async fn save(&self, worlds: &WorldMap) -> io::Result<usize> {
174        let _save_guard = self.save_lock.lock().await;
175        let mut saved = 0;
176        for (domain, storage) in &self.domains {
177            if !storage.pending_save() {
178                continue;
179            }
180            let world = domain_default_world(worlds, domain)?;
181            let snapshot = storage.snapshot();
182            world
183                .saved_data
184                .sync_save_wincode(saved_data_names::MAPS, &snapshot)
185                .map_err(|error| map_io_error(domain, error))?;
186            storage.mark_saved();
187            saved += 1;
188        }
189        Ok(saved)
190    }
191}
192
193fn domain_default_world(worlds: &WorldMap, domain: &str) -> io::Result<Arc<World>> {
194    worlds.default_world(domain).ok_or_else(|| {
195        io::Error::new(
196            io::ErrorKind::NotFound,
197            format!("domain '{domain}' has no loaded default world"),
198        )
199    })
200}
201
202fn map_io_error(domain: &str, error: io::Error) -> io::Error {
203    io::Error::new(
204        error.kind(),
205        format!("map data for domain '{domain}': {error}"),
206    )
207}

231#[derive(Debug, SchemaWrite, SchemaRead)]
232struct PersistentMapBanner {
233    pos: [i32; 3],
234    /// `DyeColor.serialized_name`, the form vanilla's codec writes.
235    color: String,
236    /// The banner's custom name as network NBT, absent when it has none.
237    name: Option<Vec<u8>>,
238}

241struct PersistentMapFrame {
242    pos: [i32; 3],
243    rotation: i32,
244    entity_id: i32,
245}

247fn persist_map(id: i32, map: &MapItemSavedData) -> PersistentMap {
248    PersistentMap {
249        id,
250        dimension: map.dimension.to_string(),
251        nether: map.nether(),
252        center_x: map.center_x,
253        center_z: map.center_z,
254        scale: map.scale,
255        tracking_position: map.tracking_position(),
256        unlimited_tracking: map.unlimited_tracking(),
257        locked: map.locked,
258        colors: map.colors().to_vec(),
259        banners: map
260            .banners()
261            .map(|banner| PersistentMapBanner {
262                pos: [banner.pos.x(), banner.pos.y(), banner.pos.z()],
263                color: banner.color.serialized_name().to_owned(),
264                name: banner.name.as_ref().map(component_to_nbt_bytes),
265            })
266            .collect(),
267        frames: map
268            .frames()
269            .map(|frame| PersistentMapFrame {
270                pos: [frame.pos.x(), frame.pos.y(), frame.pos.z()],
271                rotation: frame.rotation,
272                entity_id: frame.entity_id,
273            })
274            .collect(),
275    }
276}

278fn restore_map(entry: PersistentMap) -> MapItemSavedData {
279    let dimension = entry
280        .dimension
281        .parse::<Identifier>()
282        .unwrap_or_else(|_| Identifier::vanilla_static("overworld"));
283    let banners = entry
284        .banners
285        .into_iter()
286        .filter_map(|banner| {
287            let color = DyeColor::from_serialized_name(&banner.color)?;
288            Some(MapBanner::new(
289                BlockPos::new(banner.pos[0], banner.pos[1], banner.pos[2]),
290                color,
291                banner.name.as_deref().and_then(component_from_nbt_bytes),
292            ))
293        })
294        .collect();
295    let frames = entry
296        .frames
297        .into_iter()
298        .map(|frame| {
299            MapFrame::new(
300                BlockPos::new(frame.pos[0], frame.pos[1], frame.pos[2]),
301                frame.rotation,
302                frame.entity_id,
303            )
304        })
305        .collect();
306
307    MapItemSavedData::from_persisted(
308        dimension,
309        entry.nether,
310        entry.center_x,
311        entry.center_z,
312        entry.scale,
313        &entry.colors,
314        entry.tracking_position,
315        entry.unlimited_tracking,
316        entry.locked,
317        banners,
318        frames,
319    )
320}

322fn component_to_nbt_bytes(component: &TextComponent) -> Vec<u8> {
323    let mut bytes = Vec::new();
324    component.clone().to_codec_nbt().write(&mut bytes);
325    bytes
326}
327
328fn component_from_nbt_bytes(bytes: &[u8]) -> Option<TextComponent> {
329    let tag = read_tag(&mut Cursor::new(bytes)).ok()?;
330    let owned: NbtTag = tag.as_tag().to_owned();
331    TextComponent::from_nbt(&owned)
