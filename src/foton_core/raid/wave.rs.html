1//! What a raid wave is made of.
2//!
3//! Vanilla parity: `Raid.RaiderType`, the private enum that holds one row per
4//! mob and one column per wave. Everything about the shape of a raid is in
5//! this table: why wave one is four pillagers and nothing else, why the
6//! ravager arrives on wave three, why the witches all turn up at once on wave
7//! four.
8
9use foton_registry::entity_type::EntityTypeRef;
10use foton_registry::vanilla_entities;
11use foton_utils::types::Difficulty;
12use rand::RngExt as _;
13
14/// One kind of mob a raid wave can contain.
15///
16/// Vanilla parity: `Raid.RaiderType`. The declaration order matters: vanilla
17/// iterates `RaiderType.VALUES` and spawns each kind in turn, and the first
18/// mob spawned that can carry a banner becomes the wave's captain.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum RaiderType {
21    /// Vanilla parity: `RaiderType.VINDICATOR`.
22    Vindicator,
23    /// Vanilla parity: `RaiderType.EVOKER`.
24    Evoker,
25    /// Vanilla parity: `RaiderType.PILLAGER`.
26    Pillager,
27    /// Vanilla parity: `RaiderType.WITCH`.
28    Witch,
29    /// Vanilla parity: `RaiderType.RAVAGER`.
30    Ravager,
31}
32
33impl RaiderType {
34    /// Every kind, in vanilla's declaration order.
35    ///
36    /// Vanilla parity: `Raid.RaiderType.VALUES`.
37    pub const VALUES: [Self; 5] = [
38        Self::Vindicator,
39        Self::Evoker,
40        Self::Pillager,
41        Self::Witch,
42        Self::Ravager,
43    ];
44
45    /// Returns the entity this kind spawns.
46    #[must_use]
47    pub const fn entity_type(self) -> EntityTypeRef {
48        match self {
49            Self::Vindicator => &vanilla_entities::VINDICATOR,
50            Self::Evoker => &vanilla_entities::EVOKER,
51            Self::Pillager => &vanilla_entities::PILLAGER,
52            Self::Witch => &vanilla_entities::WITCH,
53            Self::Ravager => &vanilla_entities::RAVAGER,
54        }
55    }
56
57    /// Returns how many of this kind each wave brings, indexed by wave number.
58    ///
59    /// Vanilla parity: the `spawnsPerWaveBeforeBonus` array of each
60    /// `RaiderType`. Index zero is unused -- waves are numbered from one -- and
61    /// index seven is what a hard-difficulty final wave reads.
62    #[must_use]
63    pub const fn spawns_per_wave_before_bonus(self) -> &'static [i32; 8] {
64        match self {
65            Self::Vindicator => &[0, 0, 2, 0, 1, 4, 2, 5],
66            Self::Evoker => &[0, 0, 0, 0, 0, 1, 1, 2],
67            Self::Pillager => &[0, 4, 3, 3, 4, 4, 4, 2],
68            Self::Witch => &[0, 0, 0, 0, 3, 0, 0, 1],
69            Self::Ravager => &[0, 0, 0, 1, 0, 1, 0, 2],
70        }
71    }
72
73    /// Returns how many of this kind a wave brings before the difficulty bonus.
74    ///
75    /// Vanilla parity: `Raid.getDefaultNumSpawns`. A bonus wave reads the row
76    /// at the raid's group count rather than at its own wave number, which is
77    /// what makes it a repeat of the final wave rather than a ninth column.
78    #[must_use]
79    pub fn default_num_spawns(self, wave: i32, num_groups: i32, is_bonus_wave: bool) -> i32 {
80        let index = if is_bonus_wave { num_groups } else { wave };
81        let table = self.spawns_per_wave_before_bonus();
82        usize::try_from(index)
83            .ok()
84            .and_then(|index| table.get(index))
85            .copied()
86            .unwrap_or(0)
87    }
88
89    /// Returns the extra mobs difficulty adds to this kind's share of a wave.
90    ///
91    /// Vanilla parity: `Raid.getPotentialBonusSpawns`. Vanilla rolls
92    /// `nextInt(bonus + 1)`, so the number it names is a ceiling and not a
93    /// guarantee -- except for the easy-difficulty pillager and vindicator,
94    /// whose `nextInt(2)` ceiling is itself rolled and can come out zero twice
95    /// over.
96    #[must_use]
97    pub fn potential_bonus_spawns(
98        self,
99        wave: i32,
100        difficulty: Difficulty,
101        is_bonus_wave: bool,
102        rng: &mut impl rand::Rng,
103    ) -> i32 {
104        let is_easy = difficulty == Difficulty::Easy;
105        let is_normal = difficulty == Difficulty::Normal;
106        let bonus_spawns = match self {
107            Self::Vindicator | Self::Pillager => {
108                if is_easy {
109                    rng.random_range(0..2)
110                } else if is_normal {
111                    1
112                } else {
113                    2
114                }
115            }
116            Self::Evoker => return 0,
117            Self::Witch => {
118                if is_easy || wave <= 2 || wave == 4 {
119                    return 0;
120                }
121                1
122            }
123            Self::Ravager => i32::from(!is_easy && is_bonus_wave),
124        };
125
126        if bonus_spawns > 0 {
127            rng.random_range(0..bonus_spawns + 1)
128        } else {
129            0
130        }
131    }
132}
133
134/// Returns how many waves a raid at this difficulty runs.
135///
136/// Vanilla parity: `Raid.getNumGroups`.
137#[must_use]
138pub const fn num_groups(difficulty: Difficulty) -> i32 {
139    match difficulty {
140        Difficulty::Peaceful => 0,
141        Difficulty::Easy => 3,
142        Difficulty::Normal => 5,
143        Difficulty::Hard => 7,
144    }
145}

147#[cfg(test)]
148mod tests {
149    use super::*;
150
151    /// Vanilla indexes the wave table with the raid's group count on a bonus
152    /// wave and with the wave number otherwise. Reading the wave number in both
153    /// cases is the natural mistake, and on hard difficulty it would index one
154    /// past the final wave and spawn nothing at all.
155    #[test]
156    fn a_bonus_wave_repeats_the_final_wave_rather_than_reading_past_it() {
157        assert_eq!(
158            RaiderType::Vindicator.default_num_spawns(8, 7, true),
159            RaiderType::Vindicator.default_num_spawns(7, 7, false)
160        );
161        assert_eq!(RaiderType::Vindicator.default_num_spawns(8, 7, true), 5);
162    }
163
164    /// Wave one is four pillagers and nothing else; that is the shape a player
165    /// recognizes as the start of a raid.
166    #[test]
167    fn the_first_wave_is_pillagers_alone() {
168        for raider_type in RaiderType::VALUES {
169            let expected = i32::from(raider_type == RaiderType::Pillager) * 4;
170            assert_eq!(raider_type.default_num_spawns(1, 5, false), expected);
171        }
172    }
173
174    /// The witch's bonus is the only one with a wave-number condition, and it
175    /// reads as a list of the waves it *does* apply to unless read carefully.
176    #[test]
177    fn the_witch_takes_no_bonus_before_wave_three_or_on_wave_four() {
178        let mut rng = rand::rng();
179        for wave in [1, 2, 4] {
180            assert_eq!(
181                RaiderType::Witch.potential_bonus_spawns(wave, Difficulty::Hard, false, &mut rng),
182                0
183            );
184        }
185        // Waves three, five and up roll `nextInt(2)`, so the ceiling is one.
186        for wave in [3, 5, 6, 7] {
187            assert!(
188                RaiderType::Witch.potential_bonus_spawns(wave, Difficulty::Hard, false, &mut rng)
189                    <= 1
190            );
191        }
192    }
193
194    /// A ravager bonus only exists on a bonus wave above easy, and confusing
195    /// that with "any wave" would double the ravagers of a normal raid.
196    #[test]
197    fn the_ravager_takes_a_bonus_only_on_a_bonus_wave() {
198        let mut rng = rand::rng();
199        assert_eq!(
200            RaiderType::Ravager.potential_bonus_spawns(5, Difficulty::Hard, false, &mut rng),
201            0
202        );
203        assert_eq!(
204            RaiderType::Ravager.potential_bonus_spawns(5, Difficulty::Easy, true, &mut rng),
205            0
206        );
207        assert!(
208            RaiderType::Ravager.potential_bonus_spawns(5, Difficulty::Hard, true, &mut rng) <= 1
209        );
210    }
211
212    #[test]
213    fn peaceful_runs_no_waves_at_all() {
214        assert_eq!(num_groups(Difficulty::Peaceful), 0);
215        assert_eq!(num_groups(Difficulty::Easy), 3);
216        assert_eq!(num_groups(Difficulty::Normal), 5);
217        assert_eq!(num_groups(Difficulty::Hard), 7);
218    }
