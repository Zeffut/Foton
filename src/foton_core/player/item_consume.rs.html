1//! The moment an eaten or drunk item is about to take effect.
2
3use foton_registry::item_stack::ItemStack;
4use foton_utils::types::InteractionHand;
5
6use crate::entity::LivingEntity as _;
7use crate::event::{Event as _, PlayerItemConsumeEvent};
8use crate::player::Player;
9
10impl Player {
11    /// Asks plugins about the item this player is about to finish using.
12    ///
13    /// Paper parity: the `PlayerItemConsumeEvent` in `completeUsingItem`.
14    /// Returns what to consume and, when a listener chose one, what the hand
15    /// holds afterwards. `None` means a listener cancelled: the item is back
16    /// in the hand, the use is stopped, and the client is resynchronized.
17    ///
18    /// The caller has taken `item` out of the hand; it goes back for the
19    /// length of the event, because a listener reading the hand must find it
20    /// there.
21    pub(crate) fn fire_item_consume(
22        &self,
23        hand: InteractionHand,
24        item: ItemStack,
25    ) -> Option<(ItemStack, Option<ItemStack>)> {
26        self.set_item_in_hand(hand, item.clone());
27        let mut event = PlayerItemConsumeEvent::new(self.gameprofile.id, hand, item.clone());
28        self.fire_event(&mut event);
29        let held = self.take_item_in_hand(hand);
30        if event.is_cancelled() {
31            self.set_item_in_hand(hand, held);
32            self.stop_using_item();
33            self.send_inventory_to_remote();
34            return None;
35        }
36        // Paper consumes the listener's stack only when it differs from the
37        // one offered; otherwise the live stack, which a listener may have
38        // touched through the inventory, is what gets eaten.
39        let consumed = if event.item() == &item {
40            held
41        } else {
42            event.item().clone()
43        };
44        Some((consumed, event.replacement().cloned()))
45    }
46}
