1//! `PlayerInteractEvent`, raised where `CraftBukkit` raises it.
2
3use super::{BlockPos, ClipBlockShape, ClipFluid, Entity as _, InteractionHand, ItemStack, Player};
4use crate::event::{InteractAction, InteractTarget, PlayerInteractEvent, UseResult};
5
6/// The block interaction a use-item packet may repeat.
7///
8/// One right click on a block can reach the server twice: as use-item-on,
9/// and -- when the block did not take the action -- as use-item. Bukkit
10/// raises the event once for the click and lets the second packet reuse its
11/// decision. Paper parity: `ServerPlayerGameMode.firedInteract`.
12pub struct InteractMemo {
13    pos: BlockPos,
14    hand: InteractionHand,
15    item: ItemStack,
16    use_item: UseResult,
17}
18
19impl Player {
20    /// Raises the event for this hand and what it holds.
21    pub(super) fn fire_interact(
22        &self,
23        action: InteractAction,
24        hand: InteractionHand,
25        target: Option<InteractTarget>,
26    ) -> PlayerInteractEvent {
27        let item = self.inventory.lock().get_item_in_hand(hand).clone();
28        let mut event = PlayerInteractEvent::new(
29            self.gameprofile.id,
30            self.get_world().key.to_string(),
31            action,
32            (hand, item),
33            target,
34        );
35        self.fire_event(&mut event);
36        event
37    }
38
39    /// Remembers a right click on a block for the use-item packet that may
40    /// follow it.
41    pub(super) fn remember_interact(&self, event: &PlayerInteractEvent) {
42        let Some(target) = event.target() else {
43            return;
44        };
45        *self.interact_memo.lock() = Some(InteractMemo {
46            pos: target.pos,
47            hand: event.hand(),
48            item: event.item().clone(),
49            use_item: event.use_item(),
50        });
51    }
52
53    /// Whether a use-item packet may go ahead, raising the event for it.
54    ///
55    /// `CraftBukkit` traces the player's reach: with no block in it the use is
56    /// `RIGHT_CLICK_AIR`; with one, it is the click on that block, answered
57    /// from the remembered decision when this packet is that click's second
58    /// half and raised afresh otherwise.
59    pub(super) fn allow_use_item(&self, hand: InteractionHand) -> bool {
60        let memo = self.interact_memo.lock().take();
61        let Some(hit) = self.block_in_reach() else {
62            let event = self.fire_interact(InteractAction::RightClickAir, hand, None);
63            return event.use_item() != UseResult::Deny;
64        };
65        let held = self.inventory.lock().get_item_in_hand(hand).clone();
66        if let Some(memo) = memo
67            && memo.pos == hit.pos
68            && memo.hand == hand
69            && ItemStack::is_same_item_same_components(&memo.item, &held)
70        {
71            return memo.use_item != UseResult::Deny;
72        }
73        let event = self.fire_interact(InteractAction::RightClickBlock, hand, Some(hit));
74        event.use_item() != UseResult::Deny
75    }
76
77    /// The block the player's crosshair rests on within their reach.
78    pub(super) fn block_in_reach(&self) -> Option<InteractTarget> {
79        let eye = self.eye_position();
80        let end = eye + self.look_angle() * self.block_interaction_range();
81        let hit = self
82            .get_world()
83            .clip(eye, end, ClipBlockShape::Outline, ClipFluid::None);
84        if hit.is_miss() {
85            return None;
86        }
87        Some(InteractTarget {
88            pos: hit.block_pos,
89            face: hit.direction,
90            location: Some(hit.location),
91        })
92    }
