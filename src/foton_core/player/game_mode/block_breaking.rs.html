36/// How much each level of haste or conduit power adds.
37///
38/// Vanilla parity: the `(amplification + 1) * 0.2F` of
39/// `Player.getDestroySpeed`.
40const DIG_SPEED_PER_LEVEL: f32 = 0.2;
41
42/// How much slower mining is with both feet off the ground.
43///
44/// Vanilla parity: the `speed /= 5.0F` of `Player.getDestroySpeed`.
45const AIRBORNE_MINING_PENALTY: f32 = 5.0;

47impl Player {
48    /// Mirrors vanilla `Player.blockActionRestricted` for block breaking.
49    pub(super) fn block_action_restricted(&self, world: &World, pos: BlockPos) -> bool {
50        let game_mode = self.game_mode();
51        if !matches!(game_mode, GameType::Adventure | GameType::Spectator) {
52            return false;
53        }
54        if game_mode == GameType::Spectator {
55            return true;
56        }
57        if self.abilities.lock().may_build {
58            return false;
59        }
60
61        // TODO: Retain Vanilla's mutable per-component AdventureModePredicate
62        // cache once Foton's item components support that identity. Until then,
63        // snapshotting safely releases the inventory lock but reevaluates each use.
64        let can_break = {
65            let inventory = self.inventory.lock();
66            let item = inventory.get_selected_item();
67            if item.is_empty() {
68                return true;
69            }
70            item.get(CAN_BREAK).cloned()
71        };
72        let Some(can_break) = can_break else {
73            return true;
74        };
75        !Self::can_break_block_in_adventure_mode(&can_break, world, pos)
76    }
77
78    fn can_break_block_in_adventure_mode(
79        predicate: &AdventureModePredicate,
80        world: &World,
81        pos: BlockPos,
82    ) -> bool {
83        let state = world.get_block_state(pos);
84        // Vanilla's BlockInWorld overload intentionally does not test the
85        // predicate's block-entity component matchers.
86        predicate.predicates().iter().any(|predicate| {
87            if !predicate.matches_state(state) {
88                return false;
89            }
90            let Some(expected_nbt) = predicate.nbt() else {
91                return true;
92            };
93            let Some(block_entity) = world.get_block_entity(pos) else {
94                return false;
95            };
96            let actual_nbt = block_entity.save_with_full_metadata();
97            compare_nbt_compounds(expected_nbt.tag(), &actual_nbt, true)
98        })
99    }
100}

102/// Manages the block breaking state for a player.
103///
104/// Based on Java's `ServerPlayerGameMode` fields and logic.
105pub struct BlockBreakingManager {
106    /// Whether the player is currently breaking a block.
107    is_destroying_block: bool,
108    /// The tick when destruction started.
109    destroy_progress_start: u64,
110    /// The position of the block being destroyed.
111    destroy_pos: BlockPos,
112    /// The current game tick counter.
113    game_ticks: u64,
114    /// Whether there's a delayed destroy pending (for slow mining).
115    has_delayed_destroy: bool,
116    /// Position of the delayed destroy.
117    delayed_destroy_pos: BlockPos,
118    /// The tick when delayed destroy started.
119    delayed_tick_start: u64,
120    /// The last sent destruction progress state (0-9, or -1 for none).
121    last_sent_state: i32,
122}
123
124impl Default for BlockBreakingManager {
125    fn default() -> Self {
126        Self::new()
127    }
128}
129
130impl BlockBreakingManager {
131    /// Creates a new block breaking manager.
132    #[must_use]
133    pub const fn new() -> Self {
134        Self {
135            is_destroying_block: false,
136            destroy_progress_start: 0,
137            destroy_pos: BlockPos::new(0, 0, 0),
138            game_ticks: 0,
139            has_delayed_destroy: false,
140            delayed_destroy_pos: BlockPos::new(0, 0, 0),
141            delayed_tick_start: 0,
142            last_sent_state: -1,
143        }
144    }
145
146    /// Ticks the block breaking manager.
147    ///
148    /// This handles delayed destruction and updates break progress.
149    pub fn tick(&mut self, player: &Player, world: &Arc<World>) {
150        self.game_ticks += 1;
151
152        if self.has_delayed_destroy {
153            let state = world.get_block_state(self.delayed_destroy_pos);
154            if is_air(state) {
155                self.has_delayed_destroy = false;
156            } else {
157                let progress = self.increment_destroy_progress(
158                    player,
159                    world,
160                    state,
161                    self.delayed_destroy_pos,
162                    self.delayed_tick_start,
163                );
164                if progress >= 1.0 {
165                    self.has_delayed_destroy = false;
166                    self.destroy_block(player, world, self.delayed_destroy_pos);
167                }
168            }
169        } else if self.is_destroying_block {
170            let state = world.get_block_state(self.destroy_pos);
171            if is_air(state) {
172                // Block was broken by something else
173                world.broadcast_block_destruction(player.id(), self.destroy_pos, -1);
174                self.last_sent_state = -1;
175                self.is_destroying_block = false;
176            } else {
177                self.increment_destroy_progress(
178                    player,
179                    world,
180                    state,
181                    self.destroy_pos,
182                    self.destroy_progress_start,
183                );
184            }
185        }
186    }
187
188    /// Calculates and updates destruction progress, broadcasting to clients.
189    fn increment_destroy_progress(
190        &mut self,
191        player: &Player,
192        world: &Arc<World>,
193        block_state: BlockStateId,
194        pos: BlockPos,
195        destroy_start_tick: u64,
196    ) -> f32 {
197        let ticks_spent = self.game_ticks.saturating_sub(destroy_start_tick);
198        let destroy_speed = get_destroy_progress(player, block_state);
199        let progress = destroy_speed * (ticks_spent + 1) as f32;
200        let state = (progress * 10.0) as i32;
201
202        if state != self.last_sent_state {
203            world.broadcast_block_destruction(player.id(), pos, state);
204            self.last_sent_state = state;
205        }
206
207        progress
208    }

210    /// Handles a block break action from the client.
211    ///
212    /// Note: The caller (packet handler) is responsible for calling `ack_block_changes_up_to`
213    /// after this method returns, matching vanilla behavior.
214    pub fn handle_block_break_action(
215        &mut self,
216        player: &Player,
217        world: &Arc<World>,
218        pos: BlockPos,
219        action: BlockBreakAction,
220        _direction: Direction,
221    ) {
222        // Validate interaction range
223        if !player.is_within_block_interaction_range(pos) {
224            return;
225        }
226
227        // Validate Y coordinate
228        if pos.y() >= world.max_build_height() {
229            player.send_packet(CBlockUpdate {
230                pos,
231                block_state: world.get_block_state(pos),
232            });
233            return;
234        }
235
236        match action {
237            BlockBreakAction::Start => {
238                // Check may_interact permission
239                if !world.may_interact(player, pos) {
240                    player.send_packet(CBlockUpdate {
241                        pos,
242                        block_state: world.get_block_state(pos),
243                    });
244                    return;
245                }
246                // Restrictions are checked before firing the event. Apart from
247                // matching vanilla ordering, this keeps detached test players
248                // from requiring a live server for an action that is rejected.
249                if player.block_action_restricted(world, pos) {
250                    player.send_packet(CBlockUpdate {
251                        pos,
252                        block_state: world.get_block_state(pos),
253                    });
254                    return;
255                }
256                let mut event = BlockDamageEvent::new(player.uuid(), world.key.to_string(), pos);
257                if let Some(server) = player.server.upgrade() {
258                    server.events.fire(&mut event);
259                }
260                if event.is_cancelled() {
261                    return;
262                }
263
264                // Creative mode: instant break
265                if player.game_mode() == GameType::Creative {
266                    self.destroy_and_ack(player, world, pos);
267                    return;
268                }
269
270                self.destroy_progress_start = self.game_ticks;
271                let block_state = world.get_block_state(pos);
272
273                if !is_air(block_state) {
274                    // TODO: Call EnchantmentHelper.onHitBlock before blockState.attack.
275                    BLOCK_BEHAVIORS
276                        .get_behavior(block_state.get_block())
277                        .attack(block_state, world, pos, player);
278
279                    let progress = get_destroy_progress(player, block_state);
280
281                    if progress >= 1.0 {
282                        // Insta-mine
283                        self.destroy_and_ack(player, world, pos);
284                    } else {
285                        // Start breaking
286                        if self.is_destroying_block {
287                            // Send block update for the old position to cancel client prediction
288                            player.send_packet(CBlockUpdate {
289                                pos: self.destroy_pos,
290                                block_state: world.get_block_state(self.destroy_pos),
291                            });
292                        }
293
294                        self.is_destroying_block = true;
295                        self.destroy_pos = pos;
296                        let state = (progress * 10.0) as i32;
297                        world.broadcast_block_destruction(player.id(), pos, state);
298                        self.last_sent_state = state;
299                    }
300                }
301            }
302
303            BlockBreakAction::Stop => {
304                if pos == self.destroy_pos {
305                    let ticks_spent = self.game_ticks.saturating_sub(self.destroy_progress_start);
306                    let block_state = world.get_block_state(pos);
307
308                    if !is_air(block_state) {
309                        let destroy_speed = get_destroy_progress(player, block_state);
310                        let progress = destroy_speed * (ticks_spent + 1) as f32;
311
312                        if progress >= 0.7 {
313                            // Complete the break
314                            self.is_destroying_block = false;
315                            world.broadcast_block_destruction(player.id(), pos, -1);
316                            self.destroy_and_ack(player, world, pos);
317                            return;
318                        }
319
320                        if !self.has_delayed_destroy {
321                            // Set up delayed destroy
322                            self.is_destroying_block = false;
323                            self.has_delayed_destroy = true;
324                            self.delayed_destroy_pos = pos;
325                            self.delayed_tick_start = self.destroy_progress_start;
326                        }
327                    }
328                }
329            }
330
331            BlockBreakAction::Abort => {
332                self.is_destroying_block = false;
333
334                if self.destroy_pos != pos {
335                    log::warn!(
336                        "Mismatch in destroy block pos: {:?} vs {:?}",
337                        self.destroy_pos,
338                        pos
339                    );
340                    world.broadcast_block_destruction(player.id(), self.destroy_pos, -1);
341                }
342
343                world.broadcast_block_destruction(player.id(), pos, -1);
344            }
345        }
346    }

348    /// Destroys a block and sends appropriate response.
349    fn destroy_and_ack(&mut self, player: &Player, world: &Arc<World>, pos: BlockPos) {
350        if !self.destroy_block(player, world, pos) {
351            // Send block update to resync client
352            player.send_packet(CBlockUpdate {
353                pos,
354                block_state: world.get_block_state(pos),
355            });
356        }
357    }

359    /// Destroys a block at the given position.
360    ///
361    /// Returns true if the block was successfully destroyed.
362    #[expect(
363        clippy::unused_self,
364        reason = "method belongs logically to BlockBreakingManager and will use self when additional state is added"
365    )]
366    fn destroy_block(&self, player: &Player, world: &Arc<World>, pos: BlockPos) -> bool {
367        let state = world.get_block_state(pos);
368
369        if !item_can_destroy_block(player, world, pos, state) {
370            return false;
371        }
372
373        // Get block info
374        let Some(_block) = REGISTRY.blocks.by_state_id(state) else {
375            return false;
376        };
377
378        // Before anything touches the block, so cancelling leaves the world as
379        // it was rather than having to put it back.
380        let event_drop_items = if let Some(shared) = player.shared() {
381            let mut event = BlockBreakEvent::new(shared, pos, state);
382            player.fire_event(&mut event);
383            if event.is_cancelled() {
384                return false;
385            }
386            event.drop_items()
387        } else {
388            true
389        };
390
391        // TODO: Check for GameMasterBlock (command blocks, etc.)
392        // TODO: Check blockActionRestricted
393
394        let behavior = BLOCK_BEHAVIORS.get_behavior(state.get_block());
395        // Vanilla parity: `ServerPlayerGameMode.destroyBlock` reads the block
396        // entity before it removes the block, then hands that one to the loot
397        // roll and to `playerDestroy`. Looking it up afterwards finds nothing:
398        // removing the block detaches it.
399        let block_entity = world.get_block_entity(pos);
400        let adjusted_state = behavior.player_will_destroy(state, world, pos, player);
401        world.game_event(
402            &vanilla_game_events::BLOCK_DESTROY,
403            pos,
404            &GameEventContext::new(Some(player), Some(adjusted_state)),
405        );
406        let state_after_player_will_destroy = world.get_block_state(pos);
407
408        // Vanilla parity: fluidState.createLegacyBlock() — breaking a waterlogged
409        // block leaves water behind instead of air.
410        let replacement = fluid_state_to_block(state.get_fluid_state());
411        // Vanilla removes the live state after `playerWillDestroy`; tripwire uses
412        // that callback to set DISARMED before the same block is removed.
413        let removed_by_player_break = !state_after_player_will_destroy.is_air()
414            && world.set_block_if_unchanged(
415                pos,
416                state_after_player_will_destroy,
417                replacement,
418                UpdateFlags::UPDATE_ALL,
419            ) == ConditionalBlockSetResult::Changed;
420        let changed_by_player_will_destroy = state_after_player_will_destroy != state;
421        let changed = changed_by_player_will_destroy || removed_by_player_break;
422
423        if removed_by_player_break {
424            behavior.destroy(adjusted_state, world, pos);
425
426            // Play block destruction particles and sound (skip for fire blocks like vanilla)
427            // Exclude the breaking player as they see the effect client-side
428            let block = REGISTRY.blocks.by_state_id(adjusted_state);
429            let is_fire = block.is_some_and(|b| {
430                b.key == vanilla_blocks::FIRE.key || b.key == vanilla_blocks::SOUL_FIRE.key
431            });
432            if !is_fire {
433                world.destroy_block_effect(pos, u32::from(adjusted_state.0), Some(player.id()));
434            }
435
436            // Vanilla snapshots the tool before Item.mineBlock can damage or
437            // consume it, then uses that snapshot for loot and post-break effects.
438            let (has_correct_tool, destroyed_with) = {
439                let inv = player.inventory.lock();
440                let main_hand = inv.get_item_in_hand(InteractionHand::MainHand);
441                (
442                    main_hand.is_correct_tool_for_drops(adjusted_state)
443                        || !requires_correct_tool(adjusted_state),
444                    main_hand.copy_with_count(main_hand.count),
445                )
446            };
447
448            // Vanilla parity: `ItemStack.mineBlock`, dispatched to the item so the
449            // ones with their own durability rule get it -- shears pay for a
450            // zero-hardness plant and never pay for fire. Runs before
451            // `playerDestroy`, as vanilla does.
452            //
453            // The tool can die here, and vanilla's `hurtAndBreak` announces
454            // that itself. Foton's `ItemStack` cannot reach the miner, so the
455            // break is read off the slot: the pickaxe was there before the
456            // durability hit and is gone after.
457            let tool_broke = {
458                let mut inv = player.inventory.lock();
459                let item_behavior = ITEM_BEHAVIORS.get_behavior(inv.get_selected_item().item());
460                // with_selected_item_mut so the slot is marked changed.
461                inv.with_selected_item_mut(|main_hand| {
462                    let had_a_tool = !main_hand.is_empty();
463                    item_behavior.mine_block(main_hand, adjusted_state, player);
464                    had_a_tool && main_hand.is_empty()
465                })
466            };
467            if tool_broke {
468                LivingEntity::on_equipped_item_broken(player, EquipmentSlot::MainHand);
469            }
470
471            player.cause_food_exhaustion(food_constants::EXHAUSTION_MINE);
472
473            // Handle drops (skip for creative/spectator)
474            let game_mode = player.game_mode();
475            if event_drop_items
476                && game_mode != GameType::Spectator
477                && game_mode != GameType::Creative
478                && has_correct_tool
479            {
480                drop_block_loot(
481                    player,
482                    world,
483                    pos,
484                    adjusted_state,
485                    &destroyed_with,
486                    block_entity.as_ref(),
487                );
488                // Vanilla parity: the `awardStat(Stats.BLOCK_MINED.get(this))`
489                // that opens `Block.playerDestroy`, which Foton reaches through
490                // the same non-creative, correct-tool gate this branch is under.
491                player.award_stat(Stat::new(
492                    &vanilla_stat_types::MINED,
493                    adjusted_state.get_block(),
494                ));
495                behavior.player_destroy(
496                    world,
497                    player,
498                    pos,
499                    adjusted_state,
500                    block_entity.as_ref(),
501                    &destroyed_with,
502                );
503            }
504        }
505
506        changed
507    }

511#[derive(Debug, Clone, Copy, PartialEq, Eq)]

514    Start,
515    /// Player stopped breaking a block (finished or released).

521/// Checks if a block state is air.
522fn is_air(state: BlockStateId) -> bool {
523    let Some(block) = REGISTRY.blocks.by_state_id(state) else {
524        return true;
525    };
526    block.config.is_air
527}

529/// Checks if a block requires the correct tool for drops.
530fn requires_correct_tool(state: BlockStateId) -> bool {
531    let Some(block) = REGISTRY.blocks.by_state_id(state) else {
532        return false;
533    };
534    block.config.requires_correct_tool_for_drops
535}

537/// Gets the destroy progress per tick for a block.
538///
539/// This is based on the vanilla formula:
540/// `1.0 / (destroy_time * 30.0)` for survival with correct tool
541/// `1.0 / (destroy_time * 100.0)` for survival with wrong tool
542/// Instant break for creative mode.
543fn get_destroy_progress(player: &Player, block_state: BlockStateId) -> f32 {
544    let Some(block) = REGISTRY.blocks.by_state_id(block_state) else {
545        return 0.0;
546    };
547
548    let destroy_time = block.config.destroy_time;
549
550    // Instant break for creative
551    if player.game_mode() == GameType::Creative {
552        return 1.0;
553    }
554
555    // Unbreakable block
556    if destroy_time < 0.0 {
557        return 0.0;
558    }
559
560    // Instant break for destroy_time == 0
561    if destroy_time == 0.0 {
562        return 1.0;
563    }
564
565    // Get player's mining speed
566    let mining_speed = {
567        let inv = player.inventory.lock();
568        let main_hand = inv.get_item_in_hand(InteractionHand::MainHand);
569        main_hand.get_destroy_speed(block_state)
570    };
571
572    // Check if player has the correct tool
573    let has_correct_tool = {
574        let inv = player.inventory.lock();
575        let main_hand = inv.get_item_in_hand(InteractionHand::MainHand);
576        main_hand.is_correct_tool_for_drops(block_state)
577    };
578
579    let speed = apply_mining_speed_modifiers(player, mining_speed);
580
581    // Calculate destroy progress per tick
582    // Vanilla formula: speed / hardness / (hasCorrectTool ? 30 : 100)
583    let divisor = if has_correct_tool || !block.config.requires_correct_tool_for_drops {
584        30.0
585    } else {
586        100.0
587    };
588
589    speed / destroy_time / divisor
590}

592/// Scales a tool's raw speed by everything the digger's own state does to it.
593///
594/// Vanilla parity: the body of `Player.getDestroySpeed` after the tool has been
595/// asked, in vanilla's order -- the `MINING_EFFICIENCY` term is added before
596/// haste and mining fatigue scale the result, and `BLOCK_BREAK_SPEED` and
597/// `SUBMERGED_MINING_SPEED` multiply after them.
598#[must_use]
599pub(crate) fn apply_mining_speed_modifiers(player: &Player, tool_speed: f32) -> f32 {
600    let mut speed = tool_speed;
601
602    // Vanilla parity: `if (speed > 1.0F) speed += MINING_EFFICIENCY`. That
603    // attribute is where Efficiency lands, and the `> 1.0` guard is why the
604    // enchantment does nothing for a block the tool is not the tool for.
605    if speed > 1.0 {
606        speed += player
607            .attributes()
608            .lock()
609            .required_value(vanilla_attributes::MINING_EFFICIENCY) as f32;
610    }
611
612    // Vanilla parity: `MobEffectUtil.hasDigSpeed` and
613    // `getDigSpeedAmplification`, which take the better of haste and conduit
614    // power rather than stacking them.
615    if let Some(amplification) = dig_speed_amplification(player) {
616        speed *= 1.0 + (amplification + 1) as f32 * DIG_SPEED_PER_LEVEL;
617    }
618
619    // Vanilla parity: the `switch` of `Player.getDestroySpeed`. The steps are
620    // steeper than a multiplier would be, which is why an elder guardian's
621    // curse is worth swimming away from rather than working through.
622    if let Some(fatigue) = player.mob_effect(vanilla_mob_effects::MINING_FATIGUE) {
623        speed *= match fatigue.amplifier() {
624            0 => 0.3,
625            1 => 0.09,
626            2 => 0.0027,
627            _ => 0.000_81,
628        };
629    }
630
631    // Vanilla parity: `speed *= BLOCK_BREAK_SPEED`, the attribute a command or
632    // a plugin scales digging with. Its base value is 1, so this is a no-op
633    // until something modifies it.
634    speed *= player
635        .attributes()
636        .lock()
637        .required_value(vanilla_attributes::BLOCK_BREAK_SPEED) as f32;
638
639    // Vanilla parity: `if (isEyeInFluid(WATER)) speed *= SUBMERGED_MINING_SPEED`.
640    // The base value is 0.2, and Aqua Affinity is the modifier that lifts it
641    // back to 1 -- which is the whole of the enchantment.
642    if player.is_eye_in_water() {
643        speed *= player
644            .attributes()
645            .lock()
646            .required_value(vanilla_attributes::SUBMERGED_MINING_SPEED) as f32;
647    }
648
649    // Vanilla parity: the `if (!this.onGround()) speed /= 5.0F`, which is why
650    // mining while falling or jumping takes five times as long.
651    if !player.on_ground() {
652        speed /= AIRBORNE_MINING_PENALTY;
653    }
654
655    speed
656}

658/// Returns the better of the two effects that speed digging up.
659///
660/// Vanilla parity: `MobEffectUtil.getDigSpeedAmplification`, guarded by
661/// `hasDigSpeed`; `None` here is vanilla's "neither is active".
662fn dig_speed_amplification(player: &Player) -> Option<i32> {
663    let haste = player
664        .mob_effect(vanilla_mob_effects::HASTE)
665        .map(|active| active.amplifier());
666    let conduit = player
667        .mob_effect(vanilla_mob_effects::CONDUIT_POWER)
668        .map(|active| active.amplifier());
669    match (haste, conduit) {
670        (Some(a), Some(b)) => Some(a.max(b)),
671        (Some(a), None) | (None, Some(a)) => Some(a),
672        (None, None) => None,
673    }
674}

676/// Drops loot for a destroyed block using its loot table.
677fn drop_block_loot(
678    player: &Player,
679    world: &Arc<World>,
680    pos: BlockPos,
681    state: BlockStateId,
682    tool: &ItemStack,
683    block_entity: Option<&SharedBlockEntity>,
684) {
685    let luck = player
686        .attributes()
687        .lock()
688        .get_value(vanilla_attributes::LUCK)
689        .unwrap_or(0.0) as f32;
690
691    let drops = BlockLootContext::new(world, pos)
692        .with_luck(luck)
693        .with_tool(tool)
694        .with_block_entity(block_entity)
695        .get_drops(state);
696
697    // Spawn each dropped item using the player's world reference (Arc<World>)
698    for item in drops {
699        if !item.is_empty() {
700            player.get_world().pop_resource(pos, item);
701        }
702    }
703
704    BLOCK_BEHAVIORS
705        .get_behavior(state.get_block())
706        .spawn_after_break(state, world, pos, tool, true);
707}

709/// Runs the held item's veto on breaking `state`.
710///
711/// Vanilla parity: the `ItemStack.canDestroyBlock` guard opening
712/// `ServerPlayerGameMode.destroyBlock`. Vanilla mutates the live stack; Foton
713/// works on a copy so the hook never runs while the inventory lock is held,
714/// and writes it back only when the item actually changed it.
715fn item_can_destroy_block(
716    player: &Player,
717    world: &Arc<World>,
718    pos: BlockPos,
719    state: BlockStateId,
720) -> bool {
721    let mut held = {
722        let inventory = player.inventory.lock();
723        let item = inventory.get_item_in_hand(InteractionHand::MainHand);
724        item.copy_with_count(item.count())
725    };
726    let before = held.clone();
727
728    let allowed = ITEM_BEHAVIORS
729        .get_behavior(held.item())
730        .can_destroy_block(&mut held, state, world, pos, player);
731
732    if held != before {
733        player
734            .inventory
735            .lock()
736            .set_item_in_hand(InteractionHand::MainHand, held);
737    }
738    allowed
739}
