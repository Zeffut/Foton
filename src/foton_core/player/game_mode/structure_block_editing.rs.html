1//! Saving a structure block or a jigsaw block from its editor.
2//!
3//! Vanilla parity: `ServerGamePacketListenerImpl.handleSetStructureBlock` and
4//! `handleSetJigsawBlock`. Both are gated on `Player.canUseGameMasterBlocks`
5//! and both resend the block afterwards so an open editor agrees with the
6//! server.
7//!
8//! The structure block's editor has four buttons, and one of them Foton cannot
9//! honor yet: saving needs `StructureTemplate.fillFromWorld` and a template manager
10//! to write the file, so it reports vanilla's own `structure_block.save_failure`
11//! rather than a success message for something that did not happen. Loading works,
12//! over the structures bundled with the game -- which, until a save can write one,
13//! is all there are.
14

33impl Player {
34    /// Stores what a player typed into a structure block's editor.
35    ///
36    /// Vanilla parity: `handleSetStructureBlock`.
37    pub fn handle_set_structure_block(self: &Arc<Self>, packet: SSetStructureBlock) {
38        if !self.can_use_game_master_blocks() {
39            return;
40        }
41        let Some(update_type) = StructureUpdateType::from_wire(packet.update_type) else {
42            return;
43        };
44        let Some(mode) = structure_mode_from_ordinal(packet.mode) else {
45            return;
46        };
47
48        let world = self.get_world();
49        let pos = packet.pos;
50        let Some(shared) = world.get_block_entity(pos) else {
51            return;
52        };
53        let Some(block_entity) = shared.downcast_ref::<StructureBlockEntity>() else {
54            return;
55        };
56
57        block_entity.set_mode(mode);
58        block_entity.set_structure_name(&packet.name);
59        block_entity.set_offset((
60            i32::from(packet.offset.0),
61            i32::from(packet.offset.1),
62            i32::from(packet.offset.2),
63        ));
64        block_entity.set_size((
65            i32::from(packet.size.0),
66            i32::from(packet.size.1),
67            i32::from(packet.size.2),
68        ));
69        block_entity.set_mirror(StructureMirror::from_ordinal(packet.mirror).unwrap_or_default());
70        block_entity
71            .set_rotation(StructureRotation::from_ordinal(packet.rotation).unwrap_or_default());
72        block_entity.set_metadata(packet.data);
73        block_entity.set_ignore_entities(packet.ignore_entities);
74        block_entity.set_strict(packet.strict);
75        block_entity.set_show_air(packet.show_air);
76        block_entity.set_show_bounding_box(packet.show_bounding_box);
77        block_entity.set_integrity(packet.integrity);
78        block_entity.set_seed(packet.seed);
79
80        if block_entity.has_structure_name() {
81            let name = block_entity.structure_name();
82            self.report_structure_action(update_type, &world, block_entity, &name);
83        } else {
84            self.send_message(&TextComponent::translated(
85                translations::STRUCTURE_BLOCK_INVALID_STRUCTURE_NAME
86                    .message([TextComponent::plain(packet.name)]),
87            ));
88        }
89
90        block_entity.set_changed();
91        send_block_entity_update(self, &world, pos, block_entity);
92    }
93
94    /// Reports what the pressed button did.
95    fn report_structure_action(
96        &self,
97        update_type: StructureUpdateType,
98        world: &Arc<World>,
99        block_entity: &StructureBlockEntity,
100        name: &str,
101    ) {
102        let named = || [TextComponent::plain(name.to_owned())];
103        let message = match update_type {
104            // Vanilla parity: plain "done" says nothing at all.
105            StructureUpdateType::UpdateData => return,
106            // Foton cannot capture a structure yet, so this is always the
107            // failure branch rather than a success message for nothing.
108            StructureUpdateType::SaveArea => {
109                translations::STRUCTURE_BLOCK_SAVE_FAILURE.message(named())
110            }
111            StructureUpdateType::LoadArea => {
112                if !block_entity.is_structure_loadable() {
113                    translations::STRUCTURE_BLOCK_LOAD_NOT_FOUND.message(named())
114                } else if block_entity.place_structure_if_same_size(world) {
115                    translations::STRUCTURE_BLOCK_LOAD_SUCCESS.message(named())
116                } else {
117                    translations::STRUCTURE_BLOCK_LOAD_PREPARE.message(named())
118                }
119            }
120            StructureUpdateType::ScanArea => {
121                if block_entity.detect_size() {
122                    translations::STRUCTURE_BLOCK_SIZE_SUCCESS.message(named())
123                } else {
124                    translations::STRUCTURE_BLOCK_SIZE_FAILURE.msg()
125                }
126            }
127        };
128        self.send_message(&TextComponent::translated(message));
129    }

131    /// Stores what a player typed into a jigsaw block's editor.
132    ///
133    /// Vanilla parity: `handleSetJigsawBlock`.
134    pub fn handle_set_jigsaw_block(self: &Arc<Self>, packet: SSetJigsawBlock) {
135        if !self.can_use_game_master_blocks() {
136            return;
137        }
138
139        let world = self.get_world();
140        let pos = packet.pos;
141        let Some(shared) = world.get_block_entity(pos) else {
142            return;
143        };
144        let Some(block_entity) = shared.downcast_ref::<JigsawBlockEntity>() else {
145            return;
146        };
147
148        // Vanilla parity: `JointType.CODEC.byName(..., ALIGNED)` -- a joint
149        // name the server does not know falls back to aligned, not to the
150        // orientation default.
151        let joint = JigsawJointType::from_name(&packet.joint).unwrap_or(JigsawJointType::Aligned);
152        block_entity.configure(JigsawSettings {
153            name: identifier_or_empty(&packet.name),
154            target: identifier_or_empty(&packet.target),
155            pool: identifier_or_empty(&packet.pool),
156            final_state: packet.final_state,
157            joint,
158            selection_priority: packet.selection_priority,
159            placement_priority: packet.placement_priority,
160        });
161
162        send_block_entity_update(self, &world, pos, block_entity);
163    }
164
165    /// Runs a jigsaw block's generate button.
166    ///
167    /// Vanilla parity: `handleJigsawGenerate`. Vanilla reports nothing back
168    /// either way, so a pool the server does not know simply does nothing.
169    pub fn handle_jigsaw_generate(self: &Arc<Self>, packet: SJigsawGenerate) {
170        if !self.can_use_game_master_blocks() {
171            return;
172        }
173
174        let world = self.get_world();
175        let Some(shared) = world.get_block_entity(packet.pos) else {
176            return;
177        };
178        let Some(block_entity) = shared.downcast_ref::<JigsawBlockEntity>() else {
179            return;
180        };
181
182        block_entity.generate(&world, packet.levels, packet.keep_jigsaws);
183    }
184}

186/// Resends one block entity to everyone tracking it, and to the editor.
187fn send_block_entity_update(
188    player: &Player,
189    world: &Arc<World>,
190    pos: BlockPos,
191    block_entity: &dyn BlockEntity,
192) {
193    let Some(nbt) = block_entity.get_update_tag() else {
194        return;
195    };
196    world.broadcast_block_entity_update(pos, block_entity.get_type(), nbt.clone());
197    player.send_packet(CBlockEntityData {
198        pos,
199        block_entity_type: block_entity.get_type().id() as i32,
200        nbt: OptionalNbt(Some(nbt)),
201    });
202}

204/// Parses an identifier field, falling back to `minecraft:empty`.
205///
206/// Vanilla reads these with `readIdentifier`, which rejects a malformed one at
207/// the codec; Foton keeps the string and falls back here so a bad editor entry
208/// cannot drop the connection.
209fn identifier_or_empty(value: &str) -> Identifier {
210    value
211        .parse::<Identifier>()
212        .unwrap_or_else(|_| Identifier::vanilla_static("empty"))
213}
