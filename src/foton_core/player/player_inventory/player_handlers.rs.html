32fn menu_top_slot_count(menu: &Menu) -> Option<usize> {
33    if menu.container_id() == INVENTORY_MENU_CONTAINER_ID && menu.menu_type().is_none() {
34        // Vanilla InventoryMenu exposes the result plus the 2x2 crafting grid.
35        return Some(5);
36    }
37    menu.menu_type().map(|_| {
38        menu.behavior()
39            .slots()
40            .len()
41            .saturating_sub(PlayerInventory::MAIN.len() + PlayerInventory::HOTBAR.len())
42    })
43}

45fn menu_metadata(menu: &Menu) -> (usize, Option<String>, Vec<ItemStack>) {
46    let top_slot_count = menu_top_slot_count(menu).unwrap_or(0);
47    let menu_type = menu.menu_type().map(|menu_type| menu_type.key.to_string());
48    let snapshot = {
49        let guard = menu.behavior().lock_all_containers();
50        (0..top_slot_count)
51            .filter_map(|index| {
52                menu.behavior()
53                    .slots()
54                    .get(index)
55                    .map(|slot| slot.get_item(&guard).clone())
56            })
57            .collect()
58    };
59    (top_slot_count, menu_type, snapshot)
60}

84    ///
85    /// Vanilla parity: `ServerPlayer.openItemGui`, which does nothing for a
86    /// book that carries no `written_book_content`.
87    ///
88    /// Foton gap: Vanilla first calls `WrittenBookContent.resolveForItem`, which
89    /// expands selectors and scores inside a command-written book and rewrites
90    /// the component as resolved. Foton has no command-source text resolution,
91    /// so a book written by a player -- always already resolved -- opens
92    /// correctly while one written by `/give` shows its pages unexpanded.
93    pub fn open_item_gui(&self, stack: &ItemStack, hand: InteractionHand) {
94        if !stack.has(WRITTEN_BOOK_CONTENT) {
95            return;
96        }
97        self.send_packet(COpenBook { hand });
98    }

100    /// Applies a book-and-quill edit, signing the book when a title came with it.
101    ///
102    /// Vanilla parity: `ServerGamePacketListenerImpl.handleEditBook`.
103    pub fn handle_edit_book(&self, packet: SEditBook) {
104        let Ok(slot) = usize::try_from(packet.slot) else {
105            return;
106        };
107        if !PlayerInventory::is_hotbar_slot(slot) && slot != PlayerInventory::SLOT_OFFHAND {
108            return;
109        }
110
111        // Vanilla runs the pages through the server's `TextFilter` first and
112        // resumes on the main thread. Foton has no filter service, so both
113        // branches of `filterableFromOutgoing` collapse to a pass-through.
114        match packet.title {
115            Some(title) => self.sign_book(&title, packet.pages, slot),
116            None => self.update_book_contents(packet.pages, slot),
117        }
118    }

120    /// Vanilla parity: `ServerGamePacketListenerImpl.updateBookContents`.
121    fn update_book_contents(&self, pages: Vec<String>, slot: usize) {
122        let pages = pages.into_iter().map(Filterable::pass_through).collect();
123        let contents = match WritableBookContent::new(pages) {
124            Ok(contents) => contents,
125            Err(error) => {
126                log::debug!(
127                    "Player {} sent unusable book pages: {error}",
128                    self.gameprofile.name
129                );
130                return;
131            }
132        };
133
134        let mut inventory = self.inventory.lock();
135        let carried = inventory.get_item_mut(slot);
136        if !carried.has(WRITABLE_BOOK_CONTENT) {
137            return;
138        }
139        carried.set(WRITABLE_BOOK_CONTENT, contents);
140    }

142    /// Vanilla parity: `ServerGamePacketListenerImpl.signBook`.
143    fn sign_book(&self, title: &str, pages: Vec<String>, slot: usize) {
144        let pages = pages
145            .into_iter()
146            .map(|page| Filterable::pass_through(TextComponent::plain(page)))
147            .collect();
148        let contents = match WrittenBookContent::new(
149            Filterable::pass_through(title.to_owned()),
150            self.gameprofile.name.clone(),
151            0,
152            pages,
153            true,
154        ) {
155            Ok(contents) => contents,
156            Err(error) => {
157                log::debug!(
158                    "Player {} sent an unusable book signature: {error}",
159                    self.gameprofile.name
160                );
161                return;
162            }
163        };
164
165        let mut inventory = self.inventory.lock();
166        let carried = inventory.get_item(slot);
167        if !carried.has(WRITABLE_BOOK_CONTENT) {
168            return;
169        }
170
171        // Vanilla parity: `ItemStack.transmuteCopy` keeps the count and every
172        // component the quill carried, then swaps the two book components.
173        let mut written = ItemStack::with_count_and_patch(
174            &vanilla_items::WRITTEN_BOOK,
175            carried.count(),
176            carried.components_patch().clone(),
177        );
178        written.remove(WRITABLE_BOOK_CONTENT);
179        written.set(WRITTEN_BOOK_CONTENT, contents);
180        inventory.set_item(slot, written);
181    }

205    fn take_open_menu_for_callback(
206        &self,
207        expected_container_id: Option<i32>,
208    ) -> Result<Menu, OpenMenuUnavailable> {
209        let mut open_menu = self.open_menu.lock();
210        if open_menu.dispatch.is_some() {
211            return Err(OpenMenuUnavailable::Unavailable);
212        }
213
214        let Some(menu) = open_menu.menu.as_ref() else {
215            return Err(OpenMenuUnavailable::Closed);
216        };
217        if expected_container_id.is_some_and(|expected| i32::from(menu.container_id()) != expected)
218        {
219            return Err(OpenMenuUnavailable::Unavailable);
220        }
221
222        let overrides_player_slots = menu.overrides_player_slots();
223        let Some(menu) = open_menu.menu.take() else {
224            return Err(OpenMenuUnavailable::Unavailable);
225        };
226        let top_slot_count = menu_top_slot_count(&menu).unwrap_or(0);
227        let menu_type = menu.menu_type().map(|menu_type| menu_type.key.to_string());
228        let snapshot = {
229            let guard = menu.behavior().lock_all_containers();
230            (0..top_slot_count)
231                .filter_map(|index| {
232                    menu.behavior()
233                        .slots()
234                        .get(index)
235                        .map(|slot| slot.get_item(&guard).clone())
236                })
237                .collect()
238        };
239        open_menu.dispatch = Some(OpenMenuDispatch {
240            overrides_player_slots,
241            top_slot_count,
242            menu_type,
243            snapshot,
244            actions: Vec::new(),
245        });
246        Ok(menu)
247    }

249    fn finish_open_menu_callback(&self, menu: Menu) {
250        let metadata = menu_metadata(&menu);
251        let actions = {
252            let mut open_menu = self.open_menu.lock();
253            let Some(mut dispatch) = open_menu.dispatch.take() else {
254                open_menu.menu = Some(menu);
255                return;
256            };
257            dispatch.top_slot_count = metadata.0;
258            dispatch.menu_type = metadata.1;
259            dispatch.snapshot = metadata.2;
260            if let Some(terminal_removal) = open_menu.terminal_removal.as_mut() {
261                Self::queue_deferred_menus(terminal_removal, dispatch.actions);
262                drop(open_menu);
263                self.finish_terminal_menu_main_cleanup(Some(menu));
264                return;
265            }
266            open_menu.menu = Some(menu);
267            open_menu.active_open_operations += 1;
268            dispatch.actions
269        };
270
271        self.run_deferred_menu_actions(actions);
272    }

274    fn finish_open_menu_removal(&self) {
275        let actions = {
276            let mut open_menu = self.open_menu.lock();
277            let Some(dispatch) = open_menu.dispatch.take() else {
278                return;
279            };
280            if let Some(terminal_removal) = open_menu.terminal_removal.as_mut() {
281                Self::queue_deferred_menus(terminal_removal, dispatch.actions);
282                drop(open_menu);
283                self.finish_terminal_menu_main_cleanup(None);
284                return;
285            }
286            open_menu.active_open_operations += 1;
287            dispatch.actions
288        };
289
290        self.run_deferred_menu_actions(actions);
291    }

293    fn run_deferred_menu_actions(&self, actions: Vec<DeferredMenuAction>) {
294        for action in actions {
295            match action {
296                DeferredMenuAction::Close { send_packet } => {
297                    if send_packet {
298                        self.close_container();
299                    } else {
300                        self.do_close_container();
301                    }
302                }
303                DeferredMenuAction::Open(prepared) => {
304                    self.execute_menu_open(*prepared);
305                }
306                DeferredMenuAction::Install(prepared) => {
307                    let PreparedMenu { title, menu } = *prepared;
308                    self.open_prepared_menu(title, menu);
309                }
310                DeferredMenuAction::SetSlot { index, stack } => {
311                    let _ = self.set_open_container_item(index, stack);
312                }
313            }
314        }
315
316        self.finish_menu_open_operation();
317    }
318
319    fn queue_deferred_menus(
320        terminal_removal: &mut TerminalMenuRemoval,
321        actions: Vec<DeferredMenuAction>,
322    ) {
323        terminal_removal
324            .pending_menus
325            .extend(actions.into_iter().filter_map(|action| match action {
326                DeferredMenuAction::Install(prepared) => Some(prepared.menu),
327                DeferredMenuAction::Close { .. }
328                | DeferredMenuAction::Open(_)
329                | DeferredMenuAction::SetSlot { .. } => None,
330            }));
331    }

334        let mut open_menu = self.open_menu.lock();
335        if open_menu.terminal_removal.is_some() {
336            return false;
337        }
338        open_menu.active_open_operations += 1;

351    /// Attempts to pick up nearby item entities.
352    ///
353    /// Mirrors vanilla's `Player.aiStep()` item pickup logic:
354    /// - Calculates pickup area as bounding box inflated by (1.0, 0.5, 1.0)
355    /// - Calls `playerTouch()` on each entity in range
356    pub(in crate::player) fn touch_nearby_items(&self) {
357        if self.game_mode() == GameType::Spectator {
358            return;
359        }
360
361        let pickup_area = self.bounding_box().inflate_xyz(1.0, 0.5, 1.0);
362        let world = self.get_world();
363        let entities = world.get_entities_in_aabb(&pickup_area);
364
365        let Some(player_arc) = world.players.get_by_entity_id(self.id()) else {
366            return;
367        };
368
369        for entity in entities {
370            if entity.id() == self.id() || entity.is_removed() {
371                continue;
372            }
373
374            entity.player_touch(&player_arc);
375        }
376    }

378    /// Handles a container button click packet.
379    ///
380    /// Vanilla parity: `ServerGamePacketListenerImpl.handleContainerButtonClick`,
381    /// which is what an enchanting table's three offers, a stonecutter's recipe
382    /// list, a loom's patterns and a lectern's page arrows all arrive through.
383    pub fn handle_container_button_click(&self, packet: SContainerButtonClick) {
384        if self.game_mode() == GameType::Spectator || self.get_health() <= 0.0 {
385            return;
386        }
387
388        let Ok(mut menu) = self.take_open_menu_for_callback(Some(packet.container_id)) else {
389            return;
390        };
391        if menu.still_valid(self) {
392            menu.click_menu_button(self, packet.button_id);
393        }
394        self.finish_open_menu_callback(menu);
395        // The menu is installed before this callback so Bukkit's view accessors
396        // observe the same open view that the client received. Cancelling closes
397        // that view immediately, matching the externally visible Bukkit result.
398        let mut open_event = InventoryOpenEvent::new(self.gameprofile.id);
399        self.fire_event(&mut open_event);
400        if open_event.is_cancelled() {
401            self.close_container();
402        }
403    }

437    /// Processes a container click on any menu implementing the Menu trait.
438    ///
439    /// This is the common implementation shared between inventory menu and
440    /// external menus (crafting table, chest, etc.).
441    #[expect(
442        clippy::too_many_lines,
443        reason = "one arm per vanilla click type; splitting hides the table"
444    )]
445    fn process_container_click(&self, menu: &mut Menu, packet: SContainerClick) {
446        if self.game_mode() == GameType::Spectator || self.get_health() <= 0.0 {
447            menu.behavior_mut()
448                .send_all_data_to_remote(&self.connection);
449            return;
450        }
451
452        if !menu.still_valid(self) {
453            log::debug!(
454                "Player {} interacted with invalid menu",
455                self.gameprofile.name
456            );
457            return;
458        }
459
460        // Vanilla rejects positive out-of-range slots before applying client
461        // prediction hashes or resynchronizing the menu. Its validity check
462        // admits every negative slot because each is less than the slot count.
463        let slot_count = menu.behavior().slot_count();
464        let packet_slot_is_valid = packet.slot_num < 0
465            || usize::try_from(packet.slot_num).is_ok_and(|slot| slot < slot_count);
466        if !packet_slot_is_valid {
467            log::debug!(
468                "Player {} clicked invalid slot index: {}, available slots: {}",
469                self.gameprofile.name,
470                packet.slot_num,
471                slot_count
472            );
473            return;
474        }
475
476        // Parse and validate the remaining raw click fields once. A malformed
477        // button or drag encoding is not applied, but the state sync below
478        // still runs so the client's prediction gets corrected.
479        let click = Click::parse(
480            packet.slot_num,
481            packet.button_num,
482            packet.click_type,
483            slot_count,
484        );
485        if click.is_none() {
486            log::debug!(
487                "Player {} sent malformed container click (slot {}, button {}, {:?})",
488                self.gameprofile.name,
489                packet.slot_num,
490                packet.button_num,
491                packet.click_type
492            );
493            // Vanilla rejects positive out-of-range slots before `doClick`.
494            // Once admitted, any non-QuickCraft input cancels an active drag,
495            // and malformed QuickCraft headers/types reset their own state.
496            let quick_craft_header = packet.button_num & 3;
497            let quick_craft_type = (packet.button_num >> 2) & 3;
498            if packet.click_type != ClickType::QuickCraft
499                || quick_craft_header == 3
500                || (quick_craft_header == 0 && quick_craft_type == 3)
501            {
502                menu.behavior_mut().reset_quick_craft();
503            }
504        }
505
506        let full_resync_needed = packet.state_id as u32 != menu.behavior().state_id();
507
508        menu.behavior_mut().suppress_remote_updates();
509
510        if let Some(click) = click {
511            if matches!(click, Click::QuickCraft(QuickCraft::End)) {
512                let drag_type = menu
513                    .behavior()
514                    .quickcraft()
515                    .map_or_else(|| "UNKNOWN".to_owned(), |kind| format!("{kind:?}"));
516                let mut drag_event = InventoryDragEvent::new(
517                    self.gameprofile.id,
518                    menu.behavior().quickcraft_slots().to_vec(),
519                    menu.behavior().carried().clone(),
520                    drag_type,
521                );
522                self.fire_event(&mut drag_event);
523                if drag_event.is_cancelled() {
524                    menu.behavior_mut().reset_quick_craft();
525                    menu.behavior_mut().resume_remote_updates();
526                    menu.behavior_mut()
527                        .send_all_data_to_remote(&self.connection);
528                    return;
529                }
530            }
531            let current_item = click.slot().and_then(|slot| {
532                let guard = menu.behavior().lock_all_containers();
533                menu.behavior()
534                    .slots()
535                    .get(slot)
536                    .map(|view| view.get_item(&guard).clone())
537            });
538            let click_name = match (packet.click_type, packet.button_num) {
539                (ClickType::Pickup | ClickType::QuickCraft, 0) => "LEFT",
540                (ClickType::Pickup | ClickType::QuickCraft, 1) => "RIGHT",
541                (ClickType::QuickMove, 0) => "SHIFT_LEFT",
542                (ClickType::QuickMove, 1) => "SHIFT_RIGHT",
543                (ClickType::Swap, _) => "NUMBER_KEY",
544                (ClickType::Clone, _) => "MIDDLE",
545                (ClickType::Throw, 0) => "DROP",
546                (ClickType::Throw, 1) => "CONTROL_DROP",
547                (ClickType::PickupAll, _) => "DOUBLE_CLICK",
548                _ => "UNKNOWN",
549            };
550            let cursor_item = menu.behavior().carried().clone();
551            let mut inventory_click = InventoryClickEvent::new(
552                self.gameprofile.id,
553                current_item,
554                Some(cursor_item),
555                click_name.to_owned(),
556                click.slot(),
557            );
558            self.fire_event(&mut inventory_click);
559            if inventory_click.is_cancelled() {
560                menu.behavior_mut().resume_remote_updates();
561                menu.behavior_mut()
562                    .send_all_data_to_remote(&self.connection);
563                return;
564            }
565            menu.clicked(click, self);
566        }
567
568        for (slot, hash) in packet.changed_slots {
569            let slot = slot as usize;
570            // Result/fake slots are server-authoritative (their contents are
571            // recomputed from a recipe). Don't let the client's prediction set
572            // our view of what it knows, or `broadcast_changes` will think the
573            // client already has the freshly-crafted result and skip syncing it
574            // — leaving the slot blank until the next click forces a resend.
575            if menu
576                .behavior()
577                .slots()
578                .get(slot)
579                .is_some_and(|slot| slot.is_fake())
580            {
581                menu.behavior_mut().mark_remote_slot_unknown(slot);
582                continue;
583            }
584            menu.behavior_mut().set_remote_slot(slot, hash);
585        }
586
587        menu.behavior_mut().set_remote_carried(packet.carried_item);
588        menu.behavior_mut().resume_remote_updates();
589
590        if full_resync_needed {
591            menu.behavior_mut()
592                .send_all_data_to_remote(&self.connection);
593        } else {
594            menu.behavior_mut().broadcast_changes(&self.connection);
595        }
596    }

634    /// Handles a container slot state changed packet: the crafter's slot toggle.
635    ///
636    /// Vanilla parity: `ServerGamePacketListenerImpl.handleContainerSlotStateChanged`.
637    pub fn handle_container_slot_state_changed(
638        self: &Arc<Self>,
639        packet: SContainerSlotStateChanged,
640    ) {
641        let Ok(slot) = usize::try_from(packet.slot_id) else {
642            return;
643        };
644
645        match self.take_open_menu_for_callback(Some(packet.container_id)) {
646            Ok(mut menu) => {
647                if menu.still_valid(self) {
648                    menu.set_slot_state(self, slot, packet.new_state);

651            }
652            Err(OpenMenuUnavailable::Closed) => {
653                log::debug!("slot state change without an open menu");
654            }
655            Err(OpenMenuUnavailable::Unavailable) => {}
656        }
657    }

678    /// Handles a creative mode slot set packet.
679    pub fn handle_set_creative_mode_slot(&self, packet: SSetCreativeModeSlot) {
680        if self.game_mode() != GameType::Creative {
681            return;
682        }
683
684        let drop = packet.slot_num < 0;
685        let item_stack = packet.item_stack;
686
687        let valid_slot = packet.slot_num >= 1 && packet.slot_num <= 45;
688        let valid_data = item_stack.is_empty() || item_stack.count <= item_stack.max_stack_size();
689
690        if valid_slot && valid_data {
691            let mut menu = self.inventory_menu.lock();
692            let slot_index = packet.slot_num as usize;
693
694            {
695                let mut guard = menu.behavior().lock_all_containers();
696                if let Some(slot) = menu.behavior().slots().get(slot_index) {
697                    let previous = slot.get_item(&guard).clone();
698                    slot.set_by_player(&mut guard, item_stack.clone(), &previous);
699                }
700            }
701            if (1..=4).contains(&slot_index) {
702                menu.update_crafting_result();
703            }
704            menu.behavior_mut()
705                .set_remote_slot_known(slot_index, &item_stack);
706            menu.behavior_mut().broadcast_changes(&self.connection);
707        } else if drop && valid_data && !item_stack.is_empty() {
708            // Vanilla parity: `dropSpamThrottler.increment(); player.drop(stack,
709            // true)`. This used to log that the item "would" be dropped and then
710            // discard it, so a creative player dragging a stack out of the
711            // window watched it disappear instead of hitting the ground.
712            if self.allow_creative_drop() {
713                let _ = self.drop_item(item_stack, false, true);
714            } else {
715                log::warn!(
716                    "Player {} was dropping items too fast in creative mode, ignoring.",
717                    self.gameprofile.name
718                );
719            }
720        }
721    }

754    /// Opens a menu for this player.
755    ///
756    /// Based on Java's `ServerPlayer::openMenu`.
757    ///
758    /// # Arguments
759    /// * `title` - The display title shown in the open-screen packet.
760    /// * `create` - Factory invoked with the allocated container id, player,
761    ///   and current world. If called by a menu hook, the factory runs after
762    ///   that hook releases its container locks.
763    ///
764    /// # Panics
765    /// Panics if the created menu uses a different container id than the one
766    /// allocated for it, or has no menu type (i.e. the player's own inventory
767    /// menu, which must never be opened via `open_menu`).
768    pub fn open_menu(
769        &self,
770        title: impl Into<TextComponent>,
771        create: impl for<'a> FnOnce(MenuOpenContext<'a>) -> Menu + Send + 'static,
772    ) {
773        if !self.begin_menu_open_operation() {
774            return;
775        }
776        self.open_menu_inner(PendingMenuOpen {
777            title: title.into(),
778            create: Box::new(create),
779        });
780        self.finish_menu_open_operation();
781    }

783    fn open_menu_inner(&self, pending: PendingMenuOpen) {
784        self.do_close_container();
785
786        let mut open_menu = self.open_menu.lock();
787        if open_menu.terminal_removal.is_some() {
788            return;
789        }
790        if let Some(dispatch) = open_menu.dispatch.as_mut() {
791            dispatch
792                .actions
793                .push(DeferredMenuAction::Open(Box::new(pending)));
794            return;
795        }
796        drop(open_menu);
797
798        self.execute_menu_open(pending);
799    }

801    fn execute_menu_open(&self, pending: PendingMenuOpen) {
802        {
803            let mut open_menu = self.open_menu.lock();
804            if open_menu.terminal_removal.is_some() {
805                return;
806            }
807            if let Some(dispatch) = open_menu.dispatch.as_mut() {
808                dispatch
809                    .actions
810                    .push(DeferredMenuAction::Open(Box::new(pending)));
811                return;
812            }
813        }
814
815        let PendingMenuOpen { title, create } = pending;
816        let container_id = self.next_container_counter();
817        let world = self.get_world();
818        let menu = create(MenuOpenContext {
819            container_id,
820            player: self,
821            world: &world,
822        });
823        assert_eq!(
824            menu.container_id(),
825            container_id,
826            "open_menu factory returned container id {}, but {} was allocated",
827            menu.container_id(),
828            container_id,
829        );
830        self.open_prepared_menu(title, menu);
831    }

833    fn open_prepared_menu(&self, title: TextComponent, mut menu: Menu) {
834        loop {
835            {
836                let mut open_menu = self.open_menu.lock();
837                if let Some(terminal_removal) = open_menu.terminal_removal.as_mut() {
838                    terminal_removal.pending_menus.push(menu);
839                    return;
840                }
841            }
842
843            // A removal hook may have opened another menu while the initiating
844            // open call was closing its predecessor.
845            self.do_close_container();
846
847            let mut open_menu = self.open_menu.lock();
848            if let Some(terminal_removal) = open_menu.terminal_removal.as_mut() {
849                terminal_removal.pending_menus.push(menu);
850                return;
851            }
852            if let Some(dispatch) = open_menu.dispatch.as_mut() {
853                dispatch
854                    .actions
855                    .push(DeferredMenuAction::Install(Box::new(PreparedMenu {
856                        title,
857                        menu,
858                    })));
859                return;
860            }
861            if open_menu.menu.is_some() {
862                continue;
863            }
864            open_menu.title = Some(title.to_plain(&DisplayResolutor));
865            let top_slot_count = menu_top_slot_count(&menu).unwrap_or(0);
866            let menu_type = menu.menu_type().map(|menu_type| menu_type.key.to_string());
867            let snapshot = {
868                let guard = menu.behavior().lock_all_containers();
869                (0..top_slot_count)
870                    .filter_map(|index| {
871                        menu.behavior()
872                            .slots()
873                            .get(index)
874                            .map(|slot| slot.get_item(&guard).clone())
875                    })
876                    .collect()
877            };
878            open_menu.dispatch = Some(OpenMenuDispatch {
879                overrides_player_slots: menu.overrides_player_slots(),
880                top_slot_count,
881                menu_type,
882                snapshot,
883                actions: Vec::new(),
884            });
885            break;
886        }
887
888        match menu.screen_opener() {
889            ScreenOpener::Screen => self.send_packet(COpenScreen {
890                container_id: i32::from(menu.container_id()),
891                menu_type: menu
892                    .menu_type()
893                    .expect("a menu opened via open_menu must declare a menu type"),
894                title,
895            }),
896            // Vanilla parity: `ServerPlayer.openHorseInventory`, which sends no
897            // title -- the client names the screen after the mount.
898            ScreenOpener::Mount {
899                inventory_columns,
900                entity_id,
901            } => self.send_packet(CMountScreenOpen {
902                container_id: i32::from(menu.container_id()),
903                inventory_columns: inventory_columns as i32,
904                entity_id,
905            }),
906        }
907
908        // Fire on_open before the full sync so anything the menu populates here
909        // is included in the first render sent below.
910        menu.on_open(self);
911
912        menu.behavior_mut()
913            .send_all_data_to_remote(&self.connection);
914
915        self.finish_open_menu_callback(menu);
916    }

964    pub(in crate::player) fn remove_all_menus_with_disposition(
965        &self,
966        disposition: MenuItemDisposition,
967    ) -> MenuRemovalStatus {
968        let menu = {
969            let mut open_menu = self.open_menu.lock();
970            if let Some(terminal_removal) = open_menu.terminal_removal.as_mut() {
971                terminal_removal.disposition = terminal_removal.disposition.combine(disposition);
972                return MenuRemovalStatus::Pending;
973            }
974
975            open_menu.terminal_removal = Some(TerminalMenuRemoval {
976                disposition,
977                main_cleanup_complete: false,
978                pending_cleanup_in_progress: false,
979                pending_menus: Vec::new(),
980            });
981            if open_menu.dispatch.is_some() {
982                return MenuRemovalStatus::Pending;
983            }
984
985            let menu = open_menu.menu.take();
986            open_menu.title = None;
987            menu
988        };
989
990        self.finish_terminal_menu_main_cleanup(menu);
991        if self.open_menu.lock().terminal_removal.is_none() {
992            MenuRemovalStatus::Complete
993        } else {
994            MenuRemovalStatus::Pending
995        }
996    }

998    fn finish_terminal_menu_main_cleanup(&self, mut menu: Option<Menu>) {
999        self.inventory_menu.lock().removed(self);
1000        if let Some(menu) = menu.as_mut() {
1001            self.remove_open_menu(menu);
1002        }
1003
1004        {
1005            let mut open_menu = self.open_menu.lock();
1006            let Some(terminal_removal) = open_menu.terminal_removal.as_mut() else {
1007                return;
1008            };
1009            terminal_removal.main_cleanup_complete = true;
1010        }
1011        self.try_finish_terminal_menu_removal();
1012    }

1014    fn try_finish_terminal_menu_removal(&self) {
1015        loop {
1016            let pending_menus = {
1017                let mut open_menu = self.open_menu.lock();
1018                if open_menu.active_open_operations != 0 {
1019                    return;
1020                }
1021                let Some(terminal_removal) = open_menu.terminal_removal.as_mut() else {
1022                    return;
1023                };
1024                if !terminal_removal.main_cleanup_complete {
1025                    return;
1026                }
1027                if terminal_removal.pending_cleanup_in_progress {
1028                    return;
1029                }
1030                if terminal_removal.pending_menus.is_empty() {
1031                    open_menu.terminal_removal = None;
1032                    debug_assert!(open_menu.menu.is_none());
1033                    return;
1034                }
1035                terminal_removal.pending_cleanup_in_progress = true;
1036                mem::take(&mut terminal_removal.pending_menus)
1037            };
1038
1039            for mut pending_menu in pending_menus {
1040                pending_menu.removed(self);
1041            }
1042
1043            let mut open_menu = self.open_menu.lock();
1044            let Some(terminal_removal) = open_menu.terminal_removal.as_mut() else {
1045                return;
1046            };
1047            terminal_removal.pending_cleanup_in_progress = false;
1048        }
1049    }

1057        let menu = {
1058            let mut open_menu = self.open_menu.lock();
1059            if open_menu.terminal_removal.is_some() {
1060                return;
1061            }
1062            if let Some(dispatch) = open_menu.dispatch.as_mut() {
1063                dispatch
1064                    .actions
1065                    .push(DeferredMenuAction::Close { send_packet });
1066                return;
1067            }
1068            let Some(menu) = open_menu.menu.take() else {
1069                return;
1070            };
1071            open_menu.title = None;
1072            open_menu.dispatch = Some(OpenMenuDispatch {
1073                overrides_player_slots: menu.overrides_player_slots(),
1074                top_slot_count: 0,
1075                menu_type: None,
1076                snapshot: Vec::new(),
1077                actions: Vec::new(),
1078            });
1079            menu
1080        };
1081
1082        let mut menu = menu;
1083        if send_packet {
1084            self.send_packet(CContainerClose {
1085                container_id: i32::from(menu.container_id()),
1086            });
1087        }
1088        let mut close_event = InventoryCloseEvent::new(self.gameprofile.id);
1089        self.fire_event(&mut close_event);
1090        self.remove_open_menu(&mut menu);
1091        self.finish_open_menu_removal();
1092    }

1183    /// Replaces one slot in the currently open external menu.
1184    ///
1185    /// The menu owns the slot policy and its backing containers, so mutation
1186    /// goes through the slot abstraction rather than bypassing menu rules.
1187    pub fn set_open_container_item(&self, index: usize, stack: ItemStack) -> bool {
1188        let mut open_menu = self.open_menu.lock();
1189        if let Some(dispatch) = open_menu.dispatch.as_mut() {
1190            if index >= dispatch.top_slot_count {
1191                return false;
1192            }
1193            dispatch
1194                .actions
1195                .push(DeferredMenuAction::SetSlot { index, stack });
1196            return true;
1197        }
1198        let Some(menu) = open_menu.menu.as_ref() else {
1199            return false;
1200        };
1201        let Some(top_count) = menu_top_slot_count(menu) else {
1202            return false;
1203        };
1204        let Some(slot) = menu.behavior().slots().get(index) else {
1205            return false;
1206        };
1207        if index >= top_count {
1208            return false;
1209        }
1210        let mut guard = menu.behavior().lock_all_containers();
1211        slot.set_item(&mut guard, stack);
1212        drop(guard);
1213        drop(open_menu);
1214        self.broadcast_inventory_changes();
1215        true
1216    }

1257    /// Sends the latest values for requested logical inventory slots.
1258    pub(in crate::player) fn flush_inventory_resync(&self) {
1259        let overrides_player_slots = {
1260            let open_menu = self.open_menu.lock();
1261            open_menu
1262                .menu
1263                .as_ref()
1264                .is_some_and(Menu::overrides_player_slots)
1265                || open_menu
1266                    .dispatch
1267                    .as_ref()
1268                    .is_some_and(|dispatch| dispatch.overrides_player_slots)
1269        };
1270        let slots = self
1271            .inventory_sync
1272            .lock()
1273            .take_ready(overrides_player_slots);
1274        if slots.is_empty() {
1275            return;
1276        }
1277
1278        let packets = {
1279            let inventory = self.inventory.lock();
1280            slots
1281                .into_iter()
1282                .map(|slot| CSetPlayerInventory {
1283                    slot: slot as i32,
1284                    item_stack: inventory.get_item(slot).clone(),
1285                })
1286                .collect::<Vec<_>>()
1287        };
1288        for packet in packets {
1289            self.send_packet(packet);
1290        }
1291    }

1293    /// The stack the cursor is holding.
1294    ///
1295    /// Vanilla parity: `Player.containerMenu.getCarried()`. `containerMenu` is
1296    /// whichever menu is open and falls back to the player's own inventory
1297    /// menu when none is, which is the same pair
1298    /// [`Self::clear_or_count_matching_items`] walks.
1299    pub(crate) fn carried_item(&self) -> ItemStack {
1300        let open_menu = self.open_menu.lock();
1301        if let Some(menu) = open_menu.menu.as_ref() {
1302            return menu.behavior().carried().clone();
1303        }
1304        if open_menu.dispatch.is_some() {
1305            // A menu handed to a callback has been moved out of `open_menu`
1306            // and is not the inventory menu either, so there is nothing here
1307            // to read. Reading the inventory menu instead would answer with a
1308            // cursor the player is not holding.
1309            return ItemStack::empty();
1310        }
1311        drop(open_menu);
1312        self.inventory_menu.lock().behavior().carried().clone()
1313    }

1315    /// Removes or counts matching stacks across every location used by vanilla `/clear`.
1316    pub(crate) fn clear_or_count_matching_items(
1317        &self,
1318        predicate: &dyn Fn(&ItemStack) -> bool,
1319        amount_to_remove: i32,
1320    ) -> i32 {
1321        let counting_only = amount_to_remove == 0;
1322        let mut count = self.inventory.lock().clear_or_count_matching_items(
1323            predicate,
1324            amount_to_remove,
1325            counting_only,
1326        );
1327
1328        count += self.inventory_menu.lock().clear_or_count_crafting_items(
1329            predicate,
1330            amount_to_remove - count,
1331            counting_only,
1332        );
1333
1334        let has_open_menu = {
1335            let mut open_menu = self.open_menu.lock();
1336            if let Some(menu) = open_menu.menu.as_mut() {
1337                let behavior = menu.behavior_mut();
1338                count += clear_or_count_matching_stack(
1339                    behavior.carried_mut(),
1340                    predicate,
1341                    amount_to_remove - count,
1342                    counting_only,
1343                );
1344                if behavior.carried().is_empty() {
1345                    *behavior.carried_mut() = ItemStack::empty();
1346                }
1347                true
1348            } else {
1349                open_menu.dispatch.is_some()
1350            }
1351        };
1352        if !has_open_menu {
1353            let mut inventory_menu = self.inventory_menu.lock();
1354            let behavior = inventory_menu.behavior_mut();
1355            count += clear_or_count_matching_stack(
1356                behavior.carried_mut(),
1357                predicate,
1358                amount_to_remove - count,
1359                counting_only,
1360            );
1361            if behavior.carried().is_empty() {
1362                *behavior.carried_mut() = ItemStack::empty();
1363            }
1364        }
1365
1366        self.inventory_menu.lock().update_crafting_result();
1367        self.broadcast_inventory_changes();
1368        count
1369    }

1371    /// Drops an item from the player's selected hotbar slot.
1372    ///
1373    /// Based on Java's `ServerPlayer.drop(boolean all)`.
1374    ///
1375    /// - `all`: If true, drops the entire stack (Ctrl+Q). If false, drops one item (Q).
1376    pub fn drop_from_selected(&self, all: bool) {
1377        if !self.can_drop_items() {
1378            return;
1379        }
1380
1381        let removed = {
1382            let mut inventory = self.inventory.lock();
1383            let selected_count = inventory.get_selected_item().count();
1384            if selected_count == 0 {
1385                return;
1386            }
1387            inventory.split_item_in_hand(
1388                InteractionHand::MainHand,
1389                if all { selected_count } else { 1 },
1390            )
1391        };
1392
1393        // `drop_item` puts the stack back itself now, so restoring it here too
1394        // would hand the player a second copy.
1395        let _ = self.drop_item(removed, false, true);
1396    }

1398    /// Drops an item into the world.
1399    ///
1400    /// Based on Java's `LivingEntity.drop(ItemStack, boolean randomly, boolean thrownFromHand)`.
1401    ///
1402    /// - `throw_randomly`: If true, the item is thrown in a random direction.
1403    ///   If false, it's thrown in the direction the player is facing.
1404    /// - `thrown_from_hand`: If true, sets the thrower and uses a longer pickup delay.
1405    #[must_use]
1406    pub fn drop_item(
1407        &self,
1408        item: ItemStack,
1409        throw_randomly: bool,
1410        thrown_from_hand: bool,
1411    ) -> Option<Arc<ItemEntity>> {
1412        if item.is_empty() {
1413            return None;
1414        }
1415
1416        let pos = self.position();
1417        let (yaw, pitch) = self.rotation();
1418
1419        let spawn_y = self.get_eye_y() - 0.3;
1420
1421        let velocity = if throw_randomly {
1422            let power = rand::random::<f32>() * 0.5;
1423            let angle = rand::random::<f32>() * TAU;
1424            DVec3::new(
1425                f64::from(-angle.sin() * power),
1426                0.2,
1427                f64::from(angle.cos() * power),
1428            )
1429        } else {
1430            let pitch_rad = pitch.to_radians();
1431            let yaw_rad = yaw.to_radians();
1432
1433            let sin_pitch = pitch_rad.sin();
1434            let cos_pitch = pitch_rad.cos();
1435            let sin_yaw = yaw_rad.sin();
1436            let cos_yaw = yaw_rad.cos();
1437
1438            let angle_offset = rand::random::<f32>() * TAU;
1439            let power_offset = 0.02 * rand::random::<f32>();
1440
1441            DVec3::new(
1442                f64::from(-sin_yaw * cos_pitch * 0.3)
1443                    + f64::from(angle_offset.cos() * power_offset),
1444                f64::from(-sin_pitch * 0.3 + 0.1)
1445                    + f64::from((rand::random::<f32>() - rand::random::<f32>()) * 0.1),
1446                f64::from(cos_yaw * cos_pitch * 0.3) + f64::from(angle_offset.sin() * power_offset),
1447            )
1448        };
1449
1450        let spawn_pos = DVec3::new(pos.x, spawn_y, pos.z);
1451
1452        // Vanilla's `Level.addFreshEntity` cannot refuse, so a vanilla drop always
1453        // reaches the ground. Foton's can fail -- a cancelled `ItemSpawnEvent`, or
1454        // a spawn position whose chunk is not Full -- and a plugin can cancel
1455        // `PlayerDropItemEvent` after the fact, which Bukkit defines as "the drop
1456        // did not happen" rather than "the item is gone". Both paths used to
1457        // consume the stack and hand back `None`, and every caller but one throws
1458        // that `None` away, so the item simply ceased to exist.
1459        let undropped = item.clone();
1460        let Some(entity) = self
1461            .get_world()
1462            .spawn_item_with_velocity(spawn_pos, item, velocity)
1463        else {
1464            self.restore_undropped_item(undropped);
1465            return None;
1466        };
1467        entity.set_pickup_delay(40);
1468        if thrown_from_hand {
1469            entity.set_thrower(self.gameprofile.id);
1470        }
1471        let world = self.get_world();
1472        let mut event = PlayerDropItemEvent::new(self.gameprofile.id, entity.uuid());
1473        world.fire_event(&mut event);
1474        if event.is_cancelled() {
1475            let _ = world.remove_entity(entity.id());
1476            self.restore_undropped_item(undropped);
1477            return None;
1478        }
1479        Some(entity)
1480    }

1482    /// Puts back a stack the world refused to take.
1483    ///
1484    /// `try_lock` rather than `lock`: this runs on the failure path of a drop,
1485    /// and some callers reach `drop_item` from inside inventory handling. A
1486    /// missed restore is an item on the floor of the log; a deadlock is the
1487    /// server. Anything that still does not fit is reported rather than dropped
1488    /// silently, because a vanished stack with no trace is unanswerable when a
1489    /// player asks where it went.
1490    fn restore_undropped_item(&self, mut item: ItemStack) {
1491        if item.is_empty() {
1492            return;
1493        }
1494        if let Some(mut inventory) = self.inventory.try_lock() {
1495            let _ = inventory.add(&mut item);
1496        }
1497        if !item.is_empty() {
1498            log::warn!(
1499                "player {} could neither drop nor store {} x{}",
1500                self.gameprofile.name,
1501                item.item().key,
1502                item.count()
1503            );
1504        }
1505    }

1517    /// Returns whether items from a closing menu (crafting grid, anvil inputs,
1518    /// cursor) should be placed back into the inventory instead of dropped into
1519    /// the world.
1520    ///
1521    /// Matches vanilla's `AbstractContainerMenu.dropOrPlaceInInventory`: a
1522    /// disconnected player or one removed for any reason except a world change
1523    /// drops the items.
1524    #[must_use]
1525    pub fn returns_menu_items_to_inventory(&self) -> bool {
1526        if let Some(disposition) = self
1527            .open_menu
1528            .lock()
1529            .terminal_removal
1530            .as_ref()
1531            .map(|terminal_removal| terminal_removal.disposition)
1532        {
1533            return disposition == MenuItemDisposition::ReturnToInventory;
1534        }
1535
1536        self.default_menu_item_disposition() == MenuItemDisposition::ReturnToInventory
1537    }
1538
1539    fn default_menu_item_disposition(&self) -> MenuItemDisposition {
1540        let removed_outside_world_change =
1541            self.is_removed() && self.removal_reason() != Some(RemovalReason::ChangedWorld);
1542        if removed_outside_world_change || self.connection.closed() || self.get_health() <= 0.0 {
1543            MenuItemDisposition::Drop
1544        } else {
1545            MenuItemDisposition::ReturnToInventory
1546        }
1547    }

1549    /// Tries to add an item to the player's inventory, dropping it if it doesn't fit.
1550    ///
1551    /// Based on Java's `Inventory.placeItemBackInInventory`.
1552    pub fn add_item_or_drop(&self, mut item: ItemStack) {
1553        if item.is_empty() {
1554            return;
1555        }
1556
1557        let added = self.inventory.lock().add(&mut item);
1558        if !added || !item.is_empty() {
1559            let _ = self.drop_item(item, false, false);
1560        }
1561    }

1563    /// Tries to add an item to the player's inventory using an existing lock guard,
1564    /// dropping it if it doesn't fit.
1565    ///
1566    /// Use this variant when you already hold a `ContainerLockGuard` that includes
1567    /// the player's inventory to avoid deadlocks.
1568    pub fn add_item_or_drop_with_guard(&self, guard: &mut ContainerLockGuard, mut item: ItemStack) {
1569        if item.is_empty() {
1570            return;
1571        }
1572
1573        let inv_id = ContainerId::from_arc(&self.inventory);
1574        let should_drop = if let Some(inv) = guard.get_mut(inv_id) {
1575            let added = inv.add(&mut item);
1576            !added || !item.is_empty()
1577        } else {
1578            true
1579        };
1580        if should_drop {
1581            let _ = guard.run_unlocked(|| self.drop_item(item, false, false));
1582        }
1583    }
