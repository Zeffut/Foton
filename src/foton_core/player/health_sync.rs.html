1//! Delta-tracking state for `CSetHealth` packet deduplication.
2//!
3//! Tracks the last health/food/saturation values sent to the client so we only
4//! send `CSetHealth` when something actually changes.
5//!
6//! Vanilla: `ServerPlayer.lastSentHealth`, `lastSentFood`, `lastFoodSaturationZero`.
7//!
8//! DELIBERATE DIVERGENCE: vanilla remembers only whether saturation *was zero*,
9//! so a change from 5.0 to 14.6 sends nothing. That costs vanilla nothing --
10//! its own HUD reads saturation only to decide whether the hunger bar should
11//! wobble, which is a question about zero. But it means the value a client
12//! holds is right by accident, and goes stale the moment health and food both
13//! sit still: eat at full health and full hunger and the client is never told.
14//! Foton compares the value instead. The cost is an occasional extra packet;
15//! the gain is that anything reading saturation client-side is accurate.
16
17use crate::player::Player;
18
19/// Tracks the last health/food/saturation values sent to the client.
20pub struct HealthSyncState {
21    /// Last health value sent to the client.
22    pub last_health: f32,
23    /// Last food level sent to the client.
24    pub last_food: i32,
25    /// Last saturation value sent to the client.
26    ///
27    /// Vanilla keeps a `lastFoodSaturationZero` flag here instead -- see the
28    /// module note on why Foton keeps the value.
29    pub last_saturation: f32,
30}
31
32impl HealthSyncState {
33    /// Creates a new state that will trigger a send on the first tick.
34    #[must_use]
35    pub const fn new() -> Self {
36        Self {
37            last_health: -1.0,
38            last_food: -1,
39            last_saturation: -1.0,
40        }
41    }
42
43    /// Returns true if the given values differ from what was last sent.
44    #[expect(
45        clippy::float_cmp,
46        reason = "intentional exact comparison: we only want to send updates when the value changes from what we last sent"
47    )]
48    #[must_use]
49    pub fn needs_update(&self, health: f32, food: i32, saturation: f32) -> bool {
50        self.last_health != health || self.last_food != food || self.last_saturation != saturation
51    }
52
53    /// Records that we just sent the given values to the client.
54    pub const fn record_sent(&mut self, health: f32, food: i32, saturation: f32) {
55        self.last_health = health;
56        self.last_food = food;
57        self.last_saturation = saturation;
58    }
59
60    /// Invalidates the state so the next tick will re-send.
61    ///
62    /// Vanilla: `resetSentInfo`.
63    pub const fn invalidate(&mut self) {
64        self.last_health = -1.0e8;
65    }
66
67    /// Resets to respawn defaults (forces re-send on next tick).
68    pub const fn reset_for_respawn(&mut self) {
69        self.last_health = -1.0;
70        self.last_food = -1;
71    }
72}
73
74impl Player {
75    /// Invalidates every delta the client's HUD is drawn from, so that the next
76    /// `tick()` sends the lot again.
77    ///
78    /// Vanilla parity: the three assignments of `ServerPlayer.resetSentInfo` --
79    /// `lastSentHealth`, `lastSentFood` and `lastSentExp`. The experience third
80    /// is the one that cannot be left out: health and food are polled against a
81    /// remembered value every tick, so any stale copy eventually corrects
82    /// itself, but the experience packet is sent only when a writer flags it.
83    /// A `ClientboundRespawnPacket` throws away the `LocalPlayer` that held the
84    /// bar with no writer involved at all, and level, progress and total are
85    /// plain fields on it -- not attributes, not entity data -- so no
86    /// `dataToKeep` bit can carry them across. Without this the bar reads zero
87    /// from the moment a player walks a portal until something happens to
88    /// change their experience.
89    pub fn reset_sent_info(&self) {
90        self.health_sync.lock().invalidate();
91        self.experience.lock().dirty = true;
92    }
93}
