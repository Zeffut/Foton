15    /// Resets the player's transient state and prepares them for a new world.
16    ///
17    /// This is the shared "clean slate" path used by initial join, respawn, and
18    /// world change. If the player is currently in a different world, they are
19    /// removed from the old world first.
20    ///
21    /// Vanilla creates a fresh `ServerPlayer` for death and End-credits respawns,
22    /// but reuses it for dimension changes. Foton reuses the same `Player` for
23    /// every path, so this resets only the transient state appropriate to `reason`.
24    pub(crate) fn reset(self: &Arc<Self>, new_world: Arc<World>, reason: ResetReason) {
25        self.reset_inner_after(new_world, reason, false, || {});
26    }

39    fn reset_inner_after<F>(
40        self: &Arc<Self>,
41        new_world: Arc<World>,
42        reason: ResetReason,
43        source_world_detached: bool,
44        restore_state: F,
45    ) where
46        F: FnOnce(),
47    {
48        if reason != ResetReason::InitialJoin {
49            assert_eq!(
50                self.remove_all_menus(),
51                MenuRemovalStatus::Complete,
52                "player reset menu removal must run outside a menu callback"
53            );
54        }
55        if matches!(reason, ResetReason::Respawn | ResetReason::EndCredits) {
56            // Vanilla creates a fresh ServerPlayer and inventory menu for these paths.
57            self.inventory_menu
58                .lock()
59                .behavior_mut()
60                .reset_quick_craft();
61        }
62
63        let old_world = self.get_world();
64        let switching_worlds = !Arc::ptr_eq(&old_world, &new_world);
65
66        if switching_worlds {
67            self.send_packet(CContainerClose { container_id: 0 });
68            if !source_world_detached {
69                old_world.remove_player_for_world_change(self);
70            }
71            self.set_world(new_world.clone());
72        } else if !source_world_detached {
73            old_world.chunk_map.remove_player(self);
74        }
75
76        self.set_client_loaded(false);
77        self.set_velocity(DVec3::ZERO);
78        self.movement.lock().reset_last_known_client_movement();
79        self.set_on_ground(false);
80        self.reset_entity_state();
81        *self.block_breaking.lock() = BlockBreakingManager::new();
82
83        restore_state();
84
85        if reason != ResetReason::InitialJoin {
86            // 0x01 = keep attributes, 0x02 = keep entity data
87            let data_kept = reason.respawn_data_kept();
88
89            self.send_packet(CRespawn {
90                dimension_type: new_world.dimension_type.id() as i32,
91                dimension_name: new_world.key.clone(),
92                hashed_seed: new_world.obfuscated_seed(),
93                gamemode: self.game_mode() as u8,
94                previous_gamemode: nullable_game_mode_id(self.previous_game_mode()),
95                is_debug: false,
96                is_flat: new_world.is_flat,
97                has_death_location: false,
98                death_dimension_name: None,
99                death_location: None,
100                portal_cooldown_ticks: self.portal_cooldown(),
101                sea_level: new_world.sea_level,
102                data_kept,
103            });
104        }
105    }

145    #[must_use]
146    #[expect(
147        clippy::too_many_arguments,
148        reason = "packet-relative teleports must keep resolved and protocol values separate"
149    )]
150    pub(crate) fn spawn_with_velocity_packet(
151        self: &Arc<Self>,
152        position: DVec3,
153        rotation: (f32, f32),
154        velocity: DVec3,
155        reason: ResetReason,
156        packet_position: DVec3,
157        packet_rotation: (f32, f32),
158        packet_velocity: DVec3,
159        relatives: RelativeMovement,
160    ) -> bool {
161        let world = self.get_world();
162
163        // Set position and rotation
164        self.base.set_position_local(position);
165        self.set_rotation(rotation);
166        self.set_old_position_to_current();
167        self.movement.lock().reset_for_position_sync(position);
168
169        // Teleport sync (sends CPlayerPosition, sets awaiting_teleport for ack)
170        if let Err(error) = self.teleport_with_velocity_packet(
171            position,
172            velocity,
173            rotation,
174            packet_position,
175            packet_velocity,
176            packet_rotation,
177            relatives,
178        ) {
179            log::error!(
180                "failed to synchronize player {} spawn position: {error}",
181                self.id()
182            );
183        }
184        self.reset_flying_ticks();
185
186        self.send_spawn_state_packets(&world);
187
188        // Force health/xp resync on next tick
189        self.reset_sent_info();
190
191        // Resend client context that is not fully covered by CLogin/CRespawn.
192        self.server().resend_player_context(self);
193        self.send_active_effects_for_self();
194
195        // Add to world / re-enter chunk tracking
196        match reason {
197            ResetReason::InitialJoin | ResetReason::WorldChange => {
198                if reason == ResetReason::WorldChange {
199                    log::info!(
200                        "Player {} changed world to {}",
201                        self.gameprofile.name,
202                        world.key
203                    );
204                }
205                world.add_player(self.clone(), reason)
206            }
207            ResetReason::Respawn | ResetReason::EndCredits => {
208                if world.players.get_by_entity_id(self.id()).is_none() {
209                    return world.add_respawned_player(self.clone());
210                }
211
212                // Same world — re-enter chunk tracking
213                world.chunk_map.remove_player(self);
214                world.player_area_map.remove_by_entity_id(self.id());
215                world.entity_tracker().on_player_leave(self);
216
217                self.send_packet(CGameEvent {
218                    event: GameEventType::LevelChunksLoadStart,
219                    data: 0.0,
220                });
221                world.register_respawned_player_entity(self);
222                true
223            }
224        }
225    }

227    fn send_spawn_state_packets(&self, world: &World) {
228        self.send_abilities();
229        self.publish_client_options();
230        self.resend_attributes();
231        self.resend_entity_data();
232        self.rebroadcast_game_mode();
233        self.send_client_enforced_game_rules(world);
234        self.send_packet(CSetHeldSlot {
235            slot: i32::from(self.inventory.lock().get_selected_slot()),
236        });
237        self.send_time_sync(world);
238        self.send_packet(world.initialize_border_packet());
239        self.send_default_spawn_position(world);
240        self.send_weather_sync(world);
241    }

243    /// Hands the player their whole attribute map again.
244    ///
245    /// Vanilla has no equivalent call because it does not need one: a respawn
246    /// builds a fresh `ServerPlayer`, and `restoreFrom`'s `assignBaseValues`
247    /// dirties every attribute on that new map, so `ServerEntity` sends the lot
248    /// on the next tick. Foton reuses the player and its attribute map, while
249    /// the respawn packet still tells the client to throw its copy away -- so
250    /// any attribute whose base value differs from the client's default and
251    /// which happens not to change again is lost on the client for good. Sent
252    /// on every spawn path rather than only on respawn, because a world change
253    /// keeps its attributes and a resend there costs one packet.
254    fn resend_attributes(&self) {
255        let snapshots = self.living_base.attributes().lock().syncable_snapshots();
256        if snapshots.is_empty() {
257            return;
258        }
259        self.send_packet(CUpdateAttributes::new(self.id(), snapshots));
260    }
261
262    /// Hands the player their whole synchronized entity data again.
263    ///
264    /// The same story as the attributes above, and from the same bit: a
265    /// `CRespawn` whose `data_kept` lacks 0x02 tells the client to throw its
266    /// copy of this player's metadata away, which is what vanilla wants
267    /// because its replacement `ServerPlayer` carries a fresh
268    /// `SynchedEntityData` whose every non-default entry is dirty. Foton reuses
269    /// the entry map, and an entry only travels when its value *changes*, so a
270    /// field that already holds the right number goes quiet forever: skin
271    /// parts, main hand, score, a parrot still on a shoulder. `pack_all` is
272    /// exactly the "every non-default value" set the vanilla replacement would
273    /// have sent. An initial join needs it too -- no `CRespawn` there, but no
274    /// tracker either, because a player never tracks themselves.
275    fn resend_entity_data(&self) {
276        let values = self.pack_all_entity_data();
277        // `pack_all` reports every non-default value but, unlike `pack_dirty`,
278        // consumes nothing. Anything that changed while the player was being
279        // placed is in `values` *and* still dirty, so the tracker would send it
280        // again on the next tick -- and vanilla only ever sends a player their
281        // own metadata once per change, through `packDirty`. The audible case
282        // is reconnecting mid-glide: the shared flags arrived twice a tick
283        // apart and the client started two elytra sound loops.
284        drop(self.pack_dirty_entity_data());
285        if values.is_empty() {
286            return;
287        }
288        self.send_packet(CSetEntityData::new(self.id(), values));
289    }

291    /// Tells the arriving player which rules their new world makes them keep.
292    ///
293    /// Vanilla parity: the `reducedDebugInfo`, `showDeathScreen` and
294    /// `doLimitedCrafting` fields of `ClientboundLoginPacket`. Vanilla can
295    /// afford to send those once and never again because its game rules belong
296    /// to the server. Foton's belong to a world, `CRespawn` carries none of the
297    /// three, and the client enforces all three by itself -- so a player who
298    /// walked from a world that allows free crafting into one that does not
299    /// went on crafting freely until they reconnected.
300    fn send_client_enforced_game_rules(&self, world: &World) {
301        self.send_packet(CGameEvent {
302            event: GameEventType::LimitedCrafting,
303            data: game_event_flag(world.get_game_rule(&LIMITED_CRAFTING)),
304        });
305        self.send_packet(CGameEvent {
306            event: GameEventType::ImmediateRespawn,
307            data: game_event_flag(world.get_game_rule(&IMMEDIATE_RESPAWN)),
308        });
309        self.send_packet(CEntityEvent {
310            entity_id: self.id(),
311            event: if world.get_game_rule(&REDUCED_DEBUG_INFO) {
312                EntityStatus::ReducedDebugInfo
313            } else {
314                EntityStatus::FullDebugInfo
315            },
316        });
317    }

319    /// Tells everyone else what game mode this player is now in.
320    ///
321    /// Vanilla parity: the `broadcastAll` of
322    /// `ServerPlayerGameMode.setGameModeForPlayer`. Foton keeps the game mode
323    /// per domain and restores it with a plain setter, so a player who was
324    /// creative in one domain and survival in another switched between them
325    /// without anyone's tab list hearing about it.
326    fn rebroadcast_game_mode(&self) {
327        self.server()
328            .broadcast_to_online(CPlayerInfoUpdate::update_game_mode(
329                self.gameprofile.id,
330                self.game_mode() as i32,
331            ));
332    }

338    fn send_default_spawn_position(&self, world: &World) {
339        if let Some(server) = self.server.upgrade() {
340            match server.respawn_data_for_domain(world.domain()) {
341                Ok(respawn_data) => {
342                    self.send_packet(CSetDefaultSpawnPosition {
343                        global_pos: respawn_data.global_pos,
344                        yaw: respawn_data.yaw,
345                        pitch: respawn_data.pitch,
346                    });
347                }
348                Err(error) => {
349                    log::error!(
350                        "Failed to send default spawn position to player {}: {error}",
351                        self.gameprofile.name
352                    );
353                }
354            }
355        }
356    }

358    fn send_weather_sync(&self, world: &World) {
359        if !world.can_have_weather() || !world.is_raining() {
360            return;
361        }
362
363        let (rain_level, thunder_level) = {
364            let weather = world.weather.lock();
365            (weather.rain_level, weather.thunder_level)
366        };
367
368        self.send_packet(CGameEvent {
369            event: GameEventType::StartRaining,
370            data: 0.0,
371        });
372        self.send_packet(CGameEvent {
373            event: GameEventType::RainLevelChange,
374            data: rain_level,
375        });
376        self.send_packet(CGameEvent {
377            event: GameEventType::ThunderLevelChange,
378            data: thunder_level,
379        });
380    }

400                MobEffectSyncChange::Update {

481    /// Sending the full set on join must consume the dirty flags with it.
482    ///
483    /// Vanilla only ever hands a player their own metadata through `packDirty`
484    /// (`ServerEntity.sendToTrackingPlayersAndSelf`), which clears as it packs.
485    /// `pack_all` does not, so the shared flags went out on the join tick and
486    /// again from the tracker on the next one -- two
487    /// `ClientboundSetEntityDataPacket`s a tick apart. The client starts an
488    /// `ElytraOnPlayerSoundInstance` on each, which is the doubled, slightly
489    /// offset elytra noise the report describes.
490    #[test]
491    fn resending_entity_data_on_join_consumes_the_dirty_flags() {
492        let (_world, player) = gliding_player("join_consumes_dirty_entity_data");
493
494        player.resend_entity_data();
495
496        assert!(
497            player.pack_dirty_entity_data().is_none(),
498            "the join already sent these values; leaving them dirty makes the \
499             tracker send them a second time on the next tick"
500        );
501    }
502
503    /// The guard for the test above: without the join, the flags really are
504    /// dirty, so the assertion there is not passing for free.
505    #[test]
506    fn starting_to_glide_marks_the_shared_flags_dirty() {
507        let (_world, player) = gliding_player("gliding_marks_data_dirty");
508
509        assert!(
510            player.pack_dirty_entity_data().is_some(),
511            "fall flying has to reach the client somehow"

513    }
