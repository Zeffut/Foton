33/// Persistent player data saved by Foton's storage backend.

35/// This is Foton's runtime save snapshot. Vanilla import/export should live outside
36/// server runtime storage so compatibility logic does not constrain the native format.

43    pub motion: [f64; 3],
44
45    /// Rotation (yaw, pitch) in degrees.
46    pub rotation: [f32; 2],
47
48    /// Whether the player is on the ground.
49    pub on_ground: bool,
50
51    /// Whether the player is elytra gliding.
52    pub fall_flying: bool,
53
54    /// Vanilla `remainingFireTicks`.
55    pub remaining_fire_ticks: i32,
56
57    /// Synchronized vanilla `TicksFrozen`.
58    pub ticks_frozen: i32,
59
60    /// Vanilla `isInPowderSnow`.
61    pub is_in_powder_snow: bool,
62
63    /// Vanilla `wasInPowderSnow`.
64    pub was_in_powder_snow: bool,
65
66    /// Vanilla `hasVisualFire`.
67    pub has_visual_fire: bool,
68
69    /// Current health points.
70    pub health: f32,
71
72    /// Current game mode (0=survival, 1=creative, 2=adventure, 3=spectator).
73    pub game_mode: i32,

94    pub world: String,
95
96    /// Current food level (0–20, default 20).
97    pub food_level: i32,
98
99    /// Food saturation level (0.0–`food_level`, default 5.0).
100    pub food_saturation_level: f32,

102    /// Accumulated food exhaustion (0.0–40.0, default 0.0).
103    pub food_exhaustion_level: f32,
104
105    /// Internal tick timer for regen/starvation (default 0).
106    pub food_tick_timer: i32,
107
108    /// Data version for format migrations.
109    pub data_version: i32,
110
111    /// Current experience level
112    pub experience_level: i32,
113
114    /// Vanilla progress toward the next experience level.
115    pub experience_progress: f32,
116
117    /// Vanilla `Player.totalExperience`, updated by point grants but independent of level/progress.
118    pub experience_total: i32,
119
120    /// Vanilla `XpSeed`, the seed an enchanting table draws its three offers
121    /// from. Without it a relog silently reshuffles what the table shows.
122    pub enchantment_seed: i32,
123
124    /// Vanilla death-screen score. Point grants change it with Java `int` wrapping.
125    pub score: i32,
126
127    /// Vanilla `ServerPlayer.seenCredits`.
128    pub seen_credits: bool,

130    /// Vanilla `ServerPlayer.wardenSpawnTracker`, as its three saved counters.
131    ///
132    /// The count is what a shrieker escalates and what a warden costs, so losing it on a
133    /// reload would mean the deep dark forgave every player every log-out.
134    pub warden_spawn_tracker: [i32; 3],
135
136    /// Vanilla one-player root vehicle tree stored with the player instead of chunk data.
137    pub root_vehicle: Option<PersistentRootVehicle>,
138
139    /// Vanilla per-player respawn configuration set by beds and respawn anchors.
140    pub respawn_config: Option<PlayerRespawnConfig>,
141
142    /// Vanilla in-flight ender pearls stored with the player (`ServerPlayer.enderPearls`).
143    pub ender_pearls: Vec<PersistentEnderPearl>,
144
145    /// Advancement progress, one entry per advancement with anything to save.
146    ///
147    /// Vanilla writes this to its own `advancements/<uuid>.json` because it has
148    /// one world; Foton scopes it to the domain, beside the experience and the
149    /// inventory it was earned with.
150    pub advancements: Vec<PersistentAdvancement>,
151
152    /// Statistics, keyed by the two registry keys a statistic is made of.
153    ///
154    /// The keys travel rather than the registry ids: an id is only meaningful
155    /// against the registry that produced it, and a save has to survive a
156    /// registry growing a new entry in the middle.
157    pub statistics: Vec<PersistentStatistic>,
158
159    /// The half of the save every living entity has, as a written NBT compound.
160    ///
161    /// Vanilla parity: the `LivingEntity.addAdditionalSaveData` a player's own
162    /// `addAdditionalSaveData` sits on top of. Absorption, potion effects and
163    /// attribute modifiers live nowhere else in this file, so without it a
164    /// player logged back in with the shield, the effects and the modifiers
165    /// gone. The keys it shares with a field above -- health, equipment -- are
166    /// applied first and then overwritten by that field, which is the order
167    /// vanilla's `super`-first read runs in.
168    pub living_nbt: Vec<u8>,
169}

267impl PersistentPlayerData {
268    /// Extracts persistent data from a live player.
269    #[must_use]
270    pub fn from_player(player: &Player) -> Self {
271        // Before the inventory lock below, not after: a player's equipment is a
272        // view onto its own inventory, so `save_living` takes that same lock to
273        // read the worn items and would hang waiting on this function.
274        let mut living = NbtCompound::new();
275        player.save_living(&mut living);
276        let mut living_nbt = Vec::new();
277        living.write(&mut living_nbt);
278
279        let pos = player.position();
280        let (yaw, pitch) = player.rotation();
281        let delta = player.velocity();
282        let on_ground = player.on_ground();
283        let fall_flying = player.is_fall_flying();
284        let fire_freeze = player.fire_freeze_state();
285        let abilities = player.abilities.lock();
286        let inventory = player.inventory.lock();
287        let food_data = player.food_data.lock();
288
289        // Collect non-empty inventory slots
290        let mut slots = Vec::new();
291        // Main inventory (0-35) and equipment (36-42)
292        for slot in 0..PlayerInventory::CONTAINER_SIZE {
293            let item = inventory.get_item(slot);
294            if !item.is_empty() {
295                slots.push(PersistentSlot {
296                    slot: slot as i8,
297                    item: item.clone(),
298                });
299            }
300        }
301
302        let mut ender_slots = Vec::new();
303        {
304            let ender_chest = player.ender_chest.lock();
305            for slot in 0..ENDER_CHEST_SLOTS {
306                let item = ender_chest.get_item(slot);
307                if !item.is_empty() {
308                    #[expect(
309                        clippy::cast_possible_truncation,
310                        reason = "there are twenty-seven slots"
311                    )]
312                    ender_slots.push(PersistentSlot {
313                        slot: slot as i8,
314                        item: item.clone(),
315                    });
316                }
317            }
318        }
319
320        let (experience_level, experience_progress, experience_total) = {
321            let lock = player.experience.lock();
322            (lock.level(), lock.progress(), lock.total_points())
323        };
324        let score = player.score();
325        let advancements = Self::advancements_from_player(player);
326        let statistics = Self::statistics_from_player(player);
327        let root_vehicle = Self::root_vehicle_from_player(player)
328            .or_else(|| player.pending_root_vehicle_for_current_world());
329        let ender_pearls = Self::ender_pearls_from_player(player);
330
331        Self {
332            pos: [pos.x, pos.y, pos.z],
333            motion: [delta.x, delta.y, delta.z],
334            rotation: [yaw, pitch],
335            on_ground,
336            fall_flying,
337            remaining_fire_ticks: fire_freeze.remaining_fire_ticks(),
338            ticks_frozen: fire_freeze.ticks_frozen(),
339            is_in_powder_snow: fire_freeze.is_in_powder_snow(),
340            was_in_powder_snow: fire_freeze.was_in_powder_snow(),
341            has_visual_fire: fire_freeze.has_visual_fire(),
342            health: player.get_health(),
343            game_mode: player.game_mode() as i32,
344            prev_game_mode: player
345                .previous_game_mode()
346                .map(|game_mode| game_mode as i32),
347            abilities: PersistentAbilities {
348                invulnerable: abilities.invulnerable,
349                flying: abilities.flying,
350                may_fly: abilities.may_fly,
351                instabuild: abilities.instabuild,
352                may_build: abilities.may_build,
353                flying_speed: abilities.flying_speed,
354                walking_speed: abilities.walking_speed,
355            },
356            inventory: slots,
357            ender_items: ender_slots,
358            selected_slot: i32::from(inventory.get_selected_slot()),
359            world: player.get_world().key.to_string(),
360            food_level: food_data.food_level,
361            food_saturation_level: food_data.saturation_level,
362            food_exhaustion_level: food_data.exhaustion_level,
363            food_tick_timer: food_data.tick_timer,
364            data_version: PLAYER_DATA_VERSION,
365            experience_level,
366            experience_progress,
367            experience_total,
368            enchantment_seed: player.enchantment_seed(),
369            score,
370            seen_credits: player.has_seen_credits(),
371            warden_spawn_tracker: warden_spawn_tracker_fields(player.warden_spawn_tracker()),
372            root_vehicle,
373            respawn_config: player.respawn_config(),
374
375            ender_pearls,
376            advancements,
377            statistics,
378            living_nbt,
379        }
380    }
381
382    /// Restores the shared living half before the player's own fields.
383    ///
384    /// The compound is empty for a player who has never been saved, and
385    /// [`LivingEntity::load_living`] reads an absent key as a *default* rather
386    /// than as "leave it alone" -- so handing it nothing would clear the
387    /// effects a domain switch is carrying over instead of preserving them.
388    fn apply_living_nbt(&self, player: &Player) {
389        if self.living_nbt.is_empty() {
390            return;
391        }
392        let Ok(living) = read_borrowed_compound(&mut Cursor::new(&self.living_nbt)) else {
393            tracing::warn!(
394                uuid = ?player.uuid(),
395                "Failed to parse saved player living state, leaving it at its defaults"
396            );
397            return;
398        };
399        player.load_living((&living).into());
400    }
401
402    /// Snapshots the player's advancement progress for persistence.
403    fn advancements_from_player(player: &Player) -> Vec<PersistentAdvancement> {
404        player
405            .saved_advancements()
406            .into_iter()
407            .map(|(key, criteria)| PersistentAdvancement {
408                key: key.to_string(),
409                criteria: criteria
410                    .into_iter()
411                    .map(|(name, obtained_epoch_millis)| PersistentCriterion {
412                        name: name.to_owned(),
413                        obtained_epoch_millis,
414                    })
415                    .collect(),
416            })
417            .collect()
418    }
419
420    /// Restores the saved advancement progress.
421    ///
422    /// A key that will not parse any more is dropped with a warning rather than
423    /// failing the load, which is what
424    /// [`crate::advancement::PlayerAdvancements::load`] already does for an
425    /// advancement or a criterion the tree no longer declares.
426    fn apply_advancements(&self, player: &Player) {
427        let restored = self.advancements.iter().filter_map(|advancement| {
428            let key = match Identifier::from_str(&advancement.key) {
429                Ok(key) => key,
430                Err(error) => {
431                    tracing::warn!(
432                        key = advancement.key,
433                        %error,
434                        "Ignoring saved advancement progress under an unparsable key"
435                    );
436                    return None;
437                }
438            };
439            let criteria = advancement
440                .criteria
441                .iter()
442                .map(|criterion| (criterion.name.clone(), criterion.obtained_epoch_millis))
443                .collect();
444            Some((key, criteria))
445        });
446        player.load_advancements(restored);
447    }
448
449    /// Snapshots the player's statistics for persistence.
450    fn statistics_from_player(player: &Player) -> Vec<PersistentStatistic> {
451        player
452            .saved_statistics()
453            .into_iter()
454            .filter_map(|(stat, count)| {
455                let stat_type = REGISTRY.stat_types.by_id(stat.stat_type)?;
456                let value = stat_value_key(stat_type.value_registry, stat.value)?;
457                Some(PersistentStatistic {
458                    stat_type: stat_type.key.to_string(),
459                    value: value.to_string(),
460                    count,
461                })
462            })
463            .collect()
464    }
465
466    /// Restores the saved statistics.
467    ///
468    /// A statistic naming something no longer registered is dropped rather
469    /// than failing the load, which is what vanilla's codec does with an
470    /// unknown key.
471    fn apply_statistics(&self, player: &Player) {
472        let restored = self.statistics.iter().filter_map(|entry| {
473            let stat_type_key = Identifier::from_str(&entry.stat_type).ok()?;
474            let stat_type = REGISTRY.stat_types.by_key(&stat_type_key)?;
475            let value_key = Identifier::from_str(&entry.value).ok()?;
476            let value = stat_value_id(stat_type.value_registry, &value_key)?;
477            Some((
478                Stat {
479                    stat_type: stat_type.try_id()?,
480                    value,
481                },
482                entry.count,
483            ))
484        });
485        player.load_statistics(restored);
486    }
487
488    /// Snapshots the player's live in-flight ender pearls for persistence.
489    fn ender_pearls_from_player(player: &Player) -> Vec<PersistentEnderPearl> {
490        let mut seen = FxHashSet::default();
491        let mut pearls = player
492            .ender_pearls()
493            .iter()
494            .filter_map(|pearl| {
495                let world = pearl.level()?.key.to_string();
496                let entity = ChunkStorage::entity_tree_to_persistent(pearl)?;
497                seen.insert(entity.uuid);
498                Some(PersistentEnderPearl { world, entity })
499            })
500            .collect::<Vec<_>>();
501        pearls.extend(
502            player
503                .pending_ender_pearls()
504                .into_iter()
505                .filter(|pearl| seen.insert(pearl.entity.uuid)),
506        );
507        pearls
508    }
509
510    fn root_vehicle_from_player(player: &Player) -> Option<PersistentRootVehicle> {
511        let vehicle = player.vehicle()?;
512        let root_vehicle = player.root_vehicle()?;
513        if root_vehicle.id() == player.id() || !root_vehicle.has_exactly_one_player_passenger() {
514            return None;
515        }
516
517        let entity = ChunkStorage::entity_tree_to_persistent(&root_vehicle)?;
518        Some(PersistentRootVehicle {
519            attach: *vehicle.uuid().as_bytes(),
520            entity,
521        })
522    }
523}

525impl Player {
526    /// Resets domain-scoped gameplay data to the defaults used for a new player.
527    pub(crate) fn reset_domain_data_for_first_visit(&self) {
528        use glam::DVec3;
529
530        self.set_velocity(DVec3::ZERO);
531        self.set_on_ground(false);
532        self.set_fall_flying(false);
533        self.base()
534            .set_fire_freeze_state(EntityFireFreezeState::new());
535        self.sync_base_fire_freeze_entity_data();
536        self.set_health(self.get_max_health());
537        *self.abilities.lock() = Abilities::default();
538        *self.inventory.lock() = PlayerInventory::new();
539        *self.food_data.lock() = FoodData::new();
540
541        let mut experience = Experience::default();
542        experience.dirty = true;
543        *self.experience.lock() = experience;
544
545        self.set_score(0);
546        self.set_seen_credits(false);
547        self.reset_advancements();
548        self.load_statistics([]);
549        // Vanilla parity: the `this.wardenSpawnTracker.reset()` of `ServerPlayer.reset`,
550        // which is what makes dying to a warden clear the way to the next one.
551        let mut tracker = self.warden_spawn_tracker();
552        tracker.reset();
553        self.set_warden_spawn_tracker(tracker);
554    }
555}

557/// The registry key one statistic value stands for.
558///
559/// Vanilla dispatches on the stat type's own registry; Foton names the four
560/// registries the vanilla stat types range over, so this is that dispatch.
561fn stat_value_key(registry: StatValueRegistry, id: usize) -> Option<&'static Identifier> {
562    match registry {
563        StatValueRegistry::Block => REGISTRY.blocks.by_id(id).map(|block| &block.key),
564        StatValueRegistry::Item => REGISTRY.items.by_id(id).map(|item| &item.key),
565        StatValueRegistry::EntityType => REGISTRY
566            .entity_types
567            .by_id(id)
568            .map(|entity_type| &entity_type.key),
569        StatValueRegistry::CustomStat => REGISTRY.custom_stats.by_id(id).map(|stat| &stat.key),
570    }
571}
572
573/// The id a statistic value's key resolves to.
574fn stat_value_id(registry: StatValueRegistry, key: &Identifier) -> Option<usize> {
575    match registry {
576        StatValueRegistry::Block => REGISTRY.blocks.id_from_key(key),
577        StatValueRegistry::Item => REGISTRY.items.id_from_key(key),
578        StatValueRegistry::EntityType => REGISTRY.entity_types.id_from_key(key),
579        StatValueRegistry::CustomStat => REGISTRY.custom_stats.id_from_key(key),
580    }
581}

634impl PersistentPlayerData {
635    /// Applies the saved data to a player.
636    ///
637    /// This restores position, rotation, inventory, abilities, etc.
638    pub fn apply_to_player(&self, player: &Player) {
639        self.apply_to_player_inner(player, true);
640    }
641
642    /// Applies saved gameplay state without restoring world-local location data.
643    ///
644    /// Used when the saved world is unavailable or differs from an explicitly
645    /// selected world, which must use the target spawn instead.
646    pub fn apply_to_player_without_location(&self, player: &Player) {
647        self.apply_to_player_inner(player, false);
648    }
649
650    fn apply_to_player_inner(&self, player: &Player, restore_location: bool) {
651        use glam::DVec3;
652
653        self.apply_living_nbt(player);
654
655        if restore_location {
656            // Position
657            player
658                .base()
659                .set_position_local(DVec3::new(self.pos[0], self.pos[1], self.pos[2]));
660
661            // Rotation
662            player.set_rotation((self.rotation[0], self.rotation[1]));
663
664            // Motion/velocity
665            player.set_velocity(DVec3::new(self.motion[0], self.motion[1], self.motion[2]));
666
667            // Ground state
668            player.set_fall_flying(self.fall_flying);
669            player.set_on_ground(self.on_ground);
670        }
671
672        player
673            .base()
674            .set_fire_freeze_state(EntityFireFreezeState::from_parts(
675                self.remaining_fire_ticks,
676                self.ticks_frozen,
677                self.is_in_powder_snow,
678                self.was_in_powder_snow,
679                self.has_visual_fire,
680            ));
681        player.sync_base_fire_freeze_entity_data();
682        player.set_respawn_position(self.respawn_config.clone(), false);
683
684        // Health
685        player.set_health(self.health);
686
687        // Game mode
688        player.restore_game_modes(
689            self.game_mode.into(),
690            self.prev_game_mode.map(GameType::from),
691        );
692
693        // Abilities
694        *player.abilities.lock() = self.abilities.clone().into();
695
696        // Inventory
697        {
698            let mut inventory = player.inventory.lock();
699            // Clear existing inventory first
700            for slot in 0..PlayerInventory::CONTAINER_SIZE {
701                inventory.set_item(slot, ItemStack::empty());
702            }
703            // Restore saved items
704            for slot_data in &self.inventory {
705                let slot_index = slot_data.slot as usize;
706                if slot_index < PlayerInventory::CONTAINER_SIZE {
707                    inventory.set_item(slot_index, slot_data.item.clone());
708                }
709            }
710            // Restore selected slot
711            let selected = self.selected_slot.clamp(0, 8) as u8;
712            inventory.set_selected_slot(selected);
713        }
714
715        // Ender chest
716        {
717            let mut ender_chest = player.ender_chest.lock();
718            for slot in 0..ENDER_CHEST_SLOTS {
719                ender_chest.set_item(slot, ItemStack::empty());
720            }
721            for slot_data in &self.ender_items {
722                let slot_index = slot_data.slot as usize;
723                if slot_index < ENDER_CHEST_SLOTS {
724                    ender_chest.set_item(slot_index, slot_data.item.clone());
725                }
726            }
727        }
728
729        // Food data
730        {
731            let mut food = player.food_data.lock();
732            food.food_level = self.food_level;
733            food.saturation_level = self.food_saturation_level;
734            food.exhaustion_level = self.food_exhaustion_level;
735            food.tick_timer = self.food_tick_timer;
736        }
737
738        {
739            let mut experience = player.experience.lock();
740            *experience = Experience::from_parts(
741                self.experience_level,
742                self.experience_progress,
743                self.experience_total,
744            );
745        }
746        self.apply_advancements(player);
747        self.apply_statistics(player);
748        player.set_enchantment_seed(self.enchantment_seed);
749        player.set_score(self.score);
750        player.set_seen_credits(self.seen_credits);
751        let [ticks_since_last_warning, warning_level, cooldown_ticks] = self.warden_spawn_tracker;
752        player.set_warden_spawn_tracker(WardenSpawnTracker::new(
753            ticks_since_last_warning,
754            warning_level,
755            cooldown_ticks,
756        ));
757    }
