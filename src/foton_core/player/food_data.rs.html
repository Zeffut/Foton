102/// Computes the absolute saturation value from a nutrition count and a modifier.
103#[must_use]
104pub fn saturation_by_modifier(nutrition: i32, modifier: f32) -> f32 {
105    nutrition as f32 * modifier * 2.0
106}

108/// Tracks a player's hunger, saturation, and exhaustion state.
109/// One instance is stored per player behind a `SyncMutex`.
110#[derive(Debug, Clone)]
111pub struct FoodData {
112    /// Current food level (0–20). Displayed as the hunger bar on the client.
113    pub food_level: i32,
114    /// Saturation buffer — consumed before the food level drops.
115    pub saturation_level: f32,
116    /// Accumulated exhaustion from actions (sprinting, jumping, damage, …).
117    pub exhaustion_level: f32,
118    /// Internal tick counter shared between regeneration and starvation logic.
119    pub tick_timer: i32,
120}
121
122impl Default for FoodData {
123    fn default() -> Self {
124        Self::new()
125    }
126}
127
128impl FoodData {
129    /// Creates a new `FoodData` with default values (full hunger bar).
130    #[must_use]
131    pub const fn new() -> Self {
132        Self {
133            food_level: DEFAULT_FOOD_LEVEL,
134            saturation_level: DEFAULT_SATURATION,
135            exhaustion_level: 0.0,
136            tick_timer: 0,
137        }
138    }
139
140    /// Returns `true` if the player's food level is below maximum and they
141    /// could benefit from eating.
142    #[must_use]
143    pub const fn needs_food(&self) -> bool {
144        self.food_level < MAX_FOOD_LEVEL
145    }
146
147    /// Returns `true` if the player has enough food to perform exhaustive
148    /// maneuvers (for example sprinting).
149    #[must_use]
150    pub const fn has_enough_food(&self) -> bool {
151        self.food_level > SPRINT_LEVEL
152    }
153
154    /// Adds exhaustion from an action (sprinting, jumping, taking damage, …).
155    /// The value is clamped so `exhaustion_level` never exceeds 40.
156    pub fn add_exhaustion(&mut self, amount: f32) {
157        self.exhaustion_level = (self.exhaustion_level + amount).min(MAX_EXHAUSTION);
158    }
159
160    /// Applies food and saturation
161    fn add(&mut self, food: i32, saturation: f32) {
162        self.food_level = (self.food_level + food).clamp(0, MAX_FOOD_LEVEL);
163        self.saturation_level =
164            (self.saturation_level + saturation).clamp(0.0, self.food_level as f32);
165    }
166
167    /// Applies the nutrition from eating food, given a **saturation modifier**.
168    pub fn eat(&mut self, nutrition: i32, saturation_modifier: f32) {
169        self.add(
170            nutrition,
171            saturation_by_modifier(nutrition, saturation_modifier),
172        );
173    }
174
175    /// Applies a food item's nutrition and saturation directly.
176    ///
177    /// Vanilla parity: `FoodData.eat(FoodProperties)`. Unlike [`Self::eat`], the
178    /// saturation value is applied as-is instead of being derived from a modifier.
179    pub fn eat_food(&mut self, nutrition: i32, saturation: f32) {
180        self.add(nutrition, saturation);
181    }
182
183    /// Runs one tick of the hunger system.
184    /// Returns a [`FoodTickResult`] describing what happened this tick so the
185    /// caller (`Player::tick`) can apply healing or starvation damage
186    #[must_use]
187    pub fn tick(
188        &mut self,
189        difficulty: Difficulty,
190        natural_regen: bool,
191        current_health: f32,
192        max_health: f32,
193    ) -> FoodTickResult {
194        if self.exhaustion_level > EXHAUSTION_DROP {
195            self.exhaustion_level -= EXHAUSTION_DROP;
196
197            if self.saturation_level > 0.0 {
198                self.saturation_level = (self.saturation_level - 1.0).max(0.0);
199            } else if difficulty != Difficulty::Peaceful {
200                self.food_level = (self.food_level - 1).max(0);
201            }
202        }
203
204        let food = self.food_level;
205        let is_hurt = current_health > 0.0 && current_health < max_health;
206
207        // Fast regen
208        if natural_regen && self.saturation_level > 0.0 && is_hurt && food >= MAX_FOOD_LEVEL {
209            self.tick_timer += 1;
210
211            if self.tick_timer >= HEALTH_TICK_COUNT_SATURATED {
212                let saturation_spent = self.saturation_level.min(EXHAUSTION_HEAL);
213                let heal_amount = saturation_spent / EXHAUSTION_HEAL;
214                self.tick_timer = 0;
215                return FoodTickResult::Heal {
216                    amount: heal_amount,
217                    exhaustion: saturation_spent,
218                };
219            }
220        }
221        // Slow regen
222        else if natural_regen && food >= HEAL_LEVEL && is_hurt {
223            self.tick_timer += 1;
224
225            if self.tick_timer >= HEALTH_TICK_COUNT {
226                self.tick_timer = 0;
227                return FoodTickResult::Heal {
228                    amount: 1.0,
229                    exhaustion: EXHAUSTION_HEAL,
230                };
231            }
232        }
233        // Starving
234        else if food <= 0 {
235            self.tick_timer += 1;
236
237            if self.tick_timer >= HEALTH_TICK_COUNT {
238                self.tick_timer = 0;
239
240                if current_health > 10.0
241                    || difficulty == Difficulty::Hard
242                    || (current_health > 1.0 && difficulty == Difficulty::Normal)
243                {
244                    return FoodTickResult::Starve;
245                }
246            }
247        } else {
248            self.tick_timer = 0;
249        }
250
251        FoodTickResult::None
252    }
253}

255/// Describes what the hunger tick determined should happen this tick
256#[derive(Debug, Clone, Copy, PartialEq)]
257pub enum FoodTickResult {
258    /// Nothing happened this tick
259    None,
260    /// The player should be healed by `amount` HP and `exhaustion` should be added back
261    Heal {
262        /// Health points to restore
263        amount: f32,
264        /// Exhaustion to add as cost of this regeneration
265        exhaustion: f32,
266    },
267    /// The player should take 1 point of starvation damage
268    Starve,
269}

271impl Player {
272    /// Ticks food/hunger regeneration and starvation.
273    pub(super) fn tick_regeneration(&self) {
274        let world = self.get_world();
275        let difficulty = world.level_data.read().data().difficulty;
276        let natural_regen = world.get_game_rule(&NATURAL_HEALTH_REGENERATION);
277        let tick = self.tick_count();
278
279        if difficulty == Difficulty::Peaceful && natural_regen {
280            if tick % 20 == 0 {
281                if self.is_hurt() {
282                    self.heal_with_regain_event(1.0);
283                }
284
285                let mut food = self.food_data.lock();
286                if food.saturation_level < food_constants::MAX_SATURATION {
287                    food.saturation_level += 1.0;
288                }
289            }
290
291            if tick % 10 == 0 {
292                let mut food = self.food_data.lock();
293                let previous = food.food_level;
294                if food.needs_food() {
295                    food.food_level += 1;
296                }
297                let changed = food.food_level != previous;
298                let current = food.food_level;
299                drop(food);
300                if changed {
301                    match self.allow_food_level_change(current) {
302                        Some(level) => self.food_data.lock().food_level = level.clamp(0, 20),
303                        None => self.food_data.lock().food_level = previous,
304                    }
305                }
306            }
307        }
308
309        let current_health = self.get_health();
310        let max_health = self.get_max_health();
311
312        let mut food = self.food_data.lock();
313        let previous = food.food_level;
314        let result = food.tick(difficulty, natural_regen, current_health, max_health);
315        let current = food.food_level;
316        drop(food);
317        let applied_level = if current == previous {
318            current
319        } else {
320            let Some(level) = self.allow_food_level_change(current) else {
321                self.food_data.lock().food_level = previous;
322                return;
323            };
324            let level = level.clamp(0, 20);
325            self.food_data.lock().food_level = level;
326            level
327        };
328        let mut food = self.food_data.lock();
329
330        // The Bukkit event is authoritative: changing the level can invalidate
331        // the action the hunger tick computed from the pre-event value.
332        let result = match result {
333            FoodTickResult::Starve if applied_level > 0 => FoodTickResult::None,
334            FoodTickResult::Heal { .. } if applied_level < food_constants::HEAL_LEVEL => {
335                FoodTickResult::None
336            }
337            result => result,
338        };
339
340        match result {
341            FoodTickResult::Heal { amount, exhaustion } => {
342                food.add_exhaustion(exhaustion);
343
344                drop(food);
345                self.heal_with_regain_event(amount);
346            }
347            FoodTickResult::Starve => {
348                drop(food);
349
350                self.hurt(
351                    &world,
352                    &DamageSource::environment(&vanilla_damage_types::STARVE),
353                    1.0,
354                );
355            }
356            FoodTickResult::None => {}
357        }
358    }
359
360    fn heal_with_regain_event(&self, amount: f32) {
361        let mut event = EntityRegainHealthEvent::new(self.gameprofile.id, amount);
362        self.fire_event(&mut event);
363        if !event.is_cancelled() {
364            self.heal(event.amount());
365        }
366    }
367
368    fn allow_food_level_change(&self, level: i32) -> Option<i32> {
369        let mut event = FoodLevelChangeEvent::new(self.gameprofile.id, level);
370        self.fire_event(&mut event);
371        (!event.is_cancelled()).then_some(event.food_level())
372    }
373
374    /// Adds food exhaustion, gated by invulnerability.
375    pub fn cause_food_exhaustion(&self, amount: f32) {
376        if !self.abilities.lock().invulnerable {
377            self.food_data.lock().add_exhaustion(amount);
378        }
379    }
380
381    /// Returns `true` if the player is alive but below max health.
382    pub fn is_hurt(&self) -> bool {
383        let health = self.get_health();
384        health > 0.0 && health < self.get_max_health()
385    }
386}

394    /// A notch apple's saturation reaches a player who is not hungry.
395    ///
396    /// The enchanted golden apple is the food most likely to be eaten on a full
397    /// hunger bar -- `can_always_eat` exists for it -- so it is the one where a
398    /// break anywhere in the chain shows up to a player as "nothing happened".
399    /// This walks the same way the consume path does: the item's own component
400    /// first, then the `eat(FoodProperties)` overload, on a full bar.
401    #[test]
402    fn a_notch_apple_feeds_a_player_who_is_already_full() {
403        let food = vanilla_items::ENCHANTED_GOLDEN_APPLE
404            .components
405            .get_ref(FOOD)
406            .expect("the enchanted golden apple carries a food component");
407
408        // Vanilla parity: `Foods.ENCHANTED_GOLDEN_APPLE`.
409        assert_eq!(food.nutrition(), 4);
410        assert!(
411            (food.saturation() - 9.6).abs() < 1.0e-4,
412            "vanilla gives it 9.6 absolute saturation, got {}",
413            food.saturation()
414        );
415        assert!(
416            food.can_always_eat(),
417            "a notch apple is edible on a full hunger bar"
418        );
419
420        let mut data = FoodData::new();
421        data.food_level = MAX_FOOD_LEVEL;
422        data.saturation_level = 5.0;
423        data.eat_food(food.nutrition(), food.saturation());
424
425        assert_eq!(
426            data.food_level, MAX_FOOD_LEVEL,
427            "a full bar stays full rather than overflowing"
428        );
429        assert!(
430            (data.saturation_level - 14.6).abs() < 1.0e-4,
431            "eating on a full bar must still bank saturation: expected 14.6, got {}",
432            data.saturation_level
433        );
434    }

436    /// `eat_food` must apply the saturation value as-is.
437    ///
438    /// Vanilla has two overloads: `eat(int, float)` derives saturation through
439    /// `saturationByModifier`, while `eat(FoodProperties)` adds it directly. The
440    /// consume path uses the second one. Routing food through the first would
441    /// silently change the saturation of every food in the game, which nothing
442    /// else in the test suite would catch.
443    #[test]
444    fn eat_food_applies_saturation_directly_unlike_the_modifier_overload() {
445        let nutrition = 4;
446        let saturation = 2.4;
447
448        let mut direct = FoodData::new();
449        direct.food_level = 10;
450        direct.saturation_level = 0.0;
451        direct.eat_food(nutrition, saturation);
452
453        assert_eq!(direct.food_level, 14);
454        assert!(
455            (direct.saturation_level - saturation).abs() < f32::EPSILON,
456            "expected the raw saturation {saturation}, got {}",
457            direct.saturation_level
458        );
459
460        // The modifier overload would have produced a different value, which is
461        // exactly the mistake this test exists to catch.
462        let mut via_modifier = FoodData::new();
463        via_modifier.food_level = 10;
464        via_modifier.saturation_level = 0.0;
465        via_modifier.eat(nutrition, saturation);
466
467        assert!(
468            (via_modifier.saturation_level - direct.saturation_level).abs() > f32::EPSILON,
469            "the two overloads must not be interchangeable"
470        );
471    }

534    /// - Easy/Peaceful: stops at 10 HP

536    fn starvation_by_difficulty() {

538            let mut food = FoodData::new();

540            food.saturation_level = 0.0;
541            let mut result = FoodTickResult::None;

545            result
