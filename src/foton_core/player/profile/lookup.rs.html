11const DEFAULT_PROFILE_SERVER: &str =
12    "https://api.minecraftservices.com/minecraft/profile/lookup/name";
13const MAX_PROFILE_LOOKUP_ATTEMPTS: usize = 3;
14const PROFILE_LOOKUP_RETRY_DELAY: Duration = Duration::from_millis(750);
15/// Bounds every attempt so suspended administrative commands always release their ordering barrier.
16const PROFILE_LOOKUP_REQUEST_TIMEOUT: Duration = Duration::from_secs(5);
17const MAX_PROFILE_RESPONSE_BYTES: usize = 64 * 1024;
18
19/// Failure while resolving a player identity through the configured profile service.
20#[derive(Debug, Error)]
21pub enum ProfileLookupError {
22    /// No profile exists for the requested name.
23    #[error("Unknown player {0}")]
24    UnknownPlayer(String),
25    /// The configured profile lookup endpoint is invalid.
26    #[error("Invalid profile server URL configured: {0}")]
27    InvalidProfileServer(String),
28    /// The request failed before a response was received.
29    #[error("Profile lookup failed for {name}: {source}")]
30    Request {
31        /// Requested player name.
32        name: String,
33        /// Transport error.
34        source: reqwest::Error,
35    },
36    /// The service returned an unexpected status.
37    #[error("Profile lookup service returned status {status} for {name}")]
38    ServiceResponse {
39        /// Requested player name.
40        name: String,
41        /// HTTP response status.
42        status: StatusCode,
43    },
44    /// The service returned malformed identity data.
45    #[error("Invalid profile lookup response for {name}: {reason}")]
46    InvalidResponse {
47        /// Requested player name.
48        name: String,
49        /// Response validation failure.
50        reason: String,
51    },
52}
53
54#[derive(Deserialize)]
55struct ProfileLookupResponse {
56    id: String,
57    name: String,
58}

60/// Resolves one online-mode profile through the configured service.
61///
62/// The caller handles local caches, offline mode, and name validation first.
63#[cfg_attr(
64    not(test),
65    expect(
66        clippy::unreachable,
67        reason = "packet dispatch is split by kind before this point, and the player's inventory keeps the concrete type it was created with"
68    )
69)]
70pub async fn lookup_online_profile(
71    client: &reqwest::Client,
72    profile_server: Option<&str>,
73    name: &str,
74) -> Result<KnownPlayer, ProfileLookupError> {
75    let lookup_name = name.to_ascii_lowercase();
76    let url = profile_lookup_url(profile_server, &lookup_name)?;
77    for attempt in 1..=MAX_PROFILE_LOOKUP_ATTEMPTS {
78        let result =
79            lookup_online_profile_once(client, url.as_str(), name, PROFILE_LOOKUP_REQUEST_TIMEOUT)
80                .await;
81        match result {
82            Ok(profile) => return Ok(profile),
83            Err(error @ ProfileLookupError::UnknownPlayer(_)) => return Err(error),
84            Err(error) if attempt == MAX_PROFILE_LOOKUP_ATTEMPTS => return Err(error),
85            Err(_) => sleep(PROFILE_LOOKUP_RETRY_DELAY).await,
86        }
87    }
88    unreachable!("the profile lookup attempt range is non-empty")
89}

91async fn lookup_online_profile_once(
92    client: &reqwest::Client,
93    url: &str,
94    name: &str,
95    request_timeout: Duration,
96) -> Result<KnownPlayer, ProfileLookupError> {
97    let response = client
98        .get(url)
99        .timeout(request_timeout)
100        .send()
101        .await
102        .map_err(|source| ProfileLookupError::Request {
103            name: name.to_owned(),
104            source,
105        })?;
106
107    match response.status() {
108        StatusCode::OK => parse_profile_response(response, name).await,
109        StatusCode::NO_CONTENT | StatusCode::NOT_FOUND => {
110            Err(ProfileLookupError::UnknownPlayer(name.to_owned()))
111        }
112        status => Err(ProfileLookupError::ServiceResponse {
113            name: name.to_owned(),
114            status,
115        }),
116    }
117}

119fn profile_lookup_url(
120    profile_server: Option<&str>,
121    normalized_name: &str,
122) -> Result<Url, ProfileLookupError> {
123    let server = profile_server.unwrap_or(DEFAULT_PROFILE_SERVER);
124    let endpoint = format!("{}/{normalized_name}", server.trim_end_matches('/'));
125    Url::parse(&endpoint).map_err(|_| ProfileLookupError::InvalidProfileServer(endpoint))
126}

128async fn parse_profile_response(
129    mut response: reqwest::Response,
130    requested_name: &str,
131) -> Result<KnownPlayer, ProfileLookupError> {
132    if response
133        .content_length()
134        .is_some_and(|length| length > MAX_PROFILE_RESPONSE_BYTES as u64)
135    {
136        return Err(ProfileLookupError::InvalidResponse {
137            name: requested_name.to_owned(),
138            reason: format!("response exceeds the {MAX_PROFILE_RESPONSE_BYTES}-byte limit"),
139        });
140    }
141
142    let mut body = Vec::new();
143    while let Some(chunk) =
144        response
145            .chunk()
146            .await
147            .map_err(|source| ProfileLookupError::InvalidResponse {
148                name: requested_name.to_owned(),
149                reason: source.to_string(),
150            })?
151    {
152        let remaining = MAX_PROFILE_RESPONSE_BYTES.saturating_sub(body.len());
153        if chunk.len() > remaining {
154            return Err(ProfileLookupError::InvalidResponse {
155                name: requested_name.to_owned(),
156                reason: format!("response exceeds the {MAX_PROFILE_RESPONSE_BYTES}-byte limit"),
157            });
158        }
159        body.extend_from_slice(&chunk);
160    }
161    let profile = serde_json::from_slice::<ProfileLookupResponse>(&body).map_err(|source| {
162        ProfileLookupError::InvalidResponse {
163            name: requested_name.to_owned(),
164            reason: source.to_string(),
165        }
166    })?;
167    let uuid =
168        Uuid::parse_str(&profile.id).map_err(|source| ProfileLookupError::InvalidResponse {
169            name: requested_name.to_owned(),
170            reason: source.to_string(),
171        })?;
172    Ok(KnownPlayer::new(uuid, profile.name))
173}

210    #[tokio::test]
211    async fn rejects_an_oversized_profile_response_from_its_headers() {
212        let listener = TcpListener::bind("127.0.0.1:0")
213            .await
214            .expect("test listener should bind");
215        let address = listener
216            .local_addr()
217            .expect("listener should have an address");
218        let server = tokio::spawn(async move {
219            let (mut socket, _) = listener.accept().await.expect("client should connect");
220            let mut request = Vec::new();
221            let mut chunk = [0_u8; 1024];
222            while !request.windows(4).any(|bytes| bytes == b"\r\n\r\n") {
223                let count = socket
224                    .read(&mut chunk)
225                    .await
226                    .expect("request should be readable");
227                assert_ne!(count, 0, "request ended before its headers");
228                request.extend_from_slice(&chunk[..count]);
229            }
230            let headers = format!(
231                "HTTP/1.1 200 OK\r\nContent-Length: {}\r\nConnection: close\r\n\r\n",
232                MAX_PROFILE_RESPONSE_BYTES + 1
233            );
234            socket
235                .write_all(headers.as_bytes())
236                .await
237                .expect("headers should write");
238        });
239        let client = reqwest::Client::new();
240        let url = format!("http://{address}/steve");
241
242        let result =
243            lookup_online_profile_once(&client, &url, "Steve", Duration::from_secs(1)).await;
244
245        assert!(matches!(
246            result,
247            Err(ProfileLookupError::InvalidResponse { reason, .. })
248                if reason.contains("exceeds")
249        ));
250        server.await.expect("test server should stop cleanly");
251    }
