1//! This module is responsible for sending chunks to the client.
2//!
3//! Chunk sending runs on its own independent tick loop, separate from the game
4//! tick. The three-phase design (prepare → encode → commit) minimizes lock hold
5//! time on the per-player `ChunkSender` mutex so that game-tick operations like
6//! `mark_chunk_pending_to_send` and `drop_chunk` are never blocked for long.

80impl EncodedChunk {
81    fn is_current_for(&self, prepared: &PreparedChunk) -> bool {
82        let Some(encoded_holder) = self.holder.upgrade() else {
83            return false;
84        };
85
86        self.pos == prepared.pos
87            && Arc::ptr_eq(&encoded_holder, &prepared.holder)
88            && self.readiness == prepared.readiness
89            && prepared.readiness.is_block_ticking()
90            && prepared.holder.ticking_readiness_snapshot() == prepared.readiness
91            && prepared.holder.packet_content_revision() == self.content_revision
92    }
93
94    fn into_parts(self) -> (EncodedChunkMetadata, EncodedPacket) {
95        (
96            EncodedChunkMetadata {
97                pos: self.pos,
98                content_revision: self.content_revision,
99                holder: self.holder,
100                readiness: self.readiness,
101            },
102            self.packet,
103        )
104    }
105}

119/// This struct is responsible for sending chunks to the client.
120pub struct ChunkSender {
121    /// A list of chunks that are waiting to be sent to the client.
122    pub pending_chunks: FxHashSet<ChunkPos>,
123    /// Chunks whose initial chunk packet has been queued for this client.
124    sent_chunks: FxHashSet<ChunkPos>,
125    /// The number of batches that have been sent to the client but have not been acknowledged yet.
126    pub unacknowledged_batches: u16,
127    /// The number of chunks that should be sent to the client per tick.
128    /// This is dynamically adjusted based on client feedback.
129    pub desired_chunks_per_tick: f32,
130    /// The number of chunks that can be sent to the client in the current batch.
131    pub batch_quota: f32,
132    /// The maximum number of unacknowledged batches allowed.
133    /// Starts at 1 and increases to `MAX_UNACKNOWLEDGED_BATCHES` after first ack.
134    pub max_unacknowledged_batches: u16,
135    deferred_encoded_chunks: Vec<EncodedChunk>,
136}

159impl ChunkSender {
160    /// Marks a chunk as pending to be sent to the client.
161    pub fn mark_chunk_pending_to_send(&mut self, pos: ChunkPos) {
162        self.sent_chunks.remove(&pos);
163        self.pending_chunks.insert(pos);
164    }

166    /// Drops a chunk from the client's view.
167    pub fn drop_chunk(&mut self, connection: &PlayerConnection, pos: ChunkPos) {
168        self.pending_chunks.remove(&pos);
169        self.deferred_encoded_chunks
170            .retain(|encoded| encoded.pos != pos);
171        if self.sent_chunks.remove(&pos) && !connection.closed() {
172            Self::send_packet(
173                connection,
174                CForgetLevelChunk {
175                    pos: PackedChunkPos::from(pos),
176                },
177            );
178        }
179    }

181    /// Encodes and sends a packet through the connection.
182    fn send_packet<P: ClientPacket>(connection: &PlayerConnection, packet: P) {
183        // `from_bare` refuses a payload past `MAX_PACKET_SIZE`, and under
184        // `panic = "abort"` an `expect` here ends the server. The light-update
185        // path in `chunk_map` already treats the identical error as a dropped
186        // packet and a warning; there is no reason this one should be fatal.
187        let Ok(encoded) =
188            EncodedPacket::from_bare(packet, connection.compression(), ConnectionProtocol::Play)
189        else {
190            log::warn!("Failed to encode a chunk-sender packet; dropping it");
191            return;
192        };
193        connection.send_encoded(encoded);
194    }
195
196    fn encode_packet<P: ClientPacket>(
197        connection: &PlayerConnection,
198        packet: P,
199    ) -> Option<EncodedPacket> {
200        match EncodedPacket::from_bare(packet, connection.compression(), ConnectionProtocol::Play) {
201            Ok(encoded) => Some(encoded),
202            Err(error) => {
203                log::warn!("Failed to encode a chunk-batch boundary packet: {error}");
204                None
205            }
206        }
207    }
208
209    fn try_send_batch(
210        connection: &PlayerConnection,
211        packets: Vec<EncodedPacket>,
212    ) -> Result<(), mpsc::error::TrySendError<Vec<EncodedPacket>>> {
213        if connection.closed() {
214            return Err(mpsc::error::TrySendError::Closed(packets));
215        }
216
217        match connection {
218            PlayerConnection::Java(java) => java.try_send_chunk_batch(packets),
219            PlayerConnection::Other(other) => {
220                for packet in packets {
221                    other.send_encoded(packet);
222                }
223                if other.closed() {
224                    Err(mpsc::error::TrySendError::Closed(Vec::new()))
225                } else {
226                    Ok(())
227                }
228            }
229        }
230    }
231
232    fn batch_byte_capacity(connection: &PlayerConnection) -> usize {
233        match connection {
234            PlayerConnection::Java(java) => java.chunk_batch_byte_capacity(),
235            PlayerConnection::Other(_) => usize::MAX,
236        }
237    }

239    /// Phase 1: Lock briefly to drain pending chunks and snapshot state.
240    ///
241    /// Returns `None` if there is nothing to send this tick.
242    pub fn prepare_batch(
243        &mut self,
244        world: &Arc<World>,
245        player_chunk_pos: ChunkPos,
246        chunk_send_epoch: &SyncMutex<u32>,
247    ) -> Option<PreparedBatch> {
248        if self.unacknowledged_batches >= self.max_unacknowledged_batches {
249            return None;
250        }
251
252        let max_batch_size = self.desired_chunks_per_tick.max(1.0);
253        self.batch_quota = (self.batch_quota + self.desired_chunks_per_tick).min(max_batch_size);
254
255        if self.batch_quota < 1.0 || self.pending_chunks.is_empty() {
256            return None;
257        }
258
259        let holders = self.collect_candidates(world, player_chunk_pos);
260        if holders.is_empty() {
261            return None;
262        }
263
264        let epoch_snapshot = *chunk_send_epoch.lock();
265
266        Some(PreparedBatch {
267            chunks: holders,
268            has_skylight: world.dimension_type.has_skylight,
269            epoch_snapshot,
270            reusable_encoded_chunks: SyncMutex::new(mem::take(&mut self.deferred_encoded_chunks)),
271        })
272    }
273
274    /// Phase 2: Encode chunks without holding any lock. Called between prepare and commit.
275    ///
276    /// Uses the dedicated encoding pool to encode chunks in parallel. A per-tick
277    /// local cache prevents multiple players sharing the same chunks from
278    /// re-encoding them within the same sending tick.
279    ///
280    /// # Panics
281    /// Panics if a chunk packet fails to encode.
282    pub fn encode_batch(
283        batch: &PreparedBatch,
284        cache: &mut FxHashMap<ChunkPos, EncodedChunk>,
285        compression: Option<CompressionInfo>,
286        encoding_pool: &ThreadPool,
287    ) -> Vec<EncodedChunk> {
288        let reusable = mem::take(&mut *batch.reusable_encoded_chunks.lock());
289        let mut reusable = reusable.into_iter();
290        let mut encoded_prefix = Vec::new();
291        for prepared in &batch.chunks {
292            let Some(encoded) = reusable.next() else {
293                break;
294            };
295            if !encoded.is_current_for(prepared) {
296                break;
297            }
298            encoded_prefix.push(encoded);
299        }
300
301        let cached_chunks = &*cache;
302        let encoded_chunks = encoding_pool.install(|| {
303            batch
304                .chunks
305                .par_iter()
306                .skip(encoded_prefix.len())
307                .map(|prepared| {
308                    let holder = &prepared.holder;
309                    let pos = prepared.pos;
310
311                    if let Some(cached) = cached_chunks.get(&pos)
312                        && cached.is_current_for(prepared)
313                    {
314                        return Some(cached.clone());
315                    }
316
317                    if !prepared.readiness.is_block_ticking()
318                        || holder.ticking_readiness_snapshot() != prepared.readiness
319                    {
320                        return None;
321                    }
322                    let revision_before = holder.packet_content_revision();
323                    let chunk = holder.try_full_chunk()?;
324
325                    // A chunk packet carries every section, the heightmaps, the
326                    // light arrays and the NBT of every block entity in the
327                    // column, so a chunk dense with signs, banners or command
328                    // blocks can pass `MAX_PACKET_SIZE` -- and without
329                    // compression that ceiling is only two megabytes. This runs
330                    // on a rayon worker, where `panic = "abort"` makes one
331                    // oversized chunk the end of the server. Skipping the chunk
332                    // leaves that one column unsent, which is survivable.
333                    let Ok(packet) = EncodedPacket::from_bare(
334                        CLevelChunkWithLight {
335                            x: pos.0.x,
336                            z: pos.0.y,
337                            chunk_data: chunk.extract_chunk_data(),
338                            light_data: chunk.extract_light_data(batch.has_skylight),
339                        },
340                        compression,
341                        ConnectionProtocol::Play,
342                    ) else {
343                        log::warn!("Failed to encode chunk packet for {pos:?}; skipping it");
344                        return None;
345                    };
346                    let revision_after = holder.packet_content_revision();
347                    if revision_before != revision_after
348                        || holder.ticking_readiness_snapshot() != prepared.readiness
349                    {
350                        return None;
351                    }
352
353                    Some(EncodedChunk {
354                        pos,
355                        packet,
356                        content_revision: revision_after,
357                        holder: Arc::downgrade(holder),
358                        readiness: prepared.readiness,
359                    })
360                })
361                .collect::<Vec<_>>()
362        });
363        encoded_prefix.extend(encoded_chunks.into_iter().flatten());
364
365        for encoded in &encoded_prefix {
366            cache.insert(encoded.pos, encoded.clone());
367        }
368
369        encoded_prefix
370    }

372    /// Phase 3: Lock briefly to verify generation counter and send the batch.
373    ///
374    /// If the player teleported between prepare and commit (generation counter
375    /// changed), the batch is discarded.
376    pub fn commit_batch(
377        &mut self,
378        batch: &PreparedBatch,
379        encoded_chunks: Vec<EncodedChunk>,
380        connection: &PlayerConnection,
381        chunk_send_epoch: &SyncMutex<u32>,
382    ) -> Vec<ChunkPos> {
383        let epoch = chunk_send_epoch.lock();
384        if *epoch != batch.epoch_snapshot {
385            return Vec::new();
386        }
387        drop(epoch);
388
389        let mut valid_chunks = Vec::with_capacity(encoded_chunks.len());
390        for encoded in encoded_chunks {
391            if !self.pending_chunks.contains(&encoded.pos) {
392                continue;
393            }
394            let Some(prepared) = batch.chunks.iter().find(|chunk| chunk.pos == encoded.pos) else {
395                continue;
396            };
397            if !encoded.is_current_for(prepared) {
398                continue;
399            }
400            valid_chunks.push(encoded);
401        }
402
403        if valid_chunks.is_empty() {
404            return Vec::new();
405        }
406
407        let Some(start) = Self::encode_packet(connection, CChunkBatchStart {}) else {
408            connection.close();
409            return Vec::new();
410        };
411        let batch_capacity = Self::batch_byte_capacity(connection);
412        let mut encoded_bytes = start.encoded_data.len().max(1);
413        let mut packets = Vec::with_capacity(valid_chunks.len() + 2);
414        let mut sent_chunks = Vec::with_capacity(valid_chunks.len());
415        let mut selected_metadata = Vec::with_capacity(valid_chunks.len());
416        let mut finished = None;
417        packets.push(start);
418        for encoded in valid_chunks {
419            let next_batch_size = sent_chunks.len() + 1;
420            let Ok(next_batch_size) = i32::try_from(next_batch_size) else {
421                break;
422            };
423            let Some(next_finished) = Self::encode_packet(
424                connection,
425                CChunkBatchFinished {
426                    batch_size: next_batch_size,
427                },
428            ) else {
429                connection.close();
430                return Vec::new();
431            };
432            let Some(next_encoded_bytes) = encoded_bytes
433                .checked_add(encoded.packet.encoded_data.len().max(1))
434                .and_then(|bytes| bytes.checked_add(next_finished.encoded_data.len().max(1)))
435            else {
436                break;
437            };
438            if next_encoded_bytes > batch_capacity {
439                break;
440            }
441
442            encoded_bytes = next_encoded_bytes - next_finished.encoded_data.len().max(1);
443            let (metadata, packet) = encoded.into_parts();
444            sent_chunks.push(metadata.pos);
445            selected_metadata.push(metadata);
446            packets.push(packet);
447            finished = Some(next_finished);
448        }
449        let Some(finished) = finished else {
450            log::warn!("A chunk packet exceeded the outbound batch byte budget; deferring it");
451            return Vec::new();
452        };
453        packets.push(finished);
454
455        match Self::try_send_batch(connection, packets) {
456            Ok(()) => {}
457            Err(mpsc::error::TrySendError::Full(returned_packets)) => {
458                let mut returned_packets = returned_packets.into_iter();
459                let _boundary = returned_packets.next();
460                self.deferred_encoded_chunks.extend(
461                    selected_metadata
462                        .into_iter()
463                        .zip(returned_packets)
464                        .map(|(metadata, packet)| metadata.with_packet(packet)),
465                );
466                return Vec::new();
467            }
468            Err(mpsc::error::TrySendError::Closed(_)) => {
469                connection.close();
470                return Vec::new();
471            }
472        }
473
474        self.unacknowledged_batches += 1;
475        self.batch_quota -= sent_chunks.len() as f32;
476
477        for pos in &sent_chunks {
478            self.pending_chunks.remove(pos);
479            self.sent_chunks.insert(*pos);
480        }
481        sent_chunks
482    }

484    fn collect_candidates(
485        &mut self,
486        world: &Arc<World>,
487        player_chunk_pos: ChunkPos,
488    ) -> Vec<PreparedChunk> {
489        let max_batch_size = self.batch_quota.floor() as usize;
490        let mut candidates: Vec<ChunkPos> = self.pending_chunks.iter().copied().collect();
491
492        // Sort by distance to player
493        candidates.sort_by_key(|pos| Self::chunk_distance_squared(*pos, player_chunk_pos));
494
495        let mut chunks_to_send = Vec::new();
496
497        for pos in candidates {
498            if chunks_to_send.len() >= max_batch_size {
499                break;
500            }
501
502            if let Some(holder) = world
503                .chunk_map
504                .chunks
505                .read_sync(&pos, |_, chunk| chunk.clone())
506                && holder.published_status() == Some(ChunkStatus::Full)
507            {
508                let readiness = holder.ticking_readiness_snapshot();
509                if readiness.is_block_ticking() {
510                    chunks_to_send.push(PreparedChunk {
511                        pos,
512                        holder,
513                        readiness,
514                    });
515                }
516            }
517        }
518        chunks_to_send
519    }
520
521    fn chunk_distance_squared(pos: ChunkPos, player_chunk_pos: ChunkPos) -> u64 {
522        let dx = u64::from(pos.0.x.abs_diff(player_chunk_pos.0.x));
523        let dz = u64::from(pos.0.y.abs_diff(player_chunk_pos.0.y));
524        dx.saturating_mul(dx).saturating_add(dz.saturating_mul(dz))
525    }

527    /// Handles the acknowledgement of a chunk batch from the client.
528    ///
529    /// The client sends back its desired chunks per tick based on how fast it can
530    /// process chunks. We clamp this value and use it to adjust our sending rate.
531    pub const fn on_chunk_batch_received_by_client(
532        &mut self,
533        desired_chunks_per_tick: f32,
534    ) -> bool {
535        if self.unacknowledged_batches == 0 {
536            return false;
537        }
538
539        self.unacknowledged_batches = self.unacknowledged_batches.saturating_sub(1);
540
541        // Vanilla clamps this client-controlled pacing value to 0.01..=64.0.
542        self.desired_chunks_per_tick = if desired_chunks_per_tick.is_nan() {
543            MIN_CHUNKS_PER_TICK
544        } else {
545            desired_chunks_per_tick.clamp(MIN_CHUNKS_PER_TICK, MAX_CHUNKS_PER_TICK)
546        };
547
548        // Reset batch quota when all batches are acknowledged
549        if self.unacknowledged_batches == 0 {
550            self.batch_quota = 1.0;
551        }
552
553        // After receiving the first acknowledgement, allow more unacknowledged batches
554        // for better pipelining (vanilla behavior)
555        self.max_unacknowledged_batches = MAX_UNACKNOWLEDGED_BATCHES;
556        true
557    }

851    #[test]
852    fn saturated_commit_defers_state_and_reuses_encoding_after_drain() {
853        init_vanilla_registry();
854        init_behaviors();
855        let world = fresh_test_world("chunk-sender-saturation");
856        let pos = ChunkPos::new(0, 0);
857        insert_ready_full_chunk(&world, pos);
858        let epoch = SyncMutex::new(0);
859        let mut chunk_sender = ChunkSender::default();
860        chunk_sender.mark_chunk_pending_to_send(pos);
861        let batch = chunk_sender
862            .prepare_batch(&world, pos, &epoch)
863            .expect("the ready chunk should form a batch");
864        let quota_before_commit = chunk_sender.batch_quota;
865        let encoding_pool = rayon::ThreadPoolBuilder::new()
866            .num_threads(1)
867            .build()
868            .expect("test chunk encoding pool should initialize");
869        let mut cache = FxHashMap::default();
870        let encoded = ChunkSender::encode_batch(&batch, &mut cache, None, &encoding_pool);
871        let Some(first_encoded) = encoded.first() else {
872            panic!("the ready chunk should encode");
873        };
874        let original_packet = Arc::downgrade(&first_encoded.packet.encoded_data);
875
876        let (outgoing, mut receiver) = outbound_packet_channel();
877        for _ in 0..256 {
878            assert!(
879                outgoing
880                    .try_send_chunk_batch(vec![encoded_packet(1)])
881                    .is_ok()
882            );
883        }
884        let runtime = Builder::new_current_thread()
885            .enable_io()
886            .build()
887            .expect("test network runtime should initialize");
888        let (_client, server_stream, remote_address) = runtime.block_on(async {
889            let listener = TcpListener::bind("127.0.0.1:0")
890                .await
891                .expect("test listener should bind");
892            let address = listener
893                .local_addr()
894                .expect("test listener should have an address");
895            let (client, accepted) = tokio::join!(TcpStream::connect(address), listener.accept());
896            let client = client.expect("test client should connect");
897            let (server_stream, remote_address) = accepted.expect("test listener should accept");
898            (client, server_stream, remote_address)
899        });
900        let (_read_half, write_half) = server_stream.into_split();
901        let connection = PlayerConnection::Java(JavaConnection::new(
902            outgoing,
903            CancellationToken::new(),
904            None,
905            Arc::new(AsyncMutex::new(Some(TCPNetworkEncoder::new(
906                BufWriter::new(write_half),
907            )))),
908            1,
909            remote_address,
910            Weak::new(),
911            None,
912            None,
913        ));
914
915        let committed = chunk_sender.commit_batch(&batch, encoded, &connection, &epoch);
916        assert!(committed.is_empty());
917        assert!(chunk_sender.pending_chunks.contains(&pos));
918        assert!(!chunk_sender.is_chunk_sent(pos));
919        assert_eq!(chunk_sender.unacknowledged_batches, 0);
920        assert_eq!(
921            chunk_sender.batch_quota.to_bits(),
922            quota_before_commit.to_bits()
923        );
924        cache.clear();
925
926        for _ in 0..3 {
927            let queued = receiver
928                .try_recv()
929                .expect("the saturated queue should contain filler packets");
930            drop(queued);
931        }
932        let retry_batch = chunk_sender
933            .prepare_batch(&world, pos, &epoch)
934            .expect("the deferred chunk should retry after drain");
935        let retry_encoded =
936            ChunkSender::encode_batch(&retry_batch, &mut cache, None, &encoding_pool);
937        let Some(retry_packet) = retry_encoded.first() else {
938            panic!("the deferred chunk should retain its encoding");
939        };
940        let Some(original_packet) = original_packet.upgrade() else {
941            panic!("the deferred batch should retain the encoded packet");
942        };
943        assert!(Arc::ptr_eq(
944            &original_packet,
945            &retry_packet.packet.encoded_data
946        ));
947
948        let committed = chunk_sender.commit_batch(&retry_batch, retry_encoded, &connection, &epoch);
949        assert_eq!(committed, vec![pos]);
950        assert!(!chunk_sender.pending_chunks.contains(&pos));
951        assert!(chunk_sender.is_chunk_sent(pos));
952        assert_eq!(chunk_sender.unacknowledged_batches, 1);
953        assert_eq!(
954            chunk_sender.batch_quota.to_bits(),
955            (quota_before_commit - 1.0).to_bits()
956        );
957    }
