1//! Reading a text component from vanilla's JSON form.
2
3use serde_json::Value;
4use text_components::TextComponent;
5
6/// Parses vanilla component JSON, or `None` when it is not one.
7///
8/// `text_components` reads an RGB color as `{"rgb": [r, g, b]}` rather than
9/// vanilla's `"#rrggbb"`, so hex colors are rewritten on the way in.
10#[must_use]
11pub fn parse_component(json: &str) -> Option<TextComponent> {
12    let mut value: Value = serde_json::from_str(json).ok()?;
13    rewrite_hex_colors(&mut value);
14    serde_json::from_value(value).ok()
15}
16
17fn rewrite_hex_colors(value: &mut Value) {
18    match value {
19        Value::Object(object) => {
20            let rgb = object
21                .get("color")
22                .and_then(Value::as_str)
23                .and_then(|color| color.strip_prefix('#'))
24                .filter(|hex| hex.len() == 6)
25                .and_then(|hex| u32::from_str_radix(hex, 16).ok());
26            if let Some(rgb) = rgb {
27                let [_, red, green, blue] = rgb.to_be_bytes();
28                object.insert(
29                    "color".to_owned(),
30                    serde_json::json!({ "rgb": [red, green, blue] }),
31                );
32            }
33            object.values_mut().for_each(rewrite_hex_colors);
34        }
35        Value::Array(items) => items.iter_mut().for_each(rewrite_hex_colors),
36        _ => {}
37    }
38}

40#[cfg(test)]
41mod tests {
42    use text_components::content::Content;
43    use text_components::format::Color;
44
45    use super::parse_component;
46
47    #[test]
48    fn vanilla_json_keeps_hex_colors_decorations_and_children() {
49        let parsed = parse_component(
50            r##"{"text":"Link","color":"#12ab34","bold":true,"extra":[{"translate":"a.b","with":[{"text":"x"}],"color":"gold"}]}"##,
51        );
52        let Some(parsed) = parsed else {
53            panic!("the JSON FotonComponents writes did not decode");
54        };
55        assert_eq!(parsed.format.color, Some(Color::Rgb(0x12, 0xab, 0x34)));
56        assert_eq!(parsed.format.bold, Some(true));
57        assert!(matches!(parsed.children[0].content, Content::Translate(_)));
58        assert_eq!(parsed.children[0].format.color, Some(Color::Gold));
59    }
60}
