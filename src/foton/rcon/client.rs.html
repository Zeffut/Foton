22/// Serves one connection until it closes, misbehaves, or the server stops.
23pub(super) async fn serve(
24    mut connection: TcpStream,
25    address: SocketAddr,
26    connection_id: u64,
27    password: Arc<str>,
28    server: Arc<Server>,
29    cancel: CancellationToken,
30) {
31    let mut authenticated = false;
32    // Set when authentication fails, so the connection closes after answering.
33    let mut reject_and_close = false;
34    loop {
35        let request = select! {
36            () = cancel.cancelled() => break,
37            request = read_request(&mut connection) => request,
38        };
39        let Some(request) = request else { break };
40
41        let sent = match request.kind {
42            SERVERDATA_AUTH => {
43                // Vanilla parity: an empty password never authenticates, even
44                // against an empty configured one -- though Foton refuses to
45                // start with one, so that case cannot arise here.
46                //
47                // The comparison is constant-time. Vanilla uses `String.equals`,
48                // which returns on the first differing byte and so leaks the
49                // length of the correct prefix; RCON is admin tooling rather
50                // than gameplay, so this is one of the places Foton is allowed
51                // to be stricter than vanilla.
52                authenticated =
53                    !request.body.is_empty() && constant_time_eq(&request.body, &password);
54                if authenticated {
55                    log::info!("Rcon client {address} authenticated");
56                    send(
57                        &mut connection,
58                        request.request_id,
59                        SERVERDATA_AUTH_RESPONSE,
60                        "",
61                    )
62                    .await
63                } else {
64                    log::warn!("Rcon client {address} failed to authenticate");
65                    // Vanilla parity: `RconClient` breaks out of its read loop
66                    // on a failed auth and the `finally` closes the socket, so
67                    // a client gets one guess per connection. Staying in the
68                    // loop turned a single TCP connection into an unlimited
69                    // password oracle.
70                    reject_and_close = true;
71                    send_auth_failure(&mut connection).await
72                }
73            }
74            SERVERDATA_EXECCOMMAND if authenticated => {
75                let response = run_command(&server, connection_id, &request.body, &cancel).await;
76                let Some(response) = response else { break };
77                send_command_response(&mut connection, request.request_id, &response).await
78            }
79            SERVERDATA_EXECCOMMAND => send_auth_failure(&mut connection).await,
80            unknown => {
81                // Vanilla parity: `String.format("Unknown request %s",
82                // Integer.toHexString(cmd))`, which prints the kind unsigned.
83                let message = format!("Unknown request {:x}", unknown as u32);
84                send_command_response(&mut connection, request.request_id, &message).await
85            }
86        };
87        if sent.is_err() || reject_and_close {
88            break;
89        }
90    }
91    log::debug!("Rcon client {address} disconnected");
92}

94/// Reads one frame, or `None` once the connection ends or breaks its own rules.
95async fn read_request(connection: &mut TcpStream) -> Option<RconRequest> {
96    let mut length = [0_u8; 4];
97    connection.read_exact(&mut length).await.ok()?;
98    let length = i32::from_le_bytes(length);
99
100    // Vanilla reads one 1460-byte chunk and hangs up unless the frame fills it
101    // exactly, which makes a client that coalesces two frames into one segment
102    // look malformed. Reading the declared length instead accepts that client;
103    // a well-formed frame is byte-for-byte the same either way. The upper
104    // bound is still vanilla's, so an absurd length is refused rather than
105    // allocated.
106    let length = usize::try_from(length).ok()?;
107    if !(8..=MAX_PACKET_SIZE - 4).contains(&length) {
108        return None;
109    }
110
111    let mut contents = vec![0_u8; length];
112    connection.read_exact(&mut contents).await.ok()?;
113    decode_request(&contents)
114}

116/// Runs one command and waits for everything it prints.
117///
118/// `None` means the server is shutting down and the connection should go with
119/// it. A command that never answers cannot happen: the output sink replies
120/// when its last handle is dropped, and every way an execution can end drops
121/// it.
122async fn run_command(
123    server: &Arc<Server>,
124    connection_id: u64,
125    command: &str,
126    cancel: &CancellationToken,
127) -> Option<String> {
128    let Ok(reply) = server.submit_rcon_command(connection_id, command.to_owned()) else {
129        // Vanilla parity: the shape of `RconClient.run`'s catch, which hands
130        // the client the failure as ordinary command output.
131        return Some(format!(
132            "Error executing: {command} (the command queue is full)"
133        ));
134    };
135    select! {
136        () = cancel.cancelled() => None,
137        response = reply => Some(response.unwrap_or_default()),
138    }
139}

141async fn send_command_response(
142    connection: &mut TcpStream,
143    request_id: i32,
144    response: &str,
145) -> Result<(), io::Error> {
146    for chunk in split_response(response) {
147        send(connection, request_id, SERVERDATA_RESPONSE_VALUE, chunk).await?;
148    }
149    Ok(())
150}

152async fn send_auth_failure(connection: &mut TcpStream) -> Result<(), io::Error> {
153    send(
154        connection,
155        AUTH_FAILURE_REQUEST_ID,
156        SERVERDATA_AUTH_RESPONSE,
157        "",
158    )
159    .await
160}

162async fn send(
163    connection: &mut TcpStream,
164    request_id: i32,
165    kind: i32,
166    payload: &str,
167) -> Result<(), io::Error> {
168    connection
169        .write_all(&encode_response(request_id, kind, payload))
170        .await
171}

173/// Compares two secrets without leaking where they first differ.
174///
175/// Folds every byte into one accumulator so the work does not depend on the
176/// contents. Lengths are compared first and separately, which leaks only the
177/// length -- as any timing-safe primitive of this shape does.
178fn constant_time_eq(candidate: &str, expected: &str) -> bool {
179    let candidate = candidate.as_bytes();
180    let expected = expected.as_bytes();
181    if candidate.len() != expected.len() {
182        return false;
183    }
184    let mut difference = 0u8;
185    for (left, right) in candidate.iter().zip(expected) {
186        difference |= left ^ right;
187    }
188    difference == 0
189}

191#[cfg(test)]
192mod tests {
193    use super::constant_time_eq;
194
195    /// The comparison still answers correctly; being timing-safe is not an
196    /// excuse for being wrong.
197    #[test]
198    fn constant_time_eq_matches_ordinary_equality() {
199        for (candidate, expected) in [
200            ("hunter2", "hunter2"),
201            ("hunter2", "hunter3"),
202            ("hunter2", "hunter22"),
203            ("hunter2", "hunter"),
204            ("", ""),
205            ("", "x"),
206            ("x", ""),
207            // Multi-byte characters must not be treated as single units.
208            ("clé", "clé"),
209            ("clé", "cle"),
210        ] {
211            assert_eq!(
212                constant_time_eq(candidate, expected),
213                candidate == expected,
214                "constant_time_eq({candidate:?}, {expected:?}) disagreed with =="
215            );
216        }
217    }
218
219    /// It compares every byte rather than stopping at the first difference.
220    ///
221    /// A password oracle is built out of the timing difference between "wrong
222    /// at byte 0" and "wrong at byte 30". This cannot measure time reliably in
223    /// a unit test, so it pins the observable proxy: the two cases do the same
224    /// amount of work because neither returns early.
225    #[test]
226    fn constant_time_eq_does_not_stop_at_the_first_difference() {
227        let expected = "a".repeat(32);
228        let differs_early = format!("Z{}", "a".repeat(31));
229        let differs_late = format!("{}Z", "a".repeat(31));
230
231        assert!(!constant_time_eq(&differs_early, &expected));
232        assert!(!constant_time_eq(&differs_late, &expected));
233        // Both are the same length as the secret, so both walk all 32 bytes.
234        assert_eq!(differs_early.len(), expected.len());
235        assert_eq!(differs_late.len(), expected.len());
236    }
