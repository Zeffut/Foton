1//! Source Rcon server.
2//!
3//! Vanilla parity: `net.minecraft.server.rcon.thread.RconThread`, which owns
4//! the listening socket and gives every connection its own worker. Foton runs
5//! those as Tokio tasks on the shared tracker rather than as threads, because
6//! a connection spends nearly all of its life waiting.

27const ACCEPT_ERROR_BACKOFF: Duration = Duration::from_millis(50);
28
29struct AcceptedConnection {
30    connection: TcpStream,
31    address: SocketAddr,
32    permit: OwnedSemaphorePermit,
33}
34
35/// A bound Rcon port waiting to be served.
36pub struct RconListener {
37    listener: TcpListener,
38    password: Arc<str>,
39    connection_permits: Arc<Semaphore>,
40    next_connection: u64,
41}
42
43impl RconListener {
44    /// Binds the Rcon port.
45    ///
46    /// Binding here rather than inside the accept loop is deliberate: a port
47    /// already in use is a startup failure the operator can see, not a warning
48    /// scrolling past while remote administration silently never works.
49    pub async fn bind(
50        bind: IpAddr,
51        port: u16,
52        password: Arc<str>,
53        max_connections: usize,
54    ) -> Result<Self, io::Error> {
55        let address = SocketAddr::new(bind, port);
56        let listener = TcpListener::bind(address).await?;
57        log::info!("Rcon running on {address}");
58        Ok(Self {
59            listener,
60            password,
61            connection_permits: Arc::new(Semaphore::new(max_connections)),
62            next_connection: 0,
63        })
64    }
65
66    async fn accept_connection(&self, cancel: &CancellationToken) -> Option<AcceptedConnection> {
67        let permit = select! {
68            () = cancel.cancelled() => return None,
69            permit = Arc::clone(&self.connection_permits).acquire_owned() => permit.ok()?,
70        };
71        let (connection, address) = select! {
72            () = cancel.cancelled() => return None,
73            accepted = self.listener.accept() => {
74                accept_or_backoff(accepted, ACCEPT_ERROR_BACKOFF).await?
75            }
76        };
77        Some(AcceptedConnection {
78            connection,
79            address,
80            permit,
81        })
82    }
83
84    /// Accepts connections until the server stops.
85    pub async fn run(
86        mut self,
87        server: Arc<Server>,
88        cancel: CancellationToken,
89        task_tracker: TaskTracker,
90    ) {
91        loop {
92            let Some(accepted) = self.accept_connection(&cancel).await else {
93                if cancel.is_cancelled() {
94                    break;
95                }
96                continue;
97            };
98            let AcceptedConnection {
99                connection,
100                address,
101                permit,
102            } = accepted;
103            log::info!("Accepted Rcon connection from {address}");
104
105            let connection_id = self.next_connection;
106            self.next_connection = self.next_connection.wrapping_add(1);
107            task_tracker.spawn(client::serve(
108                connection,
109                address,
110                connection_id,
111                permit,
112                Arc::clone(&self.password),
113                Arc::clone(&server),
114                cancel.child_token(),
115            ));
116        }
117    }
118}
119
120async fn accept_or_backoff<T>(accepted: io::Result<T>, duration: Duration) -> Option<T> {
121    match accepted {
122        Ok(accepted) => Some(accepted),
123        Err(error) => {
124            log::warn!(
125                "Failed to accept Rcon connection: {error}; retrying after {} ms",
126                duration.as_millis()
127            );
128            sleep(duration).await;
129            None
130        }
131    }
132}

135mod tests {
136    use std::{
137        io,
138        net::{IpAddr, Ipv4Addr},

148    #[tokio::test]
149    async fn listener_binds_only_the_configured_loopback_address() {
150        let listener = RconListener::bind(
151            IpAddr::V4(Ipv4Addr::LOCALHOST),
152            0,
153            Arc::from("test-password"),
154            8,
155        )
156        .await
157        .expect("loopback listener should bind");
158
159        let address = listener
160            .listener
161            .local_addr()
162            .expect("bound listener should have a local address");
163        assert_eq!(address.ip(), IpAddr::V4(Ipv4Addr::LOCALHOST));
164    }
165
166    #[tokio::test]
167    async fn listener_does_not_accept_more_than_the_connection_limit() {
168        let listener = RconListener::bind(
169            IpAddr::V4(Ipv4Addr::LOCALHOST),
170            0,
171            Arc::from("test-password"),
172            1,
173        )
174        .await
175        .expect("loopback listener should bind");
176        let address = listener
177            .listener
178            .local_addr()
179            .expect("bound listener should have a local address");
180        let cancel = CancellationToken::new();
181
182        let first_client = TcpStream::connect(address);
183        let (first_connection, connected) =
184            tokio::join!(listener.accept_connection(&cancel), first_client);
185        connected.expect("first client should connect");
186        let first_connection = first_connection.expect("first connection should be accepted");
187
188        let second_client = TcpStream::connect(address)
189            .await
190            .expect("second client should reach the TCP backlog");
191        assert!(
192            timeout(
193                Duration::from_millis(50),
194                listener.accept_connection(&cancel)
195            )
196            .await
197            .is_err(),
198            "a second connection must wait for the first connection's permit"
199        );
200
201        drop(first_connection);
202        let second_connection =
203            timeout(Duration::from_secs(1), listener.accept_connection(&cancel))
204                .await
205                .expect("accept should resume after a permit is released")
206                .expect("second connection should be accepted");
207        drop(second_connection);
208        drop(second_client);
209    }

211    #[tokio::test]
212    async fn accept_failures_wait_before_retrying() {
