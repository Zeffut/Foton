1//! Wire format for the Source RCON protocol.
2//!
3//! Vanilla parity: `net.minecraft.server.rcon.thread.RconClient.send` and
4//! `PktUtils`. Every field is a little-endian `i32`, the payload is a
5//! NUL-terminated string, and one more NUL closes the frame. The length prefix
6//! counts everything after itself, so it is the payload length plus the two
7//! ids, the terminator and the pad.
8
9/// Largest request frame vanilla will read in one go.
10///
11/// Vanilla parity: `PktUtils.MAX_PACKET_SIZE`.
12pub(super) const MAX_PACKET_SIZE: usize = 1460;
13
14/// Bytes a frame carries besides its payload: two ids, a terminator and a pad.
15const FRAME_OVERHEAD: usize = 10;
16
17/// Largest response chunk vanilla emits before splitting.
18///
19/// Vanilla parity: the `4096` of `RconClient.sendCmdResponse`, which is a
20/// `String.substring` bound and therefore counts UTF-16 code units.
21const MAX_RESPONSE_UNITS: usize = 4096;
22
23/// Client-to-server login attempt. Vanilla `SERVERDATA_AUTH`.
24pub(super) const SERVERDATA_AUTH: i32 = 3;
25/// Client-to-server command. Vanilla `SERVERDATA_EXECCOMMAND`.
26pub(super) const SERVERDATA_EXECCOMMAND: i32 = 2;
27/// Server-to-client command output. Vanilla `SERVERDATA_RESPONSE_VALUE`.
28pub(super) const SERVERDATA_RESPONSE_VALUE: i32 = 0;
29/// Server-to-client login verdict. Vanilla `SERVERDATA_AUTH_RESPONSE`.
30pub(super) const SERVERDATA_AUTH_RESPONSE: i32 = 2;
31/// Request id a rejected login is answered with. Vanilla `SERVERDATA_AUTH_FAILURE`.
32pub(super) const AUTH_FAILURE_REQUEST_ID: i32 = -1;
33
34/// One decoded request frame.
35#[derive(Debug, Clone, PartialEq, Eq)]
36pub(super) struct RconRequest {
37    /// Echoed back on every response so a client can pair them up.
38    pub(super) request_id: i32,
39    /// Which of the `SERVERDATA_*` requests this is.
40    pub(super) kind: i32,
41    /// The password or the command, depending on `kind`.
42    pub(super) body: String,
43}
44
45/// Reads the id, kind and body out of a frame's contents.
46///
47/// `contents` is everything the length prefix covered. A frame shorter than
48/// the two ids is malformed; vanilla drops the connection for that, and so
49/// does the caller.
50pub(super) fn decode_request(contents: &[u8]) -> Option<RconRequest> {
51    if contents.len() < 8 {
52        return None;
53    }
54    let request_id = i32::from_le_bytes([contents[0], contents[1], contents[2], contents[3]]);
55    let kind = i32::from_le_bytes([contents[4], contents[5], contents[6], contents[7]]);
56    // Vanilla parity: `PktUtils.stringFromByteArray` stops at the first NUL and
57    // decodes the rest as UTF-8 without validating it.
58    let body = &contents[8..];
59    let body = body.split(|byte| *byte == 0).next().unwrap_or(body);
60    Some(RconRequest {
61        request_id,
62        kind,
63        body: String::from_utf8_lossy(body).into_owned(),
64    })
65}
66
67/// Builds one response frame.
68pub(super) fn encode_response(request_id: i32, kind: i32, payload: &str) -> Vec<u8> {
69    let payload = payload.as_bytes();
70    let length = payload.len() + FRAME_OVERHEAD;
71    let mut frame = Vec::with_capacity(length + 4);
72    frame.extend_from_slice(&(length as i32).to_le_bytes());
73    frame.extend_from_slice(&request_id.to_le_bytes());
74    frame.extend_from_slice(&kind.to_le_bytes());
75    frame.extend_from_slice(payload);
76    frame.extend_from_slice(&[0, 0]);
77    frame
78}

80/// Splits command output into the chunks one response is sent as.
81///
82/// Vanilla parity: the `do`/`while` of `RconClient.sendCmdResponse`. The loop
83/// body runs before the length is tested, so a command that printed nothing
84/// still sends one empty frame -- a client that got no frame at all would wait
85/// forever for a reply that already happened.
86///
87/// Vanilla cuts at a UTF-16 index and will happily halve a surrogate pair.
88/// This cuts at the last code-point boundary that fits instead, which no
89/// client can tell apart except by not receiving a broken character.
90pub(super) fn split_response(response: &str) -> Vec<&str> {
91    if response.is_empty() {
92        return vec![""];
93    }
94
95    let mut chunks = Vec::new();
96    let mut start = 0;
97    while start < response.len() {
98        let mut units = 0;
99        let mut end = start;
100        for (offset, character) in response[start..].char_indices() {
101            let next = units + character.len_utf16();
102            if next > MAX_RESPONSE_UNITS {
103                break;
104            }
105            units = next;
106            end = start + offset + character.len_utf8();
107        }
108        chunks.push(&response[start..end]);
109        start = end;
110    }
111    chunks
112}

114#[cfg(test)]
115mod tests {
116    use super::{
117        MAX_RESPONSE_UNITS, RconRequest, SERVERDATA_AUTH, SERVERDATA_RESPONSE_VALUE,
118        decode_request, encode_response, split_response,
119    };
120
121    #[test]
122    fn a_frame_round_trips_through_the_wire_layout() {
123        let frame = encode_response(7, SERVERDATA_RESPONSE_VALUE, "hello");
124        assert_eq!(&frame[..4], &15_i32.to_le_bytes(), "length excludes itself");
125        assert_eq!(&frame[frame.len() - 2..], &[0, 0], "two NULs close a frame");
126        assert_eq!(
127            decode_request(&frame[4..]),
128            Some(RconRequest {
129                request_id: 7,
130                kind: SERVERDATA_RESPONSE_VALUE,
131                body: "hello".to_owned(),
132            })
133        );
134    }
135
136    #[test]
137    fn a_body_stops_at_its_terminator() {
138        let mut contents = Vec::new();
139        contents.extend_from_slice(&3_i32.to_le_bytes());
140        contents.extend_from_slice(&SERVERDATA_AUTH.to_le_bytes());
141        contents.extend_from_slice(b"secret\0trailing junk\0");
142        let request = decode_request(&contents).expect("a full frame decodes");
143        assert_eq!(request.body, "secret");
144    }
145
146    #[test]
147    fn a_frame_without_room_for_its_ids_is_rejected() {
148        assert_eq!(decode_request(&[0, 0, 0, 0, 0, 0, 0]), None);
149    }
150
151    #[test]
152    fn an_empty_response_is_still_one_frame() {
153        assert_eq!(split_response(""), vec![""]);
154    }
155
156    #[test]
157    fn a_long_response_is_split_and_loses_nothing() {
158        let response = "x".repeat(MAX_RESPONSE_UNITS * 2 + 5);
159        let chunks = split_response(&response);
160        assert_eq!(chunks.len(), 3);
161        assert_eq!(chunks[0].len(), MAX_RESPONSE_UNITS);
162        assert_eq!(chunks[1].len(), MAX_RESPONSE_UNITS);
163        assert_eq!(chunks[2].len(), 5);
164        assert_eq!(chunks.concat(), response);
165    }
166
167    #[test]
168    fn a_split_never_lands_inside_a_character() {
169        // Every one of these is two UTF-16 units, so the cut has to fall one
170        // unit short of the bound rather than through the middle of a pair.
171        let response = "\u{1F9F1}".repeat(MAX_RESPONSE_UNITS);
172        let chunks = split_response(&response);
173        assert_eq!(chunks.concat(), response);
174        for chunk in &chunks {
175            assert!(
176                chunk.chars().count() * 2 <= MAX_RESPONSE_UNITS,
177                "a chunk may not exceed the vanilla bound"
178            );
179        }
180    }
