23/// The full server configuration as deserialized from TOML.
24///
25/// Contains both creation-time values (seed, world generator, storage)
26/// and runtime values that get moved into `RuntimeConfig`.
27#[derive(Debug, Clone, Deserialize)]
28#[serde(deny_unknown_fields)]
29pub struct ServerConfig {
30    /// The port the server will listen on.
31    pub server_port: u16,
32    /// The maximum number of players that can be on the server at once.
33    pub max_players: u32,
34    /// Allow `view_distance` above vanilla's 32-chunk cap.
35    #[serde(default)]
36    pub allow_extended_view_distance: bool,
37    /// The view distance of the server.
38    pub view_distance: u8,
39    /// The simulation distance of the server.
40    pub simulation_distance: u8,
41    /// Maximum queued neighbor-update tasks in one chained run; negative means unlimited.
42    #[serde(default = "default_max_chained_neighbor_updates")]
43    pub max_chained_neighbor_updates: i32,
44    /// Whether the server is in online mode.
45    pub online_mode: bool,
46    /// Whether admission is restricted to the persisted whitelist.
47    #[serde(default)]
48    pub whitelist_enabled: bool,
49    /// Optional authentication endpoint for online-mode `hasJoined` checks.
50    pub auth_server: Option<String>,
51    /// Optional endpoint for online-mode player name-to-profile lookups.
52    pub profile_server: Option<String>,
53    /// Optional endpoint for Mojang-compatible service public keys.
54    pub services_server: Option<String>,
55    /// Whether the server should use encryption. Required in online mode.
56    pub encryption: bool,
57    /// Whether vanilla floating/flying movement checks permit unauthorized flight.
58    #[serde(default)]
59    pub allow_flight: bool,
60    /// The message of the day.
61    pub motd: String,
62    /// Whether to use a favicon.
63    pub use_favicon: bool,
64    /// The path to the favicon.
65    pub favicon: String,
66    /// Whether to enforce secure chat.
67    pub enforce_secure_chat: bool,
68    /// Vanilla chat spam threshold window in seconds
69    #[serde(default = "default_spam_threshold_seconds")]
70    pub chat_spam_threshold_seconds: i32,
71    /// Vanilla command spam threshold window in seconds
72    #[serde(default = "default_spam_threshold_seconds")]
73    pub command_spam_threshold_seconds: i32,
74    /// The compression settings for the server.
75    pub compression: Option<CompressionInfo>,
76    /// All settings and configurations for server links.
77    pub server_links: Option<ServerLinks>,
78    /// Thread counts for server thread pools.
79    #[serde(default)]
80    pub threads: ThreadConfig,
81    /// Remote administration over the Source Rcon protocol.
82    #[serde(default)]
83    pub rcon: RconConfig,
84    /// Letting Bedrock Edition players in.
85    #[serde(default)]
86    pub bedrock: BedrockConfig,
87    /// Where player-filed bug reports are sent, on top of the local file.
88    #[serde(default)]
89    pub bug_reports: BugReportsConfig,
90}

92/// Where player-filed bug reports go once they are on disk.
93///
94/// Leaving `webhook_url` unset keeps reports local, which is the right default
95/// for a server nobody is collecting from.
96#[derive(Debug, Clone, Default, Deserialize)]
97#[serde(default, deny_unknown_fields)]
98pub struct BugReportsConfig {
99    /// Endpoint each filed report is posted to. Unset means reports stay local.
100    pub webhook_url: Option<String>,
101    /// Optional bearer token sent with each post.
102    pub webhook_token: Option<String>,
103}
104
105impl BugReportsConfig {
106    /// Resolves the configured endpoint, rejecting an address that cannot work.
107    ///
108    /// A typo here is otherwise invisible until a tester writes out a repro
109    /// that then goes nowhere, so it is a startup failure rather than a
110    /// warning -- the same reasoning `RconConfig` uses for a blank password.
111    pub(super) fn into_webhook(self) -> Result<Option<BugReportWebhook>, String> {
112        let Some(url) = self.webhook_url else {
113            return Ok(None);
114        };
115        let url = Url::parse(&url)
116            .map_err(|error| format!("bug_reports.webhook_url is not a valid URL: {error}"))?;
117        if !matches!(url.scheme(), "http" | "https") {
118            return Err(format!(
119                "bug_reports.webhook_url must be http or https, not {}",
120                url.scheme()
121            ));
122        }
123        Ok(Some(BugReportWebhook {
124            url,
125            token: self.webhook_token,
126        }))
127    }
128}

166/// Remote administration over the Source Rcon protocol.
167///
168/// Vanilla parity: `enable-rcon`, `rcon.port` and `rcon.password`. Vanilla
169/// warns and quietly disables Rcon when the password is blank; Foton refuses
170/// to start instead, because an administrator who asked for remote access and
171/// silently did not get it is worse off than one who is told why.
172#[derive(Debug, Clone, Deserialize)]
173#[serde(default, deny_unknown_fields)]
174pub struct RconConfig {
175    /// Whether the Rcon port is opened at all.
176    pub enable: bool,
177    /// The port Rcon listens on.
178    pub port: u16,
179    /// The password every client must send before it may run a command.
180    pub password: String,
181}
182
183impl Default for RconConfig {
184    fn default() -> Self {
185        Self {
186            enable: false,
187            // Vanilla's default `rcon.port`.
188            port: 25575,
189            password: String::new(),
190        }
191    }
192}

212/// Validates the server configuration.
213///
214/// # Errors
215/// This function will return an error if the configuration is invalid.
216pub(super) fn validate(config: &ServerConfig) -> Result<(), &'static str> {
217    validate_login_security(config.online_mode, config.encryption)?;
218    // A TCP bind on port 0 takes an ephemeral port instead of failing, so
219    // this is not caught by the listener's own error path -- and
220    // `bedrock.port = 0` means "share `server_port`" (`resolved_port`), so a
221    // zero `server_port` would silently drag the Bedrock port down with it
222    // too. Nonsense independently of Bedrock either way.
223    if config.server_port == 0 {
224        return Err("server_port must not be 0");
225    }
226    if !config.allow_extended_view_distance && !(1..=32).contains(&config.view_distance) {
227        return Err("View distance must in range 1..32");
228    }
229    if config.allow_extended_view_distance
230        && !(1..=MAX_SUPPORTED_VIEW_DISTANCE).contains(&config.view_distance)
231    {
232        return Err("View distance must in range 1..128");
233    }
234    if let Some(auth_server) = &config.auth_server {
235        let Ok(url) = Url::parse(auth_server) else {
236            return Err("auth_server must be an absolute URL");
237        };
238        if !matches!(url.scheme(), "http" | "https") {
239            return Err("auth_server must use http or https");
240        }
241    }
242    if let Some(profile_server) = &config.profile_server {
243        let Ok(url) = Url::parse(profile_server) else {
244            return Err("profile_server must be an absolute URL");
245        };
246        if !matches!(url.scheme(), "http" | "https") {
247            return Err("profile_server must use http or https");
248        }
249    }
250    if let Some(services_server) = &config.services_server {
251        let Ok(url) = Url::parse(services_server) else {
252            return Err("services_server must be an absolute URL");
253        };
254        if !matches!(url.scheme(), "http" | "https") {
255            return Err("services_server must use http or https");
256        }
257    }
258    if config.simulation_distance > config.view_distance {
259        return Err("Simulation distance must be less than or equal to view distance");
260    }
261    if let Some(compression) = config.compression {
262        if compression.threshold.get() < 256 {
263            return Err("Compression threshold must be greater than or equal to 256");
264        }
265        if !(1..=9).contains(&compression.level) {
266            return Err("Compression level must be between 1 and 9");
267        }
268    }
269    if config.rcon.enable {
270        if config.rcon.password.is_empty() {
271            return Err("rcon.password must be set when rcon is enabled");
272        }
273        if config.rcon.port == 0 {
274            return Err("rcon.port must be a real port when rcon is enabled");
275        }
276        if config.rcon.port == config.server_port {
277            return Err("rcon.port must differ from server_port");
278        }
279    }
280    if config.bedrock.enable {
281        // `bedrock.port` may legitimately be `0` (share `server_port`, see
282        // `BedrockConfig::resolved_port`) or equal to `server_port` itself:
283        // TCP (Java) and UDP (Bedrock/RakNet) do not share a port namespace,
284        // so neither is a collision.
285        if config.bedrock.trusted_proxies.is_empty() {
286            return Err("bedrock.trusted_proxies must list at least one address");
287        }
288        // A non-parseable entry is silently dropped by `is_trusted`
289        // (`foton-login`'s login path), not rejected -- so a list that is
290        // non-empty but entirely unparsable (`["localhost"]`, say) passes
291        // the check above and then refuses every Bedrock login. At least one
292        // entry has to actually be an address for `trusted_proxies` to do
293        // anything.
294        if !config
295            .bedrock
296            .trusted_proxies
297            .iter()
298            .any(|entry| entry.parse::<IpAddr>().is_ok())
299        {
300            return Err(
301                "bedrock.trusted_proxies must contain at least one address that parses as an IP",
302            );
303        }
304        // Bounded by bytes, not characters: `java_username` truncates the
305        // prefix and gamertag together to 16 *bytes* (what the login packet's
306        // encoder actually enforces -- see `foton-bedrock`'s `floodgate`
307        // module), so a prefix that is itself 16+ bytes leaves no room for
308        // any of the gamertag, even if it is fewer than 16 characters (a
309        // multi-byte prefix, e.g. six CJK characters at 18 bytes).
310        if config.bedrock.username_prefix.len() >= 16 {
311            return Err("bedrock.username_prefix leaves no room for a username");
312        }
313    }
314    if config.enforce_secure_chat {
315        if !config.online_mode {
316            return Err("online_mode must be true when enforce_secure_chat is enabled");
317        }
318        if !config.encryption {
319            return Err("encryption must be true when enforce_secure_chat is enabled");
320        }
321    }
322    Ok(())
