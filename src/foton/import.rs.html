1//! `foton import-anvil`: fills the configured world storage from a vanilla world.
2
3use std::path::Path;
4use std::sync::atomic::{AtomicUsize, Ordering};
5use std::thread;
6
7use foton_core::chunk_saver::anvil::{
8    ImportRequest, Progress, SourceDimension, discover_dimensions, import_world,
9    parse_dimension_name, resolve_targets,
10};
11use foton_utils::Identifier;
12
13use crate::args::ImportArgs;
14use crate::config;
15
16/// Runs the import and prints what it did.
17///
18/// # Errors
19/// Returns the message to show the user if the world cannot be read, is not at
20/// the targeted `DataVersion`, or cannot be written.
21pub fn run(args: ImportArgs) -> Result<(), String> {
22    let foton_config = config::load_or_create(Path::new("config/config.toml"))?;
23
24    let mut wanted: Vec<Identifier> = Vec::new();
25    for name in &args.dimensions {
26        wanted.push(parse_dimension_name(name).ok_or_else(|| {
27            format!(
28                "`{name}` is not a dimension name; use overworld, the_nether, the_end or a full id"
29            )
30        })?);
31    }
32    let found = discover_dimensions(&args.source)
33        .map_err(|error| format!("cannot read {}: {error}", args.source.display()))?;
34    if found.is_empty() {
35        return Err(format!(
36            "no dimension with a region folder in {}; expected dimensions/<namespace>/<path>/region, \
37             or region/, DIM-1/region and DIM1/region",
38            args.source.display()
39        ));
40    }
41    let dimensions: Vec<SourceDimension> = if wanted.is_empty() {
42        found
43    } else {
44        for key in &wanted {
45            if found.iter().all(|dimension| &dimension.key != key) {
46                return Err(format!(
47                    "{} has no dimension {key}; it has: {}",
48                    args.source.display(),
49                    found
50                        .iter()
51                        .map(|dimension| dimension.key.to_string())
52                        .collect::<Vec<_>>()
53                        .join(", ")
54                ));
55            }
56        }
57        found
58            .into_iter()
59            .filter(|dimension| wanted.contains(&dimension.key))
60            .collect()
61    };
62
63    let targets = resolve_targets(&foton_config.worlds, dimensions, args.world.as_deref())
64        .map_err(|error| error.to_string())?;
65    for target in &targets {
66        println!(
67            "importing {} from {} into {}",
68            target.dimension,
69            target.source_dir.display(),
70            target.region_dir.display()
71        );
72    }
73
74    let request = ImportRequest {
75        source_root: args.source,
76        targets,
77        replace: args.replace,
78        seed: args.seed,
79        skip_outdated: args.skip_outdated,
80        threads: thread::available_parallelism().map_or(2, usize::from),
81    };
82    let last_region = AtomicUsize::new(0);
83    let report = import_world(&request, &|progress: Progress<'_>| {
84        if last_region.swap(progress.region, Ordering::Relaxed) != progress.region {
85            println!(
86                "  {}: region {}/{}, {} chunks imported",
87                progress.dimension, progress.region, progress.region_count, progress.imported
88            );
89        }
90    })
91    .map_err(|error| error.to_string())?;
92    print!("{report}");
93    Ok(())
94}
