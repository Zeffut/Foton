1//! What an operator turns on, and what both sides of the feature read.
2//!
3//! This lives here rather than beside the rest of the server's configuration
4//! for a structural reason: the login path needs it, and `foton-login` cannot
5//! reach the `foton` binary crate — the dependency runs the other way. Both
6//! `foton` and `foton-login` already depend on `foton-bedrock`, so this is the
7//! one place both can see, and one declaration is the point. Two mirrors of
8//! the same operator-facing settings drift, and the drift is silent.
9
10use serde::Deserialize;
11
12/// Letting Bedrock Edition players in, through a Geyser this server runs.
13///
14/// Off by default: enabling it starts a Java process, and nothing should pay
15/// for a feature it did not ask for.
16#[derive(Debug, Clone, Deserialize)]
17#[serde(default, deny_unknown_fields)]
18pub struct BedrockConfig {
19    /// Whether Geyser is started and the Bedrock port opened at all.
20    ///
21    /// Off also means the login path takes every hostname down the ordinary
22    /// Java route, including one carrying a Floodgate payload.
23    pub enable: bool,
24    /// The UDP port Bedrock clients connect to.
25    ///
26    /// `0` — the default — means "share the Java server's port": one port
27    /// number for an operator to open, and it keeps following `server_port`
28    /// if that ever moves. [`BedrockConfig::resolved_port`] is the single
29    /// place this is turned into an actual port number; TCP (Java) and
30    /// UDP (Bedrock/RakNet) do not share a port namespace, so this is not a
31    /// collision.
32    ///
33    /// The honest trade-off: a Bedrock client's own "Add Server" dialog
34    /// pre-fills `19132`, so sharing the Java port means a player must type
35    /// the port in by hand. An operator who wants that automatic experience
36    /// back sets this to `19132` explicitly.
37    pub port: u16,
38    /// What Bedrock clients see in the server list; empty reuses the server MOTD.
39    pub motd: String,
40    /// Prepended to a Bedrock player's gamertag so it cannot collide with a
41    /// Java player's name. Empty means no prefix, and collisions become possible.
42    pub username_prefix: String,
43    /// Addresses a Floodgate handshake is accepted from.
44    ///
45    /// Foton starts Geyser locally, so loopback is the whole list by default. A
46    /// handshake claiming Floodgate from anywhere else is refused, because the
47    /// alternative is that anyone who can reach the Java port becomes anyone.
48    pub trusted_proxies: Vec<String>,
49    /// A JDK or JRE for Geyser; empty means `JAVA_HOME`, then `java` on the path.
50    pub java_home: String,
51    /// An operator-supplied Geyser jar; empty means fetch the pinned build.
52    pub jar_path: String,
53}
54
55impl Default for BedrockConfig {
56    fn default() -> Self {
57        Self {
58            enable: false,
59            // 0: share the Java server's port. See the field's own doc comment.
60            port: 0,
61            motd: String::new(),
62            username_prefix: ".".to_owned(),
63            trusted_proxies: vec!["127.0.0.1".to_owned(), "::1".to_owned()],
64            java_home: String::new(),
65            jar_path: String::new(),
66        }
67    }
68}
69
70impl BedrockConfig {
71    /// Resolves the actual port Geyser binds: `java_port` when
72    /// [`BedrockConfig::port`] is `0` ("share the Java server's port"),
73    /// otherwise `port` itself.
74    ///
75    /// This is the single place that resolution happens — a caller that
76    /// needs the real Bedrock port number calls this rather than reading
77    /// `port` directly, which would still be `0` when it means "shared".
78    #[must_use]
79    pub const fn resolved_port(&self, java_port: u16) -> u16 {
80        if self.port == 0 { java_port } else { self.port }
81    }
82
83    /// Whether [`BedrockConfig::username_prefix`] has stopped guaranteeing
84    /// that a Bedrock player's Java-side name (`username_prefix + gamertag`,
85    /// truncated to 16 characters) can never equal a real Java username.
86    ///
87    /// The guarantee holds only if the prefix contains at least one
88    /// character a vanilla Java username can never contain. Mojang restricts
89    /// Java usernames to ASCII letters, digits and `_`
90    /// (`foton_core::player::is_valid_player_name` enforces that same set on
91    /// this server's own Java logins) — a prefix outside that set, like the
92    /// default `"."`, therefore makes every derived name unspellable as a
93    /// real Java username. An empty prefix, or one built only from letters,
94    /// digits and `_`, gives up that guarantee: a Bedrock-derived name can
95    /// then be spelled identically to an existing or future Java player's.
96    ///
97    /// This crate does not depend on `foton-core`, so the character set is
98    /// checked directly here rather than by calling that function.
99    #[must_use]
100    pub fn username_prefix_could_collide_with_java_names(&self) -> bool {
101        !self
102            .username_prefix
103            .chars()
104            .any(|character| !(character.is_ascii_alphanumeric() || character == '_'))
105    }
106}

108#[cfg(test)]
109mod tests {
110    use super::BedrockConfig;
111
112    #[test]
113    fn the_default_is_off_and_loopback_only() {
114        // Both halves of this matter: a feature nobody asked for must not start
115        // a JVM, and a Floodgate handshake must not be trusted from anywhere
116        // but the Geyser this server launched.
117        let config = BedrockConfig::default();
118        assert!(!config.enable);
119        assert_eq!(config.trusted_proxies, vec!["127.0.0.1", "::1"]);
120    }
121
122    #[test]
123    fn resolved_port_defaults_to_sharing_the_java_servers_port() {
124        // `port`'s own default is `0`; `resolved_port` is what turns that
125        // into an actual port number.
126        let config = BedrockConfig::default();
127        assert_eq!(config.port, 0);
128        assert_eq!(config.resolved_port(25565), 25565);
129    }
130
131    #[test]
132    fn resolved_port_honors_an_explicit_port() {
133        let config = BedrockConfig {
134            port: 19132,
135            ..BedrockConfig::default()
136        };
137        assert_eq!(config.resolved_port(25565), 19132);
138    }
139
140    #[test]
141    fn the_default_prefix_cannot_collide_with_a_java_name() {
142        // "." is not a legal Java username character, so no derived name can
143        // ever be spelled the way a real Java username is.
144        let config = BedrockConfig::default();
145        assert!(!config.username_prefix_could_collide_with_java_names());
146    }
147
148    #[test]
149    fn an_empty_prefix_can_collide_with_a_java_name() {
150        let config = BedrockConfig {
151            username_prefix: String::new(),
152            ..BedrockConfig::default()
153        };
154        assert!(config.username_prefix_could_collide_with_java_names());
155    }
156
157    #[test]
158    fn an_alphanumeric_only_prefix_can_collide_with_a_java_name() {
159        // Built entirely from characters a real Java username can also
160        // contain, so the derived name is not structurally distinct.
161        let config = BedrockConfig {
162            username_prefix: "bed_".to_owned(),
163            ..BedrockConfig::default()
164        };
165        assert!(config.username_prefix_could_collide_with_java_names());
166    }
167
168    #[test]
169    fn a_prefix_with_one_illegal_character_cannot_collide_with_a_java_name() {
170        // Only one character in the prefix needs to fall outside a Java
171        // username's legal set for the whole derived name to become
172        // unspellable as a real Java username.
173        let config = BedrockConfig {
174            username_prefix: "bed-".to_owned(),
175            ..BedrockConfig::default()
176        };
177        assert!(!config.username_prefix_could_collide_with_java_names());
178    }
