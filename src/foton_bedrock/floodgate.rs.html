1//! Floodgate's identity payload, decoded.
2//!
3//! Geyser puts the Bedrock player's identity in the Java handshake's hostname,
4//! encrypted with a key both sides hold. This module is the reading half, and
5//! it is deliberately pure: bytes and a key in, a verified identity or an error
6//! out. No I/O, no clock, no process — so it is testable without a JVM.
7//!
8//! The format is read from `GeyserMC`'s own source, not recalled:
9//! `common/src/main/java/org/geysermc/floodgate/crypto/{FloodgateCipher,AesCipher}.java`.
10
11use aes_gcm::aead::{Aead, KeyInit, Payload};
12use aes_gcm::{Aes128Gcm, Nonce};
13use base64::Engine as _;
14use base64::engine::general_purpose::STANDARD;
15use thiserror::Error;
16
17/// The literal identifier text. [`IDENTIFIER`] is these bytes, and `encrypt`
18/// reuses this `&str` directly (`str::as_bytes` is valid in a const context),
19/// so there is exactly one spelling of the literal.
20const IDENTIFIER_STR: &str = "^Floodgate^";
21
22/// The 11 bytes that mark a hostname segment as Floodgate's.
23pub const IDENTIFIER: &[u8] = IDENTIFIER_STR.as_bytes();
24
25/// The format version this decoder understands.
26pub const VERSION: u8 = 0;
27
28/// `IDENTIFIER` followed by the encoded version byte: the first 12 bytes of
29/// every payload.
30pub const HEADER: [u8; 12] = [
31    b'^',
32    b'F',
33    b'l',
34    b'o',
35    b'o',
36    b'd',
37    b'g',
38    b'a',
39    b't',
40    b'e',
41    b'^',
42    0x3E + VERSION,
43];
44
45/// The IV length AES-GCM is used with here.
46pub const IV_LENGTH: usize = 12;
47
48/// Separates the base64 IV from the base64 ciphertext. Base64's alphabet never
49/// contains it, which is what makes a byte scan for it safe.
50const SPLITTER: u8 = b'!';
51
52/// Why a Floodgate payload was not accepted.
53#[derive(Debug, Error, PartialEq, Eq)]
54pub enum FloodgateError {
55    /// The data does not carry Floodgate's header at all.
56    #[error("not Floodgate data")]
57    NotFloodgateData,
58    /// The header is there but the envelope is not shaped correctly.
59    #[error("malformed Floodgate payload: {0}")]
60    Malformed(&'static str),
61    /// The ciphertext did not authenticate: wrong key, or tampered with.
62    #[error("Floodgate payload failed to decrypt")]
63    Decrypt,
64    /// The plaintext did not hold the twelve fields Floodgate sends.
65    #[error("expected 12 Floodgate fields, got {0}")]
66    WrongFieldCount(usize),
67    /// A field was present but could not be read as its type.
68    #[error("Floodgate field {0} could not be read")]
69    BadField(&'static str),
70}
71
72/// Finds the Floodgate segment in a handshake hostname.
73///
74/// Returns the payload and the hostname with that segment removed, matching
75/// `FloodgateHandshakeHandler.separateHostname`. Returns `None` for an ordinary
76/// Java client's hostname.
77#[must_use]
78pub fn extract_payload(hostname: &str) -> Option<(&str, String)> {
79    let mut payload = None;
80    let mut rest: Vec<&str> = Vec::new();
81
82    for segment in hostname.split('\0') {
83        if payload.is_none() && segment.as_bytes().starts_with(IDENTIFIER) {
84            payload = Some(segment);
85            continue;
86        }
87        rest.push(segment);
88    }
89
90    payload.map(|found| (found, rest.join("\0")))
91}

93/// Decrypts a Floodgate payload with the shared key.
94///
95/// The returned string is `BedrockData`'s null-separated form; parsing it into
96/// individual fields is a later task's job, not this module's.
97pub fn decrypt(payload: &[u8], key: &[u8; 16]) -> Result<String, FloodgateError> {
98    if payload.len() <= HEADER.len() || !payload.starts_with(IDENTIFIER) {
99        return Err(FloodgateError::NotFloodgateData);
100    }
101
102    let body = &payload[HEADER.len()..];
103    let split = body
104        .iter()
105        .position(|byte| *byte == SPLITTER)
106        .ok_or(FloodgateError::Malformed("no splitter"))?;
107
108    let iv_bytes = STANDARD
109        .decode(&body[..split])
110        .map_err(|_| FloodgateError::Malformed("iv is not base64"))?;
111    let iv = Nonce::try_from(iv_bytes.as_slice())
112        .map_err(|_| FloodgateError::Malformed("iv is the wrong length"))?;
113
114    let ciphertext = STANDARD
115        .decode(&body[split + 1..])
116        .map_err(|_| FloodgateError::Malformed("ciphertext is not base64"))?;
117
118    let cipher = Aes128Gcm::new(key.into());
119    let plaintext = cipher
120        .decrypt(
121            &iv,
122            Payload {
123                msg: &ciphertext,
124                aad: &[],
125            },
126        )
127        .map_err(|_| FloodgateError::Decrypt)?;
128
129    String::from_utf8(plaintext).map_err(|_| FloodgateError::Malformed("plaintext is not UTF-8"))
130}

132/// Encrypts a plaintext into a Floodgate wire envelope, the mirror of
133/// [`decrypt`].
134///
135/// This is ordinary public API, not a test helper: a module that implements a
136/// wire format owes both directions, and outbound Floodgate-shaped payloads
137/// are needed again once Foton drives Geyser itself (see the crate's `geyser`
138/// half). `GeyserMC`'s own encoder (`AesCipher.encrypt`) draws its IV from
139/// `SecureRandom` internally; taking `iv` as a parameter here instead keeps
140/// this function pure and its output reproducible.
141///
142/// # Errors
143///
144/// Returns [`FloodgateError::Malformed`] if AES-GCM rejects the plaintext —
145/// only possible near its multi-gigabyte single-nonce limit, never true for a
146/// handshake hostname, but the cipher's own API is fallible and this crate
147/// does not unwrap production code around it.
148pub fn encrypt(
149    plaintext: &str,
150    key: &[u8; 16],
151    iv: &[u8; IV_LENGTH],
152) -> Result<String, FloodgateError> {
153    let cipher = Aes128Gcm::new(key.into());
154    let ciphertext = cipher
155        .encrypt(
156            &Nonce::from(*iv),
157            Payload {
158                msg: plaintext.as_bytes(),
159                aad: &[],
160            },
161        )
162        .map_err(|_| FloodgateError::Malformed("plaintext too large to encrypt"))?;
163
164    // Reuses `IDENTIFIER_STR` rather than `IDENTIFIER`'s bytes, so this stays
165    // a total function: every piece appended is already a `&str` or comes
166    // from `Engine::encode`, which returns `String`, so there is no fallible
167    // UTF-8 conversion to handle without unwrapping it.
168    let mut out = String::from(IDENTIFIER_STR);
169    out.push(char::from(0x3E + VERSION));
170    out.push_str(&STANDARD.encode(iv));
171    out.push(char::from(SPLITTER));
172    out.push_str(&STANDARD.encode(ciphertext));
173    Ok(out)
174}

176/// The number of `\0`-separated fields Floodgate sends. `BedrockData.EXPECTED_LENGTH`.
177const EXPECTED_FIELDS: usize = 12;
178
179/// The longest name the Java protocol accepts, in **bytes** -- what every
180/// downstream consumer of [`BedrockData::java_username`] actually measures
181/// (`str::len()`), not [`char`] count.
182const MAX_JAVA_USERNAME: usize = 16;
183
184/// What Geyser says about the Bedrock player.
185///
186/// Field order is `BedrockData.toString()`'s, which is the order the fields are
187/// declared in Geyser's own class. Types stay as close to the wire as is useful:
188/// `xuid` is a string there and stays one here, because it is an identifier
189/// before it is a number.
190#[derive(Debug, Clone, PartialEq, Eq)]
191pub struct BedrockData {
192    /// The Bedrock client's version string.
193    pub version: String,
194    /// The Xbox Live gamertag.
195    pub username: String,
196    /// The Xbox user id, as a signed 64-bit integer in a string.
197    pub xuid: String,
198    /// Which platform the player is on.
199    pub device_os: i32,
200    /// The player's language, as `fr_FR`.
201    pub language_code: String,
202    /// Which UI the client presents.
203    pub ui_profile: i32,
204    /// Touch, controller, keyboard.
205    pub input_mode: i32,
206    /// The address Geyser saw the player come from.
207    pub ip: String,
208    /// A linked Java account, or `"null"`.
209    pub linked_player: String,
210    /// Whether the data passed through a proxy.
211    pub from_proxy: bool,
212    /// Skin upload correlation id.
213    pub subscribe_id: i32,
214    /// Skin upload verification code.
215    pub verify_code: String,
216}

218impl BedrockData {
219    /// Parses the decrypted payload.
220    ///
221    /// A field count other than twelve is refused rather than padded: Geyser
222    /// sends exactly twelve, so anything else is a version mismatch or a forgery,
223    /// and guessing which would be worse than saying no.
224    pub fn parse(plaintext: &str) -> Result<Self, FloodgateError> {
225        let fields: Vec<&str> = plaintext.split('\0').collect();
226        if fields.len() != EXPECTED_FIELDS {
227            return Err(FloodgateError::WrongFieldCount(fields.len()));
228        }
229
230        let number = |index: usize, name: &'static str| -> Result<i32, FloodgateError> {
231            fields[index]
232                .parse::<i32>()
233                .map_err(|_| FloodgateError::BadField(name))
234        };
235
236        Ok(Self {
237            version: fields[0].to_owned(),
238            username: fields[1].to_owned(),
239            xuid: fields[2].to_owned(),
240            device_os: number(3, "deviceOs")?,
241            language_code: fields[4].to_owned(),
242            ui_profile: number(5, "uiProfile")?,
243            input_mode: number(6, "inputMode")?,
244            ip: fields[7].to_owned(),
245            linked_player: fields[8].to_owned(),
246            from_proxy: fields[9] == "1",
247            subscribe_id: number(10, "subscribeId")?,
248            verify_code: fields[11].to_owned(),
249        })
250    }
251
252    /// The UUID this player gets on a Java server.
253    ///
254    /// `new UUID(0, xuid)`: zero in the high bits, the XUID in the low ones. It
255    /// is derived rather than stored, so it is the same on every join, which is
256    /// what makes player data, permissions and bans work unchanged.
257    pub fn java_uuid(&self) -> Result<uuid::Uuid, FloodgateError> {
258        let xuid: i64 = self
259            .xuid
260            .parse()
261            .map_err(|_| FloodgateError::BadField("xuid"))?;
262        Ok(uuid::Uuid::from_u64_pair(0, xuid as u64))
263    }
264
265    /// The name this player is known by, prefixed so it cannot collide with a
266    /// Java player's and truncated to what the Java protocol accepts.
267    ///
268    /// The bound is on **bytes**, not characters: every downstream consumer
269    /// (`c_login_finished`'s `Prefixed(VarInt), bound = 16` write, via
270    /// `foton_utils`' `write_prefixed_bound`, which measures `str::len()`)
271    /// enforces a 16-*byte* limit, not a 16-*character* one. Xbox gamertags
272    /// have supported non-Latin scripts since 2019, so a name that is 16
273    /// characters can easily be 17+ bytes; a character-counting truncation
274    /// produces exactly that value, which then fails to encode. With
275    /// `panic = "abort"` in the release profile, that turns into the whole
276    /// process aborting on an unprivileged remote player's gamertag.
277    ///
278    /// `prefix` shares the same byte budget as the gamertag (folded into one
279    /// `chars()` chain below) rather than being written unconditionally, so
280    /// even a prefix that is itself at or past the limit truncates safely
281    /// instead of producing a name already over budget before the gamertag is
282    /// considered. Truncation still lands on a character boundary, never a
283    /// byte one: slicing a multi-byte character in half would panic.
284    #[must_use]
285    pub fn java_username(&self, prefix: &str) -> String {
286        let mut name = String::with_capacity(MAX_JAVA_USERNAME);
287        for character in prefix.chars().chain(self.username.chars()) {
288            if name.len() + character.len_utf8() > MAX_JAVA_USERNAME {
289                break;
290            }
291            name.push(character);
292        }
293        name
294    }
295}

465    #[test]
466    fn truncates_on_a_character_boundary_and_stays_within_the_byte_bound() {
467        // 14 single-byte characters place the 2-byte "é" (U+00E9) so it
468        // straddles byte offset 15 -- exactly where the byte-bounded
469        // `java_username(".")` must stop (name is "." + 14 "A"s == 15 bytes
470        // before "é" is considered; 15 + 2 > 16, so "é" and everything after
471        // it is dropped). A naive `&self.username[..15]` byte slice would
472        // land on the second byte of "é" and panic; this must not. Built via
473        // `.repeat()`, not hand-counted, and self-checked with
474        // `is_char_boundary` so a miscount fails loudly here rather than
475        // silently passing a weaker assertion.
476        let username = format!("{}éXYZ", "A".repeat(14));
477        assert!(
478            !username.is_char_boundary(15),
479            "fixture must straddle byte offset 15 for this test to be load-bearing"
480        );
481
482        let data = BedrockData::parse(&sample(&username, "1")).expect("parses");
483        let name = data.java_username(".");
484        assert_eq!(name, format!(".{}", "A".repeat(14)));
485        assert!(name.len() <= 16, "name was {} bytes: {name:?}", name.len());
486    }

500    /// player with a Cyrillic gamertag could abort the whole server.

503    /// 16`, which the old implementation violated at 19 bytes) and passes
504    /// against the byte-bounded one.

507        let gamertag = "Ж".repeat(9);
