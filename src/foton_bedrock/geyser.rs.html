1//! The Geyser that turns Bedrock into Java, run as a child of this server.
2//!
3//! Not in the JVM `foton-plugin` starts: a process gets one `JNI_CreateJavaVM`,
4//! and sharing it would make Bedrock support depend on the plugin host's
5//! lifecycle. A child process costs one process and buys isolation — Geyser
6//! crashing restarts Geyser, and an operator who runs no Bedrock runs no JVM.
7//!
8//! [`render_config`]'s key names, and the section layout they live in, were
9//! read from a real `config.yml` a pinned Geyser 2.11.2 build 1233 generated
10//! for itself — never guessed from Geyser's own documentation, an older
11//! release's schema, or memory. `design/bedrock-stage0-findings.md`, Step 3
12//! and its addendum, records every key this module writes and the re-run
13//! that confirmed a partially-filled config of exactly this shape starts
14//! cleanly against the pinned build, with every override (including an
15//! escaped MOTD) surviving intact.
16

61/// Bounded total timeout for [`download`] — covering DNS, connect, TLS and
62/// the whole response body, not just a between-bytes idle window.
63///
64/// [`Supervisor::start`] runs synchronously during server startup, after the
65/// Java port is already bound but before the accept loop runs. A download
66/// with no timeout at all means a hung connection to
67/// `download.geysermc.org` blocks startup forever: the Java port sits bound
68/// with nothing accepting on it, which a connecting client cannot tell apart
69/// from a dead server. This bound turns that into an ordinary
70/// [`GeyserError::Download`] that the existing log-and-continue policy
71/// handles, instead of a silent hang.
72///
73/// The pinned jar is roughly 10 MB; at a conservatively slow 1 Mbps
74/// (~125 KB/s) connection that is about 80 seconds. 120 seconds leaves
75/// headroom for connection setup and jitter on top of that while staying
76/// finite. `dev/doctor.sh`'s `curl --max-time 10` is sized for a small API
77/// ping, not a multi-megabyte transfer — too short a bound to reuse here.
78const DOWNLOAD_TIMEOUT: Duration = Duration::from_secs(120);
79
80/// Bounded timeout for the `java -version` probe in [`resolve_java`].
81///
82/// [`Supervisor::start`] runs synchronously during server startup, in the
83/// same window [`DOWNLOAD_TIMEOUT`]'s documentation describes: the Java port
84/// is already bound but the accept loop has not started, so an unbounded
85/// wait here is indistinguishable from a dead server to a connecting
86/// client, not merely a slow one. A wedged `java` binary -- a stale NFS
87/// mount, a broken JVM install, a filesystem that never answers -- would
88/// otherwise block startup forever.
89///
90/// A version probe is a different scale of wait than the jar download,
91/// though: it launches a JVM just long enough to print a banner and exit,
92/// no class loading beyond the JDK's own bootstrap, no network, no user
93/// code. On any host where `java` actually works this answers in well
94/// under a second. Ten seconds is two orders of magnitude above that --
95/// generous enough to absorb a loaded CI runner or a cold page cache on
96/// first launch, while still turning a truly wedged binary into the same
97/// finite, logged-and-continued [`GeyserError::JavaNotFound`] the "could
98/// not even start the binary" branch already produces, instead of a
99/// silent hang. A healthy system pays nothing extra for this: the timer
100/// races the probe and is simply dropped once the probe answers, which it
101/// always does long before ten seconds pass.
102const JAVA_VERSION_PROBE_TIMEOUT: Duration = Duration::from_secs(10);

104/// Why the supervisor could not start or keep running Geyser.
105#[derive(Debug, Error)]
106pub enum GeyserError {
107    /// No `java` binary could be run at all.
108    #[error(
109        "no Java runtime found at {searched} ({source}); set bedrock.java_home to a JDK/JRE of Java {MIN_JAVA_VERSION} or newer, install one on the PATH, or set JAVA_HOME"
110    )]
111    JavaNotFound {
112        /// The binary path the supervisor tried to run.
113        searched: String,
114        /// Why running it failed.
115        #[source]
116        source: io::Error,
117    },
118    /// `java -version` ran, but its banner did not contain a version this
119    /// crate knows how to read.
120    #[error("could not read a Java version from {path}: {reason}")]
121    JavaVersionUnreadable {
122        /// The runtime that was probed.
123        path: String,
124        /// The output that could not be parsed.
125        reason: String,
126    },
127    /// A working `java` was found, but it reports a major version below
128    /// [`MIN_JAVA_VERSION`].
129    #[error(
130        "found Java {found} at {path}, but Geyser needs Java {MIN_JAVA_VERSION} or newer; point bedrock.java_home at a newer JDK/JRE"
131    )]
132    JavaTooOld {
133        /// The runtime that was found.
134        path: String,
135        /// The major version it reported.
136        found: u32,
137    },
138    /// The pinned jar could not be downloaded — including because it did not
139    /// finish within [`DOWNLOAD_TIMEOUT`].
140    #[error(
141        "failed to download Geyser from {url}: {source}; set bedrock.jar_path to a Geyser jar already on disk instead of relying on this download"
142    )]
143    Download {
144        /// The URL the download was attempted from.
145        url: String,
146        /// The underlying HTTP error.
147        #[source]
148        source: reqwest::Error,
149    },
150    /// A jar's bytes did not match [`GEYSER_SHA256`], whether it was already
151    /// on disk or freshly downloaded.
152    #[error(
153        "jar does not match the pinned Geyser build (expected sha256 {expected}, got {actual}); delete it and let Foton re-download the pinned build, or point bedrock.jar_path at a jar with a matching checksum"
154    )]
155    ChecksumMismatch {
156        /// The checksum this build is pinned to.
157        expected: String,
158        /// What the jar's bytes actually hashed to.
159        actual: String,
160    },
161    /// A file under the Bedrock directory could not be read or written.
162    #[error("failed to access {path}: {source}")]
163    Io {
164        /// The file that could not be read or written.
165        path: PathBuf,
166        /// The underlying I/O error.
167        #[source]
168        source: io::Error,
169    },
170    /// Starting the `java` process itself failed.
171    #[error("failed to start Geyser ({java} -jar {jar}): {source}")]
172    Spawn {
173        /// The Java binary that could not be run.
174        java: String,
175        /// The jar that could not be run.
176        jar: String,
177        /// The underlying I/O error.
178        #[source]
179        source: io::Error,
180    },
181}

183/// Everything the supervisor needs to run this server's Geyser.
184///
185/// A caller (the server, at startup) builds one of these from
186/// [`BedrockConfig`](crate::config::BedrockConfig) and passes it to
187/// [`Supervisor::start`].
188#[derive(Debug, Clone)]
189pub struct GeyserOptions {
190    /// This server's own run directory; `bedrock/` under it holds Geyser's
191    /// jar, its generated `config.yml`, and the shared Floodgate key.
192    ///
193    /// May be relative when a caller builds this: [`Supervisor::start`]
194    /// resolves it to an absolute path itself, before anything is derived
195    /// from it, since Geyser runs with a working directory one level deeper
196    /// (`bedrock_dir`) and a relative path would resolve against the wrong
197    /// place from there.
198    pub run_directory: PathBuf,
199    /// The UDP port Bedrock clients connect to.
200    ///
201    /// Already resolved — a caller building this from
202    /// [`BedrockConfig`](crate::config::BedrockConfig) must call
203    /// [`BedrockConfig::resolved_port`](crate::config::BedrockConfig::resolved_port)
204    /// first. Unlike [`BedrockConfig::port`](crate::config::BedrockConfig::port),
205    /// `0` here has no special meaning; it is simply not a usable port.
206    pub bedrock_port: u16,
207    /// This Java server's own port, which Geyser is told to connect to on
208    /// loopback.
209    pub java_port: u16,
210    /// What Bedrock clients see in the server list; empty reuses the Java
211    /// server's own MOTD instead (see [`render_config`]).
212    pub motd: String,
213    /// A JDK/JRE to run Geyser with; `None` falls back to `JAVA_HOME`, then
214    /// `java` on the `PATH`. May be relative -- it is made absolute before
215    /// use, for the same reason as [`GeyserOptions::run_directory`].
216    pub java_home: Option<PathBuf>,
217    /// An operator-supplied Geyser jar; `None` fetches and caches the pinned
218    /// build. May be relative -- it is made absolute before use, for the
219    /// same reason as [`GeyserOptions::run_directory`].
220    pub jar_path: Option<PathBuf>,
221}

223/// Escapes a string into a YAML double-quoted scalar, backslash first so the
224/// escape for `"` is not itself re-escaped.
225///
226/// Every operator-supplied string that reaches [`render_config`] — today,
227/// only the MOTD — goes through this. A config file is a parser, and an
228/// unescaped MOTD is exactly how operator input reaching a parser adds a key
229/// instead of setting one: an embedded `"` closes the scalar early, and
230/// anything after it is read as new YAML.
231fn yaml_quote(value: &str) -> String {
232    let mut escaped = String::with_capacity(value.len() + 2);
233    escaped.push('"');
234    for character in value.chars() {
235        match character {
236            '\\' => escaped.push_str("\\\\"),
237            '"' => escaped.push_str("\\\""),
238            '\n' => escaped.push_str("\\n"),
239            '\r' => escaped.push_str("\\r"),
240            '\t' => escaped.push_str("\\t"),
241            other => escaped.push(other),
242        }
243    }
244    escaped.push('"');
245    escaped
246}

248/// Renders Geyser's `config.yml`, from scratch, every time.
249///
250/// Foton owns this file: nothing here reads an existing `config.yml` back or
251/// tries to preserve edits Geyser made to it (Geyser rewrites `metrics-uuid`,
252/// migrates `config-version`, and fills in every key this function leaves
253/// out with its own defaults — confirmed by running the pinned build against
254/// a config of exactly this shape, `design/bedrock-stage0-findings.md`'s
255/// Step 3 addendum). Every [`Supervisor::start`] regenerates
256/// the whole file from `options`, so the operator's actual settings are the
257/// only source of truth, never whatever the previous Geyser process happened
258/// to leave on disk. The emitted file opens with a comment saying exactly
259/// that, so an operator who opens `config.yml` directly is not surprised the
260/// next time an edit they made there does not survive a restart.
261///
262/// An empty [`GeyserOptions::motd`] emits `passthrough-motd: true` and no
263/// `primary-motd` at all, so Geyser relays the Java server's own MOTD to
264/// Bedrock clients — the mechanism
265/// [`BedrockConfig::motd`](crate::config::BedrockConfig::motd)'s own doc
266/// comment promises ("empty reuses the server MOTD"). A non-empty one emits
267/// `passthrough-motd: false` and the quoted MOTD, so Geyser shows it instead.
268#[must_use]
269pub fn render_config(options: &GeyserOptions) -> String {
270    let key_file = key::key_path(&options.run_directory);
271    let motd_section = if options.motd.is_empty() {
272        "motd:\n  passthrough-motd: true\n".to_owned()
273    } else {
274        format!(
275            "motd:\n  primary-motd: {motd}\n  passthrough-motd: false\n",
276            motd = yaml_quote(&options.motd),
277        )
278    };
279
280    format!(
281        "\
282# Generated by Foton on every start -- edits made here are lost on the next
283# one. Configure Bedrock support through [server.bedrock] in Foton's own
284# config instead.
285bedrock:
286  address: 0.0.0.0
287  port: {bedrock_port}
288java:
289  address: 127.0.0.1
290  port: {java_port}
291  auth-type: floodgate
292  forward-hostname: false
293{motd_section}advanced:
294  floodgate-key-file: {key_file}
295  bedrock:
296    validate-bedrock-login: true
297",
298        bedrock_port = options.bedrock_port,
299        java_port = options.java_port,
300        key_file = yaml_quote(&key_file.display().to_string()),
301    )
302}

304/// Renders `bytes` as lowercase hex.
305///
306/// A hand-rolled loop rather than a `LowerHex` format: `sha2`'s digest output
307/// is a fixed-size array type from the `hybrid-array`/`crypto-common` stack
308/// that does not implement it, and pulling in the `hex` crate for one call
309/// site is not worth a sixth dependency this module's brief did not ask for.
310fn to_hex(bytes: &[u8]) -> String {
311    use std::fmt::Write as _;
312    let mut hex = String::with_capacity(bytes.len() * 2);
313    for byte in bytes {
314        // `Write for String` never fails.
315        let _ignored = write!(hex, "{byte:02x}");
316    }
317    hex
318}

320/// Hashes `bytes` with SHA-256 and compares the digest against `expected`,
321/// case-insensitively — a checksum pasted from a webpage may be either case.
322///
323/// # Errors
324///
325/// Returns [`GeyserError::ChecksumMismatch`] if the digests differ.
326pub fn verify_checksum(bytes: &[u8], expected: &str) -> Result<(), GeyserError> {
327    let digest = Sha256::digest(bytes);
328    let actual = to_hex(&digest);
329    if actual.eq_ignore_ascii_case(expected) {
330        Ok(())
331    } else {
332        Err(GeyserError::ChecksumMismatch {
333            expected: expected.to_owned(),
334            actual,
335        })
336    }
337}

339/// Resolves `path` to an absolute one.
340///
341/// Every path this module ultimately hands to [`Command::new`]/[`arg`] as a
342/// filesystem path -- the Java binary, the jar -- crosses a `current_dir`
343/// boundary: [`spawn_geyser`] sets the child's working directory to
344/// `bedrock_dir`, one level deeper than where these paths are usually
345/// resolved from. A relative path is not rebased against that -- it is
346/// handed to the OS exec call as a literal string, resolved by the *child*
347/// against *its own* cwd. Making it absolute first, while still running with
348/// this process's own (correct) working directory, is what avoids that.
349///
350/// `absolute` (`std::path::absolute`) rather than `std::fs::canonicalize`:
351/// the latter requires the target to already exist -- not yet true for a
352/// jar this call is about to download -- and on Windows returns a
353/// `\\?\`-prefixed UNC path that not every tool reading it back (Geyser's
354/// own JVM included) is guaranteed to accept.
355///
356/// [`arg`]: tokio::process::Command::arg
357fn to_absolute(path: &Path) -> Result<PathBuf, GeyserError> {
358    absolute(path).map_err(|source| GeyserError::Io {
359        path: path.to_path_buf(),
360        source,
361    })
362}

364/// The `java` executable's conventional name under a JDK/JRE's `bin/`.
365const fn java_binary_name() -> &'static str {
366    if cfg!(windows) { "java.exe" } else { "java" }
367}
368
369/// Resolves which `java` binary to run: `java_home` if given, else
370/// `JAVA_HOME`, else a bare name resolved against `PATH`.
371///
372/// A `home` from either source is made absolute (see [`to_absolute`]) before
373/// `bin`/the binary name is joined onto it, since the result is handed to
374/// [`Command::new`] in [`spawn_geyser`]. The bare-name fallback is
375/// deliberately left alone: `Command` resolves a separator-free program name
376/// against `PATH`, which does not depend on either process's working
377/// directory, so making it absolute here would break that lookup rather than
378/// fix anything.
379fn java_binary_path(java_home: Option<&Path>) -> Result<PathBuf, GeyserError> {
380    let home = java_home
381        .map(Path::to_path_buf)
382        .or_else(|| env::var_os("JAVA_HOME").map(PathBuf::from));
383    match home {
384        Some(home) => Ok(to_absolute(&home)?.join("bin").join(java_binary_name())),
385        None => Ok(PathBuf::from(java_binary_name())),
386    }
387}

389/// Reads the major version number out of `java -version`'s banner line, e.g.
390/// `openjdk version "21.0.5" 2024-10-15`, or the legacy `java version
391/// "1.8.0_292"` form, where the real major version is the second component.
392fn parse_java_major_version(banner: &str) -> Option<u32> {
393    let quoted = banner.split('"').nth(1)?;
394    let mut parts = quoted.split(['.', '-', '_', '+']);
395    let first: u32 = parts.next()?.parse().ok()?;
396    if first == 1 {
397        parts.next()?.parse().ok()
398    } else {
399        Some(first)
400    }
401}

403/// Resolves and validates the Java runtime Geyser will run under.
404///
405/// `probe_timeout` is a parameter rather than reading [`JAVA_VERSION_PROBE_TIMEOUT`]
406/// directly -- the same shape as [`download`] taking a `timeout` -- so a test
407/// can bound a deliberately wedged fake `java` on a short leash instead of
408/// waiting out the real ten seconds.
409async fn resolve_java(
410    java_home: Option<&Path>,
411    probe_timeout: Duration,
412) -> Result<PathBuf, GeyserError> {
413    let binary = java_binary_path(java_home)?;
414
415    let probe = Command::new(&binary)
416        .arg("-version")
417        // If the timeout below fires, this future -- and the `Child` it
418        // owns -- is dropped without ever being awaited to completion.
419        // Without `kill_on_drop`, a wedged `java` would be left running as
420        // an orphan process even after this function has already given up
421        // on it and moved on.
422        .kill_on_drop(true)
423        .output();
424    let output = time::timeout(probe_timeout, probe)
425        .await
426        .map_err(|_elapsed| GeyserError::JavaNotFound {
427            searched: binary.display().to_string(),
428            source: io::Error::new(
429                io::ErrorKind::TimedOut,
430                format!("no response within {probe_timeout:?}"),
431            ),
432        })?
433        .map_err(|source| GeyserError::JavaNotFound {
434            searched: binary.display().to_string(),
435            source,
436        })?;
437
438    // `java -version` writes its banner to stderr, not stdout.
439    let banner = String::from_utf8_lossy(&output.stderr);
440    let major =
441        parse_java_major_version(&banner).ok_or_else(|| GeyserError::JavaVersionUnreadable {
442            path: binary.display().to_string(),
443            reason: banner.trim().to_owned(),
444        })?;
445
446    if major < MIN_JAVA_VERSION {
447        return Err(GeyserError::JavaTooOld {
448            path: binary.display().to_string(),
449            found: major,
450        });
451    }
452
453    Ok(binary)
454}

456/// Downloads `url` whole into memory, aborting the whole attempt — DNS,
457/// connect, TLS and body included — if it has not finished within `timeout`.
458/// The pinned jar is tens of megabytes, so buffering it rather than
459/// streaming to disk keeps the checksum check — which must happen before
460/// anything is written — simple.
461///
462/// A dedicated [`reqwest::Client`] is built for this rather than the
463/// crate-wide convenience of `reqwest::get`, because that bare function uses
464/// a lazily-initialized client with no timeout at all — exactly the hang
465/// this bound exists to prevent.
466async fn download(url: &str, timeout: Duration) -> Result<Vec<u8>, GeyserError> {
467    let wrap = |source: reqwest::Error| GeyserError::Download {
468        url: url.to_owned(),
469        source,
470    };
471    let client = reqwest::Client::builder()
472        .timeout(timeout)
473        .build()
474        .map_err(wrap)?;
475    let response = client
476        .get(url)
477        .send()
478        .await
479        .map_err(wrap)?
480        .error_for_status()
481        .map_err(wrap)?;
482    let bytes = response.bytes().await.map_err(wrap)?;
483    Ok(bytes.to_vec())
484}

486/// Picks the jar path [`resolve_jar`] should use, made absolute (see
487/// [`to_absolute`]) either way: an operator-supplied [`GeyserOptions::jar_path`]
488/// may itself be relative, and the pinned-build fallback joined onto
489/// `bedrock_dir` inherits whatever `bedrock_dir` is -- absolute only because
490/// [`Supervisor::start`] already made `options.run_directory` absolute
491/// before deriving it. Kept separate from [`resolve_jar`] because this half
492/// does no I/O, so it can be tested without a filesystem or a JVM.
493fn jar_candidate_path(options: &GeyserOptions, bedrock_dir: &Path) -> Result<PathBuf, GeyserError> {
494    let jar_path = options
495        .jar_path
496        .clone()
497        .unwrap_or_else(|| bedrock_dir.join(JAR_FILE_NAME));
498    to_absolute(&jar_path)
499}

501/// Resolves the Geyser jar: an operator-supplied one, or the pinned build
502/// cached under `bedrock_dir`, downloading it first if absent. Either way,
503/// the bytes are checked against [`GEYSER_SHA256`] before this returns.
504async fn resolve_jar(options: &GeyserOptions, bedrock_dir: &Path) -> Result<PathBuf, GeyserError> {
505    let jar_path = jar_candidate_path(options, bedrock_dir)?;
506
507    if jar_path.is_file() {
508        let bytes = fs::read(&jar_path)
509            .await
510            .map_err(|source| GeyserError::Io {
511                path: jar_path.clone(),
512                source,
513            })?;
514        verify_checksum(&bytes, GEYSER_SHA256)?;
515        return Ok(jar_path);
516    }
517
518    tracing::info!(
519        target: "geyser",
520        "downloading Geyser {GEYSER_VERSION} build {GEYSER_BUILD} from {GEYSER_URL}"
521    );
522    let bytes = download(GEYSER_URL, DOWNLOAD_TIMEOUT).await?;
523    verify_checksum(&bytes, GEYSER_SHA256)?;
524    fs::write(&jar_path, &bytes)
525        .await
526        .map_err(|source| GeyserError::Io {
527            path: jar_path.clone(),
528            source,
529        })?;
530    Ok(jar_path)
531}

533/// Writes `config.yml` and ensures the shared Floodgate key exists, so both
534/// files are in place before Geyser's first start — a key created only after
535/// Geyser has already read a missing one fails silently from an operator's
536/// point of view (`design/bedrock-stage0-findings.md`, Step 3).
537async fn write_config_and_key(
538    options: &GeyserOptions,
539    bedrock_dir: &Path,
540) -> Result<(), GeyserError> {
541    let config_path = bedrock_dir.join("config.yml");
542    let yaml = render_config(options);
543    fs::write(&config_path, yaml)
544        .await
545        .map_err(|source| GeyserError::Io {
546            path: config_path,
547            source,
548        })?;
549
550    let key_path = key::key_path(&options.run_directory);
551    let key_path_for_blocking = key_path.clone();
552    // `key::load_or_create` is ordinary blocking `std::fs`, shared with the
553    // server's own startup path — run it off the async executor rather than
554    // duplicating it as an async version for this one caller.
555    task::spawn_blocking(move || key::load_or_create(&key_path_for_blocking))
556        .await
557        .map_err(|join_error| GeyserError::Io {
558            path: key_path.clone(),
559            source: io::Error::other(join_error),
560        })?
561        .map_err(|source| GeyserError::Io {
562            path: key_path,
563            source,
564        })?;
565
566    Ok(())
567}

569/// Spawns `java -jar <jar_path>`, with `bedrock_dir` as its working
570/// directory (so it reads and writes `config.yml`, `key.pem`, its cache and
571/// its logs there) and its stdout/stderr piped for [`relay_child_logs`].
572///
573/// Not `async`: `Command::spawn` starts the process and returns immediately,
574/// it does not wait on it.
575fn spawn_geyser(
576    java_binary: &Path,
577    jar_path: &Path,
578    bedrock_dir: &Path,
579) -> Result<Child, GeyserError> {
580    Command::new(java_binary)
581        .arg("-jar")
582        .arg(jar_path)
583        .current_dir(bedrock_dir)
584        .stdout(Stdio::piped())
585        .stderr(Stdio::piped())
586        .kill_on_drop(true)
587        .spawn()
588        .map_err(|source| GeyserError::Spawn {
589            java: java_binary.display().to_string(),
590            jar: jar_path.display().to_string(),
591            source,
592        })
593}

595/// Reads Geyser's own leading level marker out of a console line, e.g. the
596/// `INFO` in `[12:00:56 INFO] Started Geyser on UDP port 19132`, and maps it
597/// onto a `tracing` level. Unrecognized or missing markers default to `info`.
598fn geyser_log_level(line: &str) -> tracing::Level {
599    let marker = line.split(']').next().unwrap_or(line).to_ascii_uppercase();
600    if marker.contains("ERROR") || marker.contains("SEVERE") {
601        tracing::Level::ERROR
602    } else if marker.contains("WARN") {
603        tracing::Level::WARN
604    } else if marker.contains("DEBUG") || marker.contains("FINE") || marker.contains("TRACE") {
605        tracing::Level::DEBUG
606    } else {
607        tracing::Level::INFO
608    }
609}
610
611/// Re-emits one line of Geyser's console through `tracing`, under
612/// `target: "geyser"`.
613fn emit_geyser_line(line: &str) {
614    match geyser_log_level(line) {
615        tracing::Level::ERROR => tracing::error!(target: "geyser", "{line}"),
616        tracing::Level::WARN => tracing::warn!(target: "geyser", "{line}"),
617        tracing::Level::DEBUG | tracing::Level::TRACE => {
618            tracing::debug!(target: "geyser", "{line}");
619        }
620        tracing::Level::INFO => tracing::info!(target: "geyser", "{line}"),
621    }
622}
623
624/// Reads `stream` line by line until it closes or errors, relaying each line
625/// through [`emit_geyser_line`].
626async fn relay_stream<R>(stream: R)
627where
628    R: AsyncRead + Unpin + Send + 'static,
629{
630    let mut lines = BufReader::new(stream).lines();
631    loop {
632        match lines.next_line().await {
633            Ok(Some(line)) => emit_geyser_line(&line),
634            Ok(None) => return,
635            Err(error) => {
636                tracing::warn!(target: "geyser", "Geyser's log stream ended unexpectedly: {error}");
637                return;
638            }
639        }
640    }
641}
642
643/// Spawns one relay task per stream Geyser writes to, taking ownership of
644/// each handle so this can only be called once per child.
645fn relay_child_logs(child: &mut Child) {
646    if let Some(stdout) = child.stdout.take() {
647        tokio::spawn(relay_stream(stdout));
648    }
649    if let Some(stderr) = child.stderr.take() {
650        tokio::spawn(relay_stream(stderr));
651    }
652}

654/// Sends the platform's polite termination request, best-effort.
655///
656/// On Unix this is `SIGTERM`, which lets the JVM run its shutdown hooks — the
657/// `kill` binary is used for the one syscall rather than adding `libc`/`nix`
658/// as a dependency for it. Windows has no reachable equivalent without extra
659/// platform bindings this crate does not otherwise need; [`terminate`]'s
660/// timeout and subsequent [`Child::kill`] still apply there.
661#[cfg(unix)]
662async fn request_graceful_stop(child: &Child) {
663    let Some(pid) = child.id() else {
664        return;
665    };
666    let _ignored = Command::new("kill")
667        .arg("-TERM")
668        .arg(pid.to_string())
669        .status()
670        .await;
671}
672
673/// See the Unix implementation's documentation.
674#[cfg(not(unix))]
675#[expect(
676    clippy::unused_async,
677    reason = "the signature has to match the Unix implementation `terminate` awaits"
678)]
679async fn request_graceful_stop(_child: &Child) {}
680
681/// Stops `child`: a polite termination request, up to [`TERMINATE_GRACE`] to
682/// take effect, then an unconditional kill if it hasn't.
683async fn terminate(child: &mut Child) {
684    request_graceful_stop(child).await;
685
686    if time::timeout(TERMINATE_GRACE, child.wait()).await.is_err() {
687        tracing::warn!(
688            target: "geyser",
689            "Geyser did not stop within {TERMINATE_GRACE:?}; killing it"
690        );
691        let _ignored = child.kill().await;
692        let _ignored = child.wait().await;
693    }
694}

696/// Logs why a Geyser process just exited.
697fn log_unexpected_exit(status: &Result<ExitStatus, io::Error>) {
698    match status {
699        Ok(status) => {
700            tracing::warn!(target: "geyser", "Geyser exited unexpectedly ({status}); restarting");
701        }
702        Err(error) => {
703            tracing::warn!(target: "geyser", "lost track of the Geyser process ({error}); restarting");
704        }
705    }
706}

708/// Waits out the current backoff (or returns `None` if cancelled first),
709/// then tries to spawn a fresh Geyser, retrying under the same backoff and
710/// give-up policy if the spawn itself fails.
711///
712/// Returns `None` once [`MAX_CONSECUTIVE_FAILURES`] is reached without a
713/// successful spawn, or if `cancel` fires while waiting — either way, the
714/// caller has nothing left to supervise.
715async fn respawn(
716    java_binary: &Path,
717    jar_path: &Path,
718    bedrock_dir: &Path,
719    cancel: &CancellationToken,
720    backoff: &mut Duration,
721    consecutive_failures: &mut u32,
722) -> Option<Child> {
723    loop {
724        *consecutive_failures += 1;
725        if *consecutive_failures >= MAX_CONSECUTIVE_FAILURES {
726            tracing::error!(
727                target: "geyser",
728                "Geyser has failed {consecutive_failures} times in a row; not restarting it again",
729                consecutive_failures = *consecutive_failures
730            );
731            return None;
732        }
733
734        tracing::info!(
735            target: "geyser",
736            "restarting Geyser in {backoff:?} (attempt {consecutive_failures})",
737            backoff = *backoff,
738            consecutive_failures = *consecutive_failures
739        );
740        tokio::select! {
741            () = time::sleep(*backoff) => {}
742            () = cancel.cancelled() => return None,
743        }
744        *backoff = (*backoff * 2).min(MAX_BACKOFF);
745
746        match spawn_geyser(java_binary, jar_path, bedrock_dir) {
747            Ok(mut child) => {
748                relay_child_logs(&mut child);
749                return Some(child);
750            }
751            Err(error) => {
752                tracing::error!(target: "geyser", "failed to restart Geyser: {error}");
753            }
754        }
755    }
756}

758/// Supervises an already-running Geyser `child` until `cancel` fires: waits
759/// for it to exit, restarts it with backoff, and gives up rather than
760/// spinning forever if it keeps failing immediately.
761async fn supervise(
762    java_binary: PathBuf,
763    jar_path: PathBuf,
764    bedrock_dir: PathBuf,
765    mut child: Child,
766    cancel: CancellationToken,
767) {
768    let mut backoff = INITIAL_BACKOFF;
769    let mut consecutive_failures: u32 = 0;
770
771    loop {
772        let started_at = Instant::now();
773        let exited = tokio::select! {
774            status = child.wait() => Some(status),
775            () = cancel.cancelled() => None,
776        };
777
778        let Some(status) = exited else {
779            terminate(&mut child).await;
780            return;
781        };
782
783        log_unexpected_exit(&status);
784
785        if started_at.elapsed() >= MIN_HEALTHY_UPTIME {
786            consecutive_failures = 0;
787            backoff = INITIAL_BACKOFF;
788        }
789
790        let Some(next) = respawn(
791            &java_binary,
792            &jar_path,
793            &bedrock_dir,
794            &cancel,
795            &mut backoff,
796            &mut consecutive_failures,
797        )
798        .await
799        else {
800            return;
801        };
802        child = next;
803    }
804}

806/// Runs and supervises this server's Geyser process.
807///
808/// Dropping a `Supervisor` does not stop Geyser — the supervision task keeps
809/// running until its `CancellationToken` is cancelled. Cancel it and then
810/// call [`Supervisor::wait`] for an orderly shutdown.
811pub struct Supervisor {
812    handle: task::JoinHandle<()>,
813}
814
815impl Supervisor {
816    /// Resolves Java, fetches and verifies the pinned jar if needed, writes
817    /// `config.yml` and the shared key, starts Geyser, and returns once it
818    /// is running. From here on, keeping it running is a background task's
819    /// job, not the caller's — see the module documentation.
820    ///
821    /// # Errors
822    ///
823    /// Returns [`GeyserError`] if `run_directory` cannot be resolved to an
824    /// absolute path, no Java [`MIN_JAVA_VERSION`]+ runtime can be found, the
825    /// pinned jar cannot be fetched or does not match its checksum, the
826    /// Bedrock directory's files cannot be written, or the process itself
827    /// cannot be spawned. None of these are retried here — only a process
828    /// that started successfully and later exits is restarted automatically.
829    pub async fn start(
830        mut options: GeyserOptions,
831        cancel: CancellationToken,
832    ) -> Result<Self, GeyserError> {
833        // Made absolute here, defensively, rather than trusted from the
834        // caller: every path this function and the ones it calls derive
835        // from `run_directory` -- `bedrock_dir`, the jar's default location,
836        // the Floodgate key path written into `config.yml` -- is handed
837        // either directly or indirectly to a child process whose own
838        // working directory is `bedrock_dir`, one level deeper. A relative
839        // `run_directory` would make every one of those resolve against the
840        // wrong place from the child's point of view. See [`to_absolute`].
841        options.run_directory = to_absolute(&options.run_directory)?;
842
843        let bedrock_dir = options.run_directory.join("bedrock");
844        fs::create_dir_all(&bedrock_dir)
845            .await
846            .map_err(|source| GeyserError::Io {
847                path: bedrock_dir.clone(),
848                source,
849            })?;
850
851        let java_binary =
852            resolve_java(options.java_home.as_deref(), JAVA_VERSION_PROBE_TIMEOUT).await?;
853        let jar_path = resolve_jar(&options, &bedrock_dir).await?;
854        write_config_and_key(&options, &bedrock_dir).await?;
855
856        let mut child = spawn_geyser(&java_binary, &jar_path, &bedrock_dir)?;
857        relay_child_logs(&mut child);
858
859        let handle = tokio::spawn(supervise(java_binary, jar_path, bedrock_dir, child, cancel));
860
861        Ok(Self { handle })
862    }
863
864    /// Waits for the supervisor to stop — after its `CancellationToken` is
865    /// cancelled, or it gives up after too many crashes in a row.
866    pub async fn wait(self) {
867        // The task only ever ends by returning; nothing here aborts it, so a
868        // join error is not a case this needs to report.
869        let _ignored = self.handle.await;
870    }
871}

1152    async fn a_wedged_java_probe_times_out_instead_of_hanging_forever() {

1154        use std::io::ErrorKind;
1155        use std::os::unix::fs::PermissionsExt as _;
1156
1157        use super::resolve_java;
1158
1159        // A `java` that never answers -- the same shape as the stale NFS
1160        // mount or broken JVM install this timeout exists to survive,
1161        // without needing either to actually reproduce a hung filesystem.
1162        let java_home = tempfile::tempdir().expect("temp dir");
1163        let bin = java_home.path().join("bin");
1164        create_dir_all(&bin).expect("creates bin dir");
1165        let java = bin.join("java");
1166        write(&java, "#!/bin/sh\nsleep 5\n").expect("writes the fake java script");
1167        set_permissions(&java, Permissions::from_mode(0o755))
1168            .expect("makes the fake java script executable");

1170        let result = resolve_java(Some(java_home.path()), Duration::from_millis(200)).await;
1171
1172        match result {
1173            Err(GeyserError::JavaNotFound { source, .. }) => {
1174                assert_eq!(
1175                    source.kind(),
1176                    ErrorKind::TimedOut,
1177                    "expected the timeout's own error, got: {source:?}"
1178                );
1179            }
1180            other => panic!("expected a timed-out GeyserError::JavaNotFound, got: {other:?}"),
1181        }
