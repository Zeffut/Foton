1//! The secret Foton and its Geyser share.
2//!
3//! Sixteen raw bytes, which is what Geyser's `AesKeyProducer.produceFrom` reads
4//! a key file as. Not PEM despite the conventional name -- Geyser reads the
5//! file whole and hands it to `SecretKeySpec`.
6//!
7//! Everything about a Bedrock player's identity rests on this file staying
8//! secret: it is the only thing standing between a forged handshake and total
9//! impersonation.
10
11use std::fs::{self, File, OpenOptions};
12use std::io::{Error, ErrorKind, Result, Write as _};
13use std::path::{Path, PathBuf};
14use std::sync::OnceLock;
15
16use rand::TryRng as _;
17use rand::rngs::SysRng;
18
19/// The key size Geyser generates and expects: `AesKeyProducer.KEY_SIZE` is 128 bits.
20pub const KEY_LENGTH: usize = 16;
21
22/// The directory under a run directory that holds Bedrock's own files.
23const BEDROCK_DIR: &str = "bedrock";
24
25/// The file the shared key is stored under.
26const KEY_FILE: &str = "key.pem";
27
28/// Where the Floodgate shared key lives, for a given run directory.
29///
30/// The single definition of this path. The server (at startup), the login
31/// path (through [`shared`]) and the Geyser supervisor all need the same
32/// file, so this function -- not a repeated join -- is what each of them
33/// calls: a second definition that drifted from this one would leave Foton
34/// and its own Geyser holding different keys, and every Bedrock handshake
35/// would fail to decrypt.
36#[must_use]
37pub fn key_path(run_directory: &Path) -> PathBuf {
38    run_directory.join(BEDROCK_DIR).join(KEY_FILE)
39}

41/// Loads the shared key, generating it on first run.
42///
43/// The generated key comes from the operating system's randomness, never from a
44/// seeded generator.
45pub fn load_or_create(path: &Path) -> Result<[u8; KEY_LENGTH]> {
46    if path.is_file() {
47        return read_key(path);
48    }
49
50    if let Some(parent) = path.parent() {
51        fs::create_dir_all(parent)?;
52    }
53
54    let mut key = [0u8; KEY_LENGTH];
55    SysRng
56        .try_fill_bytes(&mut key)
57        .map_err(|error| Error::other(format!("no system randomness: {error}")))?;
58
59    match create_restricted(path) {
60        Ok(mut file) => {
61            file.write_all(&key)?;
62            Ok(key)
63        }
64        // Another process created the key first, in the window between the
65        // `is_file` check above and this call. `create_new` (inside
66        // `create_restricted`) is what makes that detectable at all instead
67        // of silently overwritten: the old `fs::write` here let both
68        // processes "win", each returning the random bytes *it* generated --
69        // bytes that might not be what ends up on disk once the other
70        // process's write lands. Geyser reads the key file directly, not
71        // this process's copy, so a process holding a key that doesn't match
72        // the file would fail every Floodgate handshake with no clear
73        // reason why. Reading back what the other process wrote, rather
74        // than erroring out, is what keeps every process -- and Geyser --
75        // agreeing on one key.
76        //
77        // This still has a narrow gap: a reader landing between the
78        // winner's `create_restricted` and its `write_all` would see a
79        // 0-byte file and get `read_key`'s "wrong size" error rather than
80        // the key. Closing that fully would mean writing to a sibling
81        // temp file and renaming it into place, which trades this gap for
82        // last-rename-wins silently discarding the loser's own generated
83        // bytes instead -- a different failure mode, not obviously a
84        // better one. Left as is because two Foton processes racing to
85        // initialize the same run directory's key is already a
86        // misconfiguration outside what this function can make safe, and
87        // the window is one small, uninterrupted `write` syscall wide.
88        Err(error) if error.kind() == ErrorKind::AlreadyExists => read_key(path),
89        Err(error) => Err(error),
90    }
91}

93/// Reads and validates an existing key file.
94fn read_key(path: &Path) -> Result<[u8; KEY_LENGTH]> {
95    let bytes = fs::read(path)?;
96    bytes.try_into().map_err(|_| {
97        Error::new(
98            ErrorKind::InvalidData,
99            format!(
100                "{} is not a {KEY_LENGTH}-byte Floodgate key",
101                path.display()
102            ),
103        )
104    })
105}

107/// Creates `path` exclusively, with the mode restricted to owner
108/// read/write from the moment it exists, where the platform has a way to
109/// say that at creation time.
110///
111/// The mode is set *at creation* (`OpenOptionsExt::mode`, not a later
112/// `set_permissions`) deliberately: applying it as a second step after
113/// `fs::write` already created the file leaves a window where the file
114/// exists at whatever the process umask allows -- commonly `0o644`, world
115/// readable -- and what is readable through that window is the key that
116/// authenticates every Bedrock player's identity to this server. There is
117/// no second factor behind it, so that window is worth closing even though
118/// it is normally short.
119///
120/// `create_new` rather than plain `create`: it fails with
121/// [`ErrorKind::AlreadyExists`] instead of silently truncating a file
122/// another process just created, which is what lets [`load_or_create`]
123/// detect and recover from a concurrent first run instead of the two
124/// processes fighting over which one's key ends up on disk.
125#[cfg(unix)]
126fn create_restricted(path: &Path) -> Result<File> {
127    use std::os::unix::fs::OpenOptionsExt as _;
128    OpenOptions::new()
129        .write(true)
130        .create_new(true)
131        .mode(0o600)
132        .open(path)
133}
134
135/// Windows has no mode to set at creation; the directory's ACL, which is
136/// the operator's to set, is what governs instead. `create_new` is kept
137/// even here so both platforms detect the same concurrent-creation race
138/// (see [`load_or_create`]'s `AlreadyExists` handling), even though this
139/// branch cannot also close the Unix permission window.
140#[cfg(not(unix))]
141fn create_restricted(path: &Path) -> Result<File> {
142    OpenOptions::new().write(true).create_new(true).open(path)
143}

145/// The process-wide shared key, set once at server startup.
146static SHARED_KEY: OnceLock<[u8; KEY_LENGTH]> = OnceLock::new();
147
148/// Sets the process-wide shared key. The server calls this once at startup,
149/// right after [`load_or_create`].
150///
151/// Returns `true` if this call is the one that set the key, `false` if it was
152/// already set -- first call wins, later calls are no-ops rather than
153/// replacing a key already in use.
154pub fn init_shared(key: [u8; KEY_LENGTH]) -> bool {
155    SHARED_KEY.set(key).is_ok()
156}
157
158/// The process-wide shared key, if [`init_shared`] has been called.
159///
160/// # Security
161///
162/// `None` is the safe default. It means Bedrock support was never
163/// initialized, and it must mean **no Floodgate login is ever accepted**: the
164/// login path is expected to treat an uninitialized key exactly like a
165/// rejected handshake, never as a transient condition worth retrying.
166#[must_use]
167pub fn shared() -> Option<&'static [u8; KEY_LENGTH]> {
168    SHARED_KEY.get()
169}

171#[cfg(test)]
172mod tests {
173    use super::{KEY_LENGTH, init_shared, key_path, load_or_create, shared};
174    use std::fs::write;
175    use std::path::Path;
176
177    #[test]
178    fn creates_a_key_once_and_reuses_it() {
179        let directory = tempfile::tempdir().expect("temp dir");
180        let path = directory.path().join("key.pem");
181
182        let first = load_or_create(&path).expect("creates");
183        assert!(path.is_file());
184        let second = load_or_create(&path).expect("loads");
185
186        assert_eq!(first, second, "the key must survive a restart");
187        assert_ne!(first, [0u8; KEY_LENGTH], "a key of zeroes is not a key");
188    }
189
190    #[test]
191    fn refuses_a_key_file_of_the_wrong_size() {
192        let directory = tempfile::tempdir().expect("temp dir");
193        let path = directory.path().join("key.pem");
194        write(&path, b"too short").expect("writes");
195
196        assert!(load_or_create(&path).is_err());
197    }
198
199    /// Unix-only: permission bits are a POSIX mode, which Windows does not
200    /// have -- there, the governing control is the directory's ACL, which
201    /// this crate never touches (see `create_restricted`'s non-Unix
202    /// doc comment), so there is nothing analogous for a Windows-only
203    /// variant of this test to assert.
204    #[cfg(unix)]
205    #[test]
206    fn the_created_key_file_is_readable_only_by_its_owner() {
207        use std::fs::metadata;
208        use std::os::unix::fs::PermissionsExt as _;
209
210        let directory = tempfile::tempdir().expect("temp dir");
211        let path = directory.path().join("key.pem");
212
213        load_or_create(&path).expect("creates");
214
215        let mode = metadata(&path)
216            .expect("the freshly created file has metadata")
217            .permissions()
218            .mode()
219            & 0o777;
220        assert_eq!(
221            mode, 0o600,
222            "the key file must be created non-world-readable, not merely fixed up afterwards"
223        );
224    }
225
226    #[test]
227    fn key_path_composes_the_documented_path() {
228        let run_directory = Path::new("/srv/foton");
229
230        assert_eq!(
231            key_path(run_directory),
232            run_directory.join("bedrock").join("key.pem")
233        );
234    }
235
236    /// A single test covering set, read, and rejected second set: a
237    /// [`OnceLock`](std::sync::OnceLock) is process-wide and persists across
238    /// every test in this binary, so a second test calling `init_shared`
239    /// would either race this one or silently observe a key some other test
240    /// already installed.
241    #[test]
242    fn init_shared_is_first_call_wins() {
243        let key = [7u8; KEY_LENGTH];
244        let other = [9u8; KEY_LENGTH];
245
246        assert!(init_shared(key), "the first call sets the key");
247        assert_eq!(shared(), Some(&key));
248
249        assert!(
250            !init_shared(other),
251            "a second call must not replace the key"
252        );
253        assert_eq!(shared(), Some(&key), "the original key must survive");
254    }
