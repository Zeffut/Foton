1//! Argument plumbing shared by the entity, player and world natives.
2//!
3//! Every native here receives Java strings and answers "nothing" when what
4//! they name no longer exists: Bukkit's contract for a stale handle is an
5//! object that quietly does nothing, and plugins rely on it.
6
7use std::ffi::c_void;

24/// A registered native, in the shape `RegisterNatives` wants.
25pub(super) fn method(name: &str, signature: &str, pointer: *mut c_void) -> jni::NativeMethod {
26    jni::NativeMethod {
27        name: name.into(),
28        sig: signature.into(),
29        fn_ptr: pointer,
30    }
31}
32
33/// A Java string, or `None` for null or an unreadable one.
34pub(super) fn text(env: &mut JNIEnv<'_>, value: &JString<'_>) -> Option<String> {
35    if value.is_null() {
36        return None;
37    }
38    env.get_string(value).ok().map(Into::into)
39}
40
41/// A Java string parsed as a registry key; a bare path is in `minecraft:`.
42pub(super) fn key(env: &mut JNIEnv<'_>, value: &JString<'_>) -> Option<Identifier> {
43    text(env, value)?.parse().ok()
44}
45
46/// The entity a Java handle names, with the world it is in, if it still exists.
47pub(super) fn entity(
48    env: &mut JNIEnv<'_>,
49    uuid: &JString<'_>,
50) -> Option<(Arc<World>, SharedEntity)> {
51    let id = Uuid::parse_str(&text(env, uuid)?).ok()?;
52    entity_by_uuid(&id)
53}
54
55/// The online player a Java handle names.
56pub(super) fn player(env: &mut JNIEnv<'_>, uuid: &JString<'_>) -> Option<Arc<Player>> {
57    let id = Uuid::parse_str(&text(env, uuid)?).ok()?;
58    server()?.online_players().get_by_uuid(&id)
59}
60
61/// A world by the key a plugin holds it under.
62pub(super) fn world(env: &mut JNIEnv<'_>, name: &JString<'_>) -> Option<Arc<World>> {
63    let key = key(env, name)?;
64    server()?.worlds.get_owned(&key)
65}

67/// A Java `double[]`, or null when there is nothing to answer.
68pub(super) fn doubles(env: &mut JNIEnv<'_>, values: Option<&[f64]>) -> jdoubleArray {
69    let Some(values) = values else {
70        return null_mut();
71    };
72    let Ok(length) = i32::try_from(values.len()) else {
73        return null_mut();
74    };
75    let Ok(array) = env.new_double_array(length) else {
76        return null_mut();
77    };
78    if env.set_double_array_region(&array, 0, values).is_err() {
79        return null_mut();
80    }
81    let array: JDoubleArray<'_> = array;
82    array.into_raw()
83}
