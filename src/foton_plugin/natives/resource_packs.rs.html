1//! Resource packs a plugin pushes to a player, takes back, and asks about,
2//! through Paper's `Player.sendResourcePacks` and friends.
3
4use std::ffi::c_void;

6use foton_protocol::packets::common::{CResourcePackPop, CResourcePackPush};
7use jni::JNIEnv;
8use jni::objects::{JClass, JString};
9use jni::sys::{jboolean, jint};
10use uuid::Uuid;
11
12use super::support::{component, method, player, text};

14/// The longest URL a push packet carries.
15///
16/// Vanilla parity: `ByteBufCodecs.STRING_UTF8`, which is `stringUtf8(32767)`.
17const MAX_PACK_URL_LENGTH: usize = 32767;
18
19/// Asks the player's client to load a resource pack.
20///
21/// Paper parity: `CraftPlayer.sendResourcePacks`. False when the packet could
22/// not be built: a packet that will not encode would drop the player instead.
23extern "system" fn send_resource_pack(
24    mut env: JNIEnv<'_>,
25    _class: JClass<'_>,
26    uuid: JString<'_>,
27    pack: JString<'_>,
28    url: JString<'_>,
29    hash: JString<'_>,
30    required: jboolean,
31    prompt: JString<'_>,
32) -> jboolean {
33    let (Some(id), Some(url), Some(hash)) = (
34        text(&mut env, &pack).and_then(|pack| Uuid::parse_str(&pack).ok()),
35        text(&mut env, &url),
36        text(&mut env, &hash),
37    ) else {
38        return 0;
39    };
40    if url.len() > MAX_PACK_URL_LENGTH || hash.len() > CResourcePackPush::MAX_HASH_LENGTH {
41        return 0;
42    }
43    let prompt = if prompt.is_null() {
44        None
45    } else if let Some(prompt) = component(&mut env, &prompt) {
46        Some(prompt)
47    } else {
48        return 0;
49    };
50    let Some(player) = player(&mut env, &uuid) else {
51        return 0;
52    };
53    player.send_packet(CResourcePackPush {
54        id,
55        url,
56        hash,
57        required: required != 0,
58        prompt,
59    });
60    1
61}

63/// Takes one resource pack, or every one the server sent, back from the client.
64extern "system" fn remove_resource_packs(
65    mut env: JNIEnv<'_>,
66    _class: JClass<'_>,
67    uuid: JString<'_>,
68    pack: JString<'_>,
69) {
70    let id = if pack.is_null() {
71        None
72    } else if let Some(id) = text(&mut env, &pack).and_then(|pack| Uuid::parse_str(&pack).ok()) {
73        Some(id)
74    } else {
75        return;
76    };
77    if let Some(player) = player(&mut env, &uuid) {
78        player.send_packet(CResourcePackPop { id });
79    }
80}

82/// The ordinal of the client's last reported pack status, or -1 before it said any.
83extern "system" fn resource_pack_status(
84    mut env: JNIEnv<'_>,
85    _class: JClass<'_>,
86    uuid: JString<'_>,
87) -> jint {
88    player(&mut env, &uuid)
89        .and_then(|player| player.resource_pack_status())
90        .map_or(-1, |status| status as jint)
91}

93pub(super) fn bindings() -> Vec<jni::NativeMethod> {
94    vec![
95        method(
96            "sendResourcePack",
97            "(Ljava/lang/String;Ljava/lang/String;Ljava/lang/String;Ljava/lang/String;ZLjava/lang/String;)Z",
98            send_resource_pack as *mut c_void,
99        ),
100        method(
101            "removeResourcePacks",
102            "(Ljava/lang/String;Ljava/lang/String;)V",
103            remove_resource_packs as *mut c_void,
104        ),
105        method(
106            "resourcePackStatus",
107            "(Ljava/lang/String;)I",
108            resource_pack_status as *mut c_void,
109        ),
110    ]
