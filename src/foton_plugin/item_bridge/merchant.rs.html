32fn integer(env: &mut JNIEnv<'_>, value: &JObject<'_>, name: &str) -> Result<i32, ItemBridgeError> {

40pub(crate) fn prepare(
41    env: &mut JNIEnv<'_>,
42    values: &JObjectArray<'_>,
43) -> Result<OfferBatch, ItemBridgeError> {
44    let size = env.get_array_length(values)?;
45    if size > 4096 {
46        return Err(ItemBridgeError::TransportLimit);
47    }
48    let mut offers = MerchantOffers::new();
49    let mut candidates = vec![store()?.stage(|| Ok(ItemStack::empty()))?];
50    for index in 0..size {
51        let value = env.get_object_array_element(values, index)?;
52        if value.is_null() {
53            return Err(ItemBridgeError::InvalidEdit("missing merchant offer"));
54        }
55        let inputs = JObjectArray::from(
56            env.get_field(&value, "ingredients", "[Lfoton/item/ItemMutation;")?
57                .l()?,
58        );
59        let count = env.get_array_length(&inputs)?;
60        if !(1..=2).contains(&count) {
61            return Err(ItemBridgeError::NativeState(
62                "merchant recipe requires one or two ingredients".into(),
63            ));
64        }
65        let mut costs = Vec::new();
66        for input in 0..count {
67            let input = env.get_object_array_element(&inputs, input)?;
68            let mut candidate = mutation::materialize(env, &input)?;
69            let cost = ItemCost::try_from_ingredient(take(&mut candidate)).ok_or(
70                ItemBridgeError::InvalidEdit("empty or invalid merchant ingredient predicate"),
71            )?;
72            costs.push(cost);
73            candidates.push(candidate);
74        }
75        let result = env
76            .get_field(&value, "result", "Lfoton/item/ItemMutation;")?
77            .l()?;
78        let mut result = mutation::materialize(env, &result)?;
79        if result.stack.is_empty() {
80            return Err(ItemBridgeError::InvalidEdit("empty merchant result"));
81        }
82        let mut costs = costs.into_iter();
83        let first = costs
84            .next()
85            .ok_or(ItemBridgeError::InvalidEdit("missing merchant ingredient"))?;
86        let mut offer = MerchantOffer::with_uses(
87            first,
88            costs.next(),
89            take(&mut result),
90            integer(env, &value, "uses")?,
91            integer(env, &value, "maxUses")?,
92            integer(env, &value, "experience")?,
93            env.get_field(&value, "priceMultiplier", "F")?.f()?,
94            integer(env, &value, "demand")?,
95        );
96        offer.set_special_price_diff(integer(env, &value, "specialPrice")?);
97        offer.set_reward_exp(env.get_field(&value, "rewardExperience", "Z")?.z()?);
98        offers.push(offer);
99        candidates.push(result);
100    }
101    Ok(OfferBatch {
102        offers,
103        _candidates: candidates,
104    })
105}

107pub(crate) extern "system" fn set(
108    mut env: JNIEnv<'_>,
109    _class: JClass<'_>,
110    uuid: JString<'_>,
111    values: JObjectArray<'_>,
112) {
113    let result = (|| -> Result<(), ItemBridgeError> {
114        let mut batch = prepare(&mut env, &values)?;
115        let id = identity(&mut env, &uuid)?;
116        if let Some((_, entity)) = bridge_native_bridge::entity_by_uuid(&id)
117            && let Some(villager) = entity.downcast_ref::<VillagerEntity>()
118        {
119            villager
120                .merchant()
121                .set_offers(bridge_mem::take(&mut batch.offers));
122        }
123        Ok(())
124    })();
125    if let Err(error) = result {
126        error.throw_java(&mut env);
127    }
128}

130pub(crate) struct CapturedOffer {
131    result: Capture<'static>,
132    first: Capture<'static>,
133    second: Option<Capture<'static>>,
134    uses: i32,
135    max_uses: i32,
136    demand: i32,
137    experience: i32,
138    special_price: i32,
139    multiplier: f32,
140    reward: bool,
141}

143pub(crate) fn capture(offers: &MerchantOffers) -> Result<Vec<CapturedOffer>, ItemBridgeError> {
144    let store = store()?;
145    if offers.len() > 4096 {
146        return Err(ItemBridgeError::TransportLimit.native_state());
147    }
148    offers
149        .iter()
150        .map(|offer| {
151            if offer.result().is_empty() {
152                return Err(ItemBridgeError::NativeState("empty merchant result".into()));
153            }
154            Ok(CapturedOffer {
155                result: store
156                    .capture(offer.result())
157                    .map_err(ItemBridgeError::native_state)?,
158                first: store
159                    .capture(offer.item_cost_a().cost_stack())
160                    .map_err(ItemBridgeError::native_state)?,
161                second: offer
162                    .item_cost_b()
163                    .map(|cost| {
164                        store
165                            .capture(cost.cost_stack())
166                            .map_err(ItemBridgeError::native_state)
167                    })
168                    .transpose()?,
169                uses: offer.uses(),
170                max_uses: offer.max_uses(),
171                demand: offer.demand(),
172                experience: offer.xp(),
173                special_price: offer.special_price_diff(),
174                multiplier: offer.price_multiplier(),
175                reward: offer.should_reward_exp(),
176            })
177        })
178        .collect()
179}

181fn read(env: &mut JNIEnv<'_>, uuid: &JString<'_>) -> Result<jobjectArray, ItemBridgeError> {
182    let id = identity(env, uuid)?;
183    let Some((_, entity)) = bridge_native_bridge::entity_by_uuid(&id) else {
184        return Ok(bridge_ptr::null_mut());
185    };
186    let Some(villager) = entity.downcast_ref::<VillagerEntity>() else {
187        return Ok(bridge_ptr::null_mut());
188    };
189    let offers = villager.with_offers(capture)?;
190    transfers(env, offers).map(JObjectArray::into_raw)
191}

193pub(crate) fn transfers<'local>(
194    env: &mut JNIEnv<'local>,
195    offers: Vec<CapturedOffer>,
196) -> Result<JObjectArray<'local>, ItemBridgeError> {
197    let result = env.new_object_array(
198        i32::try_from(offers.len()).map_err(|_| ItemBridgeError::TransportLimit)?,
199        "foton/item/MerchantOfferTransfer",
200        JObject::null(),
201    )?;
202    for (index, offer) in offers.into_iter().enumerate() {
203        let output = transfer::from_capture(env, offer.result)?;
204        let first = transfer::from_capture(env, offer.first)?;
205        let second = match offer.second {
206            Some(value) => transfer::from_capture(env, value)?,
207            None => JObject::null(),
208        };
209        let value = env.new_object(
210            "foton/item/MerchantOfferTransfer",
211            "(Lfoton/item/ItemTransfer;Lfoton/item/ItemTransfer;Lfoton/item/ItemTransfer;IIIIIFZ)V",
212            &[
213                JValue::Object(&output),
214                JValue::Object(&first),
215                JValue::Object(&second),
216                JValue::Int(offer.uses),
217                JValue::Int(offer.max_uses),
218                JValue::Int(offer.demand),
219                JValue::Int(offer.experience),
220                JValue::Int(offer.special_price),
221                JValue::Float(offer.multiplier),
222                JValue::Bool(u8::from(offer.reward)),
223            ],
224        )?;
225        env.set_object_array_element(
226            &result,
227            i32::try_from(index).map_err(|_| ItemBridgeError::TransportLimit)?,
228            value,
229        )?;
230    }
231    Ok(result)
232}
