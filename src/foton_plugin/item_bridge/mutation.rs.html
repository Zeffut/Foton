34fn text(env: &mut JNIEnv<'_>, object: JObject<'_>) -> Result<String, ItemBridgeError> {
35    let value = JString::from(object);
36    let value = env.get_string(&value)?;
37    let value = value
38        .to_str()
39        .map_err(|_| ItemBridgeError::InvalidEdit("invalid string encoding"))?;
40    if value.len() > 8_388_608 {
41        return Err(ItemBridgeError::TransportLimit);
42    }
43    Ok(value.to_owned())
44}

55pub(crate) fn materialize(
56    env: &mut JNIEnv<'_>,
57    mutation: &JObject<'_>,
58) -> Result<Candidate, ItemBridgeError> {
59    super::require_registry()?;
60    if mutation.is_null() {
61        return Err(ItemBridgeError::InvalidEdit("missing owning mutation"));
62    }
63    let projection = string_field(env, mutation, "projection")?;
64    let lease = env
65        .get_field(mutation, "lease", "Lfoton/item/NativeItemLease;")?
66        .l()?;
67    let keys = JObjectArray::from(
68        env.get_field(mutation, "keys", "[Ljava/lang/String;")?
69            .l()?,
70    );
71    let operations = JObjectArray::from(
72        env.get_field(mutation, "operations", "[Ljava/lang/String;")?
73            .l()?,
74    );
75    let size = env.get_array_length(&keys)?;
76    if size > 4096 || size != env.get_array_length(&operations)? {
77        return Err(ItemBridgeError::InvalidEdit("invalid journal length"));
78    }
79    let (item, count) = header(&projection)?;
80    let mut candidate = if lease.is_null() {
81        store()?.stage(|| {
82            bridge_native_bridge::parse_slot(&projection)
83                .ok_or(ItemBridgeError::InvalidEdit("invalid legacy item"))
84        })?
85    } else {
86        store()?.materialize(transfer::lease_key(env, &lease)?)?
87    };
88    let mut groups = rustc_hash::FxHashSet::default();
89    for index in 0..size {
90        let key_object = env.get_object_array_element(&keys, index)?;
91        let key = text(env, key_object)?;
92        let operation_object = env.get_object_array_element(&operations, index)?;
93        let operation = text(env, operation_object)?;
94        if !groups.insert(key.clone()) {
95            return Err(ItemBridgeError::InvalidEdit("duplicate metadata group"));
96        }
97        let edit = decode_edit(&candidate.stack, &projection, &key, &operation)?;
98        // Subfield groups can share a component (potion base/effects, tooltip
99        // visibility/flags). Compose on the detached candidate only. A later
100        // failure discards all edits; no destination or source lease changes.
101        candidate = Edits::new(vec![edit])?.apply(candidate)?;
102    }
103    if let Some(item) = item {
104        if !lease.is_null() && candidate.stack.item() != item {
105            return Err(ItemBridgeError::InvalidEdit(
106                "item type requires an explicit rebase",
107            ));
108        }
109        candidate.stack.count = count;
110    } else {
111        candidate.stack = ItemStack::empty();
112    }
113    Ok(candidate)
114}

116fn header(projection: &str) -> Result<(Option<bridge_items::ItemRef>, i32), ItemBridgeError> {
117    if projection.is_empty() {
118        return Ok((None, 0));
119    }
120    let header = projection
121        .split('\u{1d}')
122        .next()
123        .ok_or(ItemBridgeError::InvalidEdit("missing item header"))?;
124    let (key, count) = header
125        .rsplit_once(' ')
126        .ok_or(ItemBridgeError::InvalidEdit("invalid item header"))?;
127    let key: Identifier = key
128        .parse()
129        .map_err(|_| ItemBridgeError::InvalidEdit("invalid item key"))?;
130    let item = REGISTRY
131        .items
132        .by_key(&key)
133        .ok_or(ItemBridgeError::InvalidEdit("unknown item"))?;
134    let count: i32 = count
135        .parse()
136        .map_err(|_| ItemBridgeError::InvalidEdit("invalid count"))?;
137    if count <= 0 {
138        return Err(ItemBridgeError::InvalidEdit("nonpositive count"));
139    }
140    Ok((Some(item), count))
141}

143fn decode_potion_edit(
144    base: &ItemStack,
145    projection: &str,
146    group: &str,
147    operation: &str,
148) -> Result<Edit, ItemBridgeError> {
149    use foton_registry::data_components::{
150        components::PotionContents, vanilla_components::POTION_CONTENTS,
151    };
152    if operation != "SET" {
153        return Err(ItemBridgeError::InvalidEdit(
154            "invalid potion subfield operation",
155        ));
156    }
157    let original = base.get(POTION_CONTENTS);
158    if group == "potion_base" {
159        let fields: Vec<_> = projection
160            .split('\u{1d}')
161            .skip(1)
162            .filter(|field| field.starts_with("basepotionhex="))
163            .collect();
164        if fields.len() > 1 {
165            return Err(ItemBridgeError::InvalidEdit("duplicate base potion"));
166        }
167        let selected = bridge_iter::once(projection.split('\u{1d}').next().unwrap_or_default())
168            .chain(fields)
169            .collect::<Vec<_>>()
170            .join("\u{1d}");
171        let parsed = bridge_native_bridge::parse_slot(&selected)
172            .ok_or(ItemBridgeError::InvalidEdit("invalid base potion"))?;
173        return Ok(Edit {
174            key: POTION_CONTENTS.key().clone(),
175            operation: Operation::Set(ComponentData::new(PotionContents::new(
176                parsed.get(POTION_CONTENTS).and_then(PotionContents::potion),
177                original.and_then(PotionContents::custom_color),
178                original.map_or_else(Vec::new, |value| value.custom_effects().to_vec()),
179                original
180                    .and_then(PotionContents::custom_name)
181                    .map(str::to_owned),
182            ))),
183        });
184    }
185    if original.is_some_and(|contents| {
186        contents
187            .custom_effects()
188            .iter()
189            .any(|effect| effect.hidden_effect().is_some())
190    }) {
191        return Err(ItemBridgeError::Unsupported(
192            "unprojected native potion effects",
193        ));
194    }
195    let fields: Vec<_> = projection
196        .split('\u{1d}')
197        .skip(1)
198        .filter_map(|field| {
199            field
200                .strip_prefix("potioneffects=")
201                .or_else(|| (field.contains(',') && !field.contains('=')).then_some(field))
202        })
203        .collect();
204    if fields.len() > 1 {
205        return Err(ItemBridgeError::InvalidEdit("duplicate potion effects"));
206    }
207    let effects = match fields.first() {
208        Some(field) => super::legacy::parse_effects(field)
209            .ok_or(ItemBridgeError::InvalidEdit("invalid potion effects"))?,
210        None => Vec::new(),
211    };
212    Ok(Edit {
213        key: POTION_CONTENTS.key().clone(),
214        operation: Operation::Set(ComponentData::new(PotionContents::new(
215            original.and_then(PotionContents::potion),
216            original.and_then(PotionContents::custom_color),
217            effects,
218            original
219                .and_then(PotionContents::custom_name)
220                .map(str::to_owned),
221        ))),
222    })
223}

225/// Paper replaces the whole list; an empty one restores the item's default.
226/// Entries the stack already carried keep their tooltip display, which Bukkit
227/// cannot express.
228fn decode_attribute_edit(
229    base: &ItemStack,
230    projection: &str,
231    operation: &str,
232) -> Result<Edit, ItemBridgeError> {
233    let operation = match operation {
234        "REMOVE" => Operation::Remove,
235        "RESET" => Operation::Reset,
236        "SET" => {
237            let previous = base.get(bridge_vanilla_components::ATTRIBUTE_MODIFIERS);
238            let modifiers = projection
239                .split('\u{1d}')
240                .skip(1)
241                .filter_map(|field| field.strip_prefix("attrmod="))
242                .map(|value| {
243                    let mut entry = parse_attribute_modifier(value)
244                        .ok_or(ItemBridgeError::InvalidEdit("invalid attribute modifier"))?;
245                    if let Some(old) = previous.and_then(|list| {
246                        list.modifiers.iter().find(|old| {
247                            old.attribute.key == entry.attribute.key && old.id == entry.id
248                        })
249                    }) {
250                        entry.display = old.display.clone();
251                    }
252                    Ok(entry)
253                })
254                .collect::<Result<Vec<_>, ItemBridgeError>>()?;
255            Operation::Set(ComponentData::new(ItemAttributeModifiers { modifiers }))
256        }
257        _ => return Err(ItemBridgeError::InvalidEdit("unknown edit operation")),
258    };
259    Ok(Edit {
260        key: bridge_vanilla_components::ATTRIBUTE_MODIFIERS.key().clone(),
261        operation,
262    })
263}

305fn decode_edit(
306    base: &ItemStack,
307    projection: &str,
308    group: &str,
309    operation: &str,
310) -> Result<Edit, ItemBridgeError> {
311    if group == "potion_effects" || group == "potion_base" {
312        return decode_potion_edit(base, projection, group, operation);
313    }
314    if group == "attribute_modifiers" {
315        return decode_attribute_edit(base, projection, operation);
316    }
317    if group.starts_with("unsupported:") {
318        let capability = match group {
319            "unsupported:item_flags" => "item flags",
320            "unsupported:bundle_contents" => "bundle contents",
321            "unsupported:charged_projectiles" => "charged projectiles",
322            "unsupported:dyed_color" => "dyed color",
323            "unsupported:trim" => "armor trim",
324            "unsupported:banner" => "banner patterns/color",
325            "unsupported:fireworks" => "fireworks",
326            "unsupported:firework_explosion" => "firework explosion",
327            "unsupported:map" => "map metadata",
328            "unsupported:stew_effects" => "suspicious stew effects",
329            "unsupported:base_potion" => "base potion",
330            "unsupported:potion_color" => "potion color",
331            _ => "metadata mutation group",
332        };
333        return Err(ItemBridgeError::Unsupported(capability));
334    }
335    let prefixes = group_prefixes(group)?;
336    let component = if group == "book" {
337        match header(projection)?.0 {
338            Some(item) if item == &*bridge_vanilla_items::WRITTEN_BOOK => "written_book_content",
339            Some(item) if item == &*bridge_vanilla_items::WRITABLE_BOOK => "writable_book_content",
340            _ => return Err(ItemBridgeError::InvalidEdit("book edit on non-book item")),
341        }
342    } else if group == "tooltip_visibility" {
343        "tooltip_display"
344    } else {
345        group
346    };
347    let key: Identifier = format!("minecraft:{component}")
348        .parse()
349        .map_err(|_| ItemBridgeError::InvalidEdit("invalid component key"))?;
350    let operation = match operation {
351        "REMOVE" => Operation::Remove,
352        "RESET" => Operation::Reset,
353        "SET" => {
354            let mut fields = projection.split('\u{1d}');
355            let mut selected = fields
356                .next()
357                .ok_or(ItemBridgeError::InvalidEdit("missing edit header"))?
358                .to_owned();
359            for field in fields {
360                if prefixes.iter().any(|prefix| field.starts_with(prefix)) {
361                    selected.push('\u{1d}');
362                    selected.push_str(field);
363                }
364            }
365            let parsed = bridge_native_bridge::parse_slot(&selected)
366                .ok_or(ItemBridgeError::InvalidEdit("invalid component payload"))?;
367            if group == "tooltip_visibility" {
368                // The existing Java setter edits the global flag, not hidden component types.
369                let mut display = base
370                    .get(bridge_vanilla_components::TOOLTIP_DISPLAY)
371                    .cloned()
372                    .unwrap_or_default();
373                display.hide_tooltip = parsed
374                    .get(bridge_vanilla_components::TOOLTIP_DISPLAY)
375                    .is_some_and(|display| display.hide_tooltip);
376                Operation::Set(ComponentData::new(display))
377            } else if let Some(value) = parsed.get_effective_value_raw(&key) {
378                super::preflight::check_component(value)?;
379                Operation::Set(value.clone())
380            } else {
381                match group {
382                    "damage" => Operation::Set(ComponentData::new(0_i32)),
383                    "tooltip_display" => {
384                        Operation::Set(ComponentData::new(TooltipDisplay::new(false)))
385                    }
386                    "custom_model_data" => Operation::Set(ComponentData::new(
387                        CustomModelData::new(vec![], vec![], vec![], vec![]),
388                    )),
389                    "enchantments" | "stored_enchantments" => {
390                        Operation::Set(ComponentData::new(ItemEnchantments::empty()))
391                    }
392                    "banner_patterns" => {
393                        Operation::Set(ComponentData::new(BannerPatternLayers::empty()))
394                    }
395                    "custom_data" => Operation::Remove,
396                    _ => return Err(ItemBridgeError::InvalidEdit("missing SET value")),
397                }
398            }
399        }
400        _ => return Err(ItemBridgeError::InvalidEdit("unknown edit operation")),
401    };
402    Ok(Edit { key, operation })
403}
