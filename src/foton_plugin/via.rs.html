1//! JNI adapter for the Paper channel hook consumed by `ViaVersion`.
2
3use std::sync::{
4    Arc,
5    atomic::{AtomicBool, Ordering},
6};
7
8use foton_protocol::{
9    packet_traits::{PacketDirection, PacketTranslation, PacketTranslator, TranslationBatch},
10    utils::{MAX_PACKET_DATA_SIZE, PacketError},
11};
12use foton_utils::locks::SyncMutex;
13use jni::{
14    JNIEnv, JavaVM,
15    errors::Error as JniError,
16    objects::{GlobalRef, JByteArray, JObjectArray, JValue, JValueOwned},
17};
18use uuid::Uuid;

20const BRIDGE_CLASS: &str = "foton/network/FotonViaBridge";
21const MAX_EXCHANGE_PACKETS: usize = 256;
22const MAX_EXCHANGE_BYTES: usize = MAX_PACKET_DATA_SIZE * 4;
23
24pub(crate) struct ViaTranslator {
25    vm: Arc<JavaVM>,
26    channel: SyncMutex<Option<GlobalRef>>,
27    closed: AtomicBool,
28}
29
30impl ViaTranslator {
31    pub(crate) fn open(vm: Arc<JavaVM>) -> Result<Option<PacketTranslation>, PacketError> {
32        let mut env = vm
33            .attach_current_thread()
34            .map_err(|error| Self::jni_error("attach JVM", error))?;
35        let channel = match env
36            .call_static_method(
37                BRIDGE_CLASS,
38                "open",
39                "()Lfoton/network/FotonViaChannel;",
40                &[],
41            )
42            .and_then(JValueOwned::l)
43        {
44            Ok(channel) => channel,
45            Err(error) => {
46                Self::clear_exception(&mut env);
47                return Err(Self::jni_error("open Via channel", error));
48            }
49        };
50        if channel.is_null() {
51            return Ok(None);
52        }
53        let channel = env
54            .new_global_ref(channel)
55            .map_err(|error| Self::jni_error("retain Via channel", error))?;
56        drop(env);
57        Ok(Some(PacketTranslation::new(Arc::new(Self {
58            vm,
59            channel: SyncMutex::new(Some(channel)),
60            closed: AtomicBool::new(false),
61        }))))
62    }
63
64    fn invoke(&self, method: &str, packet: Option<&[u8]>) -> Result<TranslationBatch, PacketError> {
65        if self.closed.load(Ordering::Acquire) {
66            return Err(PacketError::ConnectionClosed);
67        }
68        let mut env = self
69            .vm
70            .attach_current_thread()
71            .map_err(|error| Self::jni_error("attach JVM", error))?;
72        let channel = self.channel.lock();
73        let channel = channel.as_ref().ok_or(PacketError::ConnectionClosed)?;
74        let output = if let Some(packet) = packet {
75            let packet = env
76                .byte_array_from_slice(packet)
77                .map_err(|error| Self::jni_error("copy packet into JVM", error))?;
78            env.call_method(
79                channel.as_obj(),
80                method,
81                "([B)[[B",
82                &[JValue::Object(packet.as_ref())],
83            )
84        } else {
85            env.call_method(channel.as_obj(), method, "()[[B", &[])
86        };
87        let output = match output.and_then(JValueOwned::l) {
88            Ok(output) => output,
89            Err(error) => {
90                Self::clear_exception(&mut env);
91                return Err(Self::jni_error("run Via pipeline", error));
92            }
93        };
94        let output = JObjectArray::from(output);
95        let count = usize::try_from(
96            env.get_array_length(&output)
97                .map_err(|error| Self::jni_error("read Via output length", error))?,
98        )
99        .map_err(|_| PacketError::OutOfBounds)?;
100        if count > MAX_EXCHANGE_PACKETS {
101            return Err(PacketError::TooLong(count));
102        }
103
104        let mut batch = TranslationBatch::default();
105        let mut total_bytes = 0usize;
106        for index in 0..count {
107            let marked = env
108                .get_object_array_element(&output, index as i32)
109                .map_err(|error| Self::jni_error("read Via output packet", error))?;
110            let marked = JByteArray::from(marked);
111            let marked = env
112                .convert_byte_array(&marked)
113                .map_err(|error| Self::jni_error("copy Via output packet", error))?;
114            let Some((&direction, packet)) = marked.split_first() else {
115                return Err(PacketError::MalformedValue(
116                    "Via emitted an empty marked packet".to_owned(),
117                ));
118            };
119            total_bytes = total_bytes
120                .checked_add(packet.len())
121                .ok_or(PacketError::OutOfBounds)?;
122            if total_bytes > MAX_EXCHANGE_BYTES || packet.len() > MAX_PACKET_DATA_SIZE {
123                return Err(PacketError::TooLong(total_bytes));
124            }
125            match direction {
126                0 => batch.serverbound.push(packet.to_vec()),
127                1 => batch.clientbound.push(packet.to_vec()),
128                _ => {
129                    return Err(PacketError::MalformedValue(
130                        "Via emitted an invalid packet direction".to_owned(),
131                    ));
132                }
133            }
134        }
135        Ok(batch)
136    }
137
138    fn jni_error(context: &str, error: JniError) -> PacketError {
139        PacketError::Other(format!("{context}: {error}"))
140    }
141
142    fn clear_exception(env: &mut JNIEnv<'_>) {
143        if matches!(env.exception_check(), Ok(true)) {
144            let _ = env.exception_describe();
145            let _ = env.exception_clear();
146        }
147    }
148}

150impl PacketTranslator for ViaTranslator {
151    fn exchange(
152        &self,
153        direction: PacketDirection,
154        packet_data: &[u8],
155    ) -> Result<TranslationBatch, PacketError> {
156        match direction {
157            PacketDirection::Serverbound => self.invoke("serverbound", Some(packet_data)),
158            PacketDirection::Clientbound => self.invoke("clientbound", Some(packet_data)),
159        }
160    }
161
162    fn poll(&self) -> Result<TranslationBatch, PacketError> {
163        self.invoke("poll", None)
164    }
165
166    fn bind_player(&self, player: Uuid) {
167        if self.closed.load(Ordering::Acquire) {
168            return;
169        }
170        let Ok(mut env) = self.vm.attach_current_thread() else {
171            log::warn!("Could not attach the JVM to bind player {player} to its Via channel");
172            return;
173        };
174        let channel = self.channel.lock();
175        let Some(channel) = channel.as_ref() else {
176            return;
177        };
178        let Ok(player_text) = env.new_string(player.to_string()) else {
179            return;
180        };
181        if let Err(error) = env.call_static_method(
182            BRIDGE_CLASS,
183            "bind",
184            "(Ljava/lang/String;Lfoton/network/FotonViaChannel;)V",
185            &[
186                JValue::Object(&player_text),
187                JValue::Object(channel.as_obj()),
188            ],
189        ) {
190            Self::clear_exception(&mut env);
191            log::warn!("Could not bind player {player} to its Via channel: {error}");
192        }
193    }
194
195    fn close(&self) -> Result<(), PacketError> {
196        if self.closed.load(Ordering::Acquire) {
197            return Ok(());
198        }
199        let mut env = self
200            .vm
201            .attach_current_thread()
202            .map_err(|error| Self::jni_error("attach JVM", error))?;
203        if self.closed.swap(true, Ordering::AcqRel) {
204            return Ok(());
205        }
206        let mut channel = self.channel.lock();
207        let Some(channel) = channel.take() else {
208            return Ok(());
209        };
210        if let Err(error) = env.call_method(channel.as_obj(), "close", "()V", &[]) {
211            Self::clear_exception(&mut env);
212            drop(channel);
213            return Err(Self::jni_error("close Via channel", error));
214        }
215        drop(channel);
216        Ok(())
217    }
218}
219
220impl Drop for ViaTranslator {
221    fn drop(&mut self) {
222        // Normal connection shutdown closes explicitly on a blocking worker.
223        // This fallback matters when opening the channel exceeded its deadline
224        // and the late worker result is dropped without reaching its caller.
225        let _ = PacketTranslator::close(self);
226    }
227}
