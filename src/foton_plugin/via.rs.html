1//! JNI adapter for the Paper channel hook consumed by `ViaVersion`.
2
3use std::sync::{
4    Arc,
5    atomic::{AtomicBool, Ordering},
6};
7
8use foton_protocol::{
9    packet_traits::{PacketDirection, PacketTranslation, PacketTranslator, TranslationBatch},
10    utils::{MAX_PACKET_DATA_SIZE, PacketError},
11};
12use foton_utils::locks::SyncMutex;
13use jni::{
14    JNIEnv, JavaVM,
15    errors::Error as JniError,
16    objects::{GlobalRef, JByteArray, JObjectArray, JValue, JValueOwned},
17};
18
19const BRIDGE_CLASS: &str = "foton/network/FotonViaBridge";
20const MAX_EXCHANGE_PACKETS: usize = 256;
21const MAX_EXCHANGE_BYTES: usize = MAX_PACKET_DATA_SIZE * 4;
22
23pub(crate) struct ViaTranslator {
24    vm: Arc<JavaVM>,
25    channel: SyncMutex<Option<GlobalRef>>,
26    closed: AtomicBool,
27}
28
29impl ViaTranslator {
30    pub(crate) fn open(vm: Arc<JavaVM>) -> Result<Option<PacketTranslation>, PacketError> {
31        let mut env = vm
32            .attach_current_thread()
33            .map_err(|error| Self::jni_error("attach JVM", error))?;
34        let channel = match env
35            .call_static_method(
36                BRIDGE_CLASS,
37                "open",
38                "()Lfoton/network/FotonViaChannel;",
39                &[],
40            )
41            .and_then(JValueOwned::l)
42        {
43            Ok(channel) => channel,
44            Err(error) => {
45                Self::clear_exception(&mut env);
46                return Err(Self::jni_error("open Via channel", error));
47            }
48        };
49        if channel.is_null() {
50            return Ok(None);
51        }
52        let channel = env
53            .new_global_ref(channel)
54            .map_err(|error| Self::jni_error("retain Via channel", error))?;
55        drop(env);
56        Ok(Some(PacketTranslation::new(Arc::new(Self {
57            vm,
58            channel: SyncMutex::new(Some(channel)),
59            closed: AtomicBool::new(false),
60        }))))
61    }
62
63    fn invoke(&self, method: &str, packet: Option<&[u8]>) -> Result<TranslationBatch, PacketError> {
64        if self.closed.load(Ordering::Acquire) {
65            return Err(PacketError::ConnectionClosed);
66        }
67        let mut env = self
68            .vm
69            .attach_current_thread()
70            .map_err(|error| Self::jni_error("attach JVM", error))?;
71        let channel = self.channel.lock();
72        let channel = channel.as_ref().ok_or(PacketError::ConnectionClosed)?;
73        let output = if let Some(packet) = packet {
74            let packet = env
75                .byte_array_from_slice(packet)
76                .map_err(|error| Self::jni_error("copy packet into JVM", error))?;
77            env.call_method(
78                channel.as_obj(),
79                method,
80                "([B)[[B",
81                &[JValue::Object(packet.as_ref())],
82            )
83        } else {
84            env.call_method(channel.as_obj(), method, "()[[B", &[])
85        };
86        let output = match output.and_then(JValueOwned::l) {
87            Ok(output) => output,
88            Err(error) => {
89                Self::clear_exception(&mut env);
90                return Err(Self::jni_error("run Via pipeline", error));
91            }
92        };
93        let output = JObjectArray::from(output);
94        let count = usize::try_from(
95            env.get_array_length(&output)
96                .map_err(|error| Self::jni_error("read Via output length", error))?,
97        )
98        .map_err(|_| PacketError::OutOfBounds)?;
99        if count > MAX_EXCHANGE_PACKETS {
100            return Err(PacketError::TooLong(count));
101        }
102
103        let mut batch = TranslationBatch::default();
104        let mut total_bytes = 0usize;
105        for index in 0..count {
106            let marked = env
107                .get_object_array_element(&output, index as i32)
108                .map_err(|error| Self::jni_error("read Via output packet", error))?;
109            let marked = JByteArray::from(marked);
110            let marked = env
111                .convert_byte_array(&marked)
112                .map_err(|error| Self::jni_error("copy Via output packet", error))?;
113            let Some((&direction, packet)) = marked.split_first() else {
114                return Err(PacketError::MalformedValue(
115                    "Via emitted an empty marked packet".to_owned(),
116                ));
117            };
118            total_bytes = total_bytes
119                .checked_add(packet.len())
120                .ok_or(PacketError::OutOfBounds)?;
121            if total_bytes > MAX_EXCHANGE_BYTES || packet.len() > MAX_PACKET_DATA_SIZE {
122                return Err(PacketError::TooLong(total_bytes));
123            }
124            match direction {
125                0 => batch.serverbound.push(packet.to_vec()),
126                1 => batch.clientbound.push(packet.to_vec()),
127                _ => {
128                    return Err(PacketError::MalformedValue(
129                        "Via emitted an invalid packet direction".to_owned(),
130                    ));
131                }
132            }
133        }
134        Ok(batch)
135    }
136
137    fn jni_error(context: &str, error: JniError) -> PacketError {
138        PacketError::Other(format!("{context}: {error}"))
139    }
140
141    fn clear_exception(env: &mut JNIEnv<'_>) {
142        if matches!(env.exception_check(), Ok(true)) {
143            let _ = env.exception_describe();
144            let _ = env.exception_clear();
145        }
146    }
147}
148
149impl PacketTranslator for ViaTranslator {
150    fn exchange(
151        &self,
152        direction: PacketDirection,
153        packet_data: &[u8],
154    ) -> Result<TranslationBatch, PacketError> {
155        match direction {
156            PacketDirection::Serverbound => self.invoke("serverbound", Some(packet_data)),
157            PacketDirection::Clientbound => self.invoke("clientbound", Some(packet_data)),
158        }
159    }
160
161    fn poll(&self) -> Result<TranslationBatch, PacketError> {
162        self.invoke("poll", None)
163    }
164
165    fn close(&self) -> Result<(), PacketError> {
166        if self.closed.load(Ordering::Acquire) {
167            return Ok(());
168        }
169        let mut env = self
170            .vm
171            .attach_current_thread()
172            .map_err(|error| Self::jni_error("attach JVM", error))?;
173        if self.closed.swap(true, Ordering::AcqRel) {
174            return Ok(());
175        }
176        let mut channel = self.channel.lock();
177        let Some(channel) = channel.take() else {
178            return Ok(());
179        };
180        if let Err(error) = env.call_method(channel.as_obj(), "close", "()V", &[]) {
181            Self::clear_exception(&mut env);
182            drop(channel);
183            return Err(Self::jni_error("close Via channel", error));
184        }
185        drop(channel);
186        Ok(())
187    }
188}
189
190impl Drop for ViaTranslator {
191    fn drop(&mut self) {
192        // Normal connection shutdown closes explicitly on a blocking worker.
193        // This fallback matters when opening the channel exceeded its deadline
194        // and the late worker result is dropped without reaching its caller.
195        let _ = PacketTranslator::close(self);
196    }
197}
