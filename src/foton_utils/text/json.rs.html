1//! Minecraft's JSON text form of a component.
2//!
3//! The component model derives its serde form field by field, and that form
4//! is Minecraft's but for two details: an RGB color is `{"rgb": [r, g, b]}`
5//! where Minecraft writes `"#rrggbb"`, and a child or argument must be an
6//! object where Minecraft also accepts a bare string. These two functions
7//! translate at the border, so the server list and Adventure's Gson
8//! serializer on the plugin side each read what they expect.
9
10use serde_json::{Value, json};
11use text_components::TextComponent;
12use text_components::format::Color;
13
14use super::DisplayResolutor;
15
16/// The JSON text of a component.
17#[must_use]
18pub fn to_json(component: &TextComponent) -> Value {
19    let Ok(mut value) = serde_json::to_value(component) else {
20        return json!({ "text": component.to_plain(&DisplayResolutor) });
21    };
22    rgb_to_hex(&mut value);
23    value
24}
25
26/// The component a JSON text object describes, or `None` when `text` is not
27/// one.
28#[must_use]
29pub fn from_json(text: &str) -> Option<TextComponent> {
30    let mut value = serde_json::from_str::<Value>(text)
31        .ok()
32        .filter(Value::is_object)?;
33    to_model(&mut value);
34    serde_json::from_value(value).ok()
35}
36
37/// Rewrites every `{"rgb": [r, g, b]}` color as `"#rrggbb"`.
38fn rgb_to_hex(value: &mut Value) {
39    match value {
40        Value::Object(map) => {
41            if let Some(color) = map.get_mut("color")
42                && let Some(rgb) = color.get("rgb").and_then(Value::as_array)
43                && let [r, g, b] = rgb.as_slice()
44                && let (Some(r), Some(g), Some(b)) = (r.as_u64(), g.as_u64(), b.as_u64())
45            {
46                *color = Value::String(format!("#{r:02x}{g:02x}{b:02x}"));
47            }
48            map.values_mut().for_each(rgb_to_hex);
49        }
50        Value::Array(items) => items.iter_mut().for_each(rgb_to_hex),
51        _ => {}
52    }
53}
54
55/// Rewrites `"#rrggbb"` colors into the model's form and expands bare-string
56/// children and arguments into text components.
57fn to_model(value: &mut Value) {
58    match value {
59        Value::Object(map) => {
60            if let Some(color) = map.get_mut("color")
61                && let Some(Color::Rgb(r, g, b)) = color.as_str().and_then(Color::from_hex)
62            {
63                *color = json!({ "rgb": [r, g, b] });
64            }
65            for key in ["extra", "with"] {
66                if let Some(Value::Array(items)) = map.get_mut(key) {
67                    for item in items.iter_mut() {
68                        if let Value::String(text) = item {
69                            *item = json!({ "text": text });
70                        }
71                    }
72                }
73            }
74            map.values_mut().for_each(to_model);
75        }
76        Value::Array(items) => items.iter_mut().for_each(to_model),
77        _ => {}
78    }
79}

81#[cfg(test)]
82mod tests {
83    use text_components::TextComponent;
84    use text_components::format::Color;
85
86    use super::{from_json, to_json};
87
88    /// What Adventure's Gson serializer writes, as the plugin API configures
89    /// it, for a MOTD like Zelda Civ's: a hex color, a named one, bold, a
90    /// newline child and click and hover events. Taken from its own output.
91    const ADVENTURE_MOTD: &str = r##"{"extra":[{"bold":true,"color":"#FDC14B","text":"Zelda "},{"color":"dark_gray","text":"| "},{"text":"\n"},{"click_event":{"action":"run_command","command":"/help"},"hover_event":{"action":"show_text","value":{"text":"hi"}},"text":"x"}],"text":""}"##;
92
93    /// A component a plugin built arrives with its structure. Reading it as
94    /// plain text instead would put the raw JSON in the server list.
95    #[test]
96    fn adventure_json_reads_as_a_component() {
97        let parsed = from_json(ADVENTURE_MOTD).expect("Adventure's JSON is a component");
98        assert_eq!(parsed.children.len(), 4, "every child survives");
99        assert_eq!(
100            parsed.children[0].format.color,
101            Some(Color::Rgb(0xfd, 0xc1, 0x4b)),
102            "a hex color is read as the color it names"
103        );
104    }
105
106    /// Minecraft's shorthand for a text child is a bare string.
107    #[test]
108    fn a_bare_string_child_is_a_text_component() {
109        let parsed = from_json(r#"{"text":"","extra":["a",{"text":"b"}]}"#).expect("a component");
110        assert_eq!(parsed.children[0], TextComponent::plain("a"));
111    }
112
113    /// Adventure and the client read a hex color, and nothing else, for an
114    /// RGB one.
115    #[test]
116    fn an_rgb_color_is_written_as_hex() {
117        let parsed = from_json(r##"{"text":"a","color":"#FDC14B"}"##).expect("a component");
118        assert_eq!(to_json(&parsed)["color"], "#fdc14b");
119    }
120
121    /// The vanilla join message goes to the plugins and comes back unchanged
122    /// when none touched it.
123    #[test]
124    fn a_translatable_message_survives_the_round_trip() {
125        let json = r#"{"translate":"multiplayer.player.joined","color":"yellow","with":[{"text":"Steve"}]}"#;
126        let parsed = from_json(json).expect("a component");
127        assert_eq!(from_json(&to_json(&parsed).to_string()), Some(parsed));
128    }
129}
