4pub(super) fn parse_uuid(tag: &NbtTag) -> Result<Uuid, ComponentDecodeError> {
5    match tag {
6        NbtTag::String(value) => {
7            Uuid::parse_str(&value.to_string()).map_err(|_| invalid("uuid", "a UUID"))
8        }
9        NbtTag::IntArray(values) | NbtTag::List(NbtList::Int(values)) if values.len() == 4 => {
10            Ok(Uuid::from_u64_pair(
11                (u64::from(values[0] as u32) << 32) | u64::from(values[1] as u32),
12                (u64::from(values[2] as u32) << 32) | u64::from(values[3] as u32),
13            ))
14        }
15        _ => Err(invalid("uuid", "a UUID")),
16    }
17}

19pub(super) fn parse_color(value: &str) -> Result<Color, ComponentDecodeError> {
20    let color = match value {
21        "aqua" => Color::Aqua,
22        "black" => Color::Black,
23        "blue" => Color::Blue,
24        "dark_aqua" => Color::DarkAqua,
25        "dark_blue" => Color::DarkBlue,
26        "dark_gray" => Color::DarkGray,
27        "dark_green" => Color::DarkGreen,
28        "dark_purple" => Color::DarkPurple,
29        "dark_red" => Color::DarkRed,
30        "gold" => Color::Gold,
31        "gray" => Color::Gray,
32        "green" => Color::Green,
33        "light_purple" => Color::LightPurple,
34        "red" => Color::Red,
35        "white" => Color::White,
36        "yellow" => Color::Yellow,
37        value => {
38            let Some(hex) = value.strip_prefix('#') else {
39                return Err(invalid("color", "a text color"));
40            };
41            let rgb = u32::from_str_radix(hex, 16)
42                .ok()
43                .filter(|_| !hex.is_empty())
44                .filter(|value| *value <= 0x00ff_ffff)
45                .ok_or_else(|| invalid("color", "a text color"))?;
46            Color::Rgb((rgb >> 16) as u8, (rgb >> 8) as u8, rgb as u8)
47        }
48    };
49    Ok(color)
50}

52pub(super) fn parse_shadow_color(tag: &NbtTag) -> Result<i32, ComponentDecodeError> {
53    if let Some(value) = as_i32(tag) {
54        return Ok(value);
55    }
56    let NbtTag::List(list) = tag else {
57        return Err(invalid(
58            "shadow_color",
59            "an ARGB integer or four-number list",
60        ));
61    };
62    if nbt_list_len(list) != 4
63        || !matches!(
64            list,
65            NbtList::Byte(_)
66                | NbtList::Short(_)
67                | NbtList::Int(_)
68                | NbtList::Long(_)
69                | NbtList::Float(_)
70                | NbtList::Double(_)
71        )
72    {
73        return Err(invalid(
74            "shadow_color",
75            "an ARGB integer or four-number list",
76        ));
77    }
78    let values = list.as_nbt_tags();
79    let mut channels = [0_u32; 4];
80    for (channel, value) in channels.iter_mut().zip(values.iter()) {
81        let value = as_f32(value)
82            .ok_or_else(|| invalid("shadow_color", "an ARGB integer or four-number list"))?;
83        *channel = ((value * 255.0).floor() as i32 as u32) & 0xff;
84    }
85    Ok(((channels[3] << 24) | (channels[0] << 16) | (channels[1] << 8) | channels[2]) as i32)
86}

88pub(super) fn required_compound<'a>(
89    compound: &'a NbtCompound,
90    field: &'static str,
91) -> Result<&'a NbtCompound, ComponentDecodeError> {
92    match compound.get(field) {
93        Some(NbtTag::Compound(value)) => Ok(value),
94        Some(_) => Err(invalid(field, "a compound")),
95        None => Err(ComponentDecodeError::MissingField(field)),
96    }
97}

121pub(super) fn required_identifier(
122    compound: &NbtCompound,
123    field: &'static str,
124) -> Result<String, ComponentDecodeError> {
125    let value = required_string(compound, field)?;
126    if !is_identifier(&value) {

145pub(super) fn required_chat_string(
146    compound: &NbtCompound,
147    field: &'static str,
148) -> Result<String, ComponentDecodeError> {
149    let value = required_string(compound, field)?;
150    if value
151        .chars()
152        .any(|character| character == '\u{a7}' || character < ' ' || character == '\u{7f}')
153    {
154        return Err(invalid(field, "a chat string"));
155    }
156    Ok(value)

223pub(super) fn validate_player_name(value: &str) -> Result<(), ComponentDecodeError> {
224    if value.encode_utf16().count() > 16
225        || value
226            .chars()
227            .any(|character| character <= ' ' || character >= '\u{7f}')
228    {
229        return Err(invalid("name", "a valid player name"));
230    }
231    Ok(())
232}

234pub(super) fn is_identifier(value: &str) -> bool {
235    let (namespace, path) = value
236        .split_once(':')
237        .map_or(("minecraft", value), |(namespace, path)| (namespace, path));
238    namespace != ".."
239        && namespace.chars().all(|character| {
240            character.is_ascii_lowercase()
241                || character.is_ascii_digit()
242                || character == '_'
243                || character == '-'
244                || character == '.'
245        })
246        && path.chars().all(|character| {
247            character.is_ascii_lowercase()
248                || character.is_ascii_digit()
249                || matches!(character, '_' | '-' | '.' | '/')
250        })
251}

253pub(super) fn is_allowed_url(value: &str) -> bool {
254    let Some((scheme, remainder)) = value.split_once(':') else {
255        return false;
256    };
257    matches!(scheme.to_ascii_lowercase().as_str(), "http" | "https")
258        && !remainder.is_empty()
259        && !value.chars().any(char::is_whitespace)
260}
