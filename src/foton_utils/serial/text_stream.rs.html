13/// Test-only observations of child-list classification work.
14#[cfg(feature = "codec-test-support")]
15pub mod child_classification_work {
16    use std::cell::Cell;
17
18    thread_local! { static CLASSIFICATIONS: Cell<usize> = const { Cell::new(0) }; }
19
20    pub(super) fn visit() {
21        CLASSIFICATIONS.set(CLASSIFICATIONS.get() + 1);
22    }
23
24    /// Counts child values classified by an actual codec route.
25    pub fn measure(operation: impl FnOnce()) -> usize {
26        CLASSIFICATIONS.set(0);
27        operation();
28        CLASSIFICATIONS.get()
29    }
30}

32/// Encodes text after a budget-limited wire-size pass.
33pub fn write_bounded(value: &TextComponent, maximum: usize, writer: &mut dyn Write) -> Result<()> {
34    let mut sink = sink();
35    write(value, &mut LimitedWriter::new(&mut sink, maximum))?;
36    write(value, &mut LimitedWriter::new(writer, maximum))
37}
38/// Encodes the component codec representation without materializing owned NBT.
39pub fn write(value: &TextComponent, writer: &mut dyn NbtWrite) -> Result<()> {
40    writer.write_all(&[if plain(value) { 8 } else { 10 }])?;
41    write_payload(value, writer, 0)
42}
43/// Returns the text codec's concrete NBT tag ID.
44pub(crate) fn tag_id(value: &TextComponent) -> u8 {
45    if plain(value) { 8 } else { 10 }
46}
47fn plain(value: &TextComponent) -> bool {
48    matches!(value.content, Content::Text { .. })
49        && value.children.is_empty()
50        && value.format.is_none()
51        && value.interactions.is_none()
52}
53fn invalid(message: &'static str) -> Error {
54    Error::new(ErrorKind::InvalidData, message)
55}
56/// Streams a Rust string as length-prefixed MUTF-8.
57pub(crate) fn write_utf(value: &str, writer: &mut dyn NbtWrite) -> Result<()> {
58    writer.ensure_remaining(value.len().saturating_add(2))?;
59    let mut length = 0usize;
60    for unit in value.encode_utf16() {
61        length += if (1..=0x7f).contains(&unit) {
62            1
63        } else if unit <= 0x7ff {
64            2
65        } else {
66            3
67        };
68        writer.ensure_remaining(length.saturating_add(2))?;
69        if length > u16::MAX as usize {
70            return Err(invalid("text NBT string exceeds u16 length"));
71        }
72    }
73    writer.write_all(&(length as u16).to_be_bytes())?;
74    for unit in value.encode_utf16() {
75        if (1..=0x7f).contains(&unit) {
76            writer.write_all(&[unit as u8])?;
77        } else if unit <= 0x7ff {
78            writer.write_all(&[0xc0 | (unit >> 6) as u8, 0x80 | (unit & 63) as u8])?;
79        } else {
80            writer.write_all(&[
81                0xe0 | (unit >> 12) as u8,
82                0x80 | ((unit >> 6) & 63) as u8,
83                0x80 | (unit & 63) as u8,
84            ])?;
85        }
86    }
87    Ok(())
88}
89fn field(kind: u8, name: &str, writer: &mut dyn NbtWrite) -> Result<()> {
90    writer.write_all(&[kind])?;
91    write_utf(name, writer)
92}
93fn compound_field(name: &str, writer: &mut dyn NbtWrite, depth: usize) -> Result<()> {
94    if depth + 1 >= writer.container_limit() {
95        return Err(invalid("text nesting exceeds limit"));
96    }
97    field(10, name, writer)
98}
99fn string(name: &str, value: &str, writer: &mut dyn NbtWrite) -> Result<()> {
100    field(8, name, writer)?;
101    write_utf(value, writer)
102}
103fn boolean(name: &str, value: bool, writer: &mut dyn NbtWrite) -> Result<()> {
104    field(1, name, writer)?;
105    writer.write_all(&[u8::from(value)])
106}
107fn integer(name: &str, value: i32, writer: &mut dyn NbtWrite) -> Result<()> {
108    field(3, name, writer)?;
109    writer.write_all(&value.to_be_bytes())
110}
111fn nested(
112    name: &str,
113    value: &TextComponent,
114    writer: &mut dyn NbtWrite,
115    depth: usize,
116) -> Result<()> {
117    field(if plain(value) { 8 } else { 10 }, name, writer)?;
118    write_payload(value, writer, depth + 1)

120fn nbt(name: &str, value: &NbtTag, writer: &mut dyn NbtWrite, depth: usize) -> Result<()> {
121    field(value.id(), name, writer)?;
122    nbt_stream::write_payload(value, writer, depth + 1)
123}
124fn end(writer: &mut dyn NbtWrite) -> Result<()> {
125    writer.write_all(&[0])
126}
127fn list(
128    name: &str,
129    values: &[TextComponent],
130    writer: &mut dyn NbtWrite,
131    depth: usize,
132) -> Result<()> {
133    field(9, name, writer)?;
134    writer.ensure_remaining(values.len().saturating_mul(2).saturating_add(5))?;
135    let all_plain = !values.is_empty()
136        && values.iter().all(|value| {
137            #[cfg(feature = "codec-test-support")]
138            child_classification_work::visit();
139            plain(value)
140        });
141    if !all_plain && depth + 1 >= writer.container_limit() {
142        return Err(invalid("text nesting exceeds limit"));
143    }
144    writer.write_all(&[if all_plain { 8 } else { 10 }])?;
145    writer.write_all(
146        &i32::try_from(values.len())
147            .map_err(|_| invalid("text list exceeds i32 length"))?
148            .to_be_bytes(),
149    )?;
150    for value in values {
151        if !all_plain && plain(value) {
152            nested("", value, writer, depth + 2)?;
153            end(writer)?;
154        } else {
155            write_payload(value, writer, depth + 2)?;
156        }
157    }
158    Ok(())
159}
160fn custom(value: &CustomData, writer: &mut dyn NbtWrite, depth: usize) -> Result<()> {
161    string("id", &value.id, writer)?;
162    if let Payload::Nbt(value) = &value.payload {
163        nbt("payload", value.as_nbt(), writer, depth)?;
164    }
165    Ok(())
166}
167/// Streams the text payload without its NBT tag ID.
168pub fn write_payload(value: &TextComponent, writer: &mut dyn NbtWrite, depth: usize) -> Result<()> {
169    if !plain(value) && depth >= writer.container_limit() {
170        return Err(invalid("text nesting exceeds limit"));
171    }
172    if plain(value)
173        && let Content::Text { text } = &value.content
174    {
175        return write_utf(text, writer);
176    }
177    content(&value.content, writer, depth)?;
178    style_fields(&value.format, &value.interactions, writer, depth)?;
179    if !value.children.is_empty() {
180        list("extra", &value.children, writer, depth)?;
181    }
182    end(writer)
183}
184/// Streams only the style of a component: `Style.Serializer.CODEC`'s compound.
185pub fn write_style(
186    format: &Format,
187    interactions: &Interactivity,
188    writer: &mut dyn NbtWrite,
189) -> Result<()> {
190    writer.write_all(&[10])?;
191    style_fields(format, interactions, writer, 0)?;
192    end(writer)
193}
194/// Streams the style fields of a component, without its content or its end tag.
195fn style_fields(
196    format: &Format,
197    interactions: &Interactivity,
198    writer: &mut dyn NbtWrite,
199    depth: usize,
200) -> Result<()> {
201    if let Some(color) = &format.color {
202        // Color's display uses the vanilla named/hex codec spelling.
203        string("color", &color.to_string(), writer)?;
204    }
205    if let Some(font) = &format.font {
206        string("font", font, writer)?;
207    }
208    for (name, value) in [
209        ("bold", format.bold),
210        ("italic", format.italic),
211        ("underlined", format.underlined),
212        ("strikethrough", format.strikethrough),
213        ("obfuscated", format.obfuscated),
214    ] {
215        if let Some(value) = value {
216            boolean(name, value, writer)?;
217        }
218    }
219    if let Some(color) = format.shadow_color {
220        integer("shadow_color", color, writer)?;
221    }
222    if let Some(insertion) = &interactions.insertion {
223        string("insertion", insertion, writer)?;
224    }
225    if let Some(hover) = &interactions.hover {
226        compound_field("hover_event", writer, depth)?;
227        hover_event(hover, writer, depth + 1)?;
228        end(writer)?;
229    }
230    if let Some(click) = &interactions.click {
231        compound_field("click_event", writer, depth)?;
232        click_event(click, writer, depth + 1)?;
233        end(writer)?;
234    }
235    Ok(())
236}

241fn content(value: &Content, writer: &mut dyn NbtWrite, depth: usize) -> Result<()> {
242    match value {
243        Content::Text { text } => string("text", text, writer),
244        Content::Keybind { keybind } => string("keybind", keybind, writer),
245        Content::Translate(msg) => {
246            string("translate", &msg.key, writer)?;
247            if let Some(fallback) = &msg.fallback {
248                string("fallback", fallback, writer)?;
249            }
250            if let Some(args) = &msg.args {
251                list("with", args, writer, depth)?;
252            }
253            Ok(())
254        }
255        Content::Custom(value) => {
256            compound_field("custom", writer, depth)?;
257            custom(value, writer, depth + 1)?;
258            end(writer)
259        }
260        Content::Object(Object::Atlas {
261            atlas,
262            sprite,
263            fallback,
264        }) => {
265            if atlas != "minecraft:blocks" {
266                string("atlas", atlas, writer)?;
267            }
268            string("sprite", sprite, writer)?;
269            if let Some(fallback) = fallback {
270                nested("fallback", fallback, writer, depth)?;
271            }
272            Ok(())
273        }
274        Content::Object(Object::Player {
275            player,
276            hat,
277            fallback,
278        }) => {
279            compound_field("player", writer, depth)?;
280            if let Some(id) = player.id {
281                field(11, "id", writer)?;
282                writer.write_all(&4_i32.to_be_bytes())?;
283                for value in id {
284                    writer.write_all(&value.to_be_bytes())?;
285                }
286            }
287            for (name, value) in [
288                ("name", &player.name),
289                ("texture", &player.texture),
290                ("cape", &player.cape),
291                ("elytra", &player.elytra),
292            ] {
293                if let Some(value) = value {
294                    string(name, value, writer)?;
295                }
296            }
297            if let Some(model) = player.model {
298                string(
299                    "model",
300                    match model {
301                        PlayerModel::Slim => "slim",
302                        PlayerModel::Wide => "wide",
303                    },
304                    writer,
305                )?;
306            }
307            if !player.properties.is_empty() {
308                if depth + 3 >= writer.container_limit() {
309                    return Err(invalid("text nesting exceeds limit"));
310                }
311                field(9, "properties", writer)?;
312                writer.write_all(&[10])?;
313                writer.write_all(
314                    &i32::try_from(player.properties.len())
315                        .map_err(|_| invalid("too many player properties"))?
316                        .to_be_bytes(),
317                )?;
318                for property in &player.properties {
319                    string("name", &property.name, writer)?;
320                    string("value", &property.value, writer)?;
321                    if let Some(signature) = &property.signature {
322                        string("signature", signature, writer)?;
323                    }
324                    end(writer)?;
325                }
326            }
327            end(writer)?;
328            if !hat {
329                boolean("hat", false, writer)?;
330            }
331            if let Some(fallback) = fallback {
332                nested("fallback", fallback, writer, depth)?;
333            }
334            Ok(())
335        }
336        Content::Resolvable(Resolvable::Scoreboard {
337            selector,
338            objective,
339        }) => {
340            compound_field("score", writer, depth)?;
341            string("name", selector, writer)?;
342            string("objective", objective, writer)?;
343            end(writer)
344        }
345        Content::Resolvable(Resolvable::Entity {
346            selector,
347            separator,
348        }) => {
349            string("selector", selector, writer)?;
350            if let Some(separator) = separator {
351                nested("separator", separator, writer, depth)?;
352            }
353            Ok(())
354        }
355        Content::Resolvable(Resolvable::NBT {
356            path,
357            interpret,
358            plain,
359            separator,
360            source,
361        }) => {
362            string("nbt", path, writer)?;
363            if *interpret {
364                boolean("interpret", true, writer)?;
365            }
366            if *plain {
367                boolean("plain", true, writer)?;
368            }
369            if let Some(separator) = separator {
370                nested("separator", separator, writer, depth)?;
371            }
372            let (name, value) = match source {
373                NbtSource::Entity(value) => ("entity", value),
374                NbtSource::Block(value) => ("block", value),
375                NbtSource::Storage(value) => ("storage", value),
376            };
377            string(name, value, writer)
378        }
379    }
380}
381fn hover_event(value: &HoverEvent, writer: &mut dyn NbtWrite, depth: usize) -> Result<()> {
382    match value {
383        HoverEvent::ShowText { value } => {
384            string("action", "show_text", writer)?;
385            nested("value", value, writer, depth)
386        }
387        HoverEvent::ShowItem {
388            id,
389            count,
390            components,
391        } => {
392            string("action", "show_item", writer)?;
393            string("id", id, writer)?;
394            if *count != 1 {
395                integer("count", *count, writer)?;
396            }
397            if let Some(value) = components
398                && !value.is_empty_compound()
399            {
400                nbt("components", value.as_nbt(), writer, depth)?;
401            }
402            Ok(())
403        }
404        HoverEvent::ShowEntity { name, id, uuid } => {
405            string("action", "show_entity", writer)?;
406            string("id", id, writer)?;
407            field(11, "uuid", writer)?;
408            writer.write_all(&4_i32.to_be_bytes())?;
409            writer.write_all(uuid.as_bytes())?;
410            if let Some(name) = name {
411                nested("name", name, writer, depth)?;
412            }
413            Ok(())
414        }
415    }
416}
417fn click_event(value: &ClickEvent, writer: &mut dyn NbtWrite, depth: usize) -> Result<()> {
418    match value {
419        ClickEvent::OpenUrl { url } => {
420            string("action", "open_url", writer)?;
421            string("url", url, writer)
422        }
423        ClickEvent::RunCommand { command } => {
424            string("action", "run_command", writer)?;
425            string("command", command, writer)
426        }
427        ClickEvent::SuggestCommand { command } => {
428            string("action", "suggest_command", writer)?;
429            string("command", command, writer)
430        }
431        ClickEvent::ChangePage { page } => {
432            string("action", "change_page", writer)?;
433            integer("page", *page, writer)
434        }
435        ClickEvent::CopyToClipboard { value } => {
436            string("action", "copy_to_clipboard", writer)?;
437            string("value", value, writer)
438        }
439        ClickEvent::ShowDialog { dialog } => {
440            string("action", "show_dialog", writer)?;
441            match dialog {
442                Dialog::Reference(value) => string("dialog", value, writer),
443                Dialog::Inline(value) => nbt("dialog", value.as_nbt(), writer, depth),
444            }
445        }
446        ClickEvent::Custom(value) => {
447            string("action", "custom", writer)?;
448            custom(value, writer, depth)
449        }
450    }
