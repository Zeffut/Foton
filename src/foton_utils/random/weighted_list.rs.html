1//! A weighted list of values.
2//!
3//! Vanilla parity: `net.minecraft.util.random.WeightedList`.
4
5use std::io::{Result, Write};
6
7use crate::codec::VarInt;
8use crate::random::Random;
9use crate::serial::WriteTo;
10
11/// One entry and the weight it is drawn with.
12///
13/// Vanilla parity: `WeightedList.Weighted`.
14#[derive(Clone, Debug, PartialEq, Eq)]
15pub struct Weighted<T> {
16    /// The value this entry hands back when it is drawn.
17    pub value: T,
18    /// Its share of the draw, against the sum of every entry's.
19    pub weight: i32,
20}
21
22/// A list of values drawn in proportion to their weights.
23///
24/// Vanilla parity: `WeightedList`. Vanilla precomputes a running total and
25/// binary-searches it; the lists Foton builds are a handful of entries long,
26/// so this walks them instead.
27#[derive(Clone, Debug, PartialEq, Eq)]
28pub struct WeightedList<T> {
29    entries: Vec<Weighted<T>>,
30    total_weight: i64,
31}
32
33impl<T> Default for WeightedList<T> {
34    fn default() -> Self {
35        Self::empty()
36    }
37}
38
39impl<T> WeightedList<T> {
40    /// Returns the list with nothing in it.
41    ///
42    /// Vanilla parity: `WeightedList.of()`.
43    #[must_use]
44    pub const fn empty() -> Self {
45        Self {
46            entries: Vec::new(),
47            total_weight: 0,
48        }
49    }
50
51    /// Returns a list holding one value at weight one.
52    ///
53    /// Vanilla parity: the single-value `WeightedList.of`.
54    #[must_use]
55    pub fn single(value: T) -> Self {
56        Self::new(vec![Weighted { value, weight: 1 }])
57    }
58
59    /// Returns a list over the given entries.
60    ///
61    /// Vanilla parity: the list-taking `WeightedList.of`. Vanilla's codec
62    /// refuses a negative weight; the callers here are generated data and
63    /// saved NBT, so a negative weight is clamped to zero instead. An entry
64    /// that can never be drawn is closer to vanilla than a spawner that
65    /// refuses to load.
66    #[must_use]
67    pub fn new(entries: Vec<Weighted<T>>) -> Self {
68        let entries: Vec<Weighted<T>> = entries
69            .into_iter()
70            .map(|entry| Weighted {
71                value: entry.value,
72                weight: entry.weight.max(0),
73            })
74            .collect();
75        let total_weight = entries.iter().map(|entry| i64::from(entry.weight)).sum();
76        Self {
77            entries,
78            total_weight,
79        }
80    }
81
82    /// Returns whether nothing can be drawn.
83    ///
84    /// Vanilla parity: `WeightedList.isEmpty`, which asks only about the entry
85    /// list -- a list of zero-weight entries is not empty to vanilla either.
86    #[must_use]
87    pub const fn is_empty(&self) -> bool {
88        self.entries.is_empty()
89    }
90
91    /// Returns how many entries the list holds.
92    #[must_use]
93    pub const fn len(&self) -> usize {
94        self.entries.len()
95    }
96
97    /// Returns the entries in declaration order.
98    #[must_use]
99    pub fn entries(&self) -> &[Weighted<T>] {
100        &self.entries
101    }
102
103    /// Returns the sum of every weight.
104    #[must_use]
105    pub const fn total_weight(&self) -> i64 {
106        self.total_weight
107    }
108
109    /// Returns the value a roll below the total weight lands on.
110    ///
111    /// Split out of the draw so the seeded and unseeded callers share the
112    /// selection itself, and so a test can name a roll.
113    #[must_use]
114    pub fn get_by_roll(&self, mut roll: i64) -> Option<&T> {
115        if self.total_weight <= 0 {
116            return None;
117        }
118        for entry in &self.entries {
119            roll -= i64::from(entry.weight);
120            if roll < 0 {
121                return Some(&entry.value);
122            }
123        }
124        None
125    }
126
127    /// Draws one value with Foton's unseeded runtime RNG.
128    ///
129    /// Vanilla parity: `WeightedList.getRandom`.
130    #[must_use]
131    pub fn get_random(&self) -> Option<&T> {
132        if self.total_weight <= 0 {
133            return None;
134        }
135        self.get_by_roll(rand::random_range(0..self.total_weight))
136    }
137
138    /// Draws one value with a seeded source.
139    #[must_use]
140    pub fn get_random_with<R: Random + ?Sized>(&self, random: &mut R) -> Option<&T> {
141        if self.total_weight <= 0 {
142            return None;
143        }
144        let bound = i32::try_from(self.total_weight).unwrap_or(i32::MAX);
145        self.get_by_roll(i64::from(random.next_i32_bounded(bound)))
146    }
147}
148
149impl<T: Clone> WeightedList<T> {
150    /// Draws one value and clones it.
151    #[must_use]
152    pub fn get_random_cloned(&self) -> Option<T> {
153        self.get_random().cloned()
154    }
155}
156
157impl<T: WriteTo> WriteTo for WeightedList<T> {
158    /// Vanilla parity: `WeightedList.streamCodec`, a length-prefixed list of
159    /// `Weighted.streamCodec` -- the value, then its weight as a `VarInt`.
160    fn write(&self, writer: &mut impl Write) -> Result<()> {
161        VarInt(self.entries.len() as i32).write(writer)?;
162        for entry in &self.entries {
163            entry.value.write(writer)?;
164            VarInt(entry.weight).write(writer)?;
165        }
166        Ok(())
167    }
168}

170#[cfg(test)]
171mod tests {
172    use super::*;
173
174    /// The roll boundaries are the whole of the selection, so an off-by-one
175    /// here would silently make one entry unreachable.
176    #[test]
177    fn every_roll_lands_inside_the_entry_that_owns_it() {
178        let list = WeightedList::new(vec![
179            Weighted {
180                value: "key",
181                weight: 3,
182            },
183            Weighted {
184                value: "consumables",
185                weight: 7,
186            },
187        ]);
188        assert_eq!(list.total_weight(), 10);
189        assert_eq!(list.get_by_roll(0), Some(&"key"));
190        assert_eq!(list.get_by_roll(2), Some(&"key"));
191        assert_eq!(list.get_by_roll(3), Some(&"consumables"));
192        assert_eq!(list.get_by_roll(9), Some(&"consumables"));
193        assert_eq!(list.get_by_roll(10), None);
194    }
195
196    /// A spawner whose potentials all carry weight zero must draw nothing
197    /// rather than always hand back the first entry.
198    #[test]
199    fn a_list_of_zero_weights_draws_nothing() {
200        let list = WeightedList::new(vec![Weighted {
201            value: 1,
202            weight: 0,
203        }]);
204        assert!(!list.is_empty(), "vanilla isEmpty only counts entries");
205        assert_eq!(list.get_random(), None);
206    }
