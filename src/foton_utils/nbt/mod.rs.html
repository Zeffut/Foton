20/// Mirrors vanilla `NbtUtils.compareNbt`.
21#[must_use]
22pub fn compare_nbt(
23    expected: Option<&NbtTag>,
24    actual: Option<&NbtTag>,
25    partial_list_matches: bool,
26) -> bool {
27    let Some(expected) = expected else {
28        return true;
29    };
30    let Some(actual) = actual else {
31        return false;
32    };
33
34    match (expected, actual) {
35        (NbtTag::Compound(expected), NbtTag::Compound(actual)) if partial_list_matches => {
36            compare_nbt_compounds(expected, actual, partial_list_matches)
37        }
38        (NbtTag::List(expected), NbtTag::List(actual)) if partial_list_matches => {
39            compare_lists_partially(expected, actual)
40        }
41        _ => nbt_tags_equal(expected, actual),
42    }
43}

45/// Compares NBT values with Vanilla's `Tag.equals` semantics.
46///
47/// Compounds are maps rather than ordered entry lists. Floating-point tags
48/// canonicalize NaNs while still distinguishing positive and negative zero.
49#[must_use]
50pub fn nbt_tags_equal(left: &NbtTag, right: &NbtTag) -> bool {
51    match (left, right) {
52        (NbtTag::Byte(left), NbtTag::Byte(right)) => left == right,
53        (NbtTag::Short(left), NbtTag::Short(right)) => left == right,
54        (NbtTag::Int(left), NbtTag::Int(right)) => left == right,
55        (NbtTag::Long(left), NbtTag::Long(right)) => left == right,
56        (NbtTag::Float(left), NbtTag::Float(right)) => float_tags_equal(*left, *right),
57        (NbtTag::Double(left), NbtTag::Double(right)) => double_tags_equal(*left, *right),
58        (NbtTag::ByteArray(left), NbtTag::ByteArray(right)) => left == right,
59        (NbtTag::String(left), NbtTag::String(right)) => left == right,
60        (NbtTag::List(left), NbtTag::List(right)) => {
61            let left = nbt_list_values(left);
62            let right = nbt_list_values(right);
63            left.len() == right.len()
64                && left
65                    .iter()
66                    .zip(&right)
67                    .all(|(left, right)| nbt_tags_equal(left, right))
68        }
69        (NbtTag::Compound(left), NbtTag::Compound(right)) => nbt_compounds_equal(left, right),
70        (NbtTag::IntArray(left), NbtTag::IntArray(right)) => left == right,
71        (NbtTag::LongArray(left), NbtTag::LongArray(right)) => left == right,
72        _ => false,
73    }
74}

76/// Compares compounds as Vanilla maps, independent of serialized entry order.
77#[must_use]
78pub fn nbt_compounds_equal(left: &NbtCompound, right: &NbtCompound) -> bool {
79    left.len() == right.len()
80        && left.iter().all(|(key, left_value)| {
81            right
82                .get(&key.to_str())
83                .is_some_and(|right_value| nbt_tags_equal(left_value, right_value))
84        })
85}
86
87const fn float_tags_equal(left: f32, right: f32) -> bool {
88    (left.is_nan() && right.is_nan()) || left.to_bits() == right.to_bits()
89}
90
91const fn double_tags_equal(left: f64, right: f64) -> bool {
92    (left.is_nan() && right.is_nan()) || left.to_bits() == right.to_bits()
93}

95/// Converts an NBT compound to Vanilla's map representation.
96///
97/// Vanilla rejects malformed modified UTF-8 and keeps the final occurrence of
98/// a duplicate compound key. `simdnbt` preserves raw strings and duplicates,
99/// so codecs crossing a trust boundary normalize them explicitly.
100#[must_use]
101pub fn normalize_nbt_compound(compound: NbtCompound) -> Option<NbtCompound> {
102    let mut normalized = NbtCompound::new();
103    for (key, value) in compound {
104        let key = key.try_into_string().ok()?;
105        let value = normalize_nbt_tag(value)?;
106        while normalized.remove(&key).is_some() {}
107        normalized.insert(key, value);
108    }
109    Some(normalized)
110}

112/// Converts an NBT value and all descendants to Vanilla's canonical in-memory
113/// representation.
114pub fn normalize_nbt_tag(tag: NbtTag) -> Option<NbtTag> {
115    match tag {
116        NbtTag::String(value) => Some(NbtTag::String(value.try_into_string().ok()?.into())),
117        NbtTag::List(list) => normalize_nbt_list(list).map(NbtTag::List),
118        NbtTag::Compound(compound) => normalize_nbt_compound(compound).map(NbtTag::Compound),
119        tag => Some(tag),
120    }
121}
122
123fn normalize_nbt_list(list: NbtList) -> Option<NbtList> {
124    match list {
125        NbtList::String(values) => values
126            .into_iter()
127            .map(|value| value.try_into_string().ok().map(Into::into))
128            .collect::<Option<Vec<_>>>()
129            .map(NbtList::String),
130        NbtList::List(values) => values
131            .into_iter()
132            .map(normalize_nbt_list)
133            .collect::<Option<Vec<_>>>()
134            .map(NbtList::List),
135        NbtList::Compound(values) => values
136            .into_iter()
137            .map(normalize_nbt_compound)
138            .collect::<Option<Vec<_>>>()
139            .map(NbtList::Compound),
140        list => Some(list),
141    }
142}

144/// Recursively applies Vanilla `CompoundTag.merge` semantics.
145pub fn merge_nbt_compounds(target: &mut NbtCompound, source: &NbtCompound) {
146    for (key, source_value) in source.iter() {
147        let key = key.to_str();
148        if let NbtTag::Compound(source_compound) = source_value
149            && let Some(NbtTag::Compound(target_compound)) = target.get_mut(&key)
150        {
151            merge_nbt_compounds(target_compound, source_compound);
152            continue;
153        }
154
155        while target.remove(&key).is_some() {}
156        target.insert(key.into_owned(), source_value.clone());
157    }
158}

160/// Returns the heap usage charged by Vanilla's `NbtAccounter` while decoding.
161///
162/// `None` indicates malformed modified UTF-8 or arithmetic overflow.
163#[must_use]
164pub fn vanilla_nbt_heap_size(tag: &NbtTag) -> Option<u64> {
165    match tag {
166        NbtTag::Byte(_) => Some(9),
167        NbtTag::Short(_) => Some(10),
168        NbtTag::Int(_) | NbtTag::Float(_) => Some(12),
169        NbtTag::Long(_) | NbtTag::Double(_) => Some(16),
170        NbtTag::ByteArray(values) => sized_array(24, 1, values.len()),
171        NbtTag::String(value) => sized_string(36, value),
172        NbtTag::List(values) => vanilla_nbt_list_heap_size(values),
173        NbtTag::Compound(values) => vanilla_nbt_compound_heap_size(values),
174        NbtTag::IntArray(values) => sized_array(24, 4, values.len()),
175        NbtTag::LongArray(values) => sized_array(24, 8, values.len()),
176    }
177}

179/// Charges a list without materializing it.
180///
181/// `NbtList::as_nbt_tags` deep-clones every element it hands back, so measuring
182/// a list through it cloned the whole subtree once per nesting level: a two
183/// megabyte `custom_data` component of 511 nested lists ending in a byte array
184/// peaked near a gigabyte of live heap, and the packet was rejected afterwards
185/// anyway. Vanilla never has this problem because `NbtAccounter` charges each
186/// tag as it is parsed rather than measuring a finished tree.
187///
188/// Every charge below is the one `vanilla_nbt_heap_size` gives the same tag, so
189/// the quota itself is unchanged.
190fn vanilla_nbt_list_heap_size(list: &NbtList) -> Option<u64> {
191    let (count, elements) = match list {
192        NbtList::Empty => (0, Some(0)),
193        NbtList::Byte(values) => (values.len(), fixed_elements(values.len(), 9)),
194        NbtList::Short(values) => (values.len(), fixed_elements(values.len(), 10)),
195        NbtList::Int(values) => (values.len(), fixed_elements(values.len(), 12)),
196        NbtList::Float(values) => (values.len(), fixed_elements(values.len(), 12)),
197        NbtList::Long(values) => (values.len(), fixed_elements(values.len(), 16)),
198        NbtList::Double(values) => (values.len(), fixed_elements(values.len(), 16)),
199        NbtList::ByteArray(arrays) => (
200            arrays.len(),
201            sum_sizes(arrays.iter().map(|array| sized_array(24, 1, array.len()))),
202        ),
203        NbtList::IntArray(arrays) => (
204            arrays.len(),
205            sum_sizes(arrays.iter().map(|array| sized_array(24, 4, array.len()))),
206        ),
207        NbtList::LongArray(arrays) => (
208            arrays.len(),
209            sum_sizes(arrays.iter().map(|array| sized_array(24, 8, array.len()))),
210        ),
211        NbtList::String(values) => (
212            values.len(),
213            sum_sizes(values.iter().map(|value| sized_string(36, value.as_str()))),
214        ),
215        NbtList::List(lists) => (
216            lists.len(),
217            sum_sizes(lists.iter().map(vanilla_nbt_list_heap_size)),
218        ),
219        NbtList::Compound(compounds) => (
220            compounds.len(),
221            sum_sizes(compounds.iter().map(vanilla_nbt_compound_heap_size)),
222        ),
223    };
224
225    let count = u64::try_from(count).ok()?;
226    36_u64
227        .checked_add(4_u64.checked_mul(count)?)?
228        .checked_add(elements?)
229}
230
231fn fixed_elements(count: usize, per_element: u64) -> Option<u64> {
232    u64::try_from(count).ok()?.checked_mul(per_element)
233}
234
235fn sum_sizes(mut sizes: impl Iterator<Item = Option<u64>>) -> Option<u64> {
236    sizes.try_fold(0_u64, |total, size| total.checked_add(size?))
237}

239fn vanilla_nbt_compound_heap_size(compound: &NbtCompound) -> Option<u64> {
240    let mut size = 48_u64;
241    let mut keys = FxHashSet::default();
242    for (key, value) in compound.iter() {
243        let key = key.to_owned().try_into_string().ok()?;
244        let key_units = u64::try_from(key.encode_utf16().count()).ok()?;
245        size = size.checked_add(28_u64.checked_add(2_u64.checked_mul(key_units)?)?)?;
246        size = size.checked_add(vanilla_nbt_heap_size(value)?)?;
247        if keys.insert(key) {
248            size = size.checked_add(36)?;
249        }
250    }
251    Some(size)
252}
253
254fn sized_string(base: u64, value: &simdnbt::Mutf8Str) -> Option<u64> {
255    let value = value.to_owned().try_into_string().ok()?;
256    let units = u64::try_from(value.encode_utf16().count()).ok()?;
257    base.checked_add(2_u64.checked_mul(units)?)
258}
259
260fn sized_array(base: u64, element_size: u64, len: usize) -> Option<u64> {
261    let len = u64::try_from(len).ok()?;
262    base.checked_add(element_size.checked_mul(len)?)
263}

265/// Compares two compounds with vanilla's partial compound semantics.
266#[must_use]
267pub fn compare_nbt_compounds(
268    expected: &NbtCompound,
269    actual: &NbtCompound,
270    partial_list_matches: bool,
271) -> bool {
272    if actual.len() < expected.len() {
273        return false;
274    }
275
276    expected.iter().all(|(key, expected_tag)| {
277        compare_nbt(
278            Some(expected_tag),
279            actual.get(&key.to_str()),
280            partial_list_matches,
281        )
282    })
283}
284
285fn compare_lists_partially(expected: &NbtList, actual: &NbtList) -> bool {
286    let expected = nbt_list_values(expected);
287    let actual = nbt_list_values(actual);
288    if expected.is_empty() {
289        return actual.is_empty();
290    }
291    if actual.len() < expected.len() {
292        return false;
293    }
294
295    expected.iter().all(|expected_tag| {
296        actual
297            .iter()
298            .any(|actual_tag| compare_nbt(Some(expected_tag), Some(actual_tag), true))
299    })
300}

302/// Returns the semantic values of a Vanilla list.
303///
304/// Vanilla 26.2 permits heterogeneous lists and wraps non-compound elements
305/// when serializing them through the homogeneous binary NBT format.
306#[must_use]
307pub fn nbt_list_values(list: &NbtList) -> Vec<NbtTag> {
308    list.as_nbt_tags()
309        .into_iter()
310        .map(unwrap_list_wrapper)
311        .collect()
312}

314/// Returns values exposed by Vanilla's `NbtOps` collection interface.
315///
316/// Numeric arrays are collections as well as ordinary list tags, so codecs
317/// built with `Codec.listOf()` accept all four representations.
318#[must_use]
319pub fn nbt_collection_values(tag: &NbtTag) -> Option<Vec<NbtTag>> {
320    match tag {
321        NbtTag::ByteArray(values) => Some(
322            values
323                .iter()
324                .map(|value| NbtTag::Byte(i8::from_ne_bytes([*value])))
325                .collect(),
326        ),
327        NbtTag::List(values) => Some(nbt_list_values(values)),
328        NbtTag::IntArray(values) => Some(values.iter().copied().map(NbtTag::Int).collect()),
329        NbtTag::LongArray(values) => Some(values.iter().copied().map(NbtTag::Long).collect()),
330        _ => None,
331    }
332}
333
334fn unwrap_list_wrapper(tag: NbtTag) -> NbtTag {
335    match tag {
336        NbtTag::Compound(mut compound) if compound.len() == 1 && compound.contains("") => {
337            let Some(value) = compound.take("") else {
338                return NbtTag::Compound(compound);
339            };
340            value
341        }
342        tag => tag,
343    }
344}

475    fn list_heap_size_charges_exactly_what_the_same_tags_charge() {
476        // `vanilla_nbt_list_heap_size` no longer goes through `as_nbt_tags`,
477        // because that clones the whole subtree once per nesting level and made

481        // here against the per-tag function it replaced. `as_nbt_tags` is the
482        // oracle, which is safe on data this small.
483        let mut nested_compound = NbtCompound::new();
484        nested_compound.insert("k", 1);

487            NbtList::Empty,
488            NbtList::Byte(vec![1, 2, 3]),
489            NbtList::Short(vec![1, 2]),
490            NbtList::Int(vec![1, 2, 3, 4]),
491            NbtList::Float(vec![1.0, 2.0]),

493            NbtList::Double(vec![1.0]),
494            NbtList::ByteArray(vec![vec![1, 2, 3], vec![]]),
495            NbtList::IntArray(vec![vec![1, 2], vec![3]]),
496            NbtList::LongArray(vec![vec![1]]),
497            NbtList::String(vec![

499                Mutf8String::from_string("e\u{301} and \u{1f600}".to_owned()),

507            let count = u64::try_from(tags.len()).expect("test list fits in u64");
508            let expected = tags.iter().try_fold(36 + 4 * count, |total, tag| {
