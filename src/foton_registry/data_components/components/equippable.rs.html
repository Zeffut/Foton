27/// The equippable component data.
28#[derive(Debug, Clone, PartialEq)]
29pub struct Equippable {
30    pub slot: EquipmentSlot,
31    pub equip_sound: SoundEventHolder,
32    pub asset_id: Option<Identifier>,
33    pub camera_overlay: Option<Identifier>,
34    pub allowed_entities: Option<EquippableAllowedEntities>,
35    pub dispensable: bool,
36    pub swappable: bool,
37    pub damage_on_hurt: bool,
38    pub equip_on_interact: bool,
39    pub can_be_sheared: bool,
40    pub shearing_sound: SoundEventHolder,
41}

43impl Equippable {
44    /// Returns whether this item can be equipped by the entity type.
45    #[must_use]
46    pub fn can_be_equipped_by(&self, entity_type: EntityTypeRef) -> bool {
47        self.allowed_entities
48            .as_ref()
49            .is_none_or(|allowed| allowed.contains(entity_type))
50    }
51}

89impl HashComponent for Equippable {
90    fn hash_component(&self, hasher: &mut ComponentHasher) {
91        let mut entries = Vec::new();
92        push_hash_entry(&mut entries, "slot", self.slot.name());
93        if self.equip_sound != SoundEventHolder::registry(&sound_events::ITEM_ARMOR_EQUIP_GENERIC) {
94            push_hash_entry(&mut entries, "equip_sound", &self.equip_sound);
95        }
96        if let Some(asset_id) = &self.asset_id {
97            push_hash_entry(&mut entries, "asset_id", &asset_id.to_string());
98        }
99        if let Some(camera_overlay) = &self.camera_overlay {
100            push_hash_entry(&mut entries, "camera_overlay", &camera_overlay.to_string());
101        }
102        if let Some(allowed_entities) = &self.allowed_entities {
103            push_hash_entry(&mut entries, "allowed_entities", allowed_entities);
104        }
105        if !self.dispensable {
106            push_hash_entry(&mut entries, "dispensable", &self.dispensable);
107        }
108        if !self.swappable {
109            push_hash_entry(&mut entries, "swappable", &self.swappable);
110        }
111        if !self.damage_on_hurt {
112            push_hash_entry(&mut entries, "damage_on_hurt", &self.damage_on_hurt);
113        }
114        if self.equip_on_interact {
115            push_hash_entry(&mut entries, "equip_on_interact", &self.equip_on_interact);
116        }
117        if self.can_be_sheared {
118            push_hash_entry(&mut entries, "can_be_sheared", &self.can_be_sheared);
119        }
120        if self.shearing_sound != SoundEventHolder::registry(&sound_events::ITEM_SHEARS_SNIP) {
121            push_hash_entry(&mut entries, "shearing_sound", &self.shearing_sound);
122        }
123
124        sort_map_entries(&mut entries);
125        hasher.start_map();
126        for entry in &entries {
127            hasher.put_raw_bytes(&entry.key_bytes);
128            hasher.put_raw_bytes(&entry.value_bytes);
129        }
130        hasher.end_map();
131    }
132}
133
134fn push_hash_entry<T: HashComponent + ?Sized>(entries: &mut Vec<HashEntry>, key: &str, value: &T) {
135    let mut key_hasher = ComponentHasher::new();
136    key_hasher.put_string(key);
137    let mut value_hasher = ComponentHasher::new();
138    value.hash_component(&mut value_hasher);
139    entries.push(HashEntry::new(key_hasher, value_hasher));
140}

142impl simdnbt::ToNbtTag for Equippable {
143    fn to_nbt_tag(self) -> simdnbt::owned::NbtTag {
144        use simdnbt::owned::{NbtCompound, NbtTag};
145
146        let mut compound = NbtCompound::new();
147        compound.insert("slot", self.slot.name());
148        if self.equip_sound != SoundEventHolder::registry(&sound_events::ITEM_ARMOR_EQUIP_GENERIC) {
149            compound.insert("equip_sound", self.equip_sound.to_nbt_tag());
150        }
151        if let Some(asset_id) = self.asset_id {
152            compound.insert("asset_id", asset_id.to_string());
153        }
154        if let Some(camera_overlay) = self.camera_overlay {
155            compound.insert("camera_overlay", camera_overlay.to_string());
156        }
157        if !self.dispensable {
158            compound.insert("dispensable", i8::from(self.dispensable));
159        }
160        if !self.swappable {
161            compound.insert("swappable", i8::from(self.swappable));
162        }
163        if !self.damage_on_hurt {
164            compound.insert("damage_on_hurt", i8::from(self.damage_on_hurt));
165        }
166        if self.equip_on_interact {
167            compound.insert("equip_on_interact", i8::from(self.equip_on_interact));
168        }
169        if self.can_be_sheared {
170            compound.insert("can_be_sheared", i8::from(self.can_be_sheared));
171        }
172        if self.shearing_sound != SoundEventHolder::registry(&sound_events::ITEM_SHEARS_SNIP) {
173            compound.insert("shearing_sound", self.shearing_sound.to_nbt_tag());
174        }
175        if let Some(allowed_entities) = self.allowed_entities {
176            compound.insert("allowed_entities", allowed_entities.to_nbt_tag());
177        }
178        NbtTag::Compound(compound)
179    }
180}

182impl simdnbt::FromNbtTag for Equippable {
183    fn from_nbt_tag(tag: simdnbt::borrow::NbtTag) -> Option<Self> {
184        let compound = tag.compound()?;
185        let slot_str = compound.get("slot")?.string()?.to_str();
186        let slot = EquipmentSlot::by_name(&slot_str)?;
187        let equip_sound = match compound.get("equip_sound") {
188            Some(tag) => SoundEventHolder::from_nbt_tag(tag)?,
189            None => SoundEventHolder::registry(&sound_events::ITEM_ARMOR_EQUIP_GENERIC),
190        };
191        let asset_id = match compound.get("asset_id") {
192            Some(tag) => Some(parse_identifier_nbt(tag)?),
193            None => None,
194        };
195        let camera_overlay = match compound.get("camera_overlay") {
196            Some(tag) => Some(parse_identifier_nbt(tag)?),
197            None => None,
198        };
199        let allowed_entities = match compound.get("allowed_entities") {
200            Some(tag) => Some(EquippableAllowedEntities::from_nbt_tag(tag)?),
201            None => None,
202        };
203        let dispensable = optional_bool(compound.get("dispensable"), true)?;
204        let swappable = optional_bool(compound.get("swappable"), true)?;
205        let damage_on_hurt = optional_bool(compound.get("damage_on_hurt"), true)?;
206        let equip_on_interact = optional_bool(compound.get("equip_on_interact"), false)?;
207        let can_be_sheared = optional_bool(compound.get("can_be_sheared"), false)?;
208        let shearing_sound = match compound.get("shearing_sound") {
209            Some(tag) => SoundEventHolder::from_nbt_tag(tag)?,
210            None => SoundEventHolder::registry(&sound_events::ITEM_SHEARS_SNIP),
211        };
212
213        Some(Self {
214            slot,
215            equip_sound,
216            asset_id,
217            camera_overlay,
218            allowed_entities,
219            dispensable,
220            swappable,
221            damage_on_hurt,
222            equip_on_interact,
223            can_be_sheared,
224            shearing_sound,
225        })
226    }
227}

229fn parse_identifier_nbt(tag: simdnbt::borrow::NbtTag) -> Option<Identifier> {
230    Identifier::from_str(&tag.string()?.to_str()).ok()
231}

233fn optional_bool(tag: Option<simdnbt::borrow::NbtTag<'_, '_>>, default: bool) -> Option<bool> {
234    match tag {
235        Some(tag) => tag.codec_bool(),
236        None => Some(default),
237    }

240impl nbt_encode::NbtEncode for Equippable {
241    fn nbt_id(&self) -> u8 {
242        10
243    }
244    fn write_nbt_payload(&self, writer: &mut dyn NbtWrite, depth: usize) -> io::Result<()> {
245        use foton_utils::serial::nbt_encode::{check_depth, end, field};
246        check_depth(depth)?;
247
248        field("slot", &(self.slot.name()), writer, depth)?;
249        if self.equip_sound != SoundEventHolder::registry(&sound_events::ITEM_ARMOR_EQUIP_GENERIC) {
250            field("equip_sound", &(self.equip_sound), writer, depth)?;
251        }
252        if let Some(asset_id) = &self.asset_id {
253            field("asset_id", &(asset_id), writer, depth)?;
254        }
255        if let Some(camera_overlay) = &self.camera_overlay {
256            field("camera_overlay", &(camera_overlay), writer, depth)?;
257        }
258        if !self.dispensable {
259            field("dispensable", &(i8::from(self.dispensable)), writer, depth)?;
260        }
261        if !self.swappable {
262            field("swappable", &(i8::from(self.swappable)), writer, depth)?;
263        }
264        if !self.damage_on_hurt {
265            field(
266                "damage_on_hurt",
267                &(i8::from(self.damage_on_hurt)),
268                writer,
269                depth,
270            )?;
271        }
272        if self.equip_on_interact {
273            field(
274                "equip_on_interact",
275                &(i8::from(self.equip_on_interact)),
276                writer,
277                depth,
278            )?;
279        }
280        if self.can_be_sheared {
281            field(
282                "can_be_sheared",
283                &(i8::from(self.can_be_sheared)),
284                writer,
285                depth,
286            )?;
287        }
288        if self.shearing_sound != SoundEventHolder::registry(&sound_events::ITEM_SHEARS_SNIP) {
289            field("shearing_sound", &(self.shearing_sound), writer, depth)?;
290        }
291        if let Some(allowed_entities) = &self.allowed_entities {
292            field("allowed_entities", &(allowed_entities), writer, depth)?;
293        }
294        end(writer)
295    }

319        let mut bytes = Vec::new();
320        tag.write(&mut bytes);

322            read_tag(&mut Cursor::new(bytes.as_slice())).expect("owned test tag should parse");
323        visitor(borrowed.as_tag())
