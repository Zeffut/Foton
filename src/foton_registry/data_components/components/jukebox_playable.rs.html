17/// A jukebox song attached to an item stack.
18///
19#[derive(Debug, Clone, PartialEq)]
20pub struct JukeboxPlayable {
21    song: RegistryHolder<JukeboxSong>,
22}
23
24impl JukeboxPlayable {
25    #[must_use]
26    pub const fn new(song: JukeboxSongRef) -> Self {
27        Self {
28            song: RegistryHolder::reference(song),
29        }
30    }
31
32    #[must_use]
33    pub const fn direct(song: JukeboxSongValue) -> Self {
34        Self {
35            song: RegistryHolder::direct(song),
36        }
37    }
38
39    #[must_use]
40    pub const fn song(&self) -> &RegistryHolder<JukeboxSong> {
41        &self.song
42    }
43
44    /// Decodes `JukeboxSong.CODEC`, which is a registry-fixed holder codec.
45    #[must_use]
46    pub fn from_persistent_nbt(tag: simdnbt::borrow::NbtTag) -> Option<Self> {
47        let key = Identifier::from_str(&tag.string()?.to_str()).ok()?;
48        REGISTRY.jukebox_songs.by_key(&key).map(Self::new)
49    }
50
51    /// Encodes `JukeboxSong.CODEC`, which is a registry-fixed holder codec.
52    pub fn to_persistent_nbt(&self) -> Result<NbtTag> {
53        let Some(song) = self.song.as_reference() else {
54            return Err(Error::other("Direct jukebox song holder is not persistent"));
55        };
56        Ok(NbtTag::String(song.key.to_string().into()))
57    }
58}
59
60impl HashComponent for JukeboxPlayable {
61    fn hash_component(&self, hasher: &mut ComponentHasher) {
62        self.song.hash_component(hasher);
63    }
64}
65
66impl ReadFrom for JukeboxPlayable {
67    fn read(data: &mut Cursor<&[u8]>) -> Result<Self> {
68        RegistryHolder::read(data).map(|song| Self { song })
69    }
70}
71
72impl WriteTo for JukeboxPlayable {
73    fn write(&self, writer: &mut impl Write) -> Result<()> {
74        self.song.write(writer)
75    }
76}
77
78impl nbt_encode::NbtEncode for JukeboxPlayable {
79    fn nbt_id(&self) -> u8 {
80        8
81    }
82    fn write_nbt_payload(&self, writer: &mut dyn NbtWrite, depth: usize) -> io::Result<()> {
83        let song = self
84            .song
85            .as_reference()
86            .ok_or_else(|| Error::other("Direct jukebox song holder is not persistent"))?;
87        song.key.write_nbt_payload(writer, depth)
88    }
89}

91#[cfg(test)]
92mod tests {
93    use std::io::Cursor;
94
95    use foton_utils::serial::{ReadFrom, WriteTo};
96    use simdnbt::owned::NbtTag;
97
98    use super::JukeboxPlayable;
99    use crate::init_vanilla_registry;
100    use crate::vanilla_jukebox_songs;
101
102    #[test]
103    fn registry_reference_round_trips_both_codecs() {
104        init_vanilla_registry();
105        let component = JukeboxPlayable::new(&vanilla_jukebox_songs::CAT);
106
107        let mut network = Vec::new();
108        component
109            .write(&mut network)
110            .expect("registry jukebox holder should encode");
111        assert_eq!(
112            JukeboxPlayable::read(&mut Cursor::new(network.as_slice()))
113                .expect("registry jukebox holder should decode"),
114            component
115        );
116
117        let nbt = component
118            .to_persistent_nbt()
119            .expect("reference is persistent");
120        assert_eq!(nbt, NbtTag::String("minecraft:cat".into()));
121    }
122
123    #[test]
124    fn direct_holder_round_trips_stream_and_is_not_persistent() {
125        use crate::jukebox_song::JukeboxSongValue;
126        use crate::sound_event::SoundEventHolder;
127        use foton_utils::Identifier;
128        use text_components::TextComponent;
129
130        init_vanilla_registry();
131        let component = JukeboxPlayable::direct(JukeboxSongValue {
132            sound_event: SoundEventHolder::Direct {
133                sound_id: Identifier::vanilla_static("custom_song"),
134                fixed_range: Some(12.0),
135            },
136            description: TextComponent::plain("Custom song"),
137            length_in_seconds: -1.0,
138            comparator_output: 99,
139        });
140        let mut network = Vec::new();
141        component
142            .write(&mut network)
143            .expect("direct holder should encode");
144        assert_eq!(
145            JukeboxPlayable::read(&mut Cursor::new(network.as_slice()))
146                .expect("direct holder should decode"),
147            component
148        );
149        assert!(component.to_persistent_nbt().is_err());
150    }
