15/// Controls movement and vibration behavior while an item is being used.
16#[derive(Debug, Clone, Copy)]
17pub struct UseEffects {
18    pub can_sprint: bool,
19    pub interact_vibrations: bool,
20    pub speed_multiplier: f32,
21}
22
23impl PartialEq for UseEffects {
24    fn eq(&self, other: &Self) -> bool {
25        self.can_sprint == other.can_sprint
26            && self.interact_vibrations == other.interact_vibrations
27            && java_float_equals(self.speed_multiplier, other.speed_multiplier)
28    }
29}

72impl ToNbtTag for UseEffects {
73    fn to_nbt_tag(self) -> NbtTag {
74        let mut compound = NbtCompound::new();
75        if self.can_sprint != Self::DEFAULT.can_sprint {
76            compound.insert("can_sprint", self.can_sprint);
77        }
78        if self.interact_vibrations != Self::DEFAULT.interact_vibrations {
79            compound.insert("interact_vibrations", self.interact_vibrations);
80        }
81        if self.speed_multiplier.to_bits() != Self::DEFAULT.speed_multiplier.to_bits() {
82            compound.insert("speed_multiplier", self.speed_multiplier);
83        }
84        NbtTag::Compound(compound)
85    }
86}

88impl FromNbtTag for UseEffects {
89    fn from_nbt_tag(tag: simdnbt::borrow::NbtTag) -> Option<Self> {
90        let compound = tag.compound()?;
91        let can_sprint = compound
92            .get("can_sprint")
93            .map_or(Some(Self::DEFAULT.can_sprint), |tag| tag.codec_bool())?;
94        let interact_vibrations = compound
95            .get("interact_vibrations")
96            .map_or(Some(Self::DEFAULT.interact_vibrations), |tag| {
97                tag.codec_bool()
98            })?;
99        let speed_multiplier = compound
100            .get("speed_multiplier")
101            .map_or(Some(Self::DEFAULT.speed_multiplier), |tag| tag.codec_f32())?;
102        if !speed_multiplier.is_finite()
103            || speed_multiplier.is_sign_negative()
104            || speed_multiplier > 1.0
105        {
106            return None;
107        }
108        Some(Self {
109            can_sprint,
110            interact_vibrations,
111            speed_multiplier,
112        })
113    }
114}

116impl HashComponent for UseEffects {
117    fn hash_component(&self, hasher: &mut ComponentHasher) {
118        let mut entries = Vec::new();
119        if self.can_sprint != Self::DEFAULT.can_sprint {
120            push_hash_entry(&mut entries, "can_sprint", &self.can_sprint);
121        }
122        if self.interact_vibrations != Self::DEFAULT.interact_vibrations {
123            push_hash_entry(
124                &mut entries,
125                "interact_vibrations",
126                &self.interact_vibrations,
127            );
128        }
129        if self.speed_multiplier.to_bits() != Self::DEFAULT.speed_multiplier.to_bits() {
130            push_hash_entry(&mut entries, "speed_multiplier", &self.speed_multiplier);
131        }
132        sort_map_entries(&mut entries);
133        hasher.start_map();
134        for entry in entries {
135            hasher.put_raw_bytes(&entry.key_bytes);
136            hasher.put_raw_bytes(&entry.value_bytes);
137        }
138        hasher.end_map();
139    }
140}

142fn push_hash_entry<T: HashComponent + ?Sized>(entries: &mut Vec<HashEntry>, key: &str, value: &T) {
143    let mut key_hasher = ComponentHasher::new();
144    key_hasher.put_string(key);
145    let mut value_hasher = ComponentHasher::new();
146    value.hash_component(&mut value_hasher);
147    entries.push(HashEntry::new(key_hasher, value_hasher));
148}

150const fn java_float_equals(left: f32, right: f32) -> bool {
151    (left.is_nan() && right.is_nan()) || left.to_bits() == right.to_bits()
152}

154impl ValidatePersistentComponent for UseEffects {
155    fn validate_persistent(&self) -> io::Result<()> {
156        if !self.speed_multiplier.is_finite()
157            || self.speed_multiplier.is_sign_negative()
158            || self.speed_multiplier > 1.0
159        {
160            return Err(io::Error::other("Invalid use-effect speed"));
161        }
162
163        Ok(())
164    }
165}

167impl nbt_encode::NbtEncode for UseEffects {
168    fn nbt_id(&self) -> u8 {
169        10
170    }
171    fn write_nbt_payload(&self, writer: &mut dyn NbtWrite, depth: usize) -> io::Result<()> {
172        use foton_utils::serial::nbt_encode::{check_depth, end, field};
173        check_depth(depth)?;
174        self.validate_persistent()?;
175
176        if self.can_sprint != Self::DEFAULT.can_sprint {
177            field("can_sprint", &(self.can_sprint), writer, depth)?;
178        }
179        if self.interact_vibrations != Self::DEFAULT.interact_vibrations {
180            field(
181                "interact_vibrations",
182                &(self.interact_vibrations),
183                writer,
184                depth,
185            )?;
186        }
187        if self.speed_multiplier.to_bits() != Self::DEFAULT.speed_multiplier.to_bits() {
188            field("speed_multiplier", &(self.speed_multiplier), writer, depth)?;
189        }
190        end(writer)
191    }
192}

205    fn parse(tag: NbtTag) -> Option<UseEffects> {
206        let mut bytes = Vec::new();
207        tag.write(&mut bytes);
208        let borrowed = read_tag(&mut Cursor::new(bytes.as_slice())).ok()?;
209        UseEffects::from_nbt_tag(borrowed.as_tag())
210    }
211
212    #[test]
213    fn empty_compound_uses_vanilla_defaults() {
214        assert_eq!(
215            parse(NbtTag::Compound(NbtCompound::new())),
216            Some(UseEffects::DEFAULT)
217        );
218    }
219
220    #[test]
221    fn persistent_codec_rejects_out_of_range_speed() {
222        for speed in [-0.0_f32, -0.1, 1.1, f32::NAN] {
223            let mut compound = NbtCompound::new();
224            compound.insert("speed_multiplier", speed);
225            assert_eq!(parse(NbtTag::Compound(compound)), None);
226        }
227    }

229    #[test]
230    fn equality_uses_java_record_float_semantics() {
231        assert_eq!(
232            UseEffects::new(false, true, f32::from_bits(0x7fc0_0001)),
233            UseEffects::new(false, true, f32::from_bits(0x7fc0_0002))
234        );
235        assert_ne!(
236            UseEffects::new(false, true, 0.0),
237            UseEffects::new(false, true, -0.0)

239    }
