35/// Enchantments stored on an item. Maps enchantment key to level.
36///
37/// Used by both the `minecraft:enchantments` component (on enchanted items)
38/// and the `minecraft:stored_enchantments` component (on enchanted books).
39///
40/// Vanilla moved tooltip visibility to the separate `TOOLTIP_DISPLAY` component.
41#[derive(Debug, Clone, PartialEq)]
42pub struct ItemEnchantments {
43    pub levels: FxHashMap<Identifier, u32>,
44}
45
46impl ItemEnchantments {
47    pub(crate) fn write_bounded(
48        &self,
49        budget: usize,
50        writer: &mut dyn io::Write,
51    ) -> io::Result<()> {
52        let mut writer = LimitedWriter::new(writer, budget);
53        writer.ensure_remaining(self.levels.len().saturating_mul(2).saturating_add(1))?;
54        // Known keys cap the scratch vector at the native registry's size.
55        if self.levels.len() > REGISTRY.enchantments.len() {
56            return Err(io::Error::other("Too many enchantments"));
57        }
58        let required = self.levels.len().saturating_mul(size_of::<(usize, u32)>());
59        let mut writer = ScratchWriter::new(&mut writer, required)?;
60        // IDs carry no identifier bytes. Bound lookup work by registered key
61        // lengths without hashing an arbitrarily long unknown identifier.
62        let maximum_key_bytes = REGISTRY
63            .enchantments
64            .iter()
65            .map(|(_, value)| {
66                value
67                    .key
68                    .namespace
69                    .len()
70                    .saturating_add(value.key.path.len())
71            })
72            .max()
73            .unwrap_or(0);
74        let mut entries = Vec::with_capacity(self.levels.len());
75        for (key, &level) in &self.levels {
76            if key.namespace.len().saturating_add(key.path.len()) > maximum_key_bytes {
77                return Err(io::Error::other("Unknown enchantment"));
78            }
79            #[cfg(test)]
80            lookup_work::visit();
81            let id = REGISTRY
82                .enchantments
83                .id_from_key(key)
84                .ok_or_else(|| io::Error::other("Unknown enchantment"))?;
85            if level > 255 {
86                return Err(io::Error::other("Invalid enchantment level"));
87            }
88            entries.push((id, level));
89        }
90        entries.sort_unstable_by_key(|(id, _)| *id);
91        VarInt(i32::try_from(entries.len()).map_err(io::Error::other)?).write(&mut writer)?;
92        for (id, level) in entries {
93            VarInt(i32::try_from(id).map_err(io::Error::other)?).write(&mut writer)?;
94            VarInt(level as i32).write(&mut writer)?;
95        }
96        Ok(())
97    }

110
111    pub fn set(&mut self, enchantment: Identifier, level: u32) {
112        if level == 0 {
113            self.levels.remove(&enchantment);
114        } else {
115            self.levels.insert(enchantment, level.min(255));
116        }
117    }
118
119    /// Vanilla `Mutable.upgrade`: keeps the higher of existing vs new level.
120    pub fn upgrade(&mut self, enchantment: Identifier, level: u32) {
121        if level > 0 {
122            let existing = self.get_level(&enchantment);
123            self.levels
124                .insert(enchantment, existing.max(level).min(255));
125        }
126    }

131    }

135        self.levels.len()
136    }

139        self.levels.iter()
140    }

159/// Network format: `VarInt` count, then (`VarInt` `enchantment_id`, `VarInt` level) pairs.
160impl WriteTo for ItemEnchantments {
161    fn write(&self, writer: &mut impl io::Write) -> io::Result<()> {
162        let count = i32::try_from(self.levels.len()).map_err(|_| {
163            io::Error::other(format!(
164                "Enchantment map too large: {} entries",
165                self.levels.len()
166            ))
167        })?;
168        VarInt(count).write(writer)?;
169        for (key, &level) in &self.levels {
170            let id = REGISTRY
171                .enchantments
172                .id_from_key(key)
173                .ok_or_else(|| io::Error::other("Unknown enchantment"))?;
174            let id = i32::try_from(id).map_err(|_| {
175                io::Error::other(format!("Enchantment id out of protocol range: {id}"))
176            })?;
177            if level > 255 {
178                return Err(io::Error::other("Invalid enchantment level"));
179            }
180            VarInt(id).write(writer)?;
181            VarInt(level as i32).write(writer)?;
182        }
183        Ok(())
184    }
185}

187impl ReadFrom for ItemEnchantments {
188    fn read(data: &mut std::io::Cursor<&[u8]>) -> io::Result<Self> {
189        let count = VarInt::read(data)?.0;
190        let count = usize::try_from(count)
191            .map_err(|_| io::Error::other(format!("Negative enchantment count: {count}")))?;
192        budget::check_collection_input(data, count, 2)?;
193        budget::charge_map::<Identifier, u32>(count)?;
194        let mut levels = FxHashMap::default();
195        levels.reserve(count.min(65_536));
196        for _ in 0..count {
197            let id = VarInt::read(data)?.0;
198            let id = usize::try_from(id)
199                .map_err(|_| io::Error::other(format!("Negative enchantment id: {id}")))?;
200            let level = VarInt::read(data)?.0;
201            if !(0..=255).contains(&level) {
202                return Err(io::Error::other(format!(
203                    "Enchantment level out of range: {level}"
204                )));
205            }
206            let enchantment = REGISTRY
207                .enchantments
208                .by_id(id)
209                .ok_or_else(|| io::Error::other(format!("Unknown enchantment id: {id}")))?;
210            levels.insert(enchantment.key.clone(), level as u32);
211        }
212        Ok(Self { levels })
213    }
214}

216/// NBT format: compound with enchantment identifiers as keys and int levels as values.
217impl ToNbtTag for ItemEnchantments {
218    fn to_nbt_tag(self) -> NbtTag {
219        let mut compound = NbtCompound::new();

222        }
223        NbtTag::Compound(compound)
224    }

227impl FromNbtTag for ItemEnchantments {
228    fn from_nbt_tag(tag: simdnbt::borrow::NbtTag) -> Option<Self> {
229        let compound = tag.compound()?;
230        let mut levels = FxHashMap::default();
231        for (key, value) in compound.iter() {
232            let ident = key.to_str().parse::<Identifier>().ok()?;
233            REGISTRY.enchantments.by_key(&ident)?;
234            let level = value.codec_i32()?;
235            if !(1..=255).contains(&level) {
236                return None;
237            }
238            if levels.insert(ident, level as u32).is_some() {
239                return None;
240            }
241        }
242        Some(Self { levels })
243    }
244}

246impl HashComponent for ItemEnchantments {
247    fn hash_component(&self, hasher: &mut ComponentHasher) {
248        hasher.start_map();
249        let mut entries: Vec<_> = self
250            .levels
251            .iter()
252            .map(|(key, &level)| {
253                let mut key_hasher = ComponentHasher::new();
254                key_hasher.put_string(&key.to_string());
255                let mut value_hasher = ComponentHasher::new();
256                value_hasher.put_int(level as i32);
257                HashEntry::new(key_hasher, value_hasher)
258            })
259            .collect();
260        sort_map_entries(&mut entries);
261        for entry in &entries {
262            hasher.put_raw_bytes(&entry.key_bytes);
263            hasher.put_raw_bytes(&entry.value_bytes);
264        }
265        hasher.end_map();
266    }
267}

269impl ValidatePersistentComponent for ItemEnchantments {
270    fn validate_persistent(&self) -> io::Result<()> {
271        for (key, level) in &self.levels {
272            if !(1..=255).contains(level) || REGISTRY.enchantments.by_key(key).is_none() {
273                return Err(io::Error::other("Invalid persistent enchantment"));
274            }
275        }
276        Ok(())
277    }

280impl nbt_encode::NbtEncode for ItemEnchantments {
281    fn nbt_id(&self) -> u8 {
282        10
283    }
284    fn write_nbt_payload(&self, writer: &mut dyn NbtWrite, depth: usize) -> io::Result<()> {
285        use foton_utils::serial::nbt_encode::{check_depth, end};
286        check_depth(depth)?;
287        let mut minimum = self.levels.len().saturating_mul(8).saturating_add(1);
288        writer.ensure_remaining(minimum)?;
289        if self.levels.len() > REGISTRY.enchantments.len() {
290            return Err(io::Error::other("Too many enchantments"));
291        }
292        for key in self.levels.keys() {
293            minimum = minimum
294                .saturating_add(key.namespace.len())
295                .saturating_add(key.path.len());
296            writer.ensure_remaining(minimum)?;
297        }
298        let required = self
299            .levels
300            .len()
301            .saturating_mul(size_of::<(&Identifier, &u32)>());
302        let mut writer = ScratchWriter::new(writer, required)?;
303        self.validate_persistent()?;
304        let mut entries: Vec<_> = self.levels.iter().collect();
305        entries.sort_unstable_by_key(|(key, _)| (key.namespace.as_ref(), key.path.as_ref()));
306        for (key, value) in entries {
307            nbt_encode::identifier_field(key, &(*value as i32), &mut *writer, depth)?;
308        }
309        end(&mut *writer)
310    }
311}
