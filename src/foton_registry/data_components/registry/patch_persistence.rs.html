30    /// Whether the persistent patch contains any registered set or removal.
31    pub(crate) fn has_persistent_entries(&self) -> bool {
32        use crate::{REGISTRY, RegistryExt};
33        self.entries.keys().any(|key| {
34            REGISTRY
35                .data_components
36                .by_key(key)
37                .is_some_and(super::codecs::ComponentEntry::is_persistent)
38        })
39    }
40
41    // Owned compounds expose their complete cardinality, so this map cannot
42    // grow after its one reservation. Charge one table, without growth overlap.
43    fn with_persistent_capacity(count: usize) -> Option<Self> {
44        let mut patch = Self::new();
45        if !budget::is_active() || count == 0 {
46            return Some(patch);
47        }
48        budget::charge::<u8>(Self::persistent_capacity_bytes(count)?).ok()?;
49        patch.entries.try_reserve(count).ok()?;
50        Some(patch)
51    }
52
53    fn persistent_capacity_bytes(count: usize) -> Option<usize> {
54        if count == 0 {
55            return Some(0);
56        }
57        // std's pinned hashbrown uses 7/8 load and at most 16-byte control
58        // groups. Include aligned entry storage, all controls, and a tail group.
59        let buckets = count
60            .checked_mul(8)?
61            .checked_div(7)?
62            .checked_add(1)?
63            .checked_next_power_of_two()?
64            .max(4);
65        let alignment = align_of::<(Identifier, ComponentPatchEntry)>().max(16);
66        let data = size_of::<(Identifier, ComponentPatchEntry)>().checked_mul(buckets)?;
67        let controls = data.checked_add(alignment - 1)? & !(alignment - 1);
68        controls.checked_add(buckets)?.checked_add(16)
69    }

71    pub(crate) fn from_owned_nbt(tag: &OwnedNbtTag) -> Option<Self> {
72        use crate::{REGISTRY, RegistryExt};
73        let compound = tag.compound()?;
74        let mut patch = Self::with_persistent_capacity(compound.len())?;
75        for (key, value) in compound.iter() {
76            let key = key.to_str();
77            let (key, removed) = match key.strip_prefix('!') {
78                Some(key) => (key, true),
79                None => (key.as_ref(), false),
80            };
81            let key = key.parse::<Identifier>().ok()?;
82            let entry = REGISTRY.data_components.by_key(&key)?;
83            if !entry.is_persistent() {
84                return None;
85            }
86            let value = if removed {
87                value.compound()?;
88                ComponentPatchEntry::Removed
89            } else {
90                ComponentPatchEntry::Set(entry.read_nbt_owned(value)?)
91            };
92            patch.entries.insert(key, value);
93        }
94        Some(patch)
95    }

102    /// Computes Vanilla's `HashOps` value for the persistent patch codec.
103    pub fn compute_persistent_hash(&self) -> Result<i32> {
104        use crate::{REGISTRY, RegistryExt};
105
106        let mut entries = Vec::new();
107        for (key, patch_entry) in &self.entries {
108            let Some(component) = REGISTRY.data_components.by_key(key) else {
109                continue;
110            };
111            if !component.is_persistent() {
112                continue;
113            }
114
115            let (encoded_key, value_hash) = match patch_entry {
116                ComponentPatchEntry::Set(data) => (key.to_string(), component.compute_hash(data)?),
117                ComponentPatchEntry::Removed => (format!("!{key}"), ().compute_hash()),
118            };
119            entries.push(hash_entry(encoded_key.compute_hash(), value_hash));
120        }
121        sort_map_entries(&mut entries);
122
123        let mut hasher = ComponentHasher::new();
124        hasher.start_map();
125        for entry in &entries {
126            hasher.put_raw_bytes(&entry.key_bytes);
127            hasher.put_raw_bytes(&entry.value_bytes);
128        }
129        hasher.end_map();
130        Ok(hasher.finish())
131    }

144    fn encode_nbt(&self, validate: bool) -> (OwnedNbtTag, Vec<io::Error>) {
145        use crate::{REGISTRY, RegistryExt};
146
147        let mut compound = NbtCompound::new();
148        let mut errors = Vec::new();
149
150        for (key, entry) in &self.entries {
151            let Some(component) = REGISTRY.data_components.by_key(key) else {
152                continue;
153            };
154            if !component.is_persistent() {
155                continue;
156            }
157            match entry {
158                ComponentPatchEntry::Set(data) => {
159                    let encoded = if validate {
160                        component.validate_persistent_encoding(data)
161                    } else {
162                        component.write_nbt(data)
163                    };
164                    match encoded {
165                        Ok(nbt) => {
166                            compound.insert(key.to_string(), nbt);
167                        }
168                        Err(error) => errors.push(io::Error::other(format!(
169                            "failed to encode component {key}: {error}"
170                        ))),
171                    }
172                }
173                ComponentPatchEntry::Removed => {
174                    compound.insert(format!("!{key}"), NbtCompound::new());
175                }
176            }
177        }
178
179        (OwnedNbtTag::Compound(compound), errors)
180    }

182    /// Strictly encodes this component patch through its persistent codecs.
183    ///
184    /// This is the equivalent of Vanilla encoding an untrusted stack through
185    /// `ItemStack.CODEC` before accepting it into server state.
186    pub fn try_to_nbt_tag_ref(&self) -> Result<OwnedNbtTag> {
187        let (tag, errors) = self.encode_nbt(true);
188        match errors.into_iter().next() {
189            Some(error) => Err(error),
190            None => Ok(tag),
191        }
192    }
193
194    /// Converts this component patch to NBT without consuming it.
195    ///
196    /// Save-time encoding mirrors Vanilla's `TagValueOutput`: invalid fields
197    /// are reported and omitted from the partial result rather than aborting
198    /// the owner save.
199    #[must_use]
200    pub fn to_nbt_tag_ref(&self) -> OwnedNbtTag {
201        let (tag, errors) = self.encode_nbt(false);
202        for error in errors {
203            log::warn!("Item component serialization error: {error}");
204        }
205        tag
206    }

233
234    fn write_persistent_nbt(
235        &self,
236        writer: &mut dyn NbtWrite,
237        depth: usize,
238        optional_name: Option<&str>,
239    ) -> Result<()> {
240        use crate::{REGISTRY, RegistryExt};
241        use foton_utils::serial::nbt_encode::{check_depth, end, field_with};
242        // Unknown and transient entries consume no output or sorting workspace.
243        // Filter them once; repeatedly finding the next key would be quadratic.
244        let mut entries = Vec::new();
245        let mut minimum = optional_name.map_or(1, |name| name.len().saturating_add(4));
246        let max_entries = writer.scratch_remaining()
247            / size_of::<(&Identifier, &ComponentPatchEntry, super::ComponentEntryRef)>();
248        for (key, value) in &self.entries {
249            #[cfg(test)]
250            persistent_filter_work::visit();
251            let Some(component) = REGISTRY.data_components.by_key(key) else {
252                continue;
253            };
254            if !component.is_persistent() {
255                continue;
256            }
257            minimum = minimum
258                .saturating_add(key.namespace.len())
259                .saturating_add(key.path.len())
260                .saturating_add(5);
261            writer.ensure_remaining(minimum)?;
262            if entries.len() == max_entries {
263                return Err(io::Error::other(
264                    "Persistent patch sorting workspace exceeds limit",
265                ));
266            }
267            if entries.len() == entries.capacity() {
268                let capacity = entries
269                    .capacity()
270                    .saturating_mul(2)
271                    .max(4)
272                    .min(self.entries.len())
273                    .min(max_entries);
274                entries
275                    .try_reserve_exact(capacity - entries.len())
276                    .map_err(io::Error::other)?;
277            }
278            entries.push((key, value, component));
279        }
280        if optional_name.is_some() && entries.is_empty() {
281            return Ok(());
282        }
283        check_depth(depth)?;
284        let used = entries.capacity().saturating_mul(size_of::<(
285            &Identifier,
286            &ComponentPatchEntry,
287            super::ComponentEntryRef,
288        )>());
289        let mut writer = foton_utils::serial::nbt_stream::ScratchWriter::new(writer, used)?;
290        entries.sort_unstable_by(|(a, av, _), (b, bv, _)| {
291            let removed = |v: &ComponentPatchEntry| !matches!(v, ComponentPatchEntry::Removed);
292            removed(av).cmp(&removed(bv)).then_with(|| {
293                a.namespace
294                    .bytes()
295                    .chain(*b":")
296                    .chain(a.path.bytes())
297                    .cmp(b.namespace.bytes().chain(*b":").chain(b.path.bytes()))
298            })
299        });
300        let write_entries = |writer: &mut dyn NbtWrite| {
301            for (key, value, component) in entries {
302                match value {
303                    ComponentPatchEntry::Set(value) => {
304                        nbt_encode::identifier_field(
305                            key,
306                            component.nbt_value(value)?,
307                            writer,
308                            depth,
309                        )?;
310                    }
311                    ComponentPatchEntry::Removed => {
312                        check_depth(depth + 1)?;
313                        // The key's length was charged before collecting the index.
314                        field_with(&format!("!{key}"), 10, writer, |writer| {
315                            writer.write_all(&[0])
316                        })?;
317                    }
318                }
319            }
320            end(writer)
321        };
322        if let Some(name) = optional_name {
323            field_with(name, 10, &mut *writer, write_entries)
324        } else {
325            write_entries(&mut *writer)
326        }
327    }
328}

347impl FromNbtTag for DataComponentPatch {
348    fn from_nbt_tag(tag: BorrowedNbtTag) -> Option<Self> {
349        use crate::{REGISTRY, RegistryExt};
350
351        let compound = tag.compound()?;
352        budget::charge_map::<Identifier, ComponentPatchEntry>(compound.len()).ok()?;
353        let mut patch = Self::new();
354
355        for (key, value) in compound.iter() {
356            let key_str = key.to_str();
357
358            if let Some(stripped) = key_str.strip_prefix('!') {
359                let id = stripped.parse::<Identifier>().ok()?;
360                let entry = REGISTRY.data_components.by_key(&id)?;
361                if !entry.is_persistent() || value.compound().is_none() {
362                    return None;
363                }
364                patch.entries.insert(id, ComponentPatchEntry::Removed);
365            } else {
366                let id = key_str.parse::<Identifier>().ok()?;
367                let entry = REGISTRY.data_components.by_key(&id)?;
368                if !entry.is_persistent() {
369                    return None;
370                }
371                let component_data = entry.read_nbt(value)?;
372                patch
373                    .entries
374                    .insert(id, ComponentPatchEntry::Set(component_data));
375            }
376        }
377
378        Some(patch)
379    }

401    fn recursive_lock(count: usize) -> LockCode {
402        let mut patch = DataComponentPatch::new();
403        for i in 0..count {
404            patch.remove(DataComponentType::<()>::new(Identifier::new(
405                "test",
406                format!("unregistered_{i:05}"),
407            )));
408        }
409        use crate::data_components::vanilla_components::{CREATIVE_SLOT_LOCK, DAMAGE, ITEM_NAME};
410        patch.set(DAMAGE, 7);
411        patch.remove(ITEM_NAME);
412        patch.remove(CREATIVE_SLOT_LOCK);
413        patch.set(
414            DataComponentType::<i32>::new(Identifier::new("test", "unknown_set")),
415            3,
416        );
417        let item = ItemStackTemplate::try_with_count_and_patch(&vanilla_items::STONE, 1, patch)
418            .expect("unknown removals omitted");
419        let component = REGISTRY
420            .data_components
421            .by_key(USE_REMAINDER.key())
422            .expect("use remainder");
423        let exact = DataComponentExactPredicate::new(vec![(
424            component,
425            ComponentData::new(UseRemainder::new(item)),
426        )])
427        .expect("exact");
428        LockCode::new(ItemPredicate::new(
429            None,
430            IntBounds::ANY,
431            DataComponentMatchers::new(exact, vec![]).expect("matchers"),
432        ))
433    }

435    #[test]
436    fn brewing_union_persistent_patch_filters_once_through_recursive_lock() {
437        init_vanilla_registry();
438        let value = recursive_lock(6000);
439        let compact = recursive_lock(0);
440        let compact_visits = super::persistent_filter_work::measure(|| {
441            compact
442                .write_bounded(1024, &mut sink())
443                .expect("compact lock");
444        });
445        let mut source_visits = 0;
446        let stats = allocation_counter::measure(|| {
447            source_visits = super::persistent_filter_work::measure(|| {
448                value
449                    .write_bounded(1024, &mut sink())
450                    .expect("unknown entries are omitted");
451            });
452        });
453        eprintln!("union 6000 persistent entries: {source_visits} visits, {stats:?}");
454        assert_eq!(
455            source_visits,
456            compact_visits + 6000,
457            "persistent filtering did not visit each added source entry exactly once"
458        );
459        assert!(
460            stats.bytes_max < 64 * 1024,
461            "workspace must omit unknowns: {stats:?}"
462        );
463        let mut actual = Vec::new();
464        value.write_bounded(1024, &mut actual).expect("lock");
465        assert!(LockCode::read(&mut Cursor::new(actual.as_slice())).is_ok());
466        let mut compact_bytes = Vec::new();
467        compact
468            .write_bounded(1024, &mut compact_bytes)
469            .expect("compact lock");
470        assert_eq!(
471            actual, compact_bytes,
472            "unknown/transient entries must not change persistent bytes"
473        );
474    }
