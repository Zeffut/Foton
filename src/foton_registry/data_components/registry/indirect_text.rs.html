22fn holder<T: RegistryHolderEntry, W: NbtWrite>(
23    value: &RegistryHolder<T>,
24    writer: &mut W,
25    direct: impl FnOnce(&T::Value, &mut W) -> Result<()>,
26) -> Result<()> {
27    let RegistryHolder::Direct(value) = value else {
28        return value.write(writer);
29    };
30    VarInt(0).write(writer)?;
31    direct(value, writer)
32}

34pub(super) fn pattern(
35    value: &RegistryHolder<TrimPattern>,
36    writer: &mut impl NbtWrite,
37) -> Result<()> {
38    holder(value, writer, |value, writer| {
39        value.asset_id().write(writer)?;
40        text_stream::write(value.description(), writer)?;
41        value.decal().write(writer)
42    })
43}

45fn optional_text(value: Option<&TextComponent>, writer: &mut impl NbtWrite) -> Result<()> {
46    value.is_some().write(writer)?;
47    if let Some(value) = value {
48        text_stream::write(value, writer)?;
49    }
50    Ok(())
51}

53fn attributes(value: &ItemAttributeModifiers, writer: &mut impl NbtWrite) -> Result<()> {
54    writer.ensure_remaining(value.modifiers.len().saturating_mul(13).saturating_add(1))?;
55    VarInt(i32::try_from(value.modifiers.len()).map_err(Error::other)?).write(writer)?;
56    for entry in &value.modifiers {
57        let id = entry
58            .attribute
59            .try_id()
60            .ok_or_else(|| Error::other("Unknown attribute"))?;
61        VarInt(i32::try_from(id).map_err(Error::other)?).write(writer)?;
62        entry.id.write(writer)?;
63        entry.amount.write(writer)?;
64        entry.operation.write(writer)?;
65        VarInt(entry.slot.id()).write(writer)?;
66        VarInt(entry.display.id()).write(writer)?;
67        if let ItemAttributeModifierDisplay::OverrideText(text) = &entry.display {
68            text_stream::write(text, writer)?;
69        }
70    }
71    Ok(())
72}

74pub(super) fn write(data: &ComponentData, writer: &mut LimitedWriter<'_>) -> Option<Result<()>> {
75    if let Some(value) = data.downcast_ref::<InstrumentComponent>() {
76        return Some(holder(value.instrument(), writer, |value, writer| {
77            value.sound_event().write(writer)?;
78            value.use_duration().write(writer)?;
79            value.range().write(writer)?;
80            text_stream::write(value.description(), writer)
81        }));
82    }
83    if let Some(value) = data.downcast_ref::<ItemAttributeModifiers>() {
84        return Some(attributes(value, writer));
85    }
86    if let Some(value) = data.downcast_ref::<JukeboxPlayable>() {
87        return Some(holder(value.song(), writer, |value, writer| {
88            value.sound_event.write(writer)?;
89            text_stream::write(&value.description, writer)?;
90            value.length_in_seconds.write(writer)?;
91            VarInt(value.comparator_output).write(writer)
92        }));
93    }
94    if let Some(value) = data.downcast_ref::<PaintingVariantComponent>() {
95        return Some(holder(value.variant(), writer, |value, writer| {
96            VarInt(value.width).write(writer)?;
97            VarInt(value.height).write(writer)?;
98            value.asset_id.write(writer)?;
99            optional_text(value.title.as_ref(), writer)?;
100            optional_text(value.author.as_ref(), writer)
101        }));
102    }
103    None
104}
