1//! Placing rolled loot into a container.
2//!
3//! Vanilla parity: `LootTable.fill` and the two helpers it is built on,
4//! `shuffleAndSplitItems` and `getAvailableSlots`. Between them they are what
5//! makes a dungeon chest look like a dungeon chest: the stacks are broken up
6//! and scattered over the free slots instead of being packed into the first
7//! few.
8
9use std::mem;
10
11use super::{ItemStack, LootContext, LootTable, RngExt, uniform_int};
12
13/// The container surface [`LootTable::fill`] needs.
14///
15/// Vanilla fills a `net.minecraft.world.Container`. Foton's `Container` trait
16/// lives in `foton-core`, which depends on this crate, so `fill` asks only for
17/// the three operations vanilla performs through it.
18pub trait LootFillContainer {
19    /// Vanilla parity: `Container.getContainerSize`.
20    fn get_container_size(&self) -> usize;
21
22    /// Vanilla parity: `Container.getItem`.
23    fn get_item(&self, slot: usize) -> &ItemStack;
24
25    /// Vanilla parity: `Container.setItem`.
26    fn set_item(&mut self, slot: usize, stack: ItemStack);
27}
28
29impl LootTable {
30    /// Rolls this table and scatters the result over the container's free slots.
31    ///
32    /// Vanilla parity: `LootTable.fill`. The roll and the scatter share one
33    /// random source, in that order, so a container filled from a fixed loot
34    /// table seed always ends up with the same items in the same slots.
35    ///
36    /// Slots that already hold something are left alone, matching vanilla's
37    /// `getAvailableSlots`.
38    pub fn fill<C, R>(&self, container: &mut C, ctx: &mut LootContext<'_, R>)
39    where
40        C: LootFillContainer + ?Sized,
41        R: rand::Rng,
42    {
43        let mut item_stacks = self.get_random_items(ctx);
44        let mut available_slots = available_slots(container, ctx.rng);
45        shuffle_and_split_items(&mut item_stacks, available_slots.len(), ctx.rng);
46
47        for item_stack in item_stacks {
48            let Some(slot) = available_slots.pop() else {
49                log::warn!("Tried to over-fill a container");
50                return;
51            };
52            container.set_item(slot, item_stack);
53        }
54    }
55}
56
57/// Breaks large stacks apart until the loot spreads over the free slots.
58///
59/// Vanilla parity: `LootTable.shuffleAndSplitItems`. Only stacks of more than
60/// one are candidates, and a split half is offered up for splitting again on a
61/// coin flip, which is why a chest ends up with uneven little piles rather than
62/// with every stack cut exactly in two.
63fn shuffle_and_split_items<R: rand::Rng>(
64    result: &mut Vec<ItemStack>,
65    available_slots: usize,
66    rng: &mut R,
67) {
68    let mut splittable_items: Vec<ItemStack> = Vec::new();
69    result.retain_mut(|item_stack| {
70        if item_stack.is_empty() {
71            return false;
72        }
73        if item_stack.count() > 1 {
74            splittable_items.push(mem::take(item_stack));
75            return false;
76        }
77        true
78    });
79
80    while !splittable_items.is_empty() && available_slots > result.len() + splittable_items.len() {
81        let index = uniform_int(rng, 0, splittable_items.len() as i32 - 1) as usize;
82        let mut item_stack = splittable_items.remove(index);
83        let remove = uniform_int(rng, 1, item_stack.count() / 2);
84        let copy = item_stack.split(remove);
85
86        if item_stack.count() > 1 && rng.random::<bool>() {
87            splittable_items.push(item_stack);
88        } else {
89            result.push(item_stack);
90        }
91
92        if copy.count() > 1 && rng.random::<bool>() {
93            splittable_items.push(copy);
94        } else {
95            result.push(copy);
96        }
97    }
98
99    result.append(&mut splittable_items);
100    shuffle(result, rng);
101}

103/// Returns the empty slots of `container`, in a random order.
104///
105/// Vanilla parity: `LootTable.getAvailableSlots`. `fill` takes them off the
106/// end, so the shuffle here is what decides where each stack lands.
107fn available_slots<C, R>(container: &C, rng: &mut R) -> Vec<usize>
108where
109    C: LootFillContainer + ?Sized,
110    R: rand::Rng,
111{
112    let mut slots: Vec<usize> = (0..container.get_container_size())
113        .filter(|&slot| container.get_item(slot).is_empty())
114        .collect();
115    shuffle(&mut slots, rng);
116    slots
117}
118
119/// Vanilla parity: `Util.shuffle`, a Fisher-Yates pass walking from the end
120/// down to the second element.
121fn shuffle<T, R: rand::Rng>(list: &mut [T], rng: &mut R) {
122    for i in (2..=list.len()).rev() {
123        let swap_to = rng.random_range(0..i);
124        list.swap(i - 1, swap_to);
125    }
126}

129mod tests {
130    use rand::SeedableRng as _;

138    /// A plain array of slots, which is all `fill` ever asks a container for.
139    struct TestContainer {
140        items: Vec<ItemStack>,
141    }
142
143    impl TestContainer {
144        fn new(size: usize) -> Self {
145            Self {
146                items: vec![ItemStack::empty(); size],
147            }
148        }
149
150        fn occupied_slots(&self) -> Vec<usize> {
151            (0..self.items.len())
152                .filter(|&slot| !self.items[slot].is_empty())
153                .collect()
154        }
155
156        fn contents(&self) -> Vec<(usize, String, i32)> {
157            (0..self.items.len())
158                .filter(|&slot| !self.items[slot].is_empty())
159                .map(|slot| {
160                    (
161                        slot,
162                        self.items[slot].item.key.to_string(),
163                        self.items[slot].count(),
164                    )
165                })
166                .collect()
167        }
168    }
169
170    impl LootFillContainer for TestContainer {
171        fn get_container_size(&self) -> usize {
172            self.items.len()
173        }
174
175        fn get_item(&self, slot: usize) -> &ItemStack {
176            &self.items[slot]
177        }
178
179        fn set_item(&mut self, slot: usize, stack: ItemStack) {
180            self.items[slot] = stack;
181        }
182    }
183
184    fn fill_dungeon_chest(seed: u64) -> TestContainer {
185        init_vanilla_registry();
186        let mut container = TestContainer::new(27);
187        let mut rng = StdRng::seed_from_u64(seed);
188        let mut ctx = LootContext::new(&mut rng);
189        vanilla_loot_tables::CHESTS_SIMPLE_DUNGEON.fill(&mut container, &mut ctx);
190        container
191    }
192
193    #[test]
194    fn the_same_seed_fills_a_chest_the_same_way_twice() {
195        assert_eq!(
196            fill_dungeon_chest(1234).contents(),
197            fill_dungeon_chest(1234).contents()
198        );
199    }

201    #[test]
202    fn a_different_seed_fills_a_chest_differently() {
203        assert_ne!(
204            fill_dungeon_chest(1234).contents(),
205            fill_dungeon_chest(9876).contents()
206        );
207    }

209    /// The point of the scatter: a dungeon chest must not be a solid block of
210    /// items in the first N slots.
211    #[test]
212    fn loot_lands_on_scattered_slots_rather_than_the_first_ones() {
213        let occupied = fill_dungeon_chest(1234).occupied_slots();
214
215        assert!(!occupied.is_empty(), "simple_dungeon should roll something");
216        let contiguous_from_zero: Vec<usize> = (0..occupied.len()).collect();
217        assert_ne!(
218            occupied, contiguous_from_zero,
219            "loot was packed into the first slots instead of being scattered"
220        );
221    }

223    /// Vanilla never overwrites what is already in the container.
224    #[test]
225    fn filling_leaves_occupied_slots_alone() {
226        init_vanilla_registry();
227        let mut container = TestContainer::new(27);
228        let kept = ItemStack::with_count(&vanilla_items::DIAMOND, 7);
229        for slot in 0..26 {
230            container.set_item(slot, kept.clone());
231        }
232
233        let mut rng = StdRng::seed_from_u64(4242);
234        let mut ctx = LootContext::new(&mut rng);
235        vanilla_loot_tables::CHESTS_SIMPLE_DUNGEON.fill(&mut container, &mut ctx);
236
237        for slot in 0..26 {
238            assert!(
239                container.get_item(slot).is(&vanilla_items::DIAMOND),
240                "slot {slot} was overwritten"
241            );
242        }
243    }

245    /// Splitting stops once there are as many piles as free slots, and every
246    /// item that went in comes back out.
247    #[test]
248    fn splitting_preserves_every_item_and_respects_the_free_slot_count() {
249        init_vanilla_registry();
250        let mut rng = StdRng::seed_from_u64(7);
251        let mut result = vec![
252            ItemStack::with_count(&vanilla_items::STONE, 32),
253            ItemStack::with_count(&vanilla_items::DIRT, 9),
254            ItemStack::empty(),
255        ];
256
257        shuffle_and_split_items(&mut result, 8, &mut rng);
258
259        assert!(
260            result.len() <= 8,
261            "split into more piles than there are slots"
262        );
263        assert!(result.len() > 2, "nothing was split apart");
264        let stone: i32 = result
265            .iter()
266            .filter(|stack| stack.is(&vanilla_items::STONE))
267            .map(ItemStack::count)
268            .sum();
269        let dirt: i32 = result
270            .iter()
271            .filter(|stack| stack.is(&vanilla_items::DIRT))
272            .map(ItemStack::count)
273            .sum();
274        assert_eq!((stone, dirt), (32, 9));
275        assert!(result.iter().all(|stack| !stack.is_empty()));
276    }
