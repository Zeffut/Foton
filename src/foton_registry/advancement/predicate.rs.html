1//! The predicates an advancement criterion is allowed to ask about.
2//!
3//! Vanilla parity: `net.minecraft.advancements.predicates` plus the
4//! `LootItemCondition`s a `ContextAwarePredicate` is made of. Vanilla reuses the
5//! loot condition machinery here, and so does Foton -- but the loot version in
6//! [`crate::loot_table::conditions`] evaluates itself against a `LootContext`
7//! that only knows what a loot roll knows. An advancement asks about vehicles,
8//! passengers, what a player is looking at and how far away a victim died, none
9//! of which a loot context carries, so these are a separate, plain-data model.
10//! Evaluation lives in `foton-core`, where the world is reachable.
11//!
12//! Only the shapes vanilla's own advancement data actually uses are modeled.
13//! The build script fails on anything else rather than emitting a predicate
14//! that asks nothing -- an advancement handed out for the wrong kill is worse
15//! than one that never fires.
16

21/// An inclusive numeric range, either bound optional.
22///
23/// Vanilla parity: `MinMaxBounds.Doubles`.
24#[derive(Debug, Clone, Copy, PartialEq, Default)]
25pub struct DoubleBounds {
26    /// The lowest accepted value, `None` for unbounded.
27    pub min: Option<f64>,
28    /// The highest accepted value, `None` for unbounded.
29    pub max: Option<f64>,
30}
31
32impl DoubleBounds {
33    /// Bounds that accept every value.
34    pub const ANY: Self = Self {
35        min: None,
36        max: None,
37    };
38
39    /// Whether neither bound was given.
40    #[must_use]
41    pub const fn is_any(&self) -> bool {
42        self.min.is_none() && self.max.is_none()
43    }
44
45    /// Whether `value` falls inside the range.
46    #[must_use]
47    pub fn matches(&self, value: f64) -> bool {
48        if let Some(min) = self.min
49            && value < min
50        {
51            return false;
52        }
53        if let Some(max) = self.max
54            && value > max
55        {
56            return false;
57        }
58        true
59    }
60
61    /// Whether the square of `value` falls inside the squared range.
62    ///
63    /// Vanilla parity: `MinMaxBounds.Doubles.matchesSqr`, which is what
64    /// `DistancePredicate` uses so it never takes a square root.
65    #[must_use]
66    pub fn matches_sqr(&self, value_sqr: f64) -> bool {
67        if let Some(min) = self.min
68            && value_sqr < min * min
69        {
70            return false;
71        }
72        if let Some(max) = self.max
73            && value_sqr > max * max
74        {
75            return false;
76        }
77        true
78    }
79}

122/// A registry set written either as a tag or as an explicit list.
123///
124/// Vanilla parity: the `HolderSet` codec every `items` / `blocks` / `biomes`
125/// field uses, which accepts `"#tag"`, `"single:id"` or `["a", "b"]`.
126#[derive(Debug, Clone, PartialEq, Eq)]
127pub enum RegistrySet {
128    /// Every entry carrying this tag.
129    Tag(Identifier),
130    /// Exactly these entries.
131    Entries(&'static [Identifier]),
132}
133
134impl RegistrySet {
135    /// Whether `key` is one of the explicit entries, ignoring tags.
136    ///
137    /// Tag membership needs a registry and is resolved by the caller.
138    #[must_use]
139    pub fn contains_key(&self, key: &Identifier) -> bool {
140        match self {
141            Self::Tag(_) => false,
142            Self::Entries(entries) => entries.contains(key),
143        }
144    }
145}

197/// The item data components a vanilla advancement checks.
198///
199/// Vanilla parity: the `components` map of `ItemPredicate`, whose codec is
200/// open-ended. Foton models exactly the three keys vanilla data uses; the
201/// build script fails on a fourth rather than ignoring it.
202#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
203pub struct ItemComponentsPredicate {
204    /// `minecraft:damage` must equal this.
205    pub damage: Option<i32>,
206    /// `minecraft:banner_patterns` must be exactly these layers, in order.
207    pub banner_patterns: Option<&'static [BannerPatternLayer]>,
208    /// `minecraft:item_name` must be this translatable component's key.
209    pub item_name_translate: Option<&'static str>,
210}
211
212impl ItemComponentsPredicate {
213    /// A predicate that checks no component.
214    pub const ANY: Self = Self {
215        damage: None,
216        banner_patterns: None,
217        item_name_translate: None,
218    };
219
220    /// Whether no component check was given.
221    #[must_use]
222    pub const fn is_any(&self) -> bool {
223        self.damage.is_none()
224            && self.banner_patterns.is_none()
225            && self.item_name_translate.is_none()
226    }
227}

418/// One term of a criterion's condition list.
419///
420/// Vanilla parity: the `LootItemCondition` subtypes an advancement's
421/// `ContextAwarePredicate` is built from. Vanilla's `EntityPredicate.wrap`
422/// always targets `LootContextParams.THIS_ENTITY`, and every `entity` field in
423/// vanilla advancement data reads `"this"`, so the target is not modeled --
424/// the subject is whatever the trigger handed to this predicate.
425#[derive(Debug, Clone, PartialEq)]
426pub enum ConditionTerm {
427    /// `minecraft:entity_properties` on the subject.
428    EntityProperties(&'static EntityPredicate),
429    /// `minecraft:location_check` at the subject's position plus an offset.
430    LocationCheck {
431        /// Block offset applied before the check.
432        offset_x: i32,
433        /// Block offset applied before the check.
434        offset_y: i32,
435        /// Block offset applied before the check.
436        offset_z: i32,
437        /// The position check itself.
438        predicate: &'static LocationPredicate,
439    },
440    /// `minecraft:match_tool` on the stack the trigger supplied.
441    MatchTool(&'static ItemPredicate),
442    /// `minecraft:block_state_property` on the block the trigger supplied.
443    BlockStateProperty {
444        /// The block the state must belong to.
445        block: Identifier,
446        /// Properties that must hold, all of them.
447        properties: &'static [StatePropertyMatch],
448    },
449    /// `minecraft:any_of`: at least one term must pass.
450    AnyOf(&'static [ConditionTerm]),
451    /// `minecraft:all_of`: every term must pass.
452    AllOf(&'static [ConditionTerm]),
453    /// `minecraft:inverted`: the term must fail.
454    Inverted(&'static ConditionTerm),
455}

457/// A criterion's condition list: every term must pass.
458///
459/// Vanilla parity: `ContextAwarePredicate`, which is a `List<LootItemCondition>`
460/// evaluated with `allMatch`. An empty list passes, which is what makes an
461/// absent `player` field accept everyone.
462pub type ContextAwarePredicate = &'static [ConditionTerm];
