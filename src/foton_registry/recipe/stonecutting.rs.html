1//! Stonecutting recipes.
2//!
3//! Vanilla parity: `StonecutterRecipe`. One ingredient in, one stack out, and
4//! no shape or timing at all -- the interesting part is that a single input
5//! usually matches many recipes and the player picks which one, which is why
6//! the registry hands back every match rather than the first.
7
8use foton_utils::Identifier;
9
10use crate::item_stack::ItemStack;
11
12use super::{Ingredient, RecipeResult};
13
14/// A stonecutter recipe.
15#[derive(Debug)]
16pub struct StonecuttingRecipe {
17    pub id: Identifier,
18    pub ingredient: Ingredient,
19    pub result: RecipeResult,
20}
21
22impl StonecuttingRecipe {
23    /// Returns whether this recipe accepts `input`.
24    #[must_use]
25    pub fn matches(&self, input: &ItemStack) -> bool {
26        self.ingredient.test(input)
27    }
28}
29
30#[cfg(test)]
31mod tests {
32    use foton_utils::Identifier;
33
34    use crate::recipe::{Ingredient, RecipeResult};
35    use crate::{init_vanilla_registry, item_stack::ItemStack, vanilla_items};
36
37    use super::*;
38
39    #[test]
40    fn a_stonecutting_recipe_matches_only_its_ingredient() {
41        init_vanilla_registry();
42        let recipe = StonecuttingRecipe {
43            id: Identifier::vanilla_static("test"),
44            ingredient: Ingredient::Item(&vanilla_items::ANDESITE),
45            result: RecipeResult {
46                item: &vanilla_items::ANDESITE_SLAB,
47                count: 2,
48            },
49        };
50
51        assert!(recipe.matches(&ItemStack::new(&vanilla_items::ANDESITE)));
52        assert!(!recipe.matches(&ItemStack::new(&vanilla_items::DIORITE)));
53    }
54}
