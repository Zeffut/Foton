1//! The recipe registry as the recipe book sees it.
2//!
3//! Vanilla parity: `RecipeManager.allDisplays` and `recipeToDisplay`, built by
4//! `unpackRecipeInfo`. Every recipe display gets the next id, and every
5//! non-empty group string the next group number, in the order recipes are
6//! indexed. Both numbers are the server's own handles -- the client only ever
7//! sends them back -- so any order works as long as it never changes while a
8//! client holds them; this index only ever appends.
9
10use foton_utils::Identifier;
11use rustc_hash::FxHashMap;
12
13use super::display::{DisplaySource, RecipeDisplayEntry};
14
15/// One recipe's standing in the recipe book.
16#[derive(Debug, Clone)]
17pub struct BookRecipe {
18    /// Vanilla parity: `Recipe.showNotification`, whether unlocking it pops
19    /// a toast.
20    pub show_notification: bool,
21    /// The displays the book draws for it. Every Foton recipe has exactly one,
22    /// except one whose result cannot be shown at all.
23    pub displays: Vec<RecipeDisplayEntry>,
24}
25
26/// Recipe displays by id and by recipe.
27#[derive(Debug, Default)]
28pub struct RecipeDisplayIndex {
29    /// The recipe each display id belongs to; the position is the id.
30    display_owners: Vec<Identifier>,
31    recipes: FxHashMap<Identifier, BookRecipe>,
32    groups: FxHashMap<String, i32>,
33}
34
35impl RecipeDisplayIndex {
36    /// Adds a recipe, numbering its display and its group. A recipe already
37    /// present keeps the ids it was given.
38    pub(crate) fn insert(&mut self, key: &Identifier, source: DisplaySource<'_>) {
39        if self.recipes.contains_key(key) {
40            return;
41        }
42        let group = if source.group.is_empty() {
43            None
44        } else {
45            let next = i32::try_from(self.groups.len()).ok();
46            match self.groups.get(source.group) {
47                Some(&group) => Some(group),
48                None => next.inspect(|&group| {
49                    self.groups.insert(source.group.to_owned(), group);
50                }),
51            }
52        };
53        let mut displays = Vec::new();
54        if let Some(display) = source.display
55            && let Ok(id) = i32::try_from(self.display_owners.len())
56        {
57            self.display_owners.push(key.clone());
58            displays.push(RecipeDisplayEntry {
59                id,
60                display,
61                group,
62                category: source.category,
63                crafting_requirements: Some(source.crafting_requirements),
64            });
65        }
66        self.recipes.insert(
67            key.clone(),
68            BookRecipe {
69                show_notification: source.show_notification,
70                displays,
71            },
72        );
73    }
74
75    /// The recipe book's view of one recipe.
76    #[must_use]
77    pub fn recipe(&self, key: &Identifier) -> Option<&BookRecipe> {
78        self.recipes.get(key)
79    }
80
81    /// Every recipe key the index holds.
82    pub(crate) fn keys(&self) -> impl Iterator<Item = &Identifier> {
83        self.recipes.keys()
84    }
85
86    /// The recipe a display id belongs to.
87    ///
88    /// Vanilla parity: `RecipeManager.getRecipeFromDisplay`, which answers
89    /// nothing for an id outside the list rather than failing.
90    #[must_use]
91    pub fn recipe_for_display(&self, id: i32) -> Option<&Identifier> {
92        usize::try_from(id)
93            .ok()
94            .and_then(|index| self.display_owners.get(index))
95    }
96}
