20pub struct JukeboxSongValue {
21    pub sound_event: SoundEventHolder,
22    pub description: TextComponent,
23    pub length_in_seconds: f32,
24    pub comparator_output: i32,
25}

27impl PartialEq for JukeboxSongValue {
28    fn eq(&self, other: &Self) -> bool {
29        self.sound_event == other.sound_event
30            && self.description == other.description
31            && ((self.length_in_seconds.is_nan() && other.length_in_seconds.is_nan())
32                || self.length_in_seconds.to_bits() == other.length_in_seconds.to_bits())
33            && self.comparator_output == other.comparator_output
34    }
35}

37impl WriteTo for JukeboxSongValue {

57impl ToNbtTag for JukeboxSongValue {
58    fn to_nbt_tag(self) -> NbtTag {
59        let mut compound = NbtCompound::new();
60        compound.insert("sound_event", self.sound_event.to_nbt_tag());
61        compound.insert("description", self.description.to_codec_nbt());
62        compound.insert("length_in_seconds", self.length_in_seconds);
63        compound.insert("comparator_output", self.comparator_output);
64        NbtTag::Compound(compound)
65    }
66}

68impl FromNbtTag for JukeboxSongValue {
69    fn from_nbt_tag(tag: simdnbt::borrow::NbtTag) -> Option<Self> {
70        let compound = tag.compound()?;
71        let length_in_seconds = compound.get("length_in_seconds")?.codec_f32()?;
72        let comparator_output = compound.get("comparator_output")?.codec_i32()?;
73        if !length_in_seconds.is_finite()
74            || length_in_seconds <= 0.0
75            || !(0..=15).contains(&comparator_output)
76        {
77            return None;
78        }
79        Some(Self {
80            sound_event: SoundEventHolder::from_nbt_tag(compound.get("sound_event")?)?,
81            description: decode_text_nbt(&compound.get("description")?.to_owned())?,
82            length_in_seconds,
83            comparator_output,
84        })
85    }
86}

91    }
