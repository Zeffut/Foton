13/// Represents a full cat variant definition from a data pack JSON file.
14#[derive(Debug)]
15pub struct CatVariant {
16    pub key: Identifier,
17    pub asset_id: Identifier,
18    pub baby_asset_id: Identifier,
19    pub spawn_conditions: &'static [SpawnConditionEntry],
20}

22/// A single entry in the list of spawn conditions.
23#[derive(Debug)]
24pub struct SpawnConditionEntry {
25    pub priority: i32,
26    pub condition: Option<SpawnCondition>,
27}
28
29/// Defines various spawn conditions for cat variants.
30#[derive(Debug)]
31pub enum SpawnCondition {
32    Structure { structures: &'static str },
33    MoonBrightness { min: Option<f32>, max: Option<f32> },
34    Biome { biomes: &'static str },
35}
36
37impl SpawnConditionEntry {
38    /// Returns whether this entry admits a cat spawning in `biome`.
39    ///
40    /// Vanilla parity: `SpawnCondition.test` against a `SpawnContext`. Foton can
41    /// answer the biome condition and the unconditioned entry.
42    ///
43    /// **Gap**: a `minecraft:structure` condition never matches, because Foton
44    /// has no structure-at-position lookup, and a `minecraft:moon_brightness`
45    /// condition never matches, because 26.2 moved day time into the world-clock
46    /// registry and Foton exposes no moon phase from it. Today that only costs
47    /// the `all_black` variant, which vanilla awards inside a swamp hut or under
48    /// a moon at least 0.9 bright.
49    #[must_use]
50    pub fn matches_biome(&self, biome: BiomeRef) -> bool {
51        match &self.condition {
52            None => true,
53            Some(SpawnCondition::Biome { biomes }) => biome_target_matches(biomes, biome),
54            Some(SpawnCondition::Structure { .. } | SpawnCondition::MoonBrightness { .. }) => false,
55        }
56    }
57}
58
59/// Returns whether a `#tag` or direct biome reference names `biome`.
60fn biome_target_matches(target: &str, biome: BiomeRef) -> bool {
61    if let Some(tag) = target.strip_prefix('#') {
62        return Identifier::from_str(tag).is_ok_and(|tag| REGISTRY.biomes.is_in_tag(biome, &tag));
63    }
64
65    Identifier::from_str(target).is_ok_and(|key| biome.key == key)
66}

68impl ToNbtTag for &CatVariant {
69    fn to_nbt_tag(self) -> NbtTag {
70        use simdnbt::owned::{NbtCompound, NbtList, NbtTag};
71        let mut compound = NbtCompound::new();
72        compound.insert("asset_id", self.asset_id.clone());
73        compound.insert("baby_asset_id", self.baby_asset_id.clone());
74        let conditions: Vec<NbtCompound> = self
75            .spawn_conditions
76            .iter()
77            .map(|entry| {
78                let mut e = NbtCompound::new();
79                e.insert("priority", entry.priority);
80                if let Some(cond) = &entry.condition {
81                    let mut c = NbtCompound::new();
82                    match cond {
83                        SpawnCondition::Structure { structures } => {
84                            c.insert("type", "minecraft:in_structure");
85                            c.insert("structures", *structures);
86                        }
87                        SpawnCondition::MoonBrightness { min, max } => {
88                            c.insert("type", "minecraft:moon_brightness");
89                            if let Some(min) = min {
90                                c.insert("min", *min);
91                            }
92                            if let Some(max) = max {
93                                c.insert("max", *max);
94                            }
95                        }
96                        SpawnCondition::Biome { biomes } => {
97                            c.insert("type", "minecraft:biome");
98                            c.insert("biomes", *biomes);
99                        }
100                    }
101                    e.insert("condition", NbtTag::Compound(c));
102                }
103                e
104            })
105            .collect();
106        compound.insert(
107            "spawn_conditions",
108            NbtTag::List(NbtList::Compound(conditions)),
109        );
110        NbtTag::Compound(compound)
111    }
112}

116pub struct CatVariantRegistry {
117    cat_variants_by_id: Vec<CatVariantRef>,
118    cat_variants_by_key: FxHashMap<Identifier, usize>,
119    allows_registering: bool,
120}
121
122impl CatVariantRegistry {
123    #[must_use]
124    pub fn new() -> Self {
125        Self {
126            cat_variants_by_id: Vec::new(),
127            cat_variants_by_key: FxHashMap::default(),
128            allows_registering: true,
129        }
130    }
131
132    /// Picks the variant a cat spawning in `biome` should wear.
133    ///
134    /// Vanilla parity: `VariantUtils.selectVariantToSpawn` over
135    /// `Registries.CAT_VARIANT`, using `PriorityProvider.pick` semantics --
136    /// highest matching priority wins, ties are broken uniformly.
137    #[must_use]
138    pub fn select_spawn_variant(
139        &self,
140        biome: BiomeRef,
141        random: &mut impl Random,
142    ) -> Option<CatVariantRef> {
143        let mut selected = Vec::new();
144        let mut highest_priority = i32::MIN;
145
146        for (_, variant) in self.iter() {
147            for entry in variant.spawn_conditions {
148                if !entry.matches_biome(biome) {
149                    continue;
150                }
151
152                match entry.priority.cmp(&highest_priority) {
153                    Ordering::Greater => {
154                        selected.clear();
155                        selected.push(variant);
156                        highest_priority = entry.priority;
157                    }
158                    Ordering::Equal => selected.push(variant),
159                    Ordering::Less => {}
160                }
161            }
162        }
163
164        let bound = i32::try_from(selected.len()).ok()?;
165        if bound == 0 {
166            return None;
167        }
168
169        selected
170            .get(random.next_i32_bounded(bound) as usize)
171            .copied()
172    }
173}
