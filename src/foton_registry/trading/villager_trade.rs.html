2//!
3//! Vanilla parity: `net.minecraft.world.item.trading.VillagerTrade` and its
4//! `TradeCost`. Since 26.2 a trade is data, not code: everything a villager is

22/// The `components` an [`ItemCost`] demands of the stack that pays it.
23///
24/// Vanilla parity: the `DataComponentExactPredicate` of `TradeCost`. Foton has
25/// no JSON-to-component-value path, so the build script lowers the one shape the
26/// vanilla data contains and fails on anything else. Dropping the requirement
27/// instead would let a wandering trader take any potion for its emerald.
28#[derive(Debug, Clone)]
29pub enum TradeCostComponents {
30    /// `minecraft:potion_contents` naming exactly this potion.
31    Potion(Identifier),
32}
33
34impl TradeCostComponents {
35    /// Resolves this into the predicate an [`ItemCost`] carries.
36    ///
37    /// Returns [`DataComponentExactPredicate::EMPTY`] when the value does not
38    /// round-trip -- the same refusal `DataComponentExactPredicate::new` makes
39    /// -- rather than producing a cost nothing can satisfy.
40    fn resolve(&self) -> DataComponentExactPredicate {
41        match self {
42            Self::Potion(id) => {
43                let (Some(entry), Some(potion)) = (
44                    REGISTRY
45                        .data_components
46                        .by_key(&crate::data_components::vanilla_components::POTION_CONTENTS.key),
47                    REGISTRY.potions.by_key(id),
48                ) else {
49                    return DataComponentExactPredicate::EMPTY;
50                };
51                let contents = crate::data_components::components::PotionContents::empty()
52                    .with_potion(crate::registry::reference::RegistryReference::new(potion));
53                DataComponentExactPredicate::new(vec![(
54                    entry,
55                    crate::data_components::ComponentData::new(contents),
56                )])
57                .unwrap_or(DataComponentExactPredicate::EMPTY)
58            }
59        }
60    }
61}

63/// One side of a trade's price, before demand and reputation touch it.
64///
65/// Vanilla parity: `net.minecraft.world.item.trading.TradeCost`.
66#[derive(Debug)]
67pub struct TradeCost {
68    /// Vanilla parity: the `Holder<Item>` of `TradeCost`.
69    pub item: LazyItemRef,
70    pub count: NumberProvider,
71    pub components: Option<TradeCostComponents>,
72}
73
74impl TradeCost {
75    /// Resolves this cost against a roll, adding `additional_cost` to the count.
76    ///
77    /// Vanilla parity: `TradeCost.toItemCost`. The clamp floor of zero is what
78    /// lets `VillagerTrade.getOffer` reject a trade whose price came out empty;
79    /// the ceiling is the item's *default* stack size, not the built stack's.
80    pub fn to_item_cost<R: Rng>(
81        &self,
82        ctx: &mut LootContext<'_, R>,
83        additional_cost: i32,
84    ) -> ItemCost {
85        let max_stack_size = self.item.components.get(MAX_STACK_SIZE).unwrap_or(1);
86        let count = (self.count.get_int(ctx.rng) + additional_cost).clamp(0, max_stack_size);
87        match &self.components {
88            None => ItemCost::new(self.item, count),
89            Some(components) => ItemCost::with_components(self.item, count, components.resolve()),
90        }
91    }
92}

94/// One trade a profession can offer at one level.
95///
96/// Vanilla parity: `net.minecraft.world.item.trading.VillagerTrade`.
97#[derive(Debug)]
98pub struct VillagerTrade {
99    pub key: Identifier,
100    pub wants: TradeCost,
101    pub additional_wants: Option<TradeCost>,
102    /// Vanilla parity: the item of the `ItemStackTemplate` a trade `gives`.
103    ///
104    /// Foton's [`crate::item_stack_template::ItemStackTemplate`] cannot be named
105    /// from a `static` -- it holds an `ItemRef`, and generated items are lazy --
106    /// and no vanilla trade gives an item carrying components up front, since
107    /// every one of them builds its result through `given_item_modifiers`. So
108    /// the template collapses to an item and a count, and the build script fails
109    /// on a `components` field rather than dropping it.
110    pub gives: LazyItemRef,
111    pub gives_count: i32,
112    pub max_uses: NumberProvider,
113    pub reputation_discount: NumberProvider,
114    pub xp: NumberProvider,
115    pub merchant_predicate: Option<LootCondition>,
116    pub given_item_modifiers: &'static [ConditionalLootFunction],
117    pub double_trade_price_enchantments: Option<EnchantmentOptions>,
118}
119
120pub type VillagerTradeRef = &'static VillagerTrade;
121
122impl VillagerTrade {
123    /// Builds the offer this trade makes, or `None` when it makes none.
124    ///
125    /// Vanilla parity: `VillagerTrade.getOffer`. There are four ways to get
126    /// `None`, and all four matter: the merchant fails the predicate (a plains
127    /// cartographer has no jungle map to sell), a modifier emptied the result
128    /// (`filtered`'s `on_fail: discard`, which is how an enchantment that did
129    /// not take withdraws the trade), or either price clamped to zero.
130    pub fn get_offer<R: Rng>(&self, ctx: &mut LootContext<'_, R>) -> Option<MerchantOffer> {
131        if let Some(predicate) = &self.merchant_predicate
132            && !predicate.test(ctx)
133        {
134            return None;
135        }
136
137        let mut result = ItemStack::with_count(self.gives, self.gives_count);
138        let mut additional_cost = 0;
139
140        for modifier in self.given_item_modifiers {
141            if modifier
142                .conditions
143                .iter()
144                .all(|condition| condition.test(ctx))
145            {
146                modifier.function.apply(&mut result, ctx);
147            }
148            if result.is_empty() {
149                return None;
150            }
151        }
152
153        if let Some(banked) = result.get(ADDITIONAL_TRADE_COST).copied() {
154            result.remove(ADDITIONAL_TRADE_COST);
155            additional_cost += banked;
156        }
157
158        if let Some(enchantments) = &self.double_trade_price_enchantments
159            && result_stores_any_of(&result, enchantments)
160        {
161            additional_cost *= 2;
162        }
163
164        let item_cost = self.wants.to_item_cost(ctx, additional_cost);
165        if item_cost.count() < 1 {
166            return None;
167        }
168
169        let additional_item_cost = self
170            .additional_wants
171            .as_ref()
172            .map(|cost| cost.to_item_cost(ctx, 0));
173        if additional_item_cost
174            .as_ref()
175            .is_some_and(|cost| cost.count() < 1)
176        {
177            return None;
178        }
179
180        Some(MerchantOffer::new(
181            item_cost,
182            additional_item_cost,
183            result,
184            self.max_uses.get_int(ctx.rng).max(1),
185            self.xp.get_int(ctx.rng).max(0),
186            self.reputation_discount.get(ctx.rng, None).max(0.0),
187        ))
188    }
189}

191/// Whether `result`'s stored enchantments include any of `options`.
192///
193/// Vanilla parity: the `STORED_ENCHANTMENTS` half of `VillagerTrade.getOffer`,
194/// which is why the doubled price only ever lands on an enchanted book.
195fn result_stores_any_of(result: &ItemStack, options: &EnchantmentOptions) -> bool {
196    let Some(stored) = result.get(crate::data_components::vanilla_components::STORED_ENCHANTMENTS)
197    else {
198        return false;
199    };
200
201    match options {
202        EnchantmentOptions::Tag(tag) => REGISTRY
203            .enchantments
204            .iter_tag(tag)
205            .any(|enchantment| stored.get_level(&enchantment.key) > 0),
206        EnchantmentOptions::List(ids) => ids.iter().any(|id| stored.get_level(id) > 0),
207    }
208}
