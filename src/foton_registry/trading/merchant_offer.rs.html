12/// A single trade: what the merchant wants, what it gives, and how its price moves.
13///
14/// Vanilla parity: `net.minecraft.world.item.trading.MerchantOffer`.
15#[derive(Debug, Clone, PartialEq)]
16pub struct MerchantOffer {
17    base_cost_a: ItemCost,
18    cost_b: Option<ItemCost>,
19    result: ItemStack,
20    uses: i32,
21    max_uses: i32,
22    reward_exp: bool,
23    special_price_diff: i32,
24    demand: i32,
25    price_multiplier: f32,
26    xp: i32,
27}

31    ///
32    /// Vanilla parity: the `true` the public constructors pass; only the codec
33    /// can produce an offer that awards no experience.

36    /// Builds an unused offer.

38    /// Vanilla parity: `MerchantOffer(ItemCost, Optional<ItemCost>, ItemStack, int maxUses, int xp, float priceMultiplier)`.

41        base_cost_a: ItemCost,
42        cost_b: Option<ItemCost>,
43        result: ItemStack,

53            max_uses,

92    /// The first price before demand and reputation move it.
93    ///
94    /// Vanilla parity: `getBaseCostA`.

97        self.base_cost_a.cost_stack()
98    }

101    ///
102    /// Vanilla parity: `getCostA`.

105        self.base_cost_a
106            .cost_stack()
107            .copy_with_count(self.modified_cost_count(&self.base_cost_a))
108    }

116            .as_ref()
117            .map_or_else(ItemStack::empty, |cost| cost.cost_stack().clone())
118    }

124    }

140    /// A fresh copy of the result, for actually giving to the player.
141    ///
142    /// Vanilla parity: `assemble`.
143    #[must_use]
144    pub fn assemble(&self) -> ItemStack {
145        self.result.clone()
146    }

148    /// The count a price actually asks for once demand and reputation apply.
149    ///
150    /// Vanilla parity: the private `getModifiedCostCount`. Demand only ever
151    /// raises the price -- the `max(0, ..)` is what stops a negative demand from
152    /// discounting a trade -- while `specialPriceDiff` carries reputation and can
153    /// go either way, down to a floor of one item.
154    fn modified_cost_count(&self, cost: &ItemCost) -> i32 {
155        let base_price = cost.count();
156        #[expect(
157            clippy::cast_precision_loss,
158            reason = "vanilla computes this product in float"
159        )]
160        let demand_diff = ((base_price * self.demand) as f32 * self.price_multiplier).floor();
161        #[expect(
162            clippy::cast_possible_truncation,
163            reason = "vanilla's Mth.floor(float) truncates to int the same way"
164        )]
165        let demand_diff = (demand_diff as i32).max(0);
166        (base_price + demand_diff + self.special_price_diff)
167            .clamp(1, cost.cost_stack().max_stack_size())
168    }
169
170    /// Folds this trade's use count into its standing demand.
171    ///
172    /// Vanilla parity: `updateDemand`, run once a day per villager.
173    pub const fn update_demand(&mut self) {
174        self.demand = self.demand + self.uses - (self.max_uses - self.uses);
175    }

277    /// Returns `true` if these two stacks pay for this trade.
278    ///
279    /// Vanilla parity: `satisfiedBy`. The first price is compared against its
280    /// *modified* count, the second against its plain one -- demand moves only
281    /// the primary cost.
282    #[must_use]
283    pub fn satisfied_by(&self, buy_a: &ItemStack, buy_b: &ItemStack) -> bool {
284        if !self.base_cost_a.test(buy_a)
285            || buy_a.count() < self.modified_cost_count(&self.base_cost_a)
286        {
287            return false;
288        }
289
290        match &self.cost_b {
291            None => buy_b.is_empty(),
292            Some(cost_b) => cost_b.test(buy_b) && buy_b.count() >= cost_b.count(),
293        }
294    }
295
296    /// Takes the price out of the two stacks, if they cover it.
297    ///
298    /// Vanilla parity: `take`.
299    pub fn take(&self, buy_a: &mut ItemStack, buy_b: &mut ItemStack) -> bool {
300        if !self.satisfied_by(buy_a, buy_b) {
301            return false;
302        }
303
304        buy_a.shrink(self.cost_a().count());
305        let cost_b = self.cost_b();
306        if !cost_b.is_empty() {
307            buy_b.shrink(cost_b.count());
308        }
309
310        true
311    }
312}

314impl WriteTo for MerchantOffer {
315    /// Vanilla parity: `MerchantOffer.writeToStream`. Note the plain `writeInt`s:
316    /// uses, maxUses, xp, specialPriceDiff and demand all go out as fixed-width
317    /// big-endian, not as varints, and `specialPriceDiff` is signed.
318    fn write(&self, writer: &mut impl Write) -> Result<()> {
319        self.base_cost_a.write(writer)?;
320        self.result.write(writer)?;
321        self.cost_b.write(writer)?;
322        self.is_out_of_stock().write(writer)?;
323        self.uses.write(writer)?;
324        self.max_uses.write(writer)?;
325        self.xp.write(writer)?;
326        self.special_price_diff.write(writer)?;
327        self.price_multiplier.write(writer)?;
328        self.demand.write(writer)
329    }
330}

332impl ReadFrom for MerchantOffer {
333    /// Vanilla parity: `MerchantOffer.createFromStream`.
334    fn read(data: &mut Cursor<&[u8]>) -> Result<Self> {
335        let base_cost_a = ItemCost::read(data)?;
336        let result = ItemStack::read(data)?;
337        let cost_b = Option::<ItemCost>::read(data)?;
338        let is_exhausted = bool::read(data)?;
339        let uses = i32::read(data)?;
340        let max_uses = i32::read(data)?;
341        let xp = i32::read(data)?;
342        let special_price_diff = i32::read(data)?;
343        let price_multiplier = f32::read(data)?;
344        let demand = i32::read(data)?;
345
346        let mut offer = Self::with_uses(
347            base_cost_a,
348            cost_b,
349            result,
350            uses,
351            max_uses,
352            xp,
353            price_multiplier,
354            demand,
355        );
356        if is_exhausted {
357            offer.set_to_out_of_stock();
358        }
359        offer.set_special_price_diff(special_price_diff);
360        Ok(offer)
361    }
362}

364/// Everything a merchant is currently willing to trade.
365///
366/// Vanilla parity: `net.minecraft.world.item.trading.MerchantOffers`, which is
367/// an `ArrayList<MerchantOffer>`; the `Deref` here is what stands in for that
368/// inheritance.
369#[derive(Debug, Clone, Default, PartialEq)]
370pub struct MerchantOffers(Vec<MerchantOffer>);
371
372impl MerchantOffers {
373    /// An empty offer list.
374    #[must_use]
375    pub const fn new() -> Self {
376        Self(Vec::new())
377    }
378
379    /// Finds the index of the trade these two stacks pay for.
380    ///
381    /// Vanilla parity: `getRecipeFor`, which returns the offer itself; the
382    /// index is what a caller needs to reach back into the merchant's own list
383    /// and change the trade it just sold.
384    ///
385    /// `selection_hint` is the trade the player clicked: when it is in range it
386    /// is the *only* one considered, so two trades wanting the same item stay
387    /// distinguishable.
388    #[must_use]
389    pub fn recipe_index_for(
390        &self,
391        buy_a: &ItemStack,
392        buy_b: &ItemStack,
393        selection_hint: i32,
394    ) -> Option<usize> {
395        // Vanilla's bound is `> 0`, not `>= 0`: trade zero never gets the
396        // shortcut and always falls through to the scan, which reaches it first
397        // anyway.
398        if selection_hint > 0 && (selection_hint as usize) < self.0.len() {
399            let index = selection_hint as usize;
400            return self.0[index].satisfied_by(buy_a, buy_b).then_some(index);
401        }
402
403        self.0
404            .iter()
405            .position(|offer| offer.satisfied_by(buy_a, buy_b))
406    }
407
408    /// The trade these two stacks pay for.
409    ///
410    /// Vanilla parity: `getRecipeFor`.
411    #[must_use]
412    pub fn recipe_for(
413        &self,
414        buy_a: &ItemStack,
415        buy_b: &ItemStack,
416        selection_hint: i32,
417    ) -> Option<&MerchantOffer> {
418        self.0
419            .get(self.recipe_index_for(buy_a, buy_b, selection_hint)?)
420    }
421}

455impl ReadFrom for MerchantOffers {
456    fn read(data: &mut Cursor<&[u8]>) -> Result<Self> {
457        let count = VarInt::read(data)?.0;
458        let count = usize::try_from(count)
459            .map_err(|_| Error::other(format!("Negative merchant offer count: {count}")))?;
460        // The list is bounded only by what the peer claims, so the capacity is
461        // not trusted; the reads themselves run out of buffer soon enough.
462        let mut offers = Vec::with_capacity(count.min(64));
463        for _ in 0..count {
464            offers.push(MerchantOffer::read(data)?);
465        }
466        Ok(Self(offers))
467    }
