1//! How hard one trial spawner fights and what it pays out.
2//!
3//! Vanilla parity: `net.minecraft.world.level.block.entity.trialspawner.TrialSpawnerConfig`.
4//! Vanilla holds these in the datapack registry `minecraft:trial_spawner`; the
5//! entries are generated from the builtin datapack by
6//! `foton-registry/build/trial_spawner_configs.rs` and looked up by key here.

20/// One trial spawner's tuning.
21///
22/// Vanilla parity: `TrialSpawnerConfig`.
23#[derive(Clone, Debug)]
24pub struct TrialSpawnerConfig {
25    pub spawn_range: i32,
26    pub total_mobs: f32,
27    pub simultaneous_mobs: f32,
28    pub total_mobs_added_per_player: f32,
29    pub simultaneous_mobs_added_per_player: f32,
30    pub ticks_between_spawn: i32,
31    pub spawn_potentials: WeightedList<SpawnData>,
32    pub loot_tables_to_eject: WeightedList<LootTableRef>,
33    pub items_to_drop_when_ominous: LootTableRef,
34}
35
36/// Vanilla parity: `TrialSpawnerConfig.Builder.spawnRange`.
37pub const DEFAULT_SPAWN_RANGE: i32 = 4;
38/// Vanilla parity: `TrialSpawnerConfig.Builder.totalMobs`.
39pub const DEFAULT_TOTAL_MOBS: f32 = 6.0;
40/// Vanilla parity: `TrialSpawnerConfig.Builder.simultaneousMobs`.
41pub const DEFAULT_SIMULTANEOUS_MOBS: f32 = 2.0;
42/// Vanilla parity: `TrialSpawnerConfig.Builder.totalMobsAddedPerPlayer`.
43pub const DEFAULT_TOTAL_MOBS_ADDED_PER_PLAYER: f32 = 2.0;
44/// Vanilla parity: `TrialSpawnerConfig.Builder.simultaneousMobsAddedPerPlayer`.
45pub const DEFAULT_SIMULTANEOUS_MOBS_ADDED_PER_PLAYER: f32 = 1.0;
46/// Vanilla parity: `TrialSpawnerConfig.Builder.ticksBetweenSpawn`.
47pub const DEFAULT_TICKS_BETWEEN_SPAWN: i32 = 40;
48
49/// Ticks between two ominous item spawners.
50///
51/// Vanilla parity: `TrialSpawnerConfig.ticksBetweenItemSpawners`, which is a

53pub const TICKS_BETWEEN_ITEM_SPAWNERS: i64 = 160;

61impl TrialSpawnerConfig {
62    /// Returns the configuration a spawner has when nothing names one.
63    ///
64    /// Vanilla parity: `TrialSpawnerConfig.DEFAULT`, which is what
65    /// `TrialSpawnerConfig.builder().build()` produces.
66    #[must_use]
67    pub fn vanilla_default() -> Self {
68        Self {
69            spawn_range: DEFAULT_SPAWN_RANGE,
70            total_mobs: DEFAULT_TOTAL_MOBS,
71            simultaneous_mobs: DEFAULT_SIMULTANEOUS_MOBS,
72            total_mobs_added_per_player: DEFAULT_TOTAL_MOBS_ADDED_PER_PLAYER,
73            simultaneous_mobs_added_per_player: DEFAULT_SIMULTANEOUS_MOBS_ADDED_PER_PLAYER,
74            ticks_between_spawn: DEFAULT_TICKS_BETWEEN_SPAWN,
75            spawn_potentials: WeightedList::empty(),
76            loot_tables_to_eject: default_loot_tables_to_eject(),
77            items_to_drop_when_ominous:
78                &vanilla_loot_tables::SPAWNERS_TRIAL_CHAMBER_ITEMS_TO_DROP_WHEN_OMINOUS,
79        }
80    }
81
82    /// Returns the registered configuration under `key`.
83    ///
84    /// Vanilla parity: a `Holder` resolved out of `Registries.TRIAL_SPAWNER_CONFIG`.
85    #[must_use]
86    pub fn by_key(key: &Identifier) -> Option<&'static Self> {
87        vanilla_trial_spawner_configs::by_key(key)
88    }
89
90    /// Replaces the spawn potentials with a single entity type.
91    ///
92    /// Vanilla parity: `TrialSpawnerConfig.withSpawning`, used by the spawner
93    /// minecart and by `/setblock`-style entity overrides.
94    #[must_use]
95    pub fn with_spawning(&self, entity_type_key: &Identifier) -> Self {
96        Self {
97            spawn_potentials: WeightedList::single(SpawnData::of_entity(entity_type_key)),
98            ..self.clone()
99        }
100    }
101
102    /// Returns how many mobs this spawner will produce in total.
103    ///
104    /// Vanilla parity: `TrialSpawnerConfig.calculateTargetTotalMobs`.
105    #[must_use]
106    pub const fn calculate_target_total_mobs(&self, additional_players: i32) -> i32 {
107        self.total_mobs_added_per_player
108            .mul_add(additional_players as f32, self.total_mobs)
109            .floor() as i32
110    }
111
112    /// Returns how many of its mobs may be alive at once.
113    ///
114    /// Vanilla parity: `TrialSpawnerConfig.calculateTargetSimultaneousMobs`.
115    #[must_use]
116    pub const fn calculate_target_simultaneous_mobs(&self, additional_players: i32) -> i32 {
117        self.simultaneous_mobs_added_per_player
118            .mul_add(additional_players as f32, self.simultaneous_mobs)
119            .floor() as i32
120    }
121}

123impl TrialSpawnerConfig {
124    /// Reads one inline configuration.
125    ///
126    /// Vanilla parity: `TrialSpawnerConfig.DIRECT_CODEC`.
127    #[must_use]
128    pub fn load(nbt: &NbtCompoundView<'_, '_>) -> Self {
129        let default = Self::vanilla_default();
130        let items_to_drop_when_ominous = nbt
131            .string("items_to_drop_when_ominous")
132            .and_then(|key| key.to_str().parse::<Identifier>().ok())
133            .and_then(|key| REGISTRY.loot_tables.by_key(&key))
134            .unwrap_or(default.items_to_drop_when_ominous);
135
136        Self {
137            spawn_range: numeric_or(nbt, "spawn_range", default.spawn_range),
138            total_mobs: float_or(nbt, "total_mobs", default.total_mobs),
139            simultaneous_mobs: float_or(nbt, "simultaneous_mobs", default.simultaneous_mobs),
140            total_mobs_added_per_player: float_or(
141                nbt,
142                "total_mobs_added_per_player",
143                default.total_mobs_added_per_player,
144            ),
145            simultaneous_mobs_added_per_player: float_or(
146                nbt,
147                "simultaneous_mobs_added_per_player",
148                default.simultaneous_mobs_added_per_player,
149            ),
150            ticks_between_spawn: numeric_or(
151                nbt,
152                "ticks_between_spawn",
153                default.ticks_between_spawn,
154            ),
155            spawn_potentials: SpawnData::load_list(nbt.list("spawn_potentials").as_ref()),
156            loot_tables_to_eject: match nbt.list("loot_tables_to_eject") {
157                Some(list) => load_loot_table_list(Some(&list)),
158                None => default.loot_tables_to_eject,
159            },
160            items_to_drop_when_ominous,
161        }
162    }
163
164    /// Writes one inline configuration.
165    #[must_use]
166    pub fn save(&self) -> NbtCompound {
167        let mut nbt = NbtCompound::new();
168        nbt.insert("spawn_range", self.spawn_range);
169        nbt.insert("total_mobs", self.total_mobs);
170        nbt.insert("simultaneous_mobs", self.simultaneous_mobs);
171        nbt.insert(
172            "total_mobs_added_per_player",
173            self.total_mobs_added_per_player,
174        );
175        nbt.insert(
176            "simultaneous_mobs_added_per_player",
177            self.simultaneous_mobs_added_per_player,
178        );
179        nbt.insert("ticks_between_spawn", self.ticks_between_spawn);
180        nbt.insert(
181            "spawn_potentials",
182            SpawnData::save_list(&self.spawn_potentials),
183        );
184        nbt.insert(
185            "loot_tables_to_eject",
186            save_loot_table_list(&self.loot_tables_to_eject),
187        );
188        nbt.insert(
189            "items_to_drop_when_ominous",
190            self.items_to_drop_when_ominous.key.to_string(),
191        );
192        nbt
193    }
194}

196/// Vanilla parity: `ValueInput.getIntOr`, which accepts any numeric tag.
197fn numeric_or(nbt: &NbtCompoundView<'_, '_>, name: &str, fallback: i32) -> i32 {
198    nbt.get(name)
199        .and_then(|tag| tag.codec_i32())
200        .unwrap_or(fallback)
201}
202
203/// Vanilla parity: `ValueInput.getFloatOr`.
204fn float_or(nbt: &NbtCompoundView<'_, '_>, name: &str, fallback: f32) -> f32 {
205    nbt.get(name)
206        .and_then(|tag| tag.codec_f32())
207        .unwrap_or(fallback)
208}

210/// A configuration a trial spawner points at, by key or inline.
211///
212/// Vanilla parity: `Holder<TrialSpawnerConfig>` under `RegistryFileCodec`,
213/// which writes a bare key for a registered value and the whole object for a
214/// direct one. Keeping the two apart is what lets a saved spawner come back
215/// pointing at the same thing it was pointing at.
216#[derive(Clone, Debug)]
217pub enum TrialSpawnerConfigHolder {
218    /// A value out of the `minecraft:trial_spawner` registry.
219    Registry {
220        /// The key it was named by, which is what gets written back.
221        key: Identifier,
222        /// The registered value.
223        value: &'static TrialSpawnerConfig,
224    },
225    /// A configuration written out in full.
226    Direct(Arc<TrialSpawnerConfig>),
227}
228
229impl TrialSpawnerConfigHolder {
230    /// Returns the configuration this holder points at.
231    #[must_use]
232    pub fn value(&self) -> &TrialSpawnerConfig {
233        match self {
234            Self::Registry { value, .. } => value,
235            Self::Direct(value) => value,
236        }
237    }
238
239    /// Returns a direct holder over `config`.
240    #[must_use]
241    pub fn direct(config: TrialSpawnerConfig) -> Self {
242        Self::Direct(Arc::new(config))
243    }
244
245    /// Reads a holder from a key string or an inline object.
246    ///
247    /// An unknown registry key falls back to a direct default and says so:
248    /// silently swapping in the default fight would make a datapack spawner
249    /// look like it worked.
250    #[must_use]
251    pub fn load(tag: BorrowedNbtTag<'_, '_>) -> Option<Self> {
252        if let Some(key) = tag.string() {
253            let key: Identifier = key.to_str().parse().ok()?;
254            let Some(value) = vanilla_trial_spawner_configs::by_key(&key) else {
255                log::warn!("unknown trial spawner config {key}; using the default");
256                return Some(Self::direct(TrialSpawnerConfig::vanilla_default()));
257            };
258            return Some(Self::Registry { key, value });
259        }
260        let compound = tag.compound()?;
261        Some(Self::direct(TrialSpawnerConfig::load(&compound)))
262    }
263
264    /// Writes this holder the way vanilla's `RegistryFileCodec` would.
265    #[must_use]
266    pub fn save(&self) -> NbtTag {
267        match self {
268            Self::Registry { key, .. } => NbtTag::String(key.to_string().into()),
269            Self::Direct(value) => NbtTag::Compound(value.save()),
270        }
271    }
272}
273
274impl Default for TrialSpawnerConfigHolder {
275    fn default() -> Self {
276        Self::direct(TrialSpawnerConfig::vanilla_default())
277    }
278}

280/// The reward split a spawner ejects when its datapack entry names none.
281///
282/// Vanilla parity: the `lootTablesToEject` field initialiser of
283/// `TrialSpawnerConfig.Builder`, in that order and both at weight one.
284#[must_use]
285pub fn default_loot_tables_to_eject() -> WeightedList<LootTableRef> {
286    WeightedList::new(vec![
287        Weighted {
288            value: &vanilla_loot_tables::SPAWNERS_TRIAL_CHAMBER_CONSUMABLES as LootTableRef,
289            weight: 1,
290        },
291        Weighted {
292            value: &vanilla_loot_tables::SPAWNERS_TRIAL_CHAMBER_KEY as LootTableRef,
293            weight: 1,
294        },
295    ])
296}

298#[cfg(test)]
299mod tests {
300    use super::*;
301    use crate::init_vanilla_registry;
302
303    /// The player scaling is the whole difficulty curve of a trial chamber, and
304    /// vanilla floors rather than rounds -- three players on a default spawner
305    /// get twelve mobs, not thirteen.
306    #[test]
307    fn player_scaling_floors_the_way_vanilla_does() {
308        let config = TrialSpawnerConfig::vanilla_default();
309
310        assert_eq!(config.calculate_target_total_mobs(0), 6);
311        assert_eq!(config.calculate_target_total_mobs(3), 12);
312        assert_eq!(config.calculate_target_simultaneous_mobs(0), 2);
313        assert_eq!(config.calculate_target_simultaneous_mobs(3), 5);
314    }
315
316    /// A fractional per-player term is exactly where a rounding mistake hides:
317    /// the breeze spawner adds half a simultaneous mob per extra player.
318    #[test]
319    fn a_half_mob_per_player_only_counts_on_the_second_player() {
320        let config = TrialSpawnerConfig {
321            simultaneous_mobs: 1.0,
322            simultaneous_mobs_added_per_player: 0.5,
323            ..TrialSpawnerConfig::vanilla_default()
324        };
325
326        assert_eq!(config.calculate_target_simultaneous_mobs(0), 1);
327        assert_eq!(config.calculate_target_simultaneous_mobs(1), 1);
328        assert_eq!(config.calculate_target_simultaneous_mobs(2), 2);
329    }
330
331    /// The generated registry is the only thing that resolves the config a
332    /// trial chamber's spawner names, so a missing entry means every chamber
333    /// spawner silently falls back to the default fight.
334    #[test]
335    fn the_generated_registry_carries_the_trial_chamber_entries() {
336        init_vanilla_registry();
337        let key = Identifier::vanilla_static("trial_chamber/melee/zombie/normal");
338        let config = TrialSpawnerConfig::by_key(&key).expect("zombie melee config");
339
340        assert_eq!(config.ticks_between_spawn, 20);
341        assert_eq!(config.spawn_potentials.len(), 1);
342        assert_eq!(
343            config.spawn_potentials.entries()[0].value.entity_type_key(),
344            Some(Identifier::vanilla_static("zombie"))
345        );
346
347        let ominous = TrialSpawnerConfig::by_key(&Identifier::vanilla_static(
348            "trial_chamber/melee/zombie/ominous",
349        ))
350        .expect("zombie melee ominous config");
351        assert_eq!(ominous.loot_tables_to_eject.total_weight(), 10);
352    }
