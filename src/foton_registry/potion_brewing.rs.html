1//! Which potion a brewing stand turns another into.
2//!
3//! Vanilla parity: `PotionBrewing`. Vanilla assembles this table in code rather
4//! than shipping it as data, so it is assembled in code here too, in the same
5//! order and with the same shape: a list of containers a potion may sit in, a
6//! list of container-to-container conversions, and a list of potion-to-potion
7//! conversions.
8//!
9//! The one structural difference is `addStartMix`. Vanilla's builder expands
10//! each start mix into two ordinary mixes as it runs; expanding them by hand
11//! here would double every entry and invite a typo, so the start mixes keep
12//! their own short table and the expansion happens where the mixes are read.
13

322const fn mix(from: PotionRef, ingredient: ItemRef, to: PotionRef) -> PotionMix {
323    PotionMix {
324        from,
325        ingredient,
326        to,
327    }
328}
329
330/// Calls `visit` with every potion-to-potion conversion there is.
331///
332/// The stated mixes come first, then the two each start mix stands for: water
333/// into mundane, and awkward into whatever that ingredient makes. No ingredient
334/// appears on both sides of that split, so the order does not decide any
335/// outcome; it only decides which entry is found first.
336fn for_each_potion_mix(mut visit: impl FnMut(PotionRef, ItemRef, PotionRef) -> bool) {
337    for entry in POTION_MIXES.iter() {
338        if visit(entry.from, entry.ingredient, entry.to) {
339            return;
340        }
341    }
342    for (ingredient, potion) in START_MIXES.iter() {
343        if visit(
344            &vanilla_potions::WATER,
345            ingredient,
346            &vanilla_potions::MUNDANE,
347        ) {
348            return;
349        }
350        if visit(&vanilla_potions::AWKWARD, ingredient, potion) {
351            return;
352        }
353    }
354}

356/// Returns whether this item does anything at all in the ingredient slot.
357///
358/// Vanilla parity: `PotionBrewing.isIngredient`.
359#[must_use]
360pub fn is_ingredient(ingredient: &ItemStack) -> bool {
361    is_container_ingredient(ingredient) || is_potion_ingredient(ingredient)
362}

364/// Returns whether this item converts one bottle into another.
365///
366/// Vanilla parity: `PotionBrewing.isContainerIngredient`.
367#[must_use]
368pub fn is_container_ingredient(ingredient: &ItemStack) -> bool {
369    CONTAINER_MIXES
370        .iter()
371        .any(|entry| ingredient.is(entry.ingredient))
372}

374/// Returns whether this item converts one potion into another.
375///
376/// Vanilla parity: `PotionBrewing.isPotionIngredient`.
377#[must_use]
378pub fn is_potion_ingredient(ingredient: &ItemStack) -> bool {
379    let mut found = false;
380    for_each_potion_mix(|_, mix_ingredient, _| {
381        found = ingredient.is(mix_ingredient);
382        found
383    });
384    found
385}

387/// Returns whether this item is a bottle a potion can sit in.
388///
389/// Vanilla parity: `PotionBrewing.isContainer`.
390#[must_use]
391pub fn is_container(input: &ItemStack) -> bool {
392    CONTAINERS.iter().any(|container| input.is(container))
393}

395/// Returns whether this ingredient changes this bottle.
396///
397/// Vanilla parity: `PotionBrewing.hasMix`. A bottle that is not a potion
398/// container is refused outright, which is what stops a brewing stand working
399/// on an arbitrary item someone hoppered in.
400#[must_use]
401pub fn has_mix(source: &ItemStack, ingredient: &ItemStack) -> bool {
402    is_container(source)
403        && (has_container_mix(source, ingredient) || has_potion_mix(source, ingredient))
404}
405
406/// Returns whether this ingredient turns this bottle into another kind.
407///
408/// Vanilla parity: `PotionBrewing.hasContainerMix`.
409#[must_use]
410pub fn has_container_mix(source: &ItemStack, ingredient: &ItemStack) -> bool {
411    CONTAINER_MIXES
412        .iter()
413        .any(|entry| source.is(entry.from) && ingredient.is(entry.ingredient))
414}
415
416/// Returns whether this ingredient turns this bottle's potion into another.
417///
418/// Vanilla parity: `PotionBrewing.hasPotionMix`.
419#[must_use]
420pub fn has_potion_mix(source: &ItemStack, ingredient: &ItemStack) -> bool {
421    let Some(potion) = potion_of(source) else {
422        return false;
423    };
424
425    let mut found = false;
426    for_each_potion_mix(|from, mix_ingredient, _| {
427        found = std::ptr::eq(from, potion) && ingredient.is(mix_ingredient);
428        found
429    });
430    found
431}

433/// Returns what this bottle becomes when brewed with this ingredient.
434///
435/// Vanilla parity: `PotionBrewing.mix`. A bottle nothing applies to comes back
436/// unchanged, which is how a stand with one matching bottle and two others
437/// leaves the other two alone.
438#[must_use]
439pub fn mix_with(ingredient: &ItemStack, source: &ItemStack) -> ItemStack {
440    if source.is_empty() {
441        return source.clone();
442    }
443    let Some(potion) = potion_of(source) else {
444        return source.clone();
445    };
446
447    for entry in CONTAINER_MIXES.iter() {
448        if source.is(entry.from) && ingredient.is(entry.ingredient) {
449            return potion_item(entry.to, potion);
450        }
451    }
452
453    let mut result = None;
454    for_each_potion_mix(|from, mix_ingredient, to| {
455        if std::ptr::eq(from, potion) && ingredient.is(mix_ingredient) {
456            result = Some(to);
457            return true;
458        }
459        false
460    });
461
462    result.map_or_else(|| source.clone(), |to| potion_item(source.item(), to))
463}

465/// Returns whether this item can fuel a brewing stand.
466///
467/// Vanilla parity: the `ItemTags.BREWING_FUEL` test of

474/// Returns the potion a bottle holds, if it holds one.
475fn potion_of(stack: &ItemStack) -> Option<PotionRef> {
476    stack
477        .get(POTION_CONTENTS)
478        .and_then(PotionContents::potion)
479        .map(|reference| reference.value())
480}
481
482/// Builds a bottle of `item` holding `potion`.
483///
484/// Vanilla parity: `PotionContents.createItemStack`.
485#[must_use]
486pub fn potion_item(item: ItemRef, potion: PotionRef) -> ItemStack {
487    let mut stack = ItemStack::new(item);
488    stack.set(
489        POTION_CONTENTS,
490        PotionContents::new(Some(RegistryReference::new(potion)), None, Vec::new(), None),
491    );
492    stack
493}

504    #[test]
505    fn nether_wart_turns_water_into_awkward() {
506        init_vanilla_registry();
507        let result = mix_with(
508            &ItemStack::new(&vanilla_items::NETHER_WART),
509            &water_bottle(),
510        );
511        assert!(result.is(&vanilla_items::POTION));
512        assert!(std::ptr::eq(
513            potion_of(&result).expect("brewed bottle holds a potion"),
514            &raw const vanilla_potions::AWKWARD

516    }

518    #[test]
519    fn a_start_mix_needs_the_awkward_step_first() {
520        init_vanilla_registry();
521        // Sugar on water gives mundane, not swiftness. Getting swiftness out of
522        // water in one step is the mistake this table exists to prevent.
523        let from_water = mix_with(&ItemStack::new(&vanilla_items::SUGAR), &water_bottle());
524        assert!(std::ptr::eq(
525            potion_of(&from_water).expect("holds a potion"),
526            &raw const vanilla_potions::MUNDANE
527        ));
528
529        let awkward = potion_item(&vanilla_items::POTION, &vanilla_potions::AWKWARD);
530        let from_awkward = mix_with(&ItemStack::new(&vanilla_items::SUGAR), &awkward);
531        assert!(std::ptr::eq(
532            potion_of(&from_awkward).expect("holds a potion"),
533            &raw const vanilla_potions::SWIFTNESS
534        ));
535    }
